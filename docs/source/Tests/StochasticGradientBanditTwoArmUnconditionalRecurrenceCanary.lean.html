6 This canary pins the source-time fence and the complete public API of the
7 unconditional recurrence leaf.  An inclusive Lean prefix index `n` denotes
8 source time `n + 2`; `tailHorizon` covers source rounds
9 `t = 2, ..., tailHorizon + 1`, while `twoArmFullFailureMassSqSum_le` adds the
10 normalized source-round-1 failure square `1 / 4`.

67 source parameter `theta_{1,n+2}`, represented by adapter time `n + 1`. -/

82 example {Env : Type*} [MeasurableSpace Env] [StandardBorelSpace Env]
83     (prior : Measure Env) [IsFiniteMeasure prior]
84     (eta Delta : Real) (heta : 0 <= eta)
85     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
86     (mean : Fin 2 -> Real)
87     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean)
88     (hgap : mean 0 - mean 1 = Delta) (n : Nat) :
89     integral (twoArmTrajectoryMeasure prior eta environment)
90         (twoArmForwardPotential (Env := Env) eta (n + 1)) <=
91       integral (twoArmTrajectoryMeasure prior eta environment)
92         (twoArmForwardPotential (Env := Env) eta n) +
93       2 * integral (twoArmTrajectoryMeasure prior eta environment)
94           (fun sample => twoArmSuccessProbability
95             (Env := Env) eta n sample ^ 2) *
96         (eta * Delta + eta ^ 2 * sourceC eta) :=
97   twoArmForwardUnconditionalRecurrence
98     prior eta Delta heta environment mean contract hgap n

100 example {Env : Type*} [MeasurableSpace Env] [StandardBorelSpace Env]
101     (prior : Measure Env) [IsProbabilityMeasure prior]
102     (eta Delta : Real) (heta : 0 < eta)
103     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
104     (mean : Fin 2 -> Real)
105     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean)
106     (hgap : mean 0 - mean 1 = Delta)
107     (hmargin : eta * sourceC eta < Delta) (tailHorizon : Nat) :
108     (1 : Real) / 4 +
109         (Finset.range tailHorizon).sum (fun n =>
110           integral (twoArmTrajectoryMeasure prior eta environment)
111             (fun sample => twoArmFailureMass
112               (Env := Env) eta n sample ^ 2)) <=
113       1 / (2 * eta * (Delta - eta * sourceC eta)) :=
114   twoArmFullFailureMassSqSum_le
115     prior eta Delta heta environment mean contract hgap hmargin tailHorizon
