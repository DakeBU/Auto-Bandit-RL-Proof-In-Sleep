6 /-!
7 # BanditRLlib proof-dependency graph exporter
8 
9 This executable reads compiled declarations from Lean's environment. It does not parse source
10 text. For each project declaration it records direct constants used by the declaration's type and
11 direct constants used only by its value/proof term. If a constant occurs in both, the type edge is
12 canonical and carries `also_in_value = true`.
13 
14 The exported graph contains all declarations owned by `BanditRLProof` modules plus the direct
15 external-library boundary they reference. It deliberately does not unfold declarations or expand
16 the transitive Mathlib graph.
17 -/

23 private def projectRoot : Name := `BanditRLProof
24 
25 private def nameLt (a b : Name) : Bool :=
26   a.toString < b.toString
27 
28 private def stringLt (a b : String) : Bool :=
29   a < b
30 
31 private def isProjectModule (moduleName : Name) : Bool :=
32   moduleName == projectRoot || moduleName.toString.startsWith "BanditRLProof."
33 
34 private def moduleOf? (env : Environment) (declName : Name) : Option Name := do
35   let idx ← env.getModuleIdxFor? declName
36   env.header.moduleNames[idx.toNat]?
37 
38 private def isProjectDecl (env : Environment) (declName : Name) : Bool :=
39   (moduleOf? env declName).any isProjectModule

41 private def declarationKind : ConstantInfo → String
42   | .axiomInfo _  => "axiom"
43   | .defnInfo _   => "definition"
44   | .thmInfo _    => "theorem"
45   | .opaqueInfo _ => "opaque"
46   | .quotInfo _   => "quotient"
47   | .inductInfo _ => "inductive"
48   | .ctorInfo _   => "constructor"
49   | .recInfo _    => "recursor"

51 private def lastComponent : Name → String
52   | .anonymous => ""
53   | .str _ s   => s
54   | .num _ n   => toString n
55 
56 private def sourcePath (moduleName : Name) : String :=
57   moduleName.toString.replace "." "/" ++ ".lean"
58 
59 private def jsonNat (n : Nat) : Json :=
60   toJson n
61 
62 private def jsonBool (b : Bool) : Json :=
63   toJson b

65 private def sourceJson (env : Environment) (declName moduleName : Name) : Json :=
66   let range? := declRangeExt.find? (level := .exported) env declName
67   Json.mkObj [
68     ("file", sourcePath moduleName),
69     ("line_start", range?.map (fun r => jsonNat (r.range.pos.line + 1)) |>.getD Json.null),
70     ("line_end", range?.map (fun r => jsonNat (r.range.endPos.line + 1)) |>.getD Json.null)
71   ]

73 private def dependencies (expr : Expr) : Array Name :=
74   expr.getUsedConstantsAsSet.toArray.qsort nameLt
75 
76 private def valueDependencies (info : ConstantInfo) : Array Name :=
77   match info.value? (allowOpaque := true) with
78   | some value => dependencies value
79   | none       => #[]
80 
81 private def exprObjectCount (expr : Expr) : IO Nat :=
82   expr.numObjs
83 
84 private def nodeJson (env : Environment) (declName : Name) (scope : String) : IO Json := do
85   let some info := env.find? declName
86     | throw <| IO.userError s!"environment is missing declaration {declName}"
87   let some moduleName := moduleOf? env declName
88     | throw <| IO.userError s!"environment is missing module ownership for {declName}"
89   let typeDeps := dependencies info.type
90   let valueDeps := valueDependencies info
91   let typeObjects ← exprObjectCount info.type
92   let valueObjects ← match info.value? (allowOpaque := true) with
93     | some value => exprObjectCount value
94     | none       => pure 0
95   pure <| Json.mkObj [
96     ("name", declName.toString),
97     ("namespace", declName.getPrefix.toString),
98     ("short_name", lastComponent declName),
99     ("kind", declarationKind info),
100     ("scope", scope),
101     ("module", moduleName.toString),
102     ("source", sourceJson env declName moduleName),
103     ("status", "compiled"),
104     ("has_value", jsonBool <| (info.value? (allowOpaque := true)).isSome),
105     ("type_expr_objects", jsonNat typeObjects),
106     ("value_expr_objects", jsonNat valueObjects),
107     ("type_dependency_count", jsonNat typeDeps.size),
108     ("value_dependency_count", jsonNat valueDeps.size)
109   ]

111 structure Edge where
112   source      : Name
113   target      : Name
114   kind        : String
115   alsoInValue : Bool
116   targetScope : String
117 
118 private def edgeKey (edge : Edge) : String :=
119   edge.source.toString ++ "\u0000" ++ edge.target.toString ++ "\u0000" ++ edge.kind
120 
121 private def edgeJson (edge : Edge) : Json :=
122   Json.mkObj [
123     ("source", edge.source.toString),
124     ("target", edge.target.toString),
125     ("kind", edge.kind),
126     ("also_in_value", jsonBool edge.alsoInValue),
127     ("target_scope", edge.targetScope)
128   ]
129 
130 private def edgesFor (env : Environment) (source : Name) (info : ConstantInfo) : Array Edge :=
131   let typeDeps := dependencies info.type
132   let valueDeps := valueDependencies info
133   let valueSet := NameSet.ofList valueDeps.toList
134   let typeSet := NameSet.ofList typeDeps.toList
135   let typeEdges := typeDeps.map fun target => {
136     source
137     target
138     kind := "type"
139     alsoInValue := valueSet.contains target
140     targetScope := if isProjectDecl env target then "project" else "external"
141   }
142   let valueEdges := (valueDeps.filter fun target => !typeSet.contains target).map fun target => {
143     source
144     target
145     kind := "value"
146     alsoInValue := false
147     targetScope := if isProjectDecl env target then "project" else "external"
148   }
149   typeEdges ++ valueEdges

151 private def projectDeclarations (env : Environment) : Array Name := Id.run do
152   -- `ModuleData.constNames` can repeat generated equation/simp declarations.  Environment names are
153   -- globally unique, so canonicalize through a set before sorting and serializing.
154   let mut result : NameSet := {}
155   for h : idx in [0:env.header.moduleData.size] do
156     let moduleName := env.header.moduleNames[idx]!
157     if isProjectModule moduleName then
158       for declName in env.header.moduleData[idx].constNames do
159         result := result.insert declName
160   return result.toArray.qsort nameLt
161 
162 private def directExternalBoundary (env : Environment) (projectDecls : Array Name) : Array Name := Id.run do
163   let mut external : NameSet := {}
164   for source in projectDecls do
165     if let some info := env.find? source then
166       for edge in edgesFor env source info do
167         if edge.targetScope == "external" && (env.find? edge.target).isSome then
168           external := external.insert edge.target
169   return external.toArray.qsort nameLt
170 
171 private def moduleImportJson (moduleName : Name) (imp : Import) : Json :=
172   Json.mkObj [
173     ("source", moduleName.toString),
174     ("target", imp.module.toString),
175     ("import_all", jsonBool imp.importAll),
176     ("exported", jsonBool imp.isExported),
177     ("meta", jsonBool imp.isMeta),
178     ("target_scope", if isProjectModule imp.module then "project" else "external")
179   ]
180 
181 private def projectModuleImports (env : Environment) : Array Json := Id.run do
182   let mut rows : Array (String × Json) := #[]
183   for h : idx in [0:env.header.moduleData.size] do
184     let moduleName := env.header.moduleNames[idx]!
185     if isProjectModule moduleName then
186       for imp in env.header.moduleData[idx].imports do
187         let key := moduleName.toString ++ "\u0000" ++ imp.module.toString ++ "\u0000" ++
188           toString imp.importAll ++ "\u0000" ++ toString imp.isExported ++ "\u0000" ++
189           toString imp.isMeta
190         rows := rows.push (key, moduleImportJson moduleName imp)
191   return (rows.qsort fun a b => stringLt a.1 b.1).map (·.2)
192 
193 def exportJson (env : Environment) : IO Json := do
194   let projectDecls := projectDeclarations env
195   let externalDecls := directExternalBoundary env projectDecls
196   let mut nodes : Array Json := #[]
197   for declName in projectDecls do
198     nodes := nodes.push (← nodeJson env declName "project")
199   for declName in externalDecls do
200     nodes := nodes.push (← nodeJson env declName "external")
201   let mut edges : Array Edge := #[]
202   for source in projectDecls do
203     if let some info := env.find? source then
204       edges := edges ++ edgesFor env source info
205   edges := edges.qsort fun a b => stringLt (edgeKey a) (edgeKey b)
206   pure <| Json.mkObj [
207     ("schema_version", jsonNat 1),
208     ("project", "BanditRLlib"),
209     ("root_module", projectRoot.toString),
210     ("namespace", projectRoot.toString),
211     ("lean_version", Lean.versionString),
212     ("extraction", Json.mkObj [
213       ("source", "compiled-environment"),
214       ("dependency_semantics", "direct-constant-occurrence"),
215       ("closure", "project-direct-with-external-boundary"),
216       ("deterministic", true),
217       ("type_precedence", true)
218     ]),
219     ("status_vocabulary", toJson #["compiled", "prototype", "partial", "planned", "blocked"]),
220     ("counts", Json.mkObj [
221       ("project_nodes", jsonNat projectDecls.size),
222       ("external_boundary_nodes", jsonNat externalDecls.size),
223       ("edges", jsonNat edges.size),
224       ("module_imports", jsonNat (projectModuleImports env).size)
225     ]),
226     ("nodes", Json.arr nodes),
227     ("edges", Json.arr <| edges.map edgeJson),
228     ("module_imports", Json.arr <| projectModuleImports env)
229   ]
230 
231 end BanditRLProof.ProofGraph

233 private def usage : String :=
234   "usage: lake exe proof_graph_export [--compact] OUTPUT.json"
235 
236 unsafe def main (args : List String) : IO UInt32 := do
237   let (compact, output?) := match args with
238     | [output]              => (false, some output)
239     | ["--compact", output] => (true, some output)
240     | _                     => (false, none)
241   let some output := output?
242     | IO.eprintln usage; return 2
243   Lean.initSearchPath (← Lean.findSysroot)
244   Lean.enableInitializersExecution
245   let env ← Lean.importModules #[{ module := `BanditRLProof }] {} (loadExts := true)
246   let graph ← BanditRLProof.ProofGraph.exportJson env
247   let rendered := if compact then graph.compress else graph.pretty
248   IO.FS.writeFile output (rendered ++ "\n")
249   return 0
