175 def module_slug(value: str, max_length: int = 96) -> str:
176     """Return a stable URL slug that stays below Windows path limits."""
177     slug = slugify(value)

185 def breakable_identifier(value: str) -> str:
186     """Escape an identifier and add visual wrap opportunities without changing copied text."""
187     pieces: list[str] = []
188     for index, char in enumerate(value):
189         previous = value[index - 1] if index else ""
190         following = value[index + 1] if index + 1 < len(value) else ""
191         camel_boundary = char.isupper() and (
192             previous.islower() or (previous.isupper() and following.islower())
193         )
194         if camel_boundary:
195             pieces.append("<wbr>")
196         pieces.append(html.escape(char))
197         if char in "._/":
198             pieces.append("<wbr>")
199     return "".join(pieces)

202 def normalize_math_source(value: str) -> str:
203     """Give every teaching formula an explicit MathJax delimiter pair."""
204     source = value.strip()
205     if not source:
206         return r"\(\text{No mathematical statement recorded.}\)"
207     inline_open = source.count(r"\(")
208     inline_close = source.count(r"\)")
209     display_open = source.count(r"\[")
210     display_close = source.count(r"\]")
211     if inline_open != inline_close or display_open != display_close:
212         source = re.sub(r"\\[()\[\]]", "", source).strip()
213         return f"\\[{source}\\]"
214     if "\\(" in source or "\\[" in source:
215         return source
216     return f"\\[{source}\\]"

239 def strip_lean_comments(line: str, block_depth: int) -> tuple[str, int]:
240     visible: list[str] = []
241     index = 0
242     in_string = False
243     escaped = False
244     while index < len(line):
245         if block_depth:
246             if line.startswith("/-", index):
247                 block_depth += 1
248                 index += 2
249             elif line.startswith("-/", index):
250                 block_depth -= 1
251                 index += 2
252             else:
253                 index += 1
254             continue
255         char = line[index]
256         if in_string:
257             visible.append(char)
258             if escaped:
259                 escaped = False
260             elif char == "\\":
261                 escaped = True
262             elif char == '"':
263                 in_string = False
264             index += 1
265             continue
266         if line.startswith("--", index):
267             break
268         if line.startswith("/-", index):
269             block_depth += 1
270             index += 2
271             continue
272         visible.append(char)
273         if char == '"':
274             in_string = True
275         index += 1
276     return "".join(visible), block_depth

279 def compact_statement(lines: list[str], start: int) -> str:
280     parts: list[str] = []
281     delimiter_depth = 0
282     block_comment_depth = 0
283     in_string = False
284     for raw in lines[start : min(len(lines), start + 90)]:
285         stripped = raw.strip()
286         if not stripped:
287             continue
288         compact = re.sub(r"\s+", " ", stripped)
289         is_result_let = stripped.startswith("let ")
290         assignment_index: int | None = None
291         index = 0
292         while index < len(raw):
293             if block_comment_depth:
294                 if raw.startswith("/-", index):
295                     block_comment_depth += 1
296                     index += 2
297                     continue
298                 if raw.startswith("-/", index):
299                     block_comment_depth -= 1
300                     index += 2
301                     continue
302                 index += 1
303                 continue
304             if in_string:
305                 if raw[index] == '"' and (index == 0 or raw[index - 1] != "\\"):
306                     in_string = False
307                 index += 1
308                 continue
309             if raw.startswith("--", index):
310                 break
311             if raw.startswith("/-", index):
312                 block_comment_depth += 1
313                 index += 2
314                 continue
315             char = raw[index]
316             if char == '"':
317                 in_string = True
318             elif char in "([{":
319                 delimiter_depth += 1
320             elif char in ")]}":
321                 delimiter_depth = max(0, delimiter_depth - 1)
322             elif raw.startswith(":=", index) and delimiter_depth == 0 and not is_result_let:
323                 assignment_index = index
324                 break
325             index += 1
326         if assignment_index is not None:
327             before = re.sub(r"\s+", " ", raw[:assignment_index].strip()).rstrip()
328             if before:
329                 parts.append(before)
330             break
331         parts.append(compact)
332         if stripped == "where" or stripped.endswith(" where"):
333             break
334         if sum(len(part) for part in parts) > 8000:
335             break
336     return " ".join(parts)[:8000]

339 def docstring_before(lines: list[str], start: int) -> str:
340     index = start - 1
341     while index >= 0 and (not lines[index].strip() or lines[index].lstrip().startswith("@[")):
342         index -= 1
343     if index < 0 or "-/" not in lines[index]:
344         return ""
345     end = index
346     while index >= 0 and "/--" not in lines[index]:
347         index -= 1
348     if index < 0:
349         return ""
350     raw = "\n".join(lines[index : end + 1])
351     if not raw.lstrip().startswith("/--"):
352         return ""
353     raw = raw[raw.find("/--") + 3 :]
354     raw = raw[: raw.rfind("-/")]
355     cleaned = []
356     for line in raw.splitlines():
357         line = re.sub(r"^\s*\*?\s?", "", line.rstrip())
358         cleaned.append(line)
359     return "\n".join(cleaned).strip()

372 def markdown_prose_summary(text: str, limit: int = 620) -> str:
373     """Return the first prose paragraph of a small Lean/Markdown doc block."""
374 
375     heading_fallback = ""
376     for block in re.split(r"\n\s*\n", text.strip()):
377         lines = [line.strip() for line in block.splitlines() if line.strip()]
378         if not lines:
379             continue
380         if lines[0].startswith("#"):
381             heading_fallback = re.sub(r"^#+\s*", "", lines[0]).strip()
382             lines = lines[1:]
383             if not lines:
384                 continue
385         prose = " ".join(lines)
386         prose = re.sub(r"\[([^\]]+)\]\([^)]+\)", r"\1", prose)
387         prose = re.sub(r"`([^`]+)`", r"\1", prose)
388         prose = re.sub(r"\s+", " ", prose).strip()
389         if prose:
390             return prose[:limit].rstrip()
391     return heading_fallback[:limit].rstrip()

394 def scan_module(path: Path) -> dict[str, Any]:
395     text = path.read_text(encoding="utf-8")
396     lines = text.splitlines()
397     imports = [
398         match.group(1)
399         for line in lines
400         if (match := IMPORT_RE.match(line)) and match.group(1).startswith("BanditRLProof")
401     ]
402     scopes: list[tuple[str, list[str], str | None]] = []
403     visible_lines: list[str] = []
404     declarations: list[dict[str, Any]] = []
405     block_depth = 0
406 
407     for lineno, raw in enumerate(lines, start=1):
408         line, block_depth = strip_lean_comments(raw, block_depth)
409         visible_lines.append(line)
410         if match := NAMESPACE_RE.match(line):
411             name = match.group(1)
412             scopes.append(("namespace", name.split("."), name))
413             continue
414         if SECTION_RE.match(line):
415             scopes.append(("section", [], None))
416             continue
417         if match := END_RE.match(line):
418             name = match.group(1)
419             if not scopes:
420                 continue
421             if name:
422                 for index in range(len(scopes) - 1, -1, -1):
423                     kind, parts, label = scopes[index]
424                     if kind == "namespace" and (label == name or parts[-1:] == name.split(".")[-1:]):
425                         del scopes[index:]
426                         break
427                 else:
428                     scopes.pop()
429             else:
430                 scopes.pop()
431             continue
432         match = DECL_RE.match(line)
433         declaration_name = match.group("name") if match else ""
434         declaration_kind = match.group("kind") if match else ""
435         declaration_modifiers = match.group("modifiers") if match else ""
436         if not match:
437             # Large generated theorem statements often put `theorem` on its
438             # own line and the (very long) declaration name on the next line.
439             # The old scanner silently omitted every declaration written in
440             # that style, including the current RL route endpoint.
441             head = DECL_HEAD_RE.match(line)
442             if not head:
443                 continue
444             for following in lines[lineno:]:
445                 candidate = re.match(r"^\s*([A-Za-z0-9_'.]+)", following)
446                 if candidate:
447                     declaration_name = candidate.group(1)
448                     break
449                 if following.strip() and not following.lstrip().startswith(("--", "/-", "-/")):
450                     break
451             if not declaration_name:
452                 continue
453             declaration_kind = head.group("kind")
454             declaration_modifiers = head.group("modifiers")
455         namespace_parts: list[str] = []
456         for kind, parts, _label in scopes:
457             if kind == "namespace":
458                 namespace_parts.extend(parts)
459         name = declaration_name
460         # A dotted declaration name is still relative to the currently open
461         # namespace in Lean.  For example, `theorem Confidence.foo` inside
462         # `MDP.EstimatedModelPlan` and inside `MDP.FiniteBatchModel` denotes
463         # two different declarations.  Treating every dotted name as absolute
464         # collapsed those declarations in the documentation index as the RL
465         # tree grew.
466         full_name = ".".join(namespace_parts + [name])
467         modifiers = declaration_modifiers.strip().split()
468         declarations.append(
469             {
470                 "kind": declaration_kind,
471                 "name": name,
472                 "full_name": full_name,
473                 "file": rel_source(path),
474                 "line": lineno,
475                 "statement": compact_statement(lines, lineno - 1),
476                 "docstring": docstring_before(lines, lineno - 1),
477                 "private": "private" in modifiers,
478                 "placeholder": False,
479                 "anchor": decl_anchor(full_name),
480             }
481         )
482 
483     for index, declaration in enumerate(declarations):
484         start = declaration["line"] - 1
485         stop = declarations[index + 1]["line"] - 1 if index + 1 < len(declarations) else len(lines)
486         body = "\n".join(visible_lines[start:stop])
487         declaration["placeholder"] = bool(PLACEHOLDER_RE.search(body))
488 
489     docstring = module_docstring(text)
490     return {
491         "name": module_name(path),
492         "file": rel_source(path),
493         "slug": module_slug(module_name(path)),
494         "imports": imports,
495         "docstring": docstring,
496         "summary": markdown_prose_summary(docstring),
497         "declarations": declarations,
498         "placeholder_count": sum(1 for decl in declarations if decl["placeholder"]),
499     }

510 def assign_chapters(modules: list[dict[str, Any]], chapters: list[dict[str, Any]]) -> None:
511     for module in modules:
512         assigned = None
513         assigned_specificity = (-1, -1)
514         for chapter in chapters:
515             for pattern in chapter["module_globs"]:
516                 if not fnmatch.fnmatch(module["file"], pattern):
517                     continue
518                 has_glob = any(token in pattern for token in "*?[")
519                 specificity = (0 if has_glob else 1, sum(token not in "*?[]" for token in pattern))
520                 if specificity > assigned_specificity:
521                     assigned = chapter
522                     assigned_specificity = specificity
523         if assigned is None:
524             assigned = chapters[0]
525         module["chapter"] = assigned["slug"]
526         module["chapter_title"] = assigned["short_title"]
527         for declaration in module["declarations"]:
528             declaration["chapter"] = assigned["slug"]
529             declaration["chapter_title"] = assigned["short_title"]
530             declaration["module"] = module["name"]
531             declaration["module_slug"] = module["slug"]

579 def highlight_lean(statement: str) -> str:
580     token_re = re.compile(
581         r'("(?:\\.|[^"\\])*")'
582         r"|(\b(?:theorem|lemma|def|abbrev|structure|class|inductive|axiom|opaque|where|by|fun|let|in|if|then|else|match|with|forall|exists|noncomputable|private|protected)\b)"
583         r"|(\b\d+(?:\.\d+)?\b)"
584     )
585     pieces: list[str] = []
586     cursor = 0
587     for match in token_re.finditer(statement):
588         pieces.append(html.escape(statement[cursor : match.start()]))
589         value = html.escape(match.group(0))
590         if match.group(1):
591             pieces.append(f'<span class="str">{value}</span>')
592         elif match.group(2):
593             pieces.append(f'<span class="kw">{value}</span>')
594         else:
595             pieces.append(f'<span class="num">{value}</span>')
596         cursor = match.end()
597     pieces.append(html.escape(statement[cursor:]))
598     return "".join(pieces)

1024 def validate_textbook_spine(
1025     spine: dict[str, Any],
1026     decl_by_name: dict[str, dict[str, Any]],
1027 ) -> None:
1028     source = spine.get("canonical_source", {})
1029     if source.get("official_url") != PRIMARY_TEXTBOOK_URL:
1030         raise ValueError("textbook spine must use the official author-hosted PDF")
1031     if source.get("doi") != "10.1017/9781108571401":
1032         raise ValueError("textbook spine DOI is not the canonical CUP identifier")
1033     chapters = spine.get("chapters", [])
1034     if [chapter.get("number") for chapter in chapters] != list(range(13, 18)):
1035         raise ValueError("textbook spine chapters must be ordered exactly 13 through 17")
1036     if len({chapter.get("slug") for chapter in chapters}) != len(chapters):
1037         raise ValueError("textbook spine chapter slugs must be unique")
1038     for chapter in chapters:
1039         if chapter.get("status") not in STATUS_LABELS:
1040             raise ValueError(f"unknown textbook spine status: {chapter.get('status')}")
1041         if not isinstance(chapter.get("pdf_page"), int) or chapter["pdf_page"] < 1:
1042             raise ValueError(f"textbook chapter {chapter.get('number')} lacks a valid PDF page")
1043         if chapter.get("core_status") and chapter.get("core_status") not in STATUS_LABELS:
1044             raise ValueError(f"unknown textbook spine core status: {chapter.get('core_status')}")
1045         for section in chapter.get("section_coverage", []):
1046             if section.get("status") not in STATUS_LABELS:
1047                 raise ValueError(
1048                     f"unknown textbook section status: {section.get('status')}"
1049                 )
1050         for item in chapter.get("lean_correspondence", []):
1051             if item.get("status") == "compiled" and item.get("name") not in decl_by_name:
1052                 raise ValueError(
1053                     f"compiled textbook spine declaration is not indexed: {item.get('name')}"
1054                 )
1055         for name in chapter.get("primary_declarations", []):
1056             if name not in decl_by_name:
1057                 raise ValueError(f"primary textbook declaration is not indexed: {name}")
1058         if chapter["number"] == 16 and chapter.get("status") == "compiled":
1059             required = {
1060                 "BanditRLProof.LowerBounds.IsConsistentPolicyOver",
1061                 "BanditRLProof.LowerBounds.oneArmMeanChange_produces_gap_contract",
