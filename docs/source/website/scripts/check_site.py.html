    website/scripts/check_site.py

    1 """Check anonymous website links, declaration anchors, statuses and metadata."""
2 from pathlib import Path
3 from html.parser import HTMLParser
4 from urllib.parse import urlsplit, unquote
5 import json
6 import re
7 from functools import lru_cache
8 
9 ROOT = Path(__file__).resolve().parents[2]
10 SITE = ROOT / 'docs'
11 
12 @lru_cache(maxsize=None)
13 def resolve_target(parent, path):
14     target = (parent / unquote(path)).resolve()
15     return target / 'index.html' if target.is_dir() else target
16 
17 class Page(HTMLParser):
18     def __init__(self, text):
19         super().__init__()
20         self.ids, self.links = set(), []
21         self.feed(text)
22     def handle_starttag(self, tag, attrs):
23         attrs = dict(attrs)
24         if 'id' in attrs:
25             self.ids.add(attrs['id'])
26         for key in ('href', 'src'):
27             if attrs.get(key):
28                 self.links.append(attrs[key])
29 
30 def main():
31     manifest = json.loads((SITE / 'site-manifest.json').read_text(encoding='utf-8'))
32     assert manifest['anonymous_review']
33     assert manifest['author_count'] == manifest['contributor_count'] == 0
34     assert 'source_commit' not in manifest
35     assert not (SITE / 'contributors/index.html').exists()
36     pages, errors = {}, []
37     for path in SITE.rglob('*.html'):
38         text = path.read_text(encoding='utf-8')
39         pages[path.resolve()] = Page(text)
40         if re.search(r'class="contributor-card|href="[^"]*contributors/index.html', text):
41             errors.append(f'Authorship UI: {path.relative_to(SITE)}')
42         if 'name="referrer" content="no-referrer"' not in text:
43             errors.append(f'Missing referrer policy: {path.relative_to(SITE)}')
44     checked = 0
45     for path, page in pages.items():
46         for link in page.links:
47             url = urlsplit(link)
48             if url.scheme in {'http', 'https', 'mailto', 'data'} or url.netloc:
49                 continue
50             if url.scheme:
51                 errors.append(f'Unresolved URL: {path.relative_to(SITE)}: {link}')
52                 continue
53             target = resolve_target(path.parent, url.path) if url.path else path
54             if not target.is_file() or not target.is_relative_to(SITE.resolve()):
55                 errors.append(f'Missing local link: {path.relative_to(SITE)}: {link}')
56             elif url.fragment and target in pages and unquote(url.fragment) not in pages[target].ids:
57                 errors.append(f'Missing anchor: {path.relative_to(SITE)}: {link}')
58             checked += 1
59     search = json.loads((SITE / 'search-index.json').read_text(encoding='utf-8'))
60     assert len(search) == manifest['declaration_count']
61     assert manifest['placeholder_count'] == 0
62     if errors:
63         print('\n'.join(errors[:40]))
64         raise SystemExit(f'{len(errors)} failures')
65     print(f'PASS: {len(pages)} HTML pages; {checked} local links/anchors; {len(search)} declarations; no authorship UI.')
66 
67 if __name__ == '__main__':
68     main()
