5 # Selected-width summation for OFUL
6 
7 This module converts the compiled logarithmic elliptical-potential endpoint
8 into cumulative clipped confidence-width bounds for arbitrary feature and
9 selected-action sequences. The route is deterministic and finite-horizon.
10 -/

17 /--
18 The confidence width clipped at the inverse-quadratic level. This equals
19 `min 1 (confidenceWidth V x)` when `V` is positive definite.
20 -/
21 noncomputable def clippedConfidenceWidth
22     {Feature : Type u} [Fintype Feature] [DecidableEq Feature]
23     (V : Matrix Feature Feature Real) (x : Feature -> Real) : Real :=
24   Real.sqrt (min 1 (dotProduct x (V⁻¹.mulVec x)))
25 
26 /-- Clipping before the square root agrees with clipping the confidence width. -/
27 theorem clippedConfidenceWidth_eq_min_one_confidenceWidth
28     {Feature : Type u} [Fintype Feature] [DecidableEq Feature]
29     (V : Matrix Feature Feature Real) (hV : V.PosDef)
30     (x : Feature -> Real) :
31     clippedConfidenceWidth V x = min 1 (confidenceWidth V x) := by
32   let q : Real := dotProduct x (V⁻¹.mulVec x)
33   have hq_nonneg : 0 <= q := hV.inv.posSemidef.dotProduct_mulVec_nonneg x
34   change Real.sqrt (min 1 q) = min 1 (Real.sqrt q)
35   by_cases hq_one : q <= 1
36   · rw [min_eq_right hq_one, min_eq_right]
37     exact (Real.sqrt_le_one).2 hq_one
38   · have hone_q : 1 <= q := le_of_lt (lt_of_not_ge hq_one)
39     rw [min_eq_left hone_q, min_eq_left, Real.sqrt_one]
40     exact (Real.one_le_sqrt).2 hone_q

42 /--
43 The cumulative clipped widths of a bounded feature sequence are controlled
44 by the square root of the standard logarithmic elliptical-potential budget.
45 -/
46 theorem sum_range_clippedConfidenceWidth_le_sqrt_mul_sqrt_log
47     {Feature : Type u}
48     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
49     (lambda : Real) (hlambda : 0 < lambda)
50     (history : Nat -> Feature -> Real) (T : Nat) (L2 : Real)
51     (hL2 : 0 <= L2)
52     (hbound : forall t, t < T ->
53       dotProduct (history t) (history t) <= L2) :
54     (Finset.range T).sum (fun t =>
55         clippedConfidenceWidth
56           (regularizedPrefixFeatureGram lambda history t) (history t)) <=
57       Real.sqrt (T : Real) *
58         Real.sqrt
59           (2 * ((Fintype.card Feature : Real) *
60             Real.log (1 +
61               (T * L2) /
62                 ((Fintype.card Feature : Real) * lambda)))) := by
63   let q : Nat -> Real := fun t =>
64     dotProduct (history t)
65       ((regularizedPrefixFeatureGram lambda history t)⁻¹.mulVec (history t))
66   have hq_nonneg : forall t, 0 <= q t := fun t =>
67     regularizedPrefixFeatureGram_inv_quadratic_nonneg
68       lambda hlambda history t (history t)
69   have hpotential :
70       (Finset.range T).sum (fun t => min 1 (q t)) <=
71         2 * ((Fintype.card Feature : Real) *
72           Real.log (1 +
73             (T * L2) /
74               ((Fintype.card Feature : Real) * lambda))) := by
75     simpa only [q] using
76       (standardLogDeterminantAndEllipticalPotential
77         lambda hlambda history T L2 hL2 hbound).2
78   have hcs :=
79     Real.sum_sqrt_mul_sqrt_le
80       (Finset.range T)
81       (f := fun _ => (1 : Real))
82       (g := fun t => min 1 (q t))
83       (fun _ => zero_le_one)
84       (fun t => le_min zero_le_one (hq_nonneg t))
85   calc
86     (Finset.range T).sum (fun t =>
87         clippedConfidenceWidth
88           (regularizedPrefixFeatureGram lambda history t) (history t)) =
89         (Finset.range T).sum (fun t =>
90           Real.sqrt 1 * Real.sqrt (min 1 (q t))) := by
91       apply Finset.sum_congr rfl
92       intro t _ht
93       simp only [clippedConfidenceWidth, q, Real.sqrt_one, one_mul]
94     _ <=
95         Real.sqrt ((Finset.range T).sum (fun _ => (1 : Real))) *
96           Real.sqrt ((Finset.range T).sum (fun t => min 1 (q t))) := hcs
97     _ =
98         Real.sqrt (T : Real) *
99           Real.sqrt ((Finset.range T).sum (fun t => min 1 (q t))) := by
100       simp
101     _ <=
102         Real.sqrt (T : Real) *
103           Real.sqrt
104             (2 * ((Fintype.card Feature : Real) *
105               Real.log (1 +
106                 (T * L2) /
107                   ((Fintype.card Feature : Real) * lambda)))) := by
108       exact mul_le_mul_of_nonneg_left
109         (Real.sqrt_le_sqrt hpotential) (Real.sqrt_nonneg _)

111 /-- Public confidence-width form of the clipped selected-feature sum bound. -/
112 theorem sum_range_min_one_confidenceWidth_le_sqrt_mul_sqrt_log
113     {Feature : Type u}
114     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
115     (lambda : Real) (hlambda : 0 < lambda)
116     (history : Nat -> Feature -> Real) (T : Nat) (L2 : Real)
117     (hL2 : 0 <= L2)
118     (hbound : forall t, t < T ->
119       dotProduct (history t) (history t) <= L2) :
120     (Finset.range T).sum (fun t =>
121         min 1
122           (confidenceWidth
123             (regularizedPrefixFeatureGram lambda history t) (history t))) <=
124       Real.sqrt (T : Real) *
125         Real.sqrt
126           (2 * ((Fintype.card Feature : Real) *
127             Real.log (1 +
128               (T * L2) /
129                 ((Fintype.card Feature : Real) * lambda)))) := by
130   have hsum :=
131     sum_range_clippedConfidenceWidth_le_sqrt_mul_sqrt_log
132       lambda hlambda history T L2 hL2 hbound
133   calc
134     (Finset.range T).sum (fun t =>
135         min 1
136           (confidenceWidth
137             (regularizedPrefixFeatureGram lambda history t) (history t))) =
138         (Finset.range T).sum (fun t =>
139           clippedConfidenceWidth
140             (regularizedPrefixFeatureGram lambda history t) (history t)) := by
141       apply Finset.sum_congr rfl
142       intro t _ht
143       exact
144         (clippedConfidenceWidth_eq_min_one_confidenceWidth
145           (regularizedPrefixFeatureGram lambda history t)
146           (regularizedPrefixFeatureGram_posDef
147             lambda hlambda history t)
148           (history t)).symm
149     _ <= _ := hsum

151 /--
152 Raw selected-feature widths satisfy the same bound when every charged width
153 is at most one.
154 -/
155 theorem sum_range_confidenceWidth_le_sqrt_mul_sqrt_log_of_width_le_one
156     {Feature : Type u}
157     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
158     (lambda : Real) (hlambda : 0 < lambda)
159     (history : Nat -> Feature -> Real) (T : Nat) (L2 : Real)
160     (hL2 : 0 <= L2)
161     (hbound : forall t, t < T ->
162       dotProduct (history t) (history t) <= L2)
163     (hwidth : forall t, t < T ->
164       confidenceWidth
165         (regularizedPrefixFeatureGram lambda history t) (history t) <= 1) :
166     (Finset.range T).sum (fun t =>
167         confidenceWidth
168           (regularizedPrefixFeatureGram lambda history t) (history t)) <=
169       Real.sqrt (T : Real) *
170         Real.sqrt
171           (2 * ((Fintype.card Feature : Real) *
172             Real.log (1 +
173               (T * L2) /
174                 ((Fintype.card Feature : Real) * lambda)))) := by
175   calc
176     (Finset.range T).sum (fun t =>
177         confidenceWidth
178           (regularizedPrefixFeatureGram lambda history t) (history t)) =
179         (Finset.range T).sum (fun t =>
180           min 1
181             (confidenceWidth
182               (regularizedPrefixFeatureGram lambda history t) (history t))) := by
183       apply Finset.sum_congr rfl
184       intro t ht
185       exact (min_eq_right (hwidth t (Finset.mem_range.mp ht))).symm
186     _ <= _ :=
187       sum_range_min_one_confidenceWidth_le_sqrt_mul_sqrt_log
188         lambda hlambda history T L2 hL2 hbound
