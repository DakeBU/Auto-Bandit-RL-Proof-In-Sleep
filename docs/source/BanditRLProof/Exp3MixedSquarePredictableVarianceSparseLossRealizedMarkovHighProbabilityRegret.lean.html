3 /-!
4 # Sparse-loss control of realized predictable-variance EXP3 regret
5 
6 This module discharges the pathwise armwise loss-mass premise of the small-loss
7 route from a per-round support-cardinality contract. The local proof route is:
8 
9 1. restrict each predictable loss vector to its nonzero support in `arms`;
10 2. use the `[0,1]` loss range to bound its mass by the support cardinality;
11 3. sum a uniform support-cardinality bound over `Finset.range horizon`;
12 4. instantiate the compiled small-loss realized Markov theorem with
13    `lossMassBudget = sparsity * horizon`.
14 
15 The primary generated theorem only requires the support contract almost
16 everywhere under its exact trajectory measure. A compatibility theorem retains
17 the stronger pathwise contract uniform over all generated samples. Neither
18 surface asserts a best-arm first-order bound, tunes `eta` or `gamma`, or
19 replaces Markov overflow.
20 -/

28 /-- Nonzero predictable-loss coordinates among the active arms at one time. -/
29 noncomputable def sampledPredictableLossSupport
30     {Env : Type u} {Action : Type v}
31     [MeasurableSpace Env] [MeasurableSpace Action]
32     (arms : Finset Action) (loss : PredictableLossVector Env Action)
33     (t : Nat) (sample : Env × ((k : Nat) → Action × Real)) : Finset Action := by
34   classical
35   exact arms.filter fun action => predictableLossAt loss t sample action ≠ 0

37 /-- The predictable loss mass at one time is at most the number of active
38 nonzero loss coordinates. -/
39 theorem sampledPredictableLossMassAt_le_supportCard
40     {Env : Type u} {Action : Type v}
41     [MeasurableSpace Env] [MeasurableSpace Action]
42     (arms : Finset Action) (loss : PredictableLossVector Env Action)
43     (t : Nat) (sample : Env × ((k : Nat) → Action × Real)) :
44     arms.sum (fun action => predictableLossAt loss t sample action) ≤
45       (sampledPredictableLossSupport arms loss t sample).card := by
46   classical
47   let support := sampledPredictableLossSupport arms loss t sample
48   have hsum :
49       arms.sum (fun action => predictableLossAt loss t sample action) =
50         support.sum (fun action => predictableLossAt loss t sample action) := by
51     symm
52     apply Finset.sum_subset
53     · exact Finset.filter_subset _ _
54     · intro action haction hnot
55       simp only [support, sampledPredictableLossSupport, Finset.mem_filter,
56         not_and, ne_eq] at hnot
57       exact not_ne_iff.mp (hnot haction)
58   rw [hsum]
59   calc
60     support.sum (fun action => predictableLossAt loss t sample action) ≤
61         support.sum (fun _action => (1 : Real)) := by
62       apply Finset.sum_le_sum
63       intro action _haction
64       exact (predictableLossAt_mem_unitInterval loss t sample action).2
65     _ = (support.card : Real) := by simp

67 /-- A per-round support-cardinality bound for one trajectory supplies its
68 armwise loss-mass budget. -/
69 theorem sampledPredictableLossMassSum_le_sparsity_mul_horizon_of_sample
70     {Env : Type u} {Action : Type v}
71     [MeasurableSpace Env] [MeasurableSpace Action]
72     (arms : Finset Action) (loss : PredictableLossVector Env Action)
73     (horizon sparsity : Nat)
74     (sample : Env × ((k : Nat) → Action × Real))
75     (hsparse : ∀ t, t < horizon →
76       (sampledPredictableLossSupport arms loss t sample).card ≤ sparsity) :
77     sampledPredictableLossMassSum arms loss horizon sample ≤
78       (sparsity : Real) * (horizon : Real) := by
79   unfold sampledPredictableLossMassSum
80   calc
81     (Finset.range horizon).sum (fun t =>
82         arms.sum (fun action => predictableLossAt loss t sample action)) ≤
83         (Finset.range horizon).sum (fun _t => (sparsity : Real)) := by
84       apply Finset.sum_le_sum
85       intro t ht
86       exact
87         (sampledPredictableLossMassAt_le_supportCard arms loss t sample).trans
88           (Nat.cast_le.2 (hsparse t (Finset.mem_range.1 ht)))
89     _ = (sparsity : Real) * (horizon : Real) := by
90       simp [mul_comm]

92 /-- A uniform per-round support-cardinality bound supplies the pathwise armwise
93 loss-mass budget used by the small-loss theorem. -/
94 theorem sampledPredictableLossMassSum_le_sparsity_mul_horizon
95     {Env : Type u} {Action : Type v}
96     [MeasurableSpace Env] [MeasurableSpace Action]
97     (arms : Finset Action) (loss : PredictableLossVector Env Action)
98     (horizon sparsity : Nat)
99     (hsparse : ∀ sample t, t < horizon →
100       (sampledPredictableLossSupport arms loss t sample).card ≤ sparsity)
101     (sample : Env × ((k : Nat) → Action × Real)) :
102     sampledPredictableLossMassSum arms loss horizon sample ≤
103       (sparsity : Real) * (horizon : Real) :=
104   sampledPredictableLossMassSum_le_sparsity_mul_horizon_of_sample
105     arms loss horizon sparsity sample (hsparse sample)
