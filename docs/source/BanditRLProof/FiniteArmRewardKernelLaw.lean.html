17 /-- The largest Hoeffding proxy among a finite family of armwise intervals. -/
18 noncomputable def finiteArmIntervalVarianceProxy
19     {K : Nat} (lo hi : Fin K -> Real) : NNReal :=
20   Finset.univ.sup fun arm => intervalVarianceProxy (lo arm) (hi arm)
21 
22 /-- Every armwise interval proxy is bounded by the finite-arm maximum proxy. -/
23 theorem intervalVarianceProxy_le_finiteArmIntervalVarianceProxy
24     {K : Nat} (lo hi : Fin K -> Real) (arm : Fin K) :
25     intervalVarianceProxy (lo arm) (hi arm) <=
26       finiteArmIntervalVarianceProxy lo hi := by
27   change intervalVarianceProxy (lo arm) (hi arm) <=
28     (Finset.univ : Finset (Fin K)).sup
29       (fun arm => intervalVarianceProxy (lo arm) (hi arm))
30   exact Finset.le_sup
31     (s := (Finset.univ : Finset (Fin K)))
32     (f := fun arm => intervalVarianceProxy (lo arm) (hi arm))
33     (Finset.mem_univ arm)
34 
35 /-- Nondegenerate armwise intervals give a positive finite-arm maximum proxy. -/
36 theorem finiteArmIntervalVarianceProxy_pos
37     {K : Nat} (hK : 0 < K) (lo hi : Fin K -> Real)
38     (hlohi : forall arm, lo arm < hi arm) :
39     0 < ((finiteArmIntervalVarianceProxy lo hi : NNReal) : Real) := by
40   let arm : Fin K := Fin.mk 0 hK
41   have harmPos :
42       0 < ((intervalVarianceProxy (lo arm) (hi arm) : NNReal) : Real) :=
43     intervalVarianceProxy_pos_of_lt (hlohi arm)
44   have harmLe :
45       intervalVarianceProxy (lo arm) (hi arm) <=
46         finiteArmIntervalVarianceProxy lo hi :=
47     intervalVarianceProxy_le_finiteArmIntervalVarianceProxy lo hi arm
48   exact harmPos.trans_le (by exact_mod_cast harmLe)

54 /-- The largest variance proxy in a finite family of arms. -/
55 noncomputable def finiteArmVarianceProxy
56     {K : Nat} (varianceProxy : Fin K -> NNReal) : NNReal :=
57   Finset.univ.sup varianceProxy
58 
59 /-- Every arm proxy is bounded by the finite-arm maximum proxy. -/
60 theorem varianceProxy_le_finiteArmVarianceProxy
61     {K : Nat} (varianceProxy : Fin K -> NNReal) (arm : Fin K) :
62     varianceProxy arm <= finiteArmVarianceProxy varianceProxy := by
63   change varianceProxy arm <=
64     (Finset.univ : Finset (Fin K)).sup varianceProxy
65   exact Finset.le_sup
66     (s := (Finset.univ : Finset (Fin K)))
67     (f := varianceProxy)
68     (Finset.mem_univ arm)
69 
70 /-- A positive member makes the finite-arm maximum proxy positive. -/
71 theorem finiteArmVarianceProxy_pos_of_exists
72     {K : Nat} (varianceProxy : Fin K -> NNReal)
73     (hpos : exists arm, 0 < ((varianceProxy arm : NNReal) : Real)) :
74     0 < ((finiteArmVarianceProxy varianceProxy : NNReal) : Real) := by
75   rcases hpos with ⟨arm, harm⟩
76   have hle : varianceProxy arm <= finiteArmVarianceProxy varianceProxy :=
77     varianceProxy_le_finiteArmVarianceProxy varianceProxy arm
78   exact harm.trans_le (by exact_mod_cast hle)

80 /--
81 The finite-arm maximum padded by one, providing a strictly positive tuning
82 proxy even when all genuine armwise proxies are zero.
83 -/
84 noncomputable def finiteArmPositiveVarianceProxy
85     {K : Nat} (varianceProxy : Fin K -> NNReal) : NNReal :=
86   max 1 (finiteArmVarianceProxy varianceProxy)
87 
88 /-- Every armwise proxy is bounded by the positive padded finite-arm proxy. -/
89 theorem varianceProxy_le_finiteArmPositiveVarianceProxy
90     {K : Nat} (varianceProxy : Fin K -> NNReal) (arm : Fin K) :
91     varianceProxy arm <= finiteArmPositiveVarianceProxy varianceProxy := by
92   exact
93     (varianceProxy_le_finiteArmVarianceProxy varianceProxy arm).trans
94       (le_max_right _ _)
95 
96 /-- The padded finite-arm proxy is always strictly positive. -/
97 theorem finiteArmPositiveVarianceProxy_pos
98     {K : Nat} (varianceProxy : Fin K -> NNReal) :
99     0 < ((finiteArmPositiveVarianceProxy varianceProxy : NNReal) : Real) := by
100   have hone : (1 : NNReal) <= finiteArmPositiveVarianceProxy varianceProxy :=
101     le_max_left _ _
102   exact zero_lt_one.trans_le (by exact_mod_cast hone)

108 /--
109 Action-indexed probability laws with exact means and direct centered
110 sub-Gaussian witnesses form a context-independent centered reward-kernel law.
111 -/
112 noncomputable def
113     contextIndependentCenteredRewardKernelLaw_of_hasSubgaussianMGF
114     {Context Action : Type}
115     [MeasurableSpace Context] [MeasurableSpace Action]
116     [MeasurableSingletonClass Action] [Countable Action]
117     (armLaw : Action -> Measure Rat)
118     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
119     (armMean : Action -> Rat)
120     (varianceProxy : Action -> NNReal)
121     (hmean : forall arm,
122       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) =
123         ((armMean arm : Rat) : Real))
124     (hsubG : forall arm,
125       HasSubgaussianMGF
126         (fun reward : Rat => (((reward - armMean arm : Rat) : Real)))
127         (varianceProxy arm) (armLaw arm)) :
128     CenteredRewardKernelLaw
129       (contextIndependentOfActionLaws
130         (Context := Context) armLaw hprob)
131       (fun _ arm => armMean arm)
132       (fun _ arm => varianceProxy arm) where
133   integrable := by
134     intro context arm
135     rw [selectedMeasure_contextIndependentOfActionLaws]
136     exact (hsubG arm).integrable
137   integral_eq_zero := by
138     intro context arm
139     haveI : IsProbabilityMeasure (armLaw arm) := hprob arm
140     have hcenter := (hsubG arm).integrable
141     have hconst : Integrable
142         (fun _reward : Rat => ((armMean arm : Rat) : Real))
143         (armLaw arm) := integrable_const _
144     have hraw : Integrable
145         (fun reward : Rat => ((reward : Rat) : Real)) (armLaw arm) := by
146       convert hcenter.add hconst using 1
147       funext reward
148       simp
149     rw [selectedMeasure_contextIndependentOfActionLaws]
150     simp_rw [Rat.cast_sub]
151     rw [integral_sub hraw hconst, hmean arm]
152     simp
153   hasSubgaussianMGF := by
154     intro context arm
155     rw [selectedMeasure_contextIndependentOfActionLaws]
156     exact hsubG arm

163     {Context Action : Type}
164     [MeasurableSpace Context] [MeasurableSpace Action]

168     (armMean : Action -> Rat)
169     (lo hi : Real)
170     (hmeas : forall arm,
171       AEMeasurable (fun reward : Rat => ((reward : Rat) : Real))
172         (armLaw arm))
173     (hbound : forall arm,
174       Filter.Eventually
175         (fun reward : Rat => Set.Icc lo hi ((reward : Rat) : Real))
176         (ae (armLaw arm)))
177     (hmean : forall arm,
178       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) =
179         ((armMean arm : Rat) : Real)) :

182         (Context := Context) armLaw hprob)

185   apply contextIndependentCenteredRewardKernelLaw_of_hasSubgaussianMGF

194       (lo := lo)
195       (hi := hi)
