6 /-!
7 # IID loss-state producer for scheduled half-Tsallis mean gaps
8 
9 This module instantiates the abstract scheduled independent-mean contract with
10 an infinite product of loss states.  The only trajectory-side input is an
11 explicit finite-prefix kernel factorization: the visible prefix through `n`
12 must depend on the loss-state stream only through its coordinates through `n`.
13 -/

22 /-- A predictable loss vector obtained by reading one fresh loss state at each
23 time.  Joint measurability of `value` is the sole evaluation regularity
24 contract; pointwise bounds make this a valid `[0,1]` loss process. -/
25 def iidLossStatePredictableLossVector
26     {LossState : Type u} {Action : Type v}
27     [MeasurableSpace LossState] [MeasurableSpace Action]
28     (value : LossState -> Action -> Real)
29     (hvalue : Measurable (fun input : LossState × Action =>
30       value input.1 input.2))
31     (hvalue_nonneg : ∀ state action, 0 <= value state action)
32     (hvalue_le_one : ∀ state action, value state action <= 1) :
33     Exp3.PredictableLossVector (Nat -> LossState) Action where
34   initial environment action := value (environment 0) action
35   successor n environment _history action := value (environment (n + 1)) action
36   measurable_initial := by
37     have hstate : Measurable (fun input : (Nat -> LossState) × Action =>
38         input.1 0) := (measurable_pi_apply 0).comp measurable_fst
39     have haction : Measurable (fun input : (Nat -> LossState) × Action =>
40         input.2) := measurable_snd
41     exact hvalue.comp (hstate.prodMk haction)
42   measurable_successor n := by
43     have hstate : Measurable (fun input :
44         (Nat -> LossState) ×
45           (History.FinitePairHistory Action Real n × Action) =>
46         input.1 (n + 1)) :=
47       (measurable_pi_apply (n + 1)).comp measurable_fst
48     have haction : Measurable (fun input :
49         (Nat -> LossState) ×
50           (History.FinitePairHistory Action Real n × Action) =>
51         input.2.2) := measurable_snd.comp measurable_snd
52     exact hvalue.comp (hstate.prodMk haction)
53   initial_nonneg environment action := hvalue_nonneg (environment 0) action
54   initial_le_one environment action := hvalue_le_one (environment 0) action
55   successor_nonneg n environment _history action :=
56     hvalue_nonneg (environment (n + 1)) action
57   successor_le_one n environment _history action :=
58     hvalue_le_one (environment (n + 1)) action

78 /-- Extend a finite loss-state prefix to an infinite stream by a fixed fallback
79 state after the prefix endpoint. -/
80 def extendLossStatePrefix
81     {LossState : Type u} (fallback : LossState) (n : Nat)
82     (statePrefix : (i : Finset.Iic n) -> LossState) : Nat -> LossState :=
83   fun t => if h : t <= n then
84     statePrefix ⟨t, Finset.mem_Iic.mpr h⟩ else fallback

224 /-- The stationary mean loss gap induced by one coordinate law. -/
225 noncomputable def iidLossStateMeanGap
226     {LossState : Type u} {Action : Type v}
227     [MeasurableSpace LossState]
228     (law : Measure LossState) (value : LossState -> Action -> Real)
229     (best action : Action) : Real :=
230   ∫ state, value state action - value state best ∂law

232 /-- Exact remaining trajectory-law obligation for the IID route.  At every
233 scheduled successor time, the generated visible pair prefix is a Markov-kernel
234 extension of the corresponding finite loss-state prefix. -/
235 def HasScheduledIIDPrefixKernelFactorization
236     {LossState : Type u} {Action : Type v}
237     [MeasurableSpace LossState] [MeasurableSpace Action]
238     (trajectoryKernel : Kernel (Nat -> LossState)
239       ((k : Nat) -> Action × Real))
240     (horizon : Nat) : Prop :=
241   ∀ n, n + 1 <= horizon ->
242     ∃ prefixKernel : Kernel
243         ((i : Finset.Iic n) -> LossState)
244         (History.FinitePairHistory Action Real n),
245       IsMarkovKernel prefixKernel ∧
246         trajectoryKernel.map (Preorder.frestrictLe n) =
247           prefixKernel.comap (Preorder.frestrictLe n)
248             (Preorder.measurable_frestrictLe n)
