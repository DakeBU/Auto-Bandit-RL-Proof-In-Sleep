5 # Explicit exploration tuning for the random-square realized EXP3 route
6 
7 The random-square theorem already tunes the learning rate independently of
8 the exploration parameter.  This module chooses the remaining exploration
9 parameter from the two confidence scales at failure budget `delta / 4` and
10 obtains a fully explicit generated realized-regret threshold.
11 -/

19 /-- Dividing a total failure probability by four changes the logarithmic
20 budget to `log (4 / delta)`. -/
21 theorem log_one_div_fourth_eq_log_four_div
22     (delta : Real) (hdelta : 0 < delta) :
23     Real.log (1 / (delta / 4)) = Real.log (4 / delta) := by
24   congr 1
25   field_simp [ne_of_gt hdelta]

27 /-- The threshold after controlling both exploration-floor Bernstein radii
28 and the realized-deviation radius by the exploration scale. -/
29 noncomputable def randomSquareBernsteinRealizedExplicitThreshold
30     {Action : Type v} (arms : Finset Action) (gamma : Real)
31     (horizon : Nat) (delta : Real) : Real :=
32   3 * Real.sqrt
33       (4 * (arms.card : Real) * (horizon : Real) *
34         Real.log (arms.card : Real) / delta) +
35     8 * gamma * (horizon : Real)

37 /-- Cubic and quadratic exploration contracts turn the three remaining
38 confidence radii in the learning-rate-tuned threshold into `7 * gamma * T`;
39 together with exploration bias this contributes `8 * gamma * T`. -/
40 theorem randomSquareBernsteinRealizedTunedThreshold_le_explicitThreshold
41     {Action : Type v} [DecidableEq Action]
42     (arms : Finset Action) (hcard_two : 2 <= arms.card)
43     (horizon : Nat) (hhorizon : 0 < horizon)
44     (gamma delta : Real) (hgamma_pos : 0 < gamma)
45     (hgamma_le_half : gamma <= 1 / 2)
46     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
47     (hcubic_confidence :
48       (arms.card : Real) * Real.log (4 / delta) <=
49         gamma ^ 3 * (horizon : Real))
50     (hquadratic_realized :
51       2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
52           Real.log (4 / delta) <=
53         gamma ^ 2 * (horizon : Real)) :
54     randomSquareBernsteinRealizedTunedThreshold
55         arms gamma horizon delta <=
56       randomSquareBernsteinRealizedExplicitThreshold
57         arms gamma horizon delta := by
58   let K : Real := arms.card
59   let T : Real := horizon
60   let budget : Real := Real.log (4 / delta)
61   let variance : Real :=
62     ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real)
63   have hK_one : 1 < K := by
64     dsimp [K]
65     exact_mod_cast hcard_two
66   have hK : 0 < K := lt_trans zero_lt_one hK_one
67   have hT : 0 < T := by
68     dsimp [T]
69     exact_mod_cast hhorizon
70   have hone_lt_four_div : 1 < 4 / delta := by
71     rw [lt_div_iff₀ hdelta]
72     nlinarith
73   have hbudget_pos : 0 < budget := Real.log_pos hone_lt_four_div
74   have hlog_fourth : Real.log (1 / (delta / 4)) = budget := by
75     dsimp [budget]
76     exact log_one_div_fourth_eq_log_four_div delta hdelta
77   have hmax_fourth : max (Real.log (1 / (delta / 4))) 0 = budget := by
78     rw [hlog_fourth]
79     exact max_eq_left hbudget_pos.le
80   have hconfidence :
81       2 * Real.sqrt (T * budget / (gamma / K)) +
82           budget / (gamma / K) <=
83         3 * gamma * T :=
84     bernsteinConfidenceRadius_le_three_mul_gamma_mul_horizon
85       K T budget gamma hK hT hbudget_pos.le hgamma_pos (by linarith)
86         (by simpa [K, T, budget] using hcubic_confidence)
87   have hvariance_pos : 0 < variance := by
88     simpa [variance] using intervalVarianceProxy_zero_one_pos
89   have hrealized :
90       Real.sqrt (2 * (T * variance) * budget) <= gamma * T :=
91     realizedDeviationRadius_le_mul_gamma_mul_horizon
92       T budget variance gamma hT hbudget_pos.le hvariance_pos.le hgamma_pos
93         (by simpa [T, budget, variance] using hquadratic_realized)
94   dsimp [randomSquareBernsteinRealizedTunedThreshold,
95     randomSquareBernsteinRealizedExplicitThreshold,
96     sampledPurePredictableMinusObservedBernsteinConfidenceRadius,
97     sampledComparatorEstimatorBernsteinConfidenceRadius,
98     sampledPredictableRealizedDeviationConfidenceRadius]
99   rw [hmax_fourth, hlog_fourth]
100   change
101     3 * Real.sqrt (4 * K * T * Real.log K / delta) +
102         gamma * T +
103         (2 * Real.sqrt (T * budget / (gamma / K)) +
104           budget / (gamma / K)) +
105         (2 * Real.sqrt (T * budget / (gamma / K)) +
106           budget / (gamma / K)) +
107         Real.sqrt (2 * (T * variance) * budget) <=
108       3 * Real.sqrt (4 * K * T * Real.log K / delta) +
109         8 * gamma * T
110   linarith

185 /-- Cube-root scale required by both `delta / 4` importance-weighted
186 Bernstein radii. -/
187 noncomputable def randomSquareBernsteinConfidenceExplorationScale
188     (K T delta : Real) : Real :=
189   (K * Real.log (4 / delta) / T) ^ (3 : Real)⁻¹
190 
191 /-- Square-root scale required by the `delta / 4` realized-deviation radius. -/
192 noncomputable def randomSquareBernsteinRealizedExplorationScale
193     (T delta : Real) : Real :=
194   Real.sqrt
195     (2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
196       Real.log (4 / delta) / T)
197 
198 /-- Unclipped exploration scale covering both remaining confidence contracts. -/
199 noncomputable def randomSquareBernsteinRawExplorationRate
200     (K T delta : Real) : Real :=
201   max (randomSquareBernsteinConfidenceExplorationScale K T delta)
202     (randomSquareBernsteinRealizedExplorationScale T delta)
203 
204 /-- Explicit exploration schedule clipped into the Hedge stability regime. -/
205 noncomputable def randomSquareBernsteinClippedExplorationRate
206     (K T delta : Real) : Real :=
207   min (1 / 2) (randomSquareBernsteinRawExplorationRate K T delta)

209 theorem randomSquareBernsteinClippedExplorationRate_le_half
210     (K T delta : Real) :
211     randomSquareBernsteinClippedExplorationRate K T delta <= 1 / 2 := by

217     randomSquareBernsteinClippedExplorationRate K T delta =

221 theorem randomSquareBernsteinRawExplorationRate_pos
222     (K T delta : Real) (hK : 0 < K) (hT : 0 < T)
223     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
224     0 < randomSquareBernsteinRawExplorationRate K T delta := by
225   have hone_lt_four_div : 1 < 4 / delta := by
226     rw [lt_div_iff₀ hdelta]
227     nlinarith
228   have hbase : 0 < K * Real.log (4 / delta) / T :=
229     div_pos (mul_pos hK (Real.log_pos hone_lt_four_div)) hT
230   have hscale :
231       0 < randomSquareBernsteinConfidenceExplorationScale K T delta := by
232     exact Real.rpow_pos_of_pos hbase _
233   exact hscale.trans_le (le_max_left _ _)

235 theorem randomSquareBernsteinClippedExplorationRate_pos
236     (K T delta : Real) (hK : 0 < K) (hT : 0 < T)
237     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
238     0 < randomSquareBernsteinClippedExplorationRate K T delta := by

240   · norm_num

244 /-- Transparent large-horizon conditions ensure clipping is inactive. -/
245 theorem randomSquareBernsteinRawExplorationRate_le_half_of_horizon_contracts
246     (K T delta : Real) (hK : 0 < K) (hT : 0 < T)
247     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
248     (hlarge_confidence : 8 * (K * Real.log (4 / delta)) <= T)
249     (hlarge_realized :
250       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
251           Real.log (4 / delta) <= T) :
252     randomSquareBernsteinRawExplorationRate K T delta <= 1 / 2 := by
253   have hone_lt_four_div : 1 < 4 / delta := by
254     rw [lt_div_iff₀ hdelta]
255     nlinarith
256   have hlog : 0 <= Real.log (4 / delta) :=
257     (Real.log_pos hone_lt_four_div).le
258   have hvariance :
259       0 <= ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) :=
260     NNReal.coe_nonneg _
261   rw [randomSquareBernsteinRawExplorationRate]
262   apply max_le
263   · exact rpow_inv_three_le_half_of_eight_mul_le
264       (K * Real.log (4 / delta)) T (mul_nonneg hK.le hlog) hT
265         hlarge_confidence
266   · apply sqrt_div_le_half_of_four_mul_le
267     · exact hT
268     · convert hlarge_realized using 1
269       all_goals ring

271 /-- The clipped schedule satisfies the exact cubic and quadratic contracts
272 consumed by the characterized random-square theorem. -/
273 theorem randomSquareBernsteinClippedExplorationRate_contracts
274     (K T delta : Real) (hK : 0 < K) (hT : 0 < T)
275     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
276     (hlarge_confidence : 8 * (K * Real.log (4 / delta)) <= T)
277     (hlarge_realized :
278       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
279           Real.log (4 / delta) <= T) :
280     let gamma := randomSquareBernsteinClippedExplorationRate K T delta
281     0 < gamma ∧ gamma <= 1 / 2 ∧
282       K * Real.log (4 / delta) <= gamma ^ 3 * T ∧
283       2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
284           Real.log (4 / delta) <= gamma ^ 2 * T := by
285   dsimp only
286   let variance : Real :=
287     ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real)
288   have hone_lt_four_div : 1 < 4 / delta := by
289     rw [lt_div_iff₀ hdelta]
290     nlinarith
291   have hlog : 0 <= Real.log (4 / delta) :=
292     (Real.log_pos hone_lt_four_div).le
293   have hvariance : 0 <= variance := NNReal.coe_nonneg _
294   have hraw :=
295     randomSquareBernsteinRawExplorationRate_le_half_of_horizon_contracts
296       K T delta hK hT hdelta hdelta_le_one hlarge_confidence hlarge_realized
297   have hclip : randomSquareBernsteinClippedExplorationRate K T delta =
298       randomSquareBernsteinRawExplorationRate K T delta :=
299     randomSquareBernsteinClippedExplorationRate_eq_raw K T delta hraw
300   have hconfidence_component :
301       randomSquareBernsteinConfidenceExplorationScale K T delta <=
302         randomSquareBernsteinClippedExplorationRate K T delta := by
303     rw [hclip, randomSquareBernsteinRawExplorationRate]
304     exact le_max_left _ _
305   have hrealized_component :
306       randomSquareBernsteinRealizedExplorationScale T delta <=
307         randomSquareBernsteinClippedExplorationRate K T delta := by
308     rw [hclip, randomSquareBernsteinRawExplorationRate]
309     exact le_max_right _ _
310   refine ⟨randomSquareBernsteinClippedExplorationRate_pos
311       K T delta hK hT hdelta hdelta_le_one,
312     randomSquareBernsteinClippedExplorationRate_le_half K T delta, ?_, ?_⟩
313   · exact numerator_le_cube_mul_of_rpow_inv_three_le
314       (K * Real.log (4 / delta)) T
315         (randomSquareBernsteinClippedExplorationRate K T delta)
316         (mul_nonneg hK.le hlog) hT (by
317           simpa [randomSquareBernsteinConfidenceExplorationScale] using
318             hconfidence_component)
319   · exact numerator_le_sq_mul_of_sqrt_div_le
320       (2 * variance * Real.log (4 / delta)) T
321         (randomSquareBernsteinClippedExplorationRate K T delta)
322         (mul_nonneg (mul_nonneg (by norm_num) hvariance) hlog) hT (by
323           simpa [randomSquareBernsteinRealizedExplorationScale, variance] using
324             hrealized_component)

326 /-- Generated realized-regret tail for the explicit clipped maximum of the
327 confidence cube-root and realized square-root scales. -/
328 theorem sampledPredictable_explicitRandomSquareBernsteinRealizedRegret_tail
329     {Env : Type u} {Action : Type v}
330     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
331     [MeasurableSpace Action] [MeasurableSingletonClass Action]
332     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]

337     (comparator : Action) (hcomparator : comparator ∈ arms)
338     (horizon : Nat) (hhorizon : 0 < horizon)
339     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
340     (hlarge_confidence :
341       8 * ((arms.card : Real) * Real.log (4 / delta)) <=
342         (horizon : Real))
343     (hlarge_realized :
344       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
345           Real.log (4 / delta) <=
346         (horizon : Real)) :

356         (by

359         loss.environment

362             arms gamma horizon delta <=
363           (Finset.range horizon).sum (fun t =>
364               sampledTrajectoryRealizedLossAt t sample) -
365             (Finset.range horizon).sum (fun t =>
366               predictableLossAt loss t sample comparator)} <=
