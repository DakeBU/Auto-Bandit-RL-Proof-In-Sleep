4 /-!
5 # Generated realized EXP3 regret with variance-sensitive mixed-square confidence
6 
7 This module composes the generated predictable Bernstein-square regret theorem
8 with the one-sided realized-minus-exploration deviation tail. The resulting
9 four-event route controls generated selected scalar loss while using the
10 mixed-square second-moment coefficient `K / epsilon`.
11 
12 The square control is fixed-tilt with a deterministic variance coefficient,
13 and the realized deviation retains its bounded-loss Hoeffding/Azuma radius.
14 This is not a general Freedman or ideal EXP3.P theorem.
15 -/

23 /-- Realized selected-loss regret budget whose predictable component uses the
24 variance-sensitive mixed estimator-square event and two Bernstein confidence
25 radii. -/
26 noncomputable def sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget
27     {Action : Type v} (arms : Finset Action) (eta gamma : Real)
28     (horizon : Nat) (deltaSquare deltaConfidence deltaRealized : Real) : Real :=
29   sampledPredictableBernsteinSquareHighProbabilityRegretBudget
30       arms eta gamma horizon deltaSquare deltaConfidence +
31     sampledPredictableRealizedDeviationConfidenceRadius horizon deltaRealized

154 /-- Total-failure form: the mixed-square, pure-cross Bernstein,
155 fixed-comparator Bernstein, and realized-deviation events each receive
156 `delta / 4`. -/
157 theorem sampledPredictable_bernsteinSquareRealizedHighProbabilityRegret_tail_total_delta
158     {Env : Type u} {Action : Type v}
159     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
160     [MeasurableSpace Action] [MeasurableSingletonClass Action]
161     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]

163     (arms : Finset Action) (harms : arms.Nonempty)
164     (eta gamma : Real) (heta : 0 < eta)
165     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
166     (loss : PredictableLossVector Env Action)
167     (comparator : Action) (hcomparator : comparator ∈ arms)
168     (horizon : Nat) (hhorizon : 0 < horizon)
169     (delta : Real) (hdelta : 0 < delta) :
170     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
171       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
172     mu {sample |

174             arms eta gamma horizon (delta / 4) (delta / 4) (delta / 4) <=
175           (Finset.range horizon).sum (fun t =>

177             (Finset.range horizon).sum (fun t =>
178               predictableLossAt loss t sample comparator)} <=
179       ENNReal.ofReal delta := by
