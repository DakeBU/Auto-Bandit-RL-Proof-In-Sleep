6 # Interiority of half-Tsallis finite-simplex minimizers
7 
8 The square-root regularizer has an infinite inward slope at a zero coordinate.
9 This module makes that boundary argument finite and algebraic: transfer a
10 sufficiently small positive mass from any positive donor coordinate to the
11 zero coordinate and contradict global minimality.
12 -/

19 /-- A one-sided transfer from a positive donor to a zero coordinate stays in
20 the finite simplex. -/
21 theorem finiteSimplex_simplexPairShift_of_eq_zero_of_le
22     {Action : Type u} [DecidableEq Action]
23     (arms : Finset Action) (p : Action -> Real) (i j : Action) (t : Real)
24     (hij : i ≠ j) (hp : FTRL.finiteSimplex arms p)
25     (hi : i ∈ arms) (hj : j ∈ arms)
26     (hpi : p i = 0) (ht : 0 <= t) (htj : t <= p j) :
27     FTRL.finiteSimplex arms (simplexPairShift p i j t) := by
28   constructor
29   · intro a ha
30     by_cases hai : a = i
31     · subst a
32       simp [simplexPairShift, pairDirection, hij, hpi, ht]
33     · by_cases haj : a = j
34       · subst a
35         simp [simplexPairShift, pairDirection, hai, sub_eq_add_neg]
36         linarith
37       · simp [simplexPairShift, pairDirection, hai, haj, hp.1 a ha]
38   · rw [sum_simplexPairShift arms p i j t hi hj, hp.2]

40 /-- Exact linear-loss change under a two-coordinate simplex transfer. -/
41 theorem linearLoss_simplexPairShift_eq
42     {Action : Type u} [DecidableEq Action]
43     (arms : Finset Action) (p score : Action -> Real) (i j : Action) (t : Real)
44     (hi : i ∈ arms) (hj : j ∈ arms) :
45     FTRL.linearLoss arms (simplexPairShift p i j t) score =
46       FTRL.linearLoss arms p score + t * (score i - score j) := by
47   unfold FTRL.linearLoss simplexPairShift
48   simp_rw [add_mul]
49   rw [Finset.sum_add_distrib]
50   simp_rw [mul_assoc]
51   rw [← Finset.mul_sum,
52     sum_pairDirection_mul arms i j score hi hj]

54 /-- Exact square-root-sum change when mass enters a zero coordinate. -/
55 theorem sum_sqrt_simplexPairShift_eq_of_eq_zero
56     {Action : Type u} [DecidableEq Action]
57     (arms : Finset Action) (p : Action -> Real) (i j : Action) (t : Real)
58     (hij : i ≠ j) (hi : i ∈ arms) (hj : j ∈ arms) (hpi : p i = 0) :
59     arms.sum (fun a => Real.sqrt (simplexPairShift p i j t a)) =
60       arms.sum (fun a => Real.sqrt (p a)) + Real.sqrt t +
61         Real.sqrt (p j - t) - Real.sqrt (p j) := by
62   have hpoint : forall a, a ∈ arms ->
63       Real.sqrt (simplexPairShift p i j t a) =
64         Real.sqrt (p a) +
65           (if a = i then Real.sqrt t - Real.sqrt (p i) else 0) +
66           (if a = j then Real.sqrt (p j - t) - Real.sqrt (p j) else 0) := by
67     intro a ha
68     by_cases hai : a = i
69     · subst a
70       simp [simplexPairShift, pairDirection, hij, hpi]
71     · by_cases haj : a = j
72       · subst a
73         simp [simplexPairShift, pairDirection, hai, sub_eq_add_neg]
74       · simp [simplexPairShift, pairDirection, hai, haj]
75   calc
76     arms.sum (fun a => Real.sqrt (simplexPairShift p i j t a)) =
77         arms.sum (fun a =>
78           Real.sqrt (p a) +
79             (if a = i then Real.sqrt t - Real.sqrt (p i) else 0) +
80             (if a = j then Real.sqrt (p j - t) - Real.sqrt (p j) else 0)) :=
81       Finset.sum_congr rfl hpoint
82     _ = arms.sum (fun a => Real.sqrt (p a)) + Real.sqrt t +
83           Real.sqrt (p j - t) - Real.sqrt (p j) := by
84       rw [Finset.sum_add_distrib, Finset.sum_add_distrib]
85       simp [hi, hj, hpi]
86       ring

88 /-- Removing mass `t` from a positive coordinate loses at most
89 `t / sqrt r` of square-root mass. -/
90 theorem sqrt_sub_sqrt_sub_le_div_sqrt
91     {r t : Real} (hr : 0 < r) (ht : 0 <= t) (htr : t <= r) :
92     Real.sqrt r - Real.sqrt (r - t) <= t / Real.sqrt r := by
93   have hsr : 0 < Real.sqrt r := Real.sqrt_pos.2 hr
94   have hrt : 0 <= r - t := sub_nonneg.2 htr
95   have hsle : Real.sqrt (r - t) <= Real.sqrt r :=
96     Real.sqrt_le_sqrt (by linarith)
97   have hsrSq : (Real.sqrt r) ^ 2 = r := Real.sq_sqrt hr.le
98   have hsubSq : (Real.sqrt (r - t)) ^ 2 = r - t := Real.sq_sqrt hrt
99   apply (le_div_iff₀ hsr).2
100   nlinarith [mul_nonneg (sub_nonneg.2 hsle) (Real.sqrt_nonneg (r - t))]

102 /-- At a zero coordinate, the square-root gain dominates any fixed linear
103 slope for a sufficiently small positive transfer from a positive donor. -/
104 theorem exists_transfer_strictly_improves_half_objective
105     (r d : Real) (hr : 0 < r) :
106     exists t, 0 < t ∧ t < r ∧
107       d * t < 2 * (Real.sqrt t + Real.sqrt (r - t) - Real.sqrt r) := by
108   let C : Real := |d| + 2 / Real.sqrt r
109   let s : Real := min (Real.sqrt r / 2) (1 / (C + 1))
110   let t : Real := s ^ 2
111   have hsr : 0 < Real.sqrt r := Real.sqrt_pos.2 hr
112   have hC : 0 < C := by
113     dsimp [C]
114     positivity
115   have hs : 0 < s := by
116     dsimp [s]
117     exact lt_min (by positivity) (by positivity)
118   have hsR : s <= Real.sqrt r / 2 := by
119     dsimp [s]
120     exact min_le_left _ _
121   have hsC : s <= 1 / (C + 1) := by
122     dsimp [s]
123     exact min_le_right _ _
124   have htr : t < r := by
125     have hsLt : s < Real.sqrt r := by linarith
126     have hsrSq : (Real.sqrt r) ^ 2 = r := Real.sq_sqrt hr.le
127     dsimp [t]
128     nlinarith [sq_nonneg (Real.sqrt r - s)]
129   have ht : 0 < t := by
130     dsimp [t]
131     positivity
132   have hsqrtT : Real.sqrt t = s := by
133     dsimp [t]
134     simpa using Real.sqrt_sq hs.le
135   have hdonor : Real.sqrt r - Real.sqrt (r - t) <= t / Real.sqrt r :=
136     sqrt_sub_sqrt_sub_le_div_sqrt hr ht.le htr.le
137   have hd : d * t <= |d| * t :=
138     mul_le_mul_of_nonneg_right (le_abs_self d) ht.le
139   have hratio : C / (C + 1) < 1 := by
140     apply (div_lt_one (by linarith)).2
141     linarith
142   have hCs : C * s < 1 := by
143     calc
144       C * s <= C * (1 / (C + 1)) :=
145         mul_le_mul_of_nonneg_left hsC hC.le
146       _ = C / (C + 1) := by ring
147       _ < 1 := hratio
148   have hCss : C * s * s < s := by
149     have := mul_lt_mul_of_pos_right hCs hs
150     simpa [mul_assoc] using this
151   refine ⟨t, ht, htr, ?_⟩
152   have hbound :
153       d * t - 2 * (Real.sqrt t + Real.sqrt (r - t) - Real.sqrt r) <=
154         |d| * t - 2 * Real.sqrt t + 2 * (t / Real.sqrt r) := by
155     linarith
156   have heq :
157       |d| * t - 2 * Real.sqrt t + 2 * (t / Real.sqrt r) =
158         C * s * s - 2 * s := by
159     rw [hsqrtT]
160     dsimp [C, t]
161     ring
162   rw [heq] at hbound
163   linarith

165 /-- Every supported coordinate of a half-Tsallis finite-simplex minimizer is
166 strictly positive.  No sign condition on `eta` or boundedness condition on the
167 finite score is needed. -/
168 theorem isRegularizedMinimizer_pos
169     {Action : Type u} [DecidableEq Action]
170     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
171     (hpMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
172       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p) :
173     forall action, action ∈ arms -> 0 < p action := by
174   intro i hi
175   have hpiNonneg : 0 <= p i := hpMin.1.1 i hi
176   by_contra hnot
177   have hpi : p i = 0 := le_antisymm (not_lt.mp hnot) hpiNonneg
178   have hsumPos : 0 < arms.sum p := by
179     rw [hpMin.1.2]
180     norm_num
181   rcases (Finset.sum_pos_iff_of_nonneg hpMin.1.1).mp hsumPos with
182     ⟨j, hj, hpj⟩
183   have hij : i ≠ j := by
184     intro heq
185     subst j
186     linarith
187   let d : Real := eta * (score i - score j)
188   rcases exists_transfer_strictly_improves_half_objective (p j) d hpj with
189     ⟨t, ht, htj, himprove⟩
190   have hshiftSimplex :
191       FTRL.finiteSimplex arms (simplexPairShift p i j t) :=
192     finiteSimplex_simplexPairShift_of_eq_zero_of_le
193       arms p i j t hij hpMin.1 hi hj hpi ht.le htj.le
194   have hlinear := linearLoss_simplexPairShift_eq
195     arms p score i j t hi hj
196   have hsqrt := sum_sqrt_simplexPairShift_eq_of_eq_zero
197     arms p i j t hij hi hj hpi
198   have hdecrease :
199       FTRL.regularizedObjective arms eta
200           (negEntropyRegularizer arms (1 / 2 : Real)) score
201           (simplexPairShift p i j t) <
202         FTRL.regularizedObjective arms eta
203           (negEntropyRegularizer arms (1 / 2 : Real)) score p := by
204     rw [regularizedObjective_half_eq, regularizedObjective_half_eq,
205       hlinear, hsqrt]
206     dsimp [d] at himprove
207     nlinarith
208   exact (not_lt_of_ge (hpMin.2 _ hshiftSimplex)) hdecrease

210 /-- A half-Tsallis simplex minimizer automatically supplies the interior
211 stationarity certificate used by the one-step stability route. -/
212 theorem exists_halfTsallisInteriorStationary_of_isRegularizedMinimizer_auto

214     (arms : Finset Action) (eta : Real) (score p : Action -> Real)

222 /--
223 Sampling-law half-Tsallis stability directly from current and chosen-update
224 minimizer certificates.  Strict positivity and common multipliers are derived
225 internally.
226 -/
227 theorem sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half_of_minimizers
228     {Action : Type u} [DecidableEq Action]
229     (arms : Finset Action) (eta : Real)
230     (score prob loss : Action -> Real)
231     (next : Action -> Action -> Real)
232     (heta : 0 < eta)
233     (hprobMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
234       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score prob)
235     (hnextMin : forall chosen, chosen ∈ arms ->
236       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
237         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
238         (fun action => score action +
239           Exp3.importanceWeightedLoss prob loss chosen action)
240         (next chosen))
241     (hloss : forall action, action ∈ arms ->
242       0 <= loss action ∧ loss action <= 1) :
243     arms.sum (fun chosen =>
244         prob chosen *
245           (FTRL.linearLoss arms prob
246               (Exp3.importanceWeightedLoss prob loss chosen) -
247             FTRL.linearLoss arms (next chosen)
248               (Exp3.importanceWeightedLoss prob loss chosen))) <=
249       2 * eta * powerSum arms (1 / 2 : Real) prob := by
250   exact
251     sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half_of_positive_minimizers
252       arms eta score prob loss next heta hprobMin
253       (isRegularizedMinimizer_pos arms eta score prob hprobMin)
254       hnextMin
255       (fun chosen hchosen =>
256         isRegularizedMinimizer_pos arms eta
257           (fun action => score action +
258             Exp3.importanceWeightedLoss prob loss chosen action)
259           (next chosen) (hnextMin chosen hchosen))
260       hloss
