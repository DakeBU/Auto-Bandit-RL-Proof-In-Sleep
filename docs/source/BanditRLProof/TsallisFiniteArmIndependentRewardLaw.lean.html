5 /-!
6 # Finite-arm independent nonidentical reward laws for scheduled half-Tsallis
7 
8 Each round may use a different probability law for every arm, while all
9 roundwise arm laws retain the means of one fixed finite-bandit model. The
10 roundwise product reward vectors are independent across time but need not be
11 identically distributed.
12 -/

19 /-- The independent finite-arm reward-vector law used at round `t`. -/
20 noncomputable def finiteArmIndependentRewardVectorLaw {K : Nat}
21     (armLaw : Nat -> Fin K -> Measure Rat) (t : Nat) :
22     Measure (Fin K -> Rat) :=
23   finiteArmIIDRewardVectorLaw (armLaw t)

25 /-- Every roundwise loss gap has the fixed finite-bandit model gap when the
26 possibly time-varying arm laws preserve the model means. -/
27 theorem independentLossStateTimeVaryingMeanGap_finiteArmIndependentRewardVectorLoss_eq_gap
28     {K : Nat} (model : FiniteBanditModel K)
29     (armLaw : Nat -> Fin K -> Measure Rat)
30     (hprob : forall t arm, IsProbabilityMeasure (armLaw t arm))
31     (hbound : forall t arm, ∀ᵐ reward ∂armLaw t arm,
32       ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1)
33     (hmean : forall t arm,
34       integral (armLaw t arm) (fun reward : Rat => ((reward : Rat) : Real)) =
35         ((model.mean arm : Rat) : Real))
36     (t : Nat) (arm : Fin K) :
37     independentLossStateTimeVaryingMeanGap
38         (finiteArmIndependentRewardVectorLaw armLaw)
39         (fun _ => finiteArmIIDRewardVectorLoss)
40         t model.bestArm arm =
41       ((model.gap arm : Rat) : Real) := by

44     (iidLossStateMeanGap_finiteArmIIDRewardVectorLoss_eq_gap
45       model (armLaw t) (hprob t) (hbound t) (hmean t) arm)
