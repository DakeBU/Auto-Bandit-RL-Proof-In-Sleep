4 /-!
5 # Finite-arm IID reward-law producer for scheduled half-Tsallis regret
6 
7 This module turns one probability law per rational-valued arm into a finite
8 product reward-vector law.  Rewards are clipped pointwise before conversion to
9 losses so that the abstract IID loss-state API has global `[0, 1]` bounds; an
10 almost-sure `[0, 1]` arm-law contract then proves that clipping does not change
11 the coordinate means or the finite-bandit gaps.
12 -/
13

15 namespace Tsallis

19 /-- Pointwise projection of a rational reward into the real unit interval. -/
20 noncomputable def clippedUnitReward (reward : Rat) : Real :=
21   max 0 (min 1 ((reward : Rat) : Real))
22 
23 theorem clippedUnitReward_nonneg (reward : Rat) :
24     0 <= clippedUnitReward reward := by
25   exact le_max_left _ _
26 
27 theorem clippedUnitReward_le_one (reward : Rat) :
28     clippedUnitReward reward <= 1 := by
29   exact max_le zero_le_one (min_le_left _ _)
30 
31 theorem clippedUnitReward_eq_of_mem_Icc (reward : Rat)
32     (hreward : ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1) :
33     clippedUnitReward reward = ((reward : Rat) : Real) := by
34   rw [clippedUnitReward, min_eq_right hreward.2, max_eq_right hreward.1]
35 
36 theorem measurable_clippedUnitReward : Measurable clippedUnitReward := by
37   exact measurable_of_countable _
38 
39 /-- The independent one-round reward vector induced by one law per arm. -/
40 noncomputable def finiteArmIIDRewardVectorLaw {K : Nat}
41     (armLaw : Fin K -> Measure Rat) : Measure (Fin K -> Rat) :=
42   Measure.pi armLaw
43 
44 /-- Convert a sampled reward vector into the selected arm's clipped loss. -/
45 noncomputable def finiteArmIIDRewardVectorLoss {K : Nat}
46     (state : Fin K -> Rat) (arm : Fin K) : Real :=
47   1 - clippedUnitReward (state arm)
48 
49 theorem measurable_finiteArmIIDRewardVectorLoss {K : Nat} :
50     Measurable (fun input : (Fin K -> Rat) × Fin K =>
51       finiteArmIIDRewardVectorLoss input.1 input.2) := by
52   exact measurable_of_countable _
53 
54 theorem finiteArmIIDRewardVectorLoss_nonneg {K : Nat}
55     (state : Fin K -> Rat) (arm : Fin K) :
56     0 <= finiteArmIIDRewardVectorLoss state arm := by
57   dsimp only [finiteArmIIDRewardVectorLoss]
58   linarith [clippedUnitReward_le_one (state arm)]
59 
60 theorem finiteArmIIDRewardVectorLoss_le_one {K : Nat}
61     (state : Fin K -> Rat) (arm : Fin K) :
62     finiteArmIIDRewardVectorLoss state arm <= 1 := by
63   dsimp only [finiteArmIIDRewardVectorLoss]
64   linarith [clippedUnitReward_nonneg (state arm)]
65 
66 theorem integrable_clippedUnitReward (mu : Measure Rat)
67     [IsFiniteMeasure mu] : Integrable clippedUnitReward mu := by
68   refine Integrable.of_bound measurable_clippedUnitReward.aestronglyMeasurable 1 ?_
69   filter_upwards [] with reward
70   rw [Real.norm_eq_abs, abs_le]
71   exact ⟨by linarith [clippedUnitReward_nonneg reward],
72     clippedUnitReward_le_one reward⟩

74 /-- Under an almost-sure unit-interval contract, the product-coordinate
75 clipped reward has exactly the supplied finite-bandit mean. -/
76 theorem integral_finiteArmIIDRewardVectorLaw_clippedUnitReward_eq_mean
77     {K : Nat} (model : FiniteBanditModel K)
78     (armLaw : Fin K -> Measure Rat)
79     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
80     (hbound : forall arm, ∀ᵐ reward ∂armLaw arm,
81       ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1)
82     (hmean : forall arm,
83       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) =
84         ((model.mean arm : Rat) : Real))
85     (arm : Fin K) :
86     integral (finiteArmIIDRewardVectorLaw armLaw)
87         (fun state => clippedUnitReward (state arm)) =
88       ((model.mean arm : Rat) : Real) := by
89   letI (i : Fin K) : IsProbabilityMeasure (armLaw i) := hprob i
90   rw [finiteArmIIDRewardVectorLaw,
91     MeasureTheory.integral_comp_eval
92       (integrable_clippedUnitReward (armLaw arm)).aestronglyMeasurable]
93   calc
94     integral (armLaw arm) clippedUnitReward =
95         integral (armLaw arm) (fun reward : Rat =>
96           ((reward : Rat) : Real)) := by
97       apply integral_congr_ae
98       filter_upwards [hbound arm] with reward hreward
99       exact clippedUnitReward_eq_of_mem_Icc reward hreward
100     _ = ((model.mean arm : Rat) : Real) := hmean arm

102 /-- The one-round IID loss-state mean gap is exactly the rational finite-bandit
103 model gap after coercion to the reals. -/
104 theorem iidLossStateMeanGap_finiteArmIIDRewardVectorLoss_eq_gap
105     {K : Nat} (model : FiniteBanditModel K)
106     (armLaw : Fin K -> Measure Rat)
107     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
108     (hbound : forall arm, ∀ᵐ reward ∂armLaw arm,
109       ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1)
110     (hmean : forall arm,
111       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) =
112         ((model.mean arm : Rat) : Real))
113     (arm : Fin K) :
114     iidLossStateMeanGap (finiteArmIIDRewardVectorLaw armLaw)
115         finiteArmIIDRewardVectorLoss model.bestArm arm =
116       ((model.gap arm : Rat) : Real) := by
117   letI (i : Fin K) : IsProbabilityMeasure (armLaw i) := hprob i
118   have hbestIntegrable : Integrable
119       (fun state : Fin K -> Rat => clippedUnitReward (state model.bestArm))
120       (finiteArmIIDRewardVectorLaw armLaw) := by
121     exact MeasureTheory.integrable_comp_eval
122       (integrable_clippedUnitReward (armLaw model.bestArm))
123   have harmIntegrable : Integrable
124       (fun state : Fin K -> Rat => clippedUnitReward (state arm))
125       (finiteArmIIDRewardVectorLaw armLaw) := by
126     exact MeasureTheory.integrable_comp_eval
127       (integrable_clippedUnitReward (armLaw arm))
128   rw [iidLossStateMeanGap]
129   calc
130     integral (finiteArmIIDRewardVectorLaw armLaw)
131         (fun state => finiteArmIIDRewardVectorLoss state arm -
132           finiteArmIIDRewardVectorLoss state model.bestArm) =
133         integral (finiteArmIIDRewardVectorLaw armLaw)
134           (fun state => clippedUnitReward (state model.bestArm) -
135             clippedUnitReward (state arm)) := by
136       congr 1
137       funext state
138       simp only [finiteArmIIDRewardVectorLoss]
139       ring
140     _ = integral (finiteArmIIDRewardVectorLaw armLaw)
141           (fun state => clippedUnitReward (state model.bestArm)) -
142         integral (finiteArmIIDRewardVectorLaw armLaw)
143           (fun state => clippedUnitReward (state arm)) :=
144       integral_sub hbestIntegrable harmIntegrable
145     _ = ((model.mean model.bestArm : Rat) : Real) -
146         ((model.mean arm : Rat) : Real) := by
147       rw [integral_finiteArmIIDRewardVectorLaw_clippedUnitReward_eq_mean
148           model armLaw hprob hbound hmean model.bestArm,
149         integral_finiteArmIIDRewardVectorLaw_clippedUnitReward_eq_mean
150           model armLaw hprob hbound hmean arm]
151     _ = ((model.gap arm : Rat) : Real) := by
152       by_cases h : arm = model.bestArm
153       · simp [h]
154       · simp [FiniteBanditModel.gap, FiniteBanditModel.bestMean, h]
