3 /-!
4 # Explicit exploration tuning for realized EXP3 Bernstein confidence
5 
6 This module discharges the three dominance premises of the characterized
7 `11 * gamma * T` theorem with an explicit maximum of two cube-root scales and
8 one square-root scale. The schedule is clipped at `1 / 2`; transparent
9 large-horizon premises ensure that the clip is inactive.
10 -/

18 /-- Cube-root scale required by the arm-entropy term. -/
19 noncomputable def bernsteinArmEntropyExplorationScale
20     (K T : Real) : Real :=
21   (K * Real.log K / T) ^ (3 : Real)⁻¹
22 
23 /-- Cube-root scale required by both importance-weighted Bernstein radii. -/
24 noncomputable def bernsteinConfidenceExplorationScale
25     (K T delta : Real) : Real :=
26   (K * Real.log (3 / delta) / T) ^ (3 : Real)⁻¹
27 
28 /-- Square-root scale required by the bounded realized-deviation radius. -/
29 noncomputable def bernsteinRealizedExplorationScale
30     (T delta : Real) : Real :=
31   Real.sqrt
32     (2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
33       Real.log (3 / delta) / T)
34 
35 /-- Unclipped exploration scale simultaneously covering arm entropy, the two
36 importance-weighted confidence radii, and the realized-deviation radius. -/
37 noncomputable def bernsteinRawExplorationRate
38     (K T delta : Real) : Real :=
39   max (bernsteinArmEntropyExplorationScale K T)
40     (max (bernsteinConfidenceExplorationScale K T delta)
41       (bernsteinRealizedExplorationScale T delta))
42 
43 /-- Explicit exploration schedule, clipped into the stability regime
44 `gamma <= 1 / 2`. -/
45 noncomputable def bernsteinClippedExplorationRate
46     (K T delta : Real) : Real :=
47   min (1 / 2) (bernsteinRawExplorationRate K T delta)

49 /-- A cube-root scale is at most one half when its numerator is at most one
50 eighth of the positive horizon. -/
51 theorem rpow_inv_three_le_half_of_eight_mul_le
52     (numerator T : Real) (hnumerator : 0 <= numerator) (hT : 0 < T)
53     (hlarge : 8 * numerator <= T) :
54     (numerator / T) ^ (3 : Real)⁻¹ <= 1 / 2 := by
55   have hcube :
56       ((numerator / T) ^ (3 : Real)⁻¹) ^ (3 : Nat) = numerator / T := by
57     convert Real.rpow_inv_natCast_pow (n := 3)
58       (div_nonneg hnumerator hT.le) (by norm_num) using 1
59   apply le_of_pow_le_pow_left₀ (n := 3) (by norm_num) (by norm_num)
60   rw [hcube]
61   norm_num
62   rw [div_le_iff₀ hT]
63   nlinarith
64 
65 /-- A square-root scale is at most one half when its numerator is at most one
66 quarter of the positive horizon. -/
67 theorem sqrt_div_le_half_of_four_mul_le
68     (numerator T : Real) (hT : 0 < T)
69     (hlarge : 4 * numerator <= T) :
70     Real.sqrt (numerator / T) <= 1 / 2 := by
71   rw [Real.sqrt_le_iff]
72   constructor
73   · norm_num
74   rw [div_le_iff₀ hT]
75   norm_num
76   nlinarith

78 /-- If the cube-root scale is below `gamma`, then the corresponding numerator
79 obeys the cubic dominance contract. -/
80 theorem numerator_le_cube_mul_of_rpow_inv_three_le
81     (numerator T gamma : Real) (hnumerator : 0 <= numerator) (hT : 0 < T)
82     (hroot : (numerator / T) ^ (3 : Real)⁻¹ <= gamma) :
83     numerator <= gamma ^ 3 * T := by
84   have hroot_nonneg : 0 <= (numerator / T) ^ (3 : Real)⁻¹ :=
85     Real.rpow_nonneg (div_nonneg hnumerator hT.le) _
86   have hcube :
87       ((numerator / T) ^ (3 : Real)⁻¹) ^ (3 : Nat) = numerator / T := by
88     convert Real.rpow_inv_natCast_pow (n := 3)
89       (div_nonneg hnumerator hT.le) (by norm_num) using 1
90   have hpow := pow_le_pow_left₀ hroot_nonneg hroot 3
91   rw [hcube] at hpow
92   calc
93     numerator = (numerator / T) * T := by
94       field_simp [ne_of_gt hT]
95     _ <= gamma ^ 3 * T := mul_le_mul_of_nonneg_right hpow hT.le
96 
97 /-- If the square-root scale is below `gamma`, then the corresponding
98 numerator obeys the quadratic dominance contract. -/
99 theorem numerator_le_sq_mul_of_sqrt_div_le
100     (numerator T gamma : Real) (hnumerator : 0 <= numerator) (hT : 0 < T)
101     (hroot : Real.sqrt (numerator / T) <= gamma) :
102     numerator <= gamma ^ 2 * T := by
103   have hpow := pow_le_pow_left₀ (Real.sqrt_nonneg _) hroot 2
104   rw [Real.sq_sqrt (div_nonneg hnumerator hT.le)] at hpow
105   calc
106     numerator = (numerator / T) * T := by
107       field_simp [ne_of_gt hT]
108     _ <= gamma ^ 2 * T := mul_le_mul_of_nonneg_right hpow hT.le

111 theorem bernsteinClippedExplorationRate_le_half (K T delta : Real) :
112     bernsteinClippedExplorationRate K T delta <= 1 / 2 := by

115 /-- If the raw schedule is already stable, clipping leaves it unchanged. -/
116 theorem bernsteinClippedExplorationRate_eq_raw
117     (K T delta : Real)
118     (hraw : bernsteinRawExplorationRate K T delta <= 1 / 2) :
119     bernsteinClippedExplorationRate K T delta =
120       bernsteinRawExplorationRate K T delta := by
121   exact min_eq_right hraw

123 /-- The raw schedule is positive as soon as there are at least two arms and
124 the horizon is positive. -/
125 theorem bernsteinRawExplorationRate_pos
126     (K T delta : Real) (hK_one : 1 < K) (hT : 0 < T) :
127     0 < bernsteinRawExplorationRate K T delta := by
128   have hK : 0 < K := lt_trans zero_lt_one hK_one
129   have hbase : 0 < K * Real.log K / T :=
130     div_pos (mul_pos hK (Real.log_pos hK_one)) hT
131   have hscale : 0 < bernsteinArmEntropyExplorationScale K T := by
132     exact Real.rpow_pos_of_pos hbase _
133   exact hscale.trans_le (le_max_left _ _)
134 
135 /-- The clipped schedule remains positive in the nondegenerate finite-arm,
136 positive-horizon regime. -/
137 theorem bernsteinClippedExplorationRate_pos
138     (K T delta : Real) (hK_one : 1 < K) (hT : 0 < T) :
139     0 < bernsteinClippedExplorationRate K T delta := by
140   apply lt_min
141   · norm_num
142   · exact bernsteinRawExplorationRate_pos K T delta hK_one hT

144 /-- Three transparent horizon inequalities ensure that every raw component is
145 at most one half, so the clipping branch is inactive. -/
146 theorem bernsteinRawExplorationRate_le_half_of_horizon_contracts
147     (K T delta : Real) (hK_one : 1 < K) (hT : 0 < T)
148     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
149     (hlarge_arm : 8 * (K * Real.log K) <= T)
150     (hlarge_confidence : 8 * (K * Real.log (3 / delta)) <= T)
151     (hlarge_realized :
152       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
153           Real.log (3 / delta) <= T) :
154     bernsteinRawExplorationRate K T delta <= 1 / 2 := by
155   have hK : 0 < K := lt_trans zero_lt_one hK_one
156   have hlogK : 0 <= Real.log K := (Real.log_pos hK_one).le
157   have hone_lt_three_div : 1 < 3 / delta := by
158     rw [lt_div_iff₀ hdelta]
159     nlinarith
160   have hlog_conf : 0 <= Real.log (3 / delta) :=
161     (Real.log_pos hone_lt_three_div).le
162   have hvariance :
163       0 <= ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) :=
164     NNReal.coe_nonneg _
165   rw [bernsteinRawExplorationRate]
166   apply max_le
167   · exact rpow_inv_three_le_half_of_eight_mul_le
168       (K * Real.log K) T (mul_nonneg hK.le hlogK) hT hlarge_arm
169   apply max_le
170   · exact rpow_inv_three_le_half_of_eight_mul_le
171       (K * Real.log (3 / delta)) T (mul_nonneg hK.le hlog_conf) hT
172         hlarge_confidence
173   · apply sqrt_div_le_half_of_four_mul_le
174     · exact hT
175     · convert hlarge_realized using 1
176       all_goals ring

178 /-- Under the large-horizon regime, the clipped schedule satisfies positivity,
179 stability, both cubic dominance contracts, and the realized quadratic
180 dominance contract required by the tuned tail theorem. -/
181 theorem bernsteinClippedExplorationRate_contracts
182     (K T delta : Real) (hK_one : 1 < K) (hT : 0 < T)
183     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
184     (hlarge_arm : 8 * (K * Real.log K) <= T)
185     (hlarge_confidence : 8 * (K * Real.log (3 / delta)) <= T)
186     (hlarge_realized :
187       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
188           Real.log (3 / delta) <= T) :
189     let gamma := bernsteinClippedExplorationRate K T delta
190     0 < gamma ∧ gamma <= 1 / 2 ∧
191       K * Real.log K <= gamma ^ 3 * T ∧
192       K * Real.log (3 / delta) <= gamma ^ 3 * T ∧
193       2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
194           Real.log (3 / delta) <= gamma ^ 2 * T := by
195   dsimp only
196   let variance : Real :=
197     ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real)
198   have hK : 0 < K := lt_trans zero_lt_one hK_one
199   have hlogK : 0 <= Real.log K := (Real.log_pos hK_one).le
200   have hone_lt_three_div : 1 < 3 / delta := by
201     rw [lt_div_iff₀ hdelta]
202     nlinarith
203   have hlog_conf : 0 <= Real.log (3 / delta) :=
204     (Real.log_pos hone_lt_three_div).le
205   have hvariance : 0 <= variance := by
206     exact NNReal.coe_nonneg _
207   have hraw := bernsteinRawExplorationRate_le_half_of_horizon_contracts
208     K T delta hK_one hT hdelta hdelta_le_one hlarge_arm hlarge_confidence
209       hlarge_realized
210   have hclip : bernsteinClippedExplorationRate K T delta =
211       bernsteinRawExplorationRate K T delta :=
212     bernsteinClippedExplorationRate_eq_raw K T delta hraw
213   have harm_component : bernsteinArmEntropyExplorationScale K T <=
214       bernsteinClippedExplorationRate K T delta := by
215     rw [hclip, bernsteinRawExplorationRate]
216     exact le_max_left _ _
217   have hconfidence_component :
218       bernsteinConfidenceExplorationScale K T delta <=
219         bernsteinClippedExplorationRate K T delta := by
220     rw [hclip, bernsteinRawExplorationRate]
221     exact le_trans (le_max_left _ _) (le_max_right _ _)
222   have hrealized_component : bernsteinRealizedExplorationScale T delta <=
223       bernsteinClippedExplorationRate K T delta := by
224     rw [hclip, bernsteinRawExplorationRate]
225     exact le_trans (le_max_right _ _) (le_max_right _ _)
226   refine ⟨bernsteinClippedExplorationRate_pos K T delta hK_one hT,
227     bernsteinClippedExplorationRate_le_half K T delta, ?_, ?_, ?_⟩
228   · exact numerator_le_cube_mul_of_rpow_inv_three_le
229       (K * Real.log K) T (bernsteinClippedExplorationRate K T delta)
230         (mul_nonneg hK.le hlogK) hT (by
231           simpa [bernsteinArmEntropyExplorationScale] using harm_component)
232   · exact numerator_le_cube_mul_of_rpow_inv_three_le
233       (K * Real.log (3 / delta)) T
234         (bernsteinClippedExplorationRate K T delta)
235         (mul_nonneg hK.le hlog_conf) hT (by
236           simpa [bernsteinConfidenceExplorationScale] using
237             hconfidence_component)
238   · exact numerator_le_sq_mul_of_sqrt_div_le
239       (2 * variance * Real.log (3 / delta)) T
240         (bernsteinClippedExplorationRate K T delta)
241         (mul_nonneg (mul_nonneg (by norm_num) hvariance) hlog_conf) hT (by
242           simpa [bernsteinRealizedExplorationScale, variance] using
243             hrealized_component)

245 /-- Generated realized-regret tail for the explicit clipped maximum of the two
246 cube-root Bernstein scales and the realized square-root scale. The three
247 large-horizon premises are sufficient conditions ensuring that clipping is
248 inactive and all contracts of the characterized `11 * gamma * T` theorem
249 hold. -/
250 theorem sampledPredictable_explicitBernsteinRealizedHighProbabilityRegret_tail
251     {Env : Type u} {Action : Type v}
252     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
253     [MeasurableSpace Action] [MeasurableSingletonClass Action]
254     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
255     (prior : Measure Env) [IsProbabilityMeasure prior]
256     (arms : Finset Action) (harms : arms.Nonempty)
257     (hcard_two : 2 <= arms.card)
258     (loss : PredictableLossVector Env Action)
259     (comparator : Action) (hcomparator : comparator ∈ arms)
260     (horizon : Nat) (hhorizon : 0 < horizon)
261     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
262     (hlarge_arm :
263       8 * ((arms.card : Real) * Real.log (arms.card : Real)) <=
264         (horizon : Real))
265     (hlarge_confidence :
266       8 * ((arms.card : Real) * Real.log (3 / delta)) <=
267         (horizon : Real))
268     (hlarge_realized :
269       8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
270           Real.log (3 / delta) <=
271         (horizon : Real)) :
272     let gamma := bernsteinClippedExplorationRate
273       (arms.card : Real) (horizon : Real) delta
274     let eta := bernsteinHighProbabilityLearningRate
275       (arms.card : Real) (horizon : Real) gamma
276     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
277       eta gamma
278         (bernsteinClippedExplorationRate_pos
279           (arms.card : Real) (horizon : Real) delta
280           (by exact_mod_cast hcard_two)
281           (by exact_mod_cast hhorizon)).le
282         (by
283           exact (bernsteinClippedExplorationRate_le_half
284             (arms.card : Real) (horizon : Real) delta).trans (by norm_num))
285         loss.environment
286     mu {sample |
287         11 * gamma * (horizon : Real) <=
288           (Finset.range horizon).sum (fun t =>
289               sampledTrajectoryRealizedLossAt t sample) -
290             (Finset.range horizon).sum (fun t =>
291               predictableLossAt loss t sample comparator)} <=
292       ENNReal.ofReal delta := by
293   dsimp only
294   have hK_one : 1 < (arms.card : Real) := by
295     exact_mod_cast hcard_two
296   have hT : 0 < (horizon : Real) := by
297     exact_mod_cast hhorizon
298   have hcontracts := bernsteinClippedExplorationRate_contracts
299     (arms.card : Real) (horizon : Real) delta hK_one hT hdelta hdelta_le_one
300       hlarge_arm hlarge_confidence hlarge_realized
301   dsimp only at hcontracts
302   rcases hcontracts with
303     ⟨hgamma_pos, hgamma_le_half, hcubic_log, hcubic_confidence,
304       hquadratic_realized⟩
305   exact sampledPredictable_tunedBernsteinRealizedHighProbabilityRegret_tail
306     prior arms harms hcard_two
307       (bernsteinClippedExplorationRate
308         (arms.card : Real) (horizon : Real) delta)
309       hgamma_pos hgamma_le_half loss comparator hcomparator horizon hhorizon
