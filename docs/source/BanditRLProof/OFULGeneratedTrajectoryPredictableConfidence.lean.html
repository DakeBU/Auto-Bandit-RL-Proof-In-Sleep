4 /-!
5 # Predictable canonical-trajectory confidence for OFUL
6 
7 The actual canonical action coordinate is only almost surely equal to the
8 deterministic history selector.  This module constructs the pointwise
9 predictable selector feature under a strict-past filtration, proves its
10 almost-everywhere alignment with the actual selected feature, and transports
11 the compiled uniform confidence and successor-gap tails back to the actual
12 trajectory.
13 -/

23 /--
24 The canonical trajectory filtration strictly before the current coordinate.
25 Level zero is trivial, while level `n + 1` contains exactly coordinates
26 `0, ..., n`.
27 -/
28 def canonicalHistoryTrajectoryBeforeFiltration
29     {K : Nat} :
30     Filtration Nat
31       (inferInstance : MeasurableSpace (Nat -> Fin K × Real)) where
32   seq
33     | 0 => ⊥
34     | n + 1 =>
35         Filtration.piLE (X := fun _ : Nat => Fin K × Real) n
36   mono' := by
37     intro i j hij
38     cases i with
39     | zero => exact bot_le
40     | succ i =>
41         cases j with
42         | zero => exact (Nat.not_succ_le_zero i hij).elim
43         | succ j =>
44             exact
45               (Filtration.piLE (X := fun _ : Nat => Fin K × Real)).mono
46                 (Nat.succ_le_succ_iff.mp hij)
47   le' := by
48     intro i
49     cases i with
50     | zero => exact bot_le
51     | succ i =>
52         exact
53           (Filtration.piLE (X := fun _ : Nat => Fin K × Real)).le i

92 /--
93 Pointwise predictable feature used by scalar-ridge confidence.
94 
95 At time zero it uses the deterministic initial arm.  At time `n + 1` it uses
96 the strict-fold selector evaluated only on coordinates through `n`.
