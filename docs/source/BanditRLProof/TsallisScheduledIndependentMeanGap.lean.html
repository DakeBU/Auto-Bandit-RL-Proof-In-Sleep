6 # Independent-mean producer for scheduled half-Tsallis expected gaps
7 
8 This module turns an independence-plus-global-mean contract for every
9 predictable loss difference into the conditional-mean law used by scheduled
10 self-bounding.  It also exposes the resulting logarithmic square-root-schedule
11 regret theorem on the generated trajectory law.
12 -/

21 /-- For every included time and suboptimal arm, the predictable loss
22 difference is independent of the pre-action trace sigma-algebra and has global
23 mean equal to the arm gap. -/
24 def HasScheduledIndependentMeanGapLaw
25     {Env : Type u} {Action : Type v}
26     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
27     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
28     (arms : Finset Action) (loss : Exp3.PredictableLossVector Env Action)
29     (best : Action) (gap : Action -> Real) (horizon : Nat) : Prop :=
30   ∀ t, t <= horizon -> ∀ action, action ∈ arms.erase best ->
31     let lossDiff := fun sample : Env × ((k : Nat) -> Action × Real) =>
32       Exp3.predictableLossAt loss t sample action -
33         Exp3.predictableLossAt loss t sample best
34     Indep (MeasurableSpace.comap lossDiff inferInstance)
35         (sampledScheduledHalfTsallisPastSigma
36           (Env := Env) (Action := Action) t) mu ∧
37       integral mu lossDiff = gap action

39 /-- Independence from the pre-action trace and the correct global mean imply
40 the scheduled conditional-mean gap law. -/
41 theorem hasScheduledConditionalMeanGapLaw_of_independentMeanGapLaw
42     {Env : Type u} {Action : Type v}
43     [mEnv : MeasurableSpace Env]
44     [mAction : MeasurableSpace Action] [DecidableEq Action]
45     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
46     [IsProbabilityMeasure mu]
47     (arms : Finset Action) (loss : Exp3.PredictableLossVector Env Action)
48     (best : Action) (gap : Action -> Real) (horizon : Nat)
49     (hgap : HasScheduledIndependentMeanGapLaw
50       mu arms loss best gap horizon) :
51     HasScheduledConditionalMeanGapLaw
52       mu arms loss best gap horizon := by
53   intro t ht action haction
54   let lossDiff := fun sample : Env × ((k : Nat) -> Action × Real) =>
55     Exp3.predictableLossAt loss t sample action -
56       Exp3.predictableLossAt loss t sample best
57   let mLoss : MeasurableSpace (Env × ((k : Nat) -> Action × Real)) :=
58     MeasurableSpace.comap lossDiff inferInstance
59   let mPast : MeasurableSpace (Env × ((k : Nat) -> Action × Real)) :=
60     sampledScheduledHalfTsallisPastSigma
61       (Env := Env) (Action := Action) t
62   have hmeasLoss : @Measurable
63       (Env × ((k : Nat) -> Action × Real)) Real
64       Prod.instMeasurableSpace inferInstance lossDiff := by
65     simpa only [lossDiff] using
66       (Exp3.measurable_predictableLossAt loss t action).sub
67         (Exp3.measurable_predictableLossAt loss t best)
68   have hleLoss : mLoss <=
69       (Prod.instMeasurableSpace : MeasurableSpace
70         (Env × ((k : Nat) -> Action × Real))) := by
71     exact Measurable.comap_le hmeasLoss
72   have hlePast : mPast <=
73       (Prod.instMeasurableSpace : MeasurableSpace
74         (Env × ((k : Nat) -> Action × Real))) := by
75     change sampledScheduledHalfTsallisPastSigma
76       (Env := Env) (Action := Action) t <= Prod.instMeasurableSpace
77     exact
78       (sampledScheduledHalfTsallisPastSigma_le
79         (Env := Env) (Action := Action) t)
80   have hstrongLoss : @StronglyMeasurable
81       (Env × ((k : Nat) -> Action × Real)) Real _ mLoss lossDiff :=
82     (comap_measurable lossDiff).stronglyMeasurable
83   rcases hgap t ht action haction with ⟨hindep, hmean⟩
84   have hindep' : Indep mLoss mPast mu := by
85     simpa only [mLoss, mPast, lossDiff] using hindep
86   have hcond :=
87     @MeasureTheory.condExp_indep_eq
88       (Env × ((k : Nat) -> Action × Real)) Real _ _ _
89       mLoss mPast
90       Prod.instMeasurableSpace
91       mu lossDiff hleLoss hlePast inferInstance hstrongLoss hindep'
92   change condExp mPast mu lossDiff =ᵐ[mu] fun _sample => gap action
93   exact hcond.mono (fun _sample hsample => by
94     rw [hsample, hmean])
