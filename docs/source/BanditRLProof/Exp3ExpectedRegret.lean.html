19 /-- Deterministic EXP3 parameter algebra.  If `eta = gamma / K`, exploration
20 is at most one half, and `gamma^2 T` covers `K log K`, then the unoptimized
21 budget is at most `4 gamma T`. -/
22 theorem expectedRegretBudget_le_four_mul_gamma_mul_horizon
23     (K T logK eta gamma : Real)
24     (hK : 0 < K) (hT : 0 <= T)
25     (hgamma_pos : 0 < gamma) (hgamma_le_half : gamma <= 1 / 2)
26     (heta : eta = gamma / K)
27     (hlog_budget : K * logK <= gamma ^ 2 * T) :
28     logK / eta + (eta * (1 / (1 - gamma))) * (K * T) + gamma * T <=
29       4 * gamma * T := by
30   have hK_ne : K ≠ 0 := ne_of_gt hK
31   have hgamma_ne : gamma ≠ 0 := ne_of_gt hgamma_pos
32   have hone_sub_pos : 0 < 1 - gamma := by nlinarith
33   have hfirst_eq : logK / (gamma / K) = K * logK / gamma := by
34     field_simp [hK_ne, hgamma_ne]
35   have hfirst : logK / (gamma / K) <= gamma * T := by
36     rw [hfirst_eq, div_le_iff₀ hgamma_pos]
37     nlinarith [hlog_budget]
38   have hsecond_eq :
39       ((gamma / K) * (1 / (1 - gamma))) * (K * T) =
40         gamma * T / (1 - gamma) := by
41     field_simp [hK_ne, ne_of_gt hone_sub_pos]
42   have hgammaT_nonneg : 0 <= gamma * T := mul_nonneg hgamma_pos.le hT
43   have hone_sub_two_gamma_nonneg : 0 <= 1 - 2 * gamma := by
44     nlinarith
45   have hsecond :
46       ((gamma / K) * (1 / (1 - gamma))) * (K * T) <=
47         2 * gamma * T := by
48     rw [hsecond_eq, div_le_iff₀ hone_sub_pos]
49     nlinarith [mul_nonneg hgammaT_nonneg hone_sub_two_gamma_nonneg]
50   rw [heta]
51   linarith

53 /-- Square-root exploration scale used by the tuned corollary. -/
54 noncomputable def tunedExplorationRate (K T : Real) : Real :=
55   Real.sqrt (K * Real.log K / T)
56 
57 /-- Learning rate paired with `tunedExplorationRate`. -/
58 noncomputable def tunedLearningRate (K T : Real) : Real :=
59   tunedExplorationRate K T / K
60 
61 theorem tunedExplorationRate_pos
62     (K T : Real) (hK_one : 1 < K) (hT : 0 < T) :
63     0 < tunedExplorationRate K T := by
64   apply Real.sqrt_pos.2
65   exact div_pos (mul_pos (by linarith) (Real.log_pos hK_one)) hT
66 
67 theorem tunedExplorationRate_le_half
68     (K T : Real) (hT : 0 < T)
69     (hscale : 4 * K * Real.log K <= T) :
70     tunedExplorationRate K T <= 1 / 2 := by
71   rw [tunedExplorationRate, Real.sqrt_le_iff]
72   constructor
73   · norm_num
74   rw [div_le_iff₀ hT]
75   nlinarith
76 
77 theorem tunedLearningRate_pos
78     (K T : Real) (hK_one : 1 < K) (hT : 0 < T) :
79     0 < tunedLearningRate K T := by
80   exact div_pos (tunedExplorationRate_pos K T hK_one hT) (by linarith)
81 
82 theorem tunedExplorationRate_sq_mul_eq
83     (K T : Real) (hK_one : 1 < K) (hT : 0 < T) :
84     tunedExplorationRate K T ^ 2 * T = K * Real.log K := by
85   rw [tunedExplorationRate,
86     Real.sq_sqrt (div_nonneg
87       (mul_nonneg (by linarith) (Real.log_pos hK_one).le) hT.le)]
88   field_simp [ne_of_gt hT]
89 
90 theorem tunedExplorationRate_mul_eq_sqrt_mul
91     (K T : Real) (hK_one : 1 < K) (hT : 0 < T) :
92     tunedExplorationRate K T * T =
93       Real.sqrt (K * T * Real.log K) := by
94   have hK : 0 <= K := by linarith
95   have hlog : 0 <= Real.log K := (Real.log_pos hK_one).le
96   have hsqrtT_pos : 0 < Real.sqrt T := Real.sqrt_pos.2 hT
97   rw [tunedExplorationRate, Real.sqrt_div' _ hT.le]
98   calc
99     Real.sqrt (K * Real.log K) / Real.sqrt T * T =
100         Real.sqrt (K * Real.log K) * Real.sqrt T := by
101       field_simp [ne_of_gt hsqrtT_pos]
102       rw [Real.sq_sqrt hT.le]
103     _ = Real.sqrt ((K * Real.log K) * T) := by
104       rw [Real.sqrt_mul (mul_nonneg hK hlog)]
105     _ = Real.sqrt (K * T * Real.log K) := by ring_nf

107 /-- The generated predictable EXP3 trajectory has regret at most
108 `4 gamma horizon` when `eta = gamma / |arms|` and the exploration budget
109 dominates `|arms| log |arms|`. -/
110 theorem sampledPredictable_expectedRegret_le_four_mul_gamma_mul_horizon
111     {Env : Type u} {Action : Type v}
112     [MeasurableSpace Env] [StandardBorelSpace Env]
113     [MeasurableSpace Action] [MeasurableSingletonClass Action]
114     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
115     (prior : Measure Env) [IsProbabilityMeasure prior]
116     (arms : Finset Action) (harms : arms.Nonempty)
117     (gamma : Real) (hgamma_pos : 0 < gamma)
118     (hgamma_le_half : gamma <= 1 / 2)
119     (loss : PredictableLossVector Env Action) (horizon : Nat)
120     (hlog_budget :
121       (arms.card : Real) * Real.log arms.card <=
122         gamma ^ 2 * (horizon : Real))
123     (comparator : Action) (hcomparator : comparator ∈ arms) :
124     let eta := gamma / (arms.card : Real)
125     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
126       eta gamma hgamma_pos.le (by linarith) loss.environment
127     integral mu (fun sample =>
128       (Finset.range horizon).sum (fun t =>
129           sampledTrajectoryExploredPredictableLossAt
130             arms eta gamma loss t sample) -
131         (Finset.range horizon).sum (fun t =>
132           predictableLossAt loss t sample comparator)) <=
133       4 * gamma * (horizon : Real) := by
134   dsimp only
135   have hcard_pos : 0 < (arms.card : Real) := by
136     exact_mod_cast harms.card_pos
137   have heta_pos : 0 < gamma / (arms.card : Real) :=
138     div_pos hgamma_pos hcard_pos
139   have hgamma_lt_one : gamma < 1 := by linarith
140   have hbase := sampledPredictable_expectedRegret_le
141     prior arms harms (gamma / (arms.card : Real)) gamma heta_pos
142       hgamma_pos hgamma_lt_one loss horizon comparator hcomparator
143   dsimp only at hbase
144   exact hbase.trans
145     (expectedRegretBudget_le_four_mul_gamma_mul_horizon
146       (arms.card : Real) (horizon : Real) (Real.log arms.card)
147       (gamma / (arms.card : Real)) gamma hcard_pos (Nat.cast_nonneg _)
148       hgamma_pos hgamma_le_half rfl hlog_budget)

150 /-- Generated predictable EXP3 kernel at the square-root exploration and
151 learning rates.  The large-horizon hypotheses discharge the kernel's
152 `0 <= gamma <= 1` contract internally. -/
153 noncomputable def tunedPredictableTrajectoryKernel
154     {Env : Type u} {Action : Type v}
155     [MeasurableSpace Env]
156     [MeasurableSpace Action] [MeasurableSingletonClass Action]
157     [Nonempty Action] [DecidableEq Action]
158     (arms : Finset Action) (harms : arms.Nonempty)
159     (hcard_two : 2 <= arms.card)
160     (loss : PredictableLossVector Env Action) (horizon : Nat)
161     (hhorizon_pos : 0 < horizon)
162     (hscale :
163       4 * (arms.card : Real) * Real.log arms.card <= (horizon : Real)) :
164     Kernel Env (Nat -> Action × Real) :=
165   sampledImportanceWeightedTrajectoryKernel arms harms
166     (tunedLearningRate (arms.card : Real) (horizon : Real))
167     (tunedExplorationRate (arms.card : Real) (horizon : Real))
168     (tunedExplorationRate_pos (arms.card : Real) (horizon : Real)
169       (by
170         have htwo : (2 : Real) <= (arms.card : Real) := by
171           exact_mod_cast hcard_two
172         linarith)
173       (by exact_mod_cast hhorizon_pos)).le
174     ((tunedExplorationRate_le_half (arms.card : Real) (horizon : Real)
175       (by exact_mod_cast hhorizon_pos) hscale).trans (by norm_num))
176     loss.environment

178 /-- Tuned generated-trajectory expected predictable-regret bound.  Under the
179 large-horizon regime `4 |A| log |A| <= T`, the square-root exploration and
180 learning rates give the classical `sqrt(|A| T log |A|)` scale. -/
181 theorem sampledPredictable_expectedRegret_le_four_mul_sqrt
182     {Env : Type u} {Action : Type v}
183     [MeasurableSpace Env] [StandardBorelSpace Env]
184     [MeasurableSpace Action] [MeasurableSingletonClass Action]
185     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
186     (prior : Measure Env) [IsProbabilityMeasure prior]
187     (arms : Finset Action) (harms : arms.Nonempty)
188     (hcard_two : 2 <= arms.card)
189     (loss : PredictableLossVector Env Action) (horizon : Nat)
190     (hhorizon_pos : 0 < horizon)
191     (hscale :
192       4 * (arms.card : Real) * Real.log arms.card <= (horizon : Real))
193     (comparator : Action) (hcomparator : comparator ∈ arms) :
194     let K := (arms.card : Real)
195     let T := (horizon : Real)
196     let gamma := tunedExplorationRate K T
197     let eta := tunedLearningRate K T
198     let mu := prior ⊗ₘ tunedPredictableTrajectoryKernel
199       arms harms hcard_two loss horizon hhorizon_pos hscale
200     integral mu (fun sample =>
201       (Finset.range horizon).sum (fun t =>
202           sampledTrajectoryExploredPredictableLossAt
203             arms eta gamma loss t sample) -
204         (Finset.range horizon).sum (fun t =>
205           predictableLossAt loss t sample comparator)) <=
206       4 * Real.sqrt (K * T * Real.log K) := by
207   dsimp only
208   have hK_one : 1 < (arms.card : Real) := by exact_mod_cast hcard_two
209   have hT : 0 < (horizon : Real) := by exact_mod_cast hhorizon_pos
210   have hgamma_pos :=
211     tunedExplorationRate_pos (arms.card : Real) (horizon : Real) hK_one hT
212   have hgamma_half :=
213     tunedExplorationRate_le_half (arms.card : Real) (horizon : Real)
214       hT hscale
215   have hbudget :
216       (arms.card : Real) * Real.log arms.card <=
217         tunedExplorationRate (arms.card : Real) (horizon : Real) ^ 2 *
218           (horizon : Real) := by
219     rw [tunedExplorationRate_sq_mul_eq (arms.card : Real) (horizon : Real)
220       hK_one hT]
221   have hbound :=
222     sampledPredictable_expectedRegret_le_four_mul_gamma_mul_horizon
223       prior arms harms
224         (tunedExplorationRate (arms.card : Real) (horizon : Real))
225         hgamma_pos hgamma_half loss horizon hbudget comparator hcomparator
226   dsimp only [tunedLearningRate] at hbound ⊢
227   have hnormalized :
228       integral
229         (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
230           (tunedExplorationRate (arms.card : Real) (horizon : Real) /
231             (arms.card : Real))
232           (tunedExplorationRate (arms.card : Real) (horizon : Real))
233           hgamma_pos.le (hgamma_half.trans (by norm_num)) loss.environment)
234         (fun sample =>
235           (Finset.range horizon).sum (fun t =>
236               sampledTrajectoryExploredPredictableLossAt arms
237                 (tunedExplorationRate (arms.card : Real) (horizon : Real) /
238                   (arms.card : Real))
239                 (tunedExplorationRate (arms.card : Real) (horizon : Real))
240                 loss t sample) -
241             (Finset.range horizon).sum (fun t =>
242               predictableLossAt loss t sample comparator)) <=
243         4 * Real.sqrt
244           ((arms.card : Real) * (horizon : Real) * Real.log arms.card) := by
245     calc
246       _ <= 4 * tunedExplorationRate (arms.card : Real) (horizon : Real) *
247           (horizon : Real) := hbound
248       _ = 4 * Real.sqrt
249           ((arms.card : Real) * (horizon : Real) * Real.log arms.card) := by
250         rw [mul_assoc, tunedExplorationRate_mul_eq_sqrt_mul
251           (arms.card : Real) (horizon : Real) hK_one hT]
252   simpa only [tunedPredictableTrajectoryKernel] using hnormalized
