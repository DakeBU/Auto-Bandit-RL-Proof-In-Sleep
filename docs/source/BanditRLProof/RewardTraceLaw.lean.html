4 /-!
5 # Reward-trace law uniqueness
6 
7 This module contains the process-law foundation shared by adaptive bandit
8 routes.  Initial reward marginals and successor regular conditional
9 distributions determine every finite prefix and hence the complete reward-trace
10 law.  The results are independent of any ETC/UCB/Thompson algorithm layer.
11 -/

19 /--
20 The zeroth coordinate of an Ionescu-Tulcea trajectory has the supplied initial
21 law.
22 
23 This is a project-level wrapper over `trajMeasure`, `traj_map_frestrictLe`, and
24 `partialTraj_self`. It is general enough to be tracked as a Mathlib candidate.
25 -/
26 theorem trajMeasure_map_eval_zero
27     {X : Nat -> Type*} [forall n, MeasurableSpace (X n)]
28     (mu0 : Measure (X 0)) [IsProbabilityMeasure mu0]
29     (kernel : (n : Nat) ->
30       ProbabilityTheory.Kernel ((i : Finset.Iic n) -> X i) (X (n + 1)))
31     [forall n, ProbabilityTheory.IsMarkovKernel (kernel n)] :
32     Measure.map (fun trajectory : ((n : Nat) -> X n) => trajectory 0)
33         (ProbabilityTheory.Kernel.trajMeasure mu0 kernel) = mu0 := by
34   rw [ProbabilityTheory.Kernel.trajMeasure]
35   rw [Measure.map_comp _ _ (measurable_pi_apply 0)]
36   let evalPrefix : ((i : Finset.Iic 0) -> X i) -> X 0 :=
37     fun history => history ⟨0, Finset.mem_Iic.mpr le_rfl⟩
38   have heval : (fun trajectory : ((n : Nat) -> X n) => trajectory 0) =
39       evalPrefix ∘ Preorder.frestrictLe 0 := by
40     rfl
41   have hrestrict : Measurable (Preorder.frestrictLe (π := X) 0) := by
42     fun_prop
43   have hevalPrefix : Measurable evalPrefix :=
44     measurable_pi_apply
45       (⟨0, Finset.mem_Iic.mpr le_rfl⟩ : Finset.Iic 0)
46   rw [heval,
47     ProbabilityTheory.Kernel.map_comp_right _ hrestrict hevalPrefix,
48     ProbabilityTheory.Kernel.traj_map_frestrictLe,
49     ProbabilityTheory.Kernel.partialTraj_self,
50     ProbabilityTheory.Kernel.id_map hevalPrefix,
51     Measure.deterministic_comp_eq_map hevalPrefix,
52     Measure.map_map hevalPrefix
53       (MeasurableEquiv.piUnique (fun i : Finset.Iic 0 => X i)).symm.measurable]
54   have hcomp : evalPrefix ∘
55       (MeasurableEquiv.piUnique (fun i : Finset.Iic 0 => X i)).symm = id := by
56     funext value
57     rfl
58   exact
59     (congrArg (fun f => Measure.map f mu0) hcomp).trans Measure.map_id

61 /--
62 Finite reward-prefix law uniqueness from the initial marginal and successor
63 conditional distributions.
64 
65 Only the conditional laws before `n` are required. The proof turns each
66 conditional-distribution identity into a joint prefix/next-reward law with
67 `condDistrib_ae_eq_iff_measure_eq_compProd`, then matches the corresponding
68 Ionescu-Tulcea recurrence.
69 -/
70 theorem rewardTrace_prefix_map_eq_trajMeasure_of_condDistrib
71     {Omega Reward : Type*}
72     [MeasurableSpace Omega]
73     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
74     (mu : Measure Omega) [IsFiniteMeasure mu]
75     (mu0 : Measure Reward) [IsProbabilityMeasure mu0]
76     (reward : Omega -> RewardTrace Reward)
77     (hreward : forall t : Nat,
78       Measurable (fun omega : Omega => reward omega t))
79     (kernel : (n : Nat) ->
80       ProbabilityTheory.Kernel ((i : Finset.Iic n) -> Reward) Reward)
81     [forall n, ProbabilityTheory.IsMarkovKernel (kernel n)]
82     (hzero : Measure.map (fun omega : Omega => reward omega 0) mu = mu0)
83     (n : Nat)

85       ProbabilityTheory.condDistrib
86           (fun omega : Omega => reward omega (i + 1))
87           (fun omega : Omega =>
88             History.finiteRewardHistoryOfTrace (reward omega) i)
89           mu =ᵐ[
90             mu.map (fun omega : Omega =>
91               History.finiteRewardHistoryOfTrace (reward omega) i)]
92         kernel i) :
93     Measure.map
94         (fun omega : Omega =>
95           History.finiteRewardHistoryOfTrace (reward omega) n) mu =
96       Measure.map
97         (fun trajectory : RewardTrace Reward =>
98           History.finiteRewardHistoryOfTrace trajectory n)
99         (ProbabilityTheory.Kernel.trajMeasure
100           (X := fun _ : Nat => Reward) mu0 kernel) := by

102     (X := fun _ : Nat => Reward) mu0 kernel
103   revert hcond
104   induction n with
105   | zero =>
106       intro _hcond
107       let singleton : Reward -> History.FiniteRewardHistory Reward 0 :=
108         (MeasurableEquiv.piUnique
109           (fun _i : Finset.Iic 0 => Reward)).symm
110       let evalExternal : Omega -> Reward := fun omega => reward omega 0
111       let evalCanonical : RewardTrace Reward -> Reward :=
112         fun trajectory => trajectory 0
113       let prefixExternal : Omega -> History.FiniteRewardHistory Reward 0 :=
114         fun omega => History.finiteRewardHistoryOfTrace (reward omega) 0
115       let prefixCanonical :
116           RewardTrace Reward -> History.FiniteRewardHistory Reward 0 :=
117         fun trajectory => History.finiteRewardHistoryOfTrace trajectory 0
118       have hsingleton : Measurable singleton :=
119         (MeasurableEquiv.piUnique
120           (fun _i : Finset.Iic 0 => Reward)).symm.measurable
121       have hevalExternal : Measurable evalExternal := hreward 0
122       have hevalCanonical : Measurable evalCanonical := measurable_pi_apply 0
123       have hprefixExternal : prefixExternal = singleton ∘ evalExternal := by
124         funext omega

126         have hi : i.1 = 0 := Nat.eq_zero_of_le_zero (Finset.mem_Iic.mp i.2)
127         simp [prefixExternal, singleton, evalExternal, hi]
128       have hprefixCanonical : prefixCanonical = singleton ∘ evalCanonical := by
129         funext trajectory

131         have hi : i.1 = 0 := Nat.eq_zero_of_le_zero (Finset.mem_Iic.mp i.2)
132         simp [prefixCanonical, singleton, evalCanonical, hi]
133       calc
134         Measure.map prefixExternal mu =
135             Measure.map (singleton ∘ evalExternal) mu := by rw [hprefixExternal]
136         _ = Measure.map singleton (Measure.map evalExternal mu) := by
137           exact (Measure.map_map hsingleton hevalExternal).symm
138         _ = Measure.map singleton mu0 := by rw [hzero]
139         _ = Measure.map singleton (Measure.map evalCanonical trajMeasure) := by
140           rw [RewardKernel.trajMeasure_map_eval_zero]
141         _ = Measure.map (singleton ∘ evalCanonical) trajMeasure := by
142           exact Measure.map_map hsingleton hevalCanonical
143         _ = Measure.map prefixCanonical trajMeasure := by rw [hprefixCanonical]
144   | succ n ih =>
145       intro hcond
146       have ih := ih (fun i hi => hcond i (lt_trans hi (Nat.lt_succ_self n)))
147       let prefixExternal : Omega -> History.FiniteRewardHistory Reward n :=
148         fun omega => History.finiteRewardHistoryOfTrace (reward omega) n
149       let prefixCanonical :
150           RewardTrace Reward -> History.FiniteRewardHistory Reward n :=
151         fun trajectory => History.finiteRewardHistoryOfTrace trajectory n
152       let nextExternal : Omega -> Reward := fun omega => reward omega (n + 1)
153       let nextCanonical : RewardTrace Reward -> Reward :=
154         fun trajectory => trajectory (n + 1)
155       let pairExternal : Omega -> History.FiniteRewardHistory Reward n × Reward :=
156         fun omega => (prefixExternal omega, nextExternal omega)
157       let pairCanonical :
158           RewardTrace Reward -> History.FiniteRewardHistory Reward n × Reward :=
159         fun trajectory => (prefixCanonical trajectory, nextCanonical trajectory)
160       let extend :
161           History.FiniteRewardHistory Reward n × Reward ->
162             History.FiniteRewardHistory Reward (n + 1) :=
163         fun value =>
164           IicProdIoc (X := fun _ : Nat => Reward) n (n + 1)
165             (value.1,
166               (MeasurableEquiv.piSingleton
167                 (X := fun _ : Nat => Reward) n) value.2)
168       have hprefixExternal : Measurable prefixExternal :=
169         History.measurable_finiteRewardHistoryOfTrace reward hreward n
170       have hprefixCanonical : Measurable prefixCanonical :=
171         History.measurable_finiteRewardHistoryOfTrace
172           (fun trajectory : RewardTrace Reward => trajectory)
173           (fun t => measurable_pi_apply t) n
174       have hnextExternal : Measurable nextExternal := hreward (n + 1)
175       have hnextCanonical : Measurable nextCanonical := measurable_pi_apply (n + 1)
176       have hpairExternal : Measurable pairExternal :=
177         Measurable.prod hprefixExternal hnextExternal
178       have hpairCanonical : Measurable pairCanonical :=
179         Measurable.prod hprefixCanonical hnextCanonical
180       have hextend : Measurable extend := by
181         exact measurable_IicProdIoc.comp
182           (Measurable.prod measurable_fst
183             ((MeasurableEquiv.piSingleton
184               (X := fun _ : Nat => Reward) n).measurable.comp measurable_snd))
185       have hpairLaw : Measure.map pairExternal mu =
186           Measure.map pairCanonical trajMeasure := by
187         calc
188           Measure.map pairExternal mu =
189               Measure.map prefixExternal mu ⊗ₘ kernel n := by
190             exact
191               (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
192                 prefixExternal hnextExternal.aemeasurable (kernel n)).mp
193                 (by simpa [prefixExternal, nextExternal, pairExternal] using
194                   hcond n (Nat.lt_succ_self n))
195           _ = Measure.map prefixCanonical trajMeasure ⊗ₘ kernel n := by
196             rw [ih]
197           _ = Measure.map pairCanonical trajMeasure := by
198             have hprefix_eq : prefixCanonical = Preorder.frestrictLe n := by
199               rfl
200             have hpair_eq : pairCanonical =
201                 (fun trajectory : RewardTrace Reward =>
202                   (Preorder.frestrictLe n trajectory, trajectory (n + 1))) := by
203               rfl
204             rw [hprefix_eq, hpair_eq]
205             exact
206               ProbabilityTheory.Kernel.map_frestrictLe_trajMeasure_compProd_eq_map_trajMeasure
207                 (X := fun _ : Nat => Reward)
208                 (μ₀ := mu0) (κ := kernel) (a := n)
209       have hextendTrace : forall trace : RewardTrace Reward,
210           extend
211               (History.finiteRewardHistoryOfTrace trace n, trace (n + 1)) =
212             History.finiteRewardHistoryOfTrace trace (n + 1) := by
213         intro trace
214         funext j
215         by_cases hj : j.1 <= n
216         · simp [extend, IicProdIoc, hj,
217             History.finiteRewardHistoryOfTrace]
218         · have hj_le : j.1 <= n + 1 := Finset.mem_Iic.mp j.2
219           have hj_eq : j.1 = n + 1 := by omega
220           simp [extend, IicProdIoc, MeasurableEquiv.piSingleton,
221             History.finiteRewardHistoryOfTrace, hj_eq]
222       have hextendExternal : forall omega : Omega,
223           extend (pairExternal omega) =
224             History.finiteRewardHistoryOfTrace (reward omega) (n + 1) := by
225         intro omega
226         change extend
227             (History.finiteRewardHistoryOfTrace (reward omega) n,
228               reward omega (n + 1)) = _
229         exact hextendTrace (reward omega)
230       have hextendCanonical : forall trajectory : RewardTrace Reward,
231           extend (pairCanonical trajectory) =
232             History.finiteRewardHistoryOfTrace trajectory (n + 1) := by
233         intro trajectory
234         change extend
235             (History.finiteRewardHistoryOfTrace trajectory n,
236               trajectory (n + 1)) = _
237         exact hextendTrace trajectory
238       calc
239         Measure.map
240             (fun omega : Omega =>
241               History.finiteRewardHistoryOfTrace (reward omega) (n + 1)) mu =
242             Measure.map (extend ∘ pairExternal) mu := by
243           exact Measure.map_congr <| Filter.Eventually.of_forall <| fun omega =>
244             (hextendExternal omega).symm
245         _ = Measure.map extend (Measure.map pairExternal mu) := by
246           exact (Measure.map_map hextend hpairExternal).symm
247         _ = Measure.map extend (Measure.map pairCanonical trajMeasure) := by
248           rw [hpairLaw]
249         _ = Measure.map (extend ∘ pairCanonical) trajMeasure := by
250           exact Measure.map_map hextend hpairCanonical
251         _ = Measure.map
252             (fun trajectory : RewardTrace Reward =>
253               History.finiteRewardHistoryOfTrace trajectory (n + 1))
254             trajMeasure := by
255           exact Measure.map_congr <| Filter.Eventually.of_forall <| fun trajectory =>
256             hextendCanonical trajectory

258 /--
259 The complete trace law is uniquely determined by its initial marginal and all
260 successor conditional distributions.
261 
262 The finite-prefix theorem above handles every `Finset.Iic n`. Any finite set
263 of time coordinates embeds measurably into one such prefix, so the external
264 law and the Ionescu-Tulcea law have the same finite-dimensional marginals.
265 Mathlib's projective-limit uniqueness then identifies the full measures.
266 -/
267 theorem rewardTrace_map_eq_trajMeasure_of_condDistrib
268     {Omega Reward : Type*}
269     [MeasurableSpace Omega]
270     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
271     (mu : Measure Omega) [IsFiniteMeasure mu]
272     (mu0 : Measure Reward) [IsProbabilityMeasure mu0]
273     (reward : Omega -> RewardTrace Reward)
274     (hreward : forall t : Nat,
275       Measurable (fun omega : Omega => reward omega t))
276     (kernel : (n : Nat) ->
277       ProbabilityTheory.Kernel ((i : Finset.Iic n) -> Reward) Reward)
278     [forall n, ProbabilityTheory.IsMarkovKernel (kernel n)]
279     (hzero : Measure.map (fun omega : Omega => reward omega 0) mu = mu0)
280     (hcond : forall i : Nat,
281       ProbabilityTheory.condDistrib
282           (fun omega : Omega => reward omega (i + 1))
283           (fun omega : Omega =>
284             History.finiteRewardHistoryOfTrace (reward omega) i)
285           mu =ᵐ[mu.map (fun omega : Omega =>
286             History.finiteRewardHistoryOfTrace (reward omega) i)]
287         kernel i) :
288     Measure.map reward mu =
289       ProbabilityTheory.Kernel.trajMeasure
290         (X := fun _ : Nat => Reward) mu0 kernel := by
291   let trajMeasure := ProbabilityTheory.Kernel.trajMeasure
292     (X := fun _ : Nat => Reward) mu0 kernel
293   let finiteLaw := fun I : Finset Nat =>
294     Measure.map I.restrict trajMeasure
295   have hrewardTrace : Measurable reward :=
296     measurable_pi_lambda _ hreward
297   have hcanonical : MeasureTheory.IsProjectiveLimit trajMeasure finiteLaw := by
298     intro I
299     rfl
300   have hexternal :
301       MeasureTheory.IsProjectiveLimit (Measure.map reward mu) finiteLaw := by
302     intro I
303     let n := I.sup id
304     have hi_le : forall i : I, i.1 <= n := by
305       intro i
306       simpa [n] using (Finset.le_sup (f := id) i.2)
307     let select : History.FiniteRewardHistory Reward n -> ((i : I) -> Reward) :=
308       fun history i =>
309         history ⟨i.1, Finset.mem_Iic.mpr (hi_le i)⟩
310     have hselect : Measurable select := by
311       exact measurable_pi_lambda _ (fun i => measurable_pi_apply
312         (⟨i.1, Finset.mem_Iic.mpr (hi_le i)⟩ : Finset.Iic n))
313     have hprefixExternal : Measurable (fun omega : Omega =>
314         History.finiteRewardHistoryOfTrace (reward omega) n) :=
315       History.measurable_finiteRewardHistoryOfTrace reward hreward n
316     have hprefixCanonical : Measurable (fun trajectory : RewardTrace Reward =>
317         History.finiteRewardHistoryOfTrace trajectory n) :=
318       History.measurable_finiteRewardHistoryOfTrace
319         (fun trajectory : RewardTrace Reward => trajectory)
320         (fun t => measurable_pi_apply t) n
321     have hprefix := rewardTrace_prefix_map_eq_trajMeasure_of_condDistrib
322       mu mu0 reward hreward kernel hzero n (fun i _hi => hcond i)
323     have hfinite := congrArg (Measure.map select) hprefix
324     rw [Measure.map_map hselect hprefixExternal,
325       Measure.map_map hselect hprefixCanonical] at hfinite
326     have hselectExternal : select ∘
327         (fun omega : Omega =>
328           History.finiteRewardHistoryOfTrace (reward omega) n) =
329         fun omega => I.restrict (reward omega) := by
330       funext omega i
331       rfl
332     have hselectCanonical : select ∘
333         (fun trajectory : RewardTrace Reward =>
334           History.finiteRewardHistoryOfTrace trajectory n) = I.restrict := by
335       funext trajectory i
336       rfl
337     rw [hselectExternal, hselectCanonical] at hfinite
338     rw [AEMeasurable.map_map_of_aemeasurable
339       (Finset.measurable_restrict I).aemeasurable hrewardTrace.aemeasurable]
340     exact hfinite
341   simpa [trajMeasure] using hexternal.unique hcanonical

343 /--
344 Two complete traces are identically distributed when they share an initial
345 marginal and the same successor conditional-distribution kernels.
346 -/
347 theorem identDistrib_rewardTrace_of_common_condDistrib
348     {Omega Xi Reward : Type*}
349     [MeasurableSpace Omega] [MeasurableSpace Xi]
350     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
351     (mu : Measure Omega) [IsFiniteMeasure mu]
352     (mu' : Measure Xi) [IsFiniteMeasure mu']
353     (mu0 : Measure Reward) [IsProbabilityMeasure mu0]
354     (reward : Omega -> RewardTrace Reward)
355     (reward' : Xi -> RewardTrace Reward)
356     (hreward : forall t : Nat,
357       Measurable (fun omega : Omega => reward omega t))
358     (hreward' : forall t : Nat,
359       Measurable (fun xi : Xi => reward' xi t))
360     (kernel : (n : Nat) ->
361       ProbabilityTheory.Kernel ((i : Finset.Iic n) -> Reward) Reward)
362     [forall n, ProbabilityTheory.IsMarkovKernel (kernel n)]
363     (hzero : Measure.map (fun omega : Omega => reward omega 0) mu = mu0)
364     (hzero' : Measure.map (fun xi : Xi => reward' xi 0) mu' = mu0)
365     (hcond : forall i : Nat,
366       ProbabilityTheory.condDistrib
367           (fun omega : Omega => reward omega (i + 1))
368           (fun omega : Omega =>
369             History.finiteRewardHistoryOfTrace (reward omega) i)
370           mu =ᵐ[mu.map (fun omega : Omega =>
371             History.finiteRewardHistoryOfTrace (reward omega) i)]
372         kernel i)
373     (hcond' : forall i : Nat,
374       ProbabilityTheory.condDistrib
375           (fun xi : Xi => reward' xi (i + 1))
376           (fun xi : Xi =>
377             History.finiteRewardHistoryOfTrace (reward' xi) i)
378           mu' =ᵐ[mu'.map (fun xi : Xi =>
379             History.finiteRewardHistoryOfTrace (reward' xi) i)]
380         kernel i) :
381     ProbabilityTheory.IdentDistrib reward reward' mu mu' := by
382   refine ⟨(measurable_pi_lambda _ hreward).aemeasurable,
383     (measurable_pi_lambda _ hreward').aemeasurable, ?_⟩
384   calc
385     Measure.map reward mu =
386         ProbabilityTheory.Kernel.trajMeasure
387           (X := fun _ : Nat => Reward) mu0 kernel :=
388       rewardTrace_map_eq_trajMeasure_of_condDistrib
389         mu mu0 reward hreward kernel hzero hcond
390     _ = Measure.map reward' mu' :=
391       (rewardTrace_map_eq_trajMeasure_of_condDistrib
392         mu' mu0 reward' hreward' kernel hzero' hcond').symm
