6 # Measurable finite Real argmax
7 
8 This module provides a fixed-enumeration maximizer for Real-valued scores on a
9 nonempty finite type.  Unlike a bare `Classical.choose` over existence of a
10 maximum, the explicit fold remains measurable when every score coordinate is
11 measurable in an external parameter.
12 -/

21 private theorem score_le_foldl_select
22     {K : Nat} (scores : Fin K -> Real) (init : Fin K) :
23     forall l : List (Fin K),
24       (forall a : Fin K, List.Mem a l ->
25         scores a <=
26           scores
27             (l.foldl
28               (fun best arm : Fin K =>
29                 if scores best < scores arm then arm else best)
30               init)) /\
31       scores init <=
32         scores
33           (l.foldl
34             (fun best arm : Fin K =>
35               if scores best < scores arm then arm else best)
36             init)
37   | [] => by
38       exact And.intro (by intro _ ha; cases ha) (by simp)
39   | arm :: rest => by
40       let select := fun best arm : Fin K =>
41         if scores best < scores arm then arm else best
42       let next := select init arm
43       have ih := score_le_foldl_select (scores := scores) next rest
44       have harm_next : scores arm <= scores next := by
45         exact if hlt : scores init < scores arm then
46           by simp [next, select, hlt]
47         else
48           by simp [next, select, hlt, le_of_not_gt hlt]
49       exact And.intro
50         (by
51           intro a ha
52           cases ha with
53           | head => exact le_trans harm_next ih.2
54           | tail _ ha => exact ih.1 a ha)
55         (by
56           have hinit_next : scores init <= scores next := by
57             exact if hlt : scores init < scores arm then
58               by simpa [next, select, hlt] using le_of_lt hlt
59             else
60               by simp [next, select, hlt]
61           exact le_trans hinit_next ih.2)

63 private noncomputable def chooseFin {K : Nat} (hK : 0 < K)
64     (scores : Fin K -> Real) : Fin K :=
65   (List.finRange K).foldl
66     (fun best arm : Fin K =>
67       if scores best < scores arm then arm else best)
68     (Fin.mk 0 hK)

70 private theorem score_le_chooseFin {K : Nat} (hK : 0 < K)
71     (scores : Fin K -> Real) (a : Fin K) :
72     scores a <= scores (chooseFin hK scores) := by
73   exact
74     (score_le_foldl_select
75       (scores := scores) (init := Fin.mk 0 hK) (List.finRange K)).1
76       a (List.mem_finRange a)

78 private theorem measurable_selected_score
79     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
80     (scores : Omega -> Fin K -> Real)
81     (hscores : forall a : Fin K,
82       Measurable (fun omega : Omega => scores omega a))
83     (best : Omega -> Fin K) (hbest : Measurable best) :
84     Measurable (fun omega : Omega => scores omega (best omega)) := by
85   classical
86   have hsum : Measurable (fun omega : Omega =>
87       (Finset.univ : Finset (Fin K)).sum (fun a =>
88         if best omega = a then scores omega a else 0)) := by
89     refine Finset.measurable_sum _ ?_
90     intro a _ha
91     exact Measurable.ite
92       (measurableSet_eq_fun hbest measurable_const)
93       (hscores a) measurable_const
94   convert hsum using 1
95   funext omega
96   simp

98 private theorem measurable_foldl_select
99     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
100     (scores : Omega -> Fin K -> Real)
101     (hscores : forall a : Fin K,
102       Measurable (fun omega : Omega => scores omega a))
103     (best : Omega -> Fin K) (hbest : Measurable best) :
104     forall l : List (Fin K),
105       Measurable (fun omega : Omega =>
106         l.foldl
107           (fun best arm : Fin K =>
108             if scores omega best < scores omega arm then arm else best)
109           (best omega))
110   | [] => by simpa using hbest
111   | arm :: rest => by
112       have hselected : Measurable (fun omega : Omega =>
113           scores omega (best omega)) :=
114         measurable_selected_score scores hscores best hbest
115       have hnext : Measurable (fun omega : Omega =>
116           if scores omega (best omega) < scores omega arm then
117             arm
118           else
119             best omega) :=
120         Measurable.ite
121           (measurableSet_lt hselected (hscores arm))
122           measurable_const hbest
123       simpa using
124         (measurable_foldl_select scores hscores
125           (fun omega : Omega =>
126             if scores omega (best omega) < scores omega arm then
127               arm
128             else
129               best omega)
130           hnext rest)

132 private theorem measurable_chooseFin_of_forall_measurable
133     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
134     (hK : 0 < K) (scores : Omega -> Fin K -> Real)
135     (hscores : forall a : Fin K,
136       Measurable (fun omega : Omega => scores omega a)) :
137     Measurable (fun omega : Omega => chooseFin hK (scores omega)) := by
138   unfold chooseFin
139   exact measurable_foldl_select scores hscores
140     (fun _omega : Omega => Fin.mk 0 hK) measurable_const (List.finRange K)

142 /-- A fixed-enumeration maximizer on a nonempty finite type. -/
143 noncomputable def choose {alpha : Type v} [Fintype alpha] [Nonempty alpha]
144     (scores : alpha -> Real) : alpha :=
145   let equiv := Fintype.equivFin alpha
146   equiv.symm
147     (chooseFin (Fintype.card_pos_iff.mpr inferInstance)
148       (fun index => scores (equiv.symm index)))
149 
150 /-- Every score is bounded by the score selected by `choose`. -/
151 theorem score_le_choose {alpha : Type v} [Fintype alpha] [Nonempty alpha]
152     (scores : alpha -> Real) (a : alpha) :
153     scores a <= scores (choose scores) := by
154   let equiv := Fintype.equivFin alpha
155   have hmax := score_le_chooseFin
156     (Fintype.card_pos_iff.mpr inferInstance)
157     (fun index => scores (equiv.symm index)) (equiv a)
158   simpa [choose, equiv] using hmax

160 /-- The fixed-enumeration maximizer is measurable in external parameters. -/
161 theorem measurable_choose_of_forall_measurable
162     {Omega : Type u} {alpha : Type v}
163     [MeasurableSpace Omega] [MeasurableSpace alpha]
164     [Fintype alpha] [Nonempty alpha]
165     (scores : Omega -> alpha -> Real)
166     (hscores : forall a : alpha,
167       Measurable (fun omega : Omega => scores omega a)) :
168     Measurable (fun omega : Omega => choose (scores omega)) := by
169   let equiv := Fintype.equivFin alpha
170   have hfin : Measurable (fun omega : Omega =>
171       chooseFin (Fintype.card_pos_iff.mpr inferInstance)
172         (fun index => scores omega (equiv.symm index))) :=
173     measurable_chooseFin_of_forall_measurable
174       (Fintype.card_pos_iff.mpr inferInstance)
175       (fun omega index => scores omega (equiv.symm index))
176       (fun index => hscores (equiv.symm index))
177   exact (measurable_of_finite equiv.symm).comp hfin
