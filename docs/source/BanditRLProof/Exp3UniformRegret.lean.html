3 /-!
4 # Uniform-horizon predictable EXP3 regret
5 
6 This module closes the small-horizon branch left by the tuned square-root
7 theorem.  It clips the exploration rate at `1/2`, uses the tuned theorem when
8 `4 K log K <= T`, and otherwise uses the pathwise `[0,1]` loss budget.
9 -/

18 /-- Any generated predictable EXP3 process has expected exploration-mixed
19 regret at most the horizon, independently of the learning rate. -/
20 theorem sampledPredictable_exploredExpectedRegret_le_horizon
21     {Env : Type u} {Action : Type v}
22     [MeasurableSpace Env] [StandardBorelSpace Env]
23     [MeasurableSpace Action] [MeasurableSingletonClass Action]
24     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
25     (prior : Measure Env) [IsProbabilityMeasure prior]
26     (arms : Finset Action) (harms : arms.Nonempty)
27     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
28     (hgamma_le_one : gamma <= 1)
29     (loss : PredictableLossVector Env Action) (horizon : Nat)
30     (comparator : Action) :
31     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
32       eta gamma hgamma_nonneg hgamma_le_one loss.environment
33     integral mu (fun sample =>
34       (Finset.range horizon).sum (fun t =>
35           sampledTrajectoryExploredPredictableLossAt
36             arms eta gamma loss t sample) -
37         (Finset.range horizon).sum (fun t =>
38           predictableLossAt loss t sample comparator)) <=
39       (horizon : Real) := by
40   dsimp only
41   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
42     eta gamma hgamma_nonneg hgamma_le_one loss.environment
43   have hexploredSum : Integrable (fun sample =>
44       (Finset.range horizon).sum (fun t =>
45         sampledTrajectoryExploredPredictableLossAt
46           arms eta gamma loss t sample)) mu :=
47     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
48       (fun t sample => sampledTrajectoryExploredPredictableLossAt
49         arms eta gamma loss t sample)
50       (fun t _ht => integrable_sampledTrajectoryExploredPredictableLossAt
51         mu arms harms eta gamma hgamma_nonneg hgamma_le_one loss t)
52   have hcomparatorSum : Integrable (fun sample =>
53       (Finset.range horizon).sum (fun t =>
54         predictableLossAt loss t sample comparator)) mu :=
55     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
56       (fun t sample => predictableLossAt loss t sample comparator)
57       (fun t _ht => integrable_predictableLossAt mu loss t comparator)
58   have hae : ∀ᵐ sample ∂mu,
59       (Finset.range horizon).sum (fun t =>
60           sampledTrajectoryExploredPredictableLossAt
61             arms eta gamma loss t sample) -
62         (Finset.range horizon).sum (fun t =>
63           predictableLossAt loss t sample comparator) <=
64         (horizon : Real) := by
65     filter_upwards [] with sample
66     have hexplored :
67         (Finset.range horizon).sum (fun t =>
68             sampledTrajectoryExploredPredictableLossAt
69               arms eta gamma loss t sample) <= (horizon : Real) := by
70       calc
71         _ <= (Finset.range horizon).sum (fun _t => (1 : Real)) := by
72           apply Finset.sum_le_sum
73           intro t _ht
74           exact (sampledTrajectoryExploredPredictableLossAt_mem_unitInterval
75             arms harms eta gamma hgamma_nonneg hgamma_le_one loss t sample).2
76         _ = (horizon : Real) := by simp
77     have hcomparator : 0 <= (Finset.range horizon).sum (fun t =>
78         predictableLossAt loss t sample comparator) := by
79       apply Finset.sum_nonneg
80       intro t _ht
81       cases t with
82       | zero => exact loss.initial_nonneg sample.1 comparator
83       | succ n =>
84           exact loss.successor_nonneg n sample.1
85             (Preorder.frestrictLe n sample.2) comparator
86     linarith
87   have hmono := integral_mono_ae
88     (hexploredSum.sub hcomparatorSum) (integrable_const _) hae
89   simpa [mu] using hmono

94     {Env : Type u} {Action : Type v}
95     [MeasurableSpace Env] [StandardBorelSpace Env]
96     [MeasurableSpace Action] [MeasurableSingletonClass Action]
97     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
98     (prior : Measure Env) [IsProbabilityMeasure prior]
99     (arms : Finset Action) (harms : arms.Nonempty)
100     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
101     (hgamma_le_one : gamma <= 1)
102     (loss : PredictableLossVector Env Action) (horizon : Nat)
103     (comparator : Action) :
104     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
105       eta gamma hgamma_nonneg hgamma_le_one loss.environment
106     integral mu (fun sample =>
107       (Finset.range horizon).sum (fun t =>
108           sampledTrajectoryRealizedLossAt t sample) -
109         (Finset.range horizon).sum (fun t =>
110           predictableLossAt loss t sample comparator)) <=
111       (horizon : Real) := by
112   dsimp only
113   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
114     eta gamma hgamma_nonneg hgamma_le_one loss.environment
115   have htransport :=
116     sampledPredictableRealized_finiteHorizon_integral_eq_explored
117       prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon
118   have hbase := sampledPredictable_exploredExpectedRegret_le_horizon
119     prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon comparator
120   dsimp only at htransport hbase
121   have hrealizedSum : Integrable (fun sample =>
122       (Finset.range horizon).sum (fun t =>
123         sampledTrajectoryRealizedLossAt t sample)) mu :=
124     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
125       (fun t sample => sampledTrajectoryRealizedLossAt t sample)
126       (fun t _ht => integrable_sampledTrajectoryRealizedLossAt
127         prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss t)

133       (fun t sample => sampledTrajectoryExploredPredictableLossAt
134         arms eta gamma loss t sample)

137   have hcomparatorSum : Integrable (fun sample =>
138       (Finset.range horizon).sum (fun t =>
139         predictableLossAt loss t sample comparator)) mu :=
140     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
141       (fun t sample => predictableLossAt loss t sample comparator)
142       (fun t _ht => integrable_predictableLossAt mu loss t comparator)
143   rw [integral_sub hexploredSum hcomparatorSum] at hbase

148 /-- Exploration rate clipped to the range supported uniformly by the compiled
149 expected-regret route. -/
150 noncomputable def clippedExplorationRate (K T : Real) : Real :=
151   min (1 / 2) (tunedExplorationRate K T)
152 
153 /-- Learning rate paired with `clippedExplorationRate`. -/
154 noncomputable def clippedLearningRate (K T : Real) : Real :=
155   clippedExplorationRate K T / K
156 
157 theorem clippedExplorationRate_nonneg (K T : Real) :
158     0 <= clippedExplorationRate K T := by
159   rw [clippedExplorationRate]
160   exact le_min (by norm_num) (Real.sqrt_nonneg _)
161 
162 theorem clippedExplorationRate_le_half (K T : Real) :
163     clippedExplorationRate K T <= 1 / 2 := by
164   exact min_le_left _ _
165 
166 theorem clippedExplorationRate_eq_tuned
167     (K T : Real) (h : tunedExplorationRate K T <= 1 / 2) :
168     clippedExplorationRate K T = tunedExplorationRate K T := by
169   exact min_eq_right h

171 /-- Generated predictable EXP3 kernel using the clipped all-horizon rates. -/
172 noncomputable def clippedPredictableTrajectoryKernel
173     {Env : Type u} {Action : Type v}
174     [MeasurableSpace Env]
175     [MeasurableSpace Action] [MeasurableSingletonClass Action]
176     [Nonempty Action] [DecidableEq Action]
177     (arms : Finset Action) (harms : arms.Nonempty)
178     (loss : PredictableLossVector Env Action) (horizon : Nat) :
179     Kernel Env (Nat -> Action × Real) :=
180   sampledImportanceWeightedTrajectoryKernel arms harms
181     (clippedLearningRate (arms.card : Real) (horizon : Real))
182     (clippedExplorationRate (arms.card : Real) (horizon : Real))
183     (clippedExplorationRate_nonneg _ _)
184     ((clippedExplorationRate_le_half _ _).trans (by norm_num))
185     loss.environment

187 /-- Uniform-horizon expected regret for the scalar loss actually generated by
188 predictable EXP3.  The minimum records the trivial short-horizon budget and the
189 large-horizon square-root theorem without imposing a regime assumption. -/
190 theorem sampledPredictable_clippedRealizedExpectedRegret_le_min
191     {Env : Type u} {Action : Type v}
192     [MeasurableSpace Env] [StandardBorelSpace Env]
193     [MeasurableSpace Action] [MeasurableSingletonClass Action]
194     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
195     (prior : Measure Env) [IsProbabilityMeasure prior]
196     (arms : Finset Action) (harms : arms.Nonempty)
197     (hcard_two : 2 <= arms.card)
198     (loss : PredictableLossVector Env Action) (horizon : Nat)
199     (comparator : Action) (hcomparator : comparator ∈ arms) :
200     let K := (arms.card : Real)
201     let T := (horizon : Real)
202     let mu := prior ⊗ₘ clippedPredictableTrajectoryKernel
203       arms harms loss horizon
204     integral mu (fun sample =>
205       (Finset.range horizon).sum (fun t =>
206           sampledTrajectoryRealizedLossAt t sample) -
207         (Finset.range horizon).sum (fun t =>
208           predictableLossAt loss t sample comparator)) <=
209       min T (4 * Real.sqrt (K * T * Real.log K)) := by
210   dsimp only
211   let K := (arms.card : Real)
212   let T := (horizon : Real)
213   let gamma := clippedExplorationRate K T
214   let eta := clippedLearningRate K T
215   let mu := prior ⊗ₘ clippedPredictableTrajectoryKernel arms harms loss horizon
216   have hgamma_nonneg : 0 <= gamma := clippedExplorationRate_nonneg K T
217   have hgamma_le_one : gamma <= 1 :=
218     (clippedExplorationRate_le_half K T).trans (by norm_num)
219   have htrivial := sampledPredictable_realizedExpectedRegret_le_horizon
220     prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon comparator
221   dsimp only at htrivial
222   have htrivial' : integral mu (fun sample =>
223       (Finset.range horizon).sum (fun t =>
224           sampledTrajectoryRealizedLossAt t sample) -
225         (Finset.range horizon).sum (fun t =>
226           predictableLossAt loss t sample comparator)) <= T := by
227     simpa [mu, eta, gamma, clippedPredictableTrajectoryKernel] using htrivial
228   have hK_one : 1 < K := by
229     dsimp [K]
230     exact_mod_cast hcard_two
231   have hKlog_pos : 0 < K * Real.log K :=
232     mul_pos (by linarith) (Real.log_pos hK_one)
233   have hT_nonneg : 0 <= T := by
234     dsimp [T]
235     exact_mod_cast (Nat.zero_le horizon)
236   by_cases hscale : 4 * K * Real.log K <= T
237   · have hT_pos : 0 < T := by nlinarith
238     have hhorizon_pos : 0 < horizon := by
239       have hT_pos' : 0 < (horizon : Real) := by simpa [T] using hT_pos
240       exact_mod_cast hT_pos'
241     have hgamma_tuned : tunedExplorationRate K T <= 1 / 2 :=
242       tunedExplorationRate_le_half K T hT_pos hscale
243     have htuned :=
244       sampledPredictable_realizedExpectedRegret_le_four_mul_sqrt
245         prior arms harms hcard_two loss horizon hhorizon_pos
246           (by simpa [K, T] using hscale) comparator hcomparator
247     dsimp only at htuned
248     have htuned' : integral mu (fun sample =>
249         (Finset.range horizon).sum (fun t =>
250             sampledTrajectoryRealizedLossAt t sample) -
251           (Finset.range horizon).sum (fun t =>
252             predictableLossAt loss t sample comparator)) <=
253         4 * Real.sqrt (K * T * Real.log K) := by
254       simpa [mu, K, T, clippedPredictableTrajectoryKernel,
255         clippedLearningRate, clippedExplorationRate_eq_tuned K T hgamma_tuned,
256         tunedPredictableTrajectoryKernel, tunedLearningRate] using htuned
257     exact le_min htrivial' htuned'
258   · have hsmall : T < 4 * K * Real.log K := lt_of_not_ge hscale
259     have hradicand : 0 <= K * T * Real.log K := by
260       nlinarith [mul_nonneg hT_nonneg hKlog_pos.le]
261     have hsquare := Real.sq_sqrt hradicand
262     have hsqrt_nonneg := Real.sqrt_nonneg (K * T * Real.log K)
263     have hmul := mul_le_mul_of_nonneg_left hsmall.le hT_nonneg
264     have hT_le_sqrt : T <= 4 * Real.sqrt (K * T * Real.log K) := by
265       nlinarith
266     rw [min_eq_left hT_le_sqrt]
267     exact htrivial'
