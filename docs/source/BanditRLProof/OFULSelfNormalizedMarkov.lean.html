19 /-- The common conditional sub-Gaussian variance proxy `R²`. -/
20 def constantSquaredVarianceProxy (R : Real) : Nat -> NNReal :=
21   fun _ => ⟨R ^ 2, sq_nonneg R⟩

23 /-- The unweighted finite-horizon feature Gram `sum_{i<n} x_i x_i^T`. -/
24 noncomputable def finiteHorizonFeatureGram
25     (feature : Nat -> Omega -> Feature -> Real)
26     (n : Nat) (omega : Omega) : Matrix Feature Feature Real :=
27   prefixFeatureGram (fun i => feature i omega) n

29 /--
30 At common proxy `R²`, the variance Gram is `R²` times the unweighted feature
31 Gram.
32 -/
33 theorem finiteHorizonVarianceGram_constantSquared_eq_smul_featureGram
34     [Fintype Feature]
35     (R : Real) (hR : 0 < R)
36     (feature : Nat -> Omega -> Feature -> Real)
37     (n : Nat) (omega : Omega) :
38     finiteHorizonVarianceGram
39         feature (constantSquaredVarianceProxy R) n omega =
40       R ^ 2 • finiteHorizonFeatureGram feature n omega := by
41   ext j k
42   simp [finiteHorizonVarianceGram, prefixFeatureGram,
43     varianceWeightedFeature, constantSquaredVarianceProxy,
44     finiteHorizonFeatureGram, Real.sqrt_sq_eq_abs, abs_of_pos hR]
45   rw [Finset.mul_sum]
46   apply Finset.sum_congr rfl
47   intro i hi
48   ring

63 /-- Positive scalar-square rescaling preserves positive definiteness. -/
64 theorem sq_smul_posDef
65     [Fintype Feature] [DecidableEq Feature]
66     (R : Real) (hR : 0 < R)
67     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef) :
68     (R ^ 2 • V0).PosDef := by
69   refine Matrix.PosDef.of_dotProduct_mulVec_pos
70     (hV0.isHermitian.smul (isSelfAdjoint_iff.mpr (by simp))) ?_
71   intro x hx
72   have hquad := hV0.dotProduct_mulVec_pos hx
73   simpa [Matrix.smul_mulVec, dotProduct_smul] using
74     mul_pos (sq_pos_of_pos hR) hquad

76 /-- A common nonzero scalar factor cancels from a determinant ratio. -/
77 theorem det_sq_smul_div_det_sq_smul
78     [Fintype Feature] [DecidableEq Feature]
79     (R : Real) (hR : 0 < R)
80     (A B : Matrix Feature Feature Real) :
81     Matrix.det (R ^ 2 • A) / Matrix.det (R ^ 2 • B) =
82       Matrix.det A / Matrix.det B := by
83   rw [Matrix.det_smul, Matrix.det_smul]
84   have hR2 : R ^ 2 ≠ 0 := pow_ne_zero 2 hR.ne'
85   field_simp [pow_ne_zero (Fintype.card Feature) hR2]

87 /--
88 The inverse quadratic form of a matrix scaled by `R²` is the original
89 quadratic form divided by `R²`.
90 -/
91 theorem dotProduct_sq_smul_inv_mulVec
92     [Fintype Feature] [DecidableEq Feature]
93     (R : Real) (hR : 0 < R)
94     (A : Matrix Feature Feature Real) (hA : A.PosDef)
95     (score : EuclideanSpace Real Feature) :
96     score ⬝ᵥ (R ^ 2 • A)⁻¹.mulVec score =
97       (score ⬝ᵥ A⁻¹.mulVec score) / R ^ 2 := by
98   letI : Invertible (R ^ 2) :=
99     invertibleOfNonzero (pow_ne_zero 2 hR.ne')
100   rw [Matrix.inv_smul A (R ^ 2)
101     (A.isUnit_iff_isUnit_det.mp hA.isUnit)]
102   simp [Matrix.smul_mulVec, dotProduct_smul, div_eq_mul_inv]
103   ring

105 /--
106 Markov's inequality for the evaluated finite-horizon Gaussian-mixture
107 surface, with an arbitrary nonzero finite `ENNReal` confidence level.
108 -/
109 theorem measure_inv_le_finiteHorizonDetRatioInvGramExponential_le
110     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
111     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
112     (mu : Measure Omega) [IsProbabilityMeasure mu]
113     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
114     (F : Filtration Nat mOmega)
115     (feature : Nat -> Omega -> Feature -> Real)
116     (noise : Nat -> Omega -> Real)
117     (varianceProxy : Nat -> NNReal)
118     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
119     (hfeature : forall i j,
120       StronglyMeasurable[F i] (fun omega => feature i omega j))
121     (hnoise : StronglyAdapted F (fun t omega =>
122       match t with
123       | 0 => 0
124       | i + 1 => noise i omega))
125     (hprojectionBound_nonneg : forall theta i,
126       0 <= projectionBound theta i)
127     (hprojectionBound : forall theta i omega,
128       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
129         projectionBound theta i)
130     (n : Nat)
131     (hsubGaussian : forall i, i < n ->
132       HasCondSubgaussianMGF
133         (F i) (F.le i) (noise i) (varianceProxy i) mu)
134     (delta : ENNReal) (hdelta_zero : delta ≠ 0)
135     (hdelta_top : delta ≠ ∞) :
136     mu {omega |
137         delta⁻¹ <=
138           finiteHorizonDetRatioInvGramExponential
139             V0 feature noise varianceProxy n omega} <= delta := by
140   have hfeatureMeas : forall i j,
141       Measurable (fun omega => feature i omega j) := by
142     intro i j
143     exact (hfeature i j).measurable.mono (F.le i) le_rfl
144   have hnoiseMeas : forall i, Measurable (noise i) := by
145     intro i
146     have hi : StronglyMeasurable[F (i + 1)] (noise i) := by
147       simpa using hnoise (i + 1)
148     exact hi.measurable.mono (F.le (i + 1)) le_rfl
149   have hmeas :
150       Measurable
151         (finiteHorizonDetRatioInvGramExponential
152           V0 feature noise varianceProxy n) :=
153     measurable_finiteHorizonDetRatioInvGramExponential
154       V0 hV0 feature noise varianceProxy n hfeatureMeas hnoiseMeas
155   have hmoment :
156       ∫⁻ omega,
157           finiteHorizonDetRatioInvGramExponential
158             V0 feature noise varianceProxy n omega
159         ∂mu <= 1 :=
160     lintegral_finiteHorizon_detRatio_invGramExponential_le_one
161       mu V0 hV0 F feature noise varianceProxy projectionBound hfeature hnoise
162       hprojectionBound_nonneg hprojectionBound n hsubGaussian
163   have hmarkov :=
164     meas_ge_le_lintegral_div (μ := mu) hmeas.aemeasurable
165       (ENNReal.inv_ne_zero.mpr hdelta_top)
166       (ENNReal.inv_ne_top.mpr hdelta_zero)
167   calc
168     mu {omega |
169         delta⁻¹ <=
170           finiteHorizonDetRatioInvGramExponential
171             V0 feature noise varianceProxy n omega} <=
172         (∫⁻ omega,
173             finiteHorizonDetRatioInvGramExponential
174               V0 feature noise varianceProxy n omega
175           ∂mu) / delta⁻¹ := hmarkov
176     _ <= 1 / delta⁻¹ := by gcongr
177     _ = delta := by simp

221 /--
222 The inverse-Gram/log-determinant bad-event inequality implies the evaluated
223 Gaussian-mixture Markov threshold, pointwise.
224 -/
225 theorem invOfReal_le_gaussianDetRatioExponential_of_invGramQuadratic_gt
226     [Fintype Feature] [DecidableEq Feature]
227     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef)
228     (hG : G.PosSemidef) (score : EuclideanSpace Real Feature)
229     (delta : Real) (hdelta : 0 < delta)
230     (hbad :
231       score ⬝ᵥ (V0 + G)⁻¹.mulVec score >
232         2 *
233           Real.log
234             (Real.sqrt
235                 (Matrix.det (V0 + G) / Matrix.det V0) / delta)) :
236     (ENNReal.ofReal delta)⁻¹ <=
237       ENNReal.ofReal
238         (Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
239           Real.exp
240             (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2)) := by
241   have hV0det : 0 < Matrix.det V0 := hV0.det_pos
242   have hsumdet : 0 < Matrix.det (V0 + G) :=
243     (hV0.add_posSemidef hG).det_pos
244   have hscale_pos :
245       0 < Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) :=
246     Real.sqrt_pos.2 (div_pos hV0det hsumdet)
247   have hratio_pos :
248       0 < Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) :=
249     Real.sqrt_pos.2 (div_pos hsumdet hV0det)
250   have hreciprocal :
251       Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
252           Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) =
253         1 := by
254     rw [Real.sqrt_div hV0det.le, Real.sqrt_div hsumdet.le]
255     field_simp [ne_of_gt (Real.sqrt_pos.2 hV0det),
256       ne_of_gt (Real.sqrt_pos.2 hsumdet)]
257   have hlogarg :
258       0 <
259         Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) / delta :=
260     div_pos hratio_pos hdelta
261   have hhalf :
262       Real.log
263           (Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) / delta) <
264         score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2 := by
265     linarith
266   have hexp :
267       Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) / delta <
268         Real.exp
269           (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2) := by
270     rw [← Real.exp_log hlogarg]
271     exact Real.exp_lt_exp.mpr hhalf
272   have hreal :
273       delta⁻¹ <
274         Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
275           Real.exp
276             (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2) := by
277     calc
278       delta⁻¹ =
279           Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
280             (Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0) /
281                 delta) := by
282                 symm
283                 rw [← mul_div_assoc, hreciprocal, one_div]
284       _ <
285           Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
286             Real.exp
287               (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2) :=
288         mul_lt_mul_of_pos_left hexp hscale_pos
289   rw [← ENNReal.ofReal_inv_of_pos hdelta]
290   exact ENNReal.ofReal_le_ofReal hreal.le

375 /--
376 Finite-horizon self-normalized tail bound with the conventional common
377 conditional sub-Gaussian scale `R`. The matrix in the reported norm is the
378 unweighted feature Gram, while the confidence radius carries the factor `R²`.
379 -/
380 theorem
381     measure_finiteHorizon_selfNormalizedQuadratic_gt_two_mul_sq_mul_log_detRatio_div_le
382     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
383     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]

388     (noise : Nat -> Omega -> Real)

394       match t with
