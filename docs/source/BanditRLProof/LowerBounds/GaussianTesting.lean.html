11 theorem log_gaussianPDFReal_div_same_variance
12     (m n x : ℝ) (v : ℝ≥0) (hv : v ≠ 0) :
13     Real.log (gaussianPDFReal m v x / gaussianPDFReal n v x) =
14       ((m - n) * x + (n ^ 2 - m ^ 2) / 2) / v := by
15   have hvpos : (0 : ℝ) < v := NNReal.coe_pos.mpr (pos_iff_ne_zero.mpr hv)
16   rw [gaussianPDFReal_def, gaussianPDFReal_def]
17   have hn : (Real.sqrt (2 * Real.pi * v))⁻¹ ≠ 0 := by positivity
18   rw [mul_div_mul_left _ _ hn, Real.log_div (Real.exp_ne_zero _) (Real.exp_ne_zero _),
19     Real.log_exp, Real.log_exp]

23     (m n : ℝ) (v : ℝ≥0) (hv : v ≠ 0) :
24     llr (gaussianReal m v) (gaussianReal n v) =ᵐ[gaussianReal m v]
25       fun x => ((m - n) * x + (n ^ 2 - m ^ 2) / 2) / v := by
26   have hP := gaussianReal_absolutelyContinuous m hv
27   have hQ := gaussianReal_absolutelyContinuous n hv
28   have hPQ := hP.trans (gaussianReal_absolutelyContinuous' n hv)

44 theorem klDiv_gaussianReal_same_variance
45     (m n : ℝ) (v : ℝ≥0) (hv : v ≠ 0) :
46     InformationTheory.klDiv (gaussianReal m v) (gaussianReal n v) =
47       ENNReal.ofReal ((m - n) ^ 2 / (2 * v)) := by

62 /-- Exact rational certification of the displayed Gaussian testing constant. -/
63 theorem three_fifths_le_exp_neg_half :
64     (3 / 5 : ℝ) ≤ Real.exp (-(1 / 2 : ℝ)) := by
65   have he : Real.exp 1 < (25 / 9 : ℝ) := Real.exp_one_lt_d9.trans (by norm_num)
66   have hi := one_div_lt_one_div_of_lt (Real.exp_pos 1) he
67   have hs : Real.exp (-(1 / 2 : ℝ)) ^ 2 = (Real.exp 1)⁻¹ := by
68     rw [pow_two, ← Real.exp_add]
69     convert Real.exp_neg 1 using 1 <;> norm_num
70   norm_num [div_eq_mul_inv] at hi
71   nlinarith [Real.exp_pos (-(1 / 2 : ℝ))]

75     (Δ : ℝ) (v : ℝ≥0) (hv : v ≠ 0) {A : Set ℝ} (hA : MeasurableSet A) :
76     (1 / 2 : ℝ) * Real.exp (-(Δ ^ 2 / (2 * v))) ≤
77       (gaussianReal 0 v).real A + (gaussianReal Δ v).real Aᶜ := by

84 /-- Under signal-to-noise ratio at most one, the sum of errors is at least 3/10. -/
85 theorem gaussian_testing_error_three_tenths
86     (Δ : ℝ) (v : ℝ≥0) (hv : v ≠ 0) (hsnr : Δ ^ 2 / (v : ℝ) ≤ 1)
87     {A : Set ℝ} (hA : MeasurableSet A) :
88     (3 / 10 : ℝ) ≤ (gaussianReal 0 v).real A + (gaussianReal Δ v).real Aᶜ := by
89   have hr : Δ ^ 2 / (2 * (v : ℝ)) ≤ 1 / 2 := by
90     have he : Δ ^ 2 / (2 * (v : ℝ)) = (Δ ^ 2 / (v : ℝ)) / 2 := by ring
91     rw [he]
92     linarith
93   have he := Real.exp_le_exp.mpr (neg_le_neg hr)
94   have ht := gaussian_testing_error_lower_bound Δ v hv hA
95   linarith [three_fifths_le_exp_neg_half]

97 /-- No measurable decision rule has both errors below 3/20 in the low-SNR regime. -/
98 theorem gaussian_testing_max_error_three_twentieths

100     {A : Set ℝ} (hA : MeasurableSet A) :
101     (3 / 20 : ℝ) ≤ max ((gaussianReal 0 v).real A) ((gaussianReal Δ v).real Aᶜ) := by

104   have hq := le_max_right ((gaussianReal 0 v).real A) ((gaussianReal Δ v).real Aᶜ)
