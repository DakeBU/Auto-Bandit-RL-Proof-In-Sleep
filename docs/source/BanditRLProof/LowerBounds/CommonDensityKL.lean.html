12 /-- The log likelihood ratio is the log ratio of two common-measure densities. -/
13 theorem llr_ae_eq_log_commonDensity
14     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
15     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
16     (hP : P ≪ μ) (hQ : Q ≪ μ) (hPQ : P ≪ Q) :
17     llr P Q =ᵐ[P] fun x => Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal) := by
18   filter_upwards [hPQ.ae_le (Measure.rnDeriv_eq_div hP hQ)] with x hx
19   simp [llr, hx, ENNReal.toReal_div]

21 /-- Integrability of the common-density logarithmic integrand is exactly that of LLR. -/
22 theorem integrable_commonDensity_iff
23     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
24     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
25     (hP : P ≪ μ) (hQ : Q ≪ μ) (hPQ : P ≪ Q) :
26     Integrable (fun x => (P.rnDeriv μ x).toReal *
27       Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal)) μ ↔
28       Integrable (llr P Q) P := by
29   rw [integrable_toReal_rnDeriv_mul_iff hP]
30   exact integrable_congr (llr_ae_eq_log_commonDensity P Q μ hP hQ hPQ).symm

32 /-- Eq. (14.6) in its supported, integrable probability-measure branch. -/
33 theorem relativeEntropy_commonDensity_of_integrable
34     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
35     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] [SigmaFinite μ]
36     (hP : P ≪ μ) (hQ : Q ≪ μ) (hPQ : P ≪ Q)
37     (hi : Integrable (fun x => (P.rnDeriv μ x).toReal *
38       Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal)) μ) :
39     relativeEntropy P Q = ENNReal.ofReal (∫ x, (P.rnDeriv μ x).toReal *
40       Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal) ∂μ) := by
41   rw [relativeEntropy_of_probability_absolutelyContinuous_of_integrable P Q hPQ
42     ((integrable_commonDensity_iff P Q μ hP hQ hPQ).1 hi)]
43   congr 1
44   exact (integral_congr_ae (llr_ae_eq_log_commonDensity P Q μ hP hQ hPQ)).trans
45     (integral_toReal_rnDeriv_mul hP).symm

48 /-- Common-density formula with singular and nonintegrable branches kept infinite. -/
49 theorem relativeEntropy_commonDensity_eq_if
50     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
51     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] [SigmaFinite μ]
52     (hP : P ≪ μ) (hQ : Q ≪ μ) :
53     relativeEntropy P Q = if P ≪ Q ∧
54       Integrable (fun x => (P.rnDeriv μ x).toReal *
55         Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal)) μ then
56       ENNReal.ofReal (∫ x, (P.rnDeriv μ x).toReal *
57         Real.log ((P.rnDeriv μ x).toReal / (Q.rnDeriv μ x).toReal) ∂μ)
58       else (⊤ : ENNReal) := by
59   classical
60   split_ifs with h
61   · exact relativeEntropy_commonDensity_of_integrable P Q μ hP hQ h.1 h.2
62   · apply not_ne_iff.mp
63     intro hne
64     have hr := relativeEntropy_ne_top_iff.1 hne
65     exact h ⟨hr.1, (integrable_commonDensity_iff P Q μ hP hQ hr.1).2 hr.2⟩

67 /-- Nonnegative common-density KL integral, valid even at infinite KL under AC. -/
68 theorem relativeEntropy_commonDensity_klFun
69     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
70     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
71     (hP : P ≪ μ) (hQ : Q ≪ μ) (hPQ : P ≪ Q) :
72     relativeEntropy P Q = ∫⁻ x, Q.rnDeriv μ x * ENNReal.ofReal
73       (InformationTheory.klFun ((P.rnDeriv μ x / Q.rnDeriv μ x).toReal)) ∂μ := by
74   calc
75     _ = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun (P.rnDeriv Q x).toReal) ∂Q :=
76       InformationTheory.klDiv_eq_lintegral_klFun_of_ac hPQ
77     _ = ∫⁻ x, ENNReal.ofReal
78         (InformationTheory.klFun ((P.rnDeriv μ x / Q.rnDeriv μ x).toReal)) ∂Q := by
79       apply lintegral_congr_ae
80       filter_upwards [Measure.rnDeriv_eq_div hP hQ] with x hx
81       rw [hx]
