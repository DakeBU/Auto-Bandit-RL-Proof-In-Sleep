5 theorem oneBitCode_optimal {α : Type*} [Fintype α] (p : α → ℝ)
6     (hp : ∀ i, 0 ≤ p i) (code : BinaryPrefixCode α)
7     (hlen : ∀ i, (code.encode i).length = 1) : IsOptimalPrefixCode p code := by
8   intro other
9   apply Finset.sum_le_sum
10   intro i _
11   rw [hlen]
12   have hn : 1 ≤ (other.encode i).length := by
13     have hz : (other.encode i).length ≠ 0 := by
14       intro he
15       exact other.nonempty i (List.length_eq_zero_iff.mp he)
16     omega
17   exact mul_le_mul_of_nonneg_left (by exact_mod_cast hn) (hp i)

45 /-- Huffman's recursive merge-two-least construction, with its correctness proof.
46 Real-weight choices are classical; the code itself is assembled recursively. -/
47 noncomputable def huffmanOptimalCode {α : Type*} [Fintype α]
48     (p : α → ℝ) (hp : ∀ i, 0 ≤ p i) : {code : BinaryPrefixCode α // IsOptimalPrefixCode p code} := by
49   classical
50   by_cases hsmall : Subsingleton α
51   · letI := hsmall
52     exact ⟨singletonPrefixCode α, oneBitCode_optimal p hp _ (fun _ => rfl)⟩
53   · letI : Nontrivial α := not_subsingleton_iff_nontrivial.mp hsmall
54     let a := Classical.choose (exists_two_least_weights p)
55     let b := Classical.choose (Classical.choose_spec (exists_two_least_weights p))
56     have hpair := Classical.choose_spec (Classical.choose_spec (exists_two_least_weights p))
57     have hab : a ≠ b := hpair.1
58     let β := HuffmanRemainder a b
59     let e := huffmanSplitEquiv a b hab
60     let mass : β ⊕ Bool → ℝ := Sum.elim (fun i => p i.val) (fun bit => if bit then p b else p a)
61     have hmass : mass = p ∘ e := by
62       funext i
63       cases i with
64       | inl i => rfl
65       | inr bit => cases bit <;> rfl
66     by_cases hempty : IsEmpty β
67     · letI := hempty
68       let root : BinaryPrefixCode (β ⊕ Bool) := emptyRemainderRoot
69       have ho : IsOptimalPrefixCode mass root := oneBitCode_optimal mass
70         (fun i => by rw [hmass]; exact hp (e i)) root (by
71           intro i
72           cases i with
73           | inl i => exact isEmptyElim i
74           | inr bit => rfl)
75       refine ⟨root.relabel e.symm, ?_⟩
76       have h := ho.relabel mass root e.symm
77       simpa [hmass, Function.comp_def] using h
78     · letI : Nonempty β := not_isEmpty_iff.mp hempty
79       let merged : Option β → ℝ := fun i => i.elim (p a + p b) (fun j => p j.val)
80       have hm : ∀ i, 0 ≤ merged i := by
81         intro i
82         cases i with
83         | none => exact add_nonneg (hp a) (hp b)
84         | some i => exact hp i.val
85       let smaller := huffmanOptimalCode merged hm
86       have ho : IsOptimalPrefixCode mass smaller.val.expandSibling :=
87         smaller.property.expand_least_weights (fun i : β => p i.val) (p a) (p b)
88           (fun i => hp i.val) (hp a) (hpair.2.1 b)
89           (fun i => hpair.2.2 i.val i.property.1) smaller.val
90       refine ⟨smaller.val.expandSibling.relabel e.symm, ?_⟩
91       have h := ho.relabel mass smaller.val.expandSibling e.symm
92       simpa [hmass, Function.comp_def] using h
93 termination_by Fintype.card α
94 decreasing_by
95   classical
96   exact huffman_merged_card_lt a b hab

98 /-- The prefix code produced by recursive Huffman merging. -/
99 noncomputable def huffmanCode {α : Type*} [Fintype α]
100     (p : α → ℝ) (hp : ∀ i, 0 ≤ p i) : BinaryPrefixCode α :=
101   (huffmanOptimalCode p hp).val
