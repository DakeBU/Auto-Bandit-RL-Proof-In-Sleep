4 /-!
5 # Same-policy finite-history KL for stochastic bandits
6 
7 This module connects the conditional-kernel KL integral to the repository's
8 kernel-valued `HistoryAlgorithm` and canonical Ionescu--Tulcea trajectory.
9 The observable history includes every sampled action and reward, so the same
10 possibly randomized nonanticipating policy is shared by both environments.
11 -/

38         rfl }

42 /-- A finite history and one next pair are measurably equivalent to the successor history. -/
43 def pairHistorySuccMeasurableEquiv
44     (Action : Type u) (Reward : Type v)
45     [MeasurableSpace Action] [MeasurableSpace Reward] (n : Nat) :
46     (History.FinitePairHistory Action Reward n × (Action × Reward)) ≃ᵐ
47       History.FinitePairHistory Action Reward (n + 1) :=
48   (MeasurableEquiv.prodCongr
49       (MeasurableEquiv.refl

52         (X := fun _ : Nat => Action × Reward) n)).trans
53     (MeasurableEquiv.IicProdIoc
54       (X := fun _ : Nat => Action × Reward) (Nat.le_succ n))

137 /-- The inclusive time-zero history is the initial action/reward law in singleton form. -/
138 theorem canonicalBanditHistoryMeasure_zero
139     {K : Nat} {Reward : Type v}
140     [MeasurableSpace Reward]
141     (algorithm : Thompson.HistoryAlgorithm (Fin K) Reward)
142     (armLaw : Kernel (Fin K) Reward) [IsMarkovKernel armLaw] :
143     canonicalBanditHistoryMeasure algorithm armLaw 0 =
144       (algorithm.initialAction ⊗ₘ armLaw).map
145         (pairHistoryZeroMeasurableEquiv (Fin K) Reward) := by
146   let trajectoryMeasure :=
147     Thompson.canonicalHistoryTrajectoryMeasure algorithm
148       (stationaryBanditHistoryEnvironment armLaw)
149   have hprefix :
150       (Preorder.frestrictLe 0 :
151           ((n : Nat) → Fin K × Reward) →
152             History.FinitePairHistory (Fin K) Reward 0) =
153         fun trajectory =>
154           pairHistoryZeroMeasurableEquiv (Fin K) Reward (trajectory 0) := by
155     funext trajectory index
156     have hindex : index = ⟨0, Finset.mem_Iic.mpr le_rfl⟩ :=
157       Subsingleton.elim _ _
158     subst index
159     rfl
160   have hinitial :
161       trajectoryMeasure.map (fun trajectory => trajectory 0) =
162         algorithm.initialAction ⊗ₘ armLaw := by
163     simpa [trajectoryMeasure, Thompson.canonicalHistoryTrajectoryAction,
164       Thompson.canonicalHistoryTrajectoryReward] using
165       (Thompson.canonicalHistoryTrajectory_initialPair_map_eq algorithm
166         (stationaryBanditHistoryEnvironment armLaw))
167   change trajectoryMeasure.map (Preorder.frestrictLe 0) = _
168   rw [hprefix]
169   change trajectoryMeasure.map
170       ((pairHistoryZeroMeasurableEquiv (Fin K) Reward) ∘
171         fun trajectory => trajectory 0) = _
172   rw [← Measure.map_map
173     (pairHistoryZeroMeasurableEquiv (Fin K) Reward).measurable
174     (measurable_pi_apply 0), hinitial]

176 /--
177 The inclusive successor history is obtained from the prefix history and the
178 canonical same-policy action/reward step, then re-encoded by a measurable
179 equivalence.
180 -/
181 theorem canonicalBanditHistoryMeasure_succ
182     {K : Nat} {Reward : Type v}
183     [MeasurableSpace Reward]
184     (algorithm : Thompson.HistoryAlgorithm (Fin K) Reward)
185     (armLaw : Kernel (Fin K) Reward) [IsMarkovKernel armLaw]
186     (n : Nat) :
187     canonicalBanditHistoryMeasure algorithm armLaw (n + 1) =
188       (canonicalBanditHistoryMeasure algorithm armLaw n ⊗ₘ
189           Thompson.historyStepKernel algorithm
190             (stationaryBanditHistoryEnvironment armLaw) n).map
191         (pairHistorySuccMeasurableEquiv (Fin K) Reward n) := by
192   let trajectoryMeasure :=
193     Thompson.canonicalHistoryTrajectoryMeasure algorithm
194       (stationaryBanditHistoryEnvironment armLaw)
195   let stepKernel := fun m =>
196     Thompson.historyStepKernel algorithm
197       (stationaryBanditHistoryEnvironment armLaw) m
198   have hjoint :
199       trajectoryMeasure.map (Preorder.frestrictLe n) ⊗ₘ stepKernel n =
200         trajectoryMeasure.map (fun trajectory =>
201           (Preorder.frestrictLe n trajectory, trajectory (n + 1))) := by
202     simpa [trajectoryMeasure, stepKernel,
203       Thompson.canonicalHistoryTrajectoryMeasure] using
204       (ProbabilityTheory.Kernel.map_frestrictLe_trajMeasure_compProd_eq_map_trajMeasure
205           (X := fun _ : Nat => Fin K × Reward)
206           (μ₀ := algorithm.initialAction ⊗ₘ armLaw)
207           (κ := stepKernel) (a := n))
208   have hencode :
209       (Preorder.frestrictLe (n + 1) :
210           ((m : Nat) → Fin K × Reward) →
211             History.FinitePairHistory (Fin K) Reward (n + 1)) =
212         fun trajectory =>
213           pairHistorySuccMeasurableEquiv (Fin K) Reward n
214             (Preorder.frestrictLe n trajectory, trajectory (n + 1)) := by
215     funext trajectory
216     rw [pairHistorySuccMeasurableEquiv_apply]
217     funext index
218     by_cases hindex : index.1 ≤ n
219     · simp [History.extendPairHistorySucc, hindex]
220     · have hle : index.1 ≤ n + 1 := Finset.mem_Iic.mp index.2
221       have hsucc : index.1 = n + 1 := by omega
222       simp [History.extendPairHistorySucc, hsucc]
223   change trajectoryMeasure.map (Preorder.frestrictLe (n + 1)) = _
224   rw [hencode]
225   change trajectoryMeasure.map
226       ((pairHistorySuccMeasurableEquiv (Fin K) Reward n) ∘
227         fun trajectory =>
228           (Preorder.frestrictLe n trajectory, trajectory (n + 1))) = _
229   rw [← Measure.map_map
230     (pairHistorySuccMeasurableEquiv (Fin K) Reward n).measurable
231     ((Preorder.measurable_frestrictLe n).prodMk
232       (measurable_pi_apply (n + 1)))]
233   rw [← hjoint]
234   rfl

258 /--
259 One successor round adds the first-environment policy average of the selected
260 arm KL divergence.  This is the exact adaptive-history chain-rule step; the
261 possibly randomized policy is shared and contributes no additional KL term.
262 -/
263 theorem klDiv_canonicalBanditHistoryMeasure_succ
264     {K : Nat} {Reward : Type v}
265     [MeasurableSpace Reward] [MeasurableSpace.CountablyGenerated Reward]
266     (algorithm : Thompson.HistoryAlgorithm (Fin K) Reward)
267     (armLaw referenceArmLaw : Kernel (Fin K) Reward)
268     [IsMarkovKernel armLaw] [IsMarkovKernel referenceArmLaw]
269     (h_ac : ∀ arm, armLaw arm ≪ referenceArmLaw arm)
270     (n : Nat) :
271     InformationTheory.klDiv
272         (canonicalBanditHistoryMeasure algorithm armLaw (n + 1))
273         (canonicalBanditHistoryMeasure algorithm referenceArmLaw (n + 1)) =
274       InformationTheory.klDiv
275           (canonicalBanditHistoryMeasure algorithm armLaw n)
276           (canonicalBanditHistoryMeasure algorithm referenceArmLaw n) +
277         ∫⁻ history, ∫⁻ arm,
278           InformationTheory.klDiv
279             (armLaw arm) (referenceArmLaw arm)
280             ∂algorithm.policy n history
281           ∂canonicalBanditHistoryMeasure algorithm armLaw n := by
282   rw [canonicalBanditHistoryMeasure_succ algorithm armLaw n,
283     canonicalBanditHistoryMeasure_succ algorithm referenceArmLaw n]
284   rw [klDiv_map_measurableEquiv]
285   rw [InformationTheory.klDiv_compProd_eq_add]
286   congr 1
287   simpa [Thompson.historyStepKernel,
288     stationaryBanditHistoryEnvironment] using
289     (klDiv_historyStep_samePolicy_eq_iterated_lintegral_armKL
290       (canonicalBanditHistoryMeasure algorithm armLaw n)
291       (algorithm.policy n) armLaw referenceArmLaw h_ac)

400 /-- Realized pull count encoded directly on an inclusive finite history. -/
401 noncomputable def finiteHistoryPullCountENNReal
402     {K : Nat} {Reward : Type v} :
403     (n : Nat) → History.FinitePairHistory (Fin K) Reward n → Fin K → ENNReal
404   | 0, history, arm =>
405       if (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 = arm then 1 else 0
406   | n + 1, history, arm =>
407       finiteHistoryPullCountENNReal n (Thompson.pairHistoryPrefix history) arm +
408         if (Thompson.pairHistoryLast history).1 = arm then 1 else 0
