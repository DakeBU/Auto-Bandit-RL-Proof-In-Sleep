6 theorem uniformPowerTwo_entropy {α : Type*} [Fintype α] (n : ℕ)
7     (hcard : Fintype.card α = 2 ^ n) :
8     discreteEntropyBaseTwo Finset.univ (fun _ : α => (1 / (2 : ℝ) ^ n)) = n := by

12 theorem fixedLength_uniformPowerTwo_optimal {α : Type*} [Fintype α]
13     (n : ℕ) (hcard : Fintype.card α = 2 ^ n) (code : BinaryPrefixCode α)
14     (hlen : ∀ a, (code.encode a).length = n) :
15     IsOptimalPrefixCode (fun _ : α => 1 / (2 : ℝ) ^ n) code := by
16   classical
17   have hs : ∑ _ : α, (1 / (2 : ℝ) ^ n) = 1 := by simp [hcard]
18   intro other
19   rw [expectedCodeLength_fixedLength _ hs code n hlen]
20   have h := discreteEntropyBaseTwo_le_expectedCodeLength
21     (fun _ : α => 1 / (2 : ℝ) ^ n) (fun _ => by positivity) hs other
22   rwa [uniformPowerTwo_entropy n hcard] at h

24 def ternaryPrefixWord (a : Fin 3) : List Bool :=
25   if a.val = 0 then [false] else if a.val = 1 then [true, false] else [true, true]
26 
27 def ternaryPrefixCode : BinaryPrefixCode (Fin 3) where
28   encode := ternaryPrefixWord
29   injective := by
30     intro a b h
31     fin_cases a <;> fin_cases b <;> simp_all [ternaryPrefixWord]
32   nonempty := by intro a; fin_cases a <;> simp [ternaryPrefixWord]
33   prefixFree := by
34     intro a b h
35     fin_cases a <;> fin_cases b <;> simp_all [ternaryPrefixWord]

41 /-- Uniform masses do not make a constant-length code optimal for every alphabet size. -/
42 theorem uniform_three_fixedLength_not_optimal (code : BinaryPrefixCode (Fin 3))
43     (hlen : ∀ a, (code.encode a).length = 2) :
44     ¬ IsOptimalPrefixCode (fun _ : Fin 3 => (1 / 3 : ℝ)) code := by
