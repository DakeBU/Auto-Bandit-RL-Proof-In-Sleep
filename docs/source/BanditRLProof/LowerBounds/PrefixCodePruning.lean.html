5 /-- A deepest leaf with no sibling has a parent incomparable with every other word. -/
6 theorem deepest_parent_incomparable
7     {α : Type*} (code : BinaryPrefixCode α) (a : α) (w : List Bool) (b : Bool)
8     (ha : code.encode a = w ++ [b])
9     (hmax : ∀ i, (code.encode i).length ≤ (code.encode a).length)
10     (hmissing : ∀ i, code.encode i ≠ w ++ [!b]) :
11     ∀ i, i ≠ a → (¬ code.encode i <+: w) ∧ (¬ w <+: code.encode i) := by
12   intro i hia
13   have hleft : ¬ code.encode i <+: w := by
14     intro h
15     have hp : code.encode i <+: code.encode a := by
16       rw [ha]
17       exact h.trans (List.prefix_append _ _)
18     exact hia (code.prefixFree hp)
19   refine ⟨hleft, ?_⟩
20   rintro ⟨t, ht⟩
21   have hlen := hmax i
22   rw [ha, ← ht] at hlen
23   simp only [List.length_append, List.length_singleton] at hlen
24   cases t with
25   | nil =>
26     have he : w = code.encode i := by simpa using ht
27     exact hleft (by rw [he])
28   | cons c t =>
29     have ht0 : t = [] := List.length_eq_zero_iff.mp (by simp only [List.length_cons] at hlen; omega)
30     subst t
31     by_cases hcb : c = b
32     · have he : code.encode i = code.encode a := by rw [ha, ← ht, hcb]
33       exact hia (code.injective he)
34     · have hc : c = !b := by cases b <;> cases c <;> simp_all
35       exact hmissing i (by rw [← ht, hc])

37 /-- Replace one word by an incomparable nonempty word, preserving code validity. -/
38 def BinaryPrefixCode.replaceWord
39     {α : Type*} [DecidableEq α] (code : BinaryPrefixCode α) (a : α) (w : List Bool)
40     (hw : w ≠ [])
41     (hsep : ∀ i, i ≠ a → (¬ code.encode i <+: w) ∧ (¬ w <+: code.encode i)) :
42     BinaryPrefixCode α := by
43   let f : α → List Bool := fun i => if i = a then w else code.encode i
44   have hf : ∀ i j, f i <+: f j → i = j := by
45     intro i j h
46     by_cases hi : i = a
47     · by_cases hj : j = a
48       · exact hi.trans hj.symm
49       · exact ((hsep j hj).2 (by simpa [f, hi, hj] using h)).elim
50     · by_cases hj : j = a
51       · exact ((hsep i hi).1 (by simpa [f, hi, hj] using h)).elim
52       · exact code.prefixFree (by simpa [f, hi, hj] using h)
53   exact ⟨f, fun i j h => hf i j (by rw [h]),
54     fun i => by dsimp [f]; split_ifs; exact hw; exact code.nonempty i,
55     fun h => hf _ _ h⟩

57 /-- Legal deletion of the last bit of a deepest sibling-free leaf. -/
58 def BinaryPrefixCode.pruneDeepest
59     {α : Type*} [DecidableEq α] (code : BinaryPrefixCode α) (a : α) (w : List Bool) (b : Bool)
60     (hw : w ≠ []) (ha : code.encode a = w ++ [b])
61     (hmax : ∀ i, (code.encode i).length ≤ (code.encode a).length)
62     (hmissing : ∀ i, code.encode i ≠ w ++ [!b]) : BinaryPrefixCode α :=
63   code.replaceWord a w hw (deepest_parent_incomparable code a w b ha hmax hmissing)

65 theorem expectedCodeLength_replaceWord
66     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
67     (code : BinaryPrefixCode α) (a : α) (w : List Bool) (hw : w ≠ [])
68     (hsep : ∀ i, i ≠ a → (¬ code.encode i <+: w) ∧ (¬ w <+: code.encode i)) :
69     expectedCodeLength p (code.replaceWord a w hw hsep) = expectedCodeLength p code +
70       p a * (w.length - (code.encode a).length : ℝ) := by
71   have ht (i) : p i * ((code.replaceWord a w hw hsep).encode i).length =
72       p i * (code.encode i).length +
73         (if i = a then p a * (w.length - (code.encode a).length : ℝ) else 0) := by
74     by_cases hi : i = a
75     · subst i
76       simp [BinaryPrefixCode.replaceWord]
77       ring
78     · simp [BinaryPrefixCode.replaceWord, hi]
79   simp only [expectedCodeLength, ht, Finset.sum_add_distrib]
80   simp

82 theorem expectedCodeLength_pruneDeepest
83     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
84     (code : BinaryPrefixCode α) (a : α) (w : List Bool) (b : Bool)
85     (hw : w ≠ []) (ha : code.encode a = w ++ [b])
86     (hmax : ∀ i, (code.encode i).length ≤ (code.encode a).length)
87     (hmissing : ∀ i, code.encode i ≠ w ++ [!b]) :
88     expectedCodeLength p (code.pruneDeepest a w b hw ha hmax hmissing) =
89       expectedCodeLength p code - p a := by
90   rw [BinaryPrefixCode.pruneDeepest, expectedCodeLength_replaceWord, ha]
91   simp only [List.length_append, List.length_singleton, Nat.cast_add, Nat.cast_one]
92   ring

95 theorem expectedCodeLength_pruneDeepest_le
96     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
97     (code : BinaryPrefixCode α) (a : α) (w : List Bool) (b : Bool)
98     (hp : 0 ≤ p a) (hw : w ≠ []) (ha : code.encode a = w ++ [b])
99     (hmax : ∀ i, (code.encode i).length ≤ (code.encode a).length)
100     (hmissing : ∀ i, code.encode i ≠ w ++ [!b]) :

105 /-- Structural termination measure, independent of source probabilities. -/
106 def totalCodeLength {α : Type*} [Fintype α] (code : BinaryPrefixCode α) : ℕ :=
107   ∑ i, (code.encode i).length

109 theorem totalCodeLength_pruneDeepest
110     {α : Type*} [Fintype α] [DecidableEq α]
111     (code : BinaryPrefixCode α) (a : α) (w : List Bool) (b : Bool)
112     (hw : w ≠ []) (ha : code.encode a = w ++ [b])
113     (hmax : ∀ i, (code.encode i).length ≤ (code.encode a).length)
114     (hmissing : ∀ i, code.encode i ≠ w ++ [!b]) :
115     totalCodeLength (code.pruneDeepest a w b hw ha hmax hmissing) + 1 = totalCodeLength code := by
116   have h := expectedCodeLength_pruneDeepest (fun _ => (1 : ℝ)) code a w b hw ha hmax hmissing
117   simp only [expectedCodeLength, one_mul] at h
118   have he : (∑ i, (((code.pruneDeepest a w b hw ha hmax hmissing).encode i).length : ℝ)) + 1 =
119       ∑ i, ((code.encode i).length : ℝ) := by linarith
120   exact_mod_cast he

122 /-- Choose a structurally minimal no-worse competitor without assuming cost-minimizer existence. -/
123 theorem exists_minimal_totalCodeLength_competitor
124     {α : Type*} [Fintype α] (p : α → ℝ) (original : BinaryPrefixCode α) :
125     ∃ code : BinaryPrefixCode α,
126       expectedCodeLength p code ≤ expectedCodeLength p original ∧
127       ∀ other : BinaryPrefixCode α,
128         expectedCodeLength p other ≤ expectedCodeLength p original →
129           totalCodeLength code ≤ totalCodeLength other := by
130   classical
131   let P : ℕ → Prop := fun n => ∃ code : BinaryPrefixCode α,
132     expectedCodeLength p code ≤ expectedCodeLength p original ∧ totalCodeLength code = n
133   have hex : ∃ n, P n := ⟨totalCodeLength original, original, le_rfl, rfl⟩
134   obtain ⟨code, hc, hn⟩ := Nat.find_spec hex
135   refine ⟨code, hc, ?_⟩
136   intro other ho
137   rw [hn]
138   exact Nat.find_min' hex ⟨other, ho, rfl⟩

140 /-- Normalize any competitor so that each deepest leaf has its sibling present. -/
141 theorem exists_competitor_with_deepest_siblings
142     {α : Type*} [Fintype α] [DecidableEq α] [Nontrivial α]
143     (p : α → ℝ) (hp : ∀ i, 0 ≤ p i) (original : BinaryPrefixCode α) :
144     ∃ code : BinaryPrefixCode α,
145       expectedCodeLength p code ≤ expectedCodeLength p original ∧
146       ∀ a w b, code.encode a = w ++ [b] →
147         (∀ i, (code.encode i).length ≤ (code.encode a).length) →
148           ∃ j, code.encode j = w ++ [!b] := by
149   classical
150   obtain ⟨code, hc, hmin⟩ := exists_minimal_totalCodeLength_competitor p original
151   refine ⟨code, hc, ?_⟩
152   intro a w b ha hmax
153   by_contra h
154   have hmissing : ∀ j, code.encode j ≠ w ++ [!b] := by simpa using h
155   have hw : w ≠ [] := by
156     intro he
157     obtain ⟨i, hi⟩ := exists_ne a
158     have hn := (deepest_parent_incomparable code a w b ha hmax hmissing i hi).2
159     apply hn
160     simp [he]
161   have hcost := expectedCodeLength_pruneDeepest_le p code a w b (hp a) hw ha hmax hmissing
162   have hbound := hmin (code.pruneDeepest a w b hw ha hmax hmissing) (hcost.trans hc)
163   have hdrop := totalCodeLength_pruneDeepest code a w b hw ha hmax hmissing
164   omega

166 /-- Every competitor has a no-worse code containing a deepest sibling pair. -/
167 theorem exists_no_worse_deepest_sibling_pair
168     {α : Type*} [Fintype α] [DecidableEq α] [Nontrivial α]
169     (p : α → ℝ) (hp : ∀ i, 0 ≤ p i) (original : BinaryPrefixCode α) :
170     ∃ code : BinaryPrefixCode α, expectedCodeLength p code ≤ expectedCodeLength p original ∧
171       ∃ a j w b, a ≠ j ∧ code.encode a = w ++ [b] ∧ code.encode j = w ++ [!b] ∧
172         ∀ i, (code.encode i).length ≤ (code.encode a).length := by
173   obtain ⟨code, hc, hf⟩ := exists_competitor_with_deepest_siblings p hp original
174   obtain ⟨a, _, hmax⟩ := (Finset.univ : Finset α).exists_max_image
175     (fun i => (code.encode i).length) Finset.univ_nonempty
176   have hsplit : ∀ v : List Bool, v ≠ [] → ∃ w b, v = w ++ [b] := by
177     intro v
178     induction v using List.reverseRecOn with
179     | nil => simp
180     | append_singleton w b _ => exact fun _ => ⟨w, b, rfl⟩
181   obtain ⟨w, b, ha⟩ := hsplit (code.encode a) (code.nonempty a)
182   have hm : ∀ i, (code.encode i).length ≤ (code.encode a).length :=
183     fun i => hmax i (Finset.mem_univ i)
184   obtain ⟨j, hj⟩ := hf a w b ha hm
185   have haj : a ≠ j := by
186     intro he
187     have hwords : w ++ [b] = w ++ [!b] := ha.symm.trans (he ▸ hj)
188     have hb := List.append_cancel_left hwords
189     cases b <;> simp at hb
190   exact ⟨code, hc, a, j, w, b, haj, ha, hj, hm⟩
