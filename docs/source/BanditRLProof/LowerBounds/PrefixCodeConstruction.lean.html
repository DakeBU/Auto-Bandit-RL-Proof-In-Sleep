6 /-- All words at depth n in the full binary tree. -/
7 def binaryWords (n : ℕ) : Finset (List Bool) :=
8   Finset.univ.image (fun f : Fin n → Bool => List.ofFn f)
9 
10 theorem card_binaryWords (n : ℕ) : (binaryWords n).card = 2 ^ n := by
11   rw [binaryWords, Finset.card_image_of_injective _ List.ofFn_injective]
12   simp

14 theorem mem_binaryWords_iff (w : List Bool) (n : ℕ) :
15     w ∈ binaryWords n ↔ w.length = n := by
16   constructor
17   · intro h
18     obtain ⟨f, _, rfl⟩ := Finset.mem_image.mp h
19     simp
20   · intro h
21     subst n
22     apply Finset.mem_image.mpr
23     exact ⟨w.get, Finset.mem_univ _, List.ofFn_get w⟩

25 /-- The descendants of a prefix after n additional bits. -/
26 def binaryExtensions (w : List Bool) (n : ℕ) : Finset (List Bool) :=
27   (binaryWords n).image (fun v => w ++ v)
28 
29 theorem card_binaryExtensions (w : List Bool) (n : ℕ) :
30     (binaryExtensions w n).card = 2 ^ n := by
31   rw [binaryExtensions, Finset.card_image_of_injective]
32   · exact card_binaryWords n
33   · intro a b h
34     exact List.append_cancel_left h
35 
36 theorem mem_binaryExtensions_iff (w v : List Bool) (n : ℕ) :
37     v ∈ binaryExtensions w n ↔ w <+: v ∧ v.length = w.length + n := by
38   constructor
39   · intro h
40     obtain ⟨u, hu, rfl⟩ := Finset.mem_image.mp h
41     exact ⟨List.prefix_append _ _, by simp [(mem_binaryWords_iff u n).1 hu]⟩
42   · rintro ⟨⟨u, rfl⟩, hlen⟩
43     apply Finset.mem_image.mpr
44     refine ⟨u, (mem_binaryWords_iff u n).2 ?_, rfl⟩
45     simpa using hlen

47 /-- Cylinders from incomparable prefixes are disjoint. -/
48 theorem binaryExtensions_disjoint_of_incomparable
49     (u v : List Bool) (m n : ℕ) (huv : ¬ u <+: v) (hvu : ¬ v <+: u) :
50     Disjoint (binaryExtensions u m) (binaryExtensions v n) := by
51   apply Finset.disjoint_left.mpr
52   intro w hu hv
53   have h1 := (mem_binaryExtensions_iff u w m).1 hu
54   have h2 := (mem_binaryExtensions_iff v w n).1 hv
55   exact (List.prefix_or_prefix_of_prefix h1.1 h2.1).elim huv hvu

57 /-- A free word exists whenever previous prefixes occupy fewer than all level nodes. -/
58 theorem exists_binaryWord_avoiding_prefixes
59     (S : Finset (List Bool)) (n : ℕ)
60     (hlen : ∀ w ∈ S, w.length ≤ n)
61     (hbudget : (∑ w ∈ S, 2 ^ (n - w.length)) < 2 ^ n) :
62     ∃ v : List Bool, v.length = n ∧ ∀ w ∈ S, ¬ w <+: v := by
63   let U := S.biUnion (fun w => binaryExtensions w (n - w.length))
64   have hcard : U.card < (binaryWords n).card := by
65     rw [card_binaryWords]
66     calc
67       U.card ≤ ∑ w ∈ S, (binaryExtensions w (n - w.length)).card :=
68         Finset.card_biUnion_le
69       _ = ∑ w ∈ S, 2 ^ (n - w.length) := by simp only [card_binaryExtensions]
70       _ < _ := hbudget
71   obtain ⟨v, hv, hnot⟩ := Finset.exists_mem_notMem_of_card_lt_card hcard
72   refine ⟨v, (mem_binaryWords_iff v n).1 hv, ?_⟩
73   intro w hw hpre
74   apply hnot
75   apply Finset.mem_biUnion.mpr
76   refine ⟨w, hw, (mem_binaryExtensions_iff w v (n - w.length)).2 ⟨hpre, ?_⟩⟩
77   rw [(mem_binaryWords_iff v n).1 hv]
78   have := hlen w hw
79   omega

81 theorem binary_level_mul_kraft_weight {k n : ℕ} (h : k ≤ n) :
82     (2 : ℝ) ^ n * (1 / 2 : ℝ) ^ k = 2 ^ (n - k) := by
83   have hn : n = (n - k) + k := by omega
84   nth_rw 1 [hn]
85   rw [pow_add, one_div_pow]
86   field_simp

88 /-- Real Kraft slack supplies the integer capacity needed to insert a word. -/
89 theorem exists_binaryWord_of_kraft_lt_one
90     (S : Finset (List Bool)) (n : ℕ)
91     (hlen : ∀ w ∈ S, w.length ≤ n)
92     (hk : (∑ w ∈ S, (1 / 2 : ℝ) ^ w.length) < 1) :
93     ∃ v : List Bool, v.length = n ∧ ∀ w ∈ S, ¬ w <+: v := by
94   apply exists_binaryWord_avoiding_prefixes S n hlen
95   have h := mul_lt_mul_of_pos_left hk (pow_pos (by norm_num : (0 : ℝ) < 2) n)
96   rw [Finset.mul_sum, mul_one] at h
97   have he : (∑ w ∈ S, (2 : ℝ) ^ n * (1 / 2 : ℝ) ^ w.length) =
98       ∑ w ∈ S, (2 : ℝ) ^ (n - w.length) := by
99     exact Finset.sum_congr rfl fun w hw => binary_level_mul_kraft_weight (hlen w hw)
100   rw [he] at h
101   exact_mod_cast h

103 /-- The greedy insertion preserves prefix freedom in both directions. -/
104 theorem exists_prefixFree_insert_of_kraft_lt_one
105     (S : Finset (List Bool)) (n : ℕ)
106     (hfree : ∀ a ∈ S, ∀ b ∈ S, a <+: b → a = b)
107     (hlen : ∀ w ∈ S, w.length ≤ n)
108     (hk : (∑ w ∈ S, (1 / 2 : ℝ) ^ w.length) < 1) :
109     ∃ v : List Bool, v.length = n ∧ v ∉ S ∧
110       ∀ a ∈ insert v S, ∀ b ∈ insert v S, a <+: b → a = b := by
111   obtain ⟨v, hvlen, hv⟩ := exists_binaryWord_of_kraft_lt_one S n hlen hk
112   have hvnot : v ∉ S := by
113     intro h
114     exact hv v h (by rfl)
115   have hreverse : ∀ w ∈ S, ¬ v <+: w := by
116     intro w hw hpre
117     obtain ⟨t, rfl⟩ := hpre
118     have ht : t.length = 0 := by
119       have h := hlen (v ++ t) hw
120       simp only [List.length_append, hvlen] at h
121       omega
122     have ht' : t = [] := List.length_eq_zero_iff.mp ht
123     simp only [ht', List.append_nil] at hw
124     exact hvnot hw
125   refine ⟨v, hvlen, hvnot, ?_⟩
126   intro a ha b hb hab
127   by_cases hav : a = v
128   · subst a
129     by_cases hbv : b = v
130     · exact hbv.symm
131     · exact (hreverse b ((Finset.mem_insert.mp hb).resolve_left hbv) hab).elim
132   · have haS := (Finset.mem_insert.mp ha).resolve_left hav
133     by_cases hbv : b = v
134     · subst b
135       exact (hv a haS hab).elim
136     · exact hfree a haS b ((Finset.mem_insert.mp hb).resolve_left hbv) hab

138 /-- Finite Kraft converse, retaining each prescribed length, including equality. -/
139 theorem exists_prefix_encoding_of_kraft_le_one
140     {α : Type*} [DecidableEq α] (s : Finset α) (l : α → ℕ)
141     (hk : (∑ i ∈ s, (1 / 2 : ℝ) ^ l i) ≤ 1) :
142     ∃ c : α → List Bool, (∀ i ∈ s, (c i).length = l i) ∧
143       (∀ i ∈ s, ∀ j ∈ s, c i <+: c j → i = j) := by
144   classical
145   induction s using Finset.strongInductionOn with
146   | _ s ih =>
147     by_cases hs : s.Nonempty
148     · obtain ⟨a, ha, hmax⟩ := s.exists_max_image l hs
149       let t := s.erase a
150       have ht : t ⊂ s := Finset.erase_ssubset ha
151       have hkt : (∑ i ∈ t, (1 / 2 : ℝ) ^ l i) < 1 := by
152         have he := Finset.sum_erase_add s (fun i => (1 / 2 : ℝ) ^ l i) ha
153         have hp : 0 < (1 / 2 : ℝ) ^ l a := by positivity
154         dsimp [t]
155         linarith
156       obtain ⟨c, hcl, hcf⟩ := ih t ht hkt.le
157       let S := t.image c
158       have hci : Set.InjOn c t := by
159         intro i hi j hj hij
160         exact hcf i hi j hj (by rw [hij])
161       have hSl : ∀ w ∈ S, w.length ≤ l a := by
162         intro w hw
163         obtain ⟨i, hi, rfl⟩ := Finset.mem_image.mp hw
164         rw [hcl i hi]
165         exact hmax i (Finset.mem_of_mem_erase hi)
166       have hSk : (∑ w ∈ S, (1 / 2 : ℝ) ^ w.length) < 1 := by
167         dsimp [S]
168         rw [Finset.sum_image hci]
169         have he : (∑ i ∈ t, (1 / 2 : ℝ) ^ (c i).length) =
170             ∑ i ∈ t, (1 / 2 : ℝ) ^ l i :=
171           Finset.sum_congr rfl fun i hi => by rw [hcl i hi]
172         rw [he]
173         exact hkt
174       have hSf : ∀ u ∈ S, ∀ v ∈ S, u <+: v → u = v := by
175         intro u hu v hv huv
176         obtain ⟨i, hi, rfl⟩ := Finset.mem_image.mp hu
177         obtain ⟨j, hj, rfl⟩ := Finset.mem_image.mp hv
178         rw [hcf i hi j hj huv]
179       obtain ⟨v, hvl, hvnot, hvfree⟩ :=
180         exists_prefixFree_insert_of_kraft_lt_one S (l a) hSf hSl hSk
181       let d : α → List Bool := fun i => if i = a then v else c i
182       have hdmem : ∀ i ∈ s, d i ∈ insert v S := by
183         intro i hi
184         by_cases hia : i = a
185         · simp [d, hia]
186         · simp only [d, if_neg hia]
187           exact Finset.mem_insert_of_mem (Finset.mem_image.mpr
188             ⟨i, Finset.mem_erase.mpr ⟨hia, hi⟩, rfl⟩)
189       have hdi : Set.InjOn d s := by
190         intro i hi j hj hd
191         by_cases hia : i = a
192         · by_cases hja : j = a
193           · exact hia.trans hja.symm
194           · have hcj : c j ∈ S := Finset.mem_image.mpr
195               ⟨j, Finset.mem_erase.mpr ⟨hja, hj⟩, rfl⟩
196             have he : v = c j := by simpa [d, hia, hja] using hd
197             exact (hvnot (he ▸ hcj)).elim
198         · by_cases hja : j = a
199           · have hci' : c i ∈ S := Finset.mem_image.mpr
200               ⟨i, Finset.mem_erase.mpr ⟨hia, hi⟩, rfl⟩
201             have he : c i = v := by simpa [d, hia, hja] using hd
202             exact (hvnot (he ▸ hci')).elim
203           · apply hci (Finset.mem_erase.mpr ⟨hia, hi⟩) (Finset.mem_erase.mpr ⟨hja, hj⟩)
204             simpa [d, hia, hja] using hd
205       refine ⟨d, ?_, ?_⟩
206       · intro i hi
207         by_cases hia : i = a
208         · simpa [d, hia] using hvl
209         · simpa [d, hia] using hcl i (Finset.mem_erase.mpr ⟨hia, hi⟩)
210       · intro i hi j hj hpre
211         exact hdi hi hj (hvfree _ (hdmem i hi) _ (hdmem j hj) hpre)
212     · have he : s = ∅ := Finset.not_nonempty_iff_eq_empty.mp hs
213       subst s
214       exact ⟨fun _ => [], by simp⟩

224 /-- Non-strict Kraft converse packaged as an actual binary prefix code. -/
225 theorem exists_binaryPrefixCode_of_kraft_le_one
226     {α : Type*} [Fintype α] [DecidableEq α] (l : α → ℕ)
227     (hl : ∀ i, 0 < l i) (hk : (∑ i, (1 / 2 : ℝ) ^ l i) ≤ 1) :
228     ∃ code : BinaryPrefixCode α, ∀ i, (code.encode i).length = l i := by
229   obtain ⟨c, hlen, hfree⟩ := exists_prefix_encoding_of_kraft_le_one Finset.univ l hk
230   have hci : Function.Injective c := by
231     intro i j hij
232     exact hfree i (Finset.mem_univ _) j (Finset.mem_univ _) (by rw [hij])
233   have hcne : ∀ i, c i ≠ [] := by
234     intro i he
235     have h := hlen i (Finset.mem_univ _)
236     rw [he] at h
237     have := hl i
238     simp only [List.length_nil] at h
239     omega
240   refine ⟨⟨c, hci, hcne, ?_⟩, ?_⟩
241   · intro i j hp
242     exact hfree i (Finset.mem_univ _) j (Finset.mem_univ _) hp
243   · intro i
244     exact hlen i (Finset.mem_univ _)

253 /-- Every finite uniquely decodable encoder has a prefix code with exactly
254 the same symbol lengths (the boxed assertion in Chapter 14, Section 14.1). -/
255 theorem exists_prefixCode_of_uniquelyDecodable
256     {α : Type*} [Fintype α] (c : α → List Bool) (hinj : Function.Injective c)
257     (hud : InformationTheory.UniquelyDecodable (Set.range c)) :
258     ∃ code : BinaryPrefixCode α, ∀ i, (code.encode i).length = (c i).length := by
259   classical
260   have hl (i : α) : 0 < (c i).length := by
261     have hn : c i ≠ [] := fun h => hud.epsilon_not_mem ⟨i, h⟩
262     exact List.length_pos_iff.mpr hn
263   have hset : ((Finset.univ.image c : Finset (List Bool)) : Set (List Bool)) =
264       Set.range c := by ext w; simp
265   have hk := InformationTheory.kraft_mcmillan_inequality (S := Finset.univ.image c)
266     (hset.symm ▸ hud)
267   have hki : (∑ i, (1 / 2 : ℝ) ^ (c i).length) ≤ 1 := by
268     simpa only [Fintype.card_bool, Nat.cast_ofNat,
269       Finset.sum_image (fun i _ j _ h => hinj h)] using hk
270   exact exists_binaryPrefixCode_of_kraft_le_one (fun i => (c i).length) hl hki

272 /-- A realizable finite prefix code attains the one-bit entropy sandwich.
273 This proves existence, not Huffman's algorithm or optimality. -/
274 theorem exists_binaryPrefixCode_entropy_sandwich
275     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
276     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
277     ∃ code : BinaryPrefixCode α,
278       discreteEntropyBaseTwo Finset.univ p ≤ expectedCodeLength p code ∧
279       expectedCodeLength p code ≤ discreteEntropyBaseTwo Finset.univ p + 1 := by
280   obtain ⟨l, hl, hk, he⟩ := exists_lengths_kraft_lt_one_entropy_bound p hp hs
281   obtain ⟨code, hc⟩ := exists_binaryPrefixCode_of_kraft_lt_one l hl hk
282   refine ⟨code, discreteEntropyBaseTwo_le_expectedCodeLength p hp hs code, ?_⟩
283   simpa only [expectedCodeLength, hc] using he
