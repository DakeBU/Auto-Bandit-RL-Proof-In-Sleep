8 noncomputable def discreteCrossEntropy {α : Type*} [Fintype α] (p q : α → ℝ) : ℝ :=
9   ∑ i, p i * Real.log (q i)⁻¹

11 /-- The unrounded information-cost interpretation of Eq. (14.4). -/
12 theorem discreteCrossEntropy_sub_entropy {α : Type*} [Fintype α] (p q : α → ℝ)
13     (hsupport : ∀ i, p i ≠ 0 → q i ≠ 0) :
14     discreteCrossEntropy p q - discreteEntropy Finset.univ p =
15       ∑ i, p i * Real.log (p i / q i) := by
16   rw [discreteCrossEntropy, discreteEntropy, ← Finset.sum_sub_distrib]
17   apply Finset.sum_congr rfl
18   intro i _
19   by_cases hi : p i = 0
20   · simp [hi]
21   · rw [Real.log_inv, Real.log_inv, Real.log_div hi (hsupport i hi)]
22     ring

24 theorem relativeEntropy_finite_crossEntropy {α : Type*} [Fintype α]
25     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α)
26     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] (h : P ≪ Q) :
27     relativeEntropy P Q = ENNReal.ofReal
28       (discreteCrossEntropy (fun i => (P {i}).toReal) (fun i => (Q {i}).toReal) -
29         discreteEntropy Finset.univ (fun i => (P {i}).toReal)) := by
30   rw [discreteCrossEntropy_sub_entropy]
31   · exact relativeEntropy_finite_sum_log P Q h
32   · intro i hi
33     have hP : P {i} ≠ 0 := by intro hz; simp [hz] at hi
34     have hQ : Q {i} ≠ 0 := fun hz => hP (h hz)
35     exact ENNReal.toReal_ne_zero.mpr ⟨hQ, measure_ne_top Q {i}⟩

37 /-- The source's zero-mass entropy convention agrees with the right-hand limit. -/
38 theorem entropyTerm_tendsto_zero_right :
39     Filter.Tendsto (fun x : ℝ => x * Real.log x⁻¹)
40       (nhdsWithin 0 (Set.Ioi 0)) (nhds 0) := by
41   have h := Real.continuous_mul_log.continuousAt.tendsto (x := (0 : ℝ))
42   have hn := h.neg
