6 /-- A finite alphabet fitting in n bits has an actual fixed-length prefix code. -/
7 theorem exists_fixedLengthPrefixCode {α : Type*} [Fintype α] (n : ℕ)
8     (hn : 0 < n) (hcapacity : Fintype.card α ≤ 2 ^ n) :
9     ∃ code : BinaryPrefixCode α, ∀ a, (code.encode a).length = n := by
10   classical
11   let e := Fintype.equivFin α
12   have hex (a : α) := exists_binaryAddress n (e a).val ((e a).isLt.trans_le hcapacity)
13   choose c hc hv using hex
14   have hinj : Function.Injective c := by
15     intro a b h
16     apply e.injective
17     apply Fin.ext
18     have he := congrArg binaryAddressValue h
19     simpa [hv] using he
20   let code : BinaryPrefixCode α := {
21     encode := c
22     injective := hinj
23     nonempty := by
24       intro a he
25       have h := hc a
26       simp [he] at h
27       omega
28     prefixFree := by
29       intro a b h
30       exact hinj (h.eq_of_length ((hc a).trans (hc b).symm)) }
31   exact ⟨code, hc⟩

33 /-- The source's ceiling-log binary code, for at least two symbols. -/
34 theorem exists_ceilingLogPrefixCode {α : Type*} [Fintype α]
35     (hcard : 1 < Fintype.card α) :
36     ∃ code : BinaryPrefixCode α, ∀ a, (code.encode a).length = Nat.clog 2 (Fintype.card α) := by
37   have hcap := Nat.le_pow_clog (by norm_num : 1 < 2) (Fintype.card α)
38   have hn : 0 < Nat.clog 2 (Fintype.card α) := by
39     by_contra h
40     have hz : Nat.clog 2 (Fintype.card α) = 0 := by omega
41     simp [hz] at hcap
42     omega
43   exact exists_fixedLengthPrefixCode _ hn hcap
