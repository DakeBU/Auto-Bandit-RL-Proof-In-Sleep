14 /-- A measurable finite-range map admits a finite code and an exact decoder. -/
15 theorem exists_fin_encoding_of_finite_range
16     {α β : Type*} [MeasurableSpace α] [MeasurableSpace β]
17     [MeasurableSingletonClass β] (f : α → β) (hf : Measurable f)
18     (hfin : (Set.range f).Finite) :
19     ∃ (n : ℕ) (g : α → Fin n), Measurable g ∧
20       ∃ d : Fin n → β, f = d ∘ g := by
21   classical
22   letI : Fintype (Set.range f) := hfin.fintype
23   let e := Fintype.equivFin (Set.range f)
24   let g : α → Fin (Fintype.card (Set.range f)) := fun x => e ⟨f x, ⟨x, rfl⟩⟩
25   refine ⟨_, g, (measurable_of_countable e).comp hf.subtype_mk,
26     (fun i => (e.symm i).val), ?_⟩
27   funext x
28   simp [g]

30 /-- Each density-approximation layer is contained in a finite observation sigma-algebra. -/
31 theorem exists_fin_observation_densityApproximation
32     {α : Type*} [m : MeasurableSpace α] (r : α → ENNReal) (n : ℕ) :
33     ∃ (k : ℕ) (g : α → Fin k) (_hg : Measurable g),
34       densityApproximationFiltration r n ≤
35         (inferInstance : MeasurableSpace (Fin k)).comap g := by
36   classical
37   let H : α → (Fin (n + 1) → ENNReal) := fun x i => SimpleFunc.eapprox r i.val x
38   have hH : Measurable H := measurable_pi_lambda _ fun i => (SimpleFunc.eapprox r i.val).measurable
39   have hfinite : (Set.range H).Finite := by
40     apply (Set.Finite.pi' (fun i : Fin (n + 1) => (SimpleFunc.eapprox r i.val).finite_range)).subset
41     rintro y ⟨x, rfl⟩ i
42     exact ⟨x, rfl⟩
43   obtain ⟨k, g, hg, d, hd⟩ := exists_fin_encoding_of_finite_range H hH hfinite
44   refine ⟨k, g, hg, ?_⟩
45   change (⨆ j ≤ n, MeasurableSpace.comap (SimpleFunc.eapprox r j) _) ≤ _
46   refine iSup₂_le fun j hj => ?_
47   apply measurable_iff_comap_le.1
48   have hcoord : (SimpleFunc.eapprox r j : α → ENNReal) =
49       (fun z => d z ⟨j, Nat.lt_succ_of_le hj⟩) ∘ g := by
50     funext x
51     exact congrFun (congrFun hd x) ⟨j, Nat.lt_succ_of_le hj⟩
52   rw [hcoord]
53   exact (measurable_of_countable _).comp (measurable_iff_comap_le.2 le_rfl)

55 /-- Refining a sub-sigma-algebra can only increase its retained KL. -/
56 theorem relativeEntropy_trim_mono
57     {α : Type*} {m₁ m₂ m₀ : MeasurableSpace α} (P Q : @Measure α m₀)
58     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h₁₂ : m₁ ≤ m₂) (h₂ : m₂ ≤ m₀) :
59     @relativeEntropy α m₁ (P.trim (h₁₂.trans h₂)) (Q.trim (h₁₂.trans h₂)) ≤
60       @relativeEntropy α m₂ (P.trim h₂) (Q.trim h₂) := by
61   simpa only [trim_trim] using
62     (relativeEntropy_trim_le (P := P.trim h₂) (Q := Q.trim h₂) h₁₂)

64 /-- Textbook Eq. (14.5) and Theorem 14.1: finite discretisations recover RN KL. -/
65 theorem finitePartitionRelativeEntropy_eq_relativeEntropy
66     {α : Type*} [m : MeasurableSpace α] (P Q : Measure α)
67     [IsFiniteMeasure P] [IsFiniteMeasure Q] :
68     finitePartitionRelativeEntropy P Q = relativeEntropy P Q := by
69   by_cases h : P ≪ Q
70   · apply le_antisymm (finitePartitionRelativeEntropy_le_relativeEntropy P Q)
71     rw [relativeEntropy_eq_iSup_densityApproximation_trim P Q h]
72     refine iSup_le fun n => ?_
73     obtain ⟨k, g, hg, hle⟩ := exists_fin_observation_densityApproximation (P.rnDeriv Q) n
74     calc
75       _ ≤ @relativeEntropy α ((inferInstance : MeasurableSpace (Fin k)).comap g)
76           (P.trim hg.comap_le) (Q.trim hg.comap_le) :=
77         relativeEntropy_trim_mono P Q hle hg.comap_le
78       _ = relativeEntropy (P.map g) (Q.map g) :=
79         (relativeEntropy_map_eq_trim_of_absolutelyContinuous P Q h g hg).symm
80       _ ≤ _ := relativeEntropy_map_le_finitePartitionRelativeEntropy P Q g hg
81   · exact finitePartitionRelativeEntropy_eq_relativeEntropy_of_not_absolutelyContinuous P Q h
82
