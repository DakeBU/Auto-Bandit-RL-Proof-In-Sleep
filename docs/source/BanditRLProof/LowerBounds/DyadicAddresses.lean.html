5 /-- Big-endian binary address, with leading zeroes retained by the word length. -/
6 def binaryAddressValue : List Bool → ℕ
7   | [] => 0
8   | b :: w => (if b then 2 ^ w.length else 0) + binaryAddressValue w
9 
10 theorem binaryAddressValue_lt (w : List Bool) : binaryAddressValue w < 2 ^ w.length := by
11   induction w with
12   | nil => simp [binaryAddressValue]
13   | cons b w ih =>
14     cases b <;> simp only [binaryAddressValue, Bool.false_eq_true, if_false, if_true,
15       zero_add, List.length_cons, pow_succ] <;> omega
16 
17 /-- Every dyadic cell index has a binary address of the specified length. -/
18 theorem exists_binaryAddress (n m : ℕ) (hm : m < 2 ^ n) :
19     ∃ w : List Bool, w.length = n ∧ binaryAddressValue w = m := by
20   induction n generalizing m with
21   | zero =>
22     have hz : m = 0 := by simpa using hm
23     exact ⟨[], rfl, hz.symm⟩
24   | succ n ih =>
25     by_cases hsmall : m < 2 ^ n
26     · obtain ⟨w, hw, hv⟩ := ih m hsmall
27       exact ⟨false :: w, by simp [hw], by simp [binaryAddressValue, hv]⟩
28     · have hrest : m - 2 ^ n < 2 ^ n := by
29         rw [pow_succ] at hm
30         omega
31       obtain ⟨w, hw, hv⟩ := ih (m - 2 ^ n) hrest
32       refine ⟨true :: w, by simp [hw], ?_⟩
33       simp only [binaryAddressValue, if_true, hw, hv]
34       omega
35 
36 theorem binaryAddressValue_append (u v : List Bool) :
37     binaryAddressValue (u ++ v) = binaryAddressValue u * 2 ^ v.length + binaryAddressValue v := by
38   induction u with
39   | nil => simp [binaryAddressValue]
40   | cons b u ih =>
41     cases b <;> simp [binaryAddressValue, ih, List.length_append, pow_add] <;> ring
42 
43 noncomputable def dyadicAddressLower (w : List Bool) : ℝ := (binaryAddressValue w : ℝ) / 2 ^ w.length
44 noncomputable def dyadicAddressUpper (w : List Bool) : ℝ := ((binaryAddressValue w : ℝ) + 1) / 2 ^ w.length
45 
46 theorem dyadicAddress_width (w : List Bool) :
47     dyadicAddressUpper w - dyadicAddressLower w = 1 / (2 : ℝ) ^ w.length := by
48   simp only [dyadicAddressUpper, dyadicAddressLower]
49   ring
50 
51 theorem dyadicAddress_nonempty (w : List Bool) : dyadicAddressLower w < dyadicAddressUpper w := by
52   have h := dyadicAddress_width w
53   have hp : 0 < 1 / (2 : ℝ) ^ w.length := by positivity
54   linarith
55 
56 theorem dyadicAddress_append_contained (u v : List Bool) :
57     dyadicAddressLower u ≤ dyadicAddressLower (u ++ v) ∧
58       dyadicAddressUpper (u ++ v) ≤ dyadicAddressUpper u := by
59   have hu : 0 < (2 : ℝ) ^ u.length := by positivity
60   have hv : 0 < (2 : ℝ) ^ v.length := by positivity
61   have hval : (binaryAddressValue v : ℝ) + 1 ≤ (2 : ℝ) ^ v.length := by
62     exact_mod_cast (Nat.succ_le_of_lt (binaryAddressValue_lt v))
63   have hn : 0 ≤ (binaryAddressValue v : ℝ) := by positivity
64   simp only [dyadicAddressLower, dyadicAddressUpper, binaryAddressValue_append,
65     List.length_append, pow_add, Nat.cast_add, Nat.cast_mul, Nat.cast_pow, Nat.cast_ofNat]
66   constructor
67   · apply (div_le_div_iff₀ hu (mul_pos hu hv)).mpr
68     nlinarith
69   · apply (div_le_div_iff₀ (mul_pos hu hv) hu).mpr
70     nlinarith
71 
72 theorem dyadicAddress_prefix_contained (u v : List Bool) (h : u <+: v) :
73     dyadicAddressLower u ≤ dyadicAddressLower v ∧ dyadicAddressUpper v ≤ dyadicAddressUpper u := by
74   obtain ⟨w, rfl⟩ := h
75   exact dyadicAddress_append_contained u w

77 /-- Select an actual binary word whose dyadic cell fits inside the given interval. -/
78 theorem exists_dyadicAddress_inside (L U : ℝ) (n : ℕ) (hL : 0 ≤ L) (hU : U ≤ 1)
79     (hwidth : 2 * (1 / (2 : ℝ) ^ n) ≤ U - L) :
80     ∃ w : List Bool, w.length = n ∧ L ≤ dyadicAddressLower w ∧ dyadicAddressUpper w < U := by
81   have hpow : 0 < (2 : ℝ) ^ n := by positivity
82   obtain ⟨m, hmL, hmU⟩ := exists_grid_cell_inside L U (1 / (2 : ℝ) ^ n) hL
83     (by positivity) hwidth
84   have hm1 : ((m : ℝ) + 1) / (2 : ℝ) ^ n < 1 := by
85     simpa [div_eq_mul_inv] using hmU.trans_le hU
86   have hm2 := (div_lt_iff₀ hpow).mp hm1
87   have hmR : (m : ℝ) < (2 : ℝ) ^ n := by linarith
88   have hm : m < 2 ^ n := by exact_mod_cast hmR
89   obtain ⟨w, hw, hv⟩ := exists_binaryAddress n m hm
90   refine ⟨w, hw, ?_, ?_⟩
91   · simpa [dyadicAddressLower, hw, hv, div_eq_mul_inv] using hmL
92   · simpa [dyadicAddressUpper, hw, hv, div_eq_mul_inv] using hmU
