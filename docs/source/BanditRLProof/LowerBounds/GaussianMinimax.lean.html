23 /-- A realized pull count, converted to `Real` for the source's regret
24 algebra. -/
25 noncomputable def finiteHistoryPullCountReal
26     {K : Nat} {Reward : Type*}
27     (n : Nat) (history : History.FinitePairHistory (Fin K) Reward n)
28     (arm : Fin K) : Real :=
29   (finiteHistoryPullCountENNReal n history arm).toReal

31 theorem finiteHistoryPullCountENNReal_ne_top
32     {K : Nat} {Reward : Type*}
33     (n : Nat) (history : History.FinitePairHistory (Fin K) Reward n)
34     (arm : Fin K) :
35     finiteHistoryPullCountENNReal n history arm ≠ ∞ := by
36   induction n with
37   | zero =>
38       simp only [finiteHistoryPullCountENNReal]
39       split <;> simp
40   | succ n ih =>
41       simp only [finiteHistoryPullCountENNReal]
42       exact ENNReal.add_ne_top.mpr
43         ⟨ih (Thompson.pairHistoryPrefix history), by split <;> simp⟩

45 theorem sum_finiteHistoryPullCountENNReal
46     {K : Nat} {Reward : Type*}
47     (n : Nat) (history : History.FinitePairHistory (Fin K) Reward n) :
48     ∑ arm : Fin K, finiteHistoryPullCountENNReal n history arm = n + 1 := by
49   classical
50   induction n with
51   | zero =>
52       simp [finiteHistoryPullCountENNReal]
53   | succ n ih =>
54       simp only [finiteHistoryPullCountENNReal, Finset.sum_add_distrib, ih]
55       have hlast : (Thompson.pairHistoryLast history).1 ∈
56           (Finset.univ : Finset (Fin K)) := Finset.mem_univ _
57       have hsingle :
58           (∑ arm : Fin K,
59               if (Thompson.pairHistoryLast history).1 = arm
60               then (1 : ENNReal) else 0) = 1 := by
61         simp
62       rw [hsingle]
63       norm_num

65 @[simp]
66 theorem sum_finiteHistoryPullCountReal
67     {K : Nat} {Reward : Type*}
68     (n : Nat) (history : History.FinitePairHistory (Fin K) Reward n) :
69     ∑ arm : Fin K, finiteHistoryPullCountReal n history arm = n + 1 := by
70   simp only [finiteHistoryPullCountReal]

92 /-- Event integration lower bound in the exact real-probability convention
93 used by the Bretagnolle--Huber theorem. -/
94 theorem ofReal_mul_probReal_le_lintegral_of_event
95     {alpha : Type*} [MeasurableSpace alpha]
96     {mu : Measure alpha} [IsFiniteMeasure mu]
97     {A : Set alpha} (hA : MeasurableSet A)
98     {c : Real} (hc : 0 ≤ c) {f : alpha -> ENNReal}
99     (hf : forall x, x ∈ A -> ENNReal.ofReal c ≤ f x) :
100     ENNReal.ofReal (c * mu.real A) ≤ ∫⁻ x, f x ∂mu := by
101   rw [ENNReal.ofReal_mul hc, ofReal_measureReal]
102   calc
103     ENNReal.ofReal c * mu A =
104         ∫⁻ x, A.indicator (fun _ => ENNReal.ofReal c) x ∂mu := by
105       rw [MeasureTheory.lintegral_indicator hA,
106         MeasureTheory.setLIntegral_const]
107     _ ≤ ∫⁻ x, f x ∂mu := by
108       apply MeasureTheory.lintegral_mono
109       intro x
110       by_cases hx : x ∈ A
111       · simpa [Set.indicator_of_mem hx] using hf x hx
112       · simp [Set.indicator, hx]

116 theorem sixteen_div_twentySeven_le_exp_neg_half :
117     (16 / 27 : Real) ≤ Real.exp (-(1 / 2 : Real)) := by
118   have hexpOne : Real.exp 1 < (729 / 256 : Real) :=
119     Real.exp_one_lt_d9.trans (by norm_num)
120   have hpos : 0 < Real.exp (-(1 / 2 : Real)) := Real.exp_pos _
121   have hsquare : Real.exp (-(1 / 2 : Real)) ^ 2 = (Real.exp 1)⁻¹ := by
122     rw [sq, ← Real.exp_add]
123     convert Real.exp_neg 1 using 1
124     norm_num

126     have h := one_div_lt_one_div_of_lt (by positivity : (0 : Real) < Real.exp 1)

132 /-- A unit-cube Gaussian environment together with a certified optimal arm.

135 structure UnitGaussianBanditEnvironment (K : Nat) where

137   bestArm : Fin K
138   mean_mem_unit : forall arm, mean arm ∈ Set.Icc (0 : Real) 1
139   isBest : forall arm, mean arm ≤ mean bestArm

159 /-- Realized pseudo-regret on an inclusive finite history. -/
160 noncomputable def finiteHistoryGaussianPseudoRegret
161     {K : Nat} (environment : UnitGaussianBanditEnvironment K)
162     (lastRound : Nat)
163     (history : History.FinitePairHistory (Fin K) Real lastRound) : ENNReal :=
164   ∑ arm : Fin K,
165     ENNReal.ofReal
166         (environment.mean environment.bestArm - environment.mean arm) *
167       finiteHistoryPullCountENNReal lastRound history arm

169 /-- Expected pseudo-regret under the canonical history law.  This is the
170 Lemma 4.5-equivalent gap-times-pull-count form of the source quantity `R_n`,
171 represented in `ENNReal` to align with Mathlib's measure-KL API.  The separate
172 reward-sum-regret equality is not asserted by this definition. -/
173 noncomputable def gaussianExpectedPseudoRegret
174     {K : Nat} (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
175     (environment : UnitGaussianBanditEnvironment K)
176     (lastRound : Nat) : ENNReal :=
177   ∫⁻ history,
178     finiteHistoryGaussianPseudoRegret environment lastRound history
179     ∂canonicalBanditHistoryMeasure algorithm
180       (unitGaussianKernel environment.mean) lastRound

182 theorem measurable_finiteHistoryGaussianPseudoRegret
183     {K : Nat} (environment : UnitGaussianBanditEnvironment K)
184     (lastRound : Nat) :
185     Measurable (finiteHistoryGaussianPseudoRegret environment lastRound) := by
186   classical
187   unfold finiteHistoryGaussianPseudoRegret
188   exact Finset.measurable_sum Finset.univ fun arm _harm =>
189     measurable_const.mul
190       (measurable_finiteHistoryPullCountENNReal lastRound arm)

192 /-- Regrouping expected Gaussian pseudo-regret by arm pulls. -/
193 theorem gaussianExpectedPseudoRegret_eq_sum_expectedPulls
194     {K : Nat} (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
195     (environment : UnitGaussianBanditEnvironment K)
196     (lastRound : Nat) :
197     gaussianExpectedPseudoRegret algorithm environment lastRound =
198       ∑ arm : Fin K,
199         ENNReal.ofReal
200             (environment.mean environment.bestArm - environment.mean arm) *
201           canonicalRealizedExpectedPullCountThrough algorithm
202             (unitGaussianKernel environment.mean) lastRound arm := by
203   classical
204   unfold gaussianExpectedPseudoRegret finiteHistoryGaussianPseudoRegret
205   rw [MeasureTheory.lintegral_finset_sum]
206   · apply Finset.sum_congr rfl
207     intro arm _harm
208     rw [MeasureTheory.lintegral_const_mul _
209       (measurable_finiteHistoryPullCountENNReal lastRound arm)]
210     rfl
211   · intro arm _harm
212     exact measurable_const.mul
213       (measurable_finiteHistoryPullCountENNReal lastRound arm)

215 /-- Every inclusive history contains exactly `lastRound + 1` pulls, hence so

217 theorem sum_canonicalRealizedExpectedPullCountThrough
218     {K : Nat} (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
219     (mean : Fin K -> Real) (lastRound : Nat) :

226   · simp_rw [sum_finiteHistoryPullCountENNReal]

245 theorem sum_gaussianExpectedPullCountReal
246     {K : Nat} (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
247     (mean : Fin K -> Real) (lastRound : Nat) :
248     ∑ arm : Fin K, gaussianExpectedPullCountReal algorithm mean lastRound arm =
249       lastRound + 1 := by
250   simp only [gaussianExpectedPullCountReal]
251   rw [← ENNReal.toReal_sum]
252   · rw [sum_canonicalRealizedExpectedPullCountThrough]
253     rw [ENNReal.toReal_add (by simp) (by simp)]
254     simp
255   · intro arm _harm
256     have hle : canonicalRealizedExpectedPullCountThrough algorithm
257         (unitGaussianKernel mean) lastRound arm ≤ lastRound + 1 := by
258       calc
259         canonicalRealizedExpectedPullCountThrough algorithm
260             (unitGaussianKernel mean) lastRound arm ≤
261             ∑ a : Fin K,
262               canonicalRealizedExpectedPullCountThrough algorithm
263                 (unitGaussianKernel mean) lastRound a :=
264           Finset.single_le_sum (fun _ _ => bot_le) (Finset.mem_univ arm)
265         _ = lastRound + 1 :=
266           sum_canonicalRealizedExpectedPullCountThrough algorithm mean lastRound
267     exact ne_top_of_le_ne_top (by simp) hle

269 /-- Base mean vector in the proof of Theorem 15.2: arm zero has mean
270 `gap`, and every alternative has mean zero. -/
271 def gaussianMinimaxBaseMean {m : Nat} (gap : Real) : Fin (m + 1) -> Real :=
272   fun arm => if arm = 0 then gap else 0
273 
274 @[simp]
275 theorem gaussianMinimaxBaseMean_zero {m : Nat} (gap : Real) :
276     gaussianMinimaxBaseMean (m := m) gap 0 = gap := by
277   simp [gaussianMinimaxBaseMean]
278 
279 @[simp]
280 theorem gaussianMinimaxBaseMean_succ {m : Nat} (gap : Real) (i : Fin m) :
281     gaussianMinimaxBaseMean (m := m) gap i.succ = 0 := by
282   simp [gaussianMinimaxBaseMean, Fin.succ_ne_zero]
283 
284 /-- Changed mean vector: the selected alternative `i.succ` is raised from
285 zero to `2*gap`, while arm zero remains at `gap`. -/
286 def gaussianMinimaxChangedMean {m : Nat} (gap : Real) (i : Fin m) :
287     Fin (m + 1) -> Real :=
288   fun arm => if arm = i.succ then 2 * gap
289     else gaussianMinimaxBaseMean gap arm
290 
291 @[simp]
292 theorem gaussianMinimaxChangedMean_selected
293     {m : Nat} (gap : Real) (i : Fin m) :
294     gaussianMinimaxChangedMean gap i i.succ = 2 * gap := by
295   simp [gaussianMinimaxChangedMean]
296 
297 @[simp]
298 theorem gaussianMinimaxChangedMean_zero
299     {m : Nat} (gap : Real) (i : Fin m) :
300     gaussianMinimaxChangedMean gap i 0 = gap := by
301   have hne : (0 : Fin (m + 1)) ≠ i.succ := (Fin.succ_ne_zero i).symm
302   simp [gaussianMinimaxChangedMean, hne]
303 
304 @[simp]
305 theorem gaussianMinimaxChangedMean_other
306     {m : Nat} (gap : Real) {i j : Fin m} (hji : j ≠ i) :
307     gaussianMinimaxChangedMean gap i j.succ = 0 := by
308   simp [gaussianMinimaxChangedMean, Fin.succ_inj, hji]

310 /-- Certified base environment. -/
311 noncomputable def gaussianMinimaxBaseEnvironment
312     {m : Nat} (gap : Real) (hgap : 0 ≤ gap) (hgap_le : gap ≤ 1 / 2) :
313     UnitGaussianBanditEnvironment (m + 1) where
314   mean := gaussianMinimaxBaseMean gap
315   bestArm := 0
316   mean_mem_unit arm := by
317     by_cases harm : arm = 0
318     · subst arm
319       simp only [gaussianMinimaxBaseMean_zero, Set.mem_Icc]
320       constructor <;> linarith
321     · simp [gaussianMinimaxBaseMean, harm]
322   isBest arm := by
323     by_cases harm : arm = 0
324     · subst arm
325       simp
326     · simp [gaussianMinimaxBaseMean, harm, hgap]

328 /-- Certified changed environment with `i.succ` optimal. -/
329 noncomputable def gaussianMinimaxChangedEnvironment
330     {m : Nat} (gap : Real) (i : Fin m)
331     (hgap : 0 ≤ gap) (hgap_le : gap ≤ 1 / 2) :
332     UnitGaussianBanditEnvironment (m + 1) where
333   mean := gaussianMinimaxChangedMean gap i
334   bestArm := i.succ
335   mean_mem_unit arm := by
336     by_cases hselected : arm = i.succ
337     · subst arm
338       simp only [gaussianMinimaxChangedMean_selected, Set.mem_Icc]
339       constructor <;> linarith
340     · by_cases hzero : arm = 0
341       · subst arm
342         simp only [gaussianMinimaxChangedMean_zero, Set.mem_Icc]
343         constructor <;> linarith
344       · simp [gaussianMinimaxChangedMean, gaussianMinimaxBaseMean,
345           hselected, hzero]
346   isBest arm := by
347     by_cases hselected : arm = i.succ
348     · subst arm
349       simp
350     · by_cases hzero : arm = 0
351       · subst arm
352         simp only [gaussianMinimaxChangedMean_zero,
353           gaussianMinimaxChangedMean_selected]
354         linarith
355       · simp [gaussianMinimaxChangedMean, gaussianMinimaxBaseMean,
356           hselected, hzero, hgap]

406     {m : Nat} (gap : Real) (hgap : 0 ≤ gap) (hgap_le : gap ≤ 1 / 2)
