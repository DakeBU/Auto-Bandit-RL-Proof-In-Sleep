9 /-- The class in Chapter 13's main prose: gaps, not means, lie in [0,1]. -/
10 structure UnitSubgaussianBanditEnvironment (k : ℕ) where
11   armLaw : Kernel (Fin k) ℝ
12   markov : IsMarkovKernel armLaw
13   mean : Fin k → ℝ
14   bestArm : Fin k
15   mean_eq : ∀ a, ∫ r, r ∂armLaw a = mean a
16   subgaussian : ∀ a, HasSubgaussianMGF (fun r => r-mean a) 1 (armLaw a)
17   isBest : ∀ a, mean a ≤ mean bestArm
18   gap_le_one : ∀ a, mean bestArm-mean a ≤ 1
19 
20 attribute [instance] UnitSubgaussianBanditEnvironment.markov

22 def UnitGaussianBanditEnvironment.toSubgaussian {k : ℕ}
23     (e : UnitGaussianBanditEnvironment k) : UnitSubgaussianBanditEnvironment k where
24   armLaw := unitGaussianKernel e.mean
25   markov := inferInstance
26   mean := e.mean
27   bestArm := e.bestArm
28   mean_eq := fun a => integral_id_gaussianReal
29   subgaussian := by
30     intro a
31     have h := (hasSubgaussianMGF_gap_sub_id_gaussianReal (e.mean a) 1).neg
32     change HasSubgaussianMGF (fun r => -(e.mean a-r)) 1 (gaussianReal (e.mean a) 1) at h
33     simpa only [neg_sub] using h
34   isBest := e.isBest
35   gap_le_one := fun a => sub_le_iff_le_add.mpr
36     ((e.mean_mem_unit e.bestArm).2.trans (le_add_of_nonneg_right (e.mean_mem_unit a).1))

38 def subgaussianExpectedPseudoRegret {k : ℕ}
39     (algorithm : Thompson.HistoryAlgorithm (Fin k) ℝ)
40     (e : UnitSubgaussianBanditEnvironment k) (t : ℕ) : ℝ≥0∞ :=
41   canonicalGapExpectedPseudoRegret algorithm e.armLaw (fun a => e.mean e.bestArm-e.mean a) t

43 def subgaussianWorstCaseExpectedPseudoRegret (k : ℕ)
44     (algorithm : Thompson.HistoryAlgorithm (Fin k) ℝ) (t : ℕ) : ℝ≥0∞ :=
45   ⨆ e : UnitSubgaussianBanditEnvironment k, subgaussianExpectedPseudoRegret algorithm e t

48   ⨅ algorithm : Thompson.HistoryAlgorithm (Fin k) ℝ,

57 /-- Gaussian subclass inclusion is on the identical policy and regret functional. -/
58 theorem unitGaussianMinimax_le_subgaussianMinimax (k t : ℕ) :
59     unitGaussianMinimaxExpectedPseudoRegret k t ≤ subgaussianMinimaxExpectedPseudoRegret k t := by
60   apply iInf_mono
61   intro algorithm
62   apply iSup_le
63   intro e
64   exact (subgaussianExpectedPseudoRegret_gaussian algorithm e t).symm.le.trans
65     (le_iSup (fun e => subgaussianExpectedPseudoRegret algorithm e t) e.toSubgaussian)

67 theorem moss_subgaussianExpectedPseudoRegret_le {k : ℕ} [NeZero k]
68     (hk : 0 < k) (t : ℕ) (hkt : k ≤ t+1) (e : UnitSubgaussianBanditEnvironment k) :
69     subgaussianExpectedPseudoRegret (MOSS.historyAlgorithm hk (t+1)) e t ≤
70       ENNReal.ofReal (40*Real.sqrt ((k : ℝ)*(t+1))) := by
71   have hu := MOSS.canonicalGapExpectedRegret_le hk e.armLaw t hkt e.mean e.bestArm
72     e.isBest e.mean_eq e.subgaussian
73   have hg : ∑ a, (e.mean e.bestArm-e.mean a) ≤ (k : ℝ) := by
74     calc
75       _ ≤ ∑ _a : Fin k, (1 : ℝ) := Finset.sum_le_sum (fun a _ => e.gap_le_one a)
76       _ = _ := by simp
77   have hc : (k : ℝ) ≤ Real.sqrt ((k : ℝ)*(t+1)) := by
78     apply Real.le_sqrt_of_sq_le
79     have ht : (k : ℝ) ≤ (t : ℝ)+1 := by exact_mod_cast hkt
80     nlinarith [show (0 : ℝ) ≤ k by positivity]
81   have hr : (subgaussianExpectedPseudoRegret (MOSS.historyAlgorithm hk (t+1)) e t).toReal ≤
82       40*Real.sqrt ((k : ℝ)*(t+1)) := by
83     change (canonicalGapExpectedPseudoRegret _ _ _ _).toReal ≤ _
84     dsimp [canonicalGapExpectedPseudoRegretReal] at hu
85     simp only [Nat.cast_add, Nat.cast_one, mul_comm ((t : ℝ)+1)] at hu
86     linarith
87   change canonicalGapExpectedPseudoRegret _ _ _ _ ≤ _
88   rw [← ENNReal.ofReal_toReal (canonicalGapExpectedPseudoRegret_ne_top _ _ _ _)]
89   exact ENNReal.ofReal_le_ofReal hr

91 /-- Chapter 13's broader-class minimax sandwich with universal constants. -/
92 theorem subgaussianMinimax_sandwich {k : ℕ} [NeZero k]
93     (hk : 1 < k) (t : ℕ) (hkt : k ≤ t+1) :
94     ENNReal.ofReal ((1/54 : ℝ)*Real.sqrt ((k : ℝ)*(t+1))) ≤
95       subgaussianMinimaxExpectedPseudoRegret k t ∧
96     subgaussianMinimaxExpectedPseudoRegret k t ≤
97       subgaussianWorstCaseExpectedPseudoRegret k (MOSS.historyAlgorithm (by omega) (t+1)) t ∧
98     subgaussianWorstCaseExpectedPseudoRegret k (MOSS.historyAlgorithm (by omega) (t+1)) t ≤
99       ENNReal.ofReal (40*Real.sqrt ((k : ℝ)*(t+1))) := by
100   refine ⟨?_, iInf_le _ _, ?_⟩
101   · have h := unitGaussianMinimaxExpectedPseudoRegret_ge_one_div_fiftyFour_sqrt hk hkt
102     simp only [Nat.add_sub_cancel, Nat.cast_add, Nat.cast_one] at h
103     exact h.trans (unitGaussianMinimax_le_subgaussianMinimax k t)
104   · exact iSup_le fun e => moss_subgaussianExpectedPseudoRegret_le (by omega) t hkt e

106 /-- Algorithm 7 is minimax optimal up to one universal multiplicative factor. -/
107 theorem moss_nearMinimax {k : ℕ} [NeZero k]
108     (hk : 1 < k) (t : ℕ) (hkt : k ≤ t+1) :
109     subgaussianWorstCaseExpectedPseudoRegret k (MOSS.historyAlgorithm (by omega) (t+1)) t ≤
110       2160 * subgaussianMinimaxExpectedPseudoRegret k t := by
111   obtain ⟨hl, _, hu⟩ := subgaussianMinimax_sandwich hk t hkt
112   calc
113     _ ≤ ENNReal.ofReal (40*Real.sqrt ((k : ℝ)*(t+1))) := hu
114     _ = 2160 * ENNReal.ofReal ((1/54 : ℝ)*Real.sqrt ((k : ℝ)*(t+1))) := by
115       rw [show 40*Real.sqrt ((k : ℝ)*(t+1)) =
116         2160*((1/54 : ℝ)*Real.sqrt ((k : ℝ)*(t+1))) by ring,
117         ENNReal.ofReal_mul (by norm_num : (0 : ℝ) ≤ 2160)]
118       norm_num
119     _ ≤ _ := mul_le_mul_left' hl _
