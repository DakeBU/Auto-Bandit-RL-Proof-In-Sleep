11 /-- On a finite alphabet, absolute continuity is exactly atomwise support inclusion. -/
12 theorem absolutelyContinuous_iff_atom_support {α : Type*} [Fintype α]
13     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α) :
14     P ≪ Q ↔ ∀ x, Q {x} = 0 → P {x} = 0 := by
15   classical
16   constructor
17   · intro h x hx
18     exact h hx
19   · intro h s hs
20     rw [← Set.coe_toFinset s, ← sum_measure_singleton]
21     apply Finset.sum_eq_zero
22     intro x hx
23     exact h x (measure_mono_null (Set.singleton_subset_iff.2
24       (Set.mem_toFinset.1 hx)) hs)

26 /-- Atomwise density identity, retaining zero-mass atoms. -/
27 theorem rnDeriv_mul_atom {α : Type*} [MeasurableSpace α]
28     [MeasurableSingletonClass α] (P Q : Measure α)
29     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q) (x : α) :
30     P.rnDeriv Q x * Q {x} = P {x} := by
31   have heq := congrArg (fun μ : Measure α => μ {x})
32     (Measure.withDensity_rnDeriv_eq P Q h)
33   simpa [mul_comm] using heq
34 
35 /-- On a positive reference atom, the RN density is the atom-mass ratio. -/
36 theorem rnDeriv_atom_eq_div {α : Type*} [MeasurableSpace α]
37     [MeasurableSingletonClass α] (P Q : Measure α)
38     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q) (x : α)
39     (hq : Q {x} ≠ 0) : P.rnDeriv Q x = P {x} / Q {x} := by
40   exact (ENNReal.eq_div_iff hq (measure_ne_top Q {x})).2
41     (by simpa [mul_comm] using rnDeriv_mul_atom P Q h x)

43 /-- A positive source atom absent from the reference law forces infinite KL. -/
44 theorem relativeEntropy_eq_top_of_atom_support_mismatch {α : Type*}
45     [MeasurableSpace α] (P Q : Measure α) (x : α)
46     (hp : P {x} ≠ 0) (hq : Q {x} = 0) : relativeEntropy P Q = ∞ := by
47   apply InformationTheory.klDiv_of_not_ac
48   intro h
49   exact hp (h hq)

51 /-- Finite-alphabet KL in its nonnegative convex-integrand form. -/
52 theorem relativeEntropy_finite_klFun {α : Type*} [Fintype α]
53     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α)
54     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q) :
55     relativeEntropy P Q = ∑ x, ENNReal.ofReal
56       (InformationTheory.klFun ((P {x} / Q {x}).toReal)) * Q {x} := by
57   rw [relativeEntropy, InformationTheory.klDiv_eq_lintegral_klFun_of_ac h,
58     lintegral_fintype]
59   apply Finset.sum_congr rfl
60   intro x _
61   by_cases hq : Q {x} = 0
62   · simp [hq]
63   · rw [rnDeriv_atom_eq_div P Q h x hq]
64 
65 /-- Textbook Eq. (14.4) on any finite alphabet in the supported branch. -/
66 theorem relativeEntropy_finite_sum_log {α : Type*} [Fintype α]
67     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α)
68     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] (h : P ≪ Q) :
69     relativeEntropy P Q = ENNReal.ofReal
70       (∑ x, (P {x}).toReal * Real.log ((P {x}).toReal / (Q {x}).toReal)) := by
71   have hi : Integrable (llr P Q) P :=
72     (SimpleFunc.ofFinite (llr P Q)).integrable_of_isFiniteMeasure
73   rw [relativeEntropy_of_probability_absolutelyContinuous_of_integrable P Q h hi,
74     integral_fintype hi]
75   congr 1
76   apply Finset.sum_congr rfl
77   intro x _
78   by_cases hp : P {x} = 0
79   · simp [measureReal_def, hp]
80   · have hq : Q {x} ≠ 0 := fun hz => hp (h hz)
81     simp [llr, rnDeriv_atom_eq_div P Q h x hq, ENNReal.toReal_div,
82       measureReal_def, smul_eq_mul]
83 
84 /-- Eq. (14.4), including the infinite branch when atomwise support fails. -/
85 theorem relativeEntropy_finite_eq_if {α : Type*} [Fintype α]
86     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α)
87     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] :
88     relativeEntropy P Q = if ∀ x, Q {x} = 0 → P {x} = 0 then
89       ENNReal.ofReal (∑ x, (P {x}).toReal *
90         Real.log ((P {x}).toReal / (Q {x}).toReal)) else ∞ := by
91   classical
92   split_ifs with h
93   · exact relativeEntropy_finite_sum_log P Q
94       ((absolutelyContinuous_iff_atom_support P Q).2 h)
95   · exact InformationTheory.klDiv_of_not_ac
96       (fun hac => h ((absolutelyContinuous_iff_atom_support P Q).1 hac))
97 
98 /-- Finite alphabets have infinite KL exactly at a support mismatch. -/
99 theorem relativeEntropy_finite_eq_top_iff {α : Type*} [Fintype α]
100     [MeasurableSpace α] [MeasurableSingletonClass α] (P Q : Measure α)
101     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] :
102     relativeEntropy P Q = ∞ ↔ ∃ x, P {x} ≠ 0 ∧ Q {x} = 0 := by
103   classical
104   rw [relativeEntropy_finite_eq_if]
105   split_ifs with h
106   · simp only [ENNReal.ofReal_ne_top, false_iff, not_exists, not_and]
107     exact fun x hp hq => hp (h x hq)
108   · simp only [not_forall] at h
109     exact iff_of_true rfl (by obtain ⟨x, hq, hp⟩ := h; exact ⟨x, hp, hq⟩)
