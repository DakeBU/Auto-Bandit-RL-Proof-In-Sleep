10 # High-probability lower-bound interfaces
11 
12 This module formalizes the source-faithful threshold surfaces and reusable
13 probability/algebra leaves from Lattimore--Szepesvari, *Bandit Algorithms*
14 (2020), Part IV, Chapter 17.
15 
16 The file closes Theorem 17.1 and Corollaries 17.2--17.3 on the canonical adaptive
17 history law, and keeps their random pseudo-regret separate from deterministic
18 expected pseudo-regret.  It also formalizes Claim 17.5, the exact correlated
19 clipped-Gaussian path construction, and Eq. (17.8) for adversarial random
20 regret. Claim 17.7 is closed at the textbook constants. The user-approved
21 correction of Claim 17.6 uses a non-strict half-pull event. Corrected Theorem
22 17.4 has `0 < delta <= 1/32`, `c=1/160`, `C=64`, a deterministic bounded
23 reward-table witness, and the strict CDF-complement tail. The same-policy
24 coupling is proved, and fixed-table random regret is measurable and integrable;
25 the false printed strict claim and full confidence domain are not asserted.

37 /-- The tail event used throughout Chapter 17: the realized quantity is at
38 least the displayed lower-bound threshold. -/
39 def tailAtLeast {Omega : Type*} (quantity : Omega -> Real) (threshold : Real) :
40     Set Omega :=
41   {omega | threshold <= quantity omega}

43 /-- The exact threshold inside Theorem 17.1, with `alternativeArms = k - 1`.
44 The theorem's factor `1/4` multiplies the whole minimum. -/
45 def stochasticHighProbabilityThreshold
46     (horizon alternativeArms : Nat) (B delta : Real) : Real :=
47   (1 / 4 : Real) *
48     min (horizon : Real)
49       ((1 / B) * Real.sqrt ((alternativeArms : Real) * (horizon : Real)) *
50         Real.log (1 / (4 * delta)))

52 /-- The exact threshold in Corollary 17.2, again with
53 `alternativeArms = k - 1`. -/
54 def stochasticMinimaxHighProbabilityThreshold
55     (horizon alternativeArms : Nat) (delta : Real) : Real :=
56   (1 / 4 : Real) *
57     min (horizon : Real)
58       (Real.sqrt
59         (((horizon : Real) * (alternativeArms : Real) / 2) *
60           Real.log (1 / (4 * delta))))

271 /-- Integrating a nonnegative bounded random variable after splitting at one
272 measurable upper-tail event. -/
273 theorem integral_le_threshold_add_bound_mul_tailMass
274     {Omega : Type*} [MeasurableSpace Omega]
275     (mu : Measure Omega) [IsProbabilityMeasure mu]
276     (quantity : Omega -> Real) (threshold bound : Real)
277     (hmeas : Measurable quantity) (hintegrable : Integrable quantity mu)
278     (hthreshold : 0 <= threshold)
279     (hbound : forall omega, quantity omega <= bound) :
280     ∫ omega, quantity omega ∂mu <=
281       threshold + bound * mu.real (tailAtLeast quantity threshold) := by
282   let bad := tailAtLeast quantity threshold
283   have hbad : MeasurableSet bad :=
284     measurableSet_le measurable_const hmeas
285   let penalty : Omega -> Real := bad.indicator (fun _ => bound)
286   have hpenalty : Integrable penalty mu :=
287     (integrable_const bound).indicator hbad
288   have hmajorant : Integrable (fun omega => threshold + penalty omega) mu :=
289     (integrable_const threshold).add hpenalty
290   calc
291     ∫ omega, quantity omega ∂mu <=
292         ∫ omega, threshold + penalty omega ∂mu := by
293       apply integral_mono hintegrable hmajorant
294       intro omega
295       by_cases hmem : omega ∈ bad
296       · simp [penalty, hmem]
297         exact (hbound omega).trans (le_add_of_nonneg_left hthreshold)
298       · simp [penalty, hmem]
299         simp only [bad, tailAtLeast, Set.mem_setOf_eq] at hmem
300         exact (lt_of_not_ge hmem).le
301     _ = threshold + bound * mu.real bad := by
302       rw [integral_add (integrable_const threshold) hpenalty]
303       simp [penalty, hbad, Measure.real_def, mul_comm]
