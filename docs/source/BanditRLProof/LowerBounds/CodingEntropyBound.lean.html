7 theorem entropy_term_le_codeLength_remainder {p : ℝ} (hp : 0 ≤ p) (l : ℕ) :
8     p * Real.log p⁻¹ ≤ p * l * Real.log 2 + (1 / 2 : ℝ) ^ l - p := by
9   rcases eq_or_lt_of_le hp with he | he
10   · subst p
11     simp only [zero_mul, sub_zero]
12     positivity
13   have hw : 0 < (1 / 2 : ℝ) ^ l := by positivity
14   have h := mul_le_mul_of_nonneg_left
15     (Real.log_le_sub_one_of_pos (div_pos hw he)) hp
16   rw [Real.log_div hw.ne' he.ne', Real.log_pow, Real.log_div (by norm_num) (by norm_num),
17     Real.log_one] at h
18   have hc : p * ((1 / 2 : ℝ) ^ l / p - 1) = (1 / 2 : ℝ) ^ l - p := by
19     field_simp
20   rw [hc] at h
21   rw [Real.log_inv]
22   nlinarith

25 theorem discreteEntropyBaseTwo_le_expectedCodeLength
26     {Symbol : Type*} [Fintype Symbol] [DecidableEq Symbol]
27     (p : Symbol → ℝ) (hp : ∀ i, 0 ≤ p i) (hsum : ∑ i, p i = 1)
28     (code : BinaryPrefixCode Symbol) :
29     discreteEntropyBaseTwo Finset.univ p ≤ expectedCodeLength p code := by
30   have hk : ∑ i, (1 / 2 : ℝ) ^ (code.encode i).length ≤ 1 := by
31     have h := code.kraft_inequality
32     rw [BinaryPrefixCode.codebook, Finset.sum_image] at h
33     · exact h
34     · intro a _ b _ hab
35       exact code.injective hab
36   have h := Finset.sum_le_sum (s := Finset.univ)
37     (fun i _ => entropy_term_le_codeLength_remainder (hp i) (code.encode i).length)
38   simp only [Finset.sum_sub_distrib, Finset.sum_add_distrib, ← Finset.sum_mul, hsum] at h
39   rw [discreteEntropyBaseTwo_eq_div_log_two]
40   apply (div_le_iff₀ (Real.log_pos (by norm_num : (1 : ℝ) < 2))).2
41   change (∑ i, p i * Real.log (p i)⁻¹) ≤
42     (∑ i, p i * (code.encode i).length) * Real.log 2
