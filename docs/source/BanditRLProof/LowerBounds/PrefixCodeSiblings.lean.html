5 theorem BinaryPrefixCode.extended_prefix_parent_eq
6     {α : Type*} (code : BinaryPrefixCode α) (a b : α) (u v : List Bool)
7     (h : code.encode a ++ u <+: code.encode b ++ v) : a = b := by
8   have ha := (List.prefix_append (code.encode a) u).trans h
9   have hb := List.prefix_append (code.encode b) v
10   rcases List.prefix_or_prefix_of_prefix ha hb with hab | hba
11   · exact code.prefixFree hab
12   · exact (code.prefixFree hba).symm

14 def siblingExpandedWord {α : Type*} (code : BinaryPrefixCode (Option α)) : α ⊕ Bool → List Bool
15   | .inl a => code.encode (some a)
16   | .inr b => code.encode none ++ [b]

18 theorem siblingExpandedWord_prefixFree {α : Type*} (code : BinaryPrefixCode (Option α))
19     {a b : α ⊕ Bool} (h : siblingExpandedWord code a <+: siblingExpandedWord code b) : a = b := by
20   cases a with
21   | inl a =>
22     cases b with
23     | inl b =>
24       have he := code.prefixFree h
25       exact congrArg Sum.inl (Option.some.inj he)
26     | inr b =>
27       have he := code.extended_prefix_parent_eq (some a) none [] [b] (by simpa [siblingExpandedWord] using h)
28       contradiction
29   | inr a =>
30     cases b with
31     | inl b =>
32       have he := code.extended_prefix_parent_eq none (some b) [a] [] (by simpa [siblingExpandedWord] using h)
33       contradiction
34     | inr b =>
35       change code.encode none ++ [a] <+: code.encode none ++ [b] at h
36       obtain ⟨t, ht⟩ := h
37       have hlen := congrArg List.length ht
38       simp only [List.length_append, List.length_singleton] at hlen
39       have ht0 : t = [] := List.length_eq_zero_iff.mp (by omega)
40       simp only [ht0, List.append_nil] at ht
41       have he := List.append_cancel_left ht
42       simp only [List.singleton_inj] at he
43       exact congrArg Sum.inr he

46 def BinaryPrefixCode.expandSibling {α : Type*} (code : BinaryPrefixCode (Option α)) :
47     BinaryPrefixCode (α ⊕ Bool) where
48   encode := siblingExpandedWord code
49   injective := fun a b h => siblingExpandedWord_prefixFree code (by rw [h])
50   nonempty := by
51     intro a
52     cases a with
53     | inl a => exact code.nonempty (some a)
54     | inr b => simp [siblingExpandedWord]

67 theorem sibling_parent_not_prefix_other
68     {α : Type*} (code : BinaryPrefixCode (α ⊕ Bool)) (w : List Bool)
69     (hs : ∀ b, code.encode (.inr b) = w ++ [b]) (a : α) :
70     ¬ w <+: code.encode (.inl a) := by
71   rintro ⟨t, ht⟩
72   cases t with
73   | nil =>
74     have he : w = code.encode (.inl a) := by simpa using ht
75     have hp : code.encode (.inl a) <+: code.encode (.inr false) := by
76       rw [hs false, ← he]
77       exact List.prefix_append _ _
78     have h := code.prefixFree hp
79     contradiction
80   | cons b t =>
81     have hp : code.encode (.inr b) <+: code.encode (.inl a) := by
82       rw [hs b]
83       exact ⟨t, by simpa [List.append_assoc] using ht⟩
84     have h := code.prefixFree hp
85     contradiction

92 theorem siblingContractedWord_prefixFree
93     {α : Type*} (code : BinaryPrefixCode (α ⊕ Bool)) (w : List Bool)
94     (hs : ∀ b, code.encode (.inr b) = w ++ [b]) {a b : Option α}
95     (h : siblingContractedWord code w a <+: siblingContractedWord code w b) : a = b := by
96   cases a with
97   | none =>
98     cases b with
99     | none => rfl
100     | some b => exact (sibling_parent_not_prefix_other code w hs b h).elim
101   | some a =>
102     cases b with
103     | none =>
104       have hp : code.encode (.inl a) <+: code.encode (.inr false) := by
105         rw [hs false]
106         exact h.trans (List.prefix_append _ _)
107       have he := code.prefixFree hp
108       contradiction
109     | some b =>
110       exact congrArg some (Sum.inl.inj (code.prefixFree h))

112 /-- Merge two actual sibling leaves whose parent is nonempty. -/
113 def BinaryPrefixCode.contractSibling
114     {α : Type*} (code : BinaryPrefixCode (α ⊕ Bool)) (w : List Bool) (hw : w ≠ [])
115     (hs : ∀ b, code.encode (.inr b) = w ++ [b]) : BinaryPrefixCode (Option α) where
116   encode := siblingContractedWord code w
117   injective := fun a b h => siblingContractedWord_prefixFree code w hs (by rw [h])
118   nonempty := by
119     intro a
120     cases a with
121     | none => exact hw
122     | some a => exact code.nonempty (.inl a)
123   prefixFree := siblingContractedWord_prefixFree code w hs

135 /-- The two-symbol root code, avoiding an invalid empty parent codeword. -/
136 def binaryRootPrefixCode : BinaryPrefixCode Bool where
137   encode := fun b => [b]
138   injective := fun _ _ h => List.singleton_inj.mp h
139   nonempty := fun _ => by simp
140   prefixFree := by
141     intro a b h
142     cases a <;> cases b <;> simp_all

144 theorem binaryRootPrefixCode_optimal (p : Bool → ℝ)
145     (hp : ∀ b, 0 ≤ p b) (hs : ∑ b, p b = 1) :
146     IsOptimalPrefixCode p binaryRootPrefixCode := by
147   intro other
