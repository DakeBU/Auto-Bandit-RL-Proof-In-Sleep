33 /-- The exact scalar asymptotic quantifier in Definition 16.1: a nonnegative
34 regret sequence is smaller than every positive polynomial order.  Regret
35 nonnegativity is kept outside this analytic predicate so callers must expose
36 the model-specific fact explicitly. -/
37 def IsConsistentRegret (regret : Nat -> Real) : Prop :=
38   forall p : Real, 0 < p ->
39     Tendsto (fun n : Nat => regret n / (n : Real) ^ p) atTop (nhds 0)
40 
41 /-- Definition 16.1 over an abstract policy/environment regret interface.
42 This preserves the source quantifier order: one policy, every environment in
43 the class, and every positive real exponent. -/
44 def IsConsistentPolicyOver
45     {Policy Environment : Type*}
46     (environmentClass : Set Environment)
47     (regret : Policy -> Environment -> Nat -> Real)
48     (policy : Policy) : Prop :=
49   forall environment, environment ∈ environmentClass ->
50     IsConsistentRegret (regret policy environment)

52 /-- A finite-armed product environment whose arm laws have certified finite
53 means.  The `mean` field is intentionally tied to the Bochner integral rather
54 than treated as an unrelated parameter: this is the source-level bridge used
55 when an unchanged arm law must imply an unchanged arm mean. -/
56 structure FiniteMeanBanditEnvironment (K : Nat) where
57   armLaw : Kernel (Fin K) Real
58   armLaw_isMarkov : IsMarkovKernel armLaw

61   mean_eq_integral : forall arm, mean arm = ∫ reward, reward ∂armLaw arm
62   bestArm : Fin K
63   isBest : forall arm, mean arm <= mean bestArm

205 /-- Source mean-to-gap producer for a one-arm change.  If the changed arm is
206 suboptimal originally, uniquely optimal after the change, and every other arm
207 law is unchanged, then it produces all sign and comparison obligations needed
208 by the exact majority-event consumer for Lemma 16.3. -/
209 theorem oneArmMeanChange_produces_gap_contract
210     {K : Nat} (original reference : FiniteMeanBanditEnvironment K)
211     (changedArm : Fin K)
212     (hsuboptimal :
213       original.mean changedArm < original.mean original.bestArm)
214     (hunique : forall arm, arm ≠ changedArm ->
215       reference.mean arm < reference.mean changedArm)
216     (hsame : forall arm, arm ≠ changedArm ->
217       original.armLaw arm = reference.armLaw arm) :
218     0 < original.gap changedArm /\
219       0 < oneArmChangedMargin original reference changedArm /\
220       (forall arm, arm ≠ changedArm ->
221         oneArmChangedMargin original reference changedArm <=
222           reference.gap arm) := by
223   have hbest_ne : original.bestArm ≠ changedArm := by
224     intro heq
225     rw [heq] at hsuboptimal
226     exact (lt_irrefl _ hsuboptimal)
227   have hbest_mean :
228       original.mean original.bestArm = reference.mean original.bestArm :=
229     FiniteMeanBanditEnvironment.mean_eq_of_armLaw_eq
230       original reference original.bestArm (hsame original.bestArm hbest_ne)
231   have hchangedAbove :
232       original.mean original.bestArm < reference.mean changedArm := by
233     rw [hbest_mean]
234     exact hunique original.bestArm hbest_ne
235   have hrefBest : reference.bestArm = changedArm := by
236     by_contra hne
237     have hlt := hunique reference.bestArm hne
238     have hle := reference.isBest changedArm
239     exact (not_lt_of_ge hle) hlt
240   refine ⟨?_, ?_, ?_⟩
241   · exact sub_pos.mpr hsuboptimal
242   · exact sub_pos.mpr hchangedAbove
243   · intro arm harm
244     have hmean := FiniteMeanBanditEnvironment.mean_eq_of_armLaw_eq
245       original reference arm (hsame arm harm)
246     rw [oneArmChangedMargin, FiniteMeanBanditEnvironment.gap,
247       hrefBest, ← hmean]
248     linarith [original.isBest arm]

526 theorem IsConsistentRegret.eventually_log_add_div_log_le
527     {first second : Nat -> Real}

529     (hsecond : IsConsistentRegret second)
530     (hpositive : ∀ᶠ n : Nat in atTop, 0 < first n + second n)
531     {p : Real} (hp : 0 < p) :

533       Real.log (first n + second n) / Real.log n <= p := by

544 /-- Consistency extracts every strict reciprocal-information lower bound
545 from the source logarithmic inequality. This eventual form does not assume
546 that the normalized pull counts converge. -/
547 theorem IsConsistentRegret.eventually_pull_div_log_ge
548     {first second pulls : Nat → Real} {c d r : Real}
549     (hfirst : IsConsistentRegret first) (hsecond : IsConsistentRegret second)
550     (hpositive : ∀ᶠ n in atTop, 0 < first n + second n)
551     (hd : 0 < d) (hr : r < 1 / d)

553       (c + Real.log n - Real.log (first n + second n)) / d ≤ pulls n) :
554     ∀ᶠ n in atTop, r ≤ pulls n / Real.log n := by
555   let p : Real := (1 - r * d) / 2
556   have hrd : r * d < 1 := (lt_div_iff₀ hd).mp hr
557   have hp : 0 < p := by dsimp [p]; linarith
558   have hrp : r < (1 - p) / d := by
559     apply (lt_div_iff₀ hd).2
560     dsimp [p]
561     linarith
562   have hlogtop : Tendsto (fun n : Nat => Real.log (n : Real)) atTop atTop :=
563     Real.tendsto_log_atTop.comp tendsto_natCast_atTop_atTop
564   have hconstant : Tendsto (fun n : Nat => c / Real.log n) atTop (nhds 0) :=
565     tendsto_const_nhds.div_atTop hlogtop
566   have hlimit : Tendsto (fun n : Nat => (c / Real.log n + 1 - p) / d)
567       atTop (nhds ((1 - p) / d)) := by
568     simpa using ((hconstant.add_const 1).sub_const p).div_const d
569   have hevent := hlimit.eventually_const_lt hrp
570   filter_upwards [hevent, hsource,

572     eventually_gt_atTop (1 : Nat)] with n hn hs hg hnlarge
573   have hnreal : (1 : Real) < n := by exact_mod_cast hnlarge
574   have hlog : 0 < Real.log (n : Real) := Real.log_pos hnreal
575   apply hn.le.trans
576   apply (div_le_iff₀ hd).2
577   rw [div_mul_eq_mul_div]
578   apply (le_div_iff₀ hlog).2
579   have hc : c / Real.log n * Real.log n = c := div_mul_cancel₀ c hlog.ne'
580   have hs' := (div_le_iff₀ hd).mp hs
581   have hg' := (div_le_iff₀ hlog).mp hg
