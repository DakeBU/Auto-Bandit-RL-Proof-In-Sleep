5 def sourceBlockList {α : Type*} : (n : ℕ) → SourceBlock α n → List α
6   | 0, _ => []
7   | n + 1, x => x.1 :: sourceBlockList n x.2

9 theorem sourceBlockList_length {α : Type*} (n : ℕ) (x : SourceBlock α n) :
10     (sourceBlockList n x).length = n := by
11   induction n with
12   | zero => rfl
13   | succ n ih => simp [sourceBlockList, ih]

15 theorem sourceBlockList_injective {α : Type*} (n : ℕ) :
16     Function.Injective (sourceBlockList (α := α) n) := by
17   induction n with
18   | zero =>
19     change Function.Injective (fun _ : PUnit => ([] : List α))
20     intro a b _
21     exact Subsingleton.elim a b
22   | succ n ih =>
23     intro a b h
24     have he := List.cons.inj h
25     exact Prod.ext he.1 (ih he.2)

27 theorem sourceBlockList_mass {α : Type*} (p : α → ℝ) (n : ℕ) (x : SourceBlock α n) :
28     ((sourceBlockList n x).map p).prod = sourceBlockMass p n x := by
29   induction n with
30   | zero => rfl
31   | succ n ih => simp [sourceBlockList, sourceBlockMass, ih]

33 theorem exists_arithmeticBlockSupport {k : ℕ} (p : Fin k → ℝ)
34     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) :
35     ∃ positive : BinaryPrefixCode {x : SourceBlock.{0,0} (Fin k) n // 0 < sourceBlockMass p n x},
36       (∀ x, (positive.encode x).length = arithmeticLength (sourceBlockMass p n x.val)) ∧
37       (∀ x, (arithmeticInterval p (sourceBlockList n x.val)).1 ≤
38         dyadicAddressLower (positive.encode x) ∧
39         dyadicAddressUpper (positive.encode x) <
40           (arithmeticInterval p (sourceBlockList n x.val)).2) ∧
41       expectedCodeLength (sourceBlockMass p n)
42         (positive.extendZeroMass (sourceBlockMass p n)
43           (huffmanCode (sourceBlockMass p n) (sourceBlockMass_nonneg p hp n))) ≤
44         n * discreteEntropyBaseTwo Finset.univ p + 3 := by
45   have h := exists_zeroSafe_arithmeticCode p hp hs (sourceBlockList n)
46     (sourceBlockList_injective n) (fun a b => (sourceBlockList_length n a).trans
47       (sourceBlockList_length n b).symm) (sourceBlockMass p n)
48     (sourceBlockMass_nonneg p hp n) (sum_sourceBlockMass p hs n) (sourceBlockList_mass p n)
49   simpa [discreteEntropyBaseTwo_sourceBlockMass p hs n] using h

51 /-- The named arithmetic block code: interval addresses on positive blocks,
52 with the one-bit tagged fallback on zero-mass blocks. -/
53 noncomputable def arithmeticBlockCode {k : ℕ} (p : Fin k → ℝ)
54     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) :
55     BinaryPrefixCode (SourceBlock.{0,0} (Fin k) n) :=
56   (Classical.choose (exists_arithmeticBlockSupport p hp hs n)).extendZeroMass
57     (sourceBlockMass p n) (huffmanCode (sourceBlockMass p n) (sourceBlockMass_nonneg p hp n))
58 
59 theorem arithmeticBlockCode_expected_length_le {k : ℕ} (p : Fin k → ℝ)
60     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) :
61     expectedCodeLength (sourceBlockMass p n) (arithmeticBlockCode p hp hs n) ≤
62       n * discreteEntropyBaseTwo Finset.univ p + 3 :=
63   (Classical.choose_spec (exists_arithmeticBlockSupport p hp hs n)).2.2

65 /-- Removing the support tag from a positive block yields an address inside
66 its actual arithmetic interval. This property belongs to the named code,
67 not merely to an unrelated witness with the same length. -/
68 theorem arithmeticBlockCode_payload_interval {k : ℕ} (p : Fin k → ℝ)
69     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ)
70     (x : SourceBlock.{0,0} (Fin k) n) (hx : 0 < sourceBlockMass p n x) :
71     (arithmeticInterval p (sourceBlockList n x)).1 ≤
72       dyadicAddressLower ((arithmeticBlockCode p hp hs n).encode x).tail ∧
73     dyadicAddressUpper ((arithmeticBlockCode p hp hs n).encode x).tail <
74       (arithmeticInterval p (sourceBlockList n x)).2 := by
75   have h := (Classical.choose_spec (exists_arithmeticBlockSupport p hp hs n)).2.1 ⟨x, hx⟩
76   simpa [arithmeticBlockCode, BinaryPrefixCode.extendZeroMass, supportTaggedWord, hx] using h

78 theorem arithmeticBlockCode_rate_sandwich {k : ℕ} (p : Fin k → ℝ)
79     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) (hn : 0 < n) :
80     discreteEntropyBaseTwo Finset.univ p ≤
81       expectedCodeLength (sourceBlockMass p n) (arithmeticBlockCode p hp hs n) / n ∧
82     expectedCodeLength (sourceBlockMass p n) (arithmeticBlockCode p hp hs n) / n ≤
83       discreteEntropyBaseTwo Finset.univ p + 3 / n := by
84   refine ⟨sourceBlock_code_rate_lower_bound p hp hs n hn _, ?_⟩
85   have hnR : (0 : ℝ) < n := by exact_mod_cast hn
86   apply (div_le_iff₀ hnR).mpr
87   have he : (discreteEntropyBaseTwo Finset.univ p + 3 / (n : ℝ)) * n =
88       n * discreteEntropyBaseTwo Finset.univ p + 3 := by field_simp
89   rw [he]
90   exact arithmeticBlockCode_expected_length_le p hp hs n

92 /-- The constructed arithmetic code's expected bits per symbol tend to entropy,
93 including sources with zero-probability symbols. -/
94 theorem arithmeticBlockCode_rate_tendsto_entropy {k : ℕ} (p : Fin k → ℝ)
95     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
96     Filter.Tendsto (fun n : ℕ => expectedCodeLength (sourceBlockMass p (n + 1))
97       (arithmeticBlockCode p hp hs (n + 1)) / (n + 1))
98       Filter.atTop (nhds (discreteEntropyBaseTwo Finset.univ p)) := by
99   have hup : Filter.Tendsto
100       (fun n : ℕ => discreteEntropyBaseTwo Finset.univ p + 3 / (n + 1 : ℝ))
101       Filter.atTop (nhds (discreteEntropyBaseTwo Finset.univ p)) := by
102     have hz := (tendsto_one_div_add_atTop_nhds_zero_nat (𝕜 := ℝ)).const_mul (3 : ℝ)
103     simpa [mul_div_assoc] using tendsto_const_nhds.add hz
104   exact tendsto_of_tendsto_of_tendsto_of_le_of_le tendsto_const_nhds hup
105     (fun n => by simpa using (arithmeticBlockCode_rate_sandwich p hp hs (n + 1) (Nat.succ_pos n)).1)
106     (fun n => by simpa using (arithmeticBlockCode_rate_sandwich p hp hs (n + 1) (Nat.succ_pos n)).2)
