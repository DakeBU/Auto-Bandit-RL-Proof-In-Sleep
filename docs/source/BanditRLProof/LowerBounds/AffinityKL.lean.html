15 theorem mul_exp_neg_half_log_eq_sqrt {r : ℝ} (hr : 0 ≤ r) :
16     r * Real.exp (-Real.log r / 2) = Real.sqrt r := by
17   rcases eq_or_lt_of_le hr with h | h
18   · simp [← h]
19   · calc
20       _ = Real.exp (Real.log r) * Real.exp (-Real.log r / 2) := by
21         rw [Real.exp_log h]
22       _ = Real.exp (Real.log r / 2) := by rw [← Real.exp_add]; congr 1; ring
23       _ = Real.sqrt r := by rw [Real.exp_half, Real.exp_log h]

48 /-- Jensen's source proof step, in the RN representation and finite-KL branch. -/
49 theorem exp_neg_half_integral_llr_le_rnAffinity
50     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
51     [IsProbabilityMeasure P] [IsProbabilityMeasure Q]
52     (hPQ : P ≪ Q) (hi : Integrable (llr P Q) P) :
53     Real.exp (-(∫ x, llr P Q x ∂P) / 2) ≤
54       ∫ x, Real.sqrt (P.rnDeriv Q x).toReal ∂Q := by
55   have h := convexOn_exp.map_integral_le Real.continuous_exp.continuousOn
56     isClosed_univ (ae_of_all P fun _ => mem_univ _)
57     (hi.neg.div_const 2) (integrable_exp_neg_half_llr P Q hPQ)
58   simp only [Pi.neg_apply] at h
59   rw [integral_div, integral_neg, integral_exp_neg_half_llr_eq P Q hPQ] at h
60   exact h

62 /-- Change the RN affinity to any common sigma-finite dominating measure. -/
63 theorem rnAffinity_eq_commonDensityAffinity
64     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
65     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
66     (hPQ : P ≪ Q) (hQ : Q ≪ μ) :
67     (∫ x, Real.sqrt (P.rnDeriv Q x).toReal ∂Q) = commonDensityAffinity P Q μ := by
68   rw [← integral_toReal_rnDeriv_mul hQ, commonDensityAffinity]
69   apply integral_congr_ae
70   filter_upwards [Measure.rnDeriv_mul_rnDeriv (κ := μ) hPQ] with x hx
71   have heq : (P.rnDeriv Q x).toReal * (Q.rnDeriv μ x).toReal =
72       (P.rnDeriv μ x).toReal := by
73     simpa only [Pi.mul_apply, ENNReal.toReal_mul] using congrArg ENNReal.toReal hx
74   rw [← heq]
75   have halg : (P.rnDeriv Q x).toReal * (Q.rnDeriv μ x).toReal *
76       (Q.rnDeriv μ x).toReal = (P.rnDeriv Q x).toReal * (Q.rnDeriv μ x).toReal ^ 2 := by ring
77   rw [halg, Real.sqrt_mul ENNReal.toReal_nonneg,
78     Real.sqrt_sq ENNReal.toReal_nonneg]
79   ring

90 /-- The squared-affinity/KL bound, with the infinite-KL branch explicit. -/
91 theorem bretagnolleHuberScale_le_half_commonDensityAffinity_sq
92     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
93     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] [SigmaFinite μ]
94     (hQ : Q ≪ μ) :
95     bretagnolleHuberScale (relativeEntropy P Q) ≤
96       (1 / 2 : ℝ) * commonDensityAffinity P Q μ ^ 2 := by
97   by_cases ht : relativeEntropy P Q = ⊤
98   · simp only [bretagnolleHuberScale, ht, ↓reduceIte]
99     positivity
100   have hr := relativeEntropy_ne_top_iff.1 ht
101   have hj := exp_neg_half_integral_llr_le_commonDensityAffinity P Q μ hr.1 hQ hr.2
102   have ha : 0 ≤ commonDensityAffinity P Q μ :=
103     integral_nonneg fun x => Real.sqrt_nonneg _
104   have hs := (sq_le_sq₀ (Real.exp_pos _).le ha).2 hj
105   have he : Real.exp (-(∫ x, llr P Q x ∂P) / 2) ^ 2 =
106       Real.exp (-(∫ x, llr P Q x ∂P)) := by
107     rw [pow_two, ← Real.exp_add]
108     congr 1
109     ring
110   have hk : (relativeEntropy P Q).toReal = ∫ x, llr P Q x ∂P := by
111     simpa only [relativeEntropy, probReal_univ, add_sub_cancel_right]
112       using InformationTheory.toReal_klDiv hr.1 hr.2
113   rw [he] at hs
114   simp only [bretagnolleHuberScale, ht, ↓reduceIte, hk]
115   linarith
