4 /-!
5 # Conditional-kernel KL integral for Chapter 15
6 
7 Mathlib's composition-product chain rule deliberately leaves the conditional
8 term as another measure-level KL divergence because measurability of
9 `x \mapsto klDiv (kappa x) (eta x)` is not automatic. For countably generated
10 target spaces, kernel Radon--Nikodym derivatives provide a measurable
11 replacement. This module proves the resulting iterated-lintegral identity.
12 
13 The result is a dependency for Lattimore--Szepesvari, *Bandit Algorithms*,
14 Lemma 15.1. It is not itself the adaptive-history divergence decomposition.
15 -/

25 /--
26 The conditional KL term for two composition products with one common left
27 measure is the iterated integral of `klFun` applied to the measurable kernel
28 Radon--Nikodym derivative.
29 
30 This is the measurable formulation of the conditional-KL integral needed by
31 the Chapter 15 history chain rule. Pointwise absolute continuity is explicit;
32 without it, individual conditional KL terms may be infinite because of
33 support mismatch.
34 -/
35 theorem klDiv_compProd_same_left_eq_lintegral_kernelRN_ae
36     {Base Target : Type*}
37     [MeasurableSpace Base] [MeasurableSpace Target]
38     [MeasurableSpace.CountablyGenerated Target]
39     (mu : Measure Base) [IsFiniteMeasure mu]
40     (kappa eta : Kernel Base Target)
41     [IsMarkovKernel kappa] [IsMarkovKernel eta]
42     (h_ac : ∀ᵐ base ∂mu, kappa base ≪ eta base) :
43     InformationTheory.klDiv (mu ⊗ₘ kappa) (mu ⊗ₘ eta) =
44       ∫⁻ x, ∫⁻ y,
45         ENNReal.ofReal
46           (InformationTheory.klFun
47             ((kappa.rnDeriv eta x y).toReal))
48           ∂eta x ∂mu := by
49   have hkappa :
50       kappa =ᵐ[mu] eta.withDensity (kappa.rnDeriv eta) := by
51     filter_upwards [h_ac] with base hbase
52     exact (Kernel.withDensity_rnDeriv_eq hbase).symm
53   have hmeasurable :
54       Measurable (Function.uncurry (kappa.rnDeriv eta)) :=
55     Kernel.measurable_rnDeriv kappa eta
56   have hjoint :
57       mu ⊗ₘ kappa =
58         (mu ⊗ₘ eta).withDensity
59           (fun pair => kappa.rnDeriv eta pair.1 pair.2) := by
60     calc
61       mu ⊗ₘ kappa =
62           mu ⊗ₘ eta.withDensity (kappa.rnDeriv eta) :=
63         Measure.compProd_congr hkappa
64       _ = (mu ⊗ₘ eta).withDensity
65           (fun pair => kappa.rnDeriv eta pair.1 pair.2) :=
66         Measure.compProd_withDensity hmeasurable
67   have hjoint_ac : mu ⊗ₘ kappa ≪ mu ⊗ₘ eta := by
68     exact Measure.AbsolutelyContinuous.compProd_right h_ac
69   rw [InformationTheory.klDiv_eq_lintegral_klFun_of_ac hjoint_ac]
70   have hrn :
71       (mu ⊗ₘ kappa).rnDeriv (mu ⊗ₘ eta) =ᵐ[mu ⊗ₘ eta]
72         fun pair => kappa.rnDeriv eta pair.1 pair.2 := by
73     rw [hjoint]
74     exact Measure.rnDeriv_withDensity _ hmeasurable
75   calc
76     ∫⁻ pair,
77         ENNReal.ofReal
78           (InformationTheory.klFun
79             (((mu ⊗ₘ kappa).rnDeriv (mu ⊗ₘ eta) pair).toReal))
80         ∂mu ⊗ₘ eta =
81         ∫⁻ pair,
82           ENNReal.ofReal
83             (InformationTheory.klFun
84               ((kappa.rnDeriv eta pair.1 pair.2).toReal))
85           ∂mu ⊗ₘ eta := by
86             refine lintegral_congr_ae ?_
87             filter_upwards [hrn] with pair hpair
88             rw [hpair]
89     _ = ∫⁻ x, ∫⁻ y,
90           ENNReal.ofReal
91             (InformationTheory.klFun
92               ((kappa.rnDeriv eta x y).toReal))
93           ∂eta x ∂mu := by
94             rw [Measure.lintegral_compProd]
95             fun_prop

99     {Base Target : Type*}
100     [MeasurableSpace Base] [MeasurableSpace Target]
101     [MeasurableSpace.CountablyGenerated Target]
102     (mu : Measure Base) [IsFiniteMeasure mu]
103     (kappa eta : Kernel Base Target)

115 /--
116 Under pointwise absolute continuity, the measurable kernel-RN formulation is
117 equal to the familiar conditional-KL integral.  This closes Mathlib's stated
118 `klDiv_compProd_eq_add` TODO for the countably generated target-space branch.
119 -/
120 theorem klDiv_compProd_same_left_eq_lintegral_klDiv
121     {Base Target : Type*}
122     [MeasurableSpace Base] [MeasurableSpace Target]
123     [MeasurableSpace.CountablyGenerated Target]
124     (mu : Measure Base) [IsFiniteMeasure mu]
125     (kappa eta : Kernel Base Target)
126     [IsMarkovKernel kappa] [IsMarkovKernel eta]
127     (h_ac : forall base, kappa base ≪ eta base) :
128     InformationTheory.klDiv (mu ⊗ₘ kappa) (mu ⊗ₘ eta) =
129       ∫⁻ base,
130         InformationTheory.klDiv (kappa base) (eta base) ∂mu := by
131   rw [klDiv_compProd_same_left_eq_lintegral_kernelRN mu kappa eta h_ac]
132   refine lintegral_congr fun base => ?_
133   rw [InformationTheory.klDiv_eq_lintegral_klFun_of_ac (h_ac base)]
134   have hmeasurable :
135       Measurable (fun target => kappa.rnDeriv eta base target) :=
136     (Kernel.measurable_rnDeriv kappa eta).comp
137       (measurable_const.prodMk measurable_id)
138   have hkappa :
139       kappa base =
140         (eta base).withDensity (fun target => kappa.rnDeriv eta base target) :=
141     by
142       rw [← Kernel.withDensity_apply eta
143         (Kernel.measurable_rnDeriv kappa eta) base]
144       exact (Kernel.withDensity_rnDeriv_eq (h_ac base)).symm
145   have hrn :
146       (kappa base).rnDeriv (eta base) =ᵐ[eta base]
147         fun target => kappa.rnDeriv eta base target := by
148     rw [hkappa]
149     exact Measure.rnDeriv_withDensity _ hmeasurable
150   refine lintegral_congr_ae ?_
151   filter_upwards [hrn] with target htarget
152   rw [htarget]

190 /--
191 If the conditional KL function is measurable, the same-left composition-product
192 identity holds without an absolute-continuity hypothesis. Singular conditional
193 fibres correctly contribute `∞` on sets of positive base measure.
194 -/
195 theorem klDiv_compProd_same_left_eq_lintegral_klDiv_of_measurable
196     {Base Target : Type*}
197     [MeasurableSpace Base] [MeasurableSpace Target]
198     [MeasurableSpace.CountablyGenerated Target]
199     (mu : Measure Base) [IsFiniteMeasure mu]
200     (kappa eta : Kernel Base Target)
201     [IsMarkovKernel kappa] [IsMarkovKernel eta]
202     (hmeasurable : Measurable (fun base =>
203       InformationTheory.klDiv (kappa base) (eta base))) :
204     InformationTheory.klDiv (mu ⊗ₘ kappa) (mu ⊗ₘ eta) =
205       ∫⁻ base,
206         InformationTheory.klDiv (kappa base) (eta base) ∂mu := by
207   by_cases hjoint : mu ⊗ₘ kappa ≪ mu ⊗ₘ eta
208   · exact klDiv_compProd_same_left_eq_lintegral_klDiv_ae
209       mu kappa eta hjoint.kernel_of_compProd
210   · have hnot_ae : ¬∀ᵐ base ∂mu, kappa base ≪ eta base := by
211       intro hae
212       exact hjoint (Measure.AbsolutelyContinuous.compProd_right hae)
213     have hintegral_top :
214         (∫⁻ base,
215           InformationTheory.klDiv (kappa base) (eta base) ∂mu) = ∞ := by
216       by_contra hne
217       have hlt := MeasureTheory.ae_lt_top hmeasurable hne
218       have hae : ∀ᵐ base ∂mu, kappa base ≪ eta base := by
219         filter_upwards [hlt] with base hbase
220         by_contra hnot_ac
221         rw [InformationTheory.klDiv_of_not_ac hnot_ac] at hbase
222         exact (not_lt_of_ge le_rfl) hbase
223       exact hnot_ae hae
224     rw [InformationTheory.klDiv_of_not_ac hjoint, hintegral_top]

226 /-- Relative entropy is invariant under a measurable equivalence. -/
227 theorem klDiv_map_measurableEquiv
228     {Source Target : Type*}
229     [MeasurableSpace Source] [MeasurableSpace Target]
230     (mu nu : Measure Source) [IsFiniteMeasure mu] [IsFiniteMeasure nu]
231     (equiv : Source ≃ᵐ Target) :
232     InformationTheory.klDiv (mu.map equiv) (nu.map equiv) =
233       InformationTheory.klDiv mu nu := by
234   by_cases h_ac : mu ≪ nu
235   · have hmap_ac : mu.map equiv ≪ nu.map equiv :=
236       equiv.measurableEmbedding.absolutelyContinuous_map h_ac
237     rw [InformationTheory.klDiv_eq_lintegral_klFun_of_ac hmap_ac,
238       InformationTheory.klDiv_eq_lintegral_klFun_of_ac h_ac]
239     calc
240       ∫⁻ target,
241           ENNReal.ofReal
242             (InformationTheory.klFun
243               (((mu.map equiv).rnDeriv (nu.map equiv) target).toReal))
244           ∂nu.map equiv =
245           ∫⁻ source,
246             ENNReal.ofReal
247               (InformationTheory.klFun
248                 (((mu.map equiv).rnDeriv (nu.map equiv)
249                   (equiv source)).toReal))
250             ∂nu :=
251         equiv.measurableEmbedding.lintegral_map _
252       _ = ∫⁻ source,
253           ENNReal.ofReal
254             (InformationTheory.klFun ((mu.rnDeriv nu source).toReal))
255           ∂nu := by
256         refine lintegral_congr_ae ?_
257         filter_upwards [equiv.measurableEmbedding.rnDeriv_map mu nu]
258           with source hsource
259         rw [hsource]
260   · have hmap_not_ac : ¬mu.map equiv ≪ nu.map equiv := by
261       intro hmap_ac
262       have hback :=
263         equiv.symm.measurableEmbedding.absolutelyContinuous_map hmap_ac
264       have hmu_back : (mu.map equiv).map equiv.symm = mu := by
265         rw [Measure.map_map equiv.symm.measurable equiv.measurable]
266         rw [show (equiv.symm : Target → Source) ∘ equiv = id from by
267           funext source
268           exact equiv.symm_apply_apply source]
269         exact Measure.map_id
270       have hnu_back : (nu.map equiv).map equiv.symm = nu := by
271         rw [Measure.map_map equiv.symm.measurable equiv.measurable]
272         rw [show (equiv.symm : Target → Source) ∘ equiv = id from by
273           funext source
274           exact equiv.symm_apply_apply source]
275         exact Measure.map_id
276       rw [hmu_back, hnu_back] at hback
277       exact h_ac hback
278     rw [InformationTheory.klDiv_of_not_ac hmap_not_ac,
279       InformationTheory.klDiv_of_not_ac h_ac]

281 /--
282 For one adaptive bandit round, a common randomized policy contributes no KL
283 cost of its own.  The conditional history-extension cost is the first-law
284 policy average of the selected arm reward-law KL divergences.
285 
286 The policy is an arbitrary Markov kernel from the complete visible history to
287 the action space; in particular, this statement is not restricted to a
288 deterministic action rule.
289 -/
290 theorem klDiv_historyStep_samePolicy_eq_iterated_lintegral_armKL
291     {History Action Reward : Type*}
292     [MeasurableSpace History]
293     [MeasurableSpace Action] [MeasurableSpace.CountablyGenerated Action]
294     [MeasurableSpace Reward] [MeasurableSpace.CountablyGenerated Reward]
295     (historyLaw : Measure History) [IsFiniteMeasure historyLaw]
296     (policy : Kernel History Action) [IsMarkovKernel policy]
297     (armLaw referenceArmLaw : Kernel Action Reward)
298     [IsMarkovKernel armLaw] [IsMarkovKernel referenceArmLaw]
299     (h_ac : forall arm, armLaw arm ≪ referenceArmLaw arm) :
300     InformationTheory.klDiv
301         (historyLaw ⊗ₘ
302           (policy ⊗ₖ armLaw.comap Prod.snd measurable_snd))
303         (historyLaw ⊗ₘ
304           (policy ⊗ₖ referenceArmLaw.comap Prod.snd measurable_snd)) =
305       ∫⁻ history, ∫⁻ arm,
306         InformationTheory.klDiv (armLaw arm) (referenceArmLaw arm)
307           ∂policy history ∂historyLaw := by
308   let feedback : Kernel (History × Action) Reward :=
309     armLaw.comap Prod.snd measurable_snd
310   let referenceFeedback : Kernel (History × Action) Reward :=
311     referenceArmLaw.comap Prod.snd measurable_snd
312   have hstep_apply (history : History) :
313       (policy ⊗ₖ feedback) history = policy history ⊗ₘ armLaw := by
314     ext event hevent
315     rw [Kernel.compProd_apply hevent, Measure.compProd_apply hevent]
316     congr with arm
317   have hreferenceStep_apply (history : History) :
318       (policy ⊗ₖ referenceFeedback) history =
319         policy history ⊗ₘ referenceArmLaw := by
320     ext event hevent
321     rw [Kernel.compProd_apply hevent, Measure.compProd_apply hevent]
322     congr with arm
323   have hstep_ac : forall history,
324       (policy ⊗ₖ feedback) history ≪
325         (policy ⊗ₖ referenceFeedback) history := by
326     intro history
327     rw [hstep_apply history, hreferenceStep_apply history]
328     exact Measure.AbsolutelyContinuous.compProd_right
329       (Filter.Eventually.of_forall h_ac)
330   change InformationTheory.klDiv
331       (historyLaw ⊗ₘ (policy ⊗ₖ feedback))
332       (historyLaw ⊗ₘ (policy ⊗ₖ referenceFeedback)) = _
333   rw [klDiv_compProd_same_left_eq_lintegral_klDiv
334     historyLaw (policy ⊗ₖ feedback) (policy ⊗ₖ referenceFeedback) hstep_ac]
335   refine lintegral_congr fun history => ?_
336   rw [hstep_apply history, hreferenceStep_apply history]
337   rw [klDiv_compProd_same_left_eq_lintegral_klDiv
338     (policy history) armLaw referenceArmLaw h_ac]

355 /-- Finite-action conditional KL written as a weighted finite sum. -/
356 theorem klDiv_finiteAction_compProd_eq_sum_mass_mul_armKL
357     {K : Nat} {Reward : Type*}
358     [MeasurableSpace Reward] [MeasurableSpace.CountablyGenerated Reward]
359     (actionLaw : Measure (Fin K)) [IsFiniteMeasure actionLaw]
360     (armLaw referenceArmLaw : Kernel (Fin K) Reward)
361     [IsMarkovKernel armLaw] [IsMarkovKernel referenceArmLaw] :
362     InformationTheory.klDiv
363         (actionLaw ⊗ₘ armLaw) (actionLaw ⊗ₘ referenceArmLaw) =
364       ∑ arm : Fin K,
365         actionLaw ({arm} : Set (Fin K)) *
366           InformationTheory.klDiv (armLaw arm) (referenceArmLaw arm) := by
367   rw [klDiv_finiteAction_compProd_eq_lintegral_armKL]
368   rw [MeasureTheory.lintegral_fintype]
369   apply Finset.sum_congr rfl
370   intro arm _harm
371   rw [mul_comm]

373 /--
374 General finite-action same-policy one-round KL identity.  Unlike the AC
375 specialization above, this theorem also covers infinite arm divergences and
376 zero-probability singular arms, with the standard `ENNReal` convention
377 `0 * ∞ = 0`.
378 -/
379 theorem klDiv_historyStep_samePolicy_eq_iterated_lintegral_armKL_general
380     {History Reward : Type*} {K : Nat}
381     [MeasurableSpace History]
382     [MeasurableSpace Reward] [MeasurableSpace.CountablyGenerated Reward]
383     (historyLaw : Measure History) [IsFiniteMeasure historyLaw]
384     (policy : Kernel History (Fin K)) [IsMarkovKernel policy]
385     (armLaw referenceArmLaw : Kernel (Fin K) Reward)
386     [IsMarkovKernel armLaw] [IsMarkovKernel referenceArmLaw] :
387     InformationTheory.klDiv
388         (historyLaw ⊗ₘ
389           (policy ⊗ₖ armLaw.comap Prod.snd measurable_snd))
390         (historyLaw ⊗ₘ
391           (policy ⊗ₖ referenceArmLaw.comap Prod.snd measurable_snd)) =
392       ∫⁻ history, ∫⁻ arm,
393         InformationTheory.klDiv (armLaw arm) (referenceArmLaw arm)
394           ∂policy history ∂historyLaw := by
395   let feedback : Kernel (History × Fin K) Reward :=
396     armLaw.comap Prod.snd measurable_snd
397   let referenceFeedback : Kernel (History × Fin K) Reward :=
398     referenceArmLaw.comap Prod.snd measurable_snd
399   have hstep_apply (history : History) :
400       (policy ⊗ₖ feedback) history = policy history ⊗ₘ armLaw := by
401     ext event hevent
402     rw [Kernel.compProd_apply hevent, Measure.compProd_apply hevent]
403     congr with arm
404   have hreferenceStep_apply (history : History) :
405       (policy ⊗ₖ referenceFeedback) history =
406         policy history ⊗ₘ referenceArmLaw := by
407     ext event hevent
408     rw [Kernel.compProd_apply hevent, Measure.compProd_apply hevent]
409     congr with arm
410   have hpointwise (history : History) :
411       InformationTheory.klDiv
412           ((policy ⊗ₖ feedback) history)
413           ((policy ⊗ₖ referenceFeedback) history) =
414         ∑ arm : Fin K,
415           policy history ({arm} : Set (Fin K)) *
416             InformationTheory.klDiv
417               (armLaw arm) (referenceArmLaw arm) := by
418     rw [hstep_apply, hreferenceStep_apply]
419     exact klDiv_finiteAction_compProd_eq_sum_mass_mul_armKL
420       (policy history) armLaw referenceArmLaw
421   have hconditionalMeasurable :
422       Measurable (fun history =>
423         InformationTheory.klDiv
424           ((policy ⊗ₖ feedback) history)
425           ((policy ⊗ₖ referenceFeedback) history)) := by
426     rw [show (fun history =>
427         InformationTheory.klDiv
428           ((policy ⊗ₖ feedback) history)
429           ((policy ⊗ₖ referenceFeedback) history)) =
430         fun history => ∑ arm : Fin K,
431           policy history ({arm} : Set (Fin K)) *
432             InformationTheory.klDiv
433               (armLaw arm) (referenceArmLaw arm) by
434       funext history
435       exact hpointwise history]
436     exact Finset.measurable_sum _ fun arm _harm =>
437       (policy.measurable_coe (MeasurableSet.singleton arm)).mul measurable_const
438   change InformationTheory.klDiv
439       (historyLaw ⊗ₘ (policy ⊗ₖ feedback))
440       (historyLaw ⊗ₘ (policy ⊗ₖ referenceFeedback)) = _
441   rw [klDiv_compProd_same_left_eq_lintegral_klDiv_of_measurable
442     historyLaw (policy ⊗ₖ feedback) (policy ⊗ₖ referenceFeedback)
443     hconditionalMeasurable]
444   refine lintegral_congr fun history => ?_
445   rw [hstep_apply, hreferenceStep_apply]
446   exact klDiv_finiteAction_compProd_eq_lintegral_armKL
447     (policy history) armLaw referenceArmLaw
