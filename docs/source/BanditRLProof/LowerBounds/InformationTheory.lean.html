10 # Information-theoretic lower-bound foundations
11 
12 This file formalizes the finite prefix-code/entropy definitions, a Kraft
13 adapter, the measure-KL and data-processing surfaces, and event testing used in
14 Part IV, Chapter 14 of Lattimore--Szepesvári, *Bandit Algorithms*. Measure-level
15 relative entropy is Mathlib's extended-real `InformationTheory.klDiv`. The
16 project-local work keeps codeword regularity, absolute continuity,
17 integrability, KL direction, Bernoulli endpoints, and the infinite-divergence
18 branch explicit. It does not claim Huffman optimality or source coding.
19 -/

31 /-- A finite-alphabet binary prefix code.  Excluding the empty codeword is
32 the regularity condition needed for concatenations of repeated messages to be
33 uniquely decodable. -/
34 structure BinaryPrefixCode (Symbol : Type*) where
35   encode : Symbol → List Bool
36   injective : Function.Injective encode
37   nonempty : ∀ symbol, encode symbol ≠ []
38   prefixFree : ∀ {left right}, encode left <+: encode right → left = right

44 /-- A prefix-free codebook with no empty codeword is uniquely decodable. -/
45 theorem uniquelyDecodable_range (code : BinaryPrefixCode Symbol) :
46     InformationTheory.UniquelyDecodable (Set.range code.encode) := by
47   intro words₁
48   induction words₁ with
49   | nil =>
50       intro words₂ hwords₁ hwords₂ hflat
51       cases words₂ with
52       | nil => rfl
53       | cons word₂ tail₂ =>
54           have hmem₂ : word₂ ∈ Set.range code.encode := hwords₂ word₂ (by simp)
55           rcases hmem₂ with ⟨symbol₂, rfl⟩
56           simp only [List.flatten_nil, List.flatten_cons] at hflat
57           have hempty_and : code.encode symbol₂ = [] ∧
58               (tail₂.map List.length).sum = 0 := by
59             simpa using congrArg List.length hflat.symm
60           exact False.elim (code.nonempty symbol₂ hempty_and.1)
61   | cons word₁ tail₁ ih =>
62       intro words₂ hwords₁ hwords₂ hflat
63       cases words₂ with
64       | nil =>
65           have hmem₁ : word₁ ∈ Set.range code.encode := hwords₁ word₁ (by simp)
66           rcases hmem₁ with ⟨symbol₁, rfl⟩
67           simp only [List.flatten_cons, List.flatten_nil] at hflat
68           have hempty_and : code.encode symbol₁ = [] ∧
69               (tail₁.map List.length).sum = 0 := by
70             simpa using congrArg List.length hflat
71           exact False.elim (code.nonempty symbol₁ hempty_and.1)
72       | cons word₂ tail₂ =>
73           have hmem₁ : word₁ ∈ Set.range code.encode := hwords₁ word₁ (by simp)
74           have hmem₂ : word₂ ∈ Set.range code.encode := hwords₂ word₂ (by simp)
75           rcases hmem₁ with ⟨symbol₁, rfl⟩
76           rcases hmem₂ with ⟨symbol₂, rfl⟩
77           simp only [List.flatten_cons] at hflat
78           have hprefix₁ : code.encode symbol₁ <+:
79               code.encode symbol₂ ++ tail₂.flatten := by
80             rw [← hflat]
81             exact List.prefix_append _ _
82           have hprefix₂ : code.encode symbol₂ <+:
83               code.encode symbol₂ ++ tail₂.flatten :=
84             List.prefix_append _ _
85           have hsymbol : symbol₁ = symbol₂ := by
86             rcases List.prefix_or_prefix_of_prefix hprefix₁ hprefix₂ with h | h
87             · exact code.prefixFree h
88             · exact (code.prefixFree h).symm
89           subst symbol₂
90           congr 1
91           apply ih
92           · intro word hword
93             exact hwords₁ word (by simp [hword])
94           · intro word hword
95             exact hwords₂ word (by simp [hword])
96           · exact List.append_cancel_left hflat

98 /-- The finite set of codewords induced by a finite source alphabet. -/
99 noncomputable def codebook [Fintype Symbol] [DecidableEq Symbol]
100     (code : BinaryPrefixCode Symbol) : Finset (List Bool) :=
101   Finset.univ.image code.encode

103 theorem coe_codebook [Fintype Symbol] [DecidableEq Symbol]
104     (code : BinaryPrefixCode Symbol) :
105     (code.codebook : Set (List Bool)) = Set.range code.encode := by
106   ext word
107   simp [codebook]

109 /-- Kraft--McMillan for a finite binary prefix code, obtained by adapting the
110 codebook to Mathlib's uniquely-decodable-code theorem. -/
111 theorem kraft_inequality [Fintype Symbol] [DecidableEq Symbol]
112     (code : BinaryPrefixCode Symbol) :
113     ∑ word ∈ code.codebook, (1 / 2 : Real) ^ word.length ≤ 1 := by
114   apply InformationTheory.kraft_mcmillan_inequality
115   rw [coe_codebook code]
116   exact code.uniquelyDecodable_range

120 /-- Natural-log entropy (nats) of a finite supported mass function, Eq. (14.3). -/
121 noncomputable def discreteEntropy (support : Finset Symbol)
122     (probability : Symbol → Real) : Real :=
123   ∑ symbol ∈ support, probability symbol * Real.log (probability symbol)⁻¹
124 
125 /-- Base-two entropy (bits) of a finite supported mass function. -/
126 noncomputable def discreteEntropyBaseTwo (support : Finset Symbol)
127     (probability : Symbol → Real) : Real :=
128   ∑ symbol ∈ support,
129     probability symbol * (Real.log (probability symbol)⁻¹ / Real.log 2)
130 
131 theorem discreteEntropyBaseTwo_eq_div_log_two
132     (support : Finset Symbol) (probability : Symbol → Real) :
133     discreteEntropyBaseTwo support probability =
134       discreteEntropy support probability / Real.log 2 := by
135   simp only [discreteEntropyBaseTwo, discreteEntropy, div_eq_mul_inv,
136     mul_assoc, Finset.sum_mul]
137 
138 theorem discreteEntropy_nonneg (support : Finset Symbol)
139     (probability : Symbol → Real)
140     (hprobability : ∀ symbol ∈ support,
141       0 ≤ probability symbol ∧ probability symbol ≤ 1) :
142     0 ≤ discreteEntropy support probability := by
143   apply Finset.sum_nonneg
144   intro symbol hsymbol
145   rcases hprobability symbol hsymbol with ⟨hzero, hone⟩
146   by_cases hp : probability symbol = 0
147   · simp [hp]
148   · exact mul_nonneg hzero
149       (Real.log_nonneg ((one_le_inv₀ (lt_of_le_of_ne hzero (Ne.symm hp))).2 hone))

151 /-- Expected binary codeword length, the objective in Eq. (14.1). -/
152 noncomputable def expectedCodeLength [Fintype Symbol]
153     (probability : Symbol → Real) (code : BinaryPrefixCode Symbol) : Real :=
154   ∑ symbol, probability symbol * (code.encode symbol).length
155 
156 theorem expectedCodeLength_nonneg [Fintype Symbol]
157     (probability : Symbol → Real) (code : BinaryPrefixCode Symbol)
158     (hprobability : ∀ symbol, 0 ≤ probability symbol) :
159     0 ≤ expectedCodeLength probability code := by
160   apply Finset.sum_nonneg
161   intro symbol _
162   exact mul_nonneg (hprobability symbol) (Nat.cast_nonneg _)

172 /-- The finite regular branch of the Radon--Nikodym representation.  The mass
173 correction vanishes when `P` and `Q` are probability measures. -/
174 theorem relativeEntropy_of_absolutelyContinuous_of_integrable
175     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
176     (hPQ : P ≪ Q) (hInt : Integrable (llr P Q) P) :
177     relativeEntropy P Q =
178       ENNReal.ofReal
179         (∫ x, llr P Q x ∂P + Q.real univ - P.real univ) :=
180   InformationTheory.klDiv_of_ac_of_integrable hPQ hInt

184 theorem relativeEntropy_of_probability_absolutelyContinuous_of_integrable

187     (hPQ : P ≪ Q) (hInt : Integrable (llr P Q) P) :

214 /-- Exercise 14.10 in its full sub-sigma-algebra form: forgetting measurable
215 sets cannot increase relative entropy.  The proof uses the Radon--Nikodym
216 conditional-expectation identity and conditional Jensen for `klFun`. -/
217 theorem relativeEntropy_trim_le
218     {α : Type*} {m m₀ : MeasurableSpace α}
219     {P Q : @Measure α m₀} [IsFiniteMeasure P] [IsFiniteMeasure Q]
220     (hm : m ≤ m₀) :
221     @relativeEntropy α m (P.trim hm) (Q.trim hm) ≤
222       @relativeEntropy α m₀ P Q := by
223   by_cases hKL : @relativeEntropy α m₀ P Q = ∞
224   · simp [hKL]
225   have hReg : P ≪ Q ∧ Integrable (llr P Q) P :=
226     relativeEntropy_ne_top_iff.mp hKL
227   have hACTrim : P.trim hm ≪ Q.trim hm := hReg.1.trim hm
228   let density : α → Real := fun x => (P.rnDeriv Q x).toReal
229   let divergence : Real → Real := InformationTheory.klFun
230   have hDensityInt : Integrable density Q := by
231     exact Measure.integrable_toReal_rnDeriv
232   have hDivergenceInt : Integrable (divergence ∘ density) Q := by
233     exact (InformationTheory.integrable_klFun_rnDeriv_iff hReg.1).2 hReg.2
234   have hDensityNonneg : ∀ᵐ x ∂Q, density x ∈ Ici (0 : Real) := by
235     exact ae_of_all Q (fun x => ENNReal.toReal_nonneg)
236   have hJensen :
237       divergence ∘ Q[density | m] ≤ᵐ[Q]
238         Q[divergence ∘ density | m] := by
239     exact InformationTheory.convexOn_klFun.map_condExp_le hm
240       (InformationTheory.continuous_klFun.lowerSemicontinuous.lowerSemicontinuousOn _)
241       hDensityNonneg isClosed_Ici hDensityInt hDivergenceInt
242   have hCondDensityNonneg : 0 ≤ᵐ[Q] Q[density | m] :=
243     condExp_nonneg (hDensityNonneg.mono fun x hx => hx)
244   have hCondDivergenceNonneg :
245       0 ≤ᵐ[Q] divergence ∘ Q[density | m] := by
246     filter_upwards [hCondDensityNonneg] with x hx
247     exact InformationTheory.klFun_nonneg hx
248   have hCondDivergenceInt :
249       Integrable (divergence ∘ Q[density | m]) Q := by
250     exact integrable_of_le_of_le (by fun_prop) hCondDivergenceNonneg hJensen
251       (integrable_zero α Real Q) integrable_condExp
252   have hRNTrim := toReal_rnDeriv_trim hm hReg.1
253   have hTrimDivergenceInt :
254       Integrable (fun x => InformationTheory.klFun
255         ((P.trim hm).rnDeriv (Q.trim hm) x).toReal) (Q.trim hm) := by
256     have hCondTrim :
257         Integrable (divergence ∘ Q[density | m]) (Q.trim hm) :=
258       hCondDivergenceInt.trim hm (by fun_prop)
259     refine hCondTrim.congr ?_
260     filter_upwards [hRNTrim] with x hx
261     simp only [Function.comp_apply, divergence]
262     rw [hx]
263   have hTrimLLRInt :
264       Integrable (@llr α m (P.trim hm) (Q.trim hm)) (P.trim hm) :=
265     (InformationTheory.integrable_klFun_rnDeriv_iff hACTrim).mp
266       hTrimDivergenceInt
267   have hTrimKL :
268       @relativeEntropy α m (P.trim hm) (Q.trim hm) ≠ ∞ :=
269     InformationTheory.klDiv_ne_top_iff.mpr ⟨hACTrim, hTrimLLRInt⟩
270   rw [← ENNReal.ofReal_toReal hTrimKL, ← ENNReal.ofReal_toReal hKL]
271   apply ENNReal.ofReal_le_ofReal
272   rw [InformationTheory.toReal_klDiv_eq_integral_klFun hACTrim,
273     InformationTheory.toReal_klDiv_eq_integral_klFun hReg.1]
274   calc
275     ∫ x, InformationTheory.klFun
276           ((P.trim hm).rnDeriv (Q.trim hm) x).toReal ∂Q.trim hm =
277         ∫ x, (divergence ∘ Q[density | m]) x ∂Q.trim hm := by
278       apply integral_congr_ae
279       filter_upwards [hRNTrim] with x hx
280       simp only [Function.comp_apply, divergence]
281       rw [hx]
282     _ = ∫ x, (divergence ∘ Q[density | m]) x ∂Q := by
283       symm
284       exact integral_trim hm (by fun_prop)
285     _ ≤ ∫ x, Q[divergence ∘ density | m] x ∂Q :=
286       integral_mono_ae hCondDivergenceInt integrable_condExp hJensen
287     _ = ∫ x, (divergence ∘ density) x ∂Q := integral_condExp hm
288     _ = ∫ x, InformationTheory.klFun (P.rnDeriv Q x).toReal ∂Q := by
289       rfl

296 /-- Restricting both laws to a measurable event preserves the original
297 Radon--Nikodym derivative almost everywhere on that event. -/
298 theorem rnDeriv_restrict_restrict
299     {α : Type*} [MeasurableSpace α] {P Q : Measure α}
300     [SigmaFinite P] [SigmaFinite Q]
301     (hPQ : P ≪ Q) {A : Set α} (hA : MeasurableSet A) :
302     (P.restrict A).rnDeriv (Q.restrict A) =ᵐ[Q.restrict A]
303       P.rnDeriv Q := by
304   have hDensity :
305       (Q.restrict A).withDensity (P.rnDeriv Q) = P.restrict A := by
306     rw [← restrict_withDensity hA,
307       Measure.withDensity_rnDeriv_eq P Q hPQ]
308   have hRN := Measure.rnDeriv_withDensity (Q.restrict A)
309     (Measure.measurable_rnDeriv P Q)
310   simpa only [hDensity] using hRN

312 /-- Relative entropy splits exactly across an event and its complement.  This
313 is the two-cell partition identity used by the event data-processing proof. -/
314 theorem relativeEntropy_restrict_add_compl
315     {α : Type*} [MeasurableSpace α] {P Q : Measure α}
316     [IsFiniteMeasure P] [IsFiniteMeasure Q]
317     (hPQ : P ≪ Q) {A : Set α} (hA : MeasurableSet A) :
318     relativeEntropy P Q =
319       relativeEntropy (P.restrict A) (Q.restrict A) +
320         relativeEntropy (P.restrict Aᶜ) (Q.restrict Aᶜ) := by
321   change InformationTheory.klDiv P Q =
322     InformationTheory.klDiv (P.restrict A) (Q.restrict A) +
323       InformationTheory.klDiv (P.restrict Aᶜ) (Q.restrict Aᶜ)
324   rw [InformationTheory.klDiv_eq_lintegral_klFun_of_ac hPQ,
325     InformationTheory.klDiv_eq_lintegral_klFun_of_ac (hPQ.restrict A),
326     InformationTheory.klDiv_eq_lintegral_klFun_of_ac (hPQ.restrict Aᶜ)]
327   have hRN := rnDeriv_restrict_restrict hPQ hA
328   have hRNc := rnDeriv_restrict_restrict hPQ hA.compl
329   let f : α → ENNReal := fun x =>
330     ENNReal.ofReal (InformationTheory.klFun (P.rnDeriv Q x).toReal)
331   change (∫⁻ x, f x ∂Q) =
332     (∫⁻ x, ENNReal.ofReal
333       (InformationTheory.klFun
334         ((P.restrict A).rnDeriv (Q.restrict A) x).toReal) ∂Q.restrict A) +
335     (∫⁻ x, ENNReal.ofReal
336       (InformationTheory.klFun
337         ((P.restrict Aᶜ).rnDeriv (Q.restrict Aᶜ) x).toReal) ∂Q.restrict Aᶜ)
338   calc
339     (∫⁻ x, f x ∂Q) =
340         ∫⁻ x, f x ∂(Q.restrict A + Q.restrict Aᶜ) := by
341       rw [Measure.restrict_add_restrict_compl hA]
342     _ = (∫⁻ x, f x ∂Q.restrict A) +
343           ∫⁻ x, f x ∂Q.restrict Aᶜ := by
344       rw [lintegral_add_measure]
345     _ = _ := by
346       congr 1
347       · apply lintegral_congr_ae
348         filter_upwards [hRN] with x hx
349         simp [f, hx]
350       · apply lintegral_congr_ae
351         filter_upwards [hRNc] with x hx
352         simp [f, hx]

354 /-- Event data processing in the finite, non-singular Bernoulli branch.  This
355 is the quantitative core of Exercise 14.10 specialized to the sigma-algebra
356 generated by one event. -/
357 theorem bernoulliKLCore_event_le
358     {α : Type*} [MeasurableSpace α] {P Q : Measure α}
359     [IsProbabilityMeasure P] [IsProbabilityMeasure Q]
360     {A : Set α} (hA : MeasurableSet A)
361     (hKL : relativeEntropy P Q ≠ ∞)
362     (hQ0 : 0 < Q.real A) (hQ1 : Q.real A < 1) :
363     KLUCB.bernoulliKLCore (P.real A) (Q.real A) ≤
364       (relativeEntropy P Q).toReal := by
365   have hReg := relativeEntropy_ne_top_iff.mp hKL
366   have hSplit := relativeEntropy_restrict_add_compl hReg.1 hA
367   have hLeftFinite :
368       relativeEntropy (P.restrict A) (Q.restrict A) ≠ ∞ := by
369     intro hTop
370     apply hKL
371     rw [hSplit, hTop, top_add]
372   have hRightFinite :
373       relativeEntropy (P.restrict Aᶜ) (Q.restrict Aᶜ) ≠ ∞ := by
374     intro hTop
375     apply hKL
376     rw [hSplit, hTop, add_top]
377   have hLeftReg := relativeEntropy_ne_top_iff.mp hLeftFinite
378   have hRightReg := relativeEntropy_ne_top_iff.mp hRightFinite
379   have hLeft := InformationTheory.mul_klFun_le_toReal_klDiv
380     hLeftReg.1 hLeftReg.2
381   have hRight := InformationTheory.mul_klFun_le_toReal_klDiv
382     hRightReg.1 hRightReg.2
383   have hLeft' :
384       Q.real A * InformationTheory.klFun (P.real A / Q.real A) ≤
385         (relativeEntropy (P.restrict A) (Q.restrict A)).toReal := by
386     simpa only [measureReal_restrict_apply_univ] using hLeft
387   have hRight' :
388       Q.real Aᶜ * InformationTheory.klFun (P.real Aᶜ / Q.real Aᶜ) ≤
389         (relativeEntropy (P.restrict Aᶜ) (Q.restrict Aᶜ)).toReal := by
390     simpa only [measureReal_restrict_apply_univ] using hRight
391   have hPCompl : P.real Aᶜ = 1 - P.real A := by
392     linarith [probReal_add_probReal_compl (μ := P) hA]
393   have hQCompl : Q.real Aᶜ = 1 - Q.real A := by
394     linarith [probReal_add_probReal_compl (μ := Q) hA]
395   rw [KLUCB.bernoulliKLCore_eq_klFun
396     (ne_of_gt hQ0) (ne_of_lt hQ1)]
397   calc
398     Q.real A * InformationTheory.klFun (P.real A / Q.real A) +
399         (1 - Q.real A) *
400           InformationTheory.klFun ((1 - P.real A) / (1 - Q.real A)) =
401       Q.real A * InformationTheory.klFun (P.real A / Q.real A) +
402         Q.real Aᶜ * InformationTheory.klFun (P.real Aᶜ / Q.real Aᶜ) := by
403           rw [hPCompl, hQCompl]
404     _ ≤ (relativeEntropy (P.restrict A) (Q.restrict A)).toReal +
405           (relativeEntropy (P.restrict Aᶜ) (Q.restrict Aᶜ)).toReal :=
406       add_le_add hLeft' hRight'
407     _ = (relativeEntropy P Q).toReal := by
408       rw [hSplit, ENNReal.toReal_add hLeftFinite hRightFinite]

410 private theorem mul_sqrt_div_eq_sqrt_mul
411     {a b : Real} (ha : 0 < a) (hb : 0 ≤ b) :
412     a * Real.sqrt (b / a) = Real.sqrt (a * b) := by
413   calc
414     a * Real.sqrt (b / a) = a * (Real.sqrt b / Real.sqrt a) := by
415       rw [Real.sqrt_div hb]
416     _ = Real.sqrt a * Real.sqrt b := by
417       calc
418         a * (Real.sqrt b / Real.sqrt a) =
419             (Real.sqrt a * Real.sqrt a) *
420               (Real.sqrt b / Real.sqrt a) := by
421           rw [Real.mul_self_sqrt ha.le]
422         _ = Real.sqrt a * Real.sqrt b := by
423           field_simp [Real.sqrt_ne_zero'.mpr ha]
424     _ = Real.sqrt (a * b) := (Real.sqrt_mul ha.le b).symm

426 /-- The binary likelihood affinity dominates `exp(-d/2)`.  This is the
427 two-atom Jensen step in the source proof of Theorem 14.2. -/
428 theorem exp_neg_half_bernoulliKLCore_le_affinity
429     {p q : Real} (hp0 : 0 < p) (hp1 : p < 1)
430     (hq0 : 0 < q) (hq1 : q < 1) :
431     Real.exp (-(KLUCB.bernoulliKLCore p q) / 2) ≤
432       Real.sqrt (p * q) + Real.sqrt ((1 - p) * (1 - q)) := by
433   let x : Real := Real.sqrt (q / p)
434   let y : Real := Real.sqrt ((1 - q) / (1 - p))
435   have hxPos : 0 < x := by
436     dsimp [x]
437     exact Real.sqrt_pos.2 (div_pos hq0 hp0)
438   have hyPos : 0 < y := by
439     dsimp [y]
440     exact Real.sqrt_pos.2 (div_pos (sub_pos.mpr hq1) (sub_pos.mpr hp1))
441   have hJensen :
442       p * Real.log x + (1 - p) * Real.log y ≤
443         Real.log (p * x + (1 - p) * y) := by
444     simpa only [smul_eq_mul] using
445       (strictConcaveOn_log_Ioi).concaveOn.2 hxPos hyPos hp0.le
446         (sub_nonneg.mpr hp1.le) (by ring : p + (1 - p) = 1)

684 /-- **Bretagnolle--Huber inequality** (Lattimore--Szepesvári, Theorem 14.2).
685 For any measurable event, the two testing errors are bounded below in the
686 source KL direction `D(P,Q)`.  The infinite-divergence case is included by
687 `bretagnolleHuberScale`. -/
688 theorem bretagnolleHuber
689     {α : Type*} [MeasurableSpace α] {P Q : Measure α}
690     [IsProbabilityMeasure P] [IsProbabilityMeasure Q]
691     {A : Set α} (hA : MeasurableSet A) :
692     bretagnolleHuberScale (relativeEntropy P Q) ≤
693       P.real A + Q.real Aᶜ := by
694   have hP : KLUCB.IsBernoulliParameter (P.real A) :=
695     ⟨measureReal_nonneg, measureReal_le_one⟩
696   have hQ : KLUCB.IsBernoulliParameter (Q.real A) :=
697     ⟨measureReal_nonneg, measureReal_le_one⟩
698   have hDPI := bernoulliRelativeEntropy_event_le (P := P) (Q := Q) hA
699   have hScale := bretagnolleHuberScale_antitone hDPI
700   have hBinary := binaryBretagnolleHuber hP hQ
701   have hQCompl : Q.real Aᶜ = 1 - Q.real A := by
702     linarith [probReal_add_probReal_compl (μ := Q) hA]
703   calc
704     bretagnolleHuberScale (relativeEntropy P Q) ≤
705         bretagnolleHuberScale
706           (bernoulliRelativeEntropy (P.real A) (Q.real A)) := hScale
707     _ ≤ P.real A + (1 - Q.real A) := hBinary
708     _ = P.real A + Q.real Aᶜ := by rw [hQCompl]
709 
710 end
