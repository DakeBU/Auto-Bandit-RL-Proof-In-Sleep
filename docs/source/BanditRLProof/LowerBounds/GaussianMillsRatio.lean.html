14 noncomputable def gaussianMillsComparison (c x : ℝ) : ℝ :=
15   Real.exp (-x ^ 2) / (x + Real.sqrt (x ^ 2 + c))

17 theorem gaussianMillsComparison_denominator_pos {c x : ℝ} (hc : 0 < c) :
18     0 < x + Real.sqrt (x ^ 2 + c) := by
19   have hp : 0 < x ^ 2 + c := by positivity
20   have hs := Real.sq_sqrt hp.le
21   have hn := Real.sqrt_nonneg (x ^ 2 + c)
22   nlinarith

24 theorem hasDerivAt_gaussianMillsComparison {c x : ℝ} (hc : 0 < c) :
25     HasDerivAt (gaussianMillsComparison c)
26       (-gaussianMillsComparison c x * (2 * x + 1 / Real.sqrt (x ^ 2 + c))) x := by
27   have hp : 0 < x ^ 2 + c := by positivity
28   have hs : Real.sqrt (x ^ 2 + c) ≠ 0 := (Real.sqrt_pos.2 hp).ne'
29   have hd := (gaussianMillsComparison_denominator_pos (x := x) hc).ne'
30   have hsq := Real.sq_sqrt hp.le
31   have h := (((hasDerivAt_id x).pow 2).neg.exp).div
32     ((hasDerivAt_id x).add ((((hasDerivAt_id x).pow 2).add_const c).sqrt hp.ne')) hd
33   convert h using 1; dsimp [gaussianMillsComparison]
34   field_simp
35   nlinarith [hsq]

37 theorem gaussianMillsComparison_lower_derivative_bound (x : ℝ) :
38     gaussianMillsComparison 2 x * (2 * x + 1 / Real.sqrt (x ^ 2 + 2)) ≤
39       Real.exp (-x ^ 2) := by
40   have hp : 0 < x ^ 2 + 2 := by positivity
41   have hs : 0 < Real.sqrt (x ^ 2 + 2) := Real.sqrt_pos.2 hp
42   have hsq := Real.sq_sqrt hp.le
43   have hd := gaussianMillsComparison_denominator_pos (x := x) (c := 2) (by norm_num)
44   have hprod : x * Real.sqrt (x ^ 2 + 2) ≤ x ^ 2 + 1 := by

46       sq_nonneg (Real.sqrt (x ^ 2 + 2) - x)]
47   unfold gaussianMillsComparison
48   apply (div_mul_eq_mul_div _ _ _).le.trans
49   rw [div_le_iff₀ hd]

51   have hi : 1 / Real.sqrt (x ^ 2 + 2) ≤ Real.sqrt (x ^ 2 + 2) - x := by
52     apply (div_le_iff₀ hs).2
53     nlinarith
54   linarith

83     -gaussianMillsComparison 2 t * (2 * t + 1 / Real.sqrt (t ^ 2 + 2))

106 /-- Algebraic sign test for the derivative of the upper comparison error. -/
107 theorem gaussianMills_sign_iff {c x : ℝ} (hc : 1 < c) (hx : 0 ≤ x) :
108     0 ≤ x ^ 2 + c - 1 - x * Real.sqrt (x ^ 2 + c) ↔
109       x ^ 2 * (2 - c) ≤ (c - 1) ^ 2 := by
110   have hp : 0 < x ^ 2 + c := by positivity
111   have hs := Real.sq_sqrt hp.le
112   have hxs : 0 ≤ x * Real.sqrt (x ^ 2 + c) := mul_nonneg hx (Real.sqrt_nonneg _)
113   have ha : 0 < x ^ 2 + c - 1 := by nlinarith [sq_nonneg x]
114   have he : (x * Real.sqrt (x ^ 2 + c)) ^ 2 = x ^ 2 * (x ^ 2 + c) := by
115     rw [mul_pow, hs]
116   constructor
117   · intro h
118     have hh := mul_self_le_mul_self hxs (show x * Real.sqrt (x ^ 2 + c) ≤ x ^ 2 + c - 1 by linarith)
119     nlinarith
120   · intro h
121     by_contra hn
122     have hh := mul_self_lt_mul_self ha.le (show x ^ 2 + c - 1 < x * Real.sqrt (x ^ 2 + c) by linarith)
123     nlinarith

125 /-- The sign changes at exactly one nonnegative threshold when `1<c<2`. -/
126 theorem gaussianMills_sign_threshold {c x : ℝ} (hc : 1 < c) (hc2 : c < 2)
127     (hx : 0 ≤ x) :
128     0 ≤ x ^ 2 + c - 1 - x * Real.sqrt (x ^ 2 + c) ↔
129       x ≤ (c - 1) / Real.sqrt (2 - c) := by
130   rw [gaussianMills_sign_iff hc hx]
131   have hs : 0 < Real.sqrt (2 - c) := Real.sqrt_pos.mpr (by linarith)
132   have he := Real.sq_sqrt (show 0 ≤ 2 - c by linarith)
133   rw [le_div_iff₀ hs]
134   have hp : 0 ≤ x * Real.sqrt (2 - c) := mul_nonneg hx hs.le
135   constructor
136   · intro h
137     nlinarith [sq_nonneg (x * Real.sqrt (2 - c) - (c - 1)),
138       show (x * Real.sqrt (2 - c)) ^ 2 = x ^ 2 * (2 - c) by rw [mul_pow, he]]
139   · intro h
140     have hh := mul_self_le_mul_self hp h
141     nlinarith [show (x * Real.sqrt (2 - c)) ^ 2 = x ^ 2 * (2 - c) by rw [mul_pow, he]]
