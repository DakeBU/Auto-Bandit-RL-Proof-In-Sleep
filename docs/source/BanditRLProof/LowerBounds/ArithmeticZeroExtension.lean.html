6 noncomputable def supportTaggedWord {α : Type*} (p : α → ℝ)
7     (positive : BinaryPrefixCode {a // 0 < p a}) (fallback : BinaryPrefixCode α)
8     (a : α) : List Bool := by
9   classical
10   exact if h : 0 < p a then false :: positive.encode ⟨a, h⟩ else true :: fallback.encode a
11 
12 theorem supportTaggedWord_prefixFree {α : Type*} (p : α → ℝ)
13     (positive : BinaryPrefixCode {a // 0 < p a}) (fallback : BinaryPrefixCode α)
14     {a b : α} (h : supportTaggedWord p positive fallback a <+:
15       supportTaggedWord p positive fallback b) : a = b := by
16   classical
17   by_cases ha : 0 < p a <;> by_cases hb : 0 < p b
18   · simp only [supportTaggedWord, dif_pos ha, dif_pos hb, List.cons_prefix_cons] at h
19     exact congrArg Subtype.val (positive.prefixFree h.2)
20   · simp [supportTaggedWord, ha, hb, List.cons_prefix_cons] at h
21   · simp [supportTaggedWord, ha, hb, List.cons_prefix_cons] at h
22   · simp only [supportTaggedWord, dif_neg ha, dif_neg hb, List.cons_prefix_cons] at h
23     exact fallback.prefixFree h.2

25 /-- Extend an arithmetic support code to all symbols, with a one-bit escape tag. -/
26 noncomputable def BinaryPrefixCode.extendZeroMass {α : Type*} (p : α → ℝ)
27     (positive : BinaryPrefixCode {a // 0 < p a}) (fallback : BinaryPrefixCode α) :
28     BinaryPrefixCode α where
29   encode := supportTaggedWord p positive fallback
30   injective := fun a b h => supportTaggedWord_prefixFree p positive fallback (by rw [h])
31   nonempty := by
32     classical
33     intro a
34     by_cases h : 0 < p a <;> simp [supportTaggedWord, h]
35   prefixFree := supportTaggedWord_prefixFree p positive fallback

37 /-- Zero-mass fallback words cost nothing in expectation. The extra support tag
38 adds only one bit to a support code with information-plus-two length bound. -/
39 theorem expectedCodeLength_extendZeroMass_le {α : Type*} [Fintype α] (p : α → ℝ)
40     (hp : ∀ a, 0 ≤ p a) (hs : ∑ a, p a = 1)
41     (positive : BinaryPrefixCode {a // 0 < p a}) (fallback : BinaryPrefixCode α)
42     (hpos : ∀ a (ha : 0 < p a), ((positive.encode ⟨a, ha⟩).length : ℝ) ≤
43       Real.log (p a)⁻¹ / Real.log 2 + 2) :
44     expectedCodeLength p (positive.extendZeroMass p fallback) ≤
45       discreteEntropyBaseTwo Finset.univ p + 3 := by
46   classical
47   have hterm (a : α) : p a * ((supportTaggedWord p positive fallback a).length : ℝ) ≤
48       p a * (Real.log (p a)⁻¹ / Real.log 2) + 3 * p a := by
49     by_cases ha : 0 < p a
50     · have h := mul_le_mul_of_nonneg_left (hpos a ha) (hp a)
51       simp only [supportTaggedWord, dif_pos ha, List.length_cons, Nat.cast_add, Nat.cast_one]
52       nlinarith
53     · have hz : p a = 0 := le_antisymm (not_lt.mp ha) (hp a)
54       simp [hz]
55   have h := Finset.sum_le_sum (s := Finset.univ) (fun a _ => hterm a)
56   simpa [expectedCodeLength, BinaryPrefixCode.extendZeroMass,
57     Finset.sum_add_distrib, ← Finset.mul_sum, hs, discreteEntropyBaseTwo] using h

59 /-- Arithmetic coding on the positive support, extended to every message.
60 Only the zero-mass escape branch uses the supplied total Huffman fallback. -/
61 theorem exists_zeroSafe_arithmeticCode {α : Type*} [Fintype α] {k : ℕ}
62     (p : Fin k → ℝ) (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
63     (message : α → List (Fin k)) (hinj : Function.Injective message)
64     (hlen : ∀ a b, (message a).length = (message b).length)
65     (q : α → ℝ) (hq : ∀ a, 0 ≤ q a) (hqs : ∑ a, q a = 1)
66     (hmass : ∀ a, ((message a).map p).prod = q a) :
67     ∃ positive : BinaryPrefixCode {a // 0 < q a},
68       (∀ a, (positive.encode a).length = arithmeticLength (q a.val)) ∧
69       (∀ a, (arithmeticInterval p (message a.val)).1 ≤
70         dyadicAddressLower (positive.encode a) ∧
71         dyadicAddressUpper (positive.encode a) < (arithmeticInterval p (message a.val)).2) ∧
72       expectedCodeLength q (positive.extendZeroMass q (huffmanCode q hq)) ≤
73         discreteEntropyBaseTwo Finset.univ q + 3 := by
74   classical
75   have hbudget (a : {a // 0 < q a}) :
76       2 * (1 / (2 : ℝ) ^ arithmeticLength (q a.val)) ≤
77         (arithmeticInterval p (message a.val)).2 - (arithmeticInterval p (message a.val)).1 := by
78     rw [arithmeticInterval_width, hmass]
79     exact (arithmeticLength_width_budget a.property).le
80   obtain ⟨positive, hl, hcell⟩ := exists_arithmeticPrefixCode p hp hs
81     (fun a : {a // 0 < q a} => message a.val)
82     (fun a b h => Subtype.ext (hinj h))
83     (fun a b => hlen a.val b.val) (fun a => arithmeticLength (q a.val)) hbudget
84   refine ⟨positive, hl, hcell,
85     expectedCodeLength_extendZeroMass_le q hq hqs positive (huffmanCode q hq) ?_⟩
86   intro a ha
87   rw [hl]
88   have hq1 : q a ≤ 1 := by
89     rw [← hqs]
90     exact Finset.single_le_sum (fun b _ => hq b) (Finset.mem_univ a)
91   exact arithmeticLength_le_information_add_two ha hq1
