7 # Lower bounds: basic ideas
8 
9 This module formalizes the semantic and deterministic interfaces developed in
10 Lattimore--Szepesvári, *Bandit Algorithms* (2020), Part IV, Chapter 13.
11 
12 The source states its finite-arm Gaussian minimax theorem in Chapter 13 but
13 defers the proof to Chapter 15. Accordingly, this module does **not** claim
14 that theorem. It provides the worst-case/minimax order surface, the
15 least-explored alternative-arm averaging leaf, and the conditional algebra
16 behind the chapter's two-environment heuristic. The cross-environment
17 information comparison remains an explicit premise for Chapters 14--15.
18 -/

25 universe u v

27 /--
28 Worst-case expected regret of one policy over an explicit environment class.
29 
30 The `ENNReal` codomain supplies the source-level supremum without a hidden
31 boundedness hypothesis. If `environmentClass` is empty, this retains the
32 standard complete-lattice value `bot`; meaningful bandit consumers should
33 prove their intended class is nonempty.
34 -/
35 noncomputable def worstCaseExpectedRegret
36     {Policy : Type u} {Environment : Type v}
37     (regret : Policy -> Environment -> ENNReal)
38     (environmentClass : Set Environment)
39     (policy : Policy) : ENNReal :=
40   ⨆ environment : environmentClass, regret policy environment.1
41 
42 /--
43 Minimax expected regret over explicit policy and environment classes.
44 
45 The definition mirrors `inf_pi sup_nu R_n(pi,nu)`. Nonemptiness of the classes
46 is intentionally a consumer-side semantic contract rather than a hidden
47 assumption of the definition.
48 -/
49 noncomputable def minimaxExpectedRegret
50     {Policy : Type u} {Environment : Type v}
51     (regret : Policy -> Environment -> ENNReal)
52     (policyClass : Set Policy)
53     (environmentClass : Set Environment) : ENNReal :=
54   ⨅ policy : policyClass,
55     worstCaseExpectedRegret regret environmentClass policy.1
56 
57 /--
58 A policy is minimax optimal for an explicit policy class, environment class,
59 and fixed-horizon regret functional when it is admissible and its worst-case
60 regret attains the minimax value.
61 
62 The horizon is carried by `regret`; keeping the two classes explicit records
63 the source warning that minimax optimality is not a property of a policy alone.
64 -/
65 def IsMinimaxOptimal
66     {Policy : Type u} {Environment : Type v}
67     (regret : Policy -> Environment -> ENNReal)
68     (policyClass : Set Policy) (environmentClass : Set Environment)
69     (policy : Policy) : Prop :=
70   policy ∈ policyClass ∧
71     worstCaseExpectedRegret regret environmentClass policy =
72       minimaxExpectedRegret regret policyClass environmentClass
73 
74 /-- A minimax-optimal policy belongs to the policy class over which the infimum is taken. -/
75 theorem IsMinimaxOptimal.mem_policyClass
76     {Policy : Type u} {Environment : Type v}
77     {regret : Policy -> Environment -> ENNReal}
78     {policyClass : Set Policy} {environmentClass : Set Environment}
79     {policy : Policy}
80     (hpolicy : IsMinimaxOptimal regret policyClass environmentClass policy) :
81     policy ∈ policyClass :=
82   hpolicy.1
83 
84 /-- A minimax-optimal policy attains the fixed-class minimax value. -/
85 theorem IsMinimaxOptimal.eq_minimaxExpectedRegret
86     {Policy : Type u} {Environment : Type v}
87     {regret : Policy -> Environment -> ENNReal}
88     {policyClass : Set Policy} {environmentClass : Set Environment}
89     {policy : Policy}
90     (hpolicy : IsMinimaxOptimal regret policyClass environmentClass policy) :
91     worstCaseExpectedRegret regret environmentClass policy =
92       minimaxExpectedRegret regret policyClass environmentClass :=
93   hpolicy.2
94 
95 /-- One environment's regret is below the worst case over any class containing it. -/
96 theorem expectedRegret_le_worstCaseExpectedRegret
97     {Policy : Type u} {Environment : Type v}
98     (regret : Policy -> Environment -> ENNReal)
99     (environmentClass : Set Environment)
100     (policy : Policy) (environment : Environment)
101     (henvironment : environment ∈ environmentClass) :
102     regret policy environment ≤
103       worstCaseExpectedRegret regret environmentClass policy := by
104   unfold worstCaseExpectedRegret
105   exact le_iSup (fun member : environmentClass => regret policy member.1)
106     ⟨environment, henvironment⟩
107 
108 /-- The minimax value is below the worst-case value of each admissible policy. -/
109 theorem minimaxExpectedRegret_le_worstCaseExpectedRegret
110     {Policy : Type u} {Environment : Type v}
111     (regret : Policy -> Environment -> ENNReal)
112     (policyClass : Set Policy) (environmentClass : Set Environment)
113     (policy : Policy) (hpolicy : policy ∈ policyClass) :
114     minimaxExpectedRegret regret policyClass environmentClass ≤
115       worstCaseExpectedRegret regret environmentClass policy := by
116   unfold minimaxExpectedRegret
117   exact iInf_le_of_le ⟨policy, hpolicy⟩ le_rfl
118 
119 /-- A uniform lower bound on every admissible policy is a minimax lower bound. -/
120 theorem le_minimaxExpectedRegret
121     {Policy : Type u} {Environment : Type v}
122     (regret : Policy -> Environment -> ENNReal)
123     (policyClass : Set Policy) (environmentClass : Set Environment)
124     (lower : ENNReal)
125     (hlower : ∀ policy : policyClass,
126       lower ≤ worstCaseExpectedRegret regret environmentClass policy.1) :
127     lower ≤ minimaxExpectedRegret regret policyClass environmentClass := by
128   unfold minimaxExpectedRegret
129   exact le_iInf hlower

131 /--
132 Finite averaging: among a nonempty family whose sum is at most `budget`, one
133 coordinate is at most `budget / m`.
134 
135 This is a Mathlib-composed deterministic leaf. It contains no bandit law,
136 expectation, measurability, or concentration assumption.
137 -/
138 theorem exists_alternative_le_average
139     {m : Nat} (hm : 0 < m)
140     (alternativeExpectedPulls : Fin m -> Real)
141     (budget : Real)
142     (hbudget : ∑ i : Fin m, alternativeExpectedPulls i ≤ budget) :
143     ∃ i : Fin m,
144       alternativeExpectedPulls i ≤ budget / (m : Real) := by
145   haveI : Nonempty (Fin m) := Fintype.card_pos_iff.mp (by simpa using hm)
146   have huniv : (Finset.univ : Finset (Fin m)).Nonempty := Finset.univ_nonempty
147   have hmReal : (m : Real) ≠ 0 := Nat.cast_ne_zero.mpr hm.ne'
148   have hconstant :
149       (∑ _i : Fin m, budget / (m : Real)) = budget := by
150     rw [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]
151     exact mul_div_cancel₀ budget hmReal
152   have hcompare :
153       (∑ i : Fin m, alternativeExpectedPulls i) ≤
154         ∑ _i : Fin m, budget / (m : Real) := by
155     simpa [hconstant] using hbudget
156   obtain ⟨i, _hi, hile⟩ :=
157     Finset.exists_le_of_sum_le huniv hcompare
158   exact ⟨i, hile⟩
159 
160 /--
161 Removing the distinguished arm zero from an exact nonnegative pull budget
162 leaves at most the full budget on the `Fin.succ` alternative arms.
163 -/
164 theorem alternativeExpectedPullBudget_le
165     {m : Nat}
166     (expectedPulls : Fin (m + 1) -> Real)
167     (budget : Real)
168     (hnonneg : ∀ arm, 0 ≤ expectedPulls arm)
169     (htotal : ∑ arm : Fin (m + 1), expectedPulls arm = budget) :
170     (∑ i : Fin m, expectedPulls i.succ) ≤ budget := by
171   rw [Fin.sum_univ_succ] at htotal
172   nlinarith [hnonneg (0 : Fin (m + 1))]
173 
174 /--
175 Chapter 13's least-explored alternative arm.
176 
177 There are `m + 1` arms: arm zero is the base arm and `i.succ`, for `i : Fin m`,
178 are the alternatives. The hypotheses are precisely the expected pull-count
179 nonnegativity and total-budget identity needed by the averaging argument.
180 -/
181 theorem exists_leastExploredAlternative
182     {m : Nat} (hm : 0 < m)
183     (expectedPulls : Fin (m + 1) -> Real)
184     (horizon : Nat)
185     (hnonneg : ∀ arm, 0 ≤ expectedPulls arm)
186     (htotal : ∑ arm : Fin (m + 1), expectedPulls arm = (horizon : Real)) :
187     ∃ i : Fin m,
188       expectedPulls i.succ ≤ (horizon : Real) / (m : Real) := by
189   exact exists_alternative_le_average hm
190     (fun i : Fin m => expectedPulls i.succ) (horizon : Real)
191     (alternativeExpectedPullBudget_le expectedPulls (horizon : Real) hnonneg htotal)

193 /-- The exact deterministic expression in the base-environment identity (13.2). -/
194 def baseEnvironmentRegret
195     (horizon : Nat) (gap baseFirstExpectedPulls : Real) : Real :=
196   gap * ((horizon : Real) - baseFirstExpectedPulls)
197 
198 /-- The changed-environment regret lower expression on the right of (13.3). -/
199 def changedEnvironmentRegretLowerBound
200     (gap changedFirstExpectedPulls : Real) : Real :=
201   gap * changedFirstExpectedPulls
202 
203 /--
204 Quantitative algebraic core of the two-environment heuristic.
205 
206 The premise bounds the cross-environment discrepancy in the expected number
207 of base-arm pulls. Chapter 13 writes these expectations as approximately
208 equal; later information-theoretic chapters must supply an actual value of
209 `error`. This theorem neither derives that premise nor proves Theorem 13.1.
210 -/
211 theorem max_base_changed_regretLowerBound_ge_half_sub_error
212     (horizon : Nat)
213     (gap baseFirstExpectedPulls changedFirstExpectedPulls error : Real)
214     (hgap : 0 ≤ gap)
215     (hpullDifference : baseFirstExpectedPulls - changedFirstExpectedPulls ≤ error) :
216     gap * ((horizon : Real) - error) / 2 ≤
217       max
218         (baseEnvironmentRegret horizon gap baseFirstExpectedPulls)
219         (changedEnvironmentRegretLowerBound gap changedFirstExpectedPulls) := by
220   have hscaled := mul_le_mul_of_nonneg_left hpullDifference hgap
221   have hsum :
222       gap * ((horizon : Real) - error) ≤
223         baseEnvironmentRegret horizon gap baseFirstExpectedPulls +
224           changedEnvironmentRegretLowerBound gap changedFirstExpectedPulls := by
225     unfold baseEnvironmentRegret changedEnvironmentRegretLowerBound
226     nlinarith
227   have hbase := le_max_left
228     (baseEnvironmentRegret horizon gap baseFirstExpectedPulls)
229     (changedEnvironmentRegretLowerBound gap changedFirstExpectedPulls)
230   have hchanged := le_max_right
231     (baseEnvironmentRegret horizon gap baseFirstExpectedPulls)
232     (changedEnvironmentRegretLowerBound gap changedFirstExpectedPulls)
233   nlinarith
234 
235 /--
236 Zero-error corollary of the quantitative two-environment algebra.
237 
238 The premise `baseFirstExpectedPulls ≤ changedFirstExpectedPulls` specializes
239 the pull discrepancy to at most zero. The quantitative predecessor is the
240 intended interface for later information theorems; neither declaration is the
241 Gaussian minimax lower bound.
242 -/
243 theorem max_base_changed_regretLowerBound_ge_half
244     (horizon : Nat) (gap baseFirstExpectedPulls changedFirstExpectedPulls : Real)
245     (hgap : 0 ≤ gap)
246     (htransport : baseFirstExpectedPulls ≤ changedFirstExpectedPulls) :
247     gap * (horizon : Real) / 2 ≤
248       max
249         (baseEnvironmentRegret horizon gap baseFirstExpectedPulls)
250         (changedEnvironmentRegretLowerBound gap changedFirstExpectedPulls) := by
251   have h := max_base_changed_regretLowerBound_ge_half_sub_error
252     horizon gap baseFirstExpectedPulls changedFirstExpectedPulls 0 hgap
253     (sub_nonpos.mpr htransport)
254   simpa using h
