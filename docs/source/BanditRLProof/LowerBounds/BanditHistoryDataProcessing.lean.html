5 # Data processing for Chapter 15 bandit histories
6 
7 This module proves the measurable-observation data-processing leaf needed by
8 the stopping-time extension in Lattimore--Szepesvari, *Bandit Algorithms*,
9 Exercise 15.7.  It also specializes the leaf to the compiled deterministic
10 finite-history divergence decomposition of Lemma 15.1.
11 
12 The final stopping-time theorem needs an additional stopped-history law whose
13 KL cost contains only pulls through the stopping time.  The results below do
14 not assume or claim that missing identity.
15 -/

27 /--
28 Kullback--Leibler divergence cannot increase under a measurable observation.
29 
30 The proof reuses Chapter 14's map/trim identity and sub-sigma-algebra
31 KL contraction. The infinite-KL branch is explicit, so absolute continuity
32 is derived only in the finite branch and is not a caller assumption.
33 -/
34 theorem klDiv_map_le
35     {Source Target : Type*}
36     [MeasurableSpace Source] [MeasurableSpace Target]
37     (mu nu : Measure Source) [IsFiniteMeasure mu] [IsFiniteMeasure nu]
38     (observe : Source -> Target) (hobserve : Measurable observe) :
39     InformationTheory.klDiv (mu.map observe) (nu.map observe) <=
40       InformationTheory.klDiv mu nu := by
41   by_cases hsource : InformationTheory.klDiv mu nu = ∞
42   · rw [hsource]
43     exact le_top
44   have h_ac := (InformationTheory.klDiv_ne_top_iff.mp hsource).1
45   change relativeEntropy (mu.map observe) (nu.map observe) ≤ relativeEntropy mu nu
46   rw [relativeEntropy_map_eq_trim_of_absolutelyContinuous mu nu h_ac observe hobserve]
47   exact relativeEntropy_trim_le hobserve.comap_le
48 
49 /--
50 Any measurable statistic of a deterministic finite bandit history has KL at
51 most the first-law expected pull-count-weighted arm information.
52 
53 This is the data-processing half of Exercise 15.7 at a fixed horizon.  It is
54 not the stopping-time theorem because the right-hand side still counts all
55 pulls through `lastRound`.
56 -/
57 theorem klDiv_observedBanditHistory_le_expectedPulls_sum
58     {K : Nat} {Reward : Type v} {Observation : Type w}
59     [MeasurableSpace Reward] [MeasurableSpace.CountablyGenerated Reward]
60     [MeasurableSpace Observation]
61     (algorithm : Thompson.HistoryAlgorithm (Fin K) Reward)
62     (armLaw referenceArmLaw : Kernel (Fin K) Reward)
63     [IsMarkovKernel armLaw] [IsMarkovKernel referenceArmLaw]
64     (lastRound : Nat)
65     (observe : History.FinitePairHistory (Fin K) Reward lastRound -> Observation)
66     (hobserve : Measurable observe) :
67     InformationTheory.klDiv
68         ((canonicalBanditHistoryMeasure algorithm armLaw lastRound).map observe)
69         ((canonicalBanditHistoryMeasure algorithm referenceArmLaw lastRound).map observe) <=
70       ∑ arm : Fin K,
71         canonicalRealizedExpectedPullCountThrough
72             algorithm armLaw lastRound arm *
73           InformationTheory.klDiv (armLaw arm) (referenceArmLaw arm) := by
74   calc
75     InformationTheory.klDiv
76         ((canonicalBanditHistoryMeasure algorithm armLaw lastRound).map observe)
77         ((canonicalBanditHistoryMeasure algorithm referenceArmLaw lastRound).map observe) <=
78         InformationTheory.klDiv
79           (canonicalBanditHistoryMeasure algorithm armLaw lastRound)
80           (canonicalBanditHistoryMeasure algorithm referenceArmLaw lastRound) :=
81       klDiv_map_le _ _ observe hobserve
82     _ = _ := banditHistoryRelativeEntropy_eq_expectedPulls_sum
83       algorithm armLaw referenceArmLaw lastRound
