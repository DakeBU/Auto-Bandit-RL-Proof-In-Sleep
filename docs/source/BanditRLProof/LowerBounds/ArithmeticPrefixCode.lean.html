5 theorem arithmeticAddress_prefix_forces_eq {k : ℕ} (p : Fin k → ℝ)
6     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
7     (u v : List (Fin k)) (hlen : u.length = v.length) (cu cv : List Bool)
8     (huL : (arithmeticInterval p u).1 ≤ dyadicAddressLower cu)
9     (huU : dyadicAddressUpper cu < (arithmeticInterval p u).2)
10     (hvL : (arithmeticInterval p v).1 ≤ dyadicAddressLower cv)
11     (hvU : dyadicAddressUpper cv < (arithmeticInterval p v).2)
12     (hprefix : cu <+: cv) : u = v := by
13   have hc := dyadicAddress_prefix_contained cu cv hprefix
14   have hn := dyadicAddress_nonempty cv
15   apply arithmeticInterval_interior_unique p hp hs u v hlen
16     ((dyadicAddressLower cv + dyadicAddressUpper cv) / 2)
17   · constructor <;> linarith
18   · constructor <;> linarith

20 /-- Assemble arithmetic-cell addresses into a genuine prefix code, preserving
21 the supplied bit lengths. The width budgets are discharged by later allocation. -/
22 theorem exists_arithmeticPrefixCode {α : Type*} {k : ℕ} (p : Fin k → ℝ)
23     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
24     (message : α → List (Fin k)) (hinj : Function.Injective message)
25     (hlen : ∀ a b, (message a).length = (message b).length) (bits : α → ℕ)
26     (hbudget : ∀ a, 2 * (1 / (2 : ℝ) ^ bits a) ≤
27       (arithmeticInterval p (message a)).2 - (arithmeticInterval p (message a)).1) :
28     ∃ code : BinaryPrefixCode α,
29       (∀ a, (code.encode a).length = bits a) ∧
30       ∀ a, (arithmeticInterval p (message a)).1 ≤ dyadicAddressLower (code.encode a) ∧
31         dyadicAddressUpper (code.encode a) < (arithmeticInterval p (message a)).2 := by
32   classical
33   have hex (a : α) := exists_dyadicAddress_inside
34     (arithmeticInterval p (message a)).1 (arithmeticInterval p (message a)).2 (bits a)
35     (arithmeticInterval_bounds p hp hs (message a)).1
36     (arithmeticInterval_bounds p hp hs (message a)).2.2 (hbudget a)
37   choose c hc hcL hcU using hex
38   have hfree {a b : α} (h : c a <+: c b) : a = b := by
39     apply hinj
40     exact arithmeticAddress_prefix_forces_eq p hp hs (message a) (message b) (hlen a b)
41       (c a) (c b) (hcL a) (hcU a) (hcL b) (hcU b) h
42   have hnonempty (a : α) : c a ≠ [] := by
43     intro he
44     have hu := hcU a
45     have hb := (arithmeticInterval_bounds p hp hs (message a)).2.2
46     simp [he, dyadicAddressUpper, binaryAddressValue] at hu
47     linarith
48   let code : BinaryPrefixCode α := {
49     encode := c
50     injective := fun a b h => hfree (by rw [h])
51     nonempty := hnonempty
52     prefixFree := hfree }
53   exact ⟨code, hc, fun a => ⟨hcL a, hcU a⟩⟩

55 /-- One extra bit beyond strict Shannon length fits a dyadic cell inside the
56 arithmetic interval, rather than merely meeting a Kraft budget. -/
57 noncomputable def arithmeticLength (mass : ℝ) : ℕ := shannonLength mass + 1

59 theorem arithmeticLength_width_budget {mass : ℝ} (hm : 0 < mass) :
60     2 * (1 / (2 : ℝ) ^ arithmeticLength mass) < mass := by
61   have h := shannonLength_kraft_weight_lt hm
62   have he : 2 * (1 / (2 : ℝ) ^ arithmeticLength mass) = (1 / 2 : ℝ) ^ shannonLength mass := by
63     simp only [arithmeticLength, pow_succ, div_pow, one_pow]
64     field_simp
65   rw [he]
66   exact h
