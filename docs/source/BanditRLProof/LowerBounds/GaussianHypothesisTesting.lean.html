8 # Gaussian two-point testing for Chapter 13
9 
10 This module formalizes the distribution-level threshold test used at the start
11 of Lattimore--Szepesvári, Chapter 13.1.  The source observes that a sample mean
12 from `n` independent unit-variance Gaussian observations has variance `1 / n`.
13 We expose that Gaussian mean-observation law, identify the two error events,
14 and prove the standard Chernoff upper bound `exp (-n * gap^2 / 8)` for both
15 hypotheses and therefore for their maximum error probability.
16 
17 The separate Mills-ratio route now proves both exact bounds in Eq. (13.1),
18 including the printed square-root denominator constants. The earlier
19 Chernoff theorem remains a weaker companion.
20 -/

30 /-- Variance `1 / n` of the mean of `n` unit-variance observations. -/
31 def gaussianSampleMeanVariance (sampleSize : Nat) : NNReal :=
32   ⟨(sampleSize : Real)⁻¹, inv_nonneg.mpr (Nat.cast_nonneg sampleSize)⟩
33 
34 /-- The source's sample-mean variance is nondegenerate for a positive sample size. -/
35 theorem gaussianSampleMeanVariance_pos
36     (sampleSize : Nat) (hsampleSize : 0 < sampleSize) :
37     0 < gaussianSampleMeanVariance sampleSize := by
38   change 0 < (sampleSize : Real)⁻¹
39   exact inv_pos.mpr (Nat.cast_pos.mpr hsampleSize)
40 
41 /-- The Gaussian law stated in Chapter 13.1 for the sample mean observation. -/
42 noncomputable def gaussianSampleMeanLaw
43     (sampleSize : Nat) (mean : Real) : Measure Real :=
44   gaussianReal mean (gaussianSampleMeanVariance sampleSize)
45 
46 /-- Canonical joint law of `n` independent `N(mean, 1)` observations. -/
47 noncomputable def gaussianIIDObservationLaw
48     (sampleSize : Nat) (mean : Real) : Measure (Fin sampleSize → Real) :=
49   Measure.pi (fun _ => gaussianReal mean 1)
50 
51 /-- Arithmetic mean of a finite coordinate family. -/
52 def gaussianCoordinateAverage (sampleSize : Nat)
53     (observations : Fin sampleSize → Real) : Real :=
54   (∑ i, observations i) / (sampleSize : Real)
55 
56 /-- The sum of the canonical iid observations has mean `n * mean` and variance `n`. -/
57 theorem gaussianIIDSumLaw (sampleSize : Nat) (mean : Real) :
58     (gaussianIIDObservationLaw sampleSize mean).map
59         (fun observations => ∑ i, observations i) =
60       gaussianReal ((sampleSize : Real) * mean) (sampleSize : NNReal) := by
61   unfold gaussianIIDObservationLaw
62   apply Measure.ext_of_charFun
63   funext t
64   rw [charFun_map_sum_pi_eq_prod]
65   simp only [Finset.prod_const, Finset.card_univ, Fintype.card_fin]
66   change (charFun (gaussianReal mean 1) t) ^ sampleSize = _
67   rw [charFun_gaussianReal, charFun_gaussianReal]
68   rw [← Complex.exp_nat_mul]
69   congr 1
70   push_cast
71   ring
72 
73 /--
74 The arithmetic mean under the canonical product law of `n > 0` independent
75 `N(mean, 1)` observations has exactly the source law `N(mean, 1 / n)`.
76 -/
77 theorem gaussianIIDSampleMeanLaw
78     (sampleSize : Nat) (mean : Real) (hsampleSize : 0 < sampleSize) :
79     (gaussianIIDObservationLaw sampleSize mean).map
80         (gaussianCoordinateAverage sampleSize) =
81       gaussianSampleMeanLaw sampleSize mean := by
82   have hcast : (sampleSize : Real) ≠ 0 := Nat.cast_ne_zero.mpr hsampleSize.ne'
83   rw [show gaussianCoordinateAverage sampleSize =
84       (fun total : Real => total / (sampleSize : Real)) ∘
85         (fun observations => ∑ i, observations i) by rfl]
86   rw [← Measure.map_map (by fun_prop) (by fun_prop)]
87   rw [gaussianIIDSumLaw, gaussianReal_map_div_const]
88   unfold gaussianSampleMeanLaw gaussianSampleMeanVariance
89   congr
90   · field_simp
91   · ext
92     simp only [NNReal.coe_div, NNReal.coe_natCast, NNReal.coe_mk]
93     field_simp

95 /--
96 Threshold rule for the two hypotheses `mean = 0` and `mean = gap`.
97 Ties are assigned to `gap`, matching the source's zero-mean error event
98 `sampleMean >= gap / 2`; for a nondegenerate Gaussian law, the tie has mass
99 zero.
100 -/
101 def twoPointGaussianThresholdDecision (gap observation : Real) : Real :=
102   if gap / 2 ≤ observation then gap else 0
103 
104 /-- Under the zero-mean hypothesis, the threshold rule errs exactly above the midpoint. -/
105 theorem twoPointGaussianThresholdDecision_zero_error_event
106     {gap : Real} (hgap : 0 < gap) :
107     {observation | twoPointGaussianThresholdDecision gap observation ≠ 0} =
108       Set.Ici (gap / 2) := by
109   ext observation
110   by_cases hthreshold : gap / 2 ≤ observation
111   · simp [twoPointGaussianThresholdDecision, hthreshold, hgap.ne']
112   · simp [twoPointGaussianThresholdDecision, hthreshold]
113 
114 /-- Under the positive-mean hypothesis, errors occur exactly below the midpoint. -/
115 theorem twoPointGaussianThresholdDecision_gap_error_event
116     {gap : Real} (hgap : 0 < gap) :
117     {observation | twoPointGaussianThresholdDecision gap observation ≠ gap} =
118       Set.Iio (gap / 2) := by
119   ext observation
120   by_cases hthreshold : gap / 2 ≤ observation
121   · simp [twoPointGaussianThresholdDecision, hthreshold]
122   · simp only [twoPointGaussianThresholdDecision, if_neg hthreshold,
123       Set.mem_setOf_eq, Set.mem_Iio]
124     constructor
125     · intro _
126       exact lt_of_not_ge hthreshold
127     · intro _
128       exact ne_of_lt hgap
129 
130 /-- Error probability of the threshold decision under the zero-mean sample-mean law. -/
131 noncomputable def gaussianSampleMeanZeroErrorProbability
132     (sampleSize : Nat) (gap : Real) : Real :=
133   (gaussianSampleMeanLaw sampleSize 0).real
134     {observation | twoPointGaussianThresholdDecision gap observation ≠ 0}
135 
136 /-- Error probability of the threshold decision under the positive-mean sample-mean law. -/
137 noncomputable def gaussianSampleMeanGapErrorProbability
138     (sampleSize : Nat) (gap : Real) : Real :=
139   (gaussianSampleMeanLaw sampleSize gap).real
140     {observation | twoPointGaussianThresholdDecision gap observation ≠ gap}

142 /-- A centered real Gaussian has its variance as a sub-Gaussian proxy. -/
143 theorem hasSubgaussianMGF_id_gaussianReal_zero (variance : NNReal) :
144     HasSubgaussianMGF id variance (gaussianReal 0 variance) := by
145   constructor
146   · intro t
147     simpa using (integrable_exp_mul_gaussianReal (μ := 0) (v := variance) t)
148   · intro t
149     rw [mgf_id_gaussianReal]
150     simp

152 /-- Reflection around the mean turns `N(gap, variance)` into a centered sub-Gaussian. -/
153 theorem hasSubgaussianMGF_gap_sub_id_gaussianReal
154     (gap : Real) (variance : NNReal) :
155     HasSubgaussianMGF (fun observation => gap - observation) variance
156       (gaussianReal gap variance) := by
157   rw [← HasSubgaussianMGF.id_map_iff (by fun_prop)]
158   rw [gaussianReal_map_const_sub]
159   simpa using hasSubgaussianMGF_id_gaussianReal_zero variance

161 /-- Chernoff upper bound for a centered Gaussian right tail. -/
162 theorem gaussianReal_zero_Ici_le_exp_neg_sq_div_two_variance
163     (variance : NNReal) (threshold : Real) (hthreshold : 0 ≤ threshold) :
164     (gaussianReal 0 variance).real (Set.Ici threshold) ≤
165       Real.exp (-threshold ^ 2 / (2 * (variance : Real))) := by
166   simpa only [Set.mem_Ici, id_eq] using
167     (hasSubgaussianMGF_id_gaussianReal_zero variance).measure_ge_le hthreshold
168 
169 /-- The positive-mean midpoint error has the same Chernoff exponent by reflection. -/
170 theorem gaussianReal_gap_Iio_half_le_exp_neg_sq_div_two_variance
171     (gap : Real) (variance : NNReal) (hgap : 0 < gap) :
172     (gaussianReal gap variance).real (Set.Iio (gap / 2)) ≤
173       Real.exp (-(gap / 2) ^ 2 / (2 * (variance : Real))) := by
174   have hsubset :
175       Set.Iio (gap / 2) ⊆
176         {observation | gap / 2 ≤ gap - observation} := by
177     intro observation hobservation
178     simp only [Set.mem_Iio] at hobservation
179     simp only [Set.mem_setOf_eq]
180     linarith
181   calc
182     (gaussianReal gap variance).real (Set.Iio (gap / 2)) ≤
183         (gaussianReal gap variance).real
184           {observation | gap / 2 ≤ gap - observation} :=
185       measureReal_mono hsubset
186     _ ≤ Real.exp (-(gap / 2) ^ 2 / (2 * (variance : Real))) :=
187       (hasSubgaussianMGF_gap_sub_id_gaussianReal gap variance).measure_ge_le
188         (by positivity)

190 /--
191 For the Chapter 13.1 zero-mean branch, the midpoint threshold has error at
192 most `exp (-n * gap^2 / 8)` under the stated `N(0, 1/n)` sample-mean law.
193 -/
194 theorem gaussianSampleMeanZeroErrorProbability_le_exp
195     (sampleSize : Nat) (gap : Real) (hgap : 0 < gap) :
196     gaussianSampleMeanZeroErrorProbability sampleSize gap ≤
197       Real.exp (-(sampleSize : Real) * gap ^ 2 / 8) := by
198   rw [gaussianSampleMeanZeroErrorProbability,
199     twoPointGaussianThresholdDecision_zero_error_event hgap]
200   unfold gaussianSampleMeanLaw gaussianSampleMeanVariance
201   calc
202     (gaussianReal 0
203         ⟨(sampleSize : Real)⁻¹, inv_nonneg.mpr (Nat.cast_nonneg sampleSize)⟩).real
204         (Set.Ici (gap / 2)) ≤
205         Real.exp (-(gap / 2) ^ 2 / (2 * (sampleSize : Real)⁻¹)) := by
206           simpa only [NNReal.coe_mk] using
207             gaussianReal_zero_Ici_le_exp_neg_sq_div_two_variance
208               ⟨(sampleSize : Real)⁻¹, inv_nonneg.mpr (Nat.cast_nonneg sampleSize)⟩
209               (gap / 2) (by positivity)
210     _ = Real.exp (-(sampleSize : Real) * gap ^ 2 / 8) := by
211       congr 1
212       rw [div_eq_mul_inv, mul_inv_rev, inv_inv]
213       ring
214 
215 /--
216 For the positive-mean branch, the same midpoint rule has the identical
217 `exp (-n * gap^2 / 8)` Chernoff upper bound.
218 -/
219 theorem gaussianSampleMeanGapErrorProbability_le_exp
220     (sampleSize : Nat) (gap : Real) (hgap : 0 < gap) :
221     gaussianSampleMeanGapErrorProbability sampleSize gap ≤
222       Real.exp (-(sampleSize : Real) * gap ^ 2 / 8) := by
223   rw [gaussianSampleMeanGapErrorProbability,
224     twoPointGaussianThresholdDecision_gap_error_event hgap]
225   unfold gaussianSampleMeanLaw gaussianSampleMeanVariance
226   calc
227     (gaussianReal gap
228         ⟨(sampleSize : Real)⁻¹, inv_nonneg.mpr (Nat.cast_nonneg sampleSize)⟩).real
229         (Set.Iio (gap / 2)) ≤
230         Real.exp (-(gap / 2) ^ 2 / (2 * (sampleSize : Real)⁻¹)) := by
231           simpa only [NNReal.coe_mk] using
232             gaussianReal_gap_Iio_half_le_exp_neg_sq_div_two_variance gap
233               ⟨(sampleSize : Real)⁻¹, inv_nonneg.mpr (Nat.cast_nonneg sampleSize)⟩ hgap
234     _ = Real.exp (-(sampleSize : Real) * gap ^ 2 / 8) := by
235       congr 1
236       rw [div_eq_mul_inv, mul_inv_rev, inv_inv]
237       ring

239 /-- Worst of the two midpoint-decision error probabilities. -/
240 noncomputable def gaussianSampleMeanThresholdRisk
241     (sampleSize : Nat) (gap : Real) : Real :=
242   max (gaussianSampleMeanZeroErrorProbability sampleSize gap)
243     (gaussianSampleMeanGapErrorProbability sampleSize gap)
244 
245 /-- Both Gaussian hypotheses obey the same source-shaped Chernoff exponent. -/
246 theorem gaussianSampleMeanThresholdRisk_le_exp
247     (sampleSize : Nat) (gap : Real) (hgap : 0 < gap) :
248     gaussianSampleMeanThresholdRisk sampleSize gap ≤
249       Real.exp (-(sampleSize : Real) * gap ^ 2 / 8) := by
250   exact max_le
251     (gaussianSampleMeanZeroErrorProbability_le_exp sampleSize gap hgap)
252     (gaussianSampleMeanGapErrorProbability_le_exp sampleSize gap hgap)

254 /-- Exact Mills bounds for the source error probability, in standardized coordinates. -/
255 theorem gaussianSampleMeanZeroErrorProbability_mills_bounds
256     (sampleSize : Nat) (hsampleSize : 0 < sampleSize)
257     (gap : Real) (hgap : 0 < gap) :
258     let z := (gap / 2) / Real.sqrt (2 * (gaussianSampleMeanVariance sampleSize : Real))
259     Real.exp (-z ^ 2) / (z + Real.sqrt (z ^ 2 + 2)) / Real.sqrt Real.pi ≤
260         gaussianSampleMeanZeroErrorProbability sampleSize gap ∧
261       gaussianSampleMeanZeroErrorProbability sampleSize gap ≤
262         Real.exp (-z ^ 2) / (z + Real.sqrt (z ^ 2 + 4 / Real.pi)) / Real.sqrt Real.pi := by
263   dsimp only
264   rw [gaussianSampleMeanZeroErrorProbability,
265     twoPointGaussianThresholdDecision_zero_error_event hgap]
266   exact gaussianReal_zero_mills_bounds (gaussianSampleMeanVariance sampleSize)
267     (gaussianSampleMeanVariance_pos sampleSize hsampleSize) (gap / 2) (by positivity)

269 /-- The exact printed two-sided Gaussian testing inequality, Eq. (13.1). -/
270 theorem gaussianSampleMeanZeroErrorProbability_source_bounds
271     (sampleSize : Nat) (hsampleSize : 0 < sampleSize)
272     (gap : Real) (hgap : 0 < gap) :
273     let q := (sampleSize : Real) * gap ^ 2
274     Real.sqrt (8 / Real.pi) * Real.exp (-q / 8) /
275         (Real.sqrt q + Real.sqrt (q + 16)) ≤
276       gaussianSampleMeanZeroErrorProbability sampleSize gap ∧
277     gaussianSampleMeanZeroErrorProbability sampleSize gap ≤
278       Real.sqrt (8 / Real.pi) * Real.exp (-q / 8) /
279         (Real.sqrt q + Real.sqrt (q + 32 / Real.pi)) := by
280   let z := (gap / 2) / Real.sqrt (2 * (gaussianSampleMeanVariance sampleSize : Real))
281   have hz : 0 ≤ z := by dsimp [z]; positivity
282   have hn : (sampleSize : Real) ≠ 0 := Nat.cast_ne_zero.mpr hsampleSize.ne'
283   have hq : (sampleSize : Real) * gap ^ 2 = 8 * z ^ 2 := by
284     dsimp [z, gaussianSampleMeanVariance]
285     rw [div_pow, Real.sq_sqrt (by positivity)]
286     field_simp
287     ring
288   have hl := gaussianMills_expression_rescale hz (show (0 : Real) < 2 by norm_num) hq
289   have hu := gaussianMills_expression_rescale hz (show (0 : Real) < 4 / Real.pi by positivity) hq
290   have hb := gaussianSampleMeanZeroErrorProbability_mills_bounds sampleSize hsampleSize gap hgap
291   change _ ≤ gaussianSampleMeanZeroErrorProbability sampleSize gap ∧
292     gaussianSampleMeanZeroErrorProbability sampleSize gap ≤ _ at hb
293   rw [hl, hu] at hb
294   simpa only [show (8 : Real) * 2 = 16 by norm_num,
295     show (8 : Real) * (4 / Real.pi) = 32 / Real.pi by ring] using hb
