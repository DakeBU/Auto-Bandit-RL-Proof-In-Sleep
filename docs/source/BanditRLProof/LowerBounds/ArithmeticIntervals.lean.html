5 /-- Left endpoint of a symbol's arithmetic-coding partition cell. -/
6 def arithmeticOffset {k : ℕ} (p : Fin k → ℝ) (a : Fin k) : ℝ :=
7   ∑ b ∈ Finset.univ.filter (fun b => b < a), p b
8 
9 theorem arithmeticOffset_nonneg {k : ℕ} (p : Fin k → ℝ) (hp : ∀ i, 0 ≤ p i)
10     (a : Fin k) : 0 ≤ arithmeticOffset p a :=
11   Finset.sum_nonneg (fun i _ => hp i)
12 
13 theorem arithmeticOffset_add_le_one {k : ℕ} (p : Fin k → ℝ)
14     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (a : Fin k) :
15     arithmeticOffset p a + p a ≤ 1 := by
16   let s := Finset.univ.filter (fun b => b < a)
17   have ha : a ∉ s := by simp [s]
18   have he : arithmeticOffset p a + p a = ∑ b ∈ insert a s, p b := by
19     rw [Finset.sum_insert ha]
20     exact add_comm _ _
21   rw [he, ← hs]
22   exact Finset.sum_le_sum_of_subset_of_nonneg (Finset.subset_univ _) (fun i _ _ => hp i)

24 /-- Successive affine subdivisions for an arithmetic-coded word. -/
25 def arithmeticInterval {k : ℕ} (p : Fin k → ℝ) : List (Fin k) → ℝ × ℝ
26   | [] => (0, 1)
27   | a :: w => (arithmeticOffset p a + p a * (arithmeticInterval p w).1,
28       arithmeticOffset p a + p a * (arithmeticInterval p w).2)
29 
30 theorem arithmeticInterval_width {k : ℕ} (p : Fin k → ℝ) (w : List (Fin k)) :
31     (arithmeticInterval p w).2 - (arithmeticInterval p w).1 = (w.map p).prod := by
32   induction w with
33   | nil => simp [arithmeticInterval]
34   | cons a w ih =>
35     simp only [arithmeticInterval, List.map_cons, List.prod_cons]
36     rw [← ih]
37     ring

39 theorem arithmeticInterval_bounds {k : ℕ} (p : Fin k → ℝ)
40     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (w : List (Fin k)) :
41     0 ≤ (arithmeticInterval p w).1 ∧
42       (arithmeticInterval p w).1 ≤ (arithmeticInterval p w).2 ∧
43       (arithmeticInterval p w).2 ≤ 1 := by
44   induction w with
45   | nil => simp [arithmeticInterval]
46   | cons a w ih =>
47     have ho := arithmeticOffset_nonneg p hp a
48     have hu := arithmeticOffset_add_le_one p hp hs a
49     simp only [arithmeticInterval]
50     constructor
51     · exact add_nonneg ho (mul_nonneg (hp a) ih.1)
52     constructor
53     · exact add_le_add (le_refl _) (mul_le_mul_of_nonneg_left ih.2.1 (hp a))
54     · have h := mul_le_mul_of_nonneg_left ih.2.2 (hp a)
55       nlinarith

57 theorem arithmeticOffset_separated {k : ℕ} (p : Fin k → ℝ)
58     (hp : ∀ i, 0 ≤ p i) (a b : Fin k) (hab : a < b) :
59     arithmeticOffset p a + p a ≤ arithmeticOffset p b := by
60   let s := Finset.univ.filter (fun i => i < a)
61   have ha : a ∉ s := by simp [s]
62   have he : arithmeticOffset p a + p a = ∑ i ∈ insert a s, p i := by
63     rw [Finset.sum_insert ha]
64     exact add_comm _ _
65   rw [he]
66   apply Finset.sum_le_sum_of_subset_of_nonneg
67   · intro i hi
68     rcases Finset.mem_insert.mp hi with rfl | hi
69     · simp [hab]
70     · exact Finset.mem_filter.mpr ⟨Finset.mem_univ i,
71         lt_trans (Finset.mem_filter.mp hi).2 hab⟩
72   · intro i _ _
73     exact hp i

75 theorem arithmeticInterval_head_separated {k : ℕ} (p : Fin k → ℝ)
76     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
77     (a b : Fin k) (u v : List (Fin k)) (hab : a < b) :
78     (arithmeticInterval p (a :: u)).2 ≤ (arithmeticInterval p (b :: v)).1 := by
79   have hsep := arithmeticOffset_separated p hp a b hab
80   have hu := (arithmeticInterval_bounds p hp hs u).2.2
81   have hv := (arithmeticInterval_bounds p hp hs v).1
82   have hmul := mul_le_mul_of_nonneg_left hu (hp a)
83   have hnonneg := mul_nonneg (hp b) hv
84   simp only [arithmeticInterval]
85   nlinarith

87 /-- Different equal-length messages occupy cells with disjoint interiors. -/
88 theorem arithmeticInterval_separated {k : ℕ} (p : Fin k → ℝ)
89     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
90     (u v : List (Fin k)) (hlen : u.length = v.length) (hne : u ≠ v) :
91     (arithmeticInterval p u).2 ≤ (arithmeticInterval p v).1 ∨
92       (arithmeticInterval p v).2 ≤ (arithmeticInterval p u).1 := by
93   induction u generalizing v with
94   | nil =>
95     have hv : v = [] := List.length_eq_zero_iff.mp hlen.symm
96     exact (hne hv.symm).elim
97   | cons a u ih =>
98     cases v with
99     | nil => simp at hlen
100     | cons b v =>
101       by_cases hab : a = b
102       · subst b
103         have ht : u ≠ v := by intro he; exact hne (congrArg (List.cons a) he)
104         have hl : u.length = v.length := by simpa using hlen
105         rcases ih v hl ht with huv | hvu
106         · exact Or.inl (add_le_add (le_refl _) (mul_le_mul_of_nonneg_left huv (hp a)))
107         · exact Or.inr (add_le_add (le_refl _) (mul_le_mul_of_nonneg_left hvu (hp a)))
108       · rcases lt_or_gt_of_ne hab with hab | hba
109         · exact Or.inl (arithmeticInterval_head_separated p hp hs a b u v hab)
110         · exact Or.inr (arithmeticInterval_head_separated p hp hs b a v u hba)

112 /-- Any point strictly inside a message cell identifies that message uniquely
113 among messages of the same length. -/
114 theorem arithmeticInterval_interior_unique {k : ℕ} (p : Fin k → ℝ)
115     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
116     (u v : List (Fin k)) (hlen : u.length = v.length) (x : ℝ)
117     (hu : (arithmeticInterval p u).1 < x ∧ x < (arithmeticInterval p u).2)
118     (hv : (arithmeticInterval p v).1 < x ∧ x < (arithmeticInterval p v).2) : u = v := by
119   by_contra hne
120   rcases arithmeticInterval_separated p hp hs u v hlen hne with h | h <;> linarith

122 /-- A positive grid cell fits in any nonnegative interval at least twice as wide.
123 Mathlib-candidate scalar rounding leaf for arithmetic-code dyadic selection. -/
124 theorem exists_grid_cell_inside (L U δ : ℝ) (hL : 0 ≤ L) (hδ : 0 < δ)
125     (hwidth : 2 * δ ≤ U - L) :
126     ∃ m : ℕ, L ≤ (m : ℝ) * δ ∧ ((m : ℝ) + 1) * δ < U := by
127   refine ⟨⌈L / δ⌉₊, ?_, ?_⟩
128   · have h := mul_le_mul_of_nonneg_right (Nat.le_ceil (L / δ)) hδ.le
129     simpa [ne_of_gt hδ] using h
130   · have h := mul_lt_mul_of_pos_right (Nat.ceil_lt_add_one (div_nonneg hL hδ.le)) hδ
131     have he : (L / δ + 1) * δ = L + δ := by
132       rw [add_mul, div_mul_cancel₀ _ (ne_of_gt hδ), one_mul]
133     rw [he] at h
134     nlinarith
