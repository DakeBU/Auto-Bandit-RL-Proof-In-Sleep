13 /-- Trimmed KL as the convex lower integral of the conditional RN density. -/
14 theorem relativeEntropy_trim_eq_lintegral_condExp
15     {α : Type*} {m m₀ : MeasurableSpace α} (P Q : @Measure α m₀)
16     [IsFiniteMeasure P] [IsFiniteMeasure Q] (hm : m ≤ m₀) (h : P ≪ Q) :
17     @relativeEntropy α m (P.trim hm) (Q.trim hm) =
18       ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun
19         (Q[fun y => (P.rnDeriv Q y).toReal | m] x)) ∂Q := by
20   calc
21     _ = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun
22         (((P.trim hm).rnDeriv (Q.trim hm)) x).toReal) ∂Q.trim hm :=
23       InformationTheory.klDiv_eq_lintegral_klFun_of_ac (h.trim hm)
24     _ = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun
25         (Q[fun y => (P.rnDeriv Q y).toReal | m] x)) ∂Q.trim hm := by
26       apply lintegral_congr_ae
27       filter_upwards [toReal_rnDeriv_trim hm h] with x hx
28       rw [hx]
29     _ = _ := lintegral_trim hm (by fun_prop)

31 /-- A measurable observation has the same KL as restriction to its generated sigma-algebra. -/
32 theorem relativeEntropy_map_eq_trim_of_absolutelyContinuous
33     {α β : Type*} [mα : MeasurableSpace α] [mβ : MeasurableSpace β]
34     (P Q : Measure α) [IsFiniteMeasure P] [IsFiniteMeasure Q]
35     (h : P ≪ Q) (f : α → β) (hf : Measurable f) :
36     relativeEntropy (P.map f) (Q.map f) =
37       @relativeEntropy α (mβ.comap f) (P.trim hf.comap_le) (Q.trim hf.comap_le) := by
38   calc
39     _ = ∫⁻ y, ENNReal.ofReal (InformationTheory.klFun
40         (((P.map f).rnDeriv (Q.map f)) y).toReal) ∂Q.map f :=
41       InformationTheory.klDiv_eq_lintegral_klFun_of_ac (h.map hf)
42     _ = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun
43         (((P.map f).rnDeriv (Q.map f)) (f x)).toReal) ∂Q :=
44       lintegral_map (by fun_prop) hf
45     _ = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun
46         (Q[fun y => (P.rnDeriv Q y).toReal | mβ.comap f] x)) ∂Q := by
47       apply lintegral_congr_ae
48       filter_upwards [toReal_rnDeriv_map h hf] with x hx
49       rw [hx]
50     _ = _ := (relativeEntropy_trim_eq_lintegral_condExp P Q hf.comap_le h).symm

52 /-- Resolving the RN density along a filtration recovers KL, including infinite KL. -/
53 theorem relativeEntropy_eq_iSup_trim_of_density_measurable
54     {α : Type*} [m₀ : MeasurableSpace α] (P Q : Measure α)
55     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q)
56     (F : Filtration ℕ m₀)
57     (hDensity : StronglyMeasurable[⨆ n, F n] (fun x => (P.rnDeriv Q x).toReal)) :
58     relativeEntropy P Q =
59       ⨆ n, @relativeEntropy α (F n) (P.trim (F.le n)) (Q.trim (F.le n)) := by
60   let d : α → ℝ := fun x => (P.rnDeriv Q x).toReal
61   let g : ℕ → α → ENNReal := fun n x =>
62     ENNReal.ofReal (InformationTheory.klFun (Q[d | F n] x))
63   have hd : Integrable d Q := Measure.integrable_toReal_rnDeriv
64   have hc := hd.tendsto_ae_condExp hDensity
65   have hlim : ∀ᵐ x ∂Q, Tendsto (fun n => g n x) atTop
66       (𝓝 (ENNReal.ofReal (InformationTheory.klFun (d x)))) := by
67     filter_upwards [hc] with x hx
68     exact ENNReal.continuous_ofReal.continuousAt.tendsto.comp
69       (InformationTheory.continuous_klFun.continuousAt.tendsto.comp hx)
70   have hg (n : ℕ) : Measurable (g n) := by
71     have hm : Measurable (Q[d | F n]) :=
72       (stronglyMeasurable_condExp.mono (F.le n)).measurable
73     exact ENNReal.continuous_ofReal.measurable.comp
74       (InformationTheory.continuous_klFun.measurable.comp hm)
75   have hterm (n : ℕ) : (∫⁻ x, g n x ∂Q) =
76       @relativeEntropy α (F n) (P.trim (F.le n)) (Q.trim (F.le n)) :=
77     (relativeEntropy_trim_eq_lintegral_condExp P Q (F.le n) h).symm
78   apply le_antisymm
79   · calc
80       relativeEntropy P Q = ∫⁻ x, ENNReal.ofReal (InformationTheory.klFun (d x)) ∂Q :=
81         InformationTheory.klDiv_eq_lintegral_klFun_of_ac h
82       _ = ∫⁻ x, liminf (fun n => g n x) atTop ∂Q := by
83         apply lintegral_congr_ae
84         filter_upwards [hlim] with x hx
85         exact hx.liminf_eq.symm
86       _ ≤ liminf (fun n => ∫⁻ x, g n x ∂Q) atTop := lintegral_liminf_le hg
87       _ ≤ _ := liminf_le_of_frequently_le' (Frequently.of_forall fun n =>
88         (hterm n).le.trans (le_iSup (fun k =>
89           @relativeEntropy α (F k) (P.trim (F.le k)) (Q.trim (F.le k))) n))
90   · exact iSup_le fun n => relativeEntropy_trim_le (F.le n)

92 /-- Natural filtration of the finite-valued lower approximations to a density. -/
93 def densityApproximationFiltration {α : Type*} [m : MeasurableSpace α]
94     (r : α → ENNReal) : Filtration ℕ m :=
95   Filtration.natural (fun n => (SimpleFunc.eapprox r n : α → ENNReal))
96     (fun n => (SimpleFunc.eapprox r n).measurable.stronglyMeasurable)
97 
98 /-- The limiting sigma-algebra of the approximation filtration resolves the density. -/
99 theorem measurable_density_iSup_approximationFiltration
100     {α : Type*} [m : MeasurableSpace α] (r : α → ENNReal) (hr : Measurable r) :
101     Measurable[⨆ n, densityApproximationFiltration r n] r := by
102   have hm (n : ℕ) :
103       StronglyMeasurable[densityApproximationFiltration r n]
104         (SimpleFunc.eapprox r n : α → ENNReal) :=
105     Filtration.stronglyAdapted_natural
106       (fun k => (SimpleFunc.eapprox r k).measurable.stronglyMeasurable) n
107   have hall (n : ℕ) :
108       Measurable[⨆ k, densityApproximationFiltration r k]
109         (SimpleFunc.eapprox r n : α → ENNReal) :=
110     ((hm n).mono (le_iSup _ n)).measurable
111   have hs := Measurable.iSup hall
112   simpa only [SimpleFunc.iSup_eapprox_apply hr] using hs

114 /-- RN KL is recovered along a concretely constructed simple-approximation filtration. -/
115 theorem relativeEntropy_eq_iSup_densityApproximation_trim
116     {α : Type*} [m : MeasurableSpace α] (P Q : Measure α)
117     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q) :
118     relativeEntropy P Q = ⨆ n,
119       @relativeEntropy α (densityApproximationFiltration (P.rnDeriv Q) n)
120         (P.trim ((densityApproximationFiltration (P.rnDeriv Q)).le n))
121         (Q.trim ((densityApproximationFiltration (P.rnDeriv Q)).le n)) := by
122   apply relativeEntropy_eq_iSup_trim_of_density_measurable P Q h
123   exact (measurable_density_iSup_approximationFiltration (P.rnDeriv Q)
124     (Measure.measurable_rnDeriv P Q)).ennreal_toReal.stronglyMeasurable
