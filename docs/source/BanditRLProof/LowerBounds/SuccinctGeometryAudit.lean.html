7 # Succinct-support geometry audit
8 
9 This module formalizes the first geometric layer of Zeng--Honorio (NeurIPS
10 2025), Definitions 3.1--3.3 and Lemmas 3.1--3.4.  It keeps the atom set
11 possibly infinite and makes every boundedness premise for real `sSup`
12 explicit.
13 
14 The final diagnostic records a source boundary rather than a source theorem:
15 if the atoms do not span the ambient inner-product space, the candidate set
16 defining the paper's globally real-valued `R` can be unbounded.  Later source
17 lemmas that consume global `R` therefore remain outside this module until a
18 spanning, extended-real, or span/quotient repair is chosen explicitly.  The
19 local Lemma-3.2 identities suffice for Lemmas 3.3--3.4 and do not assume such a
20 repair.
21 -/

34     [InnerProductSpace ℝ V] where
35   atoms : Set V
36   atoms_nonempty : atoms.Nonempty
37   norm_eq_one : ∀ e ∈ atoms, ‖e‖ = 1
38   neg_mem : ∀ e ∈ atoms, -e ∈ atoms

44 /-- The source quantity `Q(X)=sup_{E in U} <X,E>`. -/
45 def sourceQ (system : SuccinctUnitSystem V) (x : V) : ℝ :=
46   sSup ((fun e : V => ⟪x, e⟫_ℝ) '' system.atoms)

48 theorem sourceQSet_bddAbove (system : SuccinctUnitSystem V) (x : V) :
49     BddAbove ((fun e : V => ⟪x, e⟫_ℝ) '' system.atoms) := by
50   refine ⟨‖x‖, ?_⟩
51   rintro value ⟨e, he, rfl⟩
52   calc
53     ⟪x, e⟫_ℝ ≤ |⟪x, e⟫_ℝ| := le_abs_self _
54     _ ≤ ‖x‖ * ‖e‖ := abs_real_inner_le_norm x e
55     _ = ‖x‖ := by rw [system.norm_eq_one e he, mul_one]

57 theorem le_sourceQ_of_mem (system : SuccinctUnitSystem V)
58     {x e : V} (he : e ∈ system.atoms) :
59     ⟪x, e⟫_ℝ ≤ system.sourceQ x := by
60   exact le_csSup (system.sourceQSet_bddAbove x) ⟨e, he, rfl⟩

62 theorem sourceQ_le_norm (system : SuccinctUnitSystem V) (x : V) :
63     system.sourceQ x ≤ ‖x‖ := by
64   apply csSup_le
65   · exact system.atoms_nonempty.image _
66   · rintro value ⟨e, he, rfl⟩
67     calc
68       ⟪x, e⟫_ℝ ≤ |⟪x, e⟫_ℝ| := le_abs_self _
69       _ ≤ ‖x‖ * ‖e‖ := abs_real_inner_le_norm x e
70       _ = ‖x‖ := by rw [system.norm_eq_one e he, mul_one]

72 theorem sourceQ_nonneg (system : SuccinctUnitSystem V) (x : V) :

113 /-- If a nonzero ambient direction is orthogonal to every atom, the set used
114 to define its source `R` is unbounded.  This exposes the hidden global
115 regularity/codomain obligation in Definition 3.2. -/
116 theorem sourceRSet_not_bddAbove_of_nonzero_atom_orthogonal
117     (system : SuccinctUnitSystem V) {x : V} (hx : x ≠ 0)
118     (horthogonal : ∀ e ∈ system.atoms, ⟪x, e⟫_ℝ = 0) :
119     ¬ BddAbove ((fun y : V => ⟪x, y⟫_ℝ) '' {y | system.sourceQ y ≤ 1}) := by
120   intro hbounded
121   rcases hbounded with ⟨bound, hbound⟩
122   have hself : 0 < ⟪x, x⟫_ℝ := real_inner_self_pos.mpr hx
123   let scale : ℝ := (bound + 1) / ⟪x, x⟫_ℝ
124   let y : V := scale • x
125   have hqy : system.sourceQ y = 0 := by
126     apply system.sourceQ_eq_zero_of_atom_orthogonal
127     intro e he
128     simp [y, inner_smul_left, horthogonal e he]
129   have hy : y ∈ {z | system.sourceQ z ≤ 1} := by
130     simp [hqy]
131   have hvalue : ⟪x, y⟫_ℝ ≤ bound := hbound ⟨y, hy, rfl⟩
132   have hinner : ⟪x, y⟫_ℝ = bound + 1 := by
133     change ⟪x, scale • x⟫_ℝ = bound + 1
134     rw [inner_smul_right]
135     dsimp [scale]
136     field_simp
137   linarith

163 /-- The mutual-orthogonality consequence stated after source Definition 3.1. -/
164 theorem inner_basis_basis (support : IsSuccinctSupport system basis)
165     (i j : Fin s) :
166     ⟪basis i, basis j⟫_ℝ = if i = j then 1 else 0 := by
167   classical
168   by_cases hij : i = j
169   · subst j
170     rw [if_pos rfl, real_inner_self_eq_norm_sq,
171       system.norm_eq_one (basis i) (support.mem_atoms i)]
172     norm_num
173   · rw [if_neg hij]
174     have hsum := support.correlationSum_le_one (support.mem_atoms i)
175     have hdiag : |⟪basis i, basis i⟫_ℝ| = 1 := by
176       rw [real_inner_self_eq_norm_sq,
177         system.norm_eq_one (basis i) (support.mem_atoms i)]
178       norm_num
179     have herase_nonneg :
180         0 ≤ ∑ k ∈ (Finset.univ : Finset (Fin s)).erase i,
181           |⟪basis i, basis k⟫_ℝ| := by positivity
182     have herase_add := Finset.sum_erase_add (Finset.univ : Finset (Fin s))
183       (fun k => |⟪basis i, basis k⟫_ℝ|) (Finset.mem_univ i)
184     have herase_zero :
185         (∑ k ∈ (Finset.univ : Finset (Fin s)).erase i,
186           |⟪basis i, basis k⟫_ℝ|) = 0 := by
187       change
188         (∑ k ∈ (Finset.univ : Finset (Fin s)).erase i,
189           |⟪basis i, basis k⟫_ℝ|) + |⟪basis i, basis i⟫_ℝ| =
190             ∑ k, |⟪basis i, basis k⟫_ℝ| at herase_add
191       rw [hdiag] at herase_add
192       linarith
193     have hjmem : j ∈ (Finset.univ : Finset (Fin s)).erase i := by
194       simp [Ne.symm hij]
195     have hjzero : |⟪basis i, basis j⟫_ℝ| = 0 :=
196       (Finset.sum_eq_zero_iff_of_nonneg
197         (fun k _hk => abs_nonneg ⟪basis i, basis k⟫_ℝ)).mp herase_zero j hjmem
198     exact abs_eq_zero.mp hjzero

240 theorem sourceQ_supportCombination_le [Nonempty (Fin s)]
241     (support : IsSuccinctSupport system basis) (a : Fin s → ℝ) :
242     system.sourceQ (supportCombination basis a) ≤ maxAbsCoefficient a := by

532 theorem abs_inner_strictBasis_supportSignCombination_eq_one
533     (first : SuccinctRepresentation system x s)
534     (second : StrictSuccinctRepresentation system x z) (j : Fin z) :
535     |⟪second.toSuccinctRepresentation.basis j,

546         |secondRep.coefficients i| := by

552   have hweighted_le :

595 /-- Source Lemma 3.3 on explicit witnesses: if the same vector has an
596 `s`-succinct representation and a strictly `z`-succinct representation, then
597 `z <= s`. -/
598 theorem strictSize_le (first : SuccinctRepresentation system x s)
599     (second : StrictSuccinctRepresentation system x z) : z ≤ s := by
600   let secondRep := second.toSuccinctRepresentation
601   let y := IsSuccinctSupport.supportSignCombination first.basis first.coefficients
602   have hbessel := secondRep.support.orthonormal.sum_inner_products_le
603     (s := (Finset.univ : Finset (Fin z))) y
604   have hunit (i : Fin z) : |⟪secondRep.basis i, y⟫_ℝ| = 1 := by
605     change |⟪secondRep.basis i,
606       IsSuccinctSupport.supportSignCombination first.basis first.coefficients⟫_ℝ| = 1
607     exact first.abs_inner_strictBasis_supportSignCombination_eq_one second i
608   have hcast : (z : ℝ) ≤ (s : ℝ) := by
609     calc
610       (z : ℝ) =
611           ∑ i : Fin z, ‖⟪secondRep.basis i, y⟫_ℝ‖ ^ 2 := by
612         symm
613         calc
614           (∑ i : Fin z, ‖⟪secondRep.basis i, y⟫_ℝ‖ ^ 2) =
615               ∑ i : Fin z, |⟪secondRep.basis i, y⟫_ℝ| ^ 2 := by
616             simp [Real.norm_eq_abs]
617           _ = ∑ _i : Fin z, (1 : ℝ) ^ 2 := by

619             intro i _hi
