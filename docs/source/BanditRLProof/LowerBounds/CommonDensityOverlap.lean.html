13 /-- Square roots of nonnegative integrable functions belong to L2. -/
14 theorem memLp_sqrt_of_integrable_nonneg
15     {α : Type*} [MeasurableSpace α] {μ : Measure α} {f : α → ℝ}
16     (hf : Integrable f μ) (hpos : ∀ x, 0 ≤ f x) :
17     MemLp (fun x => Real.sqrt (f x)) 2 μ := by
18   apply (memLp_two_iff_integrable_sq
19     (Real.continuous_sqrt.comp_aestronglyMeasurable hf.aestronglyMeasurable)).2
20   have heq : (fun x => Real.sqrt (f x) ^ 2) = f :=
21     funext fun x => Real.sq_sqrt (hpos x)
22   rwa [heq]

24 /-- Cauchy--Schwarz for the square-root affinity integrand. -/
25 theorem integral_sqrt_mul_sq_le
26     {α : Type*} [MeasurableSpace α] {μ : Measure α} {p q : α → ℝ}
27     (hp : Integrable p μ) (hq : Integrable q μ)
28     (hp0 : ∀ x, 0 ≤ p x) (hq0 : ∀ x, 0 ≤ q x) :
29     (∫ x, Real.sqrt (p x * q x) ∂μ) ^ 2 ≤ (∫ x, p x ∂μ) * ∫ x, q x ∂μ := by
30   have hp2 := memLp_sqrt_of_integrable_nonneg hp hp0
31   have hq2 := memLp_sqrt_of_integrable_nonneg hq hq0
32   have hcs := integral_mul_le_Lp_mul_Lq_of_nonneg Real.HolderConjugate.two_two
33     (ae_of_all μ fun x => Real.sqrt_nonneg (p x))
34     (ae_of_all μ fun x => Real.sqrt_nonneg (q x))
35     (by simpa using hp2) (by simpa using hq2)
36   have hpPow (x) : Real.sqrt (p x) ^ (2 : ℝ) = p x := by
37     rw [Real.rpow_two, Real.sq_sqrt (hp0 x)]
38   have hqPow (x) : Real.sqrt (q x) ^ (2 : ℝ) = q x := by
39     rw [Real.rpow_two, Real.sq_sqrt (hq0 x)]
40   simp_rw [hpPow, hqPow, ← Real.sqrt_eq_rpow] at hcs
41   have heq : (fun x => Real.sqrt (p x) * Real.sqrt (q x)) =
42       (fun x => Real.sqrt (p x * q x)) :=
43     funext fun x => (Real.sqrt_mul (hp0 x) (q x)).symm
44   rw [heq] at hcs
45   have hs := (sq_le_sq₀ (integral_nonneg fun x => Real.sqrt_nonneg (p x * q x))
46     (mul_nonneg (Real.sqrt_nonneg _) (Real.sqrt_nonneg _))).2 hcs
47   simpa only [mul_pow, Real.sq_sqrt (integral_nonneg hp0),
48     Real.sq_sqrt (integral_nonneg hq0)] using hs

74 /-- Eq. (14.9), including integrable densities with zeros. -/
75 theorem half_commonDensityAffinity_sq_le_overlap
76     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
77     [IsProbabilityMeasure P] [IsProbabilityMeasure Q] [SigmaFinite μ]
78     (hP : P ≪ μ) (hQ : Q ≪ μ) :
79     (1 / 2 : ℝ) * commonDensityAffinity P Q μ ^ 2 ≤
80       commonDensityOverlap P Q μ := by
81   let p := fun x => (P.rnDeriv μ x).toReal
82   let q := fun x => (Q.rnDeriv μ x).toReal
83   have hp : Integrable p μ := Measure.integrable_toReal_rnDeriv
84   have hq : Integrable q μ := Measure.integrable_toReal_rnDeriv
85   have hp0 (x) : 0 ≤ p x := ENNReal.toReal_nonneg
86   have hq0 (x) : 0 ≤ q x := ENNReal.toReal_nonneg
87   have hcs := integral_sqrt_mul_sq_le (hp.inf hq) (hp.sup hq)
88     (fun x => le_min (hp0 x) (hq0 x))
89     (fun x => le_trans (hp0 x) (le_max_left _ _))
90   simp only [Pi.inf_apply, Pi.sup_apply, min_mul_max] at hcs
91   have hmax : (∫ x, max (p x) (q x) ∂μ) ≤ 2 := by
92     calc
93       _ ≤ ∫ x, p x + q x ∂μ := integral_mono (hp.sup hq) (hp.add hq)
94         (fun x => max_le (le_add_of_nonneg_right (hq0 x))
95           (le_add_of_nonneg_left (hp0 x)))
96       _ = 2 := by
97         rw [integral_add hp hq]
98         dsimp [p, q]
99         rw [Measure.integral_toReal_rnDeriv hP, Measure.integral_toReal_rnDeriv hQ]
100         norm_num
101   have hmin : 0 ≤ ∫ x, min (p x) (q x) ∂μ :=
102     integral_nonneg fun x => le_min (hp0 x) (hq0 x)
103   have hb := mul_le_mul_of_nonneg_left hmax hmin
104   change (1 / 2 : ℝ) * (∫ x, Real.sqrt (p x * q x) ∂μ) ^ 2 ≤
105     ∫ x, min (p x) (q x) ∂μ
106   nlinarith

129 /-- The likelihood comparison event attains the density-overlap testing error. -/
130 theorem commonDensityOverlap_eq_testingError
131     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
132     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
133     (hP : P ≪ μ) (hQ : Q ≪ μ) :
134     commonDensityOverlap P Q μ =
135       P.real (commonDensityComparisonEvent P Q μ) +
136         Q.real (commonDensityComparisonEvent P Q μ)ᶜ := by
137   let A := commonDensityComparisonEvent P Q μ
138   have hA : MeasurableSet A := measurableSet_commonDensityComparisonEvent P Q μ
139   rw [commonDensityOverlap, ← integral_add_compl hA (integrable_min_commonDensity P Q μ)]
140   calc
141     _ = (∫ x in A, (P.rnDeriv μ x).toReal ∂μ) +
142         ∫ x in Aᶜ, (Q.rnDeriv μ x).toReal ∂μ := by
143       congr 1
144       · apply setIntegral_congr_fun hA
145         intro x hx
146         exact min_eq_left hx
147       · apply setIntegral_congr_fun hA.compl
148         intro x hx
149         exact min_eq_right (le_of_lt (lt_of_not_ge hx))
150     _ = _ := by
151       rw [Measure.setIntegral_toReal_rnDeriv hP A,
152         Measure.setIntegral_toReal_rnDeriv hQ Aᶜ]

154 /-- The overlap is no larger than the testing error of any measurable event. -/
155 theorem commonDensityOverlap_le_testingError
156     {α : Type*} [MeasurableSpace α] (P Q μ : Measure α)
157     [IsFiniteMeasure P] [IsFiniteMeasure Q] [SigmaFinite μ]
158     (hP : P ≪ μ) (hQ : Q ≪ μ) {A : Set α} (hA : MeasurableSet A) :
159     commonDensityOverlap P Q μ ≤ P.real A + Q.real Aᶜ := by
160   rw [commonDensityOverlap, ← integral_add_compl hA (integrable_min_commonDensity P Q μ)]
161   calc
162     _ ≤ (∫ x in A, (P.rnDeriv μ x).toReal ∂μ) +
163         ∫ x in Aᶜ, (Q.rnDeriv μ x).toReal ∂μ := by
164       apply add_le_add
165       · exact integral_mono (integrable_min_commonDensity P Q μ).integrableOn
166           Measure.integrable_toReal_rnDeriv.integrableOn (fun x => min_le_left _ _)
167       · exact integral_mono (integrable_min_commonDensity P Q μ).integrableOn
168           Measure.integrable_toReal_rnDeriv.integrableOn (fun x => min_le_right _ _)
169     _ = _ := by
170       rw [Measure.setIntegral_toReal_rnDeriv hP A,
171         Measure.setIntegral_toReal_rnDeriv hQ Aᶜ]
