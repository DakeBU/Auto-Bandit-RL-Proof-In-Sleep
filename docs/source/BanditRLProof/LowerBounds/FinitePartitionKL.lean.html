3 /-!
4 # Finite measurable discretisations of relative entropy
5 
6 The supremum in textbook Eq. (14.5) ranges over every finite measurable
7 partition, represented here by measurable maps to `Fin n`. Empty cells are
8 harmless. The singular branch below is only one part of the required
9 equivalence with the Radon--Nikodym definition.
10 -/

19 /-- Convexity bounds the divergence of total masses by finite-measure KL. -/
20 theorem totalMass_klFun_le_relativeEntropy
21     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
22     [IsFiniteMeasure P] [IsFiniteMeasure Q] :
23     ENNReal.ofReal (InformationTheory.klFun ((P univ / Q univ).toReal)) * Q univ ≤
24       relativeEntropy P Q := by
25   by_cases ht : relativeEntropy P Q = (⊤ : ENNReal)
26   · simp [ht]
27   have hreg := relativeEntropy_ne_top_iff.1 ht
28   have hb := InformationTheory.mul_klFun_le_toReal_klDiv hreg.1 hreg.2
29   calc
30     _ = ENNReal.ofReal (Q.real univ *
31         InformationTheory.klFun (P.real univ / Q.real univ)) := by
32       rw [ENNReal.ofReal_mul (measureReal_nonneg)]
33       simp [measureReal_def, ENNReal.toReal_div, mul_comm]
34     _ ≤ ENNReal.ofReal (relativeEntropy P Q).toReal := ENNReal.ofReal_le_ofReal hb
35     _ = _ := ENNReal.ofReal_toReal ht

37 /-- KL splits into the restrictions to all cells of a finite observation. -/
38 theorem sum_relativeEntropy_restrict_fibers
39     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
40     [IsFiniteMeasure P] [IsFiniteMeasure Q] (h : P ≪ Q)
41     {n : ℕ} (f : α → Fin n) (hf : Measurable f) :
42     (∑ i, relativeEntropy (P.restrict (f ⁻¹' {i})) (Q.restrict (f ⁻¹' {i}))) =
43       relativeEntropy P Q := by
44   let g : α → ENNReal := fun x => ENNReal.ofReal
45     (InformationTheory.klFun (P.rnDeriv Q x).toReal)
46   have hcell (i : Fin n) : MeasurableSet (f ⁻¹' {i}) := hf (measurableSet_singleton i)
47   have hall : (⋃ i, f ⁻¹' {i}) = univ := by
48     ext x
49     simp
50   calc
51     _ = ∑ i, ∫⁻ x in f ⁻¹' {i}, g x ∂Q := by
52       apply Finset.sum_congr rfl
53       intro i _
54       rw [relativeEntropy, InformationTheory.klDiv_eq_lintegral_klFun_of_ac (h.restrict _)]
55       apply lintegral_congr_ae
56       filter_upwards [rnDeriv_restrict_restrict h (hcell i)] with x hx
57       simp [g, hx]
58     _ = ∫⁻ x, g x ∂Q := by
59       rw [← tsum_fintype (L := .unconditional _), ← lintegral_iUnion hcell
60         (Set.pairwise_univ.1 (Set.pairwiseDisjoint_fiber f univ)), hall,
61         Measure.restrict_univ]
62     _ = _ := (InformationTheory.klDiv_eq_lintegral_klFun_of_ac h).symm
63 
64 /-- Finite-valued measurable observations cannot increase KL. -/
65 theorem relativeEntropy_finite_map_le
66     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
67     [IsFiniteMeasure P] [IsFiniteMeasure Q]
68     {n : ℕ} (f : α → Fin n) (hf : Measurable f) :
69     relativeEntropy (P.map f) (Q.map f) ≤ relativeEntropy P Q := by
70   by_cases ht : relativeEntropy P Q = (⊤ : ENNReal)
71   · simp [ht]
72   have h := (relativeEntropy_ne_top_iff.1 ht).1
73   rw [relativeEntropy_finite_klFun _ _ (h.map hf)]
74   calc
75     _ ≤ ∑ i, relativeEntropy (P.restrict (f ⁻¹' {i})) (Q.restrict (f ⁻¹' {i})) := by
76       apply Finset.sum_le_sum
77       intro i _
78       simpa only [Measure.map_apply hf (measurableSet_singleton _),
79         Measure.restrict_apply_univ] using
80         totalMass_klFun_le_relativeEntropy (P.restrict (f ⁻¹' {i})) (Q.restrict (f ⁻¹' {i}))
81     _ = _ := sum_relativeEntropy_restrict_fibers P Q h f hf

83 /-- Relative entropy defined by the supremum over finite measurable observations. -/
84 def finitePartitionRelativeEntropy {α : Type*} [MeasurableSpace α]
85     (P Q : Measure α) : ENNReal :=
86   ⨆ (n : ℕ) (f : α → Fin n) (_ : Measurable f),
87     relativeEntropy (P.map f) (Q.map f)
88 
89 /-- The finite-discretisation supremum never exceeds RN relative entropy. -/
90 theorem finitePartitionRelativeEntropy_le_relativeEntropy
91     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
92     [IsFiniteMeasure P] [IsFiniteMeasure Q] :
93     finitePartitionRelativeEntropy P Q ≤ relativeEntropy P Q := by
94   refine iSup_le fun n => iSup_le fun f => iSup_le fun hf => ?_
95   exact relativeEntropy_finite_map_le P Q f hf
96 
97 /-- Every finite measurable observation is included in the defining supremum. -/
98 theorem relativeEntropy_map_le_finitePartitionRelativeEntropy
99     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
100     {n : ℕ} (f : α → Fin n) (hf : Measurable f) :
101     relativeEntropy (P.map f) (Q.map f) ≤ finitePartitionRelativeEntropy P Q := by
102   exact le_iSup_of_le n (le_iSup_of_le f (le_iSup_of_le hf le_rfl))
103 
104 /-- On an already finite observation space, the identity partition loses nothing. -/
105 theorem finitePartitionRelativeEntropy_fin_eq {n : ℕ} (P Q : Measure (Fin n))
106     [IsFiniteMeasure P] [IsFiniteMeasure Q] :
107     finitePartitionRelativeEntropy P Q = relativeEntropy P Q := by
108   apply le_antisymm (finitePartitionRelativeEntropy_le_relativeEntropy P Q)
109   simpa using relativeEntropy_map_le_finitePartitionRelativeEntropy P Q id measurable_id

111 /-- The discrete KL of an observation is the exact cell-mass formula. -/
112 theorem relativeEntropy_finite_map_eq_if
113     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
114     [IsProbabilityMeasure P] [IsProbabilityMeasure Q]
115     {n : ℕ} (f : α → Fin n) (hf : Measurable f) :
116     relativeEntropy (P.map f) (Q.map f) =
117       if ∀ i, Q (f ⁻¹' {i}) = 0 → P (f ⁻¹' {i}) = 0 then
118         ENNReal.ofReal (∑ i, (P (f ⁻¹' {i})).toReal *
119           Real.log ((P (f ⁻¹' {i})).toReal / (Q (f ⁻¹' {i})).toReal))
120       else (⊤ : ENNReal) := by
121   classical
122   letI : IsProbabilityMeasure (P.map f) := Measure.isProbabilityMeasure_map hf.aemeasurable
123   letI : IsProbabilityMeasure (Q.map f) := Measure.isProbabilityMeasure_map hf.aemeasurable
124   simpa only [Measure.map_apply hf (measurableSet_singleton _)] using
125     relativeEntropy_finite_eq_if (P.map f) (Q.map f)

127 /-- A measurable support mismatch is detected by a two-cell observation. -/
128 theorem exists_binary_map_relativeEntropy_eq_top_of_event
129     {α : Type*} [MeasurableSpace α] (P Q : Measure α)
130     {A : Set α} (hA : MeasurableSet A) (hp : P A ≠ 0) (hq : Q A = 0) :
131     ∃ f : α → Fin 2, Measurable f ∧
132       relativeEntropy (P.map f) (Q.map f) = (⊤ : ENNReal) := by
133   classical
134   let f : α → Fin 2 := fun x => if x ∈ A then 0 else 1
135   have hf : Measurable f := measurable_const.ite hA measurable_const
136   have hcell : f ⁻¹' {(0 : Fin 2)} = A := by
137     ext x
138     simp [f]
139   refine ⟨f, hf, relativeEntropy_eq_top_of_atom_support_mismatch _ _ 0 ?_ ?_⟩
140   · simpa [Measure.map_apply hf (measurableSet_singleton _), hcell] using hp
141   · simpa [Measure.map_apply hf (measurableSet_singleton _), hcell] using hq

143 /-- Non-absolute-continuity forces infinite finite-partition relative entropy. -/
144 theorem finitePartitionRelativeEntropy_eq_top_of_not_absolutelyContinuous
145     {α : Type*} [MeasurableSpace α] (P Q : Measure α) (h : ¬ P ≪ Q) :
146     finitePartitionRelativeEntropy P Q = (⊤ : ENNReal) := by
147   classical
148   have hex : ∃ A, MeasurableSet A ∧ Q A = 0 ∧ P A ≠ 0 := by
149     by_contra hn
150     apply h
151     apply Measure.AbsolutelyContinuous.mk
152     intro A hA hq
153     by_contra hp
154     exact hn ⟨A, hA, hq, hp⟩
155   obtain ⟨A, hA, hq, hp⟩ := hex
156   obtain ⟨f, hf, htop⟩ := exists_binary_map_relativeEntropy_eq_top_of_event P Q hA hp hq
157   apply top_unique
158   rw [← htop]
159   exact relativeEntropy_map_le_finitePartitionRelativeEntropy P Q f hf

161 /-- The finite-partition and RN definitions agree in the singular branch. -/
162 theorem finitePartitionRelativeEntropy_eq_relativeEntropy_of_not_absolutelyContinuous
163     {α : Type*} [MeasurableSpace α] (P Q : Measure α) (h : ¬ P ≪ Q) :
164     finitePartitionRelativeEntropy P Q = relativeEntropy P Q := by
165   rw [finitePartitionRelativeEntropy_eq_top_of_not_absolutelyContinuous P Q h,
