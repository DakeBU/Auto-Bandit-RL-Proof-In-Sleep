5 def BinaryPrefixCode.relabel {α β : Type*} (code : BinaryPrefixCode α) (e : β ≃ α) :
6     BinaryPrefixCode β where
7   encode := fun i => code.encode (e i)
8   injective := code.injective.comp e.injective
9   nonempty := fun i => code.nonempty (e i)
10   prefixFree := fun h => e.injective (code.prefixFree h)

12 /-- Full optimality against every code, not merely against a chosen candidate family. -/
13 def IsOptimalPrefixCode {α : Type*} [Fintype α] (p : α → ℝ) (code : BinaryPrefixCode α) : Prop :=
14   ∀ other : BinaryPrefixCode α, expectedCodeLength p code ≤ expectedCodeLength p other

16 theorem expectedCodeLength_swap
17     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ) (code : BinaryPrefixCode α)
18     (a b : α) (hab : a ≠ b) :
19     expectedCodeLength p (code.relabel (Equiv.swap a b)) = expectedCodeLength p code +
20       (p a - p b) * ((code.encode b).length - (code.encode a).length : ℝ) := by
21   have hterm (i : α) : p i * ((code.encode (Equiv.swap a b i)).length : ℝ) =
22       p i * (code.encode i).length +
23         (if i = a then p a * ((code.encode b).length - (code.encode a).length : ℝ) else 0) +
24         (if i = b then p b * ((code.encode a).length - (code.encode b).length : ℝ) else 0) := by
25     by_cases hia : i = a
26     · subst i
27       simp [hab]
28       ring
29     by_cases hib : i = b
30     · subst i
31       simp [hab.symm]
32       ring
33     · simp [Equiv.swap_apply_of_ne_of_ne hia hib, hia, hib]
34   change (∑ i, p i * ((code.encode (Equiv.swap a b i)).length : ℝ)) = _
35   simp_rw [hterm]
36   simp only [Finset.sum_add_distrib]
37   simp [expectedCodeLength]
38   ring

40 /-- Assigning a shorter word to a higher-probability symbol cannot increase cost. -/
41 theorem expectedCodeLength_swap_le
42     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ) (code : BinaryPrefixCode α)
43     (a b : α) (hab : a ≠ b) (hp : p a ≤ p b)
44     (hl : (code.encode a).length ≤ (code.encode b).length) :
45     expectedCodeLength p (code.relabel (Equiv.swap a b)) ≤ expectedCodeLength p code := by
46   rw [expectedCodeLength_swap p code a b hab]
47   have hlR : ((code.encode a).length : ℝ) ≤ (code.encode b).length := by exact_mod_cast hl
48   have h := mul_nonpos_of_nonpos_of_nonneg (sub_nonpos.mpr hp) (sub_nonneg.mpr hlR)
49   linarith

51 /-- An optimal code orders lengths opposite to strictly ordered probabilities. -/
52 theorem IsOptimalPrefixCode.length_antitone
53     {α : Type*} [Fintype α] [DecidableEq α] {p : α → ℝ} {code : BinaryPrefixCode α}
54     (hopt : IsOptimalPrefixCode p code) (a b : α) (hp : p a < p b) :
55     (code.encode b).length ≤ (code.encode a).length := by
56   have hab : a ≠ b := by intro h; subst b; exact (lt_irrefl _ hp)
57   have ho := hopt (code.relabel (Equiv.swap a b))
58   rw [expectedCodeLength_swap p code a b hab] at ho
59   by_contra hl
60   have hlR : ((code.encode a).length : ℝ) < (code.encode b).length := by
61     exact_mod_cast (lt_of_not_ge hl)
62   have hm := mul_neg_of_neg_of_pos (sub_neg.mpr hp) (sub_pos.mpr hlR)
63   linarith

65 /-- Any global minimizer inherits the entropy sandwich; this does not assert existence. -/
66 theorem IsOptimalPrefixCode.entropy_sandwich
67     {α : Type*} [Fintype α] [DecidableEq α] {p : α → ℝ} {code : BinaryPrefixCode α}
68     (hopt : IsOptimalPrefixCode p code) (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
69     discreteEntropyBaseTwo Finset.univ p ≤ expectedCodeLength p code ∧
70       expectedCodeLength p code ≤ discreteEntropyBaseTwo Finset.univ p + 1 := by
71   obtain ⟨other, _, hother⟩ := exists_binaryPrefixCode_entropy_sandwich p hp hs
72   exact ⟨discreteEntropyBaseTwo_le_expectedCodeLength p hp hs code, (hopt other).trans hother⟩

74 /-- The local nonempty-codeword convention forces at least one expected bit. -/
75 theorem one_le_expectedCodeLength
76     {α : Type*} [Fintype α] (p : α → ℝ) (hp : ∀ i, 0 ≤ p i)
77     (hs : ∑ i, p i = 1) (code : BinaryPrefixCode α) : 1 ≤ expectedCodeLength p code := by
78   rw [← hs]
79   apply Finset.sum_le_sum
80   intro i _
81   have hl : (1 : ℝ) ≤ (code.encode i).length := by
82     have hz : (code.encode i).length ≠ 0 := by
83       intro h
84       exact code.nonempty i (List.length_eq_zero_iff.mp h)
85     have hn : 1 ≤ (code.encode i).length := by omega
86     exact_mod_cast hn
87   nlinarith [hp i]

89 /-- The singleton alphabet's nonempty one-bit code. -/
90 def singletonPrefixCode (α : Type*) [Subsingleton α] : BinaryPrefixCode α where
91   encode := fun _ => [false]
92   injective := fun _ _ _ => Subsingleton.elim _ _
93   nonempty := fun _ => by simp
94   prefixFree := fun _ => Subsingleton.elim _ _

96 /-- The singleton base case has a genuine global optimum under the local convention. -/
97 theorem singletonPrefixCode_optimal
98     {α : Type*} [Fintype α] [Subsingleton α] (p : α → ℝ)
99     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
100     IsOptimalPrefixCode p (singletonPrefixCode α) := by
101   intro other
102   have he : expectedCodeLength p (singletonPrefixCode α) = 1 := by
103     simpa [expectedCodeLength, singletonPrefixCode] using hs
104   rw [he]
105   exact one_le_expectedCodeLength p hp hs other
