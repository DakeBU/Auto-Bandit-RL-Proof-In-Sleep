7 /-- Strict Shannon length, leaving Kraft slack on every positive-mass symbol.
8 This definition is not a prefix-code constructor. -/
9 def shannonLength (p : ℝ) : ℕ := ⌊Real.log p⁻¹ / Real.log 2⌋₊ + 1
10 
11 theorem shannonLength_pos (p : ℝ) : 0 < shannonLength p := by
12   exact Nat.succ_pos _
13 
14 /-- Each positive-mass symbol leaves strict Kraft slack. -/
15 theorem shannonLength_kraft_weight_lt {p : ℝ} (hp : 0 < p) :
16     (1 / 2 : ℝ) ^ shannonLength p < p := by
17   apply (Real.log_lt_log_iff (by positivity) hp).1
18   rw [Real.log_pow, Real.log_div (by norm_num) (by norm_num), Real.log_one]
19   have h := Nat.lt_floor_add_one (Real.log p⁻¹ / Real.log 2)
20   have hl : 0 < Real.log 2 := Real.log_pos (by norm_num)
21   have hcast : (shannonLength p : ℝ) = ⌊Real.log p⁻¹ / Real.log 2⌋₊ + 1 := by
22     simp [shannonLength]
23   rw [← hcast] at h
24   have hm := (div_lt_iff₀ hl).1 h
25   rw [Real.log_inv] at hm
26   nlinarith
27 
28 theorem shannonLength_le_information_add_one {p : ℝ} (hp : 0 < p) (hp1 : p ≤ 1) :
29     (shannonLength p : ℝ) ≤ Real.log p⁻¹ / Real.log 2 + 1 := by
30   have hn : 0 ≤ Real.log p⁻¹ / Real.log 2 :=
31     div_nonneg (Real.log_nonneg ((one_le_inv₀ hp).2 hp1))
32       (Real.log_pos (by norm_num)).le
33   simp only [shannonLength, Nat.cast_add, Nat.cast_one]
34   linarith [Nat.floor_le hn]
35 
36 theorem weighted_shannonLength_le {p : ℝ} (hp : 0 ≤ p) (hp1 : p ≤ 1) :
37     p * shannonLength p ≤ p * (Real.log p⁻¹ / Real.log 2) + p := by
38   rcases eq_or_lt_of_le hp with he | he
39   · subst p
40     simp
41   · have h := mul_le_mul_of_nonneg_left (shannonLength_le_information_add_one he hp1) hp
42     nlinarith
43 
44 /-- A numerical expected-length bound; realizability by a prefix code is separate. -/
45 theorem sum_weighted_shannonLength_le_entropy_add_one
46     {α : Type*} [Fintype α] (p : α → ℝ)
47     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
48     (∑ i, p i * shannonLength (p i)) ≤ discreteEntropyBaseTwo Finset.univ p + 1 := by
49   have hp1 (i) : p i ≤ 1 := by
50     rw [← hs]
51     exact Finset.single_le_sum (fun j _ => hp j) (Finset.mem_univ i)
52   have h := Finset.sum_le_sum (s := Finset.univ)
53     (fun i _ => weighted_shannonLength_le (hp i) (hp1 i))
54   simpa only [Finset.sum_add_distrib, hs, discreteEntropyBaseTwo] using h
55 
56 /-- The positive support occupies strictly less than the available Kraft mass. -/
57 theorem sum_positive_shannon_weights_lt_one
58     {α : Type*} [Fintype α] (p : α → ℝ)
59     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
60     (∑ i, if 0 < p i then (1 / 2 : ℝ) ^ shannonLength (p i) else 0) < 1 := by
61   classical
62   have hex : ∃ i, 0 < p i := by
63     by_contra h
64     have hz (i) : p i = 0 := le_antisymm (not_lt.mp (not_exists.mp h i)) (hp i)
65     simp [hz] at hs
66   apply lt_of_lt_of_eq ?_ hs
67   apply Finset.sum_lt_sum
68   · intro i _
69     split_ifs with hi
70     · exact (shannonLength_kraft_weight_lt hi).le
71     · exact hp i
72   · obtain ⟨i, hi⟩ := hex
73     exact ⟨i, Finset.mem_univ i, by simpa [hi] using shannonLength_kraft_weight_lt hi⟩
74 
75 /-- Complete length assignment, including zero-mass symbols; codewords are not yet constructed. -/
76 theorem exists_lengths_kraft_lt_one_entropy_bound
77     {α : Type*} [Fintype α] (p : α → ℝ)
78     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
79     ∃ l : α → ℕ, (∀ i, 0 < l i) ∧
80       (∑ i, (1 / 2 : ℝ) ^ l i) < 1 ∧
81       (∑ i, p i * l i) ≤ discreteEntropyBaseTwo Finset.univ p + 1 := by
82   classical
83   let W := ∑ i, if 0 < p i then (1 / 2 : ℝ) ^ shannonLength (p i) else 0
84   have hW : W < 1 := sum_positive_shannon_weights_lt_one p hp hs
85   let C : ℝ := Fintype.card α + 1
86   have hC : 0 < C := by dsimp [C]; positivity
87   obtain ⟨n, hn⟩ := exists_pow_lt_of_lt_one
88     (div_pos (sub_pos.mpr hW) hC) (by norm_num : (1 / 2 : ℝ) < 1)
89   let l : α → ℕ := fun i => if 0 < p i then shannonLength (p i) else n + 1
90   refine ⟨l, ?_, ?_, ?_⟩
91   · intro i
92     dsimp [l]
93     split_ifs
94     · exact shannonLength_pos _
95     · exact Nat.succ_pos _
96   · have hsum : (∑ i, (1 / 2 : ℝ) ^ l i) ≤ W + Fintype.card α * (1 / 2 : ℝ) ^ (n + 1) := by
97       calc
98         _ ≤ ∑ i, ((if 0 < p i then (1 / 2 : ℝ) ^ shannonLength (p i) else 0) +
99             (1 / 2 : ℝ) ^ (n + 1)) := by
100           apply Finset.sum_le_sum
101           intro i _
102           dsimp [l]
103           split_ifs
104           · nlinarith [pow_pos (by norm_num : (0 : ℝ) < 1 / 2) (n + 1)]
105           · simp
106         _ = _ := by simp [W, Finset.sum_add_distrib]
107     have hpow : (1 / 2 : ℝ) ^ (n + 1) ≤ (1 / 2 : ℝ) ^ n := by
108       rw [pow_succ]
109       nlinarith [pow_nonneg (by norm_num : (0 : ℝ) ≤ 1 / 2) n]
110     have hmul := (lt_div_iff₀ hC).1 hn
111     have hbound := mul_le_mul_of_nonneg_left hpow (Nat.cast_nonneg (Fintype.card α) :
112       (0 : ℝ) ≤ Fintype.card α)
113     dsimp [C] at hmul
114     nlinarith [pow_pos (by norm_num : (0 : ℝ) < 1 / 2) n]
115   · have heq : (∑ i, p i * l i) = ∑ i, p i * shannonLength (p i) := by
116       apply Finset.sum_congr rfl
117       intro i _
118       dsimp [l]
119       split_ifs with hi
120       · rfl
121       · have hz : p i = 0 := le_antisymm (not_lt.mp hi) (hp i)
122         simp [hz]
123     rw [heq]
124     exact sum_weighted_shannonLength_le_entropy_add_one p hp hs
