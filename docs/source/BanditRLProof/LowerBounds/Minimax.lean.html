6 # Finite-armed minimax lower-bound dependencies
7 
8 This file provides the unit-Gaussian analytic leaves for the source-faithful
9 Chapter 15 spine of Lattimore--Szepesvári, *Bandit Algorithms* (2020).  The
10 adaptive-history decomposition is proved in `BanditHistoryKL`; the complete
11 finite-armed Gaussian minimax endpoint is assembled in `GaussianMinimax`.
12 -/

15 namespace LowerBounds

17 open MeasureTheory ProbabilityTheory Real
18 open scoped ENNReal NNReal

20 noncomputable section

22 /-- A unit-variance Gaussian arm with mean `mu`. -/
23 abbrev unitGaussianArm (mu : Real) : Measure Real :=
24   gaussianReal mu (1 : NNReal)
25 
26 /-- The finite family of unit-variance Gaussian arms indexed by a mean vector. -/
27 abbrev unitGaussianBandit {k : Nat} (mean : Fin k -> Real) :
28     Fin k -> Measure Real :=
29   fun arm => unitGaussianArm (mean arm)
30 
31 /-- Pointwise log-density ratio for two unit-variance Gaussian laws. -/
32 theorem log_gaussianPDFReal_div_gaussianPDFReal_one
33     (mu nu x : Real) :
34     Real.log
35         (gaussianPDFReal mu (1 : NNReal) x /
36           gaussianPDFReal nu (1 : NNReal) x) =
37       (mu - nu) * x + (nu ^ 2 - mu ^ 2) / 2 := by
38   rw [gaussianPDFReal_def, gaussianPDFReal_def]
39   simp only [NNReal.coe_one, mul_one]
40   have hnorm : (Real.sqrt (2 * Real.pi))⁻¹ ≠ 0 := by positivity
41   have hden : Real.exp (-(x - nu) ^ 2 / 2) ≠ 0 := (Real.exp_pos _).ne'
42   rw [mul_div_mul_left _ _ hnorm]
43   rw [Real.log_div (Real.exp_ne_zero _) hden]
44   rw [Real.log_exp, Real.log_exp]
45   ring
46 
47 /-- The log Radon--Nikodym derivative has the expected affine form under the
48 first Gaussian law.  The direction is `N(mu,1)` to `N(nu,1)`. -/
49 theorem llr_gaussianReal_one_ae
50     (mu nu : Real) :
51     llr (unitGaussianArm mu) (unitGaussianArm nu)
52       =ᵐ[unitGaussianArm mu]
53         fun x => (mu - nu) * x + (nu ^ 2 - mu ^ 2) / 2 := by
54   let P := unitGaussianArm mu
55   let Q := unitGaussianArm nu
56   have hPvol : P ≪ volume := gaussianReal_absolutelyContinuous mu one_ne_zero
57   have hQvol : Q ≪ volume := gaussianReal_absolutelyContinuous nu one_ne_zero
58   have hvolQ : volume ≪ Q := gaussianReal_absolutelyContinuous' nu one_ne_zero
59   have hPQ : P ≪ Q := hPvol.trans hvolQ
60   have hratio := hPQ.ae_le (Measure.rnDeriv_eq_div hPvol hQvol)
61   have hPpdf := hPvol.ae_le (rnDeriv_gaussianReal mu (1 : NNReal))
62   have hQpdf := hPvol.ae_le (rnDeriv_gaussianReal nu (1 : NNReal))
63   filter_upwards [hratio, hPpdf, hQpdf] with x hratio_x hPpdf_x hQpdf_x
64   rw [llr, hratio_x, hPpdf_x, hQpdf_x, ENNReal.toReal_div,
65     toReal_gaussianPDF, toReal_gaussianPDF]
66   exact log_gaussianPDFReal_div_gaussianPDFReal_one mu nu x
67 
68 /-- The Gaussian log likelihood ratio is integrable under its first law. -/
69 theorem integrable_llr_gaussianReal_one
70     (mu nu : Real) :
71     Integrable (llr (unitGaussianArm mu) (unitGaussianArm nu))
72       (unitGaussianArm mu) := by
73   rw [integrable_congr (llr_gaussianReal_one_ae mu nu)]
74   have hid : Integrable (fun x : Real => x) (unitGaussianArm mu) :=
75     (memLp_id_gaussianReal (μ := mu) (v := (1 : NNReal)) 1).integrable
76       (by norm_num)
77   exact (hid.const_mul (mu - nu)).add (integrable_const _)
78 
79 /-- Exact KL divergence between unit-variance Gaussian laws:
80 `D(N(mu,1),N(nu,1))=(mu-nu)^2/2`.
81 
82 The result is stated in `ENNReal`, matching Mathlib's measure-KL API and
83 retaining the source direction even though this equal-variance value happens
84 to be symmetric in the two means. -/
85 theorem klDiv_gaussianReal_one
86     (mu nu : Real) :
87     InformationTheory.klDiv (unitGaussianArm mu) (unitGaussianArm nu) =
88       ENNReal.ofReal ((mu - nu) ^ 2 / 2) := by
89   have hPvol : unitGaussianArm mu ≪ volume :=
90     gaussianReal_absolutelyContinuous mu one_ne_zero
91   have hvolQ : volume ≪ unitGaussianArm nu :=
92     gaussianReal_absolutelyContinuous' nu one_ne_zero
93   have hPQ : unitGaussianArm mu ≪ unitGaussianArm nu := hPvol.trans hvolQ
94   have hllr := integrable_llr_gaussianReal_one mu nu
95   rw [InformationTheory.klDiv_of_ac_of_integrable hPQ hllr]
96   simp only [probReal_univ]
97   rw [integral_congr_ae (llr_gaussianReal_one_ae mu nu)]
98   have hid : Integrable (fun x : Real => x) (unitGaussianArm mu) :=
99     (memLp_id_gaussianReal (μ := mu) (v := (1 : NNReal)) 1).integrable
100       (by norm_num)
101   have hconst : Integrable (fun _x : Real => (nu ^ 2 - mu ^ 2) / 2)
102       (unitGaussianArm mu) := integrable_const _
103   have hlinear :
104       ∫ x, (mu - nu) * x ∂unitGaussianArm mu = (mu - nu) * mu := by
105     rw [integral_const_mul, integral_id_gaussianReal]
106   have hconstant :
107       ∫ _x, (nu ^ 2 - mu ^ 2) / 2 ∂unitGaussianArm mu =
108         (nu ^ 2 - mu ^ 2) / 2 := by
109     simp
110   rw [integral_add (hid.const_mul (mu - nu)) hconst]
111   rw [hlinear, hconstant]
112   congr 1
113   ring
114 
115 /-- Source-specialized Gaussian KL value for the changed arm in the proof of
116 Theorem 15.2. -/
117 theorem klDiv_unitGaussianArm_zero_two_mul (gap : Real) :
118     InformationTheory.klDiv (unitGaussianArm 0) (unitGaussianArm (2 * gap)) =
119       ENNReal.ofReal (2 * gap ^ 2) := by
120   rw [klDiv_gaussianReal_one]
121   congr 1
122   ring

124 /-- The source tuning `Delta=sqrt(m/(4n))`, written for positive real-valued
125 alternative count `m` and horizon `n`.  Natural-count consumers must discharge
126 their cast and positivity obligations explicitly. -/
127 noncomputable def gaussianMinimaxGap
128     (alternativeCount horizon : Real) : Real :=
129   Real.sqrt (alternativeCount / (4 * horizon))
130 
131 /-- Squared form of the Chapter 15 minimax gap choice. -/
132 theorem gaussianMinimaxGap_sq
133     {alternativeCount horizon : Real}
134     (halternatives : 0 ≤ alternativeCount) (hhorizon : 0 ≤ horizon) :
135     gaussianMinimaxGap alternativeCount horizon ^ 2 =
136       alternativeCount / (4 * horizon) := by
137   rw [gaussianMinimaxGap, sq_sqrt]
138   positivity
139 
140 /-- With the source tuning, the history-KL upper exponent
141 `2*n*Delta^2/m` is exactly `1/2`.  This is a numeric dependency only: it does
142 not supply the history-KL upper bound itself. -/
143 theorem gaussianMinimaxGap_informationExponent_eq_half
144     {alternativeCount horizon : Real}
145     (halternatives : 0 < alternativeCount) (hhorizon : 0 < horizon) :
146     2 * horizon * gaussianMinimaxGap alternativeCount horizon ^ 2 /
147         alternativeCount = 1 / 2 := by
148   rw [gaussianMinimaxGap_sq halternatives.le hhorizon.le]
149   field_simp
150   ring
151 
152 /-- If the number of alternative arms is at most the horizon, the tuned gap
153 is at most one half, so the source means `Delta` and `2*Delta` lie in the unit
154 interval after nonnegativity is combined with this leaf. -/
155 theorem gaussianMinimaxGap_le_half
156     {alternativeCount horizon : Real}
157     (hhorizon : 0 < horizon) (hcount_le : alternativeCount ≤ horizon) :
158     gaussianMinimaxGap alternativeCount horizon ≤ 1 / 2 := by
159   rw [gaussianMinimaxGap]
160   rw [Real.sqrt_le_iff]
161   constructor
162   · norm_num
163   · have hdiv : alternativeCount / horizon ≤ 1 :=
164       (div_le_one hhorizon).mpr hcount_le
165     have hfour : alternativeCount / (4 * horizon) ≤ 1 / 4 := by
166       calc
167         alternativeCount / (4 * horizon) =
168             (alternativeCount / horizon) / 4 := by field_simp
169         _ ≤ 1 / 4 := by linarith
170     norm_num at hfour ⊢
171     exact hfour
172 
173 end
