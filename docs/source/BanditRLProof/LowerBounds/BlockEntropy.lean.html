5 theorem entropy_product_term (p q : ℝ) :
6     (p * q) * Real.log (p * q)⁻¹ =
7       q * (p * Real.log p⁻¹) + p * (q * Real.log q⁻¹) := by
8   by_cases hp : p = 0
9   · simp [hp]
10   by_cases hq : q = 0
11   · simp [hq]
12   rw [Real.log_inv, Real.log_mul hp hq, Real.log_inv, Real.log_inv]
13   ring

15 /-- Entropy of a product mass function before normalization. -/
16 theorem discreteEntropy_prod
17     {α β : Type*} [Fintype α] [Fintype β] (p : α → ℝ) (q : β → ℝ) :
18     discreteEntropy Finset.univ (fun x : α × β => p x.1 * q x.2) =
19       (∑ j, q j) * discreteEntropy Finset.univ p +
20       (∑ i, p i) * discreteEntropy Finset.univ q := by
21   simp only [discreteEntropy, Fintype.sum_prod_type, entropy_product_term,
22     Finset.sum_add_distrib]
23   simp_rw [← Finset.mul_sum, ← Finset.sum_mul]
24   simp only [← Finset.mul_sum]

41 /-- An n-symbol source block, represented by a nested product. -/
42 def SourceBlock (α : Type*) : ℕ → Type _
43   | 0 => PUnit
44   | n + 1 => α × SourceBlock α n
45 
46 instance sourceBlockFintype {α : Type*} [Fintype α] (n : ℕ) : Fintype (SourceBlock α n) := by
47   induction n with
48   | zero => exact inferInstanceAs (Fintype PUnit)
49   | succ n ih => exact inferInstanceAs (Fintype (α × SourceBlock α n))
50 
51 instance sourceBlockDecidableEq {α : Type*} [DecidableEq α] (n : ℕ) :
52     DecidableEq (SourceBlock α n) := by
53   induction n with
54   | zero => exact inferInstanceAs (DecidableEq PUnit)
55   | succ n ih => exact inferInstanceAs (DecidableEq (α × SourceBlock α n))

57 /-- IID product mass, including the empty block of mass one. -/
58 def sourceBlockMass {α : Type*} (p : α → ℝ) : (n : ℕ) → SourceBlock α n → ℝ
59   | 0, _ => 1
60   | n + 1, x => p x.1 * sourceBlockMass p n x.2
61 
62 theorem sourceBlockMass_nonneg {α : Type*} (p : α → ℝ) (hp : ∀ i, 0 ≤ p i)
63     (n : ℕ) (x : SourceBlock α n) : 0 ≤ sourceBlockMass p n x := by
64   induction n with
65   | zero => exact zero_le_one
66   | succ n ih => exact mul_nonneg (hp x.1) (ih x.2)
67 
68 theorem sum_sourceBlockMass {α : Type*} [Fintype α] (p : α → ℝ)
69     (hp : ∑ i, p i = 1) (n : ℕ) : ∑ x, sourceBlockMass p n x = 1 := by
70   induction n with
71   | zero => simp [SourceBlock, sourceBlockMass]
72   | succ n ih =>
73     change (∑ x : α × SourceBlock α n, p x.1 * sourceBlockMass p n x.2) = 1
74     simp [Fintype.sum_prod_type, ← Finset.mul_sum, ih, hp]

76 theorem discreteEntropyBaseTwo_sourceBlockMass
77     {α : Type*} [Fintype α] (p : α → ℝ) (hp : ∑ i, p i = 1) (n : ℕ) :
78     discreteEntropyBaseTwo Finset.univ (sourceBlockMass p n) =
79       n * discreteEntropyBaseTwo Finset.univ p := by
80   induction n with
81   | zero => simp [discreteEntropyBaseTwo, SourceBlock, sourceBlockMass]
82   | succ n ih =>
83     change discreteEntropyBaseTwo Finset.univ
84       (fun x : α × SourceBlock α n => p x.1 * sourceBlockMass p n x.2) = _
85     rw [discreteEntropyBaseTwo_prod_probability p (sourceBlockMass p n) hp
86       (sum_sourceBlockMass p hp n), ih]

90 /-- Finite-block source coding with an explicit one-bit total overhead. -/
91 theorem exists_sourceBlock_code_rate_sandwich
92     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
93     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) (hn : 0 < n) :
94     ∃ code : BinaryPrefixCode (SourceBlock α n),
95       discreteEntropyBaseTwo Finset.univ p ≤ expectedCodeLength (sourceBlockMass p n) code / n ∧
96       expectedCodeLength (sourceBlockMass p n) code / n ≤
97         discreteEntropyBaseTwo Finset.univ p + 1 / n := by
98   obtain ⟨code, hlo, hhi⟩ := exists_binaryPrefixCode_entropy_sandwich
99     (sourceBlockMass p n) (sourceBlockMass_nonneg p hp n) (sum_sourceBlockMass p hs n)
100   rw [discreteEntropyBaseTwo_sourceBlockMass p hs n] at hlo hhi
101   have hnR : (0 : ℝ) < n := by exact_mod_cast hn
102   refine ⟨code, (le_div_iff₀ hnR).2 (by nlinarith), ?_⟩
103   apply (div_le_iff₀ hnR).2
104   have he : (discreteEntropyBaseTwo Finset.univ p + 1 / (n : ℝ)) * n =
105       n * discreteEntropyBaseTwo Finset.univ p + 1 := by field_simp
106   rw [he]
107   exact hhi

109 /-- Every finite block prefix code has rate at least the source entropy. -/
110 theorem sourceBlock_code_rate_lower_bound
111     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
112     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) (n : ℕ) (hn : 0 < n)
113     (code : BinaryPrefixCode (SourceBlock α n)) :
114     discreteEntropyBaseTwo Finset.univ p ≤ expectedCodeLength (sourceBlockMass p n) code / n := by
115   have h := discreteEntropyBaseTwo_le_expectedCodeLength (sourceBlockMass p n)
116     (sourceBlockMass_nonneg p hp n) (sum_sourceBlockMass p hs n) code
117   rw [discreteEntropyBaseTwo_sourceBlockMass p hs n] at h
118   apply (le_div_iff₀ (by exact_mod_cast hn : (0 : ℝ) < n)).2
119   nlinarith

121 /-- An actual family of block prefix codes has rate tending to entropy. -/
122 theorem exists_sourceBlock_code_family_tendsto_entropy
123     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
124     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1) :
125     ∃ code : (n : ℕ) → BinaryPrefixCode (SourceBlock α (n + 1)),
126       Filter.Tendsto
127         (fun n => expectedCodeLength (sourceBlockMass p (n + 1)) (code n) / (n + 1))
128         Filter.atTop (nhds (discreteEntropyBaseTwo Finset.univ p)) := by
129   classical
130   choose code hc using fun n : ℕ =>
131     exists_sourceBlock_code_rate_sandwich p hp hs (n + 1) (Nat.succ_pos n)
132   refine ⟨code, ?_⟩

142 theorem sourceBlock_code_family_limit_ge_entropy
143     {α : Type*} [Fintype α] [DecidableEq α] (p : α → ℝ)
144     (hp : ∀ i, 0 ≤ p i) (hs : ∑ i, p i = 1)
