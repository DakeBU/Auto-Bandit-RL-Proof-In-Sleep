3 /-!
4 # Horizon-compressed global mean-switch-count dynamic regret
5 
6 This module replaces every time-indexed global prefix count in the
7 logarithmic dynamic-regret route by the single terminal count at `horizon`.
8 The resulting bound is explicit and finite-sum free in its
9 nonstationarity term, but remains linear rather than minimax-sharp.
10 -/

39 /-- The fixed-comparator mean-deviation budget at the global prefix envelope
40 is bounded by the terminal count times horizon mass and suboptimal-arm
41 cardinality. -/
42 theorem finiteArmIndependentMeanDeviationBudget_globalMeanSwitchCount_le
43     {K : Nat} (model : FiniteBanditModel K)
44     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) :
45     finiteArmIndependentMeanDeviationBudget model horizon
46         (fun t _ =>
47           finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t) <=
48       2 *
49         (((Finset.univ : Finset (Fin K)).erase model.bestArm).card : Real) *
50         (((horizon + 1 : Nat) : Real)) *
51         finiteArmIndependentCumulativeGlobalMeanSwitchCount
52           armLaw horizon := by
53   classical
54   rw [finiteArmIndependentMeanDeviationBudget_eq]
55   let actions := (Finset.univ : Finset (Fin K)).erase model.bestArm
56   let terminalCount :=
57     finiteArmIndependentCumulativeGlobalMeanSwitchCount
58       armLaw horizon
59   change
60     (Finset.range (horizon + 1)).sum (fun t =>
61       actions.sum (fun _ =>
62         finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t +
63           finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t)) <=
64       2 * (actions.card : Real) * (((horizon + 1 : Nat) : Real)) *
65         terminalCount
66   calc
67     _ <= (Finset.range (horizon + 1)).sum (fun _ =>
68         (actions.card : Real) * (2 * terminalCount)) := by
69       apply Finset.sum_le_sum
70       intro t ht
71       have ht_le : t <= horizon := by
72         have ht_lt := Finset.mem_range.mp ht
73         omega
74       have hcount :=
75         finiteArmIndependentCumulativeGlobalMeanSwitchCount_mono
76           armLaw ht_le
77       have hinner :=
78         Finset.sum_le_card_nsmul actions
79           (fun _ =>
80             finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t +
81               finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t)
82           (2 * terminalCount)
83           (fun _ _ => by
84             linarith)
85       simpa [nsmul_eq_mul] using hinner
86     _ = 2 * (actions.card : Real) * (((horizon + 1 : Nat) : Real)) *
87         terminalCount := by
88       simp [nsmul_eq_mul]
89       ring

91 /-- The moving-comparator penalty at the global prefix envelope is bounded by
92 the same terminal-count expression as the fixed-comparator deviation budget.
93 The erased-arm cardinality makes the bound exactly zero for `Fin 1`. -/
94 theorem finiteArmIndependentDynamicComparatorPenalty_globalMeanSwitchCount_le
95     {K : Nat} (model : FiniteBanditModel K)
96     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) :
97     finiteArmIndependentDynamicComparatorPenalty model armLaw horizon
98         (fun t _ =>
99           finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t) <=
100       2 *
101         (((Finset.univ : Finset (Fin K)).erase model.bestArm).card : Real) *
102         (((horizon + 1 : Nat) : Real)) *
103         finiteArmIndependentCumulativeGlobalMeanSwitchCount
104           armLaw horizon := by
105   classical
106   let actions := (Finset.univ : Finset (Fin K)).erase model.bestArm
107   let terminalCount :=
108     finiteArmIndependentCumulativeGlobalMeanSwitchCount
109       armLaw horizon
110   unfold finiteArmIndependentDynamicComparatorPenalty
111   change
112     (Finset.range (horizon + 1)).sum (fun t =>
113       if finiteArmIndependentBestArmAt model armLaw t = model.bestArm then
114         0
115       else
116         finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t +
117           finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t) <=
118       2 * (actions.card : Real) * (((horizon + 1 : Nat) : Real)) *
119         terminalCount
120   have hterminalNonneg :
121       0 <= terminalCount := by
122     exact
123       finiteArmIndependentCumulativeGlobalMeanSwitchCount_nonneg
124         armLaw horizon
125   have hsum :=
126     Finset.sum_le_card_nsmul (Finset.range (horizon + 1))
127       (fun t =>
128         if finiteArmIndependentBestArmAt model armLaw t = model.bestArm then
129           0
130         else
131           finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t +
132             finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw t)
133       (2 * (actions.card : Real) * terminalCount)
134       (fun t ht => by
135         by_cases hsame :
136             finiteArmIndependentBestArmAt model armLaw t = model.bestArm
137         · simp only [hsame, if_true]
138           positivity
139         · simp only [hsame, if_false]
140           have ht_le : t <= horizon := by
141             have ht_lt := Finset.mem_range.mp ht
142             omega
143           have hcount :=
144             finiteArmIndependentCumulativeGlobalMeanSwitchCount_mono
145               armLaw ht_le
146           have hselectedMem : finiteArmIndependentBestArmAt model armLaw t ∈
147               actions := by
148             exact Finset.mem_erase.mpr ⟨hsame, Finset.mem_univ _⟩
149           have hcardNat : 1 <= actions.card :=
150             Finset.one_le_card.mpr ⟨_, hselectedMem⟩
151           have hcard : (1 : Real) <= (actions.card : Real) := by
152             exact_mod_cast hcardNat
153           nlinarith)
154   calc
155     _ <= (horizon + 1) •
156         (2 * (actions.card : Real) * terminalCount) := by
157       simpa only [Finset.card_range] using hsum
158     _ = 2 * (actions.card : Real) * (((horizon + 1 : Nat) : Real)) *
159         terminalCount := by
160       simp [nsmul_eq_mul]
161       ring
