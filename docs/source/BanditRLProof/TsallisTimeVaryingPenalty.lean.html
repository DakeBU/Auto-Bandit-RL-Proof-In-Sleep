9 /-!
10 # Time-varying half-Tsallis penalty
11 
12 This module formalizes the deterministic learning-rate-change term used by the
13 Tsallis-INF penalty route.  It telescopes paper-normalized half-Tsallis
14 potential values across a positive, nonincreasing schedule and retains the
15 negative terminal comparator contribution.
16 
17 The result consumes finite-simplex minimizer certificates only.  It does not
18 construct a stochastic trajectory, prove conditional stability, choose a
19 specific schedule, or conclude a bandit regret theorem.
20 -/

27 /-- The paper-normalized regularizer mass carried by the local potential. -/
28 noncomputable def halfTsallisPotentialMass {Action : Type u}
29     (arms : Finset Action) (p : Action -> Real) : Real :=
30   1 - negEntropyRegularizer arms (1 / 2 : Real) p
31 
32 /-- The regularizer mass is `2 * sum sqrt(p_a) - 1`. -/
33 theorem halfTsallisPotentialMass_eq_two_mul_powerSum_sub_one
34     {Action : Type u}
35     (arms : Finset Action) (p : Action -> Real) :
36     halfTsallisPotentialMass arms p =
37       2 * powerSum arms (1 / 2 : Real) p - 1 := by
38   simp [halfTsallisPotentialMass, negEntropyRegularizer, entropy]
39   ring

41 /-- A supported point mass has paper-normalized potential mass one. -/
42 theorem halfTsallisPotentialMass_pointMass
43     {Action : Type u} [DecidableEq Action]
44     (arms : Finset Action) {best : Action} (hbest : best ∈ arms) :
45     halfTsallisPotentialMass arms (pointMass best) = 1 := by
46   rw [halfTsallisPotentialMass_eq_two_mul_powerSum_sub_one,
47     powerSum_pointMass_half arms hbest]
48   ring

50 /-- The potential is linear loss with the regularizer mass divided by `eta`. -/
51 theorem halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
52     {Action : Type u}
53     (arms : Finset Action) (eta : Real)
54     (score p : Action -> Real) (heta : eta ≠ 0) :
55     halfTsallisPotentialValue arms eta score p =
56       -FTRL.linearLoss arms p score + halfTsallisPotentialMass arms p / eta := by
57   unfold halfTsallisPotentialValue halfTsallisPotentialMass
58   rw [FTRL.regularizedObjective]
59   field_simp [heta]
60   ring

62 /-- A regularized-objective minimizer maximizes the corresponding potential. -/
63 theorem halfTsallisPotentialValue_le_of_isRegularizedMinimizer
64     {Action : Type u}
65     (arms : Finset Action) (eta : Real)
66     (score p q : Action -> Real) (heta : 0 < eta)
67     (hp : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
68       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p)
69     (hq : FTRL.finiteSimplex arms q) :
70     halfTsallisPotentialValue arms eta score q <=
71       halfTsallisPotentialValue arms eta score p := by
72   have hobjective := hp.2 q hq
73   have hscaled :
74       FTRL.regularizedObjective arms eta
75           (negEntropyRegularizer arms (1 / 2 : Real)) score p / eta <=
76         FTRL.regularizedObjective arms eta
77           (negEntropyRegularizer arms (1 / 2 : Real)) score q / eta :=
78     (div_le_div_iff_of_pos_right heta).2 hobjective
79   unfold halfTsallisPotentialValue
80   simpa [neg_div, add_comm] using
81     add_le_add_right (neg_le_neg hscaled) (1 / eta)

83 /-- Changing the learning rate at a fixed score costs the reciprocal-rate
84 increment times the mass of the new minimizer. -/
85 theorem halfTsallisPotentialValue_new_sub_old_le_rateChange_mul_mass
86     {Action : Type u}
87     (arms : Finset Action) (etaOld etaNew : Real)
88     (score pOld pNew : Action -> Real)
89     (hetaOld : 0 < etaOld) (hetaNew : 0 < etaNew)
90     (hpOld : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
91       arms etaOld (negEntropyRegularizer arms (1 / 2 : Real)) score pOld)
92     (hpNew : FTRL.finiteSimplex arms pNew) :
93     halfTsallisPotentialValue arms etaNew score pNew -
94         halfTsallisPotentialValue arms etaOld score pOld <=
95       (1 / etaNew - 1 / etaOld) * halfTsallisPotentialMass arms pNew := by
96   have hpotential :=
97     halfTsallisPotentialValue_le_of_isRegularizedMinimizer
98       arms etaOld score pOld pNew hetaOld hpOld hpNew
99   calc
100     halfTsallisPotentialValue arms etaNew score pNew -
101         halfTsallisPotentialValue arms etaOld score pOld <=
102       halfTsallisPotentialValue arms etaNew score pNew -
103         halfTsallisPotentialValue arms etaOld score pNew :=
104       sub_le_sub_left hpotential _
105     _ = (1 / etaNew - 1 / etaOld) *
106         halfTsallisPotentialMass arms pNew := by
107       rw [halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
108           arms etaNew score pNew (ne_of_gt hetaNew),
109         halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
110           arms etaOld score pNew (ne_of_gt hetaOld)]
111       ring

113 /-- A zero-score minimizer maximizes the half-Tsallis regularizer mass. -/
114 theorem halfTsallisPotentialMass_le_of_zero_isRegularizedMinimizer
115     {Action : Type u}
116     (arms : Finset Action) (eta : Real)
117     (p q : Action -> Real) (heta : 0 < eta)
118     (hp : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
119       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) (fun _ => 0) p)
120     (hq : FTRL.finiteSimplex arms q) :
121     halfTsallisPotentialMass arms q <= halfTsallisPotentialMass arms p := by
122   have hpotential :=
123     halfTsallisPotentialValue_le_of_isRegularizedMinimizer
124       arms eta (fun _ => 0) p q heta hp hq
125   rw [halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
126         arms eta (fun _ => 0) q (ne_of_gt heta),
127       halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
128         arms eta (fun _ => 0) p (ne_of_gt heta)] at hpotential
129   simp [FTRL.linearLoss] at hpotential
130   exact (div_le_div_iff_of_pos_right heta).1 hpotential

132 /-- Algebraic telescope with different left and right endpoint processes. -/
133 theorem sum_range_succ_sub_eq_first_sub_last_add_cross
134     (A B : Nat -> Real) (n : Nat) :
135     (Finset.range (n + 1)).sum (fun t => A t - B t) =
136       A 0 - B n +
137         (Finset.range n).sum (fun t => A (t + 1) - B t) := by
138   induction n with
139   | zero => simp
140   | succ n ih =>
141       rw [Finset.sum_range_succ, ih, Finset.sum_range_succ]
142       ring

160 /-- Deterministic time-varying potential penalty with its terminal comparator
161 contribution left explicit. -/
162 theorem sum_halfTsallisScheduledPotentialPenalty_le
163     {Action : Type u}
164     (arms : Finset Action) (eta : Nat -> Real)
165     (loss : Nat -> Action -> Real)
166     (current sameRateNext : Nat -> Action -> Real)
167     (q : Action -> Real) (n : Nat)
168     (heta : forall t, t <= n -> 0 < eta t)
169     (hcurrent : forall t, t <= n ->
170       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
171         arms (eta t) (negEntropyRegularizer arms (1 / 2 : Real))
172         (FTRL.cumulativeLoss loss t) (current t))
173     (hnext : forall t, t <= n ->
174       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
175         arms (eta t) (negEntropyRegularizer arms (1 / 2 : Real))
176         (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t))
177     (hq : FTRL.finiteSimplex arms q) :
178     (Finset.range (n + 1)).sum (fun t =>
179         halfTsallisPotentialValue arms (eta t)
180             (FTRL.cumulativeLoss loss t) (current t) -
181           halfTsallisPotentialValue arms (eta t)
182             (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t) -
183           FTRL.linearLoss arms q (loss t)) <=
184       halfTsallisPotentialMass arms (current 0) / eta 0 +
185         (Finset.range n).sum (fun t =>
186           (1 / eta (t + 1) - 1 / eta t) *
187             halfTsallisPotentialMass arms (current (t + 1))) -
188         halfTsallisPotentialMass arms q / eta n := by
189   let A : Nat -> Real := fun t =>
190     halfTsallisPotentialValue arms (eta t)
191       (FTRL.cumulativeLoss loss t) (current t)
192   let B : Nat -> Real := fun t =>
193     halfTsallisPotentialValue arms (eta t)
194       (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t)
195   let C : Nat -> Real := fun t => FTRL.linearLoss arms q (loss t)
196   have htelescope :
197       (Finset.range (n + 1)).sum (fun t => (A t - B t) - C t) =
198         A 0 - B n + (Finset.range n).sum (fun t => A (t + 1) - B t) -
199           (Finset.range (n + 1)).sum C := by
200     rw [Finset.sum_sub_distrib,
201       sum_range_succ_sub_eq_first_sub_last_add_cross]
202   have hcross :
203       (Finset.range n).sum (fun t => A (t + 1) - B t) <=
204         (Finset.range n).sum (fun t =>
205           (1 / eta (t + 1) - 1 / eta t) *
206             halfTsallisPotentialMass arms (current (t + 1))) := by
207     apply Finset.sum_le_sum
208     intro t ht
209     have htn : t < n := Finset.mem_range.1 ht
210     exact halfTsallisPotentialValue_new_sub_old_le_rateChange_mul_mass
211       arms (eta t) (eta (t + 1)) (FTRL.cumulativeLoss loss (t + 1))
212       (sameRateNext t) (current (t + 1))
213       (heta t (Nat.le_of_lt htn))
214       (heta (t + 1) (Nat.succ_le_of_lt htn))
215       (hnext t (Nat.le_of_lt htn))
216       (hcurrent (t + 1) (Nat.succ_le_of_lt htn)).1
217   have hinitial :
218       A 0 = halfTsallisPotentialMass arms (current 0) / eta 0 := by
219     dsimp [A]
220     rw [halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
221       arms (eta 0) (FTRL.cumulativeLoss loss 0) (current 0)
222       (ne_of_gt (heta 0 (Nat.zero_le n)))]
223     simp [FTRL.linearLoss]
224   have hterminalPotential :=
225     halfTsallisPotentialValue_le_of_isRegularizedMinimizer
226       arms (eta n) (FTRL.cumulativeLoss loss (n + 1))
227       (sameRateNext n) q (heta n le_rfl) (hnext n le_rfl) hq
228   have hterminal :
229       -B n - (Finset.range (n + 1)).sum C <=
230         -halfTsallisPotentialMass arms q / eta n := by
231     dsimp [B, C]
232     rw [halfTsallisPotentialValue_eq_neg_linearLoss_add_mass_div
233       arms (eta n) (FTRL.cumulativeLoss loss (n + 1)) q
234       (ne_of_gt (heta n le_rfl))] at hterminalPotential
235     rw [FTRL.linearLoss_cumulativeLoss] at hterminalPotential
236     let S := (Finset.range (n + 1)).sum
237       (fun t => FTRL.linearLoss arms q (loss t))
238     let M := halfTsallisPotentialMass arms q / eta n
239     let V := halfTsallisPotentialValue arms (eta n)
240       (FTRL.cumulativeLoss loss (n + 1)) (sameRateNext n)
241     change -S + M <= V at hterminalPotential
242     have hbound : -V - S <= -M := by linarith
243     simpa [V, S, M, neg_div] using hbound
244   change (Finset.range (n + 1)).sum (fun t => (A t - B t) - C t) <= _
245   rw [htelescope, hinitial]
246   calc
247     halfTsallisPotentialMass arms (current 0) / eta 0 - B n +
248           (Finset.range n).sum (fun t => A (t + 1) - B t) -
249         (Finset.range (n + 1)).sum C =
250       halfTsallisPotentialMass arms (current 0) / eta 0 +
251           (Finset.range n).sum (fun t => A (t + 1) - B t) +
252         (-B n - (Finset.range (n + 1)).sum C) := by ring
253     _ <= halfTsallisPotentialMass arms (current 0) / eta 0 +
254           (Finset.range n).sum (fun t =>
255             (1 / eta (t + 1) - 1 / eta t) *
256               halfTsallisPotentialMass arms (current (t + 1))) +
257         (-B n - (Finset.range (n + 1)).sum C) := by
258       simpa [add_comm, add_left_comm, add_assoc] using
259         add_le_add_right
260           (add_le_add_left hcross
261             (halfTsallisPotentialMass arms (current 0) / eta 0))
262           (-B n - (Finset.range (n + 1)).sum C)
263     _ <= halfTsallisPotentialMass arms (current 0) / eta 0 +
264           (Finset.range n).sum (fun t =>
265             (1 / eta (t + 1) - 1 / eta t) *
266               halfTsallisPotentialMass arms (current (t + 1))) +
267         (-halfTsallisPotentialMass arms q / eta n) := by
268       simpa [add_comm, add_left_comm, add_assoc] using
269         add_le_add_left hterminal
270           (halfTsallisPotentialMass arms (current 0) / eta 0 +
271             (Finset.range n).sum (fun t =>
272               (1 / eta (t + 1) - 1 / eta t) *
273                 halfTsallisPotentialMass arms (current (t + 1))))
274     _ = halfTsallisPotentialMass arms (current 0) / eta 0 +
275           (Finset.range n).sum (fun t =>
276             (1 / eta (t + 1) - 1 / eta t) *
277               halfTsallisPotentialMass arms (current (t + 1))) -
278         halfTsallisPotentialMass arms q / eta n := by ring

280 /-- Under a nonincreasing positive schedule, every rate-change mass is bounded
281 by the initial zero-score mass, so the reciprocal-rate increments telescope. -/
282 theorem sum_halfTsallisScheduledPotentialPenalty_le_initial_sub_comparator_div
283     {Action : Type u}
284     (arms : Finset Action) (eta : Nat -> Real)
285     (loss : Nat -> Action -> Real)
286     (current sameRateNext : Nat -> Action -> Real)
287     (q : Action -> Real) (n : Nat)
288     (heta : forall t, t <= n -> 0 < eta t)
289     (hetaMono : forall t, t < n -> eta (t + 1) <= eta t)
290     (hcurrent : forall t, t <= n ->
291       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
292         arms (eta t) (negEntropyRegularizer arms (1 / 2 : Real))
293         (FTRL.cumulativeLoss loss t) (current t))
294     (hnext : forall t, t <= n ->
295       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
296         arms (eta t) (negEntropyRegularizer arms (1 / 2 : Real))
297         (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t))
298     (hq : FTRL.finiteSimplex arms q) :
299     (Finset.range (n + 1)).sum (fun t =>
300         halfTsallisPotentialValue arms (eta t)
301             (FTRL.cumulativeLoss loss t) (current t) -
302           halfTsallisPotentialValue arms (eta t)
303             (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t) -
304           FTRL.linearLoss arms q (loss t)) <=
305       (halfTsallisPotentialMass arms (current 0) -
306         halfTsallisPotentialMass arms q) / eta n := by
307   have hpenalty := sum_halfTsallisScheduledPotentialPenalty_le
308     arms eta loss current sameRateNext q n heta hcurrent hnext hq
309   have hmass : forall t, t < n ->
310       halfTsallisPotentialMass arms (current (t + 1)) <=
311         halfTsallisPotentialMass arms (current 0) := by
312     intro t ht
313     exact halfTsallisPotentialMass_le_of_zero_isRegularizedMinimizer
314       arms (eta 0) (current 0) (current (t + 1))
315       (heta 0 (Nat.zero_le n)) (by simpa using hcurrent 0 (Nat.zero_le n))
316       (hcurrent (t + 1) (Nat.succ_le_of_lt ht)).1
317   have hschedule :
318       (Finset.range n).sum (fun t =>
319           (1 / eta (t + 1) - 1 / eta t) *
320             halfTsallisPotentialMass arms (current (t + 1))) <=
321         (1 / eta n - 1 / eta 0) *
322           halfTsallisPotentialMass arms (current 0) := by
323     calc
324       (Finset.range n).sum (fun t =>
325           (1 / eta (t + 1) - 1 / eta t) *
326             halfTsallisPotentialMass arms (current (t + 1))) <=
327         (Finset.range n).sum (fun t =>
328           (1 / eta (t + 1) - 1 / eta t) *
329             halfTsallisPotentialMass arms (current 0)) := by
330           apply Finset.sum_le_sum
331           intro t ht
332           have htn : t < n := Finset.mem_range.1 ht
333           have hreciprocal : 1 / eta t <= 1 / eta (t + 1) :=
334             one_div_le_one_div_of_le
335               (heta (t + 1) (Nat.succ_le_of_lt htn)) (hetaMono t htn)
336           exact mul_le_mul_of_nonneg_left (hmass t htn) (sub_nonneg.2 hreciprocal)
337       _ = (1 / eta n - 1 / eta 0) *
338           halfTsallisPotentialMass arms (current 0) := by
339           rw [← Finset.sum_mul]
340           exact congrArg
341             (fun x : Real => x * halfTsallisPotentialMass arms (current 0))
342             (Exp3Potential.sum_range_forward_difference
343               (fun t => 1 / eta t) n)
344   calc
345     (Finset.range (n + 1)).sum (fun t =>
346         halfTsallisPotentialValue arms (eta t)
347             (FTRL.cumulativeLoss loss t) (current t) -
348           halfTsallisPotentialValue arms (eta t)
349             (FTRL.cumulativeLoss loss (t + 1)) (sameRateNext t) -
350           FTRL.linearLoss arms q (loss t)) <=
351       halfTsallisPotentialMass arms (current 0) / eta 0 +
352         (Finset.range n).sum (fun t =>
353           (1 / eta (t + 1) - 1 / eta t) *
354             halfTsallisPotentialMass arms (current (t + 1))) -
355         halfTsallisPotentialMass arms q / eta n := hpenalty
356     _ <= halfTsallisPotentialMass arms (current 0) / eta 0 +
357         (1 / eta n - 1 / eta 0) *
358           halfTsallisPotentialMass arms (current 0) -
359         halfTsallisPotentialMass arms q / eta n := by
360       exact sub_le_sub_right
361         (by
362           simpa [add_comm] using add_le_add_right hschedule
363             (halfTsallisPotentialMass arms (current 0) / eta 0)) _
364     _ = (halfTsallisPotentialMass arms (current 0) -
365         halfTsallisPotentialMass arms q) / eta n := by
366       field_simp [ne_of_gt (heta 0 (Nat.zero_le n)), ne_of_gt (heta n le_rfl)]
367       ring

369 /-- Canonical minimizer endpoint for the deterministic scheduled penalty. -/
370 theorem sum_halfTsallisCanonicalScheduledPotentialPenalty_le
371     {Action : Type u} [DecidableEq Action]
372     (arms : Finset Action) (harms : arms.Nonempty)
373     (eta : Nat -> Real) (loss : Nat -> Action -> Real)
374     (q : Action -> Real) (n : Nat)
375     (heta : forall t, t <= n -> 0 < eta t)
376     (hetaMono : forall t, t < n -> eta (t + 1) <= eta t)
377     (hq : FTRL.finiteSimplex arms q) :
378     (Finset.range (n + 1)).sum (fun t =>
379         halfTsallisPotentialValue arms (eta t)
380             (FTRL.cumulativeLoss loss t)
381             (halfTsallisScheduledMinimizer arms harms eta loss t) -
382           halfTsallisPotentialValue arms (eta t)
383             (FTRL.cumulativeLoss loss (t + 1))
384             (halfTsallisScheduledSameRateNext arms harms eta loss t) -
385           FTRL.linearLoss arms q (loss t)) <=
386       (halfTsallisPotentialMass arms
387           (halfTsallisScheduledMinimizer arms harms eta loss 0) -
388         halfTsallisPotentialMass arms q) / eta n := by
389   apply sum_halfTsallisScheduledPotentialPenalty_le_initial_sub_comparator_div
390     arms eta loss
391       (halfTsallisScheduledMinimizer arms harms eta loss)
392       (halfTsallisScheduledSameRateNext arms harms eta loss)
393       q n heta hetaMono
394   · intro t _ht
395     exact halfTsallisMinimizer_isRegularizedMinimizer
396       arms harms (eta t) (FTRL.cumulativeLoss loss t)
397   · intro t _ht
398     exact halfTsallisMinimizer_isRegularizedMinimizer
399       arms harms (eta t) (FTRL.cumulativeLoss loss (t + 1))
400   · exact hq

402 /-- Best-arm specialization retaining the explicit terminal `-1 / eta n`
403 contribution encoded by the point-mass comparator. -/
404 theorem sum_halfTsallisCanonicalScheduledPotentialPenalty_pointMass_le
405     {Action : Type u} [DecidableEq Action]
406     (arms : Finset Action) (harms : arms.Nonempty)
407     (eta : Nat -> Real) (loss : Nat -> Action -> Real)
408     {best : Action} (hbest : best ∈ arms) (n : Nat)
409     (heta : forall t, t <= n -> 0 < eta t)
410     (hetaMono : forall t, t < n -> eta (t + 1) <= eta t) :
411     (Finset.range (n + 1)).sum (fun t =>
412         halfTsallisPotentialValue arms (eta t)
413             (FTRL.cumulativeLoss loss t)
414             (halfTsallisScheduledMinimizer arms harms eta loss t) -
415           halfTsallisPotentialValue arms (eta t)
416             (FTRL.cumulativeLoss loss (t + 1))
417             (halfTsallisScheduledSameRateNext arms harms eta loss t) -
418           FTRL.linearLoss arms (pointMass best) (loss t)) <=
419       halfTsallisPotentialMass arms
420           (halfTsallisScheduledMinimizer arms harms eta loss 0) / eta n -
421         1 / eta n := by
422   simpa [halfTsallisPotentialMass_pointMass arms hbest, sub_div] using
423     sum_halfTsallisCanonicalScheduledPotentialPenalty_le
424       arms harms eta loss (pointMass best) n heta hetaMono
425         (finiteSimplex_pointMass arms hbest)
