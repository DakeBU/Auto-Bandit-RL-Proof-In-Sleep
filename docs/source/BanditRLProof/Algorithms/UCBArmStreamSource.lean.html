3 /-!
4 # Latent arm-stream reward source for UCB peeling
5 
6 This module implements the pathwise reward-consumption rule used by the pinned
7 LML UCB route. Whenever an action is selected, the observed reward is the next
8 unused coordinate of that arm's latent stream. Consequently, the selected
9 reward sum is exactly the stream prefix whose length is the realized pull
10 count, so it supplies `FixedArmPrefixSource` without an extra pathwise
11 hypothesis.
12 -/

22 /--
23 Read the next unused latent reward of the arm selected at time `t`.
24 
25 The action trace may depend on the whole sample point. The coordinate index is
26 the number of earlier selections of the currently selected arm.
27 -/
28 def rewardFromArmStream
29     {Omega : Type u} {K : Nat}
30     (action : Omega -> ActionTrace (Fin K))
31     (armStream : Omega -> ArmRewardStream K) :
32     Omega -> RewardTrace Real :=
33   fun omega t =>
34     armStream omega
35       (pullCount (action omega) ((action omega) t) t)
36       ((action omega) t)
37 
38 /--
39 Rewards consumed from one arm are exactly the prefix of that arm's latent
40 stream whose length is the realized pull count.
41 -/
42 theorem sumRewards_rewardFromArmStream_eq_armPrefixSum
43     {Omega : Type u} {K : Nat}
44     (action : Omega -> ActionTrace (Fin K))
45     (armStream : Omega -> ArmRewardStream K)
46     (omega : Omega) (arm : Fin K) (n : Nat) :
47     sumRewards (action omega) (rewardFromArmStream action armStream omega) arm n =
48       armPrefixSum arm (pullCount (action omega) arm n) (armStream omega) := by
49   induction n with
50   | zero => simp [armPrefixSum]
51   | succ n ih =>
52       by_cases hselected : (action omega) n = arm
53       · rw [sumRewards_succ_of_eq _ _ _ _ hselected]
54         rw [pullCount_succ_of_eq _ _ _ hselected]
55         rw [ih]
56         simp [armPrefixSum, Finset.sum_range_succ, rewardFromArmStream, hselected]
57       · rw [sumRewards_succ_of_ne _ _ _ _ (fun x : Real => add_zero x) hselected]
58         rw [pullCount_succ_of_ne _ _ _ hselected]
59         exact ih
60 
61 /--
62 The next-unused-coordinate reward rule produces a fixed-arm prefix source as
63 soon as the latent stream coordinates are measurable.
64 -/
65 def fixedArmPrefixSourceOfArmStream
66     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
67     (action : Omega -> ActionTrace (Fin K))
68     (armStream : Omega -> ArmRewardStream K)
69     (hmeasurable : forall i arm,
70       Measurable (fun omega => armStream omega i arm)) :
71     FixedArmPrefixSource action (rewardFromArmStream action armStream) where
72   armStream := armStream
73   measurable_armStream_coord := hmeasurable
74   sumRewards_eq_armPrefixSum :=
75     sumRewards_rewardFromArmStream_eq_armPrefixSum action armStream
76 
77 /--
78 Fixed-count peeling for rewards generated by the next-unused-coordinate rule,
79 with complete-stream law transport to a canonical stream.
80 -/
81 theorem measure_pullCount_prod_sumRewards_rewardFromArmStream_mem_le_identDistrib
82     {Omega : Type u} {Xi : Type v} {K : Nat}
83     [MeasurableSpace Omega] [MeasurableSpace Xi]
84     (mu : Measure Omega) (nu : Measure Xi)
85     (action : Omega -> ActionTrace (Fin K))
86     (armStream : Omega -> ArmRewardStream K)
87     (hmeasurable : forall i arm,
88       Measurable (fun omega => armStream omega i arm))
89     (canonicalStream : Xi -> ArmRewardStream K)
90     (hstreamLaw : IdentDistrib armStream canonicalStream mu nu)
91     (arm : Fin K) (n : Nat) (s : Set (Nat × Real))
92     [DecidablePred (fun k : Nat => k ∈ Prod.fst '' s)]
93     (hs : MeasurableSet s) :
94     mu {omega |
95         (pullCount (action omega) arm n,
96           sumRewards (action omega)
97             (rewardFromArmStream action armStream omega) arm n) ∈ s} ≤
98       ((Finset.range (n + 1)).filter
99         (fun k => k ∈ Prod.fst '' s)).sum (fun k =>
100           nu {xi |
101             armPrefixSum arm k (canonicalStream xi) ∈
102               Prod.mk k ⁻¹' s}) := by
103   exact measure_pullCount_prod_sumRewards_mem_le_of_fixedArmPrefixSource_identDistrib
104     mu nu action (rewardFromArmStream action armStream)
105       (fixedArmPrefixSourceOfArmStream action armStream hmeasurable)
106       canonicalStream hstreamLaw arm n s hs

108 /-- The canonical stream space supplies its own measurable latent stream. -/
109 def canonicalFixedArmPrefixSource
110     {K : Nat}
111     (action : ArmRewardStream K -> ActionTrace (Fin K)) :
112     FixedArmPrefixSource action (rewardFromArmStream action id) :=
113   fixedArmPrefixSourceOfArmStream action id (fun i arm =>
114     (measurable_pi_apply arm).comp (measurable_pi_apply i))
115 
116 /--
117 Pathwise peeling directly on the canonical latent stream space. This theorem
118 is ready for any recursively defined action trace on that space; probability
119 or stationarity assumptions enter only in later fixed-prefix tail leaves.
120 -/
121 theorem measure_pullCount_prod_sumRewards_rewardFromCanonicalArmStream_mem_le
122     {K : Nat}
123     (mu : Measure (ArmRewardStream K))
124     (action : ArmRewardStream K -> ActionTrace (Fin K))
125     (arm : Fin K) (n : Nat) (s : Set (Nat × Real))
126     [DecidablePred (fun k : Nat => k ∈ Prod.fst '' s)] :
127     mu {stream |
128         (pullCount (action stream) arm n,
129           sumRewards (action stream)
130             (rewardFromArmStream action id stream) arm n) ∈ s} ≤
131       ((Finset.range (n + 1)).filter
132         (fun k => k ∈ Prod.fst '' s)).sum (fun k =>
133           mu {stream |
