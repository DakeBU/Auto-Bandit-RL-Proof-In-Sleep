6 # Two-arm SGB Theorem 2: latent reward product bridge
7 
8 Appendix C of Baudry--Johnson--Vary--Pike-Burke--Rebeschini treats the rewards
9 collected from the optimal arm in pull order.  This module exposes the safe
10 part of that reindexing through the existing latent arm-stream coupling.
11 
12 The first `m` coordinates of one fixed arm have their finite IID product law.
13 The same law holds for the stream coordinate of the coupled SGB trajectory,
14 and every finite nth optimal-arm pull reads the corresponding latent
15 coordinate almost surely.  These results do not identify the coupling's
16 visible trajectory marginal with the native fixed-IID SGB trajectory.  They
17 also do not make totalized stopped rewards IID or condition on all requested
18 pulls occurring.  The native trajectory-law adapter and future-cylinder
19 producer remain separate obligations.
20 -/

31 /-- A fixed arm's first `m` latent rewards have the finite IID product law. -/
32 theorem armStreamMeasure_map_fixedArmFinitePrefix_eq_pi
33     {K m : Nat} (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
34     (arm : Fin K) :
35     Measure.map
36         (fun stream : ArmRewardStream K =>
37           fun i : Fin m => stream (i : Nat) arm)
38         (armStreamMeasure nu) =
39       Measure.pi (fun _ : Fin m => nu arm) := by
40   have hindex : Function.Injective
41       (fun i : Fin m => ((i : Nat), arm)) := by
42     intro i j hij
43     exact Fin.ext (congrArg Prod.fst hij)
44   have hindep :
45       iIndepFun
46         (fun i : Fin m =>
47           fun stream : ArmRewardStream K => stream (i : Nat) arm)
48         (armStreamMeasure nu) := by
49     simpa [armStreamCoordinate] using
50       (ProbabilityTheory.iIndepFun.precomp hindex
51         (iIndepFun_armStreamMeasure_coordinate nu))
52   have hcoord : forall i : Fin m,
53       AEMeasurable
54         (fun stream : ArmRewardStream K => stream (i : Nat) arm)
55         (armStreamMeasure nu) := fun i =>
56     ((measurable_pi_apply arm).comp
57       (measurable_pi_apply (i : Nat))).aemeasurable
58   have hproduct :=
59     (ProbabilityTheory.iIndepFun_iff_map_fun_eq_pi_map hcoord).1 hindep
60   calc
61     Measure.map
62         (fun stream : ArmRewardStream K =>
63           fun i : Fin m => stream (i : Nat) arm)
64         (armStreamMeasure nu) =
65       Measure.pi (fun i : Fin m =>
66         Measure.map
67           (fun stream : ArmRewardStream K => stream (i : Nat) arm)
68           (armStreamMeasure nu)) := by
69             simpa only using hproduct
70     _ = Measure.pi (fun _ : Fin m => nu arm) := by
71       congr 1
72       funext i
73       exact armStreamMeasure_map_coord nu (i : Nat) arm

79 /-- The fixed-arm product law lifted through the exact stream marginal of the
80 latent trajectory coupling. -/
81 theorem latentArmStreamTrajectoryMeasure_map_fixedArmFinitePrefix_eq_pi
82     {Env : Type u} {K m : Nat} [MeasurableSpace Env] [NeZero K]
83     (algorithm : HistoryAlgorithm (Fin K) Real) (env : Env)
84     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
85     (arm : Fin K) :
86     Measure.map
87         (fun sample :
88             UCB.ArmRewardStream K × ((n : Nat) -> Fin K × Real) =>
89           fun i : Fin m => sample.1 (i : Nat) arm)
90         (latentArmStreamTrajectoryMeasure algorithm env nu) =
91       Measure.pi (fun _ : Fin m => nu arm) := by
92   let fixedPrefix := fun stream : UCB.ArmRewardStream K =>
93     fun i : Fin m => stream (i : Nat) arm
94   have hfixedPrefix : Measurable fixedPrefix :=
95     measurable_pi_lambda _ fun i =>
96       (measurable_pi_apply arm).comp
97         (measurable_pi_apply (i : Nat))
98   have hfst :
99       Measure.map Prod.fst
100           (latentArmStreamTrajectoryMeasure algorithm env nu) =
101         UCB.armStreamMeasure nu := by
102     simpa using
103       (identDistrib_fst_latentArmStreamTrajectoryMeasure
104         algorithm env nu).map_eq
105   calc
106     Measure.map
107         (fun sample :
108             UCB.ArmRewardStream K × ((n : Nat) -> Fin K × Real) =>
109           fun i : Fin m => sample.1 (i : Nat) arm)
110         (latentArmStreamTrajectoryMeasure algorithm env nu) =
111       Measure.map fixedPrefix
112         (Measure.map Prod.fst
113           (latentArmStreamTrajectoryMeasure algorithm env nu)) := by
114             rw [Measure.map_map hfixedPrefix measurable_fst]
115             rfl
116     _ = Measure.map fixedPrefix (UCB.armStreamMeasure nu) := by
117       rw [hfst]
118     _ = Measure.pi (fun _ : Fin m => nu arm) := by
119       exact UCB.armStreamMeasure_map_fixedArmFinitePrefix_eq_pi nu arm

201 /-- At every finite nth optimal-arm pull, the observed stopped reward is the
202 corresponding latent arm-`0` coordinate almost surely.  This is pathwise
203 support, not an IID statement about totalized or occurrence-conditioned
204 stopped rewards. -/
205 theorem twoArmNthOptimalPullReward_eq_latentCoordinate_ae
206     (armLaw : Fin 2 -> Measure Real)
207     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
208     (eta : Real) (pullIndex : Nat) :
209     ∀ᵐ sample ∂twoArmFixedIIDLatentTrajectoryMeasure armLaw hprob eta,
210       ∀ t : Nat,
211         twoArmNthOptimalPullTime (Env := Unit) pullIndex ((), sample.2) =
212             (t : WithTop Nat) ->
213           twoArmNthOptimalPullReward (Env := Unit) pullIndex ((), sample.2) =
214             sample.1 pullIndex 0 := by
215   letI : IsMarkovKernel (UCB.finiteArmRealRewardKernel armLaw) :=
216     UCB.finiteArmRealRewardKernel_isMarkov armLaw hprob
217   have hsupport :=
218     Thompson.latentArmStreamTrajectoryReward_eq_rewardFromArmStream_ae
219       (historyAlgorithm (fun _ : Fin 2 => 0) eta) ()
220       (UCB.finiteArmRealRewardKernel armLaw)
221   change ∀ᵐ sample ∂Thompson.latentArmStreamTrajectoryMeasure
222       (historyAlgorithm (fun _ : Fin 2 => 0) eta) ()
223       (UCB.finiteArmRealRewardKernel armLaw),
224     ∀ t : Nat,
225       twoArmNthOptimalPullTime (Env := Unit) pullIndex ((), sample.2) =
226           (t : WithTop Nat) ->
227         twoArmNthOptimalPullReward (Env := Unit) pullIndex ((), sample.2) =
228           sample.1 pullIndex 0
229   filter_upwards [hsupport] with sample hsample
230   intro t htime
231   have hspec := twoArmNthOptimalPullTime_spec
232     (Env := Unit) pullIndex t ((), sample.2) htime
233   rw [twoArmNthOptimalPullReward_eq_of_time_eq pullIndex t ((), sample.2) htime]
