18 private theorem score_le_foldl_select
19     {K : Nat} (scores : Fin K -> Rat) (init : Fin K) :
20     forall l : List (Fin K),
21       (forall a : Fin K, List.Mem a l ->
22         scores a <=
23           scores
24             (l.foldl
25               (fun best arm : Fin K =>
26                 if scores best < scores arm then arm else best)
27               init)) /\
28       scores init <=
29         scores
30           (l.foldl
31             (fun best arm : Fin K =>
32               if scores best < scores arm then arm else best)
33             init)
34   | [] => by
35       exact And.intro
36         (by
37           intro _ ha
38           cases ha)
39         (by
40           simp)
41   | arm :: rest => by
42       let select :=
43         fun best arm : Fin K =>
44           if scores best < scores arm then arm else best
45       let next := select init arm
46       have ih := score_le_foldl_select (scores := scores) next rest
47       have harm_next : scores arm <= scores next := by
48         exact if hlt : scores init < scores arm then
49           by
50             simp [next, select, hlt]
51         else
52           by
53             simp [next, select, hlt, le_of_not_gt hlt]
54       exact And.intro
55         (by
56           intro a ha
57           cases ha with
58           | head =>
59               exact le_trans harm_next ih.2
60           | tail _ ha =>
61               exact ih.1 a ha)
62         (by
63           have hinit_next : scores init <= scores next := by
64             exact if hlt : scores init < scores arm then
65               by

67             else

72 private theorem argmax_cons_eq_some_foldl_rat_select
73     {K : Nat} (scores : Fin K -> Rat) (init : Fin K) (l : List (Fin K)) :
74     List.argmax scores (init :: l) =
75       some (l.foldl
76         (fun best arm : Fin K =>
77           if scores best < scores arm then arm else best)
78         init) := by
79   unfold List.argmax
80   simp only [List.foldl_cons, List.argAux]
81   induction l generalizing init with
82   | nil => rfl
83   | cons arm rest ih =>
84       simp only [List.foldl_cons]
85       rw [show List.argAux (fun b c : Fin K => scores c < scores b)
86           (some init) arm =
87           some (if scores init < scores arm then arm else init) by
88         by_cases h : scores init < scores arm
89         case pos => simp [List.argAux, h]
90         case neg => simp [List.argAux, h]]
91       exact ih _

93 /--
94 A concrete ETC commit oracle that selects a score-maximizing arm from `Fin K`.
95 
96 The selector scans `List.finRange K` and keeps the previous arm on ties, giving
97 a deterministic total oracle whenever `hK : 0 < K` supplies the initial arm.
98 This is the compiled `ETC-COMMIT-ORACLE-CONCRETE-ARGMAX` leaf.
99 -/
100 noncomputable def argmaxCommitOracle
101     {K : Nat} (hK : 0 < K) : ETC.CommitOracle K where
102   choose scores :=
103     (List.finRange K).foldl
104       (fun best arm : Fin K =>
105         if scores best < scores arm then arm else best)
106       (Fin.mk 0 hK)
107   card := "finite_rat_argmax_commit"
108 
109 /--
110 The concrete Rat commit oracle is Mathlib's first-occurrence list argmax.
111 
112 Because `List.finRange K` is ordered by the canonical `Fin` encoding, this
113 identity records the implementation-level tie rule used by the generated ETC
114 policy: strict score improvements replace the current arm and equal scores do
115 not.
116 -/
117 theorem argmaxCommitOracle_argmax_finRange
118     {K : Nat} (hK : 0 < K) (scores : Fin K -> Rat) :
119     List.argmax scores (List.finRange K) =
120       some ((ETC.argmaxCommitOracle hK).choose scores) := by
121   cases K with
122   | zero => omega
123   | succ k =>
124       rw [List.finRange_succ]
125       rw [argmax_cons_eq_some_foldl_rat_select]
126       simp [ETC.argmaxCommitOracle, List.finRange_succ]
127 
128 /--
129 Among arms tying the concrete Rat commit oracle's maximal score, the oracle
130 chooses the least encoded arm.
131 
132 This is the public tie-semantics certificate for the canonical generated ETC
133 route. It is deterministic and introduces no probability or concentration
134 assumption.
135 -/
136 theorem argmaxCommitOracle_encode_le_of_score_le
137     {K : Nat} (hK : 0 < K) (scores : Fin K -> Rat) (a : Fin K)
138     (hscore :
139       scores ((ETC.argmaxCommitOracle hK).choose scores) <= scores a) :
140     Encodable.encode ((ETC.argmaxCommitOracle hK).choose scores) <=
141       Encodable.encode a := by
142   have harg : Membership.mem (List.argmax scores (List.finRange K))
143       ((ETC.argmaxCommitOracle hK).choose scores) := by
144     rw [ETC.argmaxCommitOracle_argmax_finRange hK scores]
145     simp
146   have hidx := List.index_of_argmax harg (List.mem_finRange a) hscore
147   simpa only [List.idxOf_finRange] using hidx

149 /--
150 The concrete ETC argmax oracle returns an arm whose score dominates every arm.
151 
152 This is the maximality certificate needed by the already compiled abstract
153 commit-oracle wrong-event and probability consumers.  It does not introduce
154 measures, empirical-mean construction, concentration, filtration, or final ETC
155 regret.

157 theorem argmaxCommitOracle_choose_spec
158     {K : Nat} (hK : 0 < K)
159     (scores : Fin K -> Rat) (a : Fin K) :
160     scores a <= scores ((ETC.argmaxCommitOracle hK).choose scores) := by
161   unfold argmaxCommitOracle
162   dsimp
163   exact
164     (score_le_foldl_select
165       (scores := scores)
166       (init := Fin.mk 0 hK)
167       (List.finRange K)).1 a (List.mem_finRange a)

169 /--
170 Selecting a particular arm with the concrete ETC argmax implies that arm's
171 score is at least the selected model best arm's score.
172 
173 This is a deterministic single-fiber refinement of the existing wrong-commit
174 union reduction. It preserves the candidate arm instead of existentially or
175 union bounding it.
176 -/
177 theorem argmaxCommitOracle_eq_arm_subset_empMean_ge_bestArm
178     {Omega : Type u} {K : Nat}
179     (hK : 0 < K)
180     (model : FiniteBanditModel K)
181     (empMean : Omega -> Fin K -> Rat)
182     (a : Fin K) :
183     Set.Subset
184       {omega : Omega |
185         (ETC.argmaxCommitOracle hK).choose (empMean omega) = a}
186       {omega : Omega |
187         empMean omega a >= empMean omega model.bestArm} := by
188   intro omega hchoose
189   have hmax :=
190     ETC.argmaxCommitOracle_choose_spec hK (empMean omega) model.bestArm
191   rw [hchoose] at hmax
192   exact hmax
