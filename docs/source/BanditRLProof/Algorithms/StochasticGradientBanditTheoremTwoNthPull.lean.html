5 /-!
6 # Two-arm SGB Theorem 2: chronological nth-pull bridge
7 
8 Appendix C of Baudry--Johnson--Vary--Pike-Burke--Rebeschini reindexes the
9 optimal-arm dynamics by the number of times that arm has been selected.  The
10 generated Lean trajectory is instead indexed by chronological time.  This
11 module supplies the missing bridge without assuming that the adaptively
12 selected reward subsequence is IID.
13 
14 The `pullIndex` argument below is zero based: `pullIndex = 0` denotes the
15 first optimal-arm pull.  Its time is a `WithTop Nat`; `top` is the explicit
16 value when that pull never occurs.  At a finite value `t`, the prefix before
17 `t` contains exactly `pullIndex` optimal pulls, action `t` is optimal, and the
18 inclusive prefix through `t` contains exactly `pullIndex + 1` optimal pulls.
19 
20 The extracted reward and post-pull success probability are stopped values on
21 the same generated trajectory.  Their declarations establish only
22 measurability and the chronological identification.  No IID, independence,
23 future-cylinder probability, ballot, or Theorem-2 endpoint is claimed here.
24 -/

34 /-- Complete an inclusive finite prefix to an action trace, using arm `0`
35 outside the prefix.  Only coordinates through `prefix` are consumed below. -/
36 def twoArmPrefixGeneratedAction
37     {Env : Type v} (chron : Nat)
38     (context : Env × History.FinitePairHistory (Fin 2) Real chron) :
39     ActionTrace (Fin 2) :=
40   fun t =>
41     if h : t <= chron then
42       (context.2 ⟨t, Finset.mem_Iic.mpr h⟩).1
43     else
44       0

46 /-- Optimal-arm pulls visible in an inclusive finite prefix. -/
47 def twoArmPrefixOptimalPullCount
48     {Env : Type v} (chron : Nat)
49     (context : Env × History.FinitePairHistory (Fin 2) Real chron) : Nat :=
50   pullCount (twoArmPrefixGeneratedAction chron context) 0 (chron + 1)

52 theorem measurable_twoArmPrefixGeneratedAction
53     {Env : Type v} [MeasurableSpace Env] (chron t : Nat) :
54     Measurable
55       (fun context : Env × History.FinitePairHistory (Fin 2) Real chron =>
56         twoArmPrefixGeneratedAction chron context t) := by
57   by_cases ht : t <= chron

61           (context.2 index).1) :=

64   · simp [twoArmPrefixGeneratedAction, ht]

159 /-- A missing zero-based requested pull keeps every finite-horizon count below
160 the corresponding positive count level. -/
161 theorem twoArmOptimalPullCount_lt_succ_of_nthOptimalPullTime_eq_top
162     {Env : Type v} [MeasurableSpace Env] (pullIndex horizon : Nat)
163     (sample : Env × ((k : Nat) -> Fin 2 × Real))
164     (htop : twoArmNthOptimalPullTime pullIndex sample = (⊤ : WithTop Nat)) :
165     twoArmOptimalPullCount horizon sample < pullIndex + 1 := by
166   have hnever : ∀ chron,
167       pullCount (twoArmGeneratedAction sample) 0 (chron + 1) ≠
168         pullIndex + 1 := by
169     intro chron

171       ((twoArmNthOptimalPullTime_eq_top_iff pullIndex sample).mp htop chron)
172   simpa [twoArmOptimalPullCount] using
173     (pullCount_lt_of_forall_succ_ne
174       (twoArmGeneratedAction sample) (0 : Fin 2) horizon
175       (pullIndex + 1) (Nat.succ_pos pullIndex) hnever)

193     (hfinite : twoArmNthOptimalPullTime pullIndex sample ≠ (⊤ : WithTop Nat)) :

217 /-- A finite nth-pull time is a genuine selection of the optimal arm. -/
218 theorem twoArmNthOptimalPullTime_action_eq_zero
219     {Env : Type v} [MeasurableSpace Env] (pullIndex t : Nat)
220     (sample : Env × ((k : Nat) -> Fin 2 × Real))
221     (htime : twoArmNthOptimalPullTime pullIndex sample = (t : WithTop Nat)) :
222     twoArmGeneratedAction sample t = 0 := by
223   have hcount :
224       twoArmOptimalPullCount (t + 1) sample = pullIndex + 1 :=
225     twoArmNthOptimalPullTime_count_succ_eq_of_eq
226       pullIndex t sample htime
227   by_contra haction
228   have hprevious :
229       twoArmOptimalPullCount t sample = pullIndex + 1 := by
230     simpa [twoArmOptimalPullCount, pullCount_succ, haction] using hcount
231   cases t with
232   | zero =>
233       simp [twoArmOptimalPullCount] at hprevious
234   | succ t =>
235       have hmem :
236           twoArmInclusiveOptimalPullCountProcess t sample ∈
237             ({pullIndex + 1} : Set Nat) := by
238         simpa [twoArmInclusiveOptimalPullCountProcess] using hprevious
239       have hle :
240           twoArmNthOptimalPullTime pullIndex sample <= (t : WithTop Nat) :=
241         hittingAfter_le_of_mem (Nat.zero_le t) hmem
242       rw [htime] at hle
243       exact (Nat.not_succ_le_self t) (by exact_mod_cast hle)

245 /-- Immediately before the finite nth-pull time there are exactly
246 `pullIndex` optimal pulls. -/
247 theorem twoArmNthOptimalPullTime_count_eq
248     {Env : Type v} [MeasurableSpace Env] (pullIndex t : Nat)
249     (sample : Env × ((k : Nat) -> Fin 2 × Real))
250     (htime : twoArmNthOptimalPullTime pullIndex sample = (t : WithTop Nat)) :
251     twoArmOptimalPullCount t sample = pullIndex := by
252   have hcount :

254     twoArmNthOptimalPullTime_count_succ_eq_of_eq

257     twoArmNthOptimalPullTime_action_eq_zero
