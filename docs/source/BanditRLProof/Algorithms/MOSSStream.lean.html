34 def streamEmpirical {Ω : Type*} {k : ℕ} (mean : Fin k → ℝ)
35     (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) (trace : ActionTrace (Fin k))
36     (t : ℕ) (a : Fin k) : ℝ :=
37   mean a + streamMean (X a) ω (pullCount trace a t)

99 /-- Concrete MOSS execution on a fixed reward table, tracking pre-pull counts. -/
100 def streamCounts {Ω : Type*} {k : ℕ} (hk : 0 < k) (n : ℕ)
101     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) : ℕ → Fin k → ℕ
102   | 0 => fun _ => 0
103   | t+1 => fun a =>
104       let counts := streamCounts hk n mean X ω t
105       counts a + if action hk n t (fun b => mean b + streamMean (X b) ω (counts b)) counts = a
106         then 1 else 0

108 def streamTrace {Ω : Type*} {k : ℕ} (hk : 0 < k) (n : ℕ)
109     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) : ActionTrace (Fin k) :=
110   fun t => action hk n t
111     (fun a => mean a + streamMean (X a) ω (streamCounts hk n mean X ω t a))
