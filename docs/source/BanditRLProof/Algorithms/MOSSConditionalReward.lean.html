7 theorem map_condition_reward_eq_compProd {k : ℕ} (hk : 0 < k) (n : ℕ)
8     (mean : Fin k → ℝ) (t : ℕ) (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν] :
9     Measure.map (fun table => (canonicalCondition hk n mean t table, canonicalReward hk n mean table (t+1)))
10       (UCB.armStreamMeasure ν) =
11       (Measure.map (canonicalCondition hk n mean t) (UCB.armStreamMeasure ν)).compProd
12         (UCB.armStreamSelectedRewardKernel t ν) := by
13   let μ := UCB.armStreamMeasure ν
14   let C := Measure.map (canonicalCondition hk n mean t) μ
15   let P := fun table => (canonicalCondition hk n mean t table, canonicalReward hk n mean table (t+1))
16   let B := fun target : ℕ × Fin k => {table | canonicalNextCoordinate hk n mean t table = target}
17   have hB (q : ℕ × Fin k) : MeasurableSet (B q) :=
18     (measurable_canonicalNextCoordinate hk n mean t) (measurableSet_singleton q)
19   have hdB : Pairwise (Function.onFun Disjoint B) := by
20     intro i j hij
21     apply Set.disjoint_left.mpr
22     intro x hi hj
23     exact hij (hi.symm.trans hj)
24   have huB : (⋃ q, B q) = Set.univ := by
25     ext x
26     simp [B]
27   have hpB : μ = Measure.sum (fun q => μ.restrict (B q)) := by
28     rw [← Measure.restrict_iUnion hdB hB, huB, Measure.restrict_univ]
29   have hdC : Pairwise (Function.onFun Disjoint (conditionBranch (k := k) t)) := by
30     intro i j hij
31     apply Set.disjoint_left.mpr
32     intro x hi hj
33     exact hij (hi.symm.trans hj)
34   have huC : (⋃ q, conditionBranch (k := k) t q) = Set.univ := by
35     ext x
36     simp [conditionBranch]
37   have hpC : C = Measure.sum (fun q => C.restrict (conditionBranch t q)) := by
38     rw [← Measure.restrict_iUnion hdC (measurableSet_conditionBranch t), huC, Measure.restrict_univ]
39   have hP : Measurable P := (measurable_canonicalCondition hk n mean t).prodMk
40     (measurable_canonicalReward hk n mean (t+1))
41   have hl : Measure.map P μ = Measure.sum (fun q => (C.restrict (conditionBranch t q)).prod (ν q.2)) := by
42     calc
43       _ = Measure.map P (Measure.sum (fun q => μ.restrict (B q))) := by rw [← hpB]
44       _ = Measure.sum (fun q => Measure.map P (μ.restrict (B q))) := by
45         apply Measure.map_sum
46         rw [← hpB]
47         exact hP.aemeasurable
48       _ = _ := by
49         congr 1
50         funext q
51         exact map_condition_reward_restrict_branch hk n mean t ν q 0
52   have hr : C.compProd (UCB.armStreamSelectedRewardKernel t ν) =
53       Measure.sum (fun q => (C.restrict (conditionBranch t q)).prod (ν q.2)) := by
54     calc
55       _ = (Measure.sum (fun q => C.restrict (conditionBranch t q))).compProd
56           (UCB.armStreamSelectedRewardKernel t ν) := by rw [← hpC]
57       _ = Measure.sum (fun q => (C.restrict (conditionBranch t q)).compProd
58           (UCB.armStreamSelectedRewardKernel t ν)) := by rw [Measure.compProd_sum_left]
59       _ = _ := by
60         congr 1
61         funext q
62         have he : Filter.EventuallyEq (ae (C.restrict (conditionBranch t q)))
63             (UCB.armStreamSelectedRewardKernel t ν) (Kernel.const _ (ν q.2)) := by
64           filter_upwards [ae_restrict_mem (measurableSet_conditionBranch t q)] with c hc
65           have hs : c.2 = q.2 := congrArg Prod.snd hc
66           rw [Kernel.comap_apply, Kernel.const_apply, hs]
67         rw [Measure.compProd_congr he, Measure.compProd_const]
68   exact hl.trans hr.symm

73     Filter.EventuallyEq (ae ((UCB.armStreamMeasure ν).map (canonicalCondition hk n mean t)))
74       (condDistrib (fun table => canonicalReward hk n mean table (t+1))
75         (canonicalCondition hk n mean t) (UCB.armStreamMeasure ν))
