4 /-!
5 # Canonical-kernel trajectory source for stationary UCB
6 
7 This module packages the canonical arm-stream UCB split conditional laws as a
8 history algorithm and environment, then independently regenerates their
9 observable action/reward pair process with Mathlib's Ionescu-Tulcea
10 `Kernel.trajMeasure`. The resulting coordinate process supplies every field of
11 `RealStationaryUCBSequence` without a caller-provided sample space or law.
12 -/

21 /-- Canonical arm-stream initial and successor action laws as a history algorithm. -/
22 noncomputable def canonicalArmStreamHistoryAlgorithm
23     {K : Nat} [NeZero K]
24     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
25     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
26     Thompson.HistoryAlgorithm (Fin K) Real where
27   policy n := condDistrib
28     (fun stream : ArmRewardStream K =>
29       armStreamAction hK (c * (sigma2 : Real)) stream (n + 1))
30     (fun stream : ArmRewardStream K =>
31       History.finitePairHistoryOfTrace
32         (armStreamAction hK (c * (sigma2 : Real)) stream)
33         (armStreamReward hK (c * (sigma2 : Real)) stream) n)
34     (armStreamMeasure nu)
35   initialAction := Measure.map
36     (fun stream : ArmRewardStream K =>
37       armStreamAction hK (c * (sigma2 : Real)) stream 0)
38     (armStreamMeasure nu)
39   initialAction_isProbability := Measure.isProbabilityMeasure_map
40     (measurable_armStreamAction hK (c * (sigma2 : Real)) 0).aemeasurable

42 /-- Canonical arm-stream initial and successor reward laws as a history environment. -/
43 noncomputable def canonicalArmStreamHistoryEnvironment
44     {K : Nat} [NeZero K]
45     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
46     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
47     Thompson.HistoryEnvironment (Fin K) Real where
48   feedback n := condDistrib
49     (fun stream : ArmRewardStream K =>
50       armStreamReward hK (c * (sigma2 : Real)) stream (n + 1))
51     (fun stream : ArmRewardStream K =>
52       (History.finitePairHistoryOfTrace
53           (armStreamAction hK (c * (sigma2 : Real)) stream)
54           (armStreamReward hK (c * (sigma2 : Real)) stream) n,
55         armStreamAction hK (c * (sigma2 : Real)) stream (n + 1)))
56     (armStreamMeasure nu)
57   initialFeedback := condDistrib
58     (fun stream : ArmRewardStream K =>
59       armStreamReward hK (c * (sigma2 : Real)) stream 0)
60     (fun stream : ArmRewardStream K =>
61       armStreamAction hK (c * (sigma2 : Real)) stream 0)
62     (armStreamMeasure nu)

118 /--
119 The independently generated canonical-kernel trajectory satisfies all local
120 stationary UCB split conditional-law fields.
121 -/
122 theorem realStationaryUCBSequence_canonicalKernelTrajectory
123     {K : Nat} [NeZero K]
124     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
125     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
126     RealStationaryUCBSequence
127       (canonicalKernelTrajectoryMeasure hK c sigma2 nu)
128       hK c sigma2 nu
129       canonicalKernelTrajectoryAction canonicalKernelTrajectoryReward := by
130   let algorithm := canonicalArmStreamHistoryAlgorithm hK c sigma2 nu
131   let environment := canonicalArmStreamHistoryEnvironment hK c sigma2 nu
132   let source :=
133     Thompson.canonicalHistoryAlgorithmEnvironmentSplitSource
134       algorithm environment
135   exact
136     { measurable_action := source.measurable_action
137       measurable_reward := source.measurable_reward
138       hasLaw_action_zero := by
139         simpa [canonicalKernelTrajectoryMeasure,
140           canonicalKernelTrajectoryAction, algorithm,
141           canonicalArmStreamHistoryAlgorithm] using
142           source.initialAction_map_eq
143       hasCondDistrib_feedback_zero := by
144         simpa [canonicalKernelTrajectoryMeasure,
145           canonicalKernelTrajectoryAction, canonicalKernelTrajectoryReward,
146           algorithm, environment, canonicalArmStreamHistoryAlgorithm,
147           canonicalArmStreamHistoryEnvironment] using
148           source.initialFeedback_condDistrib
149       hasCondDistrib_action := by
150         intro n
151         simpa [canonicalKernelTrajectoryMeasure,
152           canonicalKernelTrajectoryAction, canonicalKernelTrajectoryReward,
153           algorithm, environment, canonicalArmStreamHistoryAlgorithm,
154           canonicalArmStreamHistoryEnvironment] using
155           source.policy_condDistrib n
156       hasCondDistrib_feedback := by
157         intro n
158         simpa [canonicalKernelTrajectoryMeasure,
159           canonicalKernelTrajectoryAction, canonicalKernelTrajectoryReward,
160           algorithm, environment, canonicalArmStreamHistoryAlgorithm,
161           canonicalArmStreamHistoryEnvironment] using
162           source.feedback_condDistrib n }

183         canonicalKernelTrajectoryAction n /\
184       realStationaryArmwiseBoundedFiniteArmExpectedRegret
185           (finiteArmCanonicalKernelTrajectoryMeasure
186             hK 4 sigma2 armLaw hprob) armLaw
187           canonicalKernelTrajectoryAction n <=
188         realStationaryArmwiseBoundedFiniteArmModelCoefficient
189             armLaw hprob lo hi *

203           canonicalKernelTrajectoryAction n <=
