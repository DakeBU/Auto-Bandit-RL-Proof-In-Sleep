7 # ETC bounded reward source from an infinite product law
8 
9 This module instantiates the action-matched bounded reward source contract for
10 a fixed product-coordinate reward trace law.  It is still a fixed-commit ETC
11 source leaf: it introduces no adaptive filtration, random commit arm, or final
12 expected-regret result.
13 -/

18 /--
19 Under an infinite product reward-coordinate law, the centered reward at time
20 `i + 1` is independent of the reward-only past sigma-algebra generated by
21 coordinates `j <= i`.
22 
23 This is the concrete product-law specialization of
24 `indep_centeredReward_succ_pastReward_iSup_of_iIndepFun_reward`.  It does not
25 add the deterministic action generators from `History.historyFiltrationSucc`;
26 use `indep_centeredReward_succ_historyFiltrationSucc_infinitePi` for that.
27 -/
28 theorem indep_centeredReward_succ_pastReward_iSup_infinitePi
29     {K : Nat}
30     (coordLaw : Nat -> MeasureTheory.Measure Rat)
31     [forall t : Nat, MeasureTheory.IsProbabilityMeasure (coordLaw t)]
32     (model : FiniteBanditModel K)
33     (b : Fin K) (i : Nat) :
34     ProbabilityTheory.Indep
35       (MeasurableSpace.comap
36         (fun omega : RewardTrace Rat =>
37           (((omega (i + 1) - model.mean b : Rat) : Real)))
38         inferInstance)
39       (iSup fun j : Nat =>
40         iSup fun _h : j <= i =>
41           MeasurableSpace.comap
42             (fun omega : RewardTrace Rat => omega j) inferInstance)
43       (MeasureTheory.Measure.infinitePi coordLaw) := by
44   exact
45     ETC.indep_centeredReward_succ_pastReward_iSup_of_iIndepFun_reward
46       (MeasureTheory.Measure.infinitePi coordLaw)
47       model
48       (fun omega : RewardTrace Rat => omega)
49       (fun t => measurable_pi_apply t)
50       (IndependenceFoundation.iIndepFun_rewardTrace_infinitePi coordLaw)
51       b i

91 /--
92 The infinite product law over reward coordinates satisfies the action-matched
93 bounded reward source contract when each coordinate law has the corresponding
94 a.s. bound and mean for the arm pulled by the fixed-commit ETC trace.
95 -/
96 theorem boundedRewardTraceSource_infinitePi_actionWithCommit
97     {K : Nat}
98     (coordLaw : Nat -> MeasureTheory.Measure Rat)
99     [forall t : Nat, MeasureTheory.IsProbabilityMeasure (coordLaw t)]
100     (spec : ETC.Spec K)
101     (model : FiniteBanditModel K)
102     (commitArm : Fin K)
103     (lo hi : Fin K -> Nat -> Real)
104     (h_coord_bound :
105       forall t, t < spec.explorationPulls * K ->
106         Filter.Eventually
107           (fun r : Rat =>
108             Set.Icc
109               (lo (ETC.actionWithCommit spec commitArm t) t)
110               (hi (ETC.actionWithCommit spec commitArm t) t)
111               (((r : Rat) : Real)))
112           (MeasureTheory.ae (coordLaw t)))
113     (h_coord_mean :
114       forall t, t < spec.explorationPulls * K ->
115         MeasureTheory.integral (coordLaw t)
116           (fun r : Rat => (((r : Rat) : Real))) =
117         (((model.mean (ETC.actionWithCommit spec commitArm t) : Rat) : Real))) :
118     ETC.BoundedRewardTraceSource
119       (MeasureTheory.Measure.infinitePi coordLaw)
120       spec model commitArm
121       (fun omega : RewardTrace Rat => omega) lo hi where
122   indep := by
123     exact IndependenceFoundation.iIndepFun_rewardTrace_infinitePi coordLaw
124   meas := by
125     intro t _ht
126     exact
127       ((measurable_of_countable (fun r : Rat => ((r : Rat) : Real))).comp
128         (measurable_pi_apply t)).aemeasurable
129   bound := by
130     intro t ht
131     have hmap :
132         MeasureTheory.Measure.map (fun omega : RewardTrace Rat => omega t)
133           (MeasureTheory.Measure.infinitePi coordLaw) = coordLaw t := by
134       simpa using (MeasureTheory.Measure.infinitePi_map_eval coordLaw t)
135     have h' :
136         Filter.Eventually
137           (fun r : Rat =>
138             Set.Icc
139               (lo (ETC.actionWithCommit spec commitArm t) t)
140               (hi (ETC.actionWithCommit spec commitArm t) t)
141               (((r : Rat) : Real)))
142           (MeasureTheory.ae
143             (MeasureTheory.Measure.map
144               (fun omega : RewardTrace Rat => omega t)
145               (MeasureTheory.Measure.infinitePi coordLaw))) := by
146       simpa [hmap] using h_coord_bound t ht
147     exact MeasureTheory.ae_of_ae_map (measurable_pi_apply t).aemeasurable h'
148   mean := by
149     intro t ht
150     have hmap :
151         MeasureTheory.Measure.map (fun omega : RewardTrace Rat => omega t)
152           (MeasureTheory.Measure.infinitePi coordLaw) = coordLaw t := by
153       simpa using (MeasureTheory.Measure.infinitePi_map_eval coordLaw t)
154     calc
155       MeasureTheory.integral (MeasureTheory.Measure.infinitePi coordLaw)
156           (fun omega : RewardTrace Rat => (((omega t : Rat) : Real)))
157           =
158         MeasureTheory.integral
159           (MeasureTheory.Measure.map
160             (fun omega : RewardTrace Rat => omega t)
161             (MeasureTheory.Measure.infinitePi coordLaw))
162           (fun r : Rat => (((r : Rat) : Real))) := by
163             exact
164               (MeasureTheory.integral_map
165                 (measurable_pi_apply t).aemeasurable
166                 ((measurable_of_countable
167                   (fun r : Rat => ((r : Rat) : Real))).aestronglyMeasurable)).symm
168       _ = MeasureTheory.integral (coordLaw t)
169             (fun r : Rat => (((r : Rat) : Real))) := by
170               rw [hmap]
171       _ = (((model.mean (ETC.actionWithCommit spec commitArm t) : Rat) : Real)) :=
172             h_coord_mean t ht
