60 private theorem score_le_foldl_select
61     {K : Nat} (scores : Fin K -> Real) (init : Fin K) :
62     forall l : List (Fin K),
63       (forall a : Fin K, List.Mem a l ->
64         scores a <=
65           scores
66             (l.foldl
67               (fun best arm : Fin K =>
68                 if scores best < scores arm then arm else best)
69               init)) /\
70       scores init <=
71         scores
72           (l.foldl
73             (fun best arm : Fin K =>
74               if scores best < scores arm then arm else best)
75             init)
76   | [] => by
77       exact And.intro
78         (by
79           intro _ ha
80           cases ha)
81         (by
82           simp)
83   | arm :: rest => by
84       let select :=
85         fun best arm : Fin K =>
86           if scores best < scores arm then arm else best
87       let next := select init arm
88       have ih := score_le_foldl_select (scores := scores) next rest
89       have harm_next : scores arm <= scores next := by
90         exact if hlt : scores init < scores arm then
91           by
92             simp [next, select, hlt]
93         else
94           by
95             simp [next, select, hlt, le_of_not_gt hlt]
96       exact And.intro
97         (by
98           intro a ha
99           cases ha with
100           | head =>
101               exact le_trans harm_next ih.2
102           | tail _ ha =>
103               exact ih.1 a ha)
104         (by
105           have hinit_next : scores init <= scores next := by
106             exact if hlt : scores init < scores arm then
107               by
108                 simpa [next, select, hlt] using le_of_lt hlt
109             else
110               by
111                 simp [next, select, hlt]
112           exact le_trans hinit_next ih.2)

114 /--
115 Concrete finite-arm Real score argmax selector.
116 
117 The selector scans `List.finRange K`, keeps the previous arm on ties, and uses
118 `hK` only to seed the nonempty finite arm set. This mirrors the ETC argmax
119 oracle but targets the Real-valued UCB confidence-score surface.
120 -/
121 noncomputable def scoreArgmax
122     {K : Nat} (hK : 0 < K) (scores : Fin K -> Real) : Fin K :=
123   (List.finRange K).foldl
124     (fun best arm : Fin K =>
125       if scores best < scores arm then arm else best)
126     (Fin.mk 0 hK)
127 
128 /-- The concrete Real score argmax dominates every arm score. -/
129 theorem scoreArgmax_spec
130     {K : Nat} (hK : 0 < K)
131     (scores : Fin K -> Real) (a : Fin K) :
132     scores a <= scores (scoreArgmax hK scores) := by
133   unfold scoreArgmax
134   exact
135     (score_le_foldl_select
136       (scores := scores)
137       (init := Fin.mk 0 hK)
138       (List.finRange K)).1 a (List.mem_finRange a)

143     (empiricalMean : Omega -> Nat -> Fin K -> Real)
144     (radius : Nat -> Fin K -> Real) :

194 /--
195 UCB good-event algebra: if the best arm's true mean is below its upper
196 confidence score, the chosen arm's true mean is above its lower confidence
197 score, and the chosen arm maximizes the confidence score against the best arm,
198 then the chosen arm's mean gap is at most twice its radius.
199 -/
200 theorem meanGap_le_two_radius_of_confidenceScore_max
201     {Arm : Type} (trueMean empiricalMean radius : Arm -> Real)
202     (best chosen : Arm)
203     (hbest_upper :
204       trueMean best <= confidenceScore empiricalMean radius best)
205     (hchosen_lower :
206       empiricalMean chosen - radius chosen <= trueMean chosen)
207     (hscore :
208       confidenceScore empiricalMean radius best <=
209         confidenceScore empiricalMean radius chosen) :
210     meanGap trueMean best chosen <= 2 * radius chosen := by
211   unfold meanGap confidenceScore at *
212   linarith

236 def upperConfidenceBad {Omega Arm : Type}
237     (trueMean : Arm -> Real) (empiricalMean : Omega -> Arm -> Real)
238     (radius : Arm -> Real) (arm : Arm) : Set Omega :=

345 upper/lower confidence failure events.

353       (Finset.univ : Finset Arm).sum

355           mu (upperConfidenceBad trueMean empiricalMean radius arm) +

360         lowerConfidenceBad trueMean empiricalMean radius arm)

362             (fun arm =>
