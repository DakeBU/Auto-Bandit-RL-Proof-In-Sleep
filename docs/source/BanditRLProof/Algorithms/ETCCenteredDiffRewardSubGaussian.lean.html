5 # ETC centered reward-difference sub-Gaussian transfer
6 
7 This module transfers per-time centered reward sub-Gaussian witnesses through
8 the deterministic centered pairwise reward-difference transform used by the
9 ETC wrong-commit probability route.
10 -/
11 
12 namespace BanditRLProof

15 /--
16 Variance proxy induced on the centered pairwise reward-difference summand by
17 per-arm centered reward variance proxies.
18 
19 Only the arm pulled by `ETC.actionWithCommit spec commitArm t` contributes:
20 arm `a`, the model best arm, or zero if neither is pulled at time `t`.
21 -/
22 noncomputable def centeredPairwiseRewardDiffVarianceProxy
23     {K : Nat}
24     (spec : ETC.Spec K)
25     (model : FiniteBanditModel K)
26     (commitArm : Fin K)
27     (cReward : Fin K -> Nat -> NNReal)
28     (a : Fin K) (t : Nat) : NNReal :=
29   if ETC.actionWithCommit spec commitArm t = a then cReward a t
30   else if ETC.actionWithCommit spec commitArm t = model.bestArm then
31     cReward model.bestArm t
32   else 0

34 /--
35 Transfer a centered reward sub-Gaussian witness at one time index to the
36 corresponding centered pairwise reward-difference summand.
37 
38 This is the `ETC-CENTERED-DIFF-SUBGAUSSIAN-REWARD-WITNESS` leaf. It handles
39 the deterministic action cases and the sign flip for the best-arm centered
40 reward. It does not prove the raw centered reward is sub-Gaussian from a
41 distributional model.
42 -/
43 theorem centeredPairwiseRewardDiff_hasSubgaussianMGF_of_centeredReward
44     {Omega : Type u} {K : Nat}
45     [MeasurableSpace Omega]
46     (mu : MeasureTheory.Measure Omega) [MeasureTheory.IsZeroOrProbabilityMeasure mu]
47     (spec : ETC.Spec K)
48     (model : FiniteBanditModel K)
49     (commitArm : Fin K)
50     (reward : Omega -> RewardTrace Rat)
51     (cReward : Fin K -> Nat -> NNReal)
52     (a : Fin K) (t : Nat)
53     (hne : a = model.bestArm -> False)
54     (h_subG :
55       forall b : Fin K, ETC.actionWithCommit spec commitArm t = b ->
56         ProbabilityTheory.HasSubgaussianMGF
57           (fun omega : Omega => (((reward omega t - model.mean b : Rat) : Real)))
58           (cReward b t) mu) :
59     ProbabilityTheory.HasSubgaussianMGF
60       (fun omega : Omega =>
61         ETC.centeredPairwiseRewardDiff spec model commitArm reward a t omega)
62       (ETC.centeredPairwiseRewardDiffVarianceProxy spec model commitArm cReward a t)
63       mu := by
64   classical
65   have hane : Not (a = model.bestArm) := hne
66   by_cases ha : ETC.actionWithCommit spec commitArm t = a
67   · have hnot_best : Not (ETC.actionWithCommit spec commitArm t = model.bestArm) := by
68       intro hbest
69       exact hne (ha.symm.trans hbest)
70     simpa [ETC.centeredPairwiseRewardDiffVarianceProxy,
71       ETC.centeredPairwiseRewardDiff, ha, hnot_best, hane] using h_subG a ha
72   · by_cases hbest : ETC.actionWithCommit spec commitArm t = model.bestArm
73     · have hbest_ne_a : Not (model.bestArm = a) := by
74         intro h
75         exact hne h.symm
76       have h := (h_subG model.bestArm hbest).neg
77       have htarget :
78           ProbabilityTheory.HasSubgaussianMGF
79             (fun omega : Omega =>
80               ((model.mean model.bestArm : Rat) : Real) - ((reward omega t : Rat) : Real))
81             (cReward model.bestArm t) mu := by
82         refine h.congr ?_
83         exact Filter.Eventually.of_forall (fun omega => by
84           simp [Pi.neg_apply, sub_eq_add_neg, add_comm])
85       simpa [ETC.centeredPairwiseRewardDiffVarianceProxy,
86         ETC.centeredPairwiseRewardDiff, ha, hbest, hbest_ne_a, Rat.cast_sub,
87         sub_eq_add_neg, add_comm, add_left_comm, add_assoc] using htarget
88     · have hfun :
89           (fun omega : Omega =>
90             ETC.centeredPairwiseRewardDiff spec model commitArm reward a t omega) =
91           (fun _omega : Omega => (0 : Real)) := by
92         funext omega
93         simp [ETC.centeredPairwiseRewardDiff, ha, hbest]
94       rw [hfun]
95       simp [ETC.centeredPairwiseRewardDiffVarianceProxy, ha, hbest]

97 /--
98 Canonical concrete argmax-oracle wrong-commit probability bound from
99 trace-level reward independence plus per-time centered reward sub-Gaussian
100 witnesses.
101 
102 This is the `ETC-WRONG-COMMIT-REWARD-LAW-SUBGAUSSIAN-BOUND` leaf. It composes
103 the reward-coordinate independence transfer, the centered reward sub-Gaussian
104 transfer, and the canonical wrong-commit probability consumer. It still leaves
105 the source of trace-level independence and centered reward sub-Gaussianity as
106 explicit assumptions.
107 -/
108 theorem prob_argmaxCommitOracle_ne_bestArm_le_filtered_sum_centeredDiffSubGaussianTail_of_reward_iIndepFun_centeredReward_subG
109     {Omega : Type u} {K : Nat}
110     [MeasurableSpace Omega]
111     (hK : 0 < K)
112     (mu : MeasureTheory.Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
113       [MeasureTheory.IsZeroOrProbabilityMeasure mu]
114     (spec : ETC.Spec K)
115     (model : FiniteBanditModel K)
116     (commitArm : Fin K)
117     (reward : Omega -> RewardTrace Rat)
118     (cReward : Fin K -> Nat -> NNReal)
119     (hexplorationPulls_pos : 0 < spec.explorationPulls)
120     (h_reward_indep :
121       ProbabilityTheory.iIndepFun
122         (fun t omega => reward omega t) mu)
123     (h_reward_subG :
124       forall b : Fin K, forall t, t < spec.explorationPulls * K ->
125         ProbabilityTheory.HasSubgaussianMGF
126           (fun omega : Omega => (((reward omega t - model.mean b : Rat) : Real)))
127           (cReward b t) mu) :
128     mu {omega : Omega |
129         (ETC.argmaxCommitOracle hK).choose
130           (fun a : Fin K =>
131             ETC.empMeanAtExploration spec commitArm (reward omega) a) =
132           model.bestArm -> False} <=
133     ((Finset.univ : Finset (Fin K)).filter
134       (fun a : Fin K => a = model.bestArm -> False)).sum
135       (ETC.centeredDiffSubGaussianTail spec model
136         (ETC.centeredPairwiseRewardDiffVarianceProxy spec model commitArm cReward)) := by
137   exact
138     ETC.prob_argmaxCommitOracle_ne_bestArm_le_filtered_sum_centeredDiffSubGaussianTail
139       hK mu spec model commitArm reward
140       (ETC.centeredPairwiseRewardDiffVarianceProxy spec model commitArm cReward)
141       hexplorationPulls_pos
142       (ETC.iIndepFun_centeredPairwiseRewardDiff_of_iIndepFun_reward
143         mu spec model commitArm reward h_reward_indep)
144       (fun a hne t ht =>
145         ETC.centeredPairwiseRewardDiff_hasSubgaussianMGF_of_centeredReward
146           mu spec model commitArm reward cReward a t hne
147           (fun b _hb => h_reward_subG b t (Finset.mem_range.mp ht)))
