16 theorem logPlus_mono {x y : ℝ} (hxy : x ≤ y) : logPlus x ≤ logPlus y :=
17   log_le_log (lt_of_lt_of_le zero_lt_one (le_max_left 1 x)) (max_le_max le_rfl hxy)
18 
19 theorem exp_neg_logPlus_inv_le (x : ℝ) (hx : 0 < x) :
20     exp (-logPlus (1 / x)) ≤ x := by
21   have h := log_le_log (show 0 < 1 / x by positivity) (le_max_right 1 (1 / x))
22   have he : -logPlus (1 / x) ≤ log x := by
23     simpa [logPlus, log_div] using neg_le_neg h
24   exact (exp_le_exp.mpr he).trans_eq (exp_log hx)

26 /-- Scaled source confidence barrier, avoiding division by the sample count. -/
27 noncomputable def peelingBarrier (δ gap s : ℝ) : ℝ :=
28   sqrt (4 * s * logPlus (1 / (s * δ))) + s * gap
29 
30 /-- Lower barrier common to one dyadic block m<=s<=2m. -/
31 noncomputable def blockBarrier (δ gap m : ℝ) : ℝ :=
32   sqrt (4 * m * logPlus (1 / (2 * m * δ))) + m * gap
33 
34 theorem blockBarrier_pos (δ gap m : ℝ) (hm : 0 < m) (hg : 0 < gap) :
35     0 < blockBarrier δ gap m := by
36   unfold blockBarrier
37   exact add_pos_of_nonneg_of_pos (sqrt_nonneg _) (mul_pos hm hg)
38 
39 theorem blockBarrier_le_peelingBarrier (δ gap m s : ℝ)
40     (hδ : 0 < δ) (hg : 0 ≤ gap) (hm : 0 < m) (hms : m ≤ s) (hsm : s ≤ 2*m) :
41     blockBarrier δ gap m ≤ peelingBarrier δ gap s := by
42   have hs : 0 < s := hm.trans_le hms
43   have hlog : logPlus (1 / (2*m*δ)) ≤ logPlus (1 / (s*δ)) := by
44     apply logPlus_mono
45     exact one_div_le_one_div_of_le (mul_pos hs hδ) (mul_le_mul_of_nonneg_right hsm hδ.le)
46   apply add_le_add
47   · apply sqrt_le_sqrt
48     exact mul_le_mul (by linarith : 4*m ≤ 4*s) hlog
49       (logPlus_nonneg _) (by positivity)
50   · exact mul_le_mul_of_nonneg_right hms hg

52 /-- Scalar exponential bound used after Doob on one dyadic block. -/
53 theorem exp_neg_blockBarrier_sq_le (δ gap m : ℝ)
54     (hδ : 0 < δ) (hg : 0 ≤ gap) (hm : 0 < m) :
55     exp (-(blockBarrier δ gap m)^2 / (4*m)) ≤
56       (2*m*δ) * exp (-(m*gap^2/4)) := by
57   have hl := logPlus_nonneg (1 / (2*m*δ))
58   have hs := sq_sqrt (show 0 ≤ 4*m*logPlus (1/(2*m*δ)) by positivity)
59   have hcross : 0 ≤ sqrt (4*m*logPlus (1/(2*m*δ))) * (m*gap) := by positivity
60   have hsq : 4*m*logPlus (1/(2*m*δ)) + m^2*gap^2 ≤
61       (blockBarrier δ gap m)^2 := by
62     unfold blockBarrier
63     nlinarith
64   have hexponent : -(blockBarrier δ gap m)^2 / (4*m) ≤
65       -logPlus (1/(2*m*δ)) - m*gap^2/4 := by
66     apply (div_le_iff₀ (show 0 < 4*m by positivity)).2
67     nlinarith
68   calc
69     exp (-(blockBarrier δ gap m)^2 / (4*m)) ≤
70         exp (-logPlus (1/(2*m*δ)) - m*gap^2/4) := exp_le_exp.mpr hexponent
71     _ = exp (-logPlus (1/(2*m*δ))) * exp (-(m*gap^2/4)) := by
72       rw [sub_eq_add_neg, exp_add]
73     _ ≤ (2*m*δ) * exp (-(m*gap^2/4)) :=
74       mul_le_mul_of_nonneg_right (exp_neg_logPlus_inv_le _ (by positivity)) (exp_pos _).le

76 /-- Centered partial sum in the source's one-based sample convention. -/
77 noncomputable def peelingSum {Ω : Type*} (X : ℕ → Ω → ℝ) (s : ℕ) (ω : Ω) : ℝ :=
78   ∑ j ∈ range s, X (j+1) ω

80 /-- Maximal partial-sum event dominating one dyadic block. -/
81 def blockBadEvent {Ω : Type*} (X : ℕ → Ω → ℝ) (δ gap : ℝ) (m : ℕ) : Set Ω :=
82   {ω | ∃ s, s ≤ 2*m ∧ peelingSum X s ω + blockBarrier δ gap (m : ℝ) ≤ 0}

86 theorem measure_blockBadEvent_le (X : ℕ → Ω → ℝ)
87     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
88     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
89     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
90     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) (m : ℕ) (hm : 0 < m) :
91     μ (blockBadEvent X δ gap m) ≤
92       ENNReal.ofReal ((2*(m : ℝ)*δ) * exp (-((m : ℝ)*gap^2/4))) := by
93   have hmR : 0 < (m : ℝ) := Nat.cast_pos.mpr hm
94   have hneg : iIndepFun (fun i ω => -X i ω) μ :=
95     hind.comp (fun (_ : ℕ) (x : ℝ) => -x) (fun _ => measurable_neg)
96   have h := Concentration.measure_exists_le_independent_partialSum_ge_le_subgaussian
97     (fun i ω => -X i ω) (fun i => (hXm i).neg) hneg
98     (fun i => by simp only [integral_neg, hmean, neg_zero])
99     1 (by norm_num) (fun i => (hsubG i).neg) (2*m) (by omega)
100     (blockBarrier δ gap (m : ℝ)) (blockBarrier_pos δ gap _ hmR hg)
101   simp only [NNReal.coe_one, mul_one, Nat.cast_mul, Nat.cast_ofNat] at h
102   rw [show (2 : ℝ)*(2*(m : ℝ)) = 4*(m : ℝ) by ring] at h
103   refine (measure_mono ?_).trans (h.trans (ENNReal.ofReal_le_ofReal
104     (exp_neg_blockBarrier_sq_le δ gap _ hδ hg.le hmR)))
105   rintro ω ⟨s, hs, hb⟩
106   refine ⟨s, hs, ?_⟩
107   simp only [sum_neg_distrib]
108   change peelingSum X s ω + blockBarrier δ gap (m : ℝ) ≤ 0 at hb
109   unfold peelingSum at hb
110   linarith

112 /-- All positive-sample source bad events, in scaled partial-sum form. -/
113 def scaledBadEvent (X : ℕ → Ω → ℝ) (δ gap : ℝ) : Set Ω :=
114   {ω | ∃ s, 0 < s ∧ peelingSum X s ω + peelingBarrier δ gap (s : ℝ) ≤ 0}

116 /-- Countable dyadic peeling with the printed constant 15. -/
117 theorem measure_scaledBadEvent_le_fifteen (X : ℕ → Ω → ℝ)
118     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
119     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
120     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
121     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) :
122     μ (scaledBadEvent X δ gap) ≤ ENNReal.ofReal (15*δ/gap^2) := by
123   have hcover : scaledBadEvent X δ gap ⊆ ⋃ j : ℕ, blockBadEvent X δ gap (2^j) := by
124     rintro ω ⟨s, hs, hb⟩
125     let j := Nat.log 2 s
126     have hlow : 2^j ≤ s := Nat.pow_log_le_self 2 (Nat.ne_of_gt hs)
127     have hupp : s ≤ 2*(2^j) := by
128       have h := (Nat.lt_pow_succ_log_self (by decide : 1 < 2) s).le
129       simpa [j, pow_succ, Nat.mul_comm] using h
130     apply Set.mem_iUnion.mpr
131     refine ⟨j, s, hupp, ?_⟩
132     have hbar := blockBarrier_le_peelingBarrier δ gap ((2^j : ℕ) : ℝ) (s : ℝ)
133       hδ hg.le (by positivity) (by exact_mod_cast hlow) (by exact_mod_cast hupp)
134     linarith
135   refine (measure_mono hcover).trans ((measure_iUnion_le _).trans ?_)
136   refine (ENNReal.tsum_le_tsum (fun j => ?_)).trans
137     (Concentration.tsum_moss_peeling_exponential_le_fifteen δ gap hδ.le hg)
138   have h := measure_blockBadEvent_le X hXm hind hmean hsubG δ gap hδ hg (2^j) (by positivity)
139   convert h using 1 <;> push_cast <;> congr 1 <;> rw [pow_succ] <;> ring

141 /-- The actual empirical-mean bad event appearing in source Lemma 9.3. -/
142 def meanBadEvent (X : ℕ → Ω → ℝ) (δ gap : ℝ) : Set Ω :=
143   {ω | ∃ s : ℕ, 0 < s ∧ peelingSum X s ω / (s : ℝ) +
144     sqrt (4 / (s : ℝ) * logPlus (1 / ((s : ℝ)*δ))) + gap ≤ 0}
145 
146 theorem meanBadEvent_subset_scaledBadEvent (X : ℕ → Ω → ℝ) (δ gap : ℝ) :
147     meanBadEvent X δ gap ⊆ scaledBadEvent X δ gap := by
148   rintro ω ⟨s, hs, hb⟩
149   have hsR : 0 < (s : ℝ) := Nat.cast_pos.mpr hs
150   have hscale : (s : ℝ) * sqrt (4 / (s : ℝ) * logPlus (1 / ((s : ℝ)*δ))) =
151       sqrt (4 * (s : ℝ) * logPlus (1 / ((s : ℝ)*δ))) := by
152     calc
153       _ = sqrt ((s : ℝ)^2) * sqrt (4 / (s : ℝ) * logPlus (1 / ((s : ℝ)*δ))) := by
154         rw [sqrt_sq hsR.le]
155       _ = sqrt ((s : ℝ)^2 * (4 / (s : ℝ) * logPlus (1 / ((s : ℝ)*δ)))) :=
156         (sqrt_mul (sq_nonneg (s : ℝ)) _).symm
157       _ = _ := by congr 1; field_simp
158   refine ⟨s, hs, ?_⟩
159   have h := mul_le_mul_of_nonneg_left hb hsR.le
160   unfold peelingBarrier
161   rw [mul_add, mul_add, mul_div_cancel₀ _ (ne_of_gt hsR), hscale, mul_zero] at h
162   simpa only [add_assoc] using h

164 /-- Source Lemma 9.3, with the printed constant and actual mean/radius event.
165 The proof works for every positive delta, hence in particular delta in (0,1).
166 The explicit centered-coordinate contracts are later instantiated by MOSS. -/
167 theorem measure_meanBadEvent_le_fifteen (X : ℕ → Ω → ℝ)
168     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
169     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
170     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
171     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) :
172     μ (meanBadEvent X δ gap) ≤ ENNReal.ofReal (15*δ/gap^2) :=
173   (measure_mono (meanBadEvent_subset_scaledBadEvent X δ gap)).trans
174     (measure_scaledBadEvent_le_fifteen X hXm hind hmean hsubG δ gap hδ hg)
