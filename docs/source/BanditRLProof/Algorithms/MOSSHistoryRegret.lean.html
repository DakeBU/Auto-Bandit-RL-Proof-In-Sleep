9 /-- Inclusive finite-history counts agree with trace counts through t+1. -/
10 theorem finiteHistoryPullCountENNReal_trace {k : ℕ} {Reward : Type*}
11     (action : ActionTrace (Fin k)) (reward : RewardTrace Reward) (t : ℕ) (a : Fin k) :
12     LowerBounds.finiteHistoryPullCountENNReal t
13       (History.finitePairHistoryOfTrace action reward t) a = (pullCount action a (t+1) : ℝ≥0∞) := by
14   induction t with
15   | zero =>
16       simp [LowerBounds.finiteHistoryPullCountENNReal, History.finitePairHistoryOfTrace,
17         pullCount_succ, pullCount_zero]
18   | succ t ih =>
19       change LowerBounds.finiteHistoryPullCountENNReal t
20         (History.finitePairHistoryOfTrace action reward t) a +
21         (if action (t+1) = a then 1 else 0) = _
22       rw [ih, pullCount_succ, Nat.cast_add]
23       split_ifs <;> simp_all

27 /-- The history gap functional is exactly the executed real pseudo-regret. -/
28 theorem canonicalHistory_gapRegret_toReal {k : ℕ} (hk : 0 < k) (n t : ℕ)
29     (mean : Fin k → ℝ) (best : Fin k) (hbest : ∀ a, mean a ≤ mean best)
30     (table : UCB.ArmRewardStream k) :
31     (LowerBounds.finiteHistoryGapPseudoRegret (fun a => mean best-mean a) t
32       (canonicalHistory hk n mean table t)).toReal =
33       realMeanRegret mean (canonicalAction hk n mean table) (t+1) := by
34   letI : Nonempty (Fin k) := ⟨best⟩
35   have hs : (⨆ a, mean a) = mean best :=
36     le_antisymm (ciSup_le hbest) (le_ciSup (Finite.bddAbove_range mean) best)
37   rw [LowerBounds.finiteHistoryGapPseudoRegret_toReal _ _ _ (fun a => sub_nonneg.mpr (hbest a)),
38     realMeanRegret_eq_sum_gap_mul_pullCount]
39   apply Finset.sum_congr rfl
40   intro a _
41   simp only [realMeanGap, hs, LowerBounds.finiteHistoryPullCountReal, canonicalHistory,
42     finiteHistoryPullCountENNReal_trace, ENNReal.toReal_natCast]

44 /-- Exact transport of the common-history expected regret to the reward table. -/
45 theorem canonicalGapExpectedRegret_eq_integral {k : ℕ} [NeZero k] (hk : 0 < k)
46     (n t : ℕ) (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν]
47     (mean : Fin k → ℝ) (best : Fin k) (hbest : ∀ a, mean a ≤ mean best) :
48     LowerBounds.canonicalGapExpectedPseudoRegretReal (historyAlgorithm hk n) ν
49       (fun a => mean best-mean a) t =
50     ∫ table, realMeanRegret mean (canonicalAction hk n mean table) (t+1)
51       ∂UCB.armStreamMeasure ν := by
52   let f := LowerBounds.finiteHistoryGapPseudoRegret (Reward := ℝ) (fun a => mean best-mean a) t
53   have hm : Measurable f := LowerBounds.measurable_finiteHistoryGapPseudoRegret _ _
54   have hf : ∀ history, f history < ∞ := fun history =>
55     lt_top_iff_ne_top.mpr (LowerBounds.finiteHistoryGapPseudoRegret_ne_top _ _ history)
56   change (∫⁻ history, f history ∂LowerBounds.canonicalBanditHistoryMeasure (historyAlgorithm hk n) ν t).toReal = _
57   rw [← integral_toReal hm.aemeasurable (Filter.Eventually.of_forall hf),
58     ← map_canonicalHistory_eq hk n mean ν t,
59     integral_map (measurable_canonicalHistory hk n mean t).aemeasurable hm.ennreal_toReal.aestronglyMeasurable]
60   apply integral_congr_ae
61   exact Filter.Eventually.of_forall (canonicalHistory_gapRegret_toReal hk n t mean best hbest)

63 /-- Source constant 39, now for the same history-law regret used by lower bounds. -/
64 theorem canonicalGapExpectedRegret_le {k : ℕ} [NeZero k] (hk : 0 < k)
65     (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν] (t : ℕ) (hkt : k ≤ t+1)
66     (mean : Fin k → ℝ) (best : Fin k) (hbest : ∀ a, mean a ≤ mean best)
67     (hmean : ∀ a, ∫ r, r ∂ν a = mean a)
68     (hsubG : ∀ a, HasSubgaussianMGF (fun r => r-mean a) 1 (ν a)) :
69     LowerBounds.canonicalGapExpectedPseudoRegretReal (historyAlgorithm hk (t+1)) ν
70       (fun a => mean best-mean a) t ≤
71       39*Real.sqrt (((t+1 : ℕ) : ℝ)*k) + ∑ a, (mean best-mean a) := by
72   rw [canonicalGapExpectedRegret_eq_integral hk (t+1) t ν mean best hbest]
73   exact integral_canonicalReward_regret_le hk ν (t+1) hkt mean best hbest hmean hsubG
