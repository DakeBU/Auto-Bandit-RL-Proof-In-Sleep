5 /-!
6 # Fixed-IID source adapter for the two-arm stochastic-gradient bandit
7 
8 This module realizes the reward model used by the two-arm source theorem from
9 an arm-indexed family of fixed probability laws.  The environment is the
10 stationary history environment over `Unit`; hence its initial and successor
11 reward fibers are exactly the selected arm law and cannot reveal a latent,
12 changing environment.
13 
14 The adapter is deliberately narrower than
15 `TwoArmBoundedFixedMeanEnvironmentContract`.  The latter also admits
16 history-dependent reward laws with fixed means, whereas this module produces
17 one fixed law per arm.  No global recurrence iteration or regret endpoint is
18 claimed here.
19 -/
20

26 /--
27 The fixed two-arm reward kernel, with a trivial environment coordinate added
28 for the measurable-history-environment interface.
29 -/
30 noncomputable def twoArmFixedIIDRewardKernel
31     (armLaw : Fin 2 -> Measure Real) : Kernel (Unit × Fin 2) Real :=
32   (UCB.finiteArmRealRewardKernel armLaw).comap Prod.snd measurable_snd
33 
34 @[simp]
35 theorem twoArmFixedIIDRewardKernel_apply
36     (armLaw : Fin 2 -> Measure Real) (env : Unit) (arm : Fin 2) :
37     twoArmFixedIIDRewardKernel armLaw (env, arm) = armLaw arm := by
38   rw [twoArmFixedIIDRewardKernel, Kernel.comap_apply,
39     UCB.finiteArmRealRewardKernel_apply]
40 
41 /-- Pointwise probability laws make the fixed two-arm kernel Markov. -/
42 theorem twoArmFixedIIDRewardKernel_isMarkov
43     (armLaw : Fin 2 -> Measure Real)
44     (hprob : forall arm, IsProbabilityMeasure (armLaw arm)) :
45     IsMarkovKernel (twoArmFixedIIDRewardKernel armLaw) := by
46   letI : IsMarkovKernel (UCB.finiteArmRealRewardKernel armLaw) :=
47     UCB.finiteArmRealRewardKernel_isMarkov armLaw hprob
48   unfold twoArmFixedIIDRewardKernel
49   infer_instance

51 /--
52 The fixed-IID source environment.  Conditional on the selected arm, every
53 round uses the same arm law and ignores the observed history.
54 -/
55 noncomputable def twoArmFixedIIDEnvironment
56     (armLaw : Fin 2 -> Measure Real)
57     (hprob : forall arm, IsProbabilityMeasure (armLaw arm)) :
58     Thompson.MeasurableHistoryEnvironment Unit (Fin 2) Real := by
59   letI : IsMarkovKernel (twoArmFixedIIDRewardKernel armLaw) :=
60     twoArmFixedIIDRewardKernel_isMarkov armLaw hprob
61   exact Thompson.stationaryMeasurableHistoryEnvironment
62     (twoArmFixedIIDRewardKernel armLaw)

90 /--
91 Fixed probability laws supported on `[-1,1]` with the stated arm means produce
92 the uniform bounded fixed-mean contract used by the compiled two-arm
93 recurrences.
94 -/
95 theorem twoArmFixedIIDEnvironment_contract
96     (armLaw : Fin 2 -> Measure Real)
97     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
98     (mean : Fin 2 -> Real)
99     (hbound : forall arm, ∀ᵐ reward ∂armLaw arm, |reward| <= 1)
100     (hmean : forall arm, integral (armLaw arm) id = mean arm) :
101     TwoArmBoundedFixedMeanEnvironmentContract
102       (twoArmFixedIIDEnvironment armLaw hprob) mean := by
103   constructor
104   · intro env selected
105     simpa using hbound selected
106   · intro env selected
107     simpa using hmean selected
108   · intro n env history selected
109     simpa using hbound selected
110   · intro n env history selected
111     simpa using hmean selected
112 
113 /-- Equation (5) on every generated fixed-IID successor history, with the
114 source-increment integrability premise discharged from the fixed reward-law
115 support.  This remains a one-step conditional-kernel identity; it does not
116 perform a global tower iteration. -/
117 theorem integral_twoArmFixedIIDHistoryStepKernel_sourceIncrement_eq_gapCoordinate
118     (armLaw : Fin 2 -> Measure Real)
119     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
120     (mean : Fin 2 -> Real)
121     (hbound : forall arm, ∀ᵐ reward ∂armLaw arm, |reward| <= 1)
122     (hmean : forall arm, integral (armLaw arm) id = mean arm)
123     (initialTheta : Fin 2 -> Real) (eta : Real)
124     (n : Nat) (history : History.FinitePairHistory (Fin 2) Real n)
125     (gap : Fin 2 -> Real) (bestMean : Real) (coordinate : Fin 2)
126     (hgap : forall action, gap action = bestMean - mean action) :
127     integral
128         (Thompson.measurableEnvironmentHistoryStepKernel
129           (historyAlgorithm initialTheta eta)
130           (twoArmFixedIIDEnvironment armLaw hprob) n ((), history))
131         (fun pair : Fin 2 × Real =>
132           sourceIncrement
133             (softmaxProbability
134               (historyParameter initialTheta eta n history))
135             pair.2 pair.1 coordinate) =
136       softmaxProbability
137           (historyParameter initialTheta eta n history) coordinate *
138         (instantaneousGap
139           (softmaxProbability
140             (historyParameter initialTheta eta n history)) gap -
141           gap coordinate) := by
142   let contract :=
143     twoArmFixedIIDEnvironment_contract armLaw hprob mean hbound hmean
144   apply
145     integral_measurableEnvironmentHistoryStepKernel_sourceIncrement_eq_gapCoordinate
146       initialTheta eta (twoArmFixedIIDEnvironment armLaw hprob)
147       n () history mean gap bestMean coordinate
148   · exact
149       integrable_measurableTwoArmHistoryStepKernel_sourceIncrement_of_contract
150         initialTheta eta (twoArmFixedIIDEnvironment armLaw hprob)
151         mean contract n () history coordinate
152   · intro selected
153     simpa using hmean selected
154   · exact hgap
