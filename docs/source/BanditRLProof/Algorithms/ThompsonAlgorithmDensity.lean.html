4 /-!
5 # Thompson posterior invariance from algorithm-density laws
6 
7 This module isolates the measure-theoretic core of LML's algorithm-density
8 posterior transport.  If the actual history law and the actual
9 history/environment joint law are obtained from their reference counterparts
10 by the same density depending only on history, then the reference and actual
11 environment posteriors agree at the actual history law.
12 -/

22 /--
23 Weighting the base measure of a composition product is the same as weighting
24 the product by the density pulled back through the first projection.
25 -/
26 theorem compProd_withDensity_left
27     {History : Type u} {Env : Type v}
28     [MeasurableSpace History] [MeasurableSpace Env]
29     (historyLaw : Measure History)
30     [SFinite historyLaw]
31     (posterior : ProbabilityTheory.Kernel History Env)
32     [ProbabilityTheory.IsSFiniteKernel posterior]
33     (density : History -> ENNReal) (hdensity : Measurable density) :
34     historyLaw.withDensity density ⊗ₘ posterior =
35       (historyLaw ⊗ₘ posterior).withDensity (density ∘ Prod.fst) := by
36   refine Measure.ext_of_lintegral _ fun f hf => ?_
37   calc
38     ∫⁻ pair, f pair ∂(historyLaw.withDensity density ⊗ₘ posterior) =
39         ∫⁻ history, ∫⁻ environment, f (history, environment) ∂posterior history
40           ∂(historyLaw.withDensity density) :=
41       Measure.lintegral_compProd hf
42     _ = ∫⁻ history, density history *
43           ∫⁻ environment, f (history, environment) ∂posterior history
44           ∂historyLaw :=
45       MeasureTheory.lintegral_withDensity_eq_lintegral_mul _ hdensity
46         hf.lintegral_kernel_prod_right'
47     _ = ∫⁻ history, ∫⁻ environment,
48           density history * f (history, environment) ∂posterior history
49           ∂historyLaw := by
50       apply lintegral_congr
51       intro history
52       exact (MeasureTheory.lintegral_const_mul _ (by fun_prop)).symm
53     _ = ∫⁻ pair, (density ∘ Prod.fst) pair * f pair
54           ∂(historyLaw ⊗ₘ posterior) :=
55       (Measure.lintegral_compProd ((hdensity.comp measurable_fst).mul hf)).symm
56     _ = ∫⁻ pair, f pair
57           ∂((historyLaw ⊗ₘ posterior).withDensity
58             (density ∘ Prod.fst)) :=
59       (MeasureTheory.lintegral_withDensity_eq_lintegral_mul _
60         (hdensity.comp measurable_fst) hf).symm

62 /--
63 Composing a kernel weighted by a density independent of its input is the same
64 as weighting the composed output measure.
65 -/
66 theorem comp_withDensity_history
67     {Env : Type u} {History : Type v}
68     [MeasurableSpace Env] [MeasurableSpace History]
69     (envLaw : Measure Env)
70     (historyKernel : ProbabilityTheory.Kernel Env History)
71     [ProbabilityTheory.IsSFiniteKernel historyKernel]
72     (density : History -> ENNReal) (hdensity : Measurable density) :
73     (historyKernel.withDensity (fun _ history => density history)) ∘ₘ envLaw =
74       (historyKernel ∘ₘ envLaw).withDensity density := by
75   ext historyEvent hhistoryEvent
76   rw [Measure.bind_apply hhistoryEvent (ProbabilityTheory.Kernel.aemeasurable _)]
77   simp_rw [ProbabilityTheory.Kernel.withDensity_apply' historyKernel
78     (show Measurable
79         (Function.uncurry (fun _ : Env => fun history => density history)) from
80       hdensity.comp measurable_snd)]
81   rw [MeasureTheory.withDensity_apply density hhistoryEvent]
82   rw [← MeasureTheory.lintegral_indicator hhistoryEvent]
83   rw [Measure.lintegral_bind historyKernel.aemeasurable
84     ((hdensity.indicator hhistoryEvent).aemeasurable)]
85   congr 1
86   funext environment
87   rw [MeasureTheory.lintegral_indicator hhistoryEvent]

89 /--
90 Swapping a joint law weighted by its second coordinate moves the density to
91 the first coordinate of the swapped law.
92 -/
93 theorem map_swap_withDensity_snd
94     {Env : Type u} {History : Type v}
95     [MeasurableSpace Env] [MeasurableSpace History]
96     (joint : Measure (Env × History))
97     (density : History -> ENNReal) (hdensity : Measurable density) :
98     (joint.withDensity (density ∘ Prod.snd)).map Prod.swap =
99       (joint.map Prod.swap).withDensity (density ∘ Prod.fst) := by
100   ext event hevent
101   rw [Measure.map_apply measurable_swap hevent]
102   rw [MeasureTheory.withDensity_apply _ (measurable_swap hevent)]
103   rw [MeasureTheory.withDensity_apply _ hevent]
104   rw [MeasureTheory.setLIntegral_map hevent
105     (hdensity.comp measurable_fst) measurable_swap]
106   rfl

108 /--
109 Composition products transport an a.e. output-only kernel density directly,
110 without requiring an `IsSFiniteKernel` instance for the weighted kernel.
111 -/
112 theorem compProd_eq_compProd_withDensity_snd_of_ae_eq
113     {Env : Type u} {History : Type v}
114     [MeasurableSpace Env] [MeasurableSpace History]
115     (envLaw : Measure Env) [SFinite envLaw]
116     (actualHistoryKernel referenceHistoryKernel :
117       ProbabilityTheory.Kernel Env History)
118     [ProbabilityTheory.IsSFiniteKernel actualHistoryKernel]
119     [ProbabilityTheory.IsSFiniteKernel referenceHistoryKernel]
120     (density : History -> ENNReal) (hdensity : Measurable density)
121     (hkernel : actualHistoryKernel =ᵐ[envLaw]
122       referenceHistoryKernel.withDensity
123         (fun _ history => density history)) :
124     envLaw ⊗ₘ actualHistoryKernel =
125       (envLaw ⊗ₘ referenceHistoryKernel).withDensity
126         (density ∘ Prod.snd) := by
127   refine Measure.ext_of_lintegral _ fun f hf => ?_
128   calc
129     ∫⁻ pair, f pair ∂(envLaw ⊗ₘ actualHistoryKernel) =
130         ∫⁻ environment, ∫⁻ history, f (environment, history)
131           ∂actualHistoryKernel environment ∂envLaw :=
132       Measure.lintegral_compProd hf
133     _ = ∫⁻ environment, ∫⁻ history, f (environment, history)
134           ∂(referenceHistoryKernel environment).withDensity density
135           ∂envLaw := by
136       apply lintegral_congr_ae
137       filter_upwards [hkernel] with environment h_environment
138       rw [h_environment, ProbabilityTheory.Kernel.withDensity_apply _
139         (show Measurable
140             (Function.uncurry
141               (fun _ : Env => fun history => density history)) from
142           hdensity.comp measurable_snd)]
143     _ = ∫⁻ environment, ∫⁻ history,
144           density history * f (environment, history)
145           ∂referenceHistoryKernel environment ∂envLaw := by
146       apply lintegral_congr
147       intro environment
148       exact MeasureTheory.lintegral_withDensity_eq_lintegral_mul _ hdensity
149         (hf.comp (measurable_const.prodMk measurable_id))
150     _ = ∫⁻ pair, (density ∘ Prod.snd) pair * f pair
151           ∂(envLaw ⊗ₘ referenceHistoryKernel) :=
152       (Measure.lintegral_compProd ((hdensity.comp measurable_snd).mul hf)).symm
153     _ = ∫⁻ pair, f pair
154           ∂((envLaw ⊗ₘ referenceHistoryKernel).withDensity
155             (density ∘ Prod.snd)) :=
156       (MeasureTheory.lintegral_withDensity_eq_lintegral_mul _
157         (hdensity.comp measurable_snd) hf).symm

159 /--
160 The exact law interface produced by an algorithm-density/change-of-algorithm
161 argument.  Both the history marginal and the history/environment joint law are
162 weighted by the same measurable density depending only on history.
163 -/
164 structure AlgorithmDensityPosteriorSource
165     {Omega : Type u} {OmegaRef : Type v}
166     {History : Type w} {Env : Type x}
167     [MeasurableSpace Omega] [MeasurableSpace OmegaRef]
168     [MeasurableSpace History] [MeasurableSpace Env]
169     (mu : Measure Omega)
170     (env : Omega -> Env) (history : Omega -> History)
171     (referenceMu : Measure OmegaRef)
172     (referenceEnv : OmegaRef -> Env)
173     (referenceHistory : OmegaRef -> History) where
174   density : History -> ENNReal
175   density_measurable : Measurable density
176   history_map_eq_withDensity :
177     mu.map history = (referenceMu.map referenceHistory).withDensity density
178   historyEnv_map_eq_withDensity :
179     mu.map (fun omega => (history omega, env omega)) =
180       (referenceMu.map
181         (fun omega => (referenceHistory omega, referenceEnv omega))).withDensity
182           (density ∘ Prod.fst)

184 /--
185 Construct the two algorithm-density pushforward laws from a closer-to-process
186 interface: the actual and reference environment marginals agree, and the
187 actual conditional history kernel is the reference conditional history kernel
188 weighted by one history-only density.
189 -/
190 noncomputable def algorithmDensityPosteriorSource_of_condDistrib_history_withDensity
191     {Omega : Type u} {OmegaRef : Type v}
192     {History : Type w} {Env : Type x}
193     [MeasurableSpace Omega] [MeasurableSpace OmegaRef]
194     [MeasurableSpace History] [StandardBorelSpace History] [Nonempty History]
195     [MeasurableSpace Env]
196     (mu : Measure Omega) [IsFiniteMeasure mu]
197     (env : Omega -> Env) (history : Omega -> History)
198     (henv : Measurable env) (hhistory : Measurable history)
199     (referenceMu : Measure OmegaRef) [IsFiniteMeasure referenceMu]
200     (referenceEnv : OmegaRef -> Env)
201     (referenceHistory : OmegaRef -> History)
202     (hreferenceEnv : Measurable referenceEnv)
203     (hreferenceHistory : Measurable referenceHistory)
204     (density : History -> ENNReal) (hdensity : Measurable density)
205     (henvLaw : mu.map env = referenceMu.map referenceEnv)
206     (hcond :
207       ProbabilityTheory.condDistrib history env mu =ᵐ[mu.map env]
208         (ProbabilityTheory.condDistrib
209           referenceHistory referenceEnv referenceMu).withDensity
210             (fun _ history => density history)) :
211     AlgorithmDensityPosteriorSource mu env history
212       referenceMu referenceEnv referenceHistory := by
213   let actualHistoryKernel := ProbabilityTheory.condDistrib history env mu
214   let referenceHistoryKernel := ProbabilityTheory.condDistrib
215     referenceHistory referenceEnv referenceMu
216   let weightedReferenceHistoryKernel := referenceHistoryKernel.withDensity
217     (fun _ history => density history)
218   have hhistoryLaw :
219       mu.map history =
220         (referenceMu.map referenceHistory).withDensity density := by
221     calc
222       mu.map history = actualHistoryKernel ∘ₘ mu.map env := by
223         simpa only [actualHistoryKernel] using
224           (ProbabilityTheory.condDistrib_comp_map
225             henv.aemeasurable hhistory.aemeasurable).symm
226       _ = weightedReferenceHistoryKernel ∘ₘ mu.map env := by
227         exact Measure.comp_congr hcond
228       _ = weightedReferenceHistoryKernel ∘ₘ
229           referenceMu.map referenceEnv := by rw [henvLaw]
230       _ = (referenceHistoryKernel ∘ₘ
231           referenceMu.map referenceEnv).withDensity density := by
232         exact comp_withDensity_history
233           (referenceMu.map referenceEnv) referenceHistoryKernel density hdensity
234       _ = (referenceMu.map referenceHistory).withDensity density := by
235         rw [ProbabilityTheory.condDistrib_comp_map
236           hreferenceEnv.aemeasurable hreferenceHistory.aemeasurable]
237   have hactualEnvHistory :
238       mu.map (fun omega => (env omega, history omega)) =
239         (referenceMu.map
240           (fun omega => (referenceEnv omega, referenceHistory omega))).withDensity
241             (density ∘ Prod.snd) := by
242     calc
243       mu.map (fun omega => (env omega, history omega)) =
244           mu.map env ⊗ₘ actualHistoryKernel := by
245         simpa only [actualHistoryKernel] using
246           (ProbabilityTheory.compProd_map_condDistrib
247             (μ := mu) (X := env) (Y := history) hhistory.aemeasurable).symm
248       _ = (mu.map env ⊗ₘ referenceHistoryKernel).withDensity
249           (density ∘ Prod.snd) :=
250         compProd_eq_compProd_withDensity_snd_of_ae_eq
251           (mu.map env) actualHistoryKernel referenceHistoryKernel
252           density hdensity hcond
253       _ = (referenceMu.map referenceEnv ⊗ₘ
254           referenceHistoryKernel).withDensity (density ∘ Prod.snd) := by
255         rw [henvLaw]
256       _ = (referenceMu.map
257           (fun omega => (referenceEnv omega, referenceHistory omega))).withDensity
258             (density ∘ Prod.snd) := by
259         rw [ProbabilityTheory.compProd_map_condDistrib
260           (μ := referenceMu) (X := referenceEnv) (Y := referenceHistory)
261           hreferenceHistory.aemeasurable]
262   have hhistoryEnvLaw :
263       mu.map (fun omega => (history omega, env omega)) =
264         (referenceMu.map
265           (fun omega => (referenceHistory omega, referenceEnv omega))).withDensity
266             (density ∘ Prod.fst) := by
267     calc
268       mu.map (fun omega => (history omega, env omega)) =
269           (mu.map (fun omega => (env omega, history omega))).map Prod.swap := by
270         rw [Measure.map_map measurable_swap (henv.prodMk hhistory)]
271         rfl
272       _ = ((referenceMu.map
273           (fun omega => (referenceEnv omega, referenceHistory omega))).withDensity
274             (density ∘ Prod.snd)).map Prod.swap := by
275         rw [hactualEnvHistory]
276       _ = ((referenceMu.map
277           (fun omega => (referenceEnv omega, referenceHistory omega))).map
278             Prod.swap).withDensity (density ∘ Prod.fst) :=
279         map_swap_withDensity_snd _ density hdensity
280       _ = (referenceMu.map
281           (fun omega => (referenceHistory omega, referenceEnv omega))).withDensity
282             (density ∘ Prod.fst) := by
283         rw [Measure.map_map measurable_swap
284           (hreferenceEnv.prodMk hreferenceHistory)]
285         rfl
286   exact
287     { density := density
288       density_measurable := hdensity
289       history_map_eq_withDensity := hhistoryLaw
290       historyEnv_map_eq_withDensity := hhistoryEnvLaw }

292 /--
293 The reference environment posterior equals the actual posterior whenever the
294 two algorithm-density pushforward laws use the same history density.
295 -/
296 theorem referencePosterior_ae_eq_condDistrib_of_algorithmDensitySource
297     {Omega : Type u} {OmegaRef : Type v}
298     {History : Type w} {Env : Type x}
299     [MeasurableSpace Omega] [MeasurableSpace OmegaRef]
300     [MeasurableSpace History]
301     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
302     (mu : Measure Omega) [IsFiniteMeasure mu]
303     (env : Omega -> Env) (history : Omega -> History)
304     (henv : Measurable env)
305     (referenceMu : Measure OmegaRef) [IsFiniteMeasure referenceMu]
306     (referenceEnv : OmegaRef -> Env)
307     (referenceHistory : OmegaRef -> History)
308     (hreferenceEnv : Measurable referenceEnv)
309     (hreferenceHistory : Measurable referenceHistory)
310     (source : AlgorithmDensityPosteriorSource mu env history
311       referenceMu referenceEnv referenceHistory) :
312     (referencePosterior referenceMu referenceEnv referenceHistory
313       hreferenceEnv hreferenceHistory).kernel =ᵐ[mu.map history]
314       ProbabilityTheory.condDistrib env history mu := by
315   let referenceHistoryLaw := referenceMu.map referenceHistory
316   let posterior := ProbabilityTheory.condDistrib
317     referenceEnv referenceHistory referenceMu
318   have hreferenceJoint :
319       referenceMu.map
320           (fun omega => (referenceHistory omega, referenceEnv omega)) =
321         referenceHistoryLaw ⊗ₘ posterior := by
322     simpa only [referenceHistoryLaw, posterior] using
323       (ProbabilityTheory.compProd_map_condDistrib
324         (μ := referenceMu) (X := referenceHistory) (Y := referenceEnv)
325         hreferenceEnv.aemeasurable).symm
326   letI : IsFiniteMeasure
327       (referenceHistoryLaw.withDensity source.density) :=
328     ⟨by
329       rw [← source.history_map_eq_withDensity]
330       exact measure_lt_top (mu.map history) Set.univ⟩
331   have hactualJoint :
332       mu.map (fun omega => (history omega, env omega)) =
333         mu.map history ⊗ₘ posterior := by
334     calc
335       _ = (referenceMu.map
336             (fun omega => (referenceHistory omega, referenceEnv omega))).withDensity
337             (source.density ∘ Prod.fst) :=
338           source.historyEnv_map_eq_withDensity
339       _ = (referenceHistoryLaw ⊗ₘ posterior).withDensity
340             (source.density ∘ Prod.fst) := by rw [hreferenceJoint]
341       _ = referenceHistoryLaw.withDensity source.density ⊗ₘ posterior :=
342           (compProd_withDensity_left referenceHistoryLaw posterior
343             source.density source.density_measurable).symm
344       _ = mu.map history ⊗ₘ posterior := by
345           rw [← source.history_map_eq_withDensity]
346   have hactualPosterior :
347       ProbabilityTheory.condDistrib env history mu =ᵐ[mu.map history]
348         posterior :=
349     ProbabilityTheory.condDistrib_ae_eq_of_measure_eq_compProd
350       history henv.aemeasurable hactualJoint
351   simpa only [referencePosterior_kernel, posterior] using hactualPosterior.symm
