6 /-!
7 # Two-arm SGB Theorem 2: Appendix-C Step-1 starvation consumer
8 
9 This module isolates the deterministic consumer in Step 1 of Appendix C of
10 Baudry--Johnson--Vary--Pike-Burke--Rebeschini (NeurIPS 2025).  On the actual
11 generated SGB action/reward trace, once exactly `n` optimal-arm pulls have
12 occurred, a path with no later optimal-arm pull has exactly
13 `Delta * (T - n)` sampled pseudo-regret.
14 
15 The source also lower-bounds the conditional probability of that no-return
16 path by `1 / 2` when the post-pull optimal-arm probability is at most
17 `1 / (2*T)`.  That statement needs a finite-future conditional-law/tower
18 bridge for the generated adaptive trajectory.  The existing API exposes the
19 one-step conditional action law but not yet the required stopped-prefix
20 future cylinder law.  No probability lower bound, phase producer, or
21 Theorem-2 endpoint is claimed in this file.
22 -/

32 /-- The action coordinate of the canonical SGB trajectory. -/
33 def twoArmGeneratedAction
34     {Env : Type v}
35     (sample : Env × ((k : Nat) → Fin 2 × Real)) : ActionTrace (Fin 2) :=
36   fun t => (sample.2 t).1
37 
38 /-- Number of optimal-arm (`0`) pulls in the first `horizon` generated rounds. -/
39 def twoArmOptimalPullCount
40     {Env : Type v} (horizon : Nat)
41     (sample : Env × ((k : Nat) → Fin 2 × Real)) : Nat :=
42   pullCount (twoArmGeneratedAction sample) 0 horizon
43 
44 /-- The source Step-1 threshold `1 / (2*T)`. -/
45 def twoArmStepOneThreshold (horizon : Nat) : Real :=
46   1 / (2 * (horizon : Real))
47 
48 /--
49 At a chronological prefix ending at `prefix`, the source Step-1 trigger says
50 that the next optimal-arm probability is at most `1/(2*T)` and that exactly
51 `n` optimal-arm pulls have occurred through that prefix.
52 
53 The distinction between chronological `prefix` and pull index `n` is
54 intentional and source-critical.
55 -/
56 def twoArmStepOneTriggerEvent
57     {Env : Type v} [MeasurableSpace Env]
58     (eta : Real) (cutoff n horizon : Nat) :
59     Set (Env × ((k : Nat) → Fin 2 × Real)) :=
60   {sample |
61     twoArmSuccessProbability eta cutoff sample ≤
62         twoArmStepOneThreshold horizon ∧
63       twoArmOptimalPullCount (cutoff + 1) sample = n}
64 
65 /--
66 The measurable part of the Appendix-C Step-1 starvation event: a trigger
67 prefix occurs and the total number of optimal-arm pulls by `horizon` remains
68 exactly `n`.  Equality of the two pull counts is the finite-trace statement
69 that there is no later optimal-arm selection.
70 -/
71 def twoArmStepOneStarvationEvent
72     {Env : Type v} [MeasurableSpace Env]
73     (eta : Real) (cutoff n horizon : Nat) :
74     Set (Env × ((k : Nat) → Fin 2 × Real)) :=
75   {sample |
76     sample ∈ twoArmStepOneTriggerEvent (Env := Env) eta cutoff n horizon ∧
77       twoArmOptimalPullCount horizon sample = n}

86 theorem measurable_twoArmOptimalPullCount
87     {Env : Type v} [MeasurableSpace Env] (horizon : Nat) :
88     Measurable
89       (twoArmOptimalPullCount (Env := Env) horizon) := by
90   induction horizon with
91   | zero =>
92       change Measurable
93         (fun _sample : Env × ((k : Nat) → Fin 2 × Real) => (0 : Nat))
94       exact measurable_const
95   | succ t ih =>
96       have hselected : Measurable
97           (fun sample : Env × ((k : Nat) → Fin 2 × Real) =>
98             twoArmGeneratedAction sample t) :=
99         measurable_twoArmGeneratedAction t
100       have hzero : MeasurableSet
101           {sample : Env × ((k : Nat) → Fin 2 × Real) |
102             twoArmGeneratedAction sample t = 0} :=
103         (measurableSet_singleton (0 : Fin 2)).preimage hselected
104       have hincrement : Measurable
105           (fun sample : Env × ((k : Nat) → Fin 2 × Real) =>
106             if twoArmGeneratedAction sample t = 0 then (1 : Nat) else 0) :=
107         Measurable.ite hzero measurable_const measurable_const
108       simpa [twoArmOptimalPullCount, pullCount_succ] using ih.add hincrement

136 /-- The below-threshold count event is the finite disjoint-by-value partition
137 into its exact terminal-count fibers. -/
138 theorem twoArmOptimalPullCountBelowEvent_eq_iUnion_terminalCount
139     {Env : Type v} [MeasurableSpace Env] (m horizon : Nat) :
140     twoArmOptimalPullCountBelowEvent (Env := Env) m horizon =
141       ⋃ n : Fin m,
142         twoArmTerminalOptimalPullCountEvent (Env := Env) (n : Nat) horizon := by
143   ext sample
144   constructor
145   · intro hbelow
146     change twoArmOptimalPullCount horizon sample < m at hbelow
147     let n : Fin m := ⟨twoArmOptimalPullCount horizon sample, hbelow⟩
148     refine Set.mem_iUnion.2 ⟨n, ?_⟩
149     change twoArmOptimalPullCount horizon sample = (n : Nat)
150     rfl
151   · intro hunion
152     rcases Set.mem_iUnion.1 hunion with ⟨n, hn⟩
153     change twoArmOptimalPullCount horizon sample = (n : Nat) at hn
154     change twoArmOptimalPullCount horizon sample < m
155     simpa [hn] using n.isLt

190 theorem twoArmSampledPseudoRegret_nonneg
191     {Env : Type v} (Delta : Real) (hDelta : 0 ≤ Delta)
192     (horizon : Nat)
193     (sample : Env × ((k : Nat) → Fin 2 × Real)) :
194     0 ≤ twoArmSampledPseudoRegret Delta horizon sample := by
195   unfold twoArmSampledPseudoRegret
196   apply Finset.sum_nonneg
197   intro t _ht
198   by_cases hzero : twoArmGeneratedAction sample t = 0
199   · change (sample.2 t).1 = 0 at hzero
200     simp [twoArmActionGap, hzero]
201   · change (sample.2 t).1 ≠ 0 at hzero
202     simpa [twoArmActionGap, hzero] using hDelta

204 /-- The two-arm sampled pseudo-regret is exactly the gap times arm-`1` pulls. -/
205 theorem twoArmSampledPseudoRegret_eq_gap_mul_suboptimalPullCount
206     {Env : Type v} (Delta : Real) (horizon : Nat)
207     (sample : Env × ((k : Nat) → Fin 2 × Real)) :
208     twoArmSampledPseudoRegret Delta horizon sample =
209       Delta * (pullCount (twoArmGeneratedAction sample) 1 horizon : Real) := by
210   induction horizon with
211   | zero => simp [twoArmSampledPseudoRegret, pullCount]
212   | succ t ih =>
213       change
214         (Finset.range (t + 1)).sum (fun s =>
215           twoArmActionGap Delta (twoArmGeneratedAction sample s)) = _
216       rw [Finset.sum_range_succ]
217       change twoArmSampledPseudoRegret Delta t sample +
218         twoArmActionGap Delta (twoArmGeneratedAction sample t) = _
219       rw [ih, pullCount_succ]
220       by_cases hzero : twoArmGeneratedAction sample t = 0
221       · simp [twoArmActionGap, hzero]
222       · have hone : twoArmGeneratedAction sample t = 1 :=
223           (finTwo_eq_zero_or_one
224             (twoArmGeneratedAction sample t)).resolve_left hzero
225         simp [twoArmActionGap, hone]
226         ring

237 /-- Exactly `n` optimal pulls force the exact source starvation charge. -/
238 theorem twoArmSampledPseudoRegret_eq_gap_mul_horizon_sub_of_optimalPullCount_eq
239     {Env : Type v} (Delta : Real) (n horizon : Nat)
240     (sample : Env × ((k : Nat) → Fin 2 × Real))
241     (hcount : twoArmOptimalPullCount horizon sample = n) :
242     twoArmSampledPseudoRegret Delta horizon sample =
243       Delta * ((horizon - n : Nat) : Real) := by

266 /--
267 The explicit chronological no-return premise constructs membership in the
268 measurable starvation event.  This is the pathwise half of Appendix-C Step 1;
269 it does not assign a probability to the event.
270 -/
271 theorem mem_twoArmStepOneStarvationEvent_of_lowProbability_noFurtherOptimalPull
272     {Env : Type v} [MeasurableSpace Env]
273     (eta : Real) (cutoff n horizon : Nat)
274     (sample : Env × ((k : Nat) → Fin 2 × Real))
275     (hcutoff : cutoff + 1 ≤ horizon)
276     (hlow : twoArmSuccessProbability eta cutoff sample ≤
277       twoArmStepOneThreshold horizon)
278     (hcount : twoArmOptimalPullCount (cutoff + 1) sample = n)
279     (hnoFurther : ∀ t, cutoff + 1 ≤ t → t < horizon →
280       twoArmGeneratedAction sample t ≠ 0) :
281     sample ∈
282       twoArmStepOneStarvationEvent (Env := Env) eta cutoff n horizon := by
283   have hdecompose :
284       cutoff + 1 + (horizon - (cutoff + 1)) = horizon :=
285     Nat.add_sub_of_le hcutoff
286   have hfinal : twoArmOptimalPullCount horizon sample =
287       twoArmOptimalPullCount (cutoff + 1) sample := by
288     rw [← hdecompose]
289     exact pullCount_add_eq_of_forall_ne_between
290       (twoArmGeneratedAction sample) 0 (cutoff + 1)
291       (horizon - (cutoff + 1))
292       (fun t ht hlt => hnoFurther t ht (by simpa [hdecompose] using hlt))
293   exact ⟨⟨hlow, hcount⟩, hfinal.trans hcount⟩

308 /-- Integrability of finite-horizon sampled regret under any finite trace law. -/
309 theorem integrable_twoArmSampledPseudoRegret_of_finiteMeasure
310     {Env : Type v} [MeasurableSpace Env]
311     (mu : Measure (Env × ((k : Nat) → Fin 2 × Real)))
312     [IsFiniteMeasure mu] (Delta : Real) (horizon : Nat) :
313     Integrable (twoArmSampledPseudoRegret (Env := Env) Delta horizon) mu := by
314   unfold twoArmSampledPseudoRegret
315   apply IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
316   intro t _ht
317   have hmeas : Measurable
318       (fun sample : Env × ((k : Nat) → Fin 2 × Real) =>
319         twoArmActionGap Delta (sample.2 t).1) :=
320     (measurable_twoArmActionGap Delta).comp
321       (measurable_fst.comp ((measurable_pi_apply t).comp measurable_snd))
322   apply Integrable.of_bound hmeas.aestronglyMeasurable |Delta|
323   filter_upwards [] with sample
324   by_cases hzero : (sample.2 t).1 = 0 <;>
325     simp [twoArmActionGap, hzero]

327 /-- Fewer than `m` optimal-arm pulls at `horizon` force at least the uniform
328 `Delta * (horizon - m)` sampled-pseudo-regret charge.  This is the
329 finite-horizon consumer for a below-count event; it does not provide any
330 positive-mass lower bound for that event.  The measure need only be finite;
331 probability and expectation terminology is reserved for probability-measure
332 wrappers such as the fixed-IID consumer below. -/
333 theorem twoArmOptimalPullCountBelowEvent_charge_mul_probability_le_integral
334     {Env : Type v} [MeasurableSpace Env]
335     (mu : Measure (Env × ((k : Nat) → Fin 2 × Real)))
336     [IsFiniteMeasure mu]
337     (Delta : Real) (hDelta : 0 ≤ Delta) (m horizon : Nat) :
338     Delta * ((horizon - m : Nat) : Real) *
339         mu.real
340           (twoArmOptimalPullCountBelowEvent
341             (Env := Env) m horizon) ≤
342       integral mu
343         (twoArmSampledPseudoRegret (Env := Env) Delta horizon) := by
344   let event := twoArmOptimalPullCountBelowEvent (Env := Env) m horizon
345   let charge : Real := Delta * ((horizon - m : Nat) : Real)
346   have hevent : MeasurableSet event :=
347     measurableSet_twoArmOptimalPullCountBelowEvent
348       (Env := Env) m horizon
349   have hregret : Integrable
350       (twoArmSampledPseudoRegret (Env := Env) Delta horizon) mu :=
351     integrable_twoArmSampledPseudoRegret_of_finiteMeasure mu Delta horizon
352   have hindicator : Integrable
353       (event.indicator (fun _sample => charge)) mu :=
354     (integrable_const charge).indicator hevent
355   have hpointwise : ∀ sample : Env × ((k : Nat) → Fin 2 × Real),
356       event.indicator (fun _sample => charge) sample ≤
357         twoArmSampledPseudoRegret Delta horizon sample := by
358     intro sample
359     by_cases hsample : sample ∈ event
360     · rw [Set.indicator_of_mem hsample]
361       have hcount_le : twoArmOptimalPullCount horizon sample ≤ m :=
362         Nat.le_of_lt hsample
363       have hsub : horizon - m ≤
364           horizon - twoArmOptimalPullCount horizon sample :=
365         Nat.sub_le_sub_left hcount_le horizon
366       calc
367         charge ≤ Delta *
368             ((horizon - twoArmOptimalPullCount horizon sample : Nat) : Real) := by
369           dsimp [charge]
370           gcongr
371         _ = twoArmSampledPseudoRegret Delta horizon sample :=
372           (twoArmSampledPseudoRegret_eq_gap_mul_horizon_sub_of_optimalPullCount_eq
373             Delta (twoArmOptimalPullCount horizon sample) horizon sample rfl).symm
374     · simp only [Set.indicator, if_neg hsample]
375       exact twoArmSampledPseudoRegret_nonneg Delta hDelta horizon sample
376   have hintegral :
377       integral mu (event.indicator (fun _sample => charge)) ≤
378         integral mu
379           (twoArmSampledPseudoRegret (Env := Env) Delta horizon) :=
380     integral_mono hindicator hregret hpointwise
381   have hindicatorIntegral :
382       integral mu (event.indicator (fun _sample => charge)) =
383         charge * mu.real event := by
384     rw [integral_indicator hevent]
385     rw [integral_const]
386     simp [Measure.real, mul_comm]
387   rw [hindicatorIntegral] at hintegral
388   simpa [event, charge, mul_assoc] using hintegral

391 Expectation-level deterministic Step-1 consumer.  It lower-bounds expected
392 sampled regret by the exact starvation charge times the probability of the
393 actual generated starvation event.  The missing source producer is precisely
394 the separate lower bound on this event probability from the trigger event.

396 theorem twoArmStepOneStarvationEvent_charge_mul_probability_le_integral
397     {Env : Type v} [MeasurableSpace Env]
398     (mu : Measure (Env × ((k : Nat) → Fin 2 × Real)))
399     [IsFiniteMeasure mu]
400     (eta Delta : Real) (hDelta : 0 ≤ Delta)
401     (cutoff n horizon : Nat) :
402     Delta * ((horizon - n : Nat) : Real) *
403         mu.real
404           (twoArmStepOneStarvationEvent
405             (Env := Env) eta cutoff n horizon) ≤
406       integral mu
407         (twoArmSampledPseudoRegret (Env := Env) Delta horizon) := by
408   let event :=
409     twoArmStepOneStarvationEvent (Env := Env) eta cutoff n horizon
410   let charge : Real := Delta * ((horizon - n : Nat) : Real)

413       (Env := Env) eta cutoff n horizon
414   have hregret : Integrable
415       (twoArmSampledPseudoRegret (Env := Env) Delta horizon) mu :=
416     integrable_twoArmSampledPseudoRegret_of_finiteMeasure mu Delta horizon
417   have hindicator : Integrable
418       (event.indicator (fun _sample => charge)) mu :=
419     (integrable_const charge).indicator hevent
420   have hpointwise : ∀ sample : Env × ((k : Nat) → Fin 2 × Real),
421       event.indicator (fun _sample => charge) sample ≤
422         twoArmSampledPseudoRegret Delta horizon sample := by
423     intro sample
424     by_cases hsample : sample ∈ event
425     · rw [Set.indicator_of_mem hsample]
426       exact le_of_eq
427         (twoArmStepOneStarvationEvent_sampledPseudoRegret_eq
428           eta Delta cutoff n horizon sample hsample).symm
429     · simp only [Set.indicator, if_neg hsample]
430       exact twoArmSampledPseudoRegret_nonneg Delta hDelta horizon sample
431   have hintegral :
432       integral mu (event.indicator (fun _sample => charge)) ≤
433         integral mu
434           (twoArmSampledPseudoRegret (Env := Env) Delta horizon) :=
435     integral_mono hindicator hregret hpointwise
436   have hindicatorIntegral :
437       integral mu (event.indicator (fun _sample => charge)) =
438         charge * mu.real event := by
439     rw [integral_indicator hevent]
440     rw [integral_const]
441     simp [Measure.real, mul_comm]
442   rw [hindicatorIntegral] at hintegral
443   simpa [event, charge, mul_assoc] using hintegral
