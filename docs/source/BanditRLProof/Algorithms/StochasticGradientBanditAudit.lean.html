7 This module formalizes the finite-action algebra in Algorithm 1 and Equations
8 (3)--(7) of Baudry--Johnson--Vary--Pike-Burke--Rebeschini (NeurIPS 2025).
9 
10 The finite weighted sums below are the exact conditional-mean calculations
11 obtained after fixing a pre-action history.  They are not a construction of
12 the paper's recursive stochastic history, reward kernel, or learning-rate
13 regret theorems.  Those trajectory and rate endpoints remain separate open
14 obligations.
15 -/

26 /-- The denominator in the source softmax rule, Equation (3). -/
27 def softmaxDenominator (theta : Action -> Real) : Real :=
28   ∑ a, Real.exp (theta a)
29 
30 /-- The source softmax sampling probability, Equation (3). -/
31 def softmaxProbability (theta : Action -> Real) (a : Action) : Real :=
32   Real.exp (theta a) / softmaxDenominator theta
33 
34 omit [DecidableEq Action] in
35 theorem softmaxDenominator_pos [Nonempty Action] (theta : Action -> Real) :
36     0 < softmaxDenominator theta := by
37   classical
38   let a : Action := Classical.choice inferInstance
39   exact Finset.sum_pos' (fun i _hi => (Real.exp_pos (theta i)).le)
40     ⟨a, Finset.mem_univ a, Real.exp_pos (theta a)⟩
41 
42 omit [DecidableEq Action] in
43 theorem softmaxProbability_pos [Nonempty Action] (theta : Action -> Real) (a : Action) :
44     0 < softmaxProbability theta a := by
45   exact div_pos (Real.exp_pos _) (softmaxDenominator_pos theta)
46 
47 omit [DecidableEq Action] in
48 theorem softmaxProbability_nonneg [Nonempty Action] (theta : Action -> Real) (a : Action) :
49     0 <= softmaxProbability theta a :=
50   (softmaxProbability_pos theta a).le
51 
52 omit [DecidableEq Action] in
53 /-- Equation (3) defines a normalized finite sampling law. -/
54 theorem softmaxProbability_sum [Nonempty Action] (theta : Action -> Real) :
55     ∑ a, softmaxProbability theta a = 1 := by
56   rw [show (∑ a, softmaxProbability theta a) =
57       (∑ a, Real.exp (theta a)) / softmaxDenominator theta by
58     simp only [softmaxProbability, Finset.sum_div]]
59   exact div_self (ne_of_gt (softmaxDenominator_pos theta))
60 
61 theorem softmaxProbability_le_one [Nonempty Action] (theta : Action -> Real) (a : Action) :
62     softmaxProbability theta a <= 1 := by
63   have hsum := softmaxProbability_sum theta
64   have hrest : 0 <= ∑ b ∈ (Finset.univ : Finset Action).erase a,
65       softmaxProbability theta b := by
66     exact Finset.sum_nonneg fun b _hb => softmaxProbability_nonneg theta b
67   have hsplit := Finset.sum_erase_add (Finset.univ : Finset Action)
68     (softmaxProbability theta) (Finset.mem_univ a)
69   linarith

71 /-- Algorithm 1 / Equation (4), before multiplication by the learning rate. -/
72 def sourceIncrement (p : Action -> Real) (reward : Real)
73     (selected k : Action) : Real :=
74   if selected = k then (1 - p k) * reward else -(p k * reward)
75 
76 omit [Fintype Action] in
77 theorem sourceIncrement_eq_indicator (p : Action -> Real) (reward : Real)
78     (selected k : Action) :
79     sourceIncrement p reward selected k =
80       reward * ((if selected = k then 1 else 0) - p k) := by
81   by_cases h : selected = k <;> simp [sourceIncrement, h] <;> ring
82 
83 /-- Algorithm 1 preserves the zero sum of its parameter vector. -/
84 theorem sum_sourceIncrement (p : Action -> Real) (reward : Real)
85     (selected : Action) (hp : ∑ k, p k = 1) :
86     ∑ k, sourceIncrement p reward selected k = 0 := by
87   classical
88   simp_rw [sourceIncrement_eq_indicator]
89   calc
90     (∑ k, reward * ((if selected = k then 1 else 0) - p k)) =
91         reward * ∑ k, ((if selected = k then 1 else 0) - p k) := by
92           rw [Finset.mul_sum]
93     _ = reward * ((∑ k, if selected = k then 1 else 0) - ∑ k, p k) := by
94           rw [Finset.sum_sub_distrib]
95     _ = 0 := by simp [hp]

97 /-- The policy value at a fixed pre-action history. -/
98 def policyValue (p mean : Action -> Real) : Real :=
99   ∑ a, p a * mean a
100 
101 /-- The finite conditional-mean version of the source expected update. -/
102 def expectedSourceIncrement (p mean : Action -> Real) (k : Action) : Real :=
103   ∑ selected, p selected * sourceIncrement p (mean selected) selected k
104 
105 /-- Equation (5), in policy-gradient-coordinate form. -/
106 theorem expectedSourceIncrement_eq_gradientCoordinate
107     (p mean : Action -> Real) (k : Action) :
108     expectedSourceIncrement p mean k =
109       p k * (mean k - policyValue p mean) := by
110   classical
111   simp_rw [expectedSourceIncrement, sourceIncrement_eq_indicator]
112   calc
113     (∑ selected,
114         p selected *
115           (mean selected * ((if selected = k then 1 else 0) - p k))) =
116         (∑ selected,
117           (p selected * mean selected) * (if selected = k then 1 else 0)) -
118         ∑ selected, (p selected * mean selected) * p k := by
119           rw [← Finset.sum_sub_distrib]
120           apply Finset.sum_congr rfl
121           intro selected _hselected
122           ring
123     _ = p k * mean k - policyValue p mean * p k := by
124           rw [← Finset.sum_mul]
125           simp [policyValue]
126     _ = p k * (mean k - policyValue p mean) := by ring

128 /-- The source instantaneous expected gap `E_t[Delta_{A_t}]`. -/
129 def instantaneousGap (p gap : Action -> Real) : Real :=
130   ∑ a, p a * gap a
131 
132 omit [DecidableEq Action] in
133 theorem instantaneousGap_eq_bestMean_sub_policyValue
134     (p mean gap : Action -> Real) (bestMean : Real)
135     (hp : ∑ a, p a = 1)
136     (hgap : ∀ a, gap a = bestMean - mean a) :
137     instantaneousGap p gap = bestMean - policyValue p mean := by
138   unfold instantaneousGap
139   calc
140     (∑ a, p a * gap a) = ∑ a, p a * (bestMean - mean a) := by
141       apply Finset.sum_congr rfl
142       intro a _ha
143       rw [hgap a]
144     _ = ∑ a, (bestMean * p a - p a * mean a) := by
145       apply Finset.sum_congr rfl
146       intro a _ha
147       ring
148     _ = (∑ a, bestMean * p a) - ∑ a, p a * mean a := by
149       rw [Finset.sum_sub_distrib]
150     _ = bestMean * (∑ a, p a) - ∑ a, p a * mean a := by
151       rw [Finset.mul_sum]
152     _ = bestMean - policyValue p mean := by rw [hp]; simp [policyValue]

154 /-- Equation (5), in instantaneous-gap-coordinate form. -/
155 theorem expectedSourceIncrement_eq_gapCoordinate
156     (p mean gap : Action -> Real) (bestMean : Real) (k : Action)
157     (hp : ∑ a, p a = 1)
158     (hgap : ∀ a, gap a = bestMean - mean a) :
159     expectedSourceIncrement p mean k =
160       p k * (instantaneousGap p gap - gap k) := by
161   rw [expectedSourceIncrement_eq_gradientCoordinate,
162     instantaneousGap_eq_bestMean_sub_policyValue p mean gap bestMean hp hgap,
163     hgap k]
164   ring

166 /-- The gap-coordinate update isolated from Equation (5). -/
167 def gapExpectedIncrement (p gap : Action -> Real) (k : Action) : Real :=
168   p k * (instantaneousGap p gap - gap k)
169 
170 theorem expectedSourceIncrement_eq_gapExpectedIncrement
171     (p mean gap : Action -> Real) (bestMean : Real) (k : Action)
172     (hp : ∑ a, p a = 1)
173     (hgap : ∀ a, gap a = bestMean - mean a) :
174     expectedSourceIncrement p mean k = gapExpectedIncrement p gap k := by
175   exact expectedSourceIncrement_eq_gapCoordinate p mean gap bestMean k hp hgap

177 theorem instantaneousGap_ge_minGap_mul_failureMass
178     (p gap : Action -> Real) (best : Action) (Delta : Real)
179     (hp : ∑ a, p a = 1)
180     (hp_nonneg : ∀ a, 0 <= p a)
181     (hgap_best : gap best = 0)
182     (hgap_min : ∀ a, a ≠ best -> Delta <= gap a) :
183     Delta * (1 - p best) <= instantaneousGap p gap := by
184   classical
185   have hprob_erase :
186       ∑ a ∈ (Finset.univ : Finset Action).erase best, p a = 1 - p best := by
187     have hsplit := Finset.sum_erase_add (Finset.univ : Finset Action) p
188       (Finset.mem_univ best)
189     rw [hp] at hsplit
190     linarith
191   calc
192     Delta * (1 - p best) =
193         ∑ a ∈ (Finset.univ : Finset Action).erase best, p a * Delta := by
194       rw [← hprob_erase]
195       rw [← Finset.sum_mul]
196       ring
197     _ <= ∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a := by
198       apply Finset.sum_le_sum
199       intro a ha
200       exact mul_le_mul_of_nonneg_left
201         (hgap_min a (by simpa using ha)) (hp_nonneg a)
202     _ = instantaneousGap p gap := by
203       rw [instantaneousGap]
204       have hsplit := Finset.sum_erase_add (Finset.univ : Finset Action)
205         (fun a => p a * gap a) (Finset.mem_univ best)
206       calc
207         (∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a) =
208             (∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a) +
209               p best * gap best := by rw [hgap_best]; ring
210         _ = ∑ a, p a * gap a := hsplit

212 /-- Pointwise Equation (6): the best coordinate gains at least the positive
213 gap times its success/failure probability product. -/
214 theorem gapExpectedIncrement_best_ge
215     (p gap : Action -> Real) (best : Action) (Delta : Real)
216     (hp : ∑ a, p a = 1)
217     (hp_nonneg : ∀ a, 0 <= p a)
218     (hgap_best : gap best = 0)
219     (hgap_min : ∀ a, a ≠ best -> Delta <= gap a) :
220     Delta * (p best * (1 - p best)) <=
221       gapExpectedIncrement p gap best := by
222   rw [gapExpectedIncrement, hgap_best, sub_zero]
223   have hinst := instantaneousGap_ge_minGap_mul_failureMass
224     p gap best Delta hp hp_nonneg hgap_best hgap_min
225   nlinarith [hp_nonneg best]

227 /-- The finite-horizon best-parameter expectation represented by Equation
228 (6), after conditioning at each round. -/
229 def bestParameterIncrementSum (eta : Real) (p : Nat -> Action -> Real)
230     (gap : Action -> Real) (best : Action) (horizon : Nat) : Real :=
231   eta * ∑ t ∈ Finset.range horizon, gapExpectedIncrement (p t) gap best
232 
233 /-- Finite-horizon Equation (6). -/
234 theorem bestParameterIncrementSum_ge
235     (eta Delta : Real) (p : Nat -> Action -> Real) (gap : Action -> Real)
236     (best : Action) (horizon : Nat)
237     (heta : 0 <= eta)
238     (hp : ∀ t, ∑ a, p t a = 1)
239     (hp_nonneg : ∀ t a, 0 <= p t a)
240     (hgap_best : gap best = 0)
241     (hgap_min : ∀ a, a ≠ best -> Delta <= gap a) :
242     eta * Delta *
243         (∑ t ∈ Finset.range horizon, p t best * (1 - p t best)) <=
244       bestParameterIncrementSum eta p gap best horizon := by
245   unfold bestParameterIncrementSum
246   have hsum :
247       Delta * (∑ t ∈ Finset.range horizon, p t best * (1 - p t best)) <=
248         ∑ t ∈ Finset.range horizon, gapExpectedIncrement (p t) gap best := by
249     rw [Finset.mul_sum]
250     apply Finset.sum_le_sum
251     intro t _ht
252     exact gapExpectedIncrement_best_ge (p t) gap best Delta
253       (hp t) (hp_nonneg t) hgap_best hgap_min
254   nlinarith

261 theorem instantaneousGap_le_maxGap_mul_failureMass
262     (p gap : Action -> Real) (best : Action) (DeltaMax : Real)
263     (hp : ∑ a, p a = 1)
264     (hp_nonneg : ∀ a, 0 <= p a)
265     (hgap_best : gap best = 0)
266     (hgap_max : ∀ a, a ≠ best -> gap a <= DeltaMax) :
267     instantaneousGap p gap <= DeltaMax * (1 - p best) := by
268   classical
269   have hprob_erase :
270       ∑ a ∈ (Finset.univ : Finset Action).erase best, p a = 1 - p best := by
271     have hsplit := Finset.sum_erase_add (Finset.univ : Finset Action) p
272       (Finset.mem_univ best)
273     rw [hp] at hsplit
274     linarith
275   calc
276     instantaneousGap p gap =
277         ∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a := by
278       rw [instantaneousGap]
279       have hsplit := Finset.sum_erase_add (Finset.univ : Finset Action)
280         (fun a => p a * gap a) (Finset.mem_univ best)
281       calc
282         (∑ a, p a * gap a) =
283             (∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a) +
284               p best * gap best := hsplit.symm
285         _ = ∑ a ∈ (Finset.univ : Finset Action).erase best, p a * gap a := by
286           rw [hgap_best]
287           ring
288     _ <= ∑ a ∈ (Finset.univ : Finset Action).erase best, p a * DeltaMax := by
289       apply Finset.sum_le_sum
290       intro a ha
291       exact mul_le_mul_of_nonneg_left
292         (hgap_max a (by simpa using ha)) (hp_nonneg a)
293     _ = DeltaMax * (1 - p best) := by
294       rw [← Finset.sum_mul, hprob_erase]
295       ring

300 /-- Equation (7), with its positive learning-rate/minimum-gap denominator and
301 maximum-gap envelope exposed explicitly. -/
302 theorem sourceRegretDecomposition_le
303     (eta Delta DeltaMax : Real) (p : Nat -> Action -> Real)
304     (gap : Action -> Real) (best : Action) (horizon : Nat)
305     (heta : 0 < eta) (hDelta : 0 < Delta) (hDeltaMax : 0 <= DeltaMax)
306     (hp : ∀ t, ∑ a, p t a = 1)
307     (hp_nonneg : ∀ t a, 0 <= p t a)
308     (hgap_best : gap best = 0)
309     (hgap_min : ∀ a, a ≠ best -> Delta <= gap a)
310     (hgap_max : ∀ a, a ≠ best -> gap a <= DeltaMax) :
311     sourceExpectedPseudoRegret p gap horizon <=
312       (DeltaMax / (eta * Delta)) *
313           bestParameterIncrementSum eta p gap best horizon +
314         DeltaMax *
315           (∑ t ∈ Finset.range horizon, (1 - p t best) ^ 2) := by
316   have hregret :
317       sourceExpectedPseudoRegret p gap horizon <=
318         DeltaMax * (∑ t ∈ Finset.range horizon, (1 - p t best)) := by
319     unfold sourceExpectedPseudoRegret
320     calc
321       (∑ t ∈ Finset.range horizon, instantaneousGap (p t) gap) <=
322           ∑ t ∈ Finset.range horizon, DeltaMax * (1 - p t best) := by
323         apply Finset.sum_le_sum
324         intro t _ht
325         exact instantaneousGap_le_maxGap_mul_failureMass
326           (p t) gap best DeltaMax (hp t) (hp_nonneg t) hgap_best hgap_max
327       _ = DeltaMax * (∑ t ∈ Finset.range horizon, (1 - p t best)) := by
328         rw [Finset.mul_sum]
329   have hparameter := bestParameterIncrementSum_ge
330     eta Delta p gap best horizon heta.le hp hp_nonneg hgap_best hgap_min
331   have hproduct : 0 < eta * Delta := mul_pos heta hDelta
332   have hsuccessFailure :
333       (∑ t ∈ Finset.range horizon, p t best * (1 - p t best)) <=
334         bestParameterIncrementSum eta p gap best horizon / (eta * Delta) := by
335     rw [le_div_iff₀ hproduct]
336     nlinarith
337   have hsplit :
338       (∑ t ∈ Finset.range horizon, (1 - p t best)) =
339         (∑ t ∈ Finset.range horizon, p t best * (1 - p t best)) +
340           ∑ t ∈ Finset.range horizon, (1 - p t best) ^ 2 := by
341     rw [← Finset.sum_add_distrib]
342     apply Finset.sum_congr rfl
343     intro t _ht
344     exact failureMass_eq_successFailure_add_sq (p t best)
345   rw [hsplit] at hregret
346   calc
347     sourceExpectedPseudoRegret p gap horizon <=
348         DeltaMax *
349           ((∑ t ∈ Finset.range horizon, p t best * (1 - p t best)) +
350             ∑ t ∈ Finset.range horizon, (1 - p t best) ^ 2) := hregret
351     _ <= DeltaMax *
352           (bestParameterIncrementSum eta p gap best horizon / (eta * Delta) +
353             ∑ t ∈ Finset.range horizon, (1 - p t best) ^ 2) := by
354       gcongr
355     _ = (DeltaMax / (eta * Delta)) *
356           bestParameterIncrementSum eta p gap best horizon +
357         DeltaMax *
358           (∑ t ∈ Finset.range horizon, (1 - p t best) ^ 2) := by
359       field_simp
