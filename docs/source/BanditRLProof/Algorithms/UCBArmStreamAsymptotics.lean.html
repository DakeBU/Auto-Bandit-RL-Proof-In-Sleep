6 # One-policy asymptotics for the canonical arm-stream UCB process
7 
8 This module keeps one recursive `armStreamAction` and one `armStreamMeasure`
9 fixed across all horizons. At exploration scale `c = 4`, the finite tail term
10 in the exact LML-shaped regret bound is uniformly bounded by a convergent
11 NNReal p-series, yielding logarithmic expected regret and vanishing expected
12 average regret for the same policy and measure.
13 -/

22 /-- The summable cubic tail that controls `indexTail 4`. -/
23 noncomputable def armStreamPSeriesTerm (t : Nat) : NNReal :=
24   1 / (((t + 1 : Nat) : NNReal) ^ (3 : Real))
25 
26 /-- A fixed finite upper bound for every `constSum 4 n`. -/
27 noncomputable def armStreamPSeriesTailBound : NNReal :=
28   ∑' t : Nat, armStreamPSeriesTerm t
29 
30 theorem armStreamPSeriesTerm_summable : Summable armStreamPSeriesTerm := by
31   unfold armStreamPSeriesTerm
32   exact (NNReal.summable_nat_add_iff 1).2
33     (NNReal.summable_one_div_rpow.mpr (by norm_num : (1 : Real) < 3))

35 theorem indexTail_four_eq_coe_armStreamPSeriesTerm (t : Nat) :
36     indexTail 4 t = (armStreamPSeriesTerm t : ENNReal) := by
37   unfold indexTail armStreamPSeriesTerm
38   norm_num

40 theorem constSum_four_le_armStreamPSeriesTailBound (n : Nat) :
41     constSum 4 n <= (armStreamPSeriesTailBound : ENNReal) := by
42   have hnn :
43       (Finset.range n).sum armStreamPSeriesTerm <=
44         armStreamPSeriesTailBound := by
45     exact Summable.sum_le_tsum
46       (s := Finset.range n) (fun _t _ht => zero_le _)
47       armStreamPSeriesTerm_summable
48   rw [constSum]
49   simp_rw [indexTail_four_eq_coe_armStreamPSeriesTerm]
50   exact_mod_cast hnn

52 theorem constSum_four_toReal_le_armStreamPSeriesTailBound (n : Nat) :
53     (constSum 4 n).toReal <= (armStreamPSeriesTailBound : Real) := by
54   rw [← ENNReal.coe_toReal armStreamPSeriesTailBound]
55   exact

59 /--
60 Fixed kernel-dependent coefficient for the one-policy logarithmic envelope.
61 -/
62 noncomputable def armStreamAsymptoticModelCoefficient {K : Nat}
63     (nu : Kernel (Fin K) Real) (sigma2 : NNReal) : Real :=
64   (Finset.univ : Finset (Fin K)).sum (fun arm =>
65     32 * (sigma2 : Real) / realKernelGap nu arm +
66       realKernelGap nu arm *
67         (2 + 2 * (armStreamPSeriesTailBound : Real)))
68 
69 theorem armStreamAsymptoticModelCoefficient_nonneg
70     {K : Nat} (hK : 0 < K)
71     (nu : Kernel (Fin K) Real) (sigma2 : NNReal) :
72     0 <= armStreamAsymptoticModelCoefficient nu sigma2 := by
73   letI : Nonempty (Fin K) := ⟨Fin.mk 0 hK⟩
74   unfold armStreamAsymptoticModelCoefficient
75   apply Finset.sum_nonneg
76   intro arm _harm
77   have hgap : 0 <= realKernelGap nu arm := realKernelGap_nonneg nu arm
78   have hfirst : 0 <= 32 * (sigma2 : Real) / realKernelGap nu arm :=
79     div_nonneg (by positivity) hgap
80   have hsecond :
81       0 <= realKernelGap nu arm *
82         (2 + 2 * (armStreamPSeriesTailBound : Real)) := by
83     positivity
84   linarith

86 theorem lml_sum_four_le_armStreamAsymptoticModelCoefficient
87     {K : Nat} (hK : 0 < K)
88     (nu : Kernel (Fin K) Real) (sigma2 : NNReal) (n : Nat) :
89     (Finset.univ : Finset (Fin K)).sum (fun arm =>
90         8 * 4 * (sigma2 : Real) * Real.log ((n + 1 : Nat) : Real) /
91             realKernelGap nu arm +
92           realKernelGap nu arm * (2 + 2 * (constSum 4 n).toReal)) <=
93       armStreamAsymptoticModelCoefficient nu sigma2 *
94         (1 + Real.log ((n + 1 : Nat) : Real)) := by
95   letI : Nonempty (Fin K) := ⟨Fin.mk 0 hK⟩
96   have hlog : 0 <= Real.log ((n + 1 : Nat) : Real) := by
97     apply Real.log_nonneg
98     exact_mod_cast Nat.succ_le_succ (Nat.zero_le n)
99   have htail := constSum_four_toReal_le_armStreamPSeriesTailBound n
100   calc
101     _ <= (Finset.univ : Finset (Fin K)).sum (fun arm =>
102         (32 * (sigma2 : Real) / realKernelGap nu arm +
103           realKernelGap nu arm *
104             (2 + 2 * (armStreamPSeriesTailBound : Real))) *
105           (1 + Real.log ((n + 1 : Nat) : Real))) := by
106       apply Finset.sum_le_sum
107       intro arm _harm
108       have hgap : 0 <= realKernelGap nu arm := realKernelGap_nonneg nu arm
109       by_cases hgapZero : realKernelGap nu arm = 0
110       · simp [hgapZero]
111       · have hA :
112             0 <= 32 * (sigma2 : Real) / realKernelGap nu arm :=
113           div_nonneg (by positivity) hgap
114         have hB :
115             0 <= realKernelGap nu arm *
116               (2 + 2 * (armStreamPSeriesTailBound : Real)) := by
117           positivity
118         have htailTerm :
119             realKernelGap nu arm * (2 + 2 * (constSum 4 n).toReal) <=
120               realKernelGap nu arm *
121                 (2 + 2 * (armStreamPSeriesTailBound : Real)) := by
122           gcongr
123         calc
124           8 * 4 * (sigma2 : Real) * Real.log ((n + 1 : Nat) : Real) /
125                 realKernelGap nu arm +
126               realKernelGap nu arm * (2 + 2 * (constSum 4 n).toReal) =
127               (32 * (sigma2 : Real) / realKernelGap nu arm) *
128                   Real.log ((n + 1 : Nat) : Real) +
129                 realKernelGap nu arm *
130                   (2 + 2 * (constSum 4 n).toReal) := by
131             ring
132           _ <= (32 * (sigma2 : Real) / realKernelGap nu arm) *
133                   Real.log ((n + 1 : Nat) : Real) +
134                 realKernelGap nu arm *
135                   (2 + 2 * (armStreamPSeriesTailBound : Real)) :=
136             add_le_add_right htailTerm _
137           _ <= (32 * (sigma2 : Real) / realKernelGap nu arm +
138                   realKernelGap nu arm *
139                     (2 + 2 * (armStreamPSeriesTailBound : Real))) *
140                 (1 + Real.log ((n + 1 : Nat) : Real)) := by
141             nlinarith [
142               mul_nonneg hA (show 0 <= (1 : Real) by norm_num),
143               mul_nonneg hB hlog]
144     _ = armStreamAsymptoticModelCoefficient nu sigma2 *
145         (1 + Real.log ((n + 1 : Nat) : Real)) := by
146       rw [armStreamAsymptoticModelCoefficient, Finset.sum_mul]

148 /-- Expected regret of one fixed recursive arm-stream UCB process. -/
149 noncomputable def armStreamExpectedRegret {K : Nat}
150     (hK : 0 < K) (sigma2 : NNReal) (nu : Kernel (Fin K) Real)
151     [IsMarkovKernel nu] (n : Nat) : Real :=
152   ∫ stream : ArmRewardStream K,
153     realKernelRegret nu
154       (armStreamAction hK (4 * (sigma2 : Real)) stream) n
155     ∂armStreamMeasure nu

157 theorem armStreamExpectedRegret_nonneg_and_le
158     {K : Nat} (hK : 0 < K) (sigma2 : NNReal)
159     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
160     (hsigma2 : sigma2 ≠ 0)
161     (hsubG : ∀ arm : Fin K, HasSubgaussianMGF
162       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm))
163     (n : Nat) :
164     0 <= armStreamExpectedRegret hK sigma2 nu n ∧
165       armStreamExpectedRegret hK sigma2 nu n <=
166         armStreamAsymptoticModelCoefficient nu sigma2 *
167           (1 + Real.log ((n + 1 : Nat) : Real)) := by
168   letI : Nonempty (Fin K) := ⟨Fin.mk 0 hK⟩
169   constructor
170   · unfold armStreamExpectedRegret
171     refine integral_nonneg (fun stream => ?_)
172     change 0 <= realKernelRegret nu
173       (armStreamAction hK (4 * (sigma2 : Real)) stream) n
174     rw [realKernelRegret_eq_finset_sum_gap]
175     exact Finset.sum_nonneg fun _t _ht => realKernelGap_nonneg nu _
176   · unfold armStreamExpectedRegret
177     refine (integral_realKernelRegret_armStreamAction_le_lml_sum
178       hK 4 sigma2 nu n (by norm_num) hsigma2 hsubG).trans ?_
179     exact lml_sum_four_le_armStreamAsymptoticModelCoefficient
180       hK nu sigma2 n

182 theorem armStreamExpectedRegret_isBigO_log
183     {K : Nat} (hK : 0 < K) (sigma2 : NNReal)
184     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
185     (hsigma2 : sigma2 ≠ 0)
186     (hsubG : ∀ arm : Fin K, HasSubgaussianMGF
187       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm)) :
188     (armStreamExpectedRegret hK sigma2 nu) =O[atTop]
189       (fun n : Nat => Real.log ((n + 1 : Nat) : Real)) := by
190   have hcoefficient :=
191     armStreamAsymptoticModelCoefficient_nonneg hK nu sigma2
192   have henvelope :
193       (fun n : Nat => armStreamAsymptoticModelCoefficient nu sigma2 *
194         (1 + Real.log ((n + 1 : Nat) : Real))) =O[atTop]
195       (fun n : Nat => Real.log ((n + 1 : Nat) : Real)) := by
196     simpa using
197       one_add_log_natCast_succ_isBigO_log_natCast_succ.const_mul_left
198         (armStreamAsymptoticModelCoefficient nu sigma2)
199   rw [Asymptotics.isBigO_iff] at henvelope ⊢
200   obtain ⟨c, hc⟩ := henvelope
201   refine ⟨c, ?_⟩
202   filter_upwards [hc] with n henvelopeN
203   have hpoint :=
204     armStreamExpectedRegret_nonneg_and_le
205       hK sigma2 nu hsigma2 hsubG n
206   have henvelopeNonneg :
207       0 <= armStreamAsymptoticModelCoefficient nu sigma2 *
208         (1 + Real.log ((n + 1 : Nat) : Real)) := by
209     exact mul_nonneg hcoefficient
210       (add_nonneg zero_le_one (Real.log_nonneg (by
211         exact_mod_cast Nat.succ_le_succ (Nat.zero_le n))))
212   calc
213     ‖armStreamExpectedRegret hK sigma2 nu n‖ =
214         armStreamExpectedRegret hK sigma2 nu n :=
215       Real.norm_of_nonneg hpoint.1
216     _ <= armStreamAsymptoticModelCoefficient nu sigma2 *
217         (1 + Real.log ((n + 1 : Nat) : Real)) := hpoint.2
218     _ = ‖armStreamAsymptoticModelCoefficient nu sigma2 *
219         (1 + Real.log ((n + 1 : Nat) : Real))‖ :=
220       (Real.norm_of_nonneg henvelopeNonneg).symm
221     _ <= c * ‖Real.log ((n + 1 : Nat) : Real)‖ := henvelopeN

223 theorem armStreamExpectedRegret_isLittleO_natCast_succ
224     {K : Nat} (hK : 0 < K) (sigma2 : NNReal)
225     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
226     (hsigma2 : sigma2 ≠ 0)
227     (hsubG : ∀ arm : Fin K, HasSubgaussianMGF
228       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm)) :
229     (armStreamExpectedRegret hK sigma2 nu) =o[atTop]

235 /-- Expected regret of the fixed arm-stream policy normalized by `n + 1`. -/
236 noncomputable def armStreamExpectedAverageRegret {K : Nat}
237     (hK : 0 < K) (sigma2 : NNReal) (nu : Kernel (Fin K) Real)
238     [IsMarkovKernel nu] (n : Nat) : Real :=
239   armStreamExpectedRegret hK sigma2 nu n / ((n + 1 : Nat) : Real)

241 /--
242 One fixed canonical arm-stream UCB policy has vanishing expected average
243 regret under its one fixed product measure.
244 -/
245 theorem armStreamExpectedAverageRegret_tendsto_zero
246     {K : Nat} (hK : 0 < K) (sigma2 : NNReal)
247     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
248     (hsigma2 : sigma2 ≠ 0)
249     (hsubG : ∀ arm : Fin K, HasSubgaussianMGF
250       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm)) :
251     Tendsto (armStreamExpectedAverageRegret hK sigma2 nu)
252       atTop (nhds 0) := by
253   have hlimit :=
254     (armStreamExpectedRegret_isLittleO_natCast_succ
255       hK sigma2 nu hsigma2 hsubG).tendsto_div_nhds_zero
