7 # Exact common-sub-Gaussian ETC tail constant
8 
9 This module normalizes the canonical common-proxy ETC pairwise tail to the
10 closed exponential used by the exact LML theorem route.  It remains over the
11 project's existing `Rat` reward-law model, with all random variables embedded
12 in `Real`; transport to native `Real` reward kernels is downstream work.
13 -/

18 /--
19 Over the round-robin exploration horizon, the constant common-proxy pairwise
20 process charges exactly `m` pulls of the candidate arm and `m` pulls of the
21 selected best arm.
22 -/
23 theorem sum_centeredPairwiseRewardDiffVarianceProxy_const_eq_two_mul
24     {K : Nat}
25     (spec : ETC.Spec K) (model : FiniteBanditModel K)
26     (sigma2 : NNReal) (a : Fin K)
27     (hne : a = model.bestArm -> False) :
28     (Finset.range (spec.explorationPulls * K)).sum
29         (fun t =>
30           ETC.centeredPairwiseRewardDiffVarianceProxy
31             spec model model.bestArm (fun _ _ => sigma2) a t) =
32       (2 : NNReal) * (spec.explorationPulls : NNReal) * sigma2 := by
33   classical
34   let action : ActionTrace (Fin K) :=
35     ETC.actionWithCommit spec model.bestArm
36   have hane : a ≠ model.bestArm := by
37     intro h
38     exact hne h
39   have hpoint : forall t : Nat,
40       ETC.centeredPairwiseRewardDiffVarianceProxy
41           spec model model.bestArm (fun _ _ => sigma2) a t =
42         (if action t = a then sigma2 else 0) +
43           (if action t = model.bestArm then sigma2 else 0) := by
44     intro t
45     by_cases ha : action t = a
46     · simp [ETC.centeredPairwiseRewardDiffVarianceProxy, action, ha, hane]
47     · by_cases hbest : action t = model.bestArm
48       · have hbest_ne_a : model.bestArm ≠ a := Ne.symm hane
49         simp [ETC.centeredPairwiseRewardDiffVarianceProxy, action, hbest,
50           hbest_ne_a]
51       · simp [ETC.centeredPairwiseRewardDiffVarianceProxy, action, ha, hbest]
52   have hselected : forall b : Fin K,
53       (Finset.range (spec.explorationPulls * K)).sum
54           (fun t => if action t = b then sigma2 else 0) =
55         (spec.explorationPulls : NNReal) * sigma2 := by
56     intro b
57     calc
58       (Finset.range (spec.explorationPulls * K)).sum
59           (fun t => if action t = b then sigma2 else 0) =
60         ((Finset.range (spec.explorationPulls * K)).filter
61           (fun t => action t = b)).sum (fun _t => sigma2) := by
62             rw [Finset.sum_filter]
63       _ = (((Finset.range (spec.explorationPulls * K)).filter
64           (fun t => action t = b)).card : NNReal) * sigma2 := by
65             simp [nsmul_eq_mul]
66       _ = (spec.explorationPulls : NNReal) * sigma2 := by
67             have hcount :=
68               pullCount_eq_finset_filter_card
69                 (action := action) (a := b)
70                 (t := spec.explorationPulls * K)
71             rw [← hcount]
72             rw [show action = ETC.actionWithCommit spec model.bestArm by rfl]
73             rw [ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq]
74   rw [Finset.sum_congr rfl (fun t _ht => hpoint t)]
75   rw [Finset.sum_add_distrib, hselected a, hselected model.bestArm]
76   ring

79 theorem centeredPairwiseGapThreshold_eq_explorationPulls_mul_gap
80     {K : Nat}
81     (spec : ETC.Spec K) (model : FiniteBanditModel K) (a : Fin K)
82     (hne : a = model.bestArm -> False) :
83     ETC.centeredPairwiseGapThreshold spec model a =
84       (spec.explorationPulls : Real) * ((model.gap a : Rat) : Real) := by

91 /--
92 The canonical direct-MGF pairwise tail equals the exact LML exponential
93 `exp (-m * gap^2 / (4 * sigma2))` for every non-best arm.
94 
95 Only `m > 0` is needed to cancel the exploration multiplicity.  The
96 `sigma2 = 0` boundary is handled separately and remains a valid equality in
97 Lean's total division.
98 -/
99 theorem canonicalSubGaussianArmPairwiseTailReal_eq_exp_neg_explorationPulls_mul_gap_sq_div_four_mul
100     {K : Nat}
101     (spec : ETC.Spec K) (model : FiniteBanditModel K)
102     (sigma2 : NNReal) (a : Fin K)
103     (hexplorationPulls_pos : 0 < spec.explorationPulls)
104     (hne : a = model.bestArm -> False) :
105     ETC.canonicalSubGaussianArmPairwiseTailReal spec model sigma2 a =
106       Real.exp
107         (-(spec.explorationPulls : Real) *
108           ((model.gap a : Rat) : Real) ^ 2 /
109             (4 * (sigma2 : Real))) := by
110   let pairwiseProxy :=
111     ETC.centeredPairwiseRewardDiffVarianceProxy
112       spec model model.bestArm (fun _ _ => sigma2)
113   have hproxyNNReal :
114       (Finset.range (spec.explorationPulls * K)).sum (pairwiseProxy a) =
115         (2 : NNReal) * (spec.explorationPulls : NNReal) * sigma2 := by
116     exact
117       ETC.sum_centeredPairwiseRewardDiffVarianceProxy_const_eq_two_mul
118         spec model sigma2 a hne
119   have hproxyReal :
120       (((Finset.range (spec.explorationPulls * K)).sum (pairwiseProxy a) : NNReal) :
121           Real) =
122         2 * (spec.explorationPulls : Real) * (sigma2 : Real) := by
123     rw [hproxyNNReal]
124     norm_num
125   have hthreshold :=
126     ETC.centeredPairwiseGapThreshold_eq_explorationPulls_mul_gap
127       spec model a hne
128   rw [ETC.canonicalSubGaussianArmPairwiseTailReal]
129   rw [ETC.centeredDiffSubGaussianTail]
130   rw [ENNReal.toReal_ofReal (le_of_lt (Real.exp_pos _))]
131   rw [show
132     ETC.centeredPairwiseRewardDiffVarianceProxy
133         spec model model.bestArm (fun _ _ => sigma2) = pairwiseProxy by rfl]
134   rw [hthreshold, hproxyReal]
135   congr 1
136   have hm_ne : (spec.explorationPulls : Real) ≠ 0 := by
137     exact_mod_cast (Nat.ne_of_gt hexplorationPulls_pos)
138   by_cases hsigma : (sigma2 : Real) = 0
139   · simp [hsigma]
140   · field_simp [hm_ne, hsigma]
141     ring

143 /--
144 Exact-LML-constant form of the canonical single-arm commit-fiber probability
145 bound under the existing common-sub-Gaussian `Rat` arm laws.
146 
147 This is a genuine concentration producer, but not yet the native `Real`
148 reward-kernel theorem: the arm laws are measures on `Rat`, their centered casts
149 have a common `NNReal` proxy, and the generated trajectory is the local
150 canonical history process.

152 theorem real_measure_explorationArgmaxCommit_eq_arm_le_exp_neg_explorationPulls_mul_gap_sq_div_four_mul_of_armLaws

161         (((model.mean arm : Rat) : Real)))

196 /--
197 Canonical generated-history per-arm expected pull-count bound with the exact
198 LML exponential constant, under the existing common-sub-Gaussian `Rat` arm
199 laws.
200 
201 This theorem composes the concrete commit-fiber concentration producer with
202 the generic Real/Bochner `actionWithCommit` expected-count consumer.  It is the
203 first local theorem with the full `m + (n - K*m) * exp (...)` per-arm shape.
204 Native `Real` reward-kernel transport and external algorithm/environment-law
205 alignment remain downstream.
206 -/
