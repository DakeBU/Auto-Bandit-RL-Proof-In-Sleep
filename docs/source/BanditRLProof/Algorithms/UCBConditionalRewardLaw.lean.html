4 /-!
5 # UCB score consumers for the practical conditional reward law
6 
7 This module connects the selected-policy simultaneous empirical-mean event to
8 the deterministic UCB score algebra.  Its confidence width depends on the
9 realized pull count, so it deliberately does not use the older
10 `UCB.finiteHorizonConfidenceBadEvent`, whose radius is deterministic in the
11 sample point.
12 -/

21 /-- Successor empirical mean at the positive horizon `t + 1`. -/
22 noncomputable def selectedPolicySuccessorEmpiricalMeanAt
23     {Omega : Type u} {Action : Type} [DecidableEq Action]
24     (action : Omega -> ActionTrace Action)
25     (reward : Omega -> RewardTrace Rat)

42 /-- Practical UCB index with a sample-dependent realized-count width. -/
43 noncomputable def selectedPolicySuccessorIndexAt
44     {Omega : Type u} {Action : Type} [DecidableEq Action]
45     (action : Omega -> ActionTrace Action)
46     (reward : Omega -> RewardTrace Rat)
47     (sigma2 : NNReal) (arms : Finset Action) (T : Nat) (delta : Real)
48     (omega : Omega) (t : Nat) (arm : Action) : Real :=

57 /--
58 Initialization and score-maximality contract for the finite set of charged UCB
59 times.  The time set may omit initialization rounds; every retained time must
60 lie below `T`, and both the designated best arm and the selected arm must have
61 positive realized successor pull counts.
62 -/
63 structure SelectedPolicySuccessorInitializedScoreMaxSource
64     {Omega : Type u} {Action : Type} [DecidableEq Action]
65     (action : Omega -> ActionTrace Action)
66     (reward : Omega -> RewardTrace Rat)
67     (arms : Finset Action) (armMean : Action -> Rat)
68     (sigma2 : NNReal) (T : Nat) (delta : Real) where
69   times : Finset Nat
70   best : Action
71   chosen : Omega -> Nat -> Action
72   times_lt : forall t, t ∈ times -> t < T
73   best_mem : best ∈ arms
74   chosen_mem : forall omega t, t ∈ times -> chosen omega t ∈ arms
75   best_count_pos : forall omega t, t ∈ times ->
76     0 < ConditionalExpectationReward.successorArmPullCount
77       (action omega) best (t + 1)
78   chosen_count_pos : forall omega t, t ∈ times ->
79     0 < ConditionalExpectationReward.successorArmPullCount
80       (action omega) (chosen omega t) (t + 1)
81   score_max : forall omega t, t ∈ times ->
82     selectedPolicySuccessorIndexAt action reward sigma2 arms T delta
83         omega t best <=
84       selectedPolicySuccessorIndexAt action reward sigma2 arms T delta
85         omega t (chosen omega t)

87 /-- Large-gap selected-time event charged by the practical confidence event. -/
88 noncomputable def selectedPolicySuccessorLargeGapEvent
89     {Omega : Type u} {Action : Type} [DecidableEq Action]
90     {action : Omega -> ActionTrace Action}
91     {reward : Omega -> RewardTrace Rat}
92     {arms : Finset Action} {armMean : Action -> Rat}
93     {sigma2 : NNReal} {T : Nat} {delta : Real}
94     (source : SelectedPolicySuccessorInitializedScoreMaxSource
95       action reward arms armMean sigma2 T delta) : Set Omega :=
96   {omega | exists t, t ∈ source.times ∧
97     2 * selectedPolicySuccessorRadiusAt action sigma2 arms T delta
98           omega t (source.chosen omega t) <
99       meanGap (fun arm => (armMean arm : Real))
100         source.best (source.chosen omega t)}

102 /--
103 Outside the practical simultaneous confidence event, score maximality implies
104 the standard UCB gap bound at every initialized charged time.
105 -/
106 theorem SelectedPolicySuccessorInitializedScoreMaxSource.meanGap_le_two_radius_of_not_badEvent
107     {Omega : Type u} {Action : Type} [DecidableEq Action]
108     {action : Omega -> ActionTrace Action}
109     {reward : Omega -> RewardTrace Rat}
110     {arms : Finset Action} {armMean : Action -> Rat}
111     {sigma2 : NNReal} {T : Nat} {delta : Real}
112     (source : SelectedPolicySuccessorInitializedScoreMaxSource
113       action reward arms armMean sigma2 T delta)
114     (omega : Omega) (t : Nat) (ht : t ∈ source.times)
115     (hgood : omega ∉
116       ConditionalExpectationReward.successorArmEmpiricalMeanFiniteArmTimeBadEvent
117         action reward arms armMean sigma2 T delta) :
118     meanGap (fun arm => (armMean arm : Real))
119         source.best (source.chosen omega t) <=
120       2 * selectedPolicySuccessorRadiusAt action sigma2 arms T delta
121         omega t (source.chosen omega t) := by
122   let empiricalMean : Action -> Real := fun arm =>
123     selectedPolicySuccessorEmpiricalMeanAt action reward omega t arm
124   let radius : Action -> Real := fun arm =>
125     selectedPolicySuccessorRadiusAt action sigma2 arms T delta omega t arm
126   have hpair (arm : Action) (harm : arm ∈ arms)
127       (hcount : 0 < ConditionalExpectationReward.successorArmPullCount
128         (action omega) arm (t + 1))
129       (hdeviation : radius arm <=
130         |empiricalMean arm - (armMean arm : Real)|) :
131       omega ∈
132         ConditionalExpectationReward.successorArmEmpiricalMeanFiniteArmTimeBadEvent
133           action reward arms armMean sigma2 T delta := by
134     unfold ConditionalExpectationReward.successorArmEmpiricalMeanFiniteArmTimeBadEvent
135     exact Set.mem_iUnion.mpr
136       ⟨(arm, t), Set.mem_iUnion.mpr
137         ⟨Finset.mem_product.mpr
138           ⟨harm, Finset.mem_range.mpr (source.times_lt t ht)⟩,
139           by simpa [empiricalMean, radius,
140             selectedPolicySuccessorEmpiricalMeanAt,
141             selectedPolicySuccessorRadiusAt] using
142               And.intro hcount hdeviation⟩⟩
143   have hbestAbs :
144       |empiricalMean source.best - (armMean source.best : Real)| <
145         radius source.best := by
146     by_contra hnot
147     exact hgood (hpair source.best source.best_mem
148       (source.best_count_pos omega t ht) (le_of_not_gt hnot))
149   have hchosenAbs :
150       |empiricalMean (source.chosen omega t) -
151           (armMean (source.chosen omega t) : Real)| <
152         radius (source.chosen omega t) := by
153     by_contra hnot
154     exact hgood (hpair (source.chosen omega t)
155       (source.chosen_mem omega t ht)
156       (source.chosen_count_pos omega t ht) (le_of_not_gt hnot))
157   have hbestUpper :
158       (armMean source.best : Real) <=
159         confidenceScore empiricalMean radius source.best := by
160     have hneg :=
161       neg_le_abs (empiricalMean source.best - (armMean source.best : Real))
162     simp only [confidenceScore]
163     linarith
164   have hchosenLower :
165       empiricalMean (source.chosen omega t) -
166           radius (source.chosen omega t) <=
167         (armMean (source.chosen omega t) : Real) := by
168     have hself := le_abs_self
169       (empiricalMean (source.chosen omega t) -
170         (armMean (source.chosen omega t) : Real))
171     linarith
172   have hscore :
173       confidenceScore empiricalMean radius source.best <=
174         confidenceScore empiricalMean radius (source.chosen omega t) := by
175     simpa [empiricalMean, radius, selectedPolicySuccessorIndexAt] using
176       source.score_max omega t ht
177   simpa [radius] using
178     (meanGap_le_two_radius_of_confidenceScore_max
179       (fun arm => (armMean arm : Real)) empiricalMean radius
180       source.best (source.chosen omega t) hbestUpper hchosenLower hscore)
