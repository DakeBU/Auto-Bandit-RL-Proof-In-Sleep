8 /-- One-based centered reward table; sample zero is unused by MOSS. -/
9 def centeredRewardTable {k : ℕ} (mean : Fin k → ℝ) :
10     Fin k → ℕ → UCB.ArmRewardStream k → ℝ :=
11   fun a i table => table i a - mean a

13 /-- Exact MOSS bound on the canonical product space of arbitrary arm laws. -/
14 theorem integral_canonicalReward_regret_le {k : ℕ} (hk : 0 < k)
15     (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν] (n : ℕ) (hkn : k ≤ n)
16     (mean : Fin k → ℝ) (best : Fin k) (hbest : ∀ a, mean a ≤ mean best)
17     (hmean : ∀ a, ∫ r, r ∂ν a = mean a)
18     (hsubG : ∀ a, HasSubgaussianMGF (fun r => r-mean a) 1 (ν a)) :
19     (∫ table, realMeanRegret mean (streamTrace hk n mean (centeredRewardTable mean) table) n
20       ∂UCB.armStreamMeasure ν) ≤ 39*Real.sqrt ((n : ℝ)*k) + ∑ a, (mean best-mean a) := by
21   have hm (a : Fin k) (i : ℕ) : Measurable (fun table : UCB.ArmRewardStream k => table i a) :=
22     (measurable_pi_apply a).comp (measurable_pi_apply i)
23   apply integral_streamTrace_regret_le _ hk n hkn mean _ best hbest
24   · intro a i
25     exact ((hm a i).sub measurable_const).stronglyMeasurable
26   · intro a
27     exact UCB.iIndepFun_armStreamMeasure_coord_sub ν a (mean a)
28   · intro a i
29     have hr : Integrable (fun r : ℝ => r) (ν a) := by
30       have h := (hsubG a).integrable.add (integrable_const (mean a))
31       change Integrable (fun r : ℝ => (r-mean a)+mean a) (ν a) at h
32       simpa only [Pi.add_apply, sub_add_cancel] using h
33     have he := integral_map (μ := UCB.armStreamMeasure ν) (f := fun r : ℝ => r-mean a) (hm a i).aemeasurable
34       (measurable_id.sub measurable_const).aestronglyMeasurable
35     rw [UCB.armStreamMeasure_map_coord ν i a] at he
36     change (∫ table : UCB.ArmRewardStream k, table i a - mean a ∂UCB.armStreamMeasure ν) = 0
37     rw [← he, integral_sub hr (integrable_const _)]
38     simp [hmean a]
39   · intro a i
40     exact UCB.hasSubgaussianMGF_armStreamMeasure_coord_sub ν a (mean a) 1 (hsubG a) i

42 /-- Centering is analytical only: positive-count empirical means use raw rewards. -/
43 theorem mean_add_centeredRewardTable_average {k : ℕ} (mean : Fin k → ℝ)
44     (table : UCB.ArmRewardStream k) (a : Fin k) (s : ℕ) (hs : 0 < s) :
45     mean a + streamMean (centeredRewardTable mean a) table s =
46       (∑ j ∈ Finset.range s, table (j+1) a)/(s : ℝ) := by
47   have hsR : (s : ℝ) ≠ 0 := by positivity
48   simp only [streamMean, peelingSum, centeredRewardTable, Finset.sum_sub_distrib,
49     Finset.sum_const, Finset.card_range, nsmul_eq_mul]
50   field_simp
51   ring

53 /-- After initialization each arm has a positive realized count. -/
54 theorem streamTrace_pullCount_pos {Ω : Type*} {k : ℕ} (hk : 0 < k) (n t : ℕ)
55     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) (a : Fin k) (ht : k ≤ t) :
56     0 < pullCount (streamTrace hk n mean X ω) a t := by
57   apply pullCount_pos_of_eq_before _ _ (lt_of_lt_of_le a.isLt ht)
58   exact action_initial_arm hk n _ _ a

60 /-- The centered execution selects solely from raw empirical rewards and counts.
61 The unknown means cancel; zero-count states are handled by initialization. -/
62 theorem canonicalReward_action_eq_raw {k : ℕ} (hk : 0 < k) (n t : ℕ)
63     (mean : Fin k → ℝ) (table : UCB.ArmRewardStream k) :
64     streamTrace hk n mean (centeredRewardTable mean) table t =
65       action hk n t
66         (fun a => (∑ j ∈ Finset.range (pullCount (streamTrace hk n mean (centeredRewardTable mean) table) a t),
67           table (j+1) a)/(pullCount (streamTrace hk n mean (centeredRewardTable mean) table) a t : ℝ))
68         (fun a => pullCount (streamTrace hk n mean (centeredRewardTable mean) table) a t) := by
69   rw [streamTrace_policy]
70   by_cases ht : t < k
71   · rw [action_of_lt hk n t _ _ ht, action_of_lt hk n t _ _ ht]
72   · congr 1
73     funext a
74     apply mean_add_centeredRewardTable_average
75     exact streamTrace_pullCount_pos hk n t mean _ table a (Nat.le_of_not_gt ht)
