145 /-- Squaring the good-event gap inequality gives the standard UCB count bound. -/
146 theorem pullCount_le_eight_scale_log_div_gap_sq

172 /-- Real threshold used for the horizon-wide pull-count split. -/
173 noncomputable def realPullThreshold
174     (c : Real) (sigma2 : NNReal) (gap : Real) (n : Nat) : Real :=
175   8 * c * (sigma2 : Real) * Real.log ((n + 1 : Nat) : Real) / gap ^ 2
176 
177 /-- Integer split point: one more than the ceiling of the real threshold. -/
178 noncomputable def pullThreshold
179     (c : Real) (sigma2 : NNReal) (gap : Real) (n : Nat) : Nat :=
180   Nat.ceil (realPullThreshold c sigma2 gap n) + 1

182 /-- The one-sided inverse-power tail budget at time `t`. -/
183 noncomputable def indexTail (c : Real) (t : Nat) : ENNReal :=
184   (1 : ENNReal) / (((t + 1 : Nat) : ENNReal) ^ (c - 1))

186 /-- Finite-horizon sum of the source-faithful one-sided index tails. -/
187 noncomputable def constSum (c : Real) (n : Nat) : ENNReal :=
188   (Finset.range n).sum (indexTail c)
