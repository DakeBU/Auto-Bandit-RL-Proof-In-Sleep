4 /-!
5 # Measurable environment-indexed Thompson trajectory kernels
6 
7 The pointwise `HistoryEnvironment` API does not by itself say that feedback
8 laws vary measurably with the environment.  This module records that missing
9 joint regularity and uses Mathlib's kernel-valued Ionescu-Tulcea theorem to
10 construct the complete environment-indexed pair-trajectory kernel.
11 -/

186     {Input : Type u} {Env : Type v} {Output : Type w}

248 /-- Stable internal step family used by the measurable trajectory producer. -/
249 noncomputable def canonicalMeasurableEnvironmentStepKernel
250     {Env : Type u} {Action : Type v} {Reward : Type w}
251     [MeasurableSpace Env] [MeasurableSpace Action] [MeasurableSpace Reward]
252     (algorithm : HistoryAlgorithm Action Reward)
253     (environment : MeasurableHistoryEnvironment Env Action Reward) :
254     (n : Nat) -> ProbabilityTheory.Kernel
255       ((i : Finset.Iic n) -> Env × (Action × Reward))
256       (Env × (Action × Reward))
257   | 0 => retainEnvironmentKernel measurableTrajectoryPrefixEnvironment
258       measurable_measurableTrajectoryPrefixEnvironment
259       ((measurableEnvironmentInitialPairKernel algorithm environment).comap
260         measurableTrajectoryPrefixEnvironment
261         measurable_measurableTrajectoryPrefixEnvironment)
262   | n + 1 => retainEnvironmentKernel measurableTrajectoryPrefixEnvironment
263       (measurable_measurableTrajectoryPrefixEnvironment (n := n + 1))
264       ((measurableEnvironmentHistoryStepKernel algorithm environment n).comap
265         (fun statePrefix : (i : Finset.Iic (n + 1)) ->
266             Env × (Action × Reward) =>
267           (measurableTrajectoryPrefixEnvironment (n := n + 1) statePrefix,
268             measurableTrajectoryPrefixHistory (n := n) statePrefix))
269         measurable_measurableTrajectoryPrefixEnvironmentHistory)

398 /-- Under a fixed input environment, every finite internal prefix retains it. -/
399 theorem measurableTrajectoryPrefixEnvironment_ae_eq_of_traj
400     {Env : Type u} {Action : Type v} {Reward : Type w}
401     [MeasurableSpace Env] [StandardBorelSpace Env]
402     [MeasurableSpace Action] [MeasurableSpace Reward]
403     [Nonempty Action] [Nonempty Reward]
404     (algorithm : HistoryAlgorithm Action Reward)
405     (environment : MeasurableHistoryEnvironment Env Action Reward)
406     (env : Env) (n : Nat) :
407     (measurableTrajectoryPrefixEnvironment
408       (Env := Env) (Action := Action) (Reward := Reward) (n := n + 1)) =ᵐ[
409         ((ProbabilityTheory.Kernel.traj
410           (X := fun _ : Nat => Env × (Action × Reward))
411           (canonicalMeasurableEnvironmentStepKernel algorithm environment) 0)
412           (measurableEnvironmentInitialStatePrefix
413             (Action := Action) (Reward := Reward) env)).map
414           (Preorder.frestrictLe (n + 1))]
415       fun _ : ((i : Finset.Iic (n + 1)) -> Env × (Action × Reward)) => env := by
416   let stepKernel :=
417     canonicalMeasurableEnvironmentStepKernel algorithm environment
418   let seed := measurableEnvironmentInitialStatePrefix
419     (Env := Env) (Action := Action) (Reward := Reward)
420   let internalMeasure : Measure ((k : Nat) -> Env × (Action × Reward)) :=
421     ProbabilityTheory.Kernel.traj stepKernel 0 (seed env)
422   let restrictInternal :
423       ((k : Nat) -> Env × (Action × Reward)) ->
424         ((i : Finset.Iic (n + 1)) -> Env × (Action × Reward)) :=
425     Preorder.frestrictLe (n + 1)
426   have hrestrictInternal : Measurable restrictInternal :=
427     Preorder.measurable_frestrictLe (n + 1)
428   change ∀ᵐ statePrefix ∂internalMeasure.map restrictInternal,
429       measurableTrajectoryPrefixEnvironment
430         (Env := Env) (Action := Action) (Reward := Reward) statePrefix = env
431   apply (ae_map_iff (μ := internalMeasure) hrestrictInternal.aemeasurable
432     (by
433       change MeasurableSet
434         (measurableTrajectoryPrefixEnvironment ⁻¹' {env})
435       exact (measurableSet_singleton env).preimage
436         measurable_measurableTrajectoryPrefixEnvironment)).2
437   let updateInitial := fun trajectory : (k : Nat) -> Env × (Action × Reward) =>
438     Function.updateFinset trajectory (Finset.Iic 0) (seed env)
439   have hupdate : internalMeasure.map updateInitial = internalMeasure := by
440     dsimp [internalMeasure, updateInitial, stepKernel, seed]
441     exact ProbabilityTheory.Kernel.traj_map_updateFinset
442       (X := fun _ : Nat => Env × (Action × Reward))
443       (κ := canonicalMeasurableEnvironmentStepKernel algorithm environment)
444       (measurableEnvironmentInitialStatePrefix
445         (Action := Action) (Reward := Reward) env)
446   rw [← hupdate]
447   apply (ae_map_iff (μ := internalMeasure) (f := updateInitial)
448     (by fun_prop)
449     (by
450       change MeasurableSet
451         ((measurableTrajectoryPrefixEnvironment ∘
452           restrictInternal) ⁻¹' {env})
453       exact (measurableSet_singleton env).preimage
454         (measurable_measurableTrajectoryPrefixEnvironment.comp
455           hrestrictInternal))).2
456   filter_upwards [] with trajectory
457   rfl

459 /-- The generated trajectory kernel has the configured initial pair law. -/
460 theorem canonicalMeasurableEnvironmentTrajectoryKernel_map_eval_zero
461     {Env : Type u} {Action : Type v} {Reward : Type w}
462     [MeasurableSpace Env] [MeasurableSpace Action] [MeasurableSpace Reward]
463     [Nonempty Action] [Nonempty Reward]
464     (algorithm : HistoryAlgorithm Action Reward)
465     (environment : MeasurableHistoryEnvironment Env Action Reward) :
466     (canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment).map
467         (fun trajectory => trajectory 0) =
468       measurableEnvironmentInitialPairKernel algorithm environment := by
469   let stepKernel :=
470     canonicalMeasurableEnvironmentStepKernel algorithm environment
471   let seed := measurableEnvironmentInitialStatePrefix
472     (Env := Env) (Action := Action) (Reward := Reward)
473   let pairTrace := measurableEnvironmentPairTrace
474     (Env := Env) (Action := Action) (Reward := Reward)
475   let evalPairZero := fun trajectory : (n : Nat) -> Action × Reward =>
476     trajectory 0
477   let evalStateOne := fun trajectory : (n : Nat) ->
478       Env × (Action × Reward) => trajectory 1
479   let prefixEnv := measurableTrajectoryPrefixEnvironment
480     (Env := Env) (Action := Action) (Reward := Reward) (n := 0)
481   have hseed : Measurable seed :=
482     measurable_measurableEnvironmentInitialStatePrefix
483   have hpairTrace : Measurable pairTrace :=
484     measurable_measurableEnvironmentPairTrace
485   have hevalPairZero : Measurable evalPairZero := measurable_pi_apply 0
486   have hevalStateOne : Measurable evalStateOne := measurable_pi_apply 1
487   have hprefixEnv : Measurable prefixEnv :=
488     measurable_measurableTrajectoryPrefixEnvironment
489   have hcompDrop : evalPairZero ∘ pairTrace = Prod.snd ∘ evalStateOne := by
490     rfl
