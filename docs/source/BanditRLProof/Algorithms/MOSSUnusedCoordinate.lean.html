23 theorem canonicalHistory_eq_of_complement_eq {k : ℕ} (hk : 0 < k) (n : ℕ)
24     (mean : Fin k → ℝ) (t : ℕ) (target : ℕ × Fin k) (table table' : UCB.ArmRewardStream k)
25     (hc : UCB.armStreamWithoutCoordinate target table = UCB.armStreamWithoutCoordinate target table')
26     (hf : pullCount (canonicalAction hk n mean table) target.2 (t+1) < target.1) :
27     canonicalHistory hk n mean table t = canonicalHistory hk n mean table' t := by
28   apply canonicalHistory_eq_of_eq_consumed
29   intro a j hj
30   have hne : (j+1,a) ≠ target := by
31     intro he
32     have ha : a = target.2 := congrArg Prod.snd he
33     have hi : j+1 = target.1 := congrArg Prod.fst he
34     subst a
35     omega
36   have h := congrFun hc ⟨(j+1,a), hne⟩
37   exact h
