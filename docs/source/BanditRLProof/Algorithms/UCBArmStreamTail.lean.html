5 /-!
6 # Fixed-arm and adaptive-count tails for the UCB arm-stream process
7 
8 The stationary product arm-stream measure makes each fixed arm an independent
9 reward trace with its prescribed kernel law. This module transports centered
10 sub-Gaussian witnesses to that stream space and combines the resulting fixed
11 prefix tails with the compiled UCB fixed-count peeling theorem.
12 -/

20 /-- Every time/arm coordinate has its prescribed stationary kernel law. -/
21 theorem armStreamMeasure_map_coord
22     {K : Nat} (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
23     (i : Nat) (arm : Fin K) :
24     Measure.map (fun stream : ArmRewardStream K => stream i arm)
25         (armStreamMeasure nu) =
26       nu arm := by
27   calc
28     Measure.map (fun stream : ArmRewardStream K => stream i arm)
29         (armStreamMeasure nu) =
30         Measure.map
31           (fun row : Fin K -> Real => row arm)
32           (Measure.map
33             (fun stream : ArmRewardStream K => stream i)
34             (armStreamMeasure nu)) := by
35       rw [Measure.map_map (measurable_pi_apply arm) (measurable_pi_apply i)]
36       rfl
37     _ = Measure.map
38           (fun row : Fin K -> Real => row arm)
39           (Measure.infinitePi fun candidate : Fin K => nu candidate) := by
40       rw [armStreamMeasure, Measure.infinitePi_map_eval]
41     _ = nu arm := by
42       exact Measure.infinitePi_map_eval (fun candidate : Fin K => nu candidate) arm

181 /--
182 Two-sided `delta` confidence theorem for the empirical mean of exactly `k`
183 latent rewards of one arm under the stationary product arm-stream law.
184 
185 The theorem derives positivity of the total proxy variance from `0 < k` and
186 `sigma2 ≠ 0`; callers do not supply a separate aggregate-variance contract.
187 -/
188 theorem measure_armPrefixAverageConfidenceRadius_le_abs_empiricalMean_sub
189     {K : Nat}
190     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
191     (arm : Fin K) (mean : Real) (sigma2 : NNReal)
192     (hsubG : HasSubgaussianMGF
193       (fun reward => reward - mean) sigma2 (nu arm))
194     (k : Nat) (hk : 0 < k) (hsigma2 : sigma2 ≠ 0)
195     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
196     armStreamMeasure nu {stream : ArmRewardStream K |
197         armPrefixAverageConfidenceRadius sigma2 k delta <=
198           |armPrefixEmpiricalMean arm k stream - mean|} <=
199       ENNReal.ofReal delta := by
200   have hkReal : 0 < (k : Real) := by exact_mod_cast hk
201   have hkReal_ne : (k : Real) ≠ 0 := ne_of_gt hkReal
202   have hsigma2Real : 0 < (sigma2 : Real) := by
203     exact_mod_cast (pos_iff_ne_zero.mpr hsigma2)
204   have hvariance :
205       0 < (((Finset.range k).sum (fun _ => sigma2) : NNReal) : Real) := by
206     simpa [nsmul_eq_mul] using mul_pos hkReal hsigma2Real
207   have htail :=
208     Concentration.subGaussian_average_abs_tail_ennreal_delta_of_iIndepFun
209       (armStreamMeasure nu)
210       (iIndepFun_armStreamMeasure_coord_sub nu arm mean)
211       (c := fun _ => sigma2)
212       k hk
213       (fun i _hi =>
214         hasSubgaussianMGF_armStreamMeasure_coord_sub
215           nu arm mean sigma2 hsubG i)
216       hvariance delta hdelta hdelta_le_one
217   have hcenter : forall stream : ArmRewardStream K,
218       ((Finset.range k).sum (fun i => stream i arm - mean)) / (k : Real) =
219         armPrefixEmpiricalMean arm k stream - mean := by
220     intro stream
221     rw [sum_coord_sub_eq_armPrefixSum_sub]
222     unfold armPrefixEmpiricalMean
223     field_simp [hkReal_ne]
224   have hvariance_eq :
225       (Finset.range k).sum (fun _ => sigma2) = k • sigma2 := by
226     simp
227   calc
228     armStreamMeasure nu {stream : ArmRewardStream K |
229         armPrefixAverageConfidenceRadius sigma2 k delta <=
230           |armPrefixEmpiricalMean arm k stream - mean|} =
231         armStreamMeasure nu {stream : ArmRewardStream K |
232           Concentration.subGaussianAverageConfidenceRadius
233               ((Finset.range k).sum (fun _ => sigma2)) k delta <=
234             |((Finset.range k).sum (fun i => stream i arm - mean)) /
235               (k : Real)|} := by
236       congr 1
237       ext stream
238       simp only [Set.mem_setOf_eq, armPrefixAverageConfidenceRadius]
239       rw [hvariance_eq, hcenter stream]
240     _ <= ENNReal.ofReal delta := htail

242 /-- Pair event used to peel an upper selected-reward deviation by pull count. -/
243 def upperDeviationPairs (mean : Real) (threshold : Nat -> Real) :
244     Set (Nat × Real) :=
245   {pair | threshold pair.1 <= pair.2 - (pair.1 : Real) * mean}
246 
247 /-- Pair event used to peel a lower selected-reward deviation by pull count. -/
248 def lowerDeviationPairs (mean : Real) (threshold : Nat -> Real) :
249     Set (Nat × Real) :=
250   {pair | threshold pair.1 <= (pair.1 : Real) * mean - pair.2}
251 
252 /-- Positive-count upper deviation pair event used by UCB index tails. -/
253 def positiveUpperDeviationPairs (mean : Real) (threshold : Nat -> Real) :
254     Set (Nat × Real) :=
255   {pair | 0 < pair.1 ∧
256     threshold pair.1 <= pair.2 - (pair.1 : Real) * mean}
257 
258 /-- Positive-count lower deviation pair event used by UCB index tails. -/
259 def positiveLowerDeviationPairs (mean : Real) (threshold : Nat -> Real) :
260     Set (Nat × Real) :=
261   {pair | 0 < pair.1 ∧
262     threshold pair.1 <= (pair.1 : Real) * mean - pair.2}

312 /--
313 Adaptive-count upper selected-reward tail for the recursive UCB process.
314 
315 The threshold may depend on the realized pull count. Peeling turns the event
316 into a finite sum over all fixed counts `k <= n`, and each term is discharged
317 by the stationary fixed-arm sub-Gaussian prefix tail.
318 -/
319 theorem measure_sumRewards_sub_pullCount_mul_ge_le
320     {K : Nat} (hK : 0 < K) (c : Real)
321     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
322     (arm : Fin K) (mean : Real) (sigma2 : NNReal)
323     (hsubG : HasSubgaussianMGF
324       (fun reward => reward - mean) sigma2 (nu arm))
325     (n : Nat) (threshold : Nat -> Real)
326     (hthreshold : forall k, k <= n -> 0 <= threshold k) :
327     armStreamMeasure nu {stream : ArmRewardStream K |
328         threshold (pullCount (armStreamAction hK c stream) arm n) <=
329           sumRewards (armStreamAction hK c stream)
330               (armStreamReward hK c stream) arm n -
331             (pullCount (armStreamAction hK c stream) arm n : Real) * mean} <=
332       (Finset.range (n + 1)).sum (fun k =>
333         ENNReal.ofReal
334           (Real.exp
335             (-(threshold k) ^ 2 /
336               (2 * (k : Real) * (sigma2 : Real))))) := by
337   classical
338   have hpeel := measure_pullCount_prod_sumRewards_armStreamUCB_mem_le
339     hK c nu arm n (upperDeviationPairs mean threshold)
340   have hprojected : forall k : Nat,
341       exists value : Real,
342         threshold k <= value - (k : Real) * mean := by
343     intro k
344     refine ⟨threshold k + (k : Real) * mean, ?_⟩
345     linarith
346   have hpeel' :
347       armStreamMeasure nu {stream : ArmRewardStream K |
348           threshold (pullCount (armStreamAction hK c stream) arm n) <=
349             sumRewards (armStreamAction hK c stream)
350                 (armStreamReward hK c stream) arm n -
351               (pullCount (armStreamAction hK c stream) arm n : Real) * mean} <=
352         (Finset.range (n + 1)).sum (fun k =>
353           armStreamMeasure nu {stream : ArmRewardStream K |
354             threshold k <=
355               armPrefixSum arm k stream - (k : Real) * mean}) := by
356     simpa [upperDeviationPairs, hprojected] using hpeel
357   refine hpeel'.trans (Finset.sum_le_sum ?_)
358   intro k hk
359   exact measure_armPrefixSum_sub_mul_ge_le
360     nu arm mean sigma2 hsubG k
361       (hthreshold k (Nat.le_of_lt_succ (Finset.mem_range.mp hk)))
