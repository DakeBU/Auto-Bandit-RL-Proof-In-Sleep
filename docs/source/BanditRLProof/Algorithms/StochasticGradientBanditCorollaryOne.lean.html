3 /-!
4 # Two-arm stochastic-gradient bandit: Corollary 1 companion
5 
6 This module closes the bounded Corollary-1 companion from Baudry, Johnson,
7 Vary, Pike-Burke, and Rebeschini, *Does Stochastic Gradient really succeed
8 for Bandits?* (NeurIPS 2025).  It remains on the generated Algorithm-1
9 trajectory and uses a separate fixed learning rate
10 `sqrt (log T / T)` for each source horizon `T`.
11 
12 The proof has the source's two branches.  When
13 `2 * eta * C_eta <= Delta`, the already compiled Theorem 1 applies and its
14 constant term is at most `1 / eta`.  Otherwise the pathwise regret of the
15 actually sampled actions is at most `Delta * T`.  The final theorem combines
16 the branches into an explicit absolute-constant multiple of
17 `sqrt (T * log T)`.
18 
19 This is a direct consumer of Theorem 1.  It is not evidence for the paper's
20 polynomial-regret Theorem 2, a time-varying learning-rate policy, or a
21 general-`K` result.
22 -/

34 /-- Every realized two-arm gap is at most `Delta` when `Delta` is
35 nonnegative. -/
36 theorem twoArmActionGap_le_gap (Delta : Real) (hDelta : 0 <= Delta)
37     (action : Fin 2) :
38     twoArmActionGap Delta action <= Delta := by
39   by_cases haction : action = 0
40   · simp [twoArmActionGap, haction, hDelta]
41   · simp [twoArmActionGap, haction]

43 /-- Pathwise trivial regret bound for the actions actually sampled by the
44 generated process. -/
45 theorem twoArmSampledPseudoRegret_le_gap_mul_horizon
46     {Env : Type v} (Delta : Real) (hDelta : 0 <= Delta) (horizon : Nat)
47     (sample : Env × ((k : Nat) -> Fin 2 × Real)) :
48     twoArmSampledPseudoRegret Delta horizon sample <=
49       Delta * (horizon : Real) := by
50   rw [twoArmSampledPseudoRegret]
51   calc
52     (Finset.range horizon).sum (fun t =>
53         twoArmActionGap Delta (sample.2 t).1) <=
54         (Finset.range horizon).sum (fun _ => Delta) := by
55       apply Finset.sum_le_sum
56       intro t _ht
57       exact twoArmActionGap_le_gap Delta hDelta (sample.2 t).1
58     _ = Delta * (horizon : Real) := by simp [mul_comm]

66 theorem integrable_twoArmSampledPseudoRegret
67     {Env : Type v} [MeasurableSpace Env]
68     (mu : Measure (Env × ((k : Nat) -> Fin 2 × Real))) [IsFiniteMeasure mu]
69     (Delta : Real) (horizon : Nat) :
70     Integrable (twoArmSampledPseudoRegret (Env := Env) Delta horizon) mu := by
71   apply Integrable.of_bound
72     (measurable_twoArmSampledPseudoRegret Delta horizon).aestronglyMeasurable
73     ((horizon : Real) * |Delta|)

75   rw [Real.norm_eq_abs, twoArmSampledPseudoRegret]

77     |(Finset.range horizon).sum (fun t =>

80           |twoArmActionGap Delta (sample.2 t).1|) :=

87       · simp [twoArmActionGap, haction]

90 /-- Integral form of the pathwise `Delta * T` bound.  The measure is the
91 actual generated trajectory measure in the Corollary-1 consumer below. -/
92 theorem integral_twoArmSampledPseudoRegret_le_gap_mul_horizon
93     {Env : Type v} [MeasurableSpace Env]
94     (mu : Measure (Env × ((k : Nat) -> Fin 2 × Real)))
95     [IsProbabilityMeasure mu]
96     (Delta : Real) (hDelta : 0 <= Delta) (horizon : Nat) :
97     integral mu (twoArmSampledPseudoRegret (Env := Env) Delta horizon) <=
98       Delta * (horizon : Real) := by
99   calc
100     integral mu (twoArmSampledPseudoRegret (Env := Env) Delta horizon) <=

177 /-- The horizon-indexed learning rate times the horizon is exactly the
178 square-root rate. -/
179 theorem corollaryOneEta_mul_horizon_eq_rate
180     (horizon : Nat) (hhorizon : 2 <= horizon) :
181     corollaryOneEta horizon * (horizon : Real) =
182       corollaryOneRate horizon := by
183   have hx : (0 : Real) < (horizon : Real) := by positivity
184   have hlog0 : 0 <= Real.log (horizon : Real) := by
185     have hhorizon_one : 1 <= horizon := by omega
186     exact Real.log_nonneg (by exact_mod_cast hhorizon_one)
187   have hetasq := corollaryOneEta_sq horizon hhorizon
188   have hratesq : corollaryOneRate horizon ^ 2 =
189       (horizon : Real) * Real.log (horizon : Real) := by
190     unfold corollaryOneRate
191     exact Real.sq_sqrt (mul_nonneg hx.le hlog0)
192   have hleft0 : 0 <= corollaryOneEta horizon * (horizon : Real) :=
193     mul_nonneg (Real.sqrt_nonneg _) hx.le
194   have hright0 := corollaryOneRate_nonneg horizon
195   have hsq :
196       (corollaryOneEta horizon * (horizon : Real)) ^ 2 =
197         corollaryOneRate horizon ^ 2 := by
198     rw [mul_pow, hetasq, hratesq]
199     field_simp
200   nlinarith

202 theorem corollaryOneEta_mul_rate_eq_log
203     (horizon : Nat) (hhorizon : 2 <= horizon) :
204     corollaryOneEta horizon * corollaryOneRate horizon =
205       Real.log (horizon : Real) := by
206   have hx0 : (horizon : Real) ≠ 0 := by positivity
207   rw [← corollaryOneEta_mul_horizon_eq_rate horizon hhorizon]
208   calc
209     corollaryOneEta horizon *
210           (corollaryOneEta horizon * (horizon : Real)) =
211         corollaryOneEta horizon ^ 2 * (horizon : Real) := by ring
212     _ = Real.log (horizon : Real) := by
213       rw [corollaryOneEta_sq horizon hhorizon]
214       field_simp

216 /-- The inverse learning-rate term is an absolute-constant multiple of the
217 square-root rate.  We keep the exact constant `1 / log 2`, avoiding a hidden
218 asymptotic threshold. -/
219 theorem corollaryOne_inv_eta_le_inv_log_two_mul_rate
220     (horizon : Nat) (hhorizon : 2 <= horizon) :
221     1 / corollaryOneEta horizon <=
222       (1 / Real.log 2) * corollaryOneRate horizon := by
223   have heta := corollaryOneEta_pos horizon hhorizon
224   have hlogTwo : 0 < Real.log 2 := Real.log_pos (by norm_num)
225   have htwoCast : (2 : Real) <= (horizon : Real) := by
226     exact_mod_cast hhorizon
227   have hlogMono : Real.log 2 <= Real.log (horizon : Real) :=
228     Real.log_le_log (by norm_num) htwoCast
229   rw [div_le_iff₀ heta]
230   calc
231     1 <= Real.log (horizon : Real) / Real.log 2 := by
232       exact (le_div_iff₀ hlogTwo).2 (by simpa using hlogMono)
233     _ = (1 / Real.log 2) * corollaryOneRate horizon *
234           corollaryOneEta horizon := by
235       rw [← corollaryOneEta_mul_rate_eq_log horizon hhorizon]
236       field_simp

238 /-- For `T >= 2`, `0 < Delta < 1`, and the Corollary-1 learning rate, the
239 argument of the Theorem-1 logarithm is at most `T^4`. -/
240 theorem corollaryOne_log_argument_le_horizon_pow_four
241     (horizon : Nat) (hhorizon : 2 <= horizon)
242     (Delta : Real) (hDelta : 0 < Delta) (hDelta_lt_one : Delta < 1) :
243     1 + 4 * corollaryOneEta horizon * Delta * (horizon : Real) <=
244       (horizon : Real) ^ 4 := by
245   let x : Real := (horizon : Real)
246   let eta := corollaryOneEta horizon
247   have hx2 : (2 : Real) <= x := by
248     dsimp [x]
249     exact_mod_cast hhorizon
250   have hx0 : 0 <= x := le_trans (by norm_num) hx2
251   have heta0 : 0 <= eta := by
252     dsimp [eta]
253     exact Real.sqrt_nonneg _
254   have heta_le : eta <= 1 := by
255     dsimp [eta]
256     exact corollaryOneEta_le_one horizon hhorizon
257   have hetaDelta_le : eta * Delta <= 1 := by
258     have hDelta_le : Delta <= 1 := hDelta_lt_one.le
259     nlinarith [mul_nonneg (sub_nonneg.mpr heta_le)
260       (sub_nonneg.mpr hDelta_le)]
261   have hscaled : eta * Delta * x <= x := by
262     simpa using mul_le_mul_of_nonneg_right hetaDelta_le hx0
263   have hxsq_ge_two_mul : 2 * x <= x ^ 2 := by
264     nlinarith [mul_nonneg hx0 (sub_nonneg.mpr hx2)]
265   have hxsq_ge_four : 4 <= x ^ 2 := by nlinarith
266   have hfour_sq_le_fourth : 4 * x ^ 2 <= x ^ 4 := by
267     calc
268       4 * x ^ 2 <= x ^ 2 * x ^ 2 :=
269         mul_le_mul_of_nonneg_right hxsq_ge_four (sq_nonneg x)
270       _ = x ^ 4 := by ring
271   have hlinear_le_fourth : 1 + 4 * x <= x ^ 4 := by
272     calc
273       1 + 4 * x <= 8 * x := by nlinarith
274       _ = 4 * (2 * x) := by ring
275       _ <= 4 * x ^ 2 := by nlinarith
276       _ <= x ^ 4 := hfour_sq_le_fourth
277   dsimp [x, eta] at hscaled hlinear_le_fourth ⊢
278   linarith

281 square-root rate. -/
282 theorem corollaryOne_log_term_le_two_mul_rate
283     (horizon : Nat) (hhorizon : 2 <= horizon)
284     (Delta : Real) (hDelta : 0 < Delta) (hDelta_lt_one : Delta < 1) :
285     Real.log
286         (1 + 4 * corollaryOneEta horizon * Delta * (horizon : Real)) /

311 /-- On the complementary branch, the pathwise `Delta * T` bound is still a
312 square-root rate because failure of `2 * eta * C_eta <= Delta` forces the gap
313 below the learning-rate scale. -/
314 theorem corollaryOne_gap_mul_horizon_le_exp_constant_mul_rate
315     (horizon : Nat) (hhorizon : 2 <= horizon) (Delta : Real)
316     (hlarge : ¬ 2 * corollaryOneEta horizon *
317       sourceC (corollaryOneEta horizon) <= Delta) :
318     Delta * (horizon : Real) <=
319       (2 * Real.exp 2) * corollaryOneRate horizon := by
320   let eta := corollaryOneEta horizon
321   have hx0 : 0 <= (horizon : Real) := by positivity
322   have heta0 : 0 <= eta := by
323     dsimp [eta]
324     exact Real.sqrt_nonneg _
325   have heta_le : eta <= 1 := by
326     dsimp [eta]
327     exact corollaryOneEta_le_one horizon hhorizon
328   have hC : sourceC eta <= Real.exp (2 * eta) :=
329     sourceC_le_exp_two_mul eta heta0
330   have hexp : Real.exp (2 * eta) <= Real.exp 2 := by
331     exact Real.exp_le_exp.mpr (by linarith)
332   have hCtwo : sourceC eta <= Real.exp 2 := hC.trans hexp
333   have hgap : Delta <= 2 * eta * sourceC eta := by
334     exact (lt_of_not_ge (by simpa [eta] using hlarge)).le
335   have hcoef : 2 * eta * sourceC eta <= 2 * eta * Real.exp 2 := by
336     exact mul_le_mul_of_nonneg_left hCtwo (by positivity)
337   calc
338     Delta * (horizon : Real) <=
339         (2 * eta * sourceC eta) * (horizon : Real) :=
340       mul_le_mul_of_nonneg_right hgap hx0
341     _ <= (2 * eta * Real.exp 2) * (horizon : Real) :=
342       mul_le_mul_of_nonneg_right hcoef hx0
343     _ = (2 * Real.exp 2) * corollaryOneRate horizon := by
344       rw [← corollaryOneEta_mul_horizon_eq_rate horizon hhorizon]
345       dsimp [eta]

471     (mean : Fin 2 -> Real)
472     (hbound : forall arm, ∀ᵐ reward ∂armLaw arm, |reward| <= 1)
