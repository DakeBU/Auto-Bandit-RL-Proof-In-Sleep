4 # Measurable-environment and trajectory interfaces for the two-arm recurrences
5 
6 This module lifts the fixed-history recurrences to the jointly measurable
7 environment kernel used by the canonical SGB trajectory.  It also records one
8 uniform, source-faithful environment contract: every initial and successor
9 reward fiber is supported in `[-1,1]` almost everywhere and has the same fixed
10 arm mean, uniformly over environment values, times, and finite histories.
11 
12 Process / filtration ledger:
13 
14 * the ambient finite measure is the canonical measure
15   `prior ⊗ₘ trajectoryKernel 0 eta environment` on an environment and an
16   infinite action/reward trace; it is a probability measure only when the
17   supplied `prior` is one;
18 * at successor index `n`, the conditioning statistic retains the environment
19   and trace pairs `0,...,n`; its generated next pair is trace pair `n+1`;
20 * the retained history parameter is source `theta_{.,n+2}` and adding the
21   next source update produces `theta_{.,n+3}`;
22 * the prefix statistic retains the latent environment as well as the observed
23   pairs.  Thus the filtration below is an environment-revealed filtration for
24   a general prior; it agrees with the fixed-instance source reading after
25   specializing to a deterministic/Dirac environment;
26 * `trajectoryMeasure_condDistrib_nextPair_given_environment_prefix` identifies
27   the regular conditional distribution of that next pair with the actual
28   measurable history-step kernel;
29 * no independence or `Delta > eta * sourceC eta` premise is used.
30 
31 The final trajectory declarations are conditional-distribution integral
32 bounds.  Mathlib's `condExp_prod_ae_eq_integral_condDistrib` would turn these
33 into function-valued conditional-expectation bounds after proving global
34 integrability of the successor exponential potential.  The current local
35 fiber bounds do not yet establish that path-level integrability over every
36 generated prefix.  This module does construct the nested prefix filtration
37 needed by `condExp_condExp_of_le`; global integrability of the two exponential
38 processes is the remaining conditional-expectation/tower blocker and is left
39 as a named downstream obligation rather than a hidden assumption.
40 -/

52 /-- The forward successor exponential at a retained finite two-arm history. -/
53 def twoArmForwardSuccessorPotential (eta : Real) {n : Nat}
54     (history : History.FinitePairHistory (Fin 2) Real n)
55     (pair : Fin 2 × Real) : Real :=
56   Real.exp
57     (2 *
58       (historyParameter (fun _ : Fin 2 => 0) eta n history 0 +
59         eta * sourceIncrement
60           (softmaxProbability
61             (historyParameter (fun _ : Fin 2 => 0) eta n history))
62           pair.2 pair.1 0))

67     (pair : Fin 2 × Real) : Real :=
68   Real.exp
69     (-2 *
70       (historyParameter (fun _ : Fin 2 => 0) eta n history 0 +
71         eta * sourceIncrement
72           (softmaxProbability
73             (historyParameter (fun _ : Fin 2 => 0) eta n history))
74           pair.2 pair.1 0))

76 /-- The additive right side of the forward conditional recurrence. -/
77 def twoArmForwardRecurrenceBound (eta Delta : Real) {n : Nat}
78     (history : History.FinitePairHistory (Fin 2) Real n) : Real :=
79   Real.exp
80       (2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) +
81     2 *
82       softmaxProbability
83           (historyParameter (fun _ : Fin 2 => 0) eta n history) 0 ^ 2 *
84       (eta * Delta + eta ^ 2 * sourceC eta)
85 
86 /-- The additive right side of the inverse conditional recurrence. -/
87 def twoArmInverseRecurrenceBound (eta Delta : Real) {n : Nat}
88     (history : History.FinitePairHistory (Fin 2) Real n) : Real :=
89   Real.exp
90       (-2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) -
91     2 * eta *
92       (1 - softmaxProbability
93           (historyParameter (fun _ : Fin 2 => 0) eta n history) 0) ^ 2 *
94       (Delta - eta * sourceC eta)

96 theorem measurable_twoArmForwardSuccessorPotential (eta : Real) (n : Nat) :
97     Measurable
98       (fun input : History.FinitePairHistory (Fin 2) Real n ×
99           (Fin 2 × Real) =>
100         twoArmForwardSuccessorPotential eta input.1 input.2) := by
101   let theta := fun input : History.FinitePairHistory (Fin 2) Real n ×
102       (Fin 2 × Real) =>
103     historyParameter (fun _ : Fin 2 => 0) eta n input.1
104   have htheta (arm : Fin 2) : Measurable (fun input => theta input arm) :=
105     (measurable_historyParameter (fun _ : Fin 2 => 0) eta n arm).comp
106       measurable_fst
107   have hprob : Measurable
108       (fun input => softmaxProbability (theta input) 0) :=
109     measurable_softmaxProbability theta htheta 0
110   have hincrement : Measurable
111       (fun input => sourceIncrement (softmaxProbability (theta input))
112         input.2.2 input.2.1 0) :=
113     measurable_sourceIncrement
114       (fun input => softmaxProbability (theta input))
115       (fun input => input.2.2) (fun input => input.2.1) 0 hprob
116       (measurable_snd.comp measurable_snd)
117       (measurable_fst.comp measurable_snd)
118   have hexponent : Measurable
119       (fun input => 2 * (theta input 0 + eta *
120         sourceIncrement (softmaxProbability (theta input))
121           input.2.2 input.2.1 0)) :=
122     measurable_const.mul ((htheta 0).add (measurable_const.mul hincrement))
123   simpa [twoArmForwardSuccessorPotential, theta] using
124     Real.measurable_exp.comp hexponent

189 /--
190 Uniform source model for the two-arm Theorem-1 route.  The same `mean` is used
191 at the initial pair and at every history-dependent successor reward law.  This
192 is a conditional fixed-mean contract, not an independence assertion.
193 -/
194 structure TwoArmBoundedFixedMeanEnvironmentContract
195     {Env : Type v} [MeasurableSpace Env]
196     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
197     (mean : Fin 2 -> Real) : Prop where
198   initial_reward_bound : forall env selected,
199     ∀ᵐ reward ∂environment.initialFeedback (env, selected), |reward| <= 1
200   initial_mean : forall env selected,
201     integral (environment.initialFeedback (env, selected)) id = mean selected
202   successor_reward_bound : forall n env
203       (history : History.FinitePairHistory (Fin 2) Real n) selected,
204     ∀ᵐ reward ∂environment.feedback n (env, (history, selected)),
205       |reward| <= 1
206   successor_mean : forall n env
207       (history : History.FinitePairHistory (Fin 2) Real n) selected,
208     integral (environment.feedback n (env, (history, selected))) id =
209       mean selected

368 theorem twoArmPrefixSigma_mono
369     {Env : Type v} [MeasurableSpace Env] {n m : Nat} (hnm : n <= m) :
370     twoArmPrefixSigma (Env := Env) n <= twoArmPrefixSigma (Env := Env) m := by
371   let reduce :
372       Env × History.FinitePairHistory (Fin 2) Real m ->
373         Env × History.FinitePairHistory (Fin 2) Real n :=
374     fun context =>
375       (context.1,
376         Preorder.frestrictLe₂ (π := fun _ : Nat => Fin 2 × Real)
377           hnm context.2)
378   have hreduce : Measurable reduce :=
379     measurable_fst.prodMk
380       ((Preorder.measurable_frestrictLe₂ hnm).comp measurable_snd)
381   have hfactor :
382       twoArmEnvironmentPrefix (Env := Env) n =
383         reduce ∘ twoArmEnvironmentPrefix m := by
384     funext sample
385     apply Prod.ext
386     · rfl
387     · exact (congrFun
388         (Preorder.frestrictLe₂_comp_frestrictLe
389           (π := fun _ : Nat => Fin 2 × Real) hnm) sample.2).symm
390   unfold twoArmPrefixSigma
391   rw [hfactor, ← MeasurableSpace.comap_comp]
392   exact MeasurableSpace.comap_mono hreduce.comap_le

427 /-- On the actual canonical trajectory, the conditional-distribution integral
428 of the forward successor potential obeys the source recurrence for almost
429 every retained environment/prefix. -/
430 theorem trajectoryPrefix_condDistrib_integral_forwardSuccessor_le
431     {Env : Type v} [MeasurableSpace Env] [StandardBorelSpace Env]
432     (prior : Measure Env) [IsFiniteMeasure prior]
433     (eta Delta : Real) (heta : 0 <= eta)
434     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
435     (mean : Fin 2 -> Real)
436     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean)
437     (hgap : mean 0 - mean 1 = Delta)
438     (n : Nat) :
439     ∀ᵐ context
440         ∂(twoArmTrajectoryMeasure prior eta environment).map
441           (twoArmEnvironmentPrefix n),
442       integral
443           (condDistrib (twoArmNextPair n) (twoArmEnvironmentPrefix n)
444             (twoArmTrajectoryMeasure prior eta environment) context)
445           (twoArmForwardSuccessorPotential eta context.2) <=
446         twoArmForwardRecurrenceBound eta Delta context.2 := by
447   have hcond :
448       condDistrib (twoArmNextPair n) (twoArmEnvironmentPrefix n)
449           (twoArmTrajectoryMeasure prior eta environment) =ᵐ[
450         (twoArmTrajectoryMeasure prior eta environment).map
451           (twoArmEnvironmentPrefix n)]
452         Thompson.measurableEnvironmentHistoryStepKernel
453           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment n := by
454     simpa [twoArmTrajectoryMeasure, twoArmEnvironmentPrefix,
455       twoArmNextPair] using
456       (trajectoryMeasure_condDistrib_nextPair_given_environment_prefix
457         prior (fun _ : Fin 2 => 0) eta environment n)
458   filter_upwards [hcond] with context hkernel
459   rw [hkernel]
460   exact
461     integral_measurableTwoArmHistoryStepKernel_forwardSuccessor_le_of_contract
462       eta Delta heta environment mean contract hgap n context.1 context.2
