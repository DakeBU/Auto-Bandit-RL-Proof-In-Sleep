4 /-!
5 # Rat arm-law transport to the Real ETC kernel regret surface
6 
7 This module pushes the existing `Rat` arm laws forward along the cast to
8 `Real`, identifies the resulting identity-integral kernel means and gaps, and
9 then assembles the canonical exact per-arm ETC pull-count bounds into the
10 LML-shaped finite sum for Real kernel regret.
11 
12 The result is still tied to the canonical generated `Rat` reward history.  It
13 does not transport an arbitrary native `Real` `IsAlgEnvSeq` process or identify
14 the local fold argmax with upstream `measurableArgmax` tie semantics.
15 -/

23 /-- Push each countable `Rat` arm law forward to a Real-valued arm kernel. -/
24 noncomputable def ratArmLawRealKernel {K : Nat}
25     (armLaw : Fin K -> Measure Rat) :
26     ProbabilityTheory.Kernel (Fin K) Real :=
27   ProbabilityTheory.Kernel.ofFunOfCountable
28     (fun arm =>
29       Measure.map (fun reward : Rat => ((reward : Rat) : Real)) (armLaw arm))
30 
31 @[simp]
32 theorem ratArmLawRealKernel_apply {K : Nat}
33     (armLaw : Fin K -> Measure Rat) (arm : Fin K) :
34     ratArmLawRealKernel armLaw arm =
35       Measure.map (fun reward : Rat => ((reward : Rat) : Real))
36         (armLaw arm) := by
37   rfl
38 
39 /-- Probability arm laws give a Markov Real pushforward kernel. -/
40 theorem isMarkovKernel_ratArmLawRealKernel {K : Nat}
41     (armLaw : Fin K -> Measure Rat)
42     (hprob : forall arm, IsProbabilityMeasure (armLaw arm)) :
43     ProbabilityTheory.IsMarkovKernel (ratArmLawRealKernel armLaw) := by
44   constructor
45   intro arm
46   rw [ratArmLawRealKernel_apply]
47   exact
48     Measure.isProbabilityMeasure_map
49       (measurable_of_countable
50         (fun reward : Rat => ((reward : Rat) : Real))).aemeasurable
51 
52 /-- The pushforward kernel identity integral is the original casted mean. -/
53 theorem realKernelMean_ratArmLawRealKernel_eq_integral_cast {K : Nat}
54     (armLaw : Fin K -> Measure Rat) (arm : Fin K) :
55     realKernelMean (ratArmLawRealKernel armLaw) arm =
56       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) := by
57   rw [realKernelMean, ratArmLawRealKernel_apply]
58   exact
59     integral_map
60       (measurable_of_countable
61         (fun reward : Rat => ((reward : Rat) : Real))).aemeasurable
62       measurable_id.aestronglyMeasurable

64 /-- Exact arm means identify the pushforward Real kernel mean with the model mean. -/
65 theorem realKernelMean_ratArmLawRealKernel_eq_modelMean {K : Nat}
66     (model : FiniteBanditModel K)
67     (armLaw : Fin K -> Measure Rat)
68     (hmean : forall arm, integral (armLaw arm)
69       (fun reward : Rat => ((reward : Rat) : Real)) =
70         ((model.mean arm : Rat) : Real))
71     (arm : Fin K) :
72     realKernelMean (ratArmLawRealKernel armLaw) arm =
73       ((model.mean arm : Rat) : Real) := by
74   rw [realKernelMean_ratArmLawRealKernel_eq_integral_cast]
75   exact hmean arm
76 
77 /-- The supremum of the cast model means is attained at the local best arm. -/
78 theorem ciSup_modelMean_cast_eq_bestArm {K : Nat}
79     (model : FiniteBanditModel K) :
80     (⨆ arm : Fin K, ((model.mean arm : Rat) : Real)) =
81       ((model.mean model.bestArm : Rat) : Real) := by
82   letI : Nonempty (Fin K) := ⟨model.bestArm⟩
83   apply le_antisymm
84   · exact ciSup_le (fun arm => by
85       exact_mod_cast FiniteBanditModel.mean_le_bestArm_mean model arm)
86   · exact le_ciSup (f := fun arm : Fin K => ((model.mean arm : Rat) : Real))
87       (by simp) model.bestArm
88 
89 /-- The pushforward Real kernel gap is exactly the cast local model gap. -/
90 theorem realKernelGap_ratArmLawRealKernel_eq_modelGap {K : Nat}
91     (model : FiniteBanditModel K)
92     (armLaw : Fin K -> Measure Rat)
93     (hmean : forall arm, integral (armLaw arm)
94       (fun reward : Rat => ((reward : Rat) : Real)) =
95         ((model.mean arm : Rat) : Real))
96     (arm : Fin K) :
97     realKernelGap (ratArmLawRealKernel armLaw) arm =
98       ((model.gap arm : Rat) : Real) := by
99   rw [realKernelGap, realMeanGap]
100   simp_rw [realKernelMean_ratArmLawRealKernel_eq_modelMean model armLaw hmean]
101   rw [ciSup_modelMean_cast_eq_bestArm]
102   by_cases hbest : arm = model.bestArm
103   · subst arm
104     simp
105   · simp [FiniteBanditModel.gap, FiniteBanditModel.bestMean, hbest,
106       Rat.cast_sub]

108 /--
109 Canonical exact ETC expected regret against the Real pushforward arm kernel.
110 
111 This is the finite-arm assembly of the exact per-arm expected pull-count leaf:
112 the best-arm summand vanishes, while each non-best summand uses the exact
113 `exp (-m * gap^2 / (4 * sigma2))` count bound.
114 -/
115 theorem integral_realKernelRegret_explorationArgmaxAction_le_exact_sum_of_armLaws
116     {K : Nat} {Context : Type}
117     [MeasurableSpace Context]
118     (spec : ETC.Spec K) (model : FiniteBanditModel K)
119     (armLaw : Fin K -> Measure Rat)
120     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
121     (sigma2 : NNReal)
122     (hmean : forall arm, integral (armLaw arm)
123       (fun reward : Rat => ((reward : Rat) : Real)) =
124         ((model.mean arm : Rat) : Real))
125     (hsubG : forall arm, ProbabilityTheory.HasSubgaussianMGF
126       (fun reward : Rat =>
127         ((reward : Rat) : Real) - ((model.mean arm : Rat) : Real))
128       sigma2 (armLaw arm))
129     (context : (n : Nat) -> ((j : Finset.Iic n) -> Rat) -> Context)
130     (hcontext : forall n : Nat, Measurable (context n))
131     (hexplorationPulls_pos : 0 < spec.explorationPulls)
132     (n : Nat) (hn : K * spec.explorationPulls <= n) :
133     let defaultAction := ETC.exploreArm spec 0
134     let rewardKernel := RewardKernel.contextIndependentOfActionLaws
135       (Context := Context) armLaw hprob
136     let policy := fun t => ETC.explorationArgmaxHistoryPolicy spec model t
137     let state := fun t history => ETC.explorationArgmaxHistoryState t history
138     let stepKernel :=
139       RewardKernel.historyStepKernelFamily rewardKernel policy context state
140         hcontext (fun t => ETC.measurable_explorationArgmaxHistoryState t)
141     let trajMeasure := ProbabilityTheory.Kernel.trajMeasure
142       (X := fun _ : Nat => Rat) (armLaw defaultAction) stepKernel
143     integral trajMeasure
144         (fun trajectory : RewardTrace Rat =>
145           realKernelRegret (ratArmLawRealKernel armLaw)
146             (ETC.explorationArgmaxAction spec model trajectory) n) <=
147       (Finset.univ : Finset (Fin K)).sum (fun arm =>
148         ((model.gap arm : Rat) : Real) *
149           ((spec.explorationPulls : Real) +
150             ((n - K * spec.explorationPulls : Nat) : Real) *
151               Real.exp
152                 (-(spec.explorationPulls : Real) *
153                   ((model.gap arm : Rat) : Real) ^ 2 /
154                     (4 * (sigma2 : Real))))) := by
155   dsimp only
156   let defaultAction := ETC.exploreArm spec 0
157   let rewardKernel := RewardKernel.contextIndependentOfActionLaws
158     (Context := Context) armLaw hprob
159   let policy := fun t => ETC.explorationArgmaxHistoryPolicy spec model t
160   let state := fun t history => ETC.explorationArgmaxHistoryState t history
161   let stepKernel :=
162     RewardKernel.historyStepKernelFamily rewardKernel policy context state
163       hcontext (fun t => ETC.measurable_explorationArgmaxHistoryState t)
164   letI : IsProbabilityMeasure (armLaw defaultAction) := hprob defaultAction
165   let trajMeasure := ProbabilityTheory.Kernel.trajMeasure
166     (X := fun _ : Nat => Rat) (armLaw defaultAction) stepKernel
167   let commit : RewardTrace Rat -> Fin K :=
168     fun trajectory => ETC.explorationArgmaxCommit spec model trajectory
169   let action : RewardTrace Rat -> ActionTrace (Fin K) :=
170     fun trajectory => ETC.explorationArgmaxAction spec model trajectory
171   have hmeas_coord : forall arm : Fin K,
172       Measurable (fun trajectory : RewardTrace Rat =>
173         ETC.empMeanAtExploration spec model.bestArm trajectory arm) := by
174     exact ETC.measurable_empMeanAtExploration_coordinates
175       (spec := spec) (commitArm := model.bestArm)
176       (reward := fun trajectory : RewardTrace Rat => trajectory)
177       (hreward := fun t => measurable_pi_apply t)
178   have hmeas_commit : Measurable commit := by
179     simpa [commit, ETC.explorationArgmaxCommit,
180       ETC.fixedProductArgmaxCommit] using
181       (ETC.measurable_commitOracle_choose_of_forall_measurable_empMean
182         (oracle := ETC.argmaxCommitOracle model.hK)
183         (empMean := fun trajectory : RewardTrace Rat => fun arm : Fin K =>
184           ETC.empMeanAtExploration spec model.bestArm trajectory arm)
185         hmeas_coord)
186   have hcount : forall arm : Fin K,
187       Integrable
188         (fun trajectory : RewardTrace Rat =>
189           (pullCount (action trajectory) arm n : Real)) trajMeasure := by
190     intro arm
191     simpa [action, commit, ETC.explorationArgmaxAction,
192       ETC.fixedProductArgmaxAction] using
193       (ETC.integrable_real_pullCount_actionWithCommit_choice_of_measurable_commit
194         trajMeasure spec commit arm n hmeas_commit)
195   rw [show (fun trajectory : RewardTrace Rat =>
196       realKernelRegret (ratArmLawRealKernel armLaw)
197         (ETC.explorationArgmaxAction spec model trajectory) n) =
198       (fun trajectory : RewardTrace Rat =>
199         realKernelRegret (ratArmLawRealKernel armLaw)
200           (action trajectory) n) by rfl]
201   rw [integral_realKernelRegret_eq_sum_gap_mul_integral_pullCount
202     trajMeasure (ratArmLawRealKernel armLaw) action n hcount]
203   refine Finset.sum_le_sum ?_
204   intro arm _harm
205   rw [realKernelGap_ratArmLawRealKernel_eq_modelGap model armLaw hmean]
206   by_cases hbest : arm = model.bestArm
207   · subst arm
208     simp
209   · exact mul_le_mul_of_nonneg_left
210       (ETC.integral_real_pullCount_explorationArgmaxAction_le_exploration_add_remaining_mul_exp_of_armLaws
211         spec model armLaw hprob sigma2 hmean hsubG context hcontext
212         hexplorationPulls_pos arm n hn (fun h => hbest h))
213       ((Rat.cast_nonneg (K := Real)).2
214         (FiniteBanditModel.gap_nonneg model arm))
