4 # Native Real UCB LML field compatibility surface
5 
6 The pinned LML source currently uses a newer Lean/mathlib toolchain, so ABRL
7 cannot import its `IsAlgEnvSeq` declaration directly. This module packages the
8 exact measurable action/feedback and split conditional-law consequences used
9 by the local UCB trajectory and regret route. It is a local compatibility
10 structure, not an imported LML proof.
11 -/

18 /--
19 The exact `IsAlgEnvSeq`-shaped fields consumed by the local stationary Real UCB
20 route.
21 
22 The law fields correspond to the pinned source's initial action law, initial
23 feedback law, successor action law given finite observable history, and
24 successor feedback law given history and the new action. They are stated as
25 Mathlib `condDistrib` equalities because LML's `HasCondDistrib` symbol is not a
26 local dependency.
27 -/
28 structure RealStationaryUCBSequence
29     {Omega : Type u} {K : Nat} [NeZero K] [MeasurableSpace Omega]
30     (mu : Measure Omega) [IsFiniteMeasure mu]
31     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
32     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
33     (action : Omega -> ActionTrace (Fin K))
34     (reward : Omega -> RewardTrace Real) : Prop where
35   measurable_action : forall t, Measurable (fun omega => action omega t)
36   measurable_reward : forall t, Measurable (fun omega => reward omega t)
37   hasLaw_action_zero :
38     Measure.map (fun omega => action omega 0) mu =
39       Measure.map
40         (fun stream : ArmRewardStream K =>
41           armStreamAction hK (c * (sigma2 : Real)) stream 0)
42         (armStreamMeasure nu)
43   hasCondDistrib_feedback_zero :
44     condDistrib (fun omega => reward omega 0)
45         (fun omega => action omega 0) mu =ᵐ[
46           mu.map (fun omega => action omega 0)]
47       condDistrib
48         (fun stream : ArmRewardStream K =>
49           armStreamReward hK (c * (sigma2 : Real)) stream 0)
50         (fun stream : ArmRewardStream K =>
51           armStreamAction hK (c * (sigma2 : Real)) stream 0)
52         (armStreamMeasure nu)
53   hasCondDistrib_action : forall i,
54     condDistrib (fun omega => action omega (i + 1))
55         (fun omega => History.finitePairHistoryOfTrace
56           (action omega) (reward omega) i) mu =ᵐ[
57           mu.map (fun omega => History.finitePairHistoryOfTrace
58             (action omega) (reward omega) i)]
59       condDistrib
60         (fun stream : ArmRewardStream K =>
61           armStreamAction hK (c * (sigma2 : Real)) stream (i + 1))
62         (fun stream : ArmRewardStream K =>
63           History.finitePairHistoryOfTrace
64             (armStreamAction hK (c * (sigma2 : Real)) stream)
65             (armStreamReward hK (c * (sigma2 : Real)) stream) i)
66         (armStreamMeasure nu)
67   hasCondDistrib_feedback : forall i,
68     condDistrib (fun omega => reward omega (i + 1))
69         (fun omega =>
70           (History.finitePairHistoryOfTrace
71               (action omega) (reward omega) i,
72             action omega (i + 1))) mu =ᵐ[
73           mu.map (fun omega =>
74             (History.finitePairHistoryOfTrace
75                 (action omega) (reward omega) i,
76               action omega (i + 1)))]
77       condDistrib
78         (fun stream : ArmRewardStream K =>
79           armStreamReward hK (c * (sigma2 : Real)) stream (i + 1))
80         (fun stream : ArmRewardStream K =>
81           (History.finitePairHistoryOfTrace
82               (armStreamAction hK (c * (sigma2 : Real)) stream)
83               (armStreamReward hK (c * (sigma2 : Real)) stream) i,
84             armStreamAction hK (c * (sigma2 : Real)) stream (i + 1)))
85         (armStreamMeasure nu)

88 theorem realStationaryUCBSequence_armStream
89     {K : Nat} [NeZero K] (hK : 0 < K) (c : Real) (sigma2 : NNReal)
90     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
91     RealStationaryUCBSequence (armStreamMeasure nu) hK c sigma2 nu
92       (armStreamAction hK (c * (sigma2 : Real)))
93       (armStreamReward hK (c * (sigma2 : Real))) := by
94   refine
95     { measurable_action := measurable_armStreamAction hK (c * (sigma2 : Real))
96       measurable_reward := measurable_armStreamReward hK (c * (sigma2 : Real))
97       hasLaw_action_zero := rfl
98       hasCondDistrib_feedback_zero :=
99         Filter.Eventually.of_forall (fun _action => rfl)
100       hasCondDistrib_action := fun _i =>
101         Filter.Eventually.of_forall (fun _history => rfl)
102       hasCondDistrib_feedback := fun _i =>
103         Filter.Eventually.of_forall (fun _historyAction => rfl) }

136 theorem regret_le_of_realStationaryUCBSequence
137     {Omega : Type u} {K : Nat} [NeZero K] [MeasurableSpace Omega]
138     (mu : Measure Omega) [IsFiniteMeasure mu]
139     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
140     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
141     (action : Omega -> ActionTrace (Fin K))
142     (reward : Omega -> RewardTrace Real)
143     (h : RealStationaryUCBSequence mu hK c sigma2 nu action reward)
144     (n : Nat) (hc : 0 < c) (hsigma2 : sigma2 ≠ 0)
145     (hsubG : forall arm : Fin K, HasSubgaussianMGF
146       (fun x => x - realKernelMean nu arm) sigma2 (nu arm)) :
147     integral mu (fun omega => realKernelRegret nu (action omega) n) <=
148       (Finset.univ : Finset (Fin K)).sum (fun arm =>
149         8 * c * (sigma2 : Real) * Real.log ((n + 1 : Nat) : Real) /
150             realKernelGap nu arm +
151           realKernelGap nu arm * (2 + 2 * (constSum c n).toReal)) := by
152   exact
153     integral_realKernelRegret_externalAction_le_lml_sum_of_split_condDistrib_eq_armStream
154       hK c sigma2 mu nu action reward h.measurable_action h.measurable_reward
155       h.hasLaw_action_zero h.hasCondDistrib_feedback_zero
