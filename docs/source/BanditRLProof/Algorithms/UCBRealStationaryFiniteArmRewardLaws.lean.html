5 # External stationary UCB expected consistency
6 
7 This module transports the canonical one-policy arm-stream asymptotics through

9 endpoint instantiates the route from armwise-bounded Real reward laws.

19 /-- Expected Real pseudo-regret of one fixed external action process. -/
20 noncomputable def realStationaryExpectedRegret
21     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
22     (mu : Measure Omega) (nu : Kernel (Fin K) Real)
23     (action : Omega -> ActionTrace (Fin K)) (n : Nat) : Real :=
24   integral mu (fun omega => realKernelRegret nu (action omega) n)

26 /--
27 The stationary UCB field bundle identifies each external expected-regret term
28 exactly with the corresponding canonical arm-stream term at `c = 4`.
29 -/
30 theorem realStationaryExpectedRegret_eq_armStreamExpectedRegret
31     {Omega : Type u} {K : Nat} [NeZero K] [MeasurableSpace Omega]
32     (mu : Measure Omega) [IsFiniteMeasure mu]
33     (hK : 0 < K) (sigma2 : NNReal)
34     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
35     (action : Omega -> ActionTrace (Fin K))
36     (reward : Omega -> RewardTrace Real)
37     (h : RealStationaryUCBSequence mu hK 4 sigma2 nu action reward)
38     (n : Nat) :
39     realStationaryExpectedRegret mu nu action n =
40       armStreamExpectedRegret hK sigma2 nu n := by
41   have htrajectory :=
42     identDistrib_actionRewardTrace_of_realStationaryUCBSequence
43       mu hK 4 sigma2 nu action reward h
44   have haction :=
45     identDistrib_action_of_identDistrib_actionRewardTrace
46       mu (armStreamMeasure nu) action reward
47       (armStreamAction hK (4 * (sigma2 : Real)))
48       (armStreamReward hK (4 * (sigma2 : Real))) htrajectory
49   have hregret :=
50     (haction.comp (measurable_realKernelRegret_actionTrace nu n)).integral_eq
51   simpa [realStationaryExpectedRegret, armStreamExpectedRegret] using hregret

53 /-- External expected regret normalized by `n + 1`. -/
54 noncomputable def realStationaryExpectedAverageRegret
55     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
56     (mu : Measure Omega) (nu : Kernel (Fin K) Real)
57     (action : Omega -> ActionTrace (Fin K)) (n : Nat) : Real :=
58   realStationaryExpectedRegret mu nu action n / ((n + 1 : Nat) : Real)

60 /--
61 Any fixed external process satisfying the stationary UCB field bundle inherits
62 the canonical sub-Gaussian expected-average consistency theorem.
63 -/
64 theorem realStationaryExpectedAverageRegret_tendsto_zero
65     {Omega : Type u} {K : Nat} [NeZero K] [MeasurableSpace Omega]
66     (mu : Measure Omega) [IsFiniteMeasure mu]
67     (hK : 0 < K) (sigma2 : NNReal)
68     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
69     (action : Omega -> ActionTrace (Fin K))
70     (reward : Omega -> RewardTrace Real)
71     (h : RealStationaryUCBSequence mu hK 4 sigma2 nu action reward)
72     (hsigma2 : Ne sigma2 0)
73     (hsubG : forall arm : Fin K, HasSubgaussianMGF
74       (fun x => x - realKernelMean nu arm) sigma2 (nu arm)) :
75     Tendsto (realStationaryExpectedAverageRegret mu nu action)
76       atTop (nhds 0) := by
77   have heq : realStationaryExpectedAverageRegret mu nu action =
78       armStreamExpectedAverageRegret hK sigma2 nu := by
79     funext n
80     simp only [realStationaryExpectedAverageRegret,
81       armStreamExpectedAverageRegret]
82     rw [realStationaryExpectedRegret_eq_armStreamExpectedRegret
83       mu hK sigma2 nu action reward h n]
84   rw [heq]
85   exact armStreamExpectedAverageRegret_tendsto_zero
86     hK sigma2 nu hsigma2 hsubG
