22 theorem canonicalKernelTrajectory_historyAction_map_eq_armStream
23     {K : Nat} [NeZero K]
24     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
25     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] (n : Nat) :
26     Measure.map
27         (fun trajectory =>
28           (History.finitePairHistoryOfTrace
29               (canonicalKernelTrajectoryAction trajectory)
30               (canonicalKernelTrajectoryReward trajectory) n,

32         (canonicalKernelTrajectoryMeasure hK c sigma2 nu) =
33       Measure.map
34         (armStreamHistoryAction hK (c * (sigma2 : Real)) n)
35         (armStreamMeasure nu) := by
36   let pairTrace := fun trajectory : (t : Nat) -> Fin K × Real =>
37     fun t => (canonicalKernelTrajectoryAction trajectory t,
38       canonicalKernelTrajectoryReward trajectory t)
39   let armPairTrace := fun stream : ArmRewardStream K =>
40     fun t => (armStreamAction hK (c * (sigma2 : Real)) stream t,
41       armStreamReward hK (c * (sigma2 : Real)) stream t)
42   let condition := fun trace : (t : Nat) -> Fin K × Real =>
43     (History.finitePairHistoryOfTrace
44         (fun t => (trace t).1) (fun t => (trace t).2) n,
45       (trace (n + 1)).1)
46   have hhistory : Measurable
47       (fun trace : (t : Nat) -> Fin K × Real =>
48         History.finitePairHistoryOfTrace
49           (fun t => (trace t).1) (fun t => (trace t).2) n) :=
50     History.measurable_finitePairHistoryOfTrace
51       (fun trace : (t : Nat) -> Fin K × Real => fun t => (trace t).1)
52       (fun trace : (t : Nat) -> Fin K × Real => fun t => (trace t).2)
53       (fun t => measurable_fst.comp (measurable_pi_apply t))
54       (fun t => measurable_snd.comp (measurable_pi_apply t)) n
55   have hcondition : Measurable condition :=
56     hhistory.prodMk (measurable_fst.comp (measurable_pi_apply (n + 1)))
57   have htrace :=
58     identDistrib_actionRewardTrace_of_realStationaryUCBSequence
59       (canonicalKernelTrajectoryMeasure hK c sigma2 nu)
60       hK c sigma2 nu canonicalKernelTrajectoryAction
61       canonicalKernelTrajectoryReward
62       (realStationaryUCBSequence_canonicalKernelTrajectory
63         hK c sigma2 nu)
64   have hmarginal := htrace.comp hcondition
65   have harmCondition :
66       (fun stream : ArmRewardStream K =>
67         (History.finitePairHistoryOfTrace
68             (armStreamAction hK (c * (sigma2 : Real)) stream)
69             (armStreamReward hK (c * (sigma2 : Real)) stream) n,
70           armStreamAction hK (c * (sigma2 : Real)) stream (n + 1))) =
71         armStreamHistoryAction hK (c * (sigma2 : Real)) n := by
72     funext stream
73     rw [armStreamAction_succ]
74     simp [armStreamHistoryAction,
75       armStreamHistory_eq_finitePairHistoryOfTrace]
76   have hgeneratedComp :
77       condition ∘ pairTrace =
78         (fun trajectory =>
79           (History.finitePairHistoryOfTrace
80               (canonicalKernelTrajectoryAction trajectory)
81               (canonicalKernelTrajectoryReward trajectory) n,
82             canonicalKernelTrajectoryAction trajectory (n + 1))) := by
83     rfl
84   have harmComp :
85       condition ∘ armPairTrace =
86         armStreamHistoryAction hK (c * (sigma2 : Real)) n := by
87     simpa only [condition, armPairTrace, Function.comp_apply] using
88       harmCondition
89   rw [hgeneratedComp, harmComp] at hmarginal
90   exact hmarginal.map_eq
