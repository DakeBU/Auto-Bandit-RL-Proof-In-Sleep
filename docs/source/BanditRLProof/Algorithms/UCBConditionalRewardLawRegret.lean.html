6 # Practical selected-policy UCB pseudo-regret assembly
7 
8 This module connects the explicit expected successor pull-count theorem to the
9 local finite-bandit pseudo-regret surface. The regret action shifts generated
10 coordinates `1, ..., T` to the standard pull-count coordinates `0, ..., T-1`.

28 /-- Textbook-style real contribution of one positive-gap arm's count threshold. -/
29 noncomputable def selectedPolicySuccessorTextbookGapBudget
30     (K : Nat) (sigma2 : NNReal) (T : Nat) (delta gap : Real) : Real :=
31   32 * (((sigma2 : NNReal) : Real)) *
32         selectedPolicySuccessorFiniteArmTimeLogBudget K T T delta / gap +
33     4 * selectedPolicySuccessorFiniteArmTimeLogBudget K T T delta +
34     2 * gap

36 /-- The explicit integer pull threshold is at most its real max plus two. -/
37 theorem selectedPolicySuccessorPullThreshold_cast_le_realThreshold_add_two
38     (K : Nat) (sigma2 : NNReal) (T : Nat) (delta gap : Real)
39     (hgap : 0 < gap) :
40     (selectedPolicySuccessorPullThreshold K sigma2 T delta gap : Real) <=
41       selectedPolicySuccessorRealPullThreshold K sigma2 T delta gap + 2 := by
42   have hbudget :
43       0 <= selectedPolicySuccessorFiniteArmTimeLogBudget K T T delta := by
44     simp [selectedPolicySuccessorFiniteArmTimeLogBudget]
45   have hthreshold :
46       0 <= selectedPolicySuccessorRealPullThreshold K sigma2 T delta gap := by
47     unfold selectedPolicySuccessorRealPullThreshold
48     positivity
49   rw [selectedPolicySuccessorPullThreshold, Nat.cast_add, Nat.cast_one]
50   exact le_of_lt (by
51     have hceil := Nat.ceil_lt_add_one hthreshold
52     linarith)

54 /-- Multiplying the explicit threshold by a positive gap removes one gap power. -/
55 theorem gap_mul_selectedPolicySuccessorPullThreshold_cast_le_textbookGapBudget
56     (K : Nat) (sigma2 : NNReal) (T : Nat) (delta gap : Real)
57     (hgap : 0 < gap) :
58     gap * (selectedPolicySuccessorPullThreshold K sigma2 T delta gap : Real) <=
59       selectedPolicySuccessorTextbookGapBudget K sigma2 T delta gap := by
60   let budget := selectedPolicySuccessorFiniteArmTimeLogBudget K T T delta
61   let variance := (((sigma2 : NNReal) : Real))
62   let quadratic := 32 * variance * budget / gap ^ 2
63   let linear := 4 * budget / gap
64   have hbudget : 0 <= budget := by
65     simp [budget, selectedPolicySuccessorFiniteArmTimeLogBudget]
66   have hquadratic : 0 <= quadratic := by
67     dsimp [quadratic]
68     positivity
69   have hlinear : 0 <= linear := by
70     dsimp [linear]
71     positivity
72   have hmax : max quadratic linear <= quadratic + linear := by
73     exact max_le (le_add_of_nonneg_right hlinear)
74       (le_add_of_nonneg_left hquadratic)
75   have hthreshold :=
76     selectedPolicySuccessorPullThreshold_cast_le_realThreshold_add_two
77       K sigma2 T delta gap hgap
78   have hmul := mul_le_mul_of_nonneg_left hthreshold hgap.le
79   calc
80     gap * (selectedPolicySuccessorPullThreshold K sigma2 T delta gap : Real) <=
81         gap * (max quadratic linear + 2) := by
82       simpa [selectedPolicySuccessorRealPullThreshold, quadratic, linear,
83         variance, budget] using hmul
84     _ <= gap * (quadratic + linear + 2) := by
85       apply mul_le_mul_of_nonneg_left _ hgap.le
86       linarith
87     _ = selectedPolicySuccessorTextbookGapBudget K sigma2 T delta gap := by
88       dsimp [selectedPolicySuccessorTextbookGapBudget, quadratic, linear,
89         variance, budget]
90       field_simp [ne_of_gt hgap]

94     (K : Nat) (sigma2 : NNReal) (T : Nat) (delta gap : Real)

187 /--
188 Generic finite-arm ENNReal pseudo-regret assembly. Only positive-gap arms need
189 a pull-count bound; zero-gap arms disappear after multiplication.
190 -/
191 theorem lintegral_ofReal_pseudoRegret_le_sum_gap_mul_bound_of_positiveGap_pullCount
192     {Omega : Type} [MeasurableSpace Omega]
193     {K : Nat}
194     (mu : Measure Omega)
195     (model : FiniteBanditModel K)
196     (action : Omega -> ActionTrace (Fin K))
197     (haction : forall t : Nat,
198       Measurable (fun omega : Omega => action omega t))
199     (n : Nat) (bound : Fin K -> ENNReal)
200     (hcount : forall arm : Fin K,
201       0 < (((model.gap arm : Rat) : Real)) ->
202         ∫⁻ omega,
203             ((pullCount (action omega) arm n : Nat) : ENNReal) ∂mu <=
204           bound arm) :
205     ∫⁻ omega,
206         ENNReal.ofReal
207           (((pseudoRegret model (action omega) n : Rat) : Real)) ∂mu <=
208       (Finset.univ : Finset (Fin K)).sum (fun arm =>
209         ENNReal.ofReal (((model.gap arm : Rat) : Real)) * bound arm) := by
210   let gap : Fin K -> ENNReal := fun arm =>
211     ENNReal.ofReal (((model.gap arm : Rat) : Real))
212   let count : Fin K -> Omega -> ENNReal := fun arm omega =>
213     ((pullCount (action omega) arm n : Nat) : ENNReal)
214   have hgap_nonneg : forall arm : Fin K,
215       0 <= (((model.gap arm : Rat) : Real)) := by
216     intro arm
217     exact_mod_cast FiniteBanditModel.gap_nonneg model arm
218   have hcount_meas : forall arm : Fin K, Measurable (count arm) := by
219     intro arm
220     exact measurable_natCast_pullCount
221       (Beta := ENNReal) action haction arm n
222   calc
223     ∫⁻ omega,
224         ENNReal.ofReal
225           (((pseudoRegret model (action omega) n : Rat) : Real)) ∂mu =
226       ∫⁻ omega,
227         (Finset.univ : Finset (Fin K)).sum
228           (fun arm => gap arm * count arm omega) ∂mu := by
229         apply lintegral_congr
230         intro omega
231         simpa [gap, count] using
232           ENNReal.ofReal_pseudoRegret_eq_univ_sum_model_gap_ofReal_mul_natCast_pullCount_of_nonneg
233             model (action omega) hgap_nonneg n
234     _ = (Finset.univ : Finset (Fin K)).sum (fun arm =>
235         ∫⁻ omega, gap arm * count arm omega ∂mu) := by
236       rw [lintegral_finset_sum]
237       intro arm _harm
238       exact (hcount_meas arm).const_mul (gap arm)
239     _ = (Finset.univ : Finset (Fin K)).sum (fun arm =>
240         gap arm * ∫⁻ omega, count arm omega ∂mu) := by
241       apply Finset.sum_congr rfl
242       intro arm _harm
243       exact lintegral_const_mul (gap arm) (hcount_meas arm)
244     _ <= (Finset.univ : Finset (Fin K)).sum (fun arm =>
245         gap arm * bound arm) := by
246       apply Finset.sum_le_sum
247       intro arm _harm
248       by_cases hzero : (((model.gap arm : Rat) : Real)) = 0
249       · simp [gap, hzero]
250       · have hpos : 0 < (((model.gap arm : Rat) : Real)) :=
251           lt_of_le_of_ne (hgap_nonneg arm) (Ne.symm hzero)
252         exact mul_le_mul_left' (hcount arm hpos) (gap arm)
253     _ = (Finset.univ : Finset (Fin K)).sum (fun arm =>
254         ENNReal.ofReal (((model.gap arm : Rat) : Real)) * bound arm) := by
255       rfl
