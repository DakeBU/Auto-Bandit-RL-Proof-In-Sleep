5 # Deterministic ETC regret lemmas

7 This module contains ETC-specific deterministic regret scaffolds.  It consumes

9 behavior, empirical means, probability, concentration, and final ETC regret
10 theorems.

16 The pseudo-regret of the pure round-robin ETC exploration prefix is bounded by
17 the sum of arm gaps times the configured number of exploration pulls per arm.
18 
19 This is the `ETC-EXPLORATION-REGRET-BOUND` project-local deterministic regret
20 scaffold.
21 -/
22 theorem ETC.pseudoRegret_exploreArm_explorationPulls_mul_K_le_sum_gap_mul_explorationPulls

24     (spec : ETC.Spec K) (model : FiniteBanditModel K) :
25     pseudoRegret model (ETC.exploreArm spec) (spec.explorationPulls * K) <=
26       ((Finset.univ : Finset (Fin K)).sum
27         (fun a : Fin K => model.gap a)) *
28         (((spec.explorationPulls : Nat) : Rat)) := by
29   exact
30     pseudoRegret_le_sum_gap_mul_uniform_nat_count_bound
31       (model := model)
32       (action := ETC.exploreArm spec)
33       (n := spec.explorationPulls * K)
34       (B := spec.explorationPulls)
35       (hB := by
36         intro a
37         exact le_of_eq
38           (ETC.pullCount_exploreArm_explorationPulls_mul_K_eq spec a))

70 /--
71 The fixed-commit ETC trace at an exploration horizon plus suffix `r` is bounded
72 by the gap-weighted count budget that allocates `explorationPulls` to every arm
73 and the suffix budget only to the committed arm.
74 
75 This is the `ETC-ACTION-WITH-COMMIT-SUFFIX-COUNT-BUDGET-REGRET` project-local
76 deterministic regret scaffold.  It deliberately keeps the right-hand side in
77 the unsimplified per-arm budget form.
78 -/
79 theorem ETC.pseudoRegret_actionWithCommit_explorationPulls_mul_K_add_le_sum_gap_mul_suffix_count_budget
80     {K : Nat}
81     (spec : ETC.Spec K) (model : FiniteBanditModel K)
82     (commitArm : Fin K) (r : Nat) :
83     pseudoRegret model (ETC.actionWithCommit spec commitArm)
84         (spec.explorationPulls * K + r) <=
85       (Finset.univ : Finset (Fin K)).sum
86         (fun a : Fin K =>
87           model.gap a *
88             (((spec.explorationPulls +
89                 (if commitArm = a then r else 0) : Nat) : Rat))) := by

98         intro a
99         exact le_of_eq
100           (ETC.pullCount_actionWithCommit_explorationPulls_mul_K_add_eq

102             (commitArm := commitArm)
103             (a := a)
104             (r := r)))

141 /--
142 The fixed-commit ETC trace splits pseudo-regret after the exploration horizon
143 into the exploration-horizon pseudo-regret plus one committed-arm gap for each
144 suffix pull.
145 
146 This is the `ETC-ACTION-WITH-COMMIT-PHASE-SPLIT-REGRET` project-local
147 deterministic regret scaffold.
148 -/
149 theorem ETC.pseudoRegret_actionWithCommit_explorationPulls_mul_K_add_eq_add_suffix_gap
150     {K : Nat}
151     (spec : ETC.Spec K) (model : FiniteBanditModel K)
152     (commitArm : Fin K) (r : Nat) :
153     pseudoRegret model (ETC.actionWithCommit spec commitArm)
154         (spec.explorationPulls * K + r) =
155       pseudoRegret model (ETC.actionWithCommit spec commitArm)
156         (spec.explorationPulls * K) +
157         (((r : Nat) : Rat) * model.gap commitArm) := by
158   induction r with
159   | zero =>
160       simp
161   | succ r ih =>
162       have hge :
163           spec.explorationPulls * K <= spec.explorationPulls * K + r := by
164         exact Nat.le_add_right (spec.explorationPulls * K) r
165       have hact :
166           ETC.actionWithCommit spec commitArm
167               (spec.explorationPulls * K + r) = commitArm :=
168         ETC.actionWithCommit_eq_commitArm_of_ge
169           (spec := spec)
170           (commitArm := commitArm)
171           (t := spec.explorationPulls * K + r)
172           hge
173       rw [Nat.add_succ]
174       rw [pseudoRegret_succ, hact, ih]
175       rw [Nat.cast_succ, add_mul, one_mul, add_assoc]
