5 /-!
6 # ETC empirical-mean measurability support
7 
8 This module starts wiring the deterministic fixed-commit ETC empirical-mean
9 surface to stochastic reward traces. It proves numerator measurability and the
10 first full empirical-mean measurability wrapper under an explicit
11 division-by-constant measurability contract, then discharges that contract
12 using the local Rat measurable-singleton wrapper, plus a coordinate-shaped
13 wrapper for downstream event measurability. Argmax wiring, concentration, and
14 filtration remain separate leaves.
15 -/
16 
17 namespace BanditRLProof
18 namespace ETC

20 /--
21 The selected-reward numerator of the fixed-commit ETC empirical mean is
22 measurable for stochastic reward traces with timewise measurable coordinates.
23 
24 This is the `ETC-MEASURABLE-SUMREWARDS-ACTION-WITH-COMMIT-EXPLORATION`
25 project-local leaf.
26 -/
27 theorem measurable_sumRewards_actionWithCommit_exploration
28     {Omega : Type u} {K : Nat}
29     [MeasurableSpace Omega]
30     [MeasurableSpace (Fin K)] [MeasurableSingletonClass (Fin K)]
31     [MeasurableSpace Rat] [MeasurableAdd₂ Rat]
32     (spec : ETC.Spec K) (commitArm a : Fin K)
33     (reward : Omega -> RewardTrace Rat)
34     (hreward : forall t : Nat,
35       Measurable (fun omega : Omega => reward omega t)) :
36     Measurable (fun omega : Omega =>
37       sumRewards (ETC.actionWithCommit spec commitArm) (reward omega) a
38         (spec.explorationPulls * K)) := by
39   have haction : forall t : Nat,
40       Measurable (fun omega : Omega =>
41         (ETC.actionWithCommit spec commitArm) t) := by
42     intro _t
43     exact measurable_const
44   simpa using
45     (measurable_sumRewards
46       (action := fun _ : Omega => ETC.actionWithCommit spec commitArm)
47       (reward := reward)
48       haction hreward a (spec.explorationPulls * K))

50 /--
51 The fixed-commit ETC empirical mean is measurable under stochastic reward
52 traces once measurability of division by a constant Rat is supplied.
53 
54 This is the
55 `ETC-MEASURABLE-EMPMEAN-ACTION-WITH-COMMIT-EXPLORATION-OF-DIV-CONST`
56 project-local leaf. It deliberately leaves the Mathlib import/wrapper decision
57 for Rat division measurability to a later leaf.
58 -/
59 theorem measurable_empMeanAtExploration_of_measurable_div_const
60     {Omega : Type u} {K : Nat}
61     [MeasurableSpace Omega]
62     [MeasurableSpace (Fin K)] [MeasurableSingletonClass (Fin K)]
63     [MeasurableSpace Rat] [MeasurableAdd₂ Rat]
64     (spec : ETC.Spec K) (commitArm a : Fin K)
65     (reward : Omega -> RewardTrace Rat)
66     (hreward : forall t : Nat,
67       Measurable (fun omega : Omega => reward omega t))
68     (hdiv_const : forall c : Rat, Measurable (fun x : Rat => x / c)) :
69     Measurable (fun omega : Omega =>
70       ETC.empMeanAtExploration spec commitArm (reward omega) a) := by
71   have hnum : Measurable (fun omega : Omega =>
72       sumRewards (ETC.actionWithCommit spec commitArm) (reward omega) a
73         (spec.explorationPulls * K)) :=
74     ETC.measurable_sumRewards_actionWithCommit_exploration
75       spec commitArm a reward hreward
76   have hdiv : Measurable (fun omega : Omega =>
77       sumRewards (ETC.actionWithCommit spec commitArm) (reward omega) a
78         (spec.explorationPulls * K) /
79         (pullCount (ETC.actionWithCommit spec commitArm) a
80           (spec.explorationPulls * K) : Rat)) := by
81     exact (hdiv_const _).comp hnum
82   simpa [ETC.empMeanAtExploration] using hdiv

84 /--
85 The fixed-commit ETC empirical mean is measurable under stochastic reward
86 traces, using the local Rat division-by-constant measurability wrapper.
87 
88 This is the `ETC-MEASURABLE-EMPMEAN-ACTION-WITH-COMMIT-EXPLORATION`
89 project-local leaf. It removes the explicit `hdiv_const` argument from
90 `measurable_empMeanAtExploration_of_measurable_div_const` by requiring
91 measurable singletons on `Rat`.
92 -/
93 theorem measurable_empMeanAtExploration
94     {Omega : Type u} {K : Nat}
95     [MeasurableSpace Omega]
96     [MeasurableSpace (Fin K)] [MeasurableSingletonClass (Fin K)]
97     [MeasurableSpace Rat] [MeasurableSingletonClass Rat] [MeasurableAdd₂ Rat]
98     (spec : ETC.Spec K) (commitArm a : Fin K)
99     (reward : Omega -> RewardTrace Rat)
100     (hreward : forall t : Nat,
101       Measurable (fun omega : Omega => reward omega t)) :
102     Measurable (fun omega : Omega =>
103       ETC.empMeanAtExploration spec commitArm (reward omega) a) := by
104   exact
105     ETC.measurable_empMeanAtExploration_of_measurable_div_const
106       spec commitArm a reward hreward measurable_rat_div_const

108 /--
109 The fixed-commit ETC empirical-mean coordinates are measurable under
110 stochastic reward traces.
111 
112 This is the `ETC-MEASURABLE-EMPMEAN-AT-EXPLORATION-COORDINATES`
113 project-local leaf. It packages `measurable_empMeanAtExploration` in the
114 `forall a : Fin K, Measurable ...` shape used by empirical-mean event
115 measurability lemmas; it does not add argmax, concentration, or filtration
116 contracts.
117 -/
118 theorem measurable_empMeanAtExploration_coordinates
119     {Omega : Type u} {K : Nat}
120     [MeasurableSpace Omega]
121     [MeasurableSpace (Fin K)] [MeasurableSingletonClass (Fin K)]
122     [MeasurableSpace Rat] [MeasurableSingletonClass Rat] [MeasurableAdd₂ Rat]
123     (spec : ETC.Spec K) (commitArm : Fin K)
124     (reward : Omega -> RewardTrace Rat)
125     (hreward : forall t : Nat,
126       Measurable (fun omega : Omega => reward omega t)) :
127     forall a : Fin K,
128       Measurable (fun omega : Omega =>
129         (fun b : Fin K =>
130           ETC.empMeanAtExploration spec commitArm (reward omega) b) a) := by
131   intro a
132   simpa using
133     (ETC.measurable_empMeanAtExploration
134       spec commitArm a reward hreward)
