4 # Measure-preserving external sources for stationary UCB
5 
6 This module constructs `RealStationaryUCBSequence` by pulling the canonical
7 arm-stream process through a measure-preserving source map. A product-space
8 specialization permits arbitrary independent nuisance randomness and closes
9 the armwise-bounded expected-average consistency route without caller-supplied
10 split conditional laws.
11 -/

21 Mathlib's `condDistrib_map` has the required transport content, but rewriting
22 its pushforward measure can expose proof-valued finite-measure instances. This
23 equivalent uniqueness proof transports the conditioning and joint maps first.
24 -/

26 /-- Conditional distributions pull back through a measure-preserving map. -/
27 theorem condDistrib_comp_measurePreserving
28     {Alpha : Type u} {Beta : Type v} {Output : Type w}
29     {Source : Type x}
30     [MeasurableSpace Alpha] [MeasurableSpace Beta]
31     [MeasurableSpace Output] [StandardBorelSpace Output] [Nonempty Output]
32     [MeasurableSpace Source]
33     (mu : Measure Source) [IsFiniteMeasure mu]
34     (nu : Measure Alpha) [IsFiniteMeasure nu]
35     (source : Source -> Alpha) (hsource : MeasurePreserving source mu nu)
36     (X : Alpha -> Beta) (Y : Alpha -> Output)
37     (hX : Measurable X) (hY : Measurable Y) :
38     condDistrib (Y ∘ source) (X ∘ source) mu =ᵐ[mu.map (X ∘ source)]
39       condDistrib Y X nu := by
40   apply ProbabilityTheory.condDistrib_ae_eq_of_measure_eq_compProd_of_measurable
41     (hX.comp hsource.measurable) (hY.comp hsource.measurable)
42   calc
43     mu.map (fun omega => ((X ∘ source) omega, (Y ∘ source) omega)) =
44         nu.map (fun alpha => (X alpha, Y alpha)) := by
45       change mu.map ((fun alpha => (X alpha, Y alpha)) ∘ source) = _
46       rw [← Measure.map_map (hX.prod hY) hsource.measurable,
47         hsource.map_eq]
48     _ = nu.map X ⊗ₘ condDistrib Y X nu := by
49       exact (ProbabilityTheory.compProd_map_condDistrib hY.aemeasurable).symm
50     _ = mu.map (X ∘ source) ⊗ₘ condDistrib Y X nu := by
51       rw [← Measure.map_map hX hsource.measurable, hsource.map_eq]

54 noncomputable def measurePreservingArmStreamAction
55     {Omega : Type u} {K : Nat} [NeZero K]
56     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
57     (source : Omega -> ArmRewardStream K) :

69 /--
70 A measure-preserving source of canonical arm streams produces every field of
71 the local stationary UCB compatibility bundle on the external sample space.
72 -/
73 theorem realStationaryUCBSequence_comp_measurePreserving_armStream
74     {Omega : Type u} {K : Nat} [NeZero K] [MeasurableSpace Omega]
75     (mu : Measure Omega) [IsFiniteMeasure mu]
76     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
77     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
78     (source : Omega -> ArmRewardStream K)
79     (hsource : MeasurePreserving source mu (armStreamMeasure nu)) :
80     RealStationaryUCBSequence mu hK c sigma2 nu
81       (measurePreservingArmStreamAction hK c sigma2 source)
82       (measurePreservingArmStreamReward hK c sigma2 source) := by
83   refine
84     { measurable_action := fun t =>
85         (measurable_armStreamAction hK (c * (sigma2 : Real)) t).comp
86           hsource.measurable
87       measurable_reward := fun t =>
88         (measurable_armStreamReward hK (c * (sigma2 : Real)) t).comp
89           hsource.measurable
90       hasLaw_action_zero := ?_
91       hasCondDistrib_feedback_zero := ?_
92       hasCondDistrib_action := ?_
93       hasCondDistrib_feedback := ?_ }
94   · change Measure.map
95       ((fun stream : ArmRewardStream K =>
96         armStreamAction hK (c * (sigma2 : Real)) stream 0) ∘ source) mu = _
97     rw [← Measure.map_map
98       (measurable_armStreamAction hK (c * (sigma2 : Real)) 0)
99       hsource.measurable]
100     rw [hsource.map_eq]
101   · simpa [measurePreservingArmStreamAction,
102       measurePreservingArmStreamReward, Function.comp_def] using
103       (condDistrib_comp_measurePreserving mu (armStreamMeasure nu)
104         source hsource
105         (fun stream : ArmRewardStream K =>
106           armStreamAction hK (c * (sigma2 : Real)) stream 0)
107         (fun stream : ArmRewardStream K =>
108           armStreamReward hK (c * (sigma2 : Real)) stream 0)
109         (measurable_armStreamAction hK (c * (sigma2 : Real)) 0)
110         (measurable_armStreamReward hK (c * (sigma2 : Real)) 0))
111   · intro i
112     simpa [measurePreservingArmStreamAction,
113       measurePreservingArmStreamReward, Function.comp_def] using
114       (condDistrib_comp_measurePreserving mu (armStreamMeasure nu)
115         source hsource
116         (fun stream : ArmRewardStream K =>
117           History.finitePairHistoryOfTrace
118             (armStreamAction hK (c * (sigma2 : Real)) stream)
119             (armStreamReward hK (c * (sigma2 : Real)) stream) i)
120         (fun stream : ArmRewardStream K =>
121           armStreamAction hK (c * (sigma2 : Real)) stream (i + 1))
122         (History.measurable_finitePairHistoryOfTrace
123           (armStreamAction hK (c * (sigma2 : Real)))
124           (armStreamReward hK (c * (sigma2 : Real)))
125           (measurable_armStreamAction hK (c * (sigma2 : Real)))
126           (measurable_armStreamReward hK (c * (sigma2 : Real))) i)
127         (measurable_armStreamAction hK (c * (sigma2 : Real)) (i + 1)))
128   · intro i
129     simpa [measurePreservingArmStreamAction,
130       measurePreservingArmStreamReward, Function.comp_def] using
131       (condDistrib_comp_measurePreserving mu (armStreamMeasure nu)
132         source hsource
133         (fun stream : ArmRewardStream K =>
134           (History.finitePairHistoryOfTrace
135               (armStreamAction hK (c * (sigma2 : Real)) stream)
136               (armStreamReward hK (c * (sigma2 : Real)) stream) i,
137             armStreamAction hK (c * (sigma2 : Real)) stream (i + 1)))
138         (fun stream : ArmRewardStream K =>
139           armStreamReward hK (c * (sigma2 : Real)) stream (i + 1))
140         ((History.measurable_finitePairHistoryOfTrace
141             (armStreamAction hK (c * (sigma2 : Real)))
142             (armStreamReward hK (c * (sigma2 : Real)))
143             (measurable_armStreamAction hK (c * (sigma2 : Real)))
144             (measurable_armStreamReward hK (c * (sigma2 : Real))) i).prod
145           (measurable_armStreamAction hK (c * (sigma2 : Real)) (i + 1)))
146         (measurable_armStreamReward hK (c * (sigma2 : Real)) (i + 1)))

162 /-- Product-first projection is a concrete external stationary UCB source. -/
163 theorem realStationaryUCBSequence_productNoise_armStream
164     {K : Nat} [NeZero K] {Aux : Type u} [MeasurableSpace Aux]
165     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
166     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu]
167     (auxMu : Measure Aux) [IsProbabilityMeasure auxMu] :
168     RealStationaryUCBSequence ((armStreamMeasure nu).prod auxMu)
169       hK c sigma2 nu
170       (productNoiseArmStreamAction hK c sigma2)
171       (productNoiseArmStreamReward hK c sigma2) := by
172   exact realStationaryUCBSequence_comp_measurePreserving_armStream
173     ((armStreamMeasure nu).prod auxMu) hK c sigma2 nu
174     Prod.fst measurePreserving_fst

176 /-- Product law for finite Real arm streams and independent auxiliary noise. -/
177 noncomputable def finiteArmProductNoiseMeasure
178     {K : Nat} {Aux : Type u} [MeasurableSpace Aux]
179     (armLaw : Fin K -> Measure Real)
180     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
181     (auxMu : Measure Aux) : Measure (ArmRewardStream K × Aux) :=
182   let nu := finiteArmRealRewardKernel armLaw
183   letI : IsMarkovKernel nu :=
184     finiteArmRealRewardKernel_isMarkov armLaw hprob
185   (armStreamMeasure nu).prod auxMu

191 theorem productNoiseArmwiseBoundedFiniteArmExpectedRegret_nonneg_and_le

196     (lo hi : Fin K -> Real)

198     (hbound : forall arm,

200         (fun x : Real => Set.Icc (lo arm) (hi arm) x) (ae (armLaw arm)))
201     (n : Nat) :
202     let sigma2 := Concentration.finiteArmPositiveVarianceProxy
203       (fun arm => Concentration.intervalVarianceProxy (lo arm) (hi arm))
204     0 <= realStationaryArmwiseBoundedFiniteArmExpectedRegret
205         (finiteArmProductNoiseMeasure armLaw hprob auxMu) armLaw
206         (productNoiseArmStreamAction hK 4 sigma2) n /\

208           (finiteArmProductNoiseMeasure armLaw hprob auxMu) armLaw
209           (productNoiseArmStreamAction hK 4 sigma2) n <=
210         realStationaryArmwiseBoundedFiniteArmModelCoefficient
211             armLaw hprob lo hi *
212           (1 + Real.log ((n + 1 : Nat) : Real)) := by

214   let nu := finiteArmRealRewardKernel armLaw
215   let sigma2 := Concentration.finiteArmPositiveVarianceProxy
216     (fun arm => Concentration.intervalVarianceProxy (lo arm) (hi arm))
217   letI : IsMarkovKernel nu :=
218     finiteArmRealRewardKernel_isMarkov armLaw hprob
219   change
220     0 <= realStationaryArmwiseBoundedFiniteArmExpectedRegret
221         ((armStreamMeasure nu).prod auxMu) armLaw
222         (productNoiseArmStreamAction hK 4 sigma2) n /\
223       realStationaryArmwiseBoundedFiniteArmExpectedRegret
224           ((armStreamMeasure nu).prod auxMu) armLaw

226         realStationaryArmwiseBoundedFiniteArmModelCoefficient
227             armLaw hprob lo hi *
