4 /-!
5 # ETC pairwise sub-Gaussian tail producer surface
6 
7 This module connects the reusable independent sub-Gaussian finite-sum tail
8 wrapper to the fixed-commit ETC pairwise empirical-mean tail contract.  It
9 does not instantiate a reward law, prove independence for ETC rewards, add
10 filtration, or prove final ETC regret.
11 -/

13 namespace BanditRLProof
14 namespace ETC

16 /--
17 Build the fixed-commit ETC pairwise empirical-mean tail contract from abstract
18 independent sub-Gaussian sum witnesses for every non-best arm.
19 
20 This is the `ETC-PAIRWISE-TAIL-PRODUCER-SUBGAUSS` bridge.  The event inclusion
21 field is intentionally explicit: later leaves must prove that the ETC
22 empirical-mean comparison event is contained in the corresponding centered
23 reward-difference sum event.
24 -/
25 theorem pairwiseEmpMeanTailContract_of_subGaussian_event_bounds
26     {Omega : Type u} {K : Nat}
27     [MeasurableSpace Omega]
28     (mu : MeasureTheory.Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
29     (spec : ETC.Spec K)
30     (model : FiniteBanditModel K)
31     (commitArm : Fin K)
32     (reward : Omega -> RewardTrace Rat)
33     (tail : Fin K -> ENNReal)
34     {Idx : Type v}
35     (idx : Finset Idx)
36     (X : Fin K -> Idx -> Omega -> Real)
37     (c : Fin K -> Idx -> NNReal)
38     (eps : Fin K -> Real)
39     (h_indep :
40       forall a : Fin K, (a = model.bestArm -> False) ->
41         ProbabilityTheory.iIndepFun (X a) mu)
42     (h_subG :
43       forall a : Fin K, (a = model.bestArm -> False) ->
44         forall i, i ∈ idx ->
45           ProbabilityTheory.HasSubgaussianMGF ((X a) i) ((c a) i) mu)
46     (heps :
47       forall a : Fin K, (a = model.bestArm -> False) ->
48         0 <= eps a)
49     (hsubset :
50       forall a : Fin K, (a = model.bestArm -> False) ->
51         Set.Subset
52           {omega : Omega |
53             ETC.empMeanAtExploration spec commitArm (reward omega) a >=
54               ETC.empMeanAtExploration spec commitArm
55                 (reward omega) model.bestArm}
56           {omega : Omega | eps a <= idx.sum (fun i => X a i omega)})
57     (htail :
58       forall a : Fin K, (a = model.bestArm -> False) ->
59         ENNReal.ofReal
60           (Real.exp
61             (-(eps a) ^ 2 /
62               (2 * ((idx.sum (c a) : NNReal) : Real)))) <= tail a) :
63     ETC.PairwiseEmpMeanTailContract
64       mu spec model commitArm reward tail := by
65   refine ⟨?_⟩
66   intro a hne
67   exact
68     (mu.mono (hsubset a hne)).trans
69       ((Concentration.subGaussian_sum_tail_ennreal_of_iIndepFun
70         (mu := mu)
71         (X := X a)
72         (c := c a)
73         (s := idx)
74         (h_indep a hne)
75         (h_subG a hne)
76         (heps a hne)).trans
77         (htail a hne))
78 
79 end ETC
80 end BanditRLProof
