4 /-!
5 # Canonical one-step Thompson sampler
6 
7 This module constructs the one-step joint law obtained by sampling an
8 environment/history pair from a Bayesian prior-likelihood model and then
9 sampling an action from the canonical posterior pushed through a measurable
10 best-action selector.  The resulting probability-matching theorem has no
11 separate pair-law or action-law premise.
12 -/

22 /-- The Thompson action kernel induced by Mathlib's canonical posterior. -/
23 noncomputable def canonicalActionKernel
24     {History : Type u} {Env : Type v} {Action : Type w}
25     [MeasurableSpace History]
26     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
27     [MeasurableSpace Action]
28     (prior : Measure Env) [IsProbabilityMeasure prior]
29     (likelihood : ProbabilityTheory.Kernel Env History)
30     [ProbabilityTheory.IsMarkovKernel likelihood]
31     (bestAction : Env -> Action) (_hbestAction : Measurable bestAction) :
32     ProbabilityTheory.Kernel History Action :=
33   (PosteriorKernel.canonicalPosterior prior likelihood).kernel.map bestAction

77 /--
78 The canonical one-step Thompson law on `(Env × History) × Action`.
79 
80 The first component is sampled from `prior ⊗ₘ likelihood`; the action is then
81 sampled from the canonical posterior mapped by `bestAction` at that history.
82 -/
83 noncomputable def canonicalSamplerMeasure
84     {History : Type u} {Env : Type v} {Action : Type w}
85     [MeasurableSpace History]
86     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
87     [MeasurableSpace Action]
88     (prior : Measure Env) [IsProbabilityMeasure prior]
89     (likelihood : ProbabilityTheory.Kernel Env History)
90     [ProbabilityTheory.IsMarkovKernel likelihood]
91     (bestAction : Env -> Action) (hbestAction : Measurable bestAction) :
92     Measure ((Env × History) × Action) :=
93   PosteriorKernel.canonicalJointMeasure prior likelihood ⊗ₘ
94     canonicalActionKernelOnPair prior likelihood bestAction hbestAction

143 /--
144 Projecting a composition product whose kernel depends only on the second base
145 coordinate gives the second-coordinate marginal composed with that kernel.
146 -/
147 theorem map_compProd_comap_snd
148     {Env History Action : Type*}
149     [MeasurableSpace Env] [MeasurableSpace History] [MeasurableSpace Action]
150     (mu : Measure (Env × History)) [IsFiniteMeasure mu]
151     (actionKernel : ProbabilityTheory.Kernel History Action)
152     [ProbabilityTheory.IsMarkovKernel actionKernel] :
153     (mu ⊗ₘ actionKernel.comap Prod.snd measurable_snd).map
154         (fun sample => (sample.1.2, sample.2)) =
155       mu.map Prod.snd ⊗ₘ actionKernel := by
156   apply Measure.ext_prod
157   intro historyEvent actionEvent hhistoryEvent hactionEvent
158   have hmap : Measurable
159       (fun sample : (Env × History) × Action => (sample.1.2, sample.2)) :=
160     (measurable_snd.comp measurable_fst).prodMk measurable_snd
161   rw [Measure.map_apply hmap (hhistoryEvent.prod hactionEvent)]
162   have hpreimage :
163       (fun sample : (Env × History) × Action => (sample.1.2, sample.2)) ⁻¹'
164           (historyEvent ×ˢ actionEvent) =
165         (Set.univ ×ˢ historyEvent) ×ˢ actionEvent := by
166     ext sample
167     simp
168   rw [hpreimage,
169     Measure.compProd_apply_prod (MeasurableSet.univ.prod hhistoryEvent) hactionEvent,
170     Measure.compProd_apply_prod hhistoryEvent hactionEvent]
171   simp_rw [ProbabilityTheory.Kernel.comap_apply]
172   rw [MeasureTheory.setLIntegral_map hhistoryEvent
173     (actionKernel.measurable_coe hactionEvent) measurable_snd]
174   have hsndPreimage :
175       Prod.snd ⁻¹' historyEvent = (Set.univ ×ˢ historyEvent : Set (Env × History)) := by
176     ext sample
177     simp
178   rw [hsndPreimage]

180 /-- The canonical sampler preserves the prescribed environment/history law. -/
181 theorem canonicalSampler_env_history_map_eq
182     {History : Type u} {Env : Type v} {Action : Type w}
183     [MeasurableSpace History]
184     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
185     [MeasurableSpace Action]
186     (prior : Measure Env) [IsProbabilityMeasure prior]
187     (likelihood : ProbabilityTheory.Kernel Env History)
188     [ProbabilityTheory.IsMarkovKernel likelihood]
189     (bestAction : Env -> Action) (hbestAction : Measurable bestAction) :
190     (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
191         (fun sample =>
192           (canonicalSamplerEnv sample, canonicalSamplerHistory sample)) =
193       PosteriorKernel.canonicalJointMeasure prior likelihood := by
194   change
195     (canonicalSamplerMeasure prior likelihood bestAction hbestAction).fst =
196       PosteriorKernel.canonicalJointMeasure prior likelihood
197   rw [canonicalSamplerMeasure, Measure.fst_compProd]

199 /-- The history/action marginal is generated by the canonical action kernel. -/
200 theorem canonicalSampler_history_action_map_eq
201     {History : Type u} {Env : Type v} {Action : Type w}
202     [MeasurableSpace History]
203     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
204     [MeasurableSpace Action]
205     (prior : Measure Env) [IsProbabilityMeasure prior]
206     (likelihood : ProbabilityTheory.Kernel Env History)
207     [ProbabilityTheory.IsMarkovKernel likelihood]
208     (bestAction : Env -> Action) (hbestAction : Measurable bestAction) :
209     (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
210         (fun sample =>
211           (canonicalSamplerHistory sample, canonicalSamplerAction sample)) =
212       (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
213           canonicalSamplerHistory ⊗ₘ
214         canonicalActionKernel prior likelihood bestAction hbestAction := by
215   have hhistoryLaw :
216       (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
217           canonicalSamplerHistory =
218         (PosteriorKernel.canonicalJointMeasure prior likelihood).map Prod.snd := by
219     calc
220       (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
221           canonicalSamplerHistory =
222           ((canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
223             Prod.fst).map Prod.snd := by
224               rw [Measure.map_map measurable_snd measurable_fst]
225               rfl
226       _ = (PosteriorKernel.canonicalJointMeasure prior likelihood).map Prod.snd := by
227         rw [show
228           (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
229               Prod.fst =
230             PosteriorKernel.canonicalJointMeasure prior likelihood by
231           simpa only [canonicalSamplerEnv, canonicalSamplerHistory] using
232             canonicalSampler_env_history_map_eq
233               prior likelihood bestAction hbestAction]
234   have htransport := map_compProd_comap_snd
235     (PosteriorKernel.canonicalJointMeasure prior likelihood)
236     (canonicalActionKernel prior likelihood bestAction hbestAction)
237   rw [hhistoryLaw]
238   simpa only [canonicalSamplerMeasure, canonicalActionKernelOnPair,
239     canonicalSamplerHistory, canonicalSamplerAction] using htransport

245     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
246     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]

252         (canonicalSamplerMeasure prior likelihood bestAction hbestAction) =ᵐ[

263 /--
264 Premise-free one-step Thompson probability matching for the canonical sampler.
265 
266 Both law premises of the generic theorem are discharged by the constructed
267 composition-product measure: its environment/history marginal is the canonical
268 Bayesian joint law, and its history/action marginal is generated by the mapped
269 canonical posterior.
270 -/
271 theorem canonicalSampler_condDistrib_action_ae_eq_bestAction
272     {History : Type u} {Env : Type v} {Action : Type w}
273     [MeasurableSpace History]
274     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
275     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
276     (prior : Measure Env) [IsProbabilityMeasure prior]
277     (likelihood : ProbabilityTheory.Kernel Env History)
278     [ProbabilityTheory.IsMarkovKernel likelihood]
279     (bestAction : Env -> Action) (hbestAction : Measurable bestAction) :
280     ProbabilityTheory.condDistrib canonicalSamplerAction canonicalSamplerHistory
281         (canonicalSamplerMeasure prior likelihood bestAction hbestAction) =ᵐ[
282       (canonicalSamplerMeasure prior likelihood bestAction hbestAction).map
283         canonicalSamplerHistory]
284       ProbabilityTheory.condDistrib (bestAction ∘ canonicalSamplerEnv)
285         canonicalSamplerHistory
286         (canonicalSamplerMeasure prior likelihood bestAction hbestAction) := by
287   exact condDistrib_action_ae_eq_bestAction_of_bayesianPairMap
288     (canonicalSamplerMeasure prior likelihood bestAction hbestAction)
289     canonicalSamplerEnv canonicalSamplerHistory canonicalSamplerAction
290     prior likelihood bestAction hbestAction
291     canonicalSamplerEnv_measurable canonicalSamplerHistory_measurable
292     canonicalSamplerAction_measurable
293     (canonicalSampler_env_history_map_eq prior likelihood bestAction hbestAction)
294     (canonicalSampler_condDistrib_action_ae_eq_actionKernel
