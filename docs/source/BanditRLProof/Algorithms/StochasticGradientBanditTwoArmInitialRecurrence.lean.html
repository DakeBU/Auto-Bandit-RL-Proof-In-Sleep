4 # Two-arm stochastic-gradient-bandit initial recurrences
5 
6 This module closes the source-round `t = 1` base case for the two exponential
7 recurrences used in Theorem 1 of
8 Baudry--Johnson--Vary--Pike-Burke--Rebeschini (NeurIPS 2025).  The generated
9 initial pair kernel samples from the untouched source parameter
10 `theta_1 = 0`, hence its action law is uniform on `Fin 2`; consuming that pair
11 produces `theta_2`.
12 
13 Probability / Equation-(8) ledger:
14 
15 * the random object is the initial `(selected action, reward)` pair drawn from
16   `Thompson.measurableEnvironmentInitialPairKernel`;
17 * the action law is the exact two-arm softmax law at the zero vector, so each
18   arm has probability `1 / 2`;
19 * `hreward` supplies the source support `|reward| <= 1` almost everywhere on
20   each initial reward fiber, while `hmean` fixes its mean;
21 * no independence or filtration premise is needed at this base step;
22 * measurability and exponential integrability are discharged by
23   `integral_measurableEnvironmentInitialPairKernel_exp_actionReward_le_sourceEqEight_of_mean`;
24 * `twoArmForwardEqEightRemainder_le` and
25   `twoArmInverseEqEightRemainder_le` specialize the two action-dependent
26   Equation-(8) coefficients at `p_1 = 1 / 2`.
27 
28 The inverse bound is algebraically valid without assuming
29 `Delta > eta * sourceC eta`; that strict gap condition is needed only by later
30 summation/division consumers.  These declarations establish the two base
31 moments, not the tower-property iteration, expected failure-mass summation,
32 or the finite-horizon regret theorem.
33 -/

45 /-- Zero initialization gives the exact source probability `p_1 = 1 / 2` on
46 both arms. -/
47 @[simp]
48 theorem softmaxProbability_zeroInitialization_finTwo (selected : Fin 2) :
49     softmaxProbability (fun _ : Fin 2 => 0) selected = (1 : Real) / 2 := by
50   fin_cases selected <;>
51     norm_num [softmaxProbability, softmaxDenominator, Fin.sum_univ_two]
52 
53 /-- Source-round `t = 1` forward exponential base recurrence.  The left side
54 is `E[exp(2 * theta_{1,2})]`, because `theta_{1,1} = 0` and the initial pair
55 contributes exactly `eta * sourceIncrement`. -/
56 theorem integral_twoArmInitialPairKernel_exp_forwardIncrement_le
57     {Env : Type v} [MeasurableSpace Env]
58     (eta Delta : Real) (heta : 0 <= eta)
59     (environment :
60       Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
61     (env : Env) (mean : Fin 2 -> Real)
62     (hreward : forall selected,
63       ∀ᵐ reward ∂environment.initialFeedback (env, selected),
64         |reward| <= 1)
65     (hmean : forall selected,
66       integral (environment.initialFeedback (env, selected)) id =
67         mean selected)
68     (hgap : mean 0 - mean 1 = Delta) :
69     integral
70         (Thompson.measurableEnvironmentInitialPairKernel
71           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
72         (fun pair : Fin 2 × Real =>
73           Real.exp
74             (2 * eta *
75               sourceIncrement (fun _ : Fin 2 => (1 : Real) / 2)
76                 pair.2 pair.1 0)) <=
77       1 + (eta * Delta + eta ^ 2 * sourceC eta) / 2 := by
78   let prob : Fin 2 -> Real :=
79     softmaxProbability (fun _ : Fin 2 => 0)
80   let q : Fin 2 -> Real := twoArmForwardQ eta prob
81   have hp_nonneg : 0 <= prob 0 := softmaxProbability_nonneg _ _
82   have hp_le_one : prob 0 <= 1 := softmaxProbability_le_one _ _
83   have hEqEight :
84       integral
85           (Thompson.measurableEnvironmentInitialPairKernel
86             (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
87           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
88         1 + ∑ selected, prob selected *
89           (q selected * mean selected +
90             q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := by
91     simpa [prob] using
92       integral_measurableEnvironmentInitialPairKernel_exp_actionReward_le_sourceEqEight_of_mean
93         (fun _ : Fin 2 => 0) eta environment env q mean hreward hmean
94   have hprobOne : prob 1 = 1 - prob 0 := by
95     simpa [prob] using
96       softmaxProbability_one_eq_one_sub_zero
97         (fun _ : Fin 2 => (0 : Real))
98   have hrem :
99       (∑ selected, prob selected *
100         (q selected * mean selected +
101           q selected ^ 2 / 2 * sourceC (|q selected| / 2))) <=
102         2 * prob 0 * (1 - prob 0) *
103           (eta * Delta + eta ^ 2 * sourceC eta) := by
104     simp only [Fin.sum_univ_two]
105     rw [hprobOne]
106     simpa [q, twoArmForwardQ] using
107       twoArmForwardEqEightRemainder_le eta (prob 0)
108         (mean 0) (mean 1) Delta heta hp_nonneg hp_le_one hgap
109   have hbase :
110       integral
111           (Thompson.measurableEnvironmentInitialPairKernel
112             (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
113           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
114         1 + (eta * Delta + eta ^ 2 * sourceC eta) / 2 := by
115     calc
116       _ <= 1 + ∑ selected, prob selected *
117           (q selected * mean selected +
118             q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := hEqEight
119       _ <= 1 + 2 * prob 0 * (1 - prob 0) *
120           (eta * Delta + eta ^ 2 * sourceC eta) :=
121         add_le_add_right hrem 1
122       _ = 1 + (eta * Delta + eta ^ 2 * sourceC eta) / 2 := by
123         rw [show prob 0 = (1 : Real) / 2 by
124           exact softmaxProbability_zeroInitialization_finTwo 0]
125         ring
126   have hintegrand :
127       (fun pair : Fin 2 × Real =>
128           Real.exp
129             (2 * eta *
130               sourceIncrement (fun _ : Fin 2 => (1 : Real) / 2)
131                 pair.2 pair.1 0)) =
132         (fun pair : Fin 2 × Real =>
133           Real.exp (q pair.1 * pair.2)) := by
134     funext pair
135     apply congrArg Real.exp
136     rw [twoArmForwardQ_mul_reward_eq_sourceIncrement]
137     congr 2
138     funext selected
139     exact (softmaxProbability_zeroInitialization_finTwo selected).symm
140   rw [hintegrand]
141   exact hbase

143 /-- Source-round `t = 1` inverse exponential base recurrence.  This is the
144 initial value of the inverse-odds potential used to telescope the expected
145 squared failure mass. -/
146 theorem integral_twoArmInitialPairKernel_exp_inverseIncrement_le
147     {Env : Type v} [MeasurableSpace Env]
148     (eta Delta : Real) (heta : 0 <= eta)
149     (environment :
150       Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
151     (env : Env) (mean : Fin 2 -> Real)
152     (hreward : forall selected,
153       ∀ᵐ reward ∂environment.initialFeedback (env, selected),
154         |reward| <= 1)
155     (hmean : forall selected,
156       integral (environment.initialFeedback (env, selected)) id =
157         mean selected)
158     (hgap : mean 0 - mean 1 = Delta) :
159     integral
160         (Thompson.measurableEnvironmentInitialPairKernel
161           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
162         (fun pair : Fin 2 × Real =>
163           Real.exp
164             (-2 * eta *
165               sourceIncrement (fun _ : Fin 2 => (1 : Real) / 2)
166                 pair.2 pair.1 0)) <=
167       1 - eta / 2 * (Delta - eta * sourceC eta) := by
168   let prob : Fin 2 -> Real :=
169     softmaxProbability (fun _ : Fin 2 => 0)
170   let q : Fin 2 -> Real := twoArmInverseQ eta prob
171   have hp_nonneg : 0 <= prob 0 := softmaxProbability_nonneg _ _
172   have hp_le_one : prob 0 <= 1 := softmaxProbability_le_one _ _
173   have hEqEight :
174       integral
175           (Thompson.measurableEnvironmentInitialPairKernel
176             (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
177           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
178         1 + ∑ selected, prob selected *
179           (q selected * mean selected +
180             q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := by
181     simpa [prob] using
182       integral_measurableEnvironmentInitialPairKernel_exp_actionReward_le_sourceEqEight_of_mean
183         (fun _ : Fin 2 => 0) eta environment env q mean hreward hmean
184   have hprobOne : prob 1 = 1 - prob 0 := by
185     simpa [prob] using
186       softmaxProbability_one_eq_one_sub_zero
187         (fun _ : Fin 2 => (0 : Real))
188   have hrem :
189       (∑ selected, prob selected *
190         (q selected * mean selected +
191           q selected ^ 2 / 2 * sourceC (|q selected| / 2))) <=
192         -2 * eta * prob 0 * (1 - prob 0) *
193           (Delta - eta * sourceC eta) := by
194     simp only [Fin.sum_univ_two]
195     rw [hprobOne]
196     simpa [q, twoArmInverseQ, twoArmForwardQ] using
197       twoArmInverseEqEightRemainder_le eta (prob 0)
198         (mean 0) (mean 1) Delta heta hp_nonneg hp_le_one hgap
199   have hbase :
200       integral
201           (Thompson.measurableEnvironmentInitialPairKernel
202             (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment env)
203           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
204         1 - eta / 2 * (Delta - eta * sourceC eta) := by
205     calc
206       _ <= 1 + ∑ selected, prob selected *
207           (q selected * mean selected +
208             q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := hEqEight
209       _ <= 1 + -2 * eta * prob 0 * (1 - prob 0) *
210           (Delta - eta * sourceC eta) :=
211         add_le_add_right hrem 1
212       _ = 1 - eta / 2 * (Delta - eta * sourceC eta) := by
213         rw [show prob 0 = (1 : Real) / 2 by
214           exact softmaxProbability_zeroInitialization_finTwo 0]
215         ring
216   have hintegrand :
217       (fun pair : Fin 2 × Real =>
218           Real.exp
219             (-2 * eta *
220               sourceIncrement (fun _ : Fin 2 => (1 : Real) / 2)
221                 pair.2 pair.1 0)) =
222         (fun pair : Fin 2 × Real =>
223           Real.exp (q pair.1 * pair.2)) := by
224     funext pair
225     apply congrArg Real.exp
226     rw [twoArmInverseQ_mul_reward_eq_sourceIncrement]
227     congr 2
228     funext selected
229     exact (softmaxProbability_zeroInitialization_finTwo selected).symm
230   rw [hintegrand]
231   exact hbase
