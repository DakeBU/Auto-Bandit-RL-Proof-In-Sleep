11       (canonicalAction hk n mean table 0, canonicalReward hk n mean table 0)) =
12       (historyAlgorithm hk n).initialAction.compProd ν := by

14       (fun r : ℝ => ((⟨0,hk⟩ : Fin k),r)) ∘ (fun table : UCB.ArmRewardStream k => table 1 ⟨0,hk⟩) := by

62 theorem map_canonicalHistory_eq {k : ℕ} [NeZero k] (hk : 0 < k) (n : ℕ)
63     (mean : Fin k → ℝ) (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν] (t : ℕ) :
64     (UCB.armStreamMeasure ν).map (fun table => canonicalHistory hk n mean table t) =
65       LowerBounds.canonicalBanditHistoryMeasure (historyAlgorithm hk n) ν t := by
66   induction t with
67   | zero =>
68       rw [LowerBounds.canonicalBanditHistoryMeasure_zero]
69       have he : (fun table => canonicalHistory hk n mean table 0) =
70           (LowerBounds.pairHistoryZeroMeasurableEquiv (Fin k) ℝ) ∘
71             (fun table => (canonicalAction hk n mean table 0, canonicalReward hk n mean table 0)) := by
72         funext table i
73         have hi : i.val = 0 := by have := Finset.mem_Iic.mp i.property; omega
74         simp [canonicalHistory, History.finitePairHistoryOfTrace,
75           LowerBounds.pairHistoryZeroMeasurableEquiv, Thompson.singletonPairHistory, hi]
76       rw [he, ← Measure.map_map (LowerBounds.pairHistoryZeroMeasurableEquiv (Fin k) ℝ).measurable
77         ((measurable_canonicalAction hk n mean 0).prodMk (measurable_canonicalReward hk n mean 0)),
78         canonical_initialPair_map]
79   | succ t ih =>
80       rw [LowerBounds.canonicalBanditHistoryMeasure_succ]
81       let J := fun table => (canonicalHistory hk n mean table t,
82         (canonicalAction hk n mean table (t+1), canonicalReward hk n mean table (t+1)))
83       have hJ : Measurable J := (measurable_canonicalHistory hk n mean t).prodMk
84         ((measurable_canonicalAction hk n mean (t+1)).prodMk (measurable_canonicalReward hk n mean (t+1)))
85       have he : (fun table => canonicalHistory hk n mean table (t+1)) =
86           (LowerBounds.pairHistorySuccMeasurableEquiv (Fin k) ℝ t) ∘ J := by
87         funext table
88         rw [Function.comp_apply, LowerBounds.pairHistorySuccMeasurableEquiv_apply]
89         exact History.finitePairHistoryOfTrace_succ _ _ t
90       have hj := Thompson.nextPairJointLaw_eq_compProd (UCB.armStreamMeasure ν)
91         (canonicalAction hk n mean) (canonicalReward hk n mean) (historyAlgorithm hk n)
92         (LowerBounds.stationaryBanditHistoryEnvironment ν) (canonical_historySequence hk n mean ν) t
93       change (UCB.armStreamMeasure ν).map J =
94         ((UCB.armStreamMeasure ν).map (fun table => canonicalHistory hk n mean table t)).compProd
95           (Thompson.historyStepKernel (historyAlgorithm hk n) (LowerBounds.stationaryBanditHistoryEnvironment ν) t) at hj
96       rw [he, ← Measure.map_map (LowerBounds.pairHistorySuccMeasurableEquiv (Fin k) ℝ t).measurable hJ,
97         hj, ih]
