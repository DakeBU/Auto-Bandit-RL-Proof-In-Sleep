5 # Native Real ETC least-encoded argmax and action alignment
6 
7 This module identifies the strict-update fold used by the native Real ETC
8 route with the least-encoded maximizing arm selected by the `Nat.find` scheme
9 used in LML's measurable argmax. It then assembles round-robin exploration, the
10 commit action, and post-commit persistence into the action equality consumed by
11 the exact native Real source-law theorem.
12 -/

19 private theorem argmax_cons_eq_some_foldl_real_select
20     {K : Nat} (scores : Fin K -> Real) (init : Fin K) (l : List (Fin K)) :
21     List.argmax scores (init :: l) =
22       some (l.foldl
23         (fun best arm : Fin K =>
24           if scores best < scores arm then arm else best)
25         init) := by
26   unfold List.argmax
27   simp only [List.foldl_cons, List.argAux]
28   induction l generalizing init with
29   | nil => rfl
30   | cons arm rest ih =>
31       simp only [List.foldl_cons]
32       rw [show List.argAux (fun b c : Fin K => scores c < scores b)
33           (some init) arm =
34           some (if scores init < scores arm then arm else init) by
35         by_cases h : scores init < scores arm
36         case pos => simp [List.argAux, h]
37         case neg => simp [List.argAux, h]]
38       exact ih _

40 /-- The native strict-update fold is Mathlib's first-occurrence list argmax. -/
41 theorem realArgmaxCommit_argmax_finRange {K : Nat} (hK : 0 < K)
42     (scores : Fin K -> Real) :
43     List.argmax scores (List.finRange K) =
44       some (ETC.realArgmaxCommit hK scores) := by
45   cases K with
46   | zero => omega
47   | succ k =>
48       rw [List.finRange_succ]
49       rw [argmax_cons_eq_some_foldl_real_select]
50       simp [ETC.realArgmaxCommit, List.finRange_succ]

52 /-- Among score maximizers, the native fold chooses the least encoded arm. -/
53 theorem realArgmaxCommit_encode_le_of_score_le {K : Nat} (hK : 0 < K)
54     (scores : Fin K -> Real) (a : Fin K)
55     (hscore : scores (ETC.realArgmaxCommit hK scores) <= scores a) :
56     Encodable.encode (ETC.realArgmaxCommit hK scores) <= Encodable.encode a := by
57   have harg : Membership.mem (List.argmax scores (List.finRange K))
58       (ETC.realArgmaxCommit hK scores) := by
59     rw [ETC.realArgmaxCommit_argmax_finRange hK scores]
60     simp
61   have hidx := List.index_of_argmax harg (List.mem_finRange a) hscore
62   simpa only [List.idxOf_finRange] using hidx

64 /-- A natural number encodes a maximizing arm for the supplied score vector. -/
65 def RealEncodedArgmaxCandidate {K : Nat} (scores : Fin K -> Real)
66     (n : Nat) : Prop :=
67   Exists fun a : Fin K =>
68     And (n = Encodable.encode a)
69       (forall z : Fin K, scores z <= scores a)

71 theorem exists_realEncodedArgmaxCandidate {K : Nat} (hK : 0 < K)
72     (scores : Fin K -> Real) :
73     Exists fun n : Nat => ETC.RealEncodedArgmaxCandidate scores n := by
74   refine Exists.intro (Encodable.encode (ETC.realArgmaxCommit hK scores)) ?_
75   refine Exists.intro (ETC.realArgmaxCommit hK scores) ?_
76   exact And.intro rfl (ETC.realArgmaxCommit_spec hK scores)

93 noncomputable def realLeastEncodedArgmax {K : Nat} (hK : 0 < K)

110 theorem realLeastEncodedArgmax_encode_le_of_isMax {K : Nat} (hK : 0 < K)
111     (scores : Fin K -> Real) (a : Fin K)
112     (ha : forall z : Fin K, scores z <= scores a) :
113     Encodable.encode (ETC.realLeastEncodedArgmax hK scores) <=
114       Encodable.encode a := by
115   classical
116   rw [ETC.realLeastEncodedArgmax_encode_eq_index]
117   unfold ETC.realLeastEncodedArgmaxIndex
118   exact Nat.find_min'

122 /-- The LML-shaped least-encoded selector equals the native strict fold. -/
123 theorem realLeastEncodedArgmax_eq_realArgmaxCommit {K : Nat} (hK : 0 < K)
124     (scores : Fin K -> Real) :
125     ETC.realLeastEncodedArgmax hK scores =
126       ETC.realArgmaxCommit hK scores := by
127   apply Encodable.encode_injective
128   have hleft : Encodable.encode (ETC.realLeastEncodedArgmax hK scores) <=
129       Encodable.encode (ETC.realArgmaxCommit hK scores) :=
130     ETC.realLeastEncodedArgmax_encode_le_of_isMax hK scores
131       (ETC.realArgmaxCommit hK scores)
132       (ETC.realArgmaxCommit_spec hK scores)
133   have hright : Encodable.encode (ETC.realArgmaxCommit hK scores) <=
134       Encodable.encode (ETC.realLeastEncodedArgmax hK scores) :=
135     ETC.realArgmaxCommit_encode_le_of_score_le hK scores
136       (ETC.realLeastEncodedArgmax hK scores)
137       (ETC.realLeastEncodedArgmax_spec hK scores
138         (ETC.realArgmaxCommit hK scores))
139   exact Nat.le_antisymm hleft hright

141 /--
142 Round-robin exploration, least-encoded commit, and persistence determine the
143 native Real ETC action trace. The boundary is written as `K * m` to match the
144 upstream ETC behavior lemmas; the local trace uses the definitionally
145 equivalent `m * K` boundary.
146 -/
147 theorem eventually_realExplorationArgmaxAction_eq_of_roundRobin_leastEncodedCommit_persist
148     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
149     (mu : Measure Omega) (spec : ETC.Spec K)
150     (baseCommitArm : Fin K)
151     (action : Omega -> ActionTrace (Fin K))
152     (reward : Omega -> RewardTrace Real)
153     (hactionExplore : forall t, t < K * spec.explorationPulls ->
154       Filter.EventuallyEq (ae mu)
155         (fun omega => action omega t)
156         (fun _omega => ETC.exploreArm spec t))
157     (hactionCommit : Filter.EventuallyEq (ae mu)
158       (fun omega => action omega (K * spec.explorationPulls))
159       (fun omega => ETC.realLeastEncodedArgmax spec.hK
160         (fun arm => ETC.realEmpMeanAtExploration spec baseCommitArm
161           (reward omega) arm)))
162     (hactionPersist : forall t, K * spec.explorationPulls <= t ->
163       Filter.EventuallyEq (ae mu)
164         (fun omega => action omega t)
165         (fun omega => action omega (K * spec.explorationPulls))) :
166     Filter.Eventually
167       (fun omega => forall t,
168         action omega t =
169           ETC.realExplorationArgmaxAction spec baseCommitArm
170             (reward omega) t)
171       (ae mu) := by
172   rw [ae_all_iff]
173   intro t
174   by_cases ht : t < spec.explorationPulls * K
175   case pos =>
176     have htUpstream : t < K * spec.explorationPulls := by
177       simpa [Nat.mul_comm] using ht
178     filter_upwards [hactionExplore t htUpstream] with omega homega
179     simpa [ETC.realExplorationArgmaxAction,
180       ETC.actionWithCommit_eq_exploreArm_of_lt spec
181         (ETC.realExplorationArgmaxCommit spec baseCommitArm (reward omega)) ht]
182       using homega
183   case neg =>
184     have hge : spec.explorationPulls * K <= t := Nat.le_of_not_gt ht
185     have hgeUpstream : K * spec.explorationPulls <= t := by
186       simpa [Nat.mul_comm] using hge
187     filter_upwards [hactionPersist t hgeUpstream, hactionCommit]
188       with omega hpersist hcommit
189     calc
190       action omega t = action omega (K * spec.explorationPulls) := hpersist
191       _ = ETC.realLeastEncodedArgmax spec.hK
192           (fun arm => ETC.realEmpMeanAtExploration spec baseCommitArm
193             (reward omega) arm) := hcommit
194       _ = ETC.realExplorationArgmaxCommit spec baseCommitArm (reward omega) := by
195         simpa [ETC.realExplorationArgmaxCommit] using
196           (ETC.realLeastEncodedArgmax_eq_realArgmaxCommit spec.hK
197             (fun arm => ETC.realEmpMeanAtExploration spec baseCommitArm
198               (reward omega) arm))
199       _ = ETC.realExplorationArgmaxAction spec baseCommitArm
200           (reward omega) t := by
201         symm
202         exact ETC.actionWithCommit_eq_commitArm_of_ge spec
203           (ETC.realExplorationArgmaxCommit spec baseCommitArm (reward omega)) hge

211 theorem integral_realKernelRegret_externalAction_le_exact_sum_of_actionDependent_actionRewardHistory_condDistrib_of_leastEncodedCommit_persist
212     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
213     (mu : Measure Omega) [IsProbabilityMeasure mu]
214     (spec : ETC.Spec K)
215     (nu : ProbabilityTheory.Kernel (Fin K) Real)
216     [ProbabilityTheory.IsMarkovKernel nu]
217     (sigma2 : NNReal)
218     (hsubG : forall arm, ProbabilityTheory.HasSubgaussianMGF
219       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm))
220     (hm : 0 < spec.explorationPulls) (n : Nat)
221     (hn : K * spec.explorationPulls <= n)
222     (action : Omega -> ActionTrace (Fin K))
223     (reward : Omega -> RewardTrace Real)
224     (haction : forall t, Measurable (fun omega => action omega t))
225     (hreward : forall t, Measurable (fun omega => reward omega t))
226     (hactionExplore : forall t, t < K * spec.explorationPulls ->
227       Filter.EventuallyEq (ae mu)
228         (fun omega => action omega t)
229         (fun _omega => ETC.exploreArm spec t))
230     (hactionCommit : Filter.EventuallyEq (ae mu)
231       (fun omega => action omega (K * spec.explorationPulls))
232       (fun omega => ETC.realLeastEncodedArgmax spec.hK
233         (fun arm => ETC.realEmpMeanAtExploration spec
234           (ETC.realKernelBestArm spec.hK nu) (reward omega) arm)))
235     (hactionPersist : forall t, K * spec.explorationPulls <= t ->
236       Filter.EventuallyEq (ae mu)
237         (fun omega => action omega t)
238         (fun omega => action omega (K * spec.explorationPulls)))
239     (hzero : ProbabilityTheory.condDistrib
240         (fun omega => reward omega 0)
241         (fun omega => action omega 0) mu =ᵐ[mu.map (fun omega => action omega 0)]
242       ProbabilityTheory.Kernel.ofFunOfCountable (fun arm : Fin K => nu arm))
243     (hcond : forall i, i < spec.explorationPulls * K - 1 ->
244       let fullCondition := fun omega : Omega =>
245         (History.finitePairHistoryOfTrace
246             (action omega) (reward omega) i,
247           action omega (i + 1))
248       ProbabilityTheory.condDistrib
249           (fun omega => reward omega (i + 1))
250           fullCondition mu =ᵐ[mu.map fullCondition]
251         (RewardKernel.contextIndependentOfActionLaws
252           (Context := History.FinitePairHistory (Fin K) Real i)
253           (fun arm : Fin K => nu arm)
254           (fun _arm => inferInstance)).kernel) :
255     integral mu (fun omega => realKernelRegret nu (action omega) n) <=
256       (Finset.univ : Finset (Fin K)).sum (fun arm =>
257         realKernelGap nu arm *
258           ((spec.explorationPulls : Real) +
259             ((n - K * spec.explorationPulls : Nat) : Real) *
260               Real.exp
261                 (-(spec.explorationPulls : Real) *
262                   (realKernelGap nu arm) ^ 2 /
263                     (4 * (sigma2 : Real))))) := by

266       mu spec nu sigma2 hsubG hm n hn action reward haction hreward
267   case hactionExplore =>
268     intro t ht
269     exact hactionExplore t (by simpa [Nat.mul_comm] using ht)
270   case hzero => exact hzero
271   case hcond => exact hcond
272   case hactionETC =>
273     have hall :=
274       ETC.eventually_realExplorationArgmaxAction_eq_of_roundRobin_leastEncodedCommit_persist
275         mu spec (ETC.realKernelBestArm spec.hK nu) action reward
