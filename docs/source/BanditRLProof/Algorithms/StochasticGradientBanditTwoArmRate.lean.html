3 /-!
4 # Two-arm stochastic-gradient-bandit rate identities
5 
6 This module starts the source-facing rate layer for Theorem 1 of
7 Baudry--Johnson--Vary--Pike-Burke--Rebeschini (NeurIPS 2025).  It proves that
8 Algorithm 1 preserves the zero sum of its parameter vector on every generated
9 finite history, then specializes the two-arm softmax law to the exact odds
10 identities used as Equation (11) in the source proof.
11 
12 These pathwise identities are inputs to the exponential-moment and
13 telescoping arguments.  They do not by themselves prove Equation (8), the
14 failure-mass estimate, or Theorem 1's regret bound.
15 -/

26 /-- Algorithm 1 preserves the sum of an arbitrary initial parameter vector
27 on every inclusive finite action/reward history. -/
28 theorem historyParameter_sum_eq_initial
29     {Action : Type u} [Fintype Action] [DecidableEq Action] [Nonempty Action]
30     (initialTheta : Action -> Real) (eta : Real) :
31     forall n (history : History.FinitePairHistory Action Real n),
32       (∑ coordinate, historyParameter initialTheta eta n history coordinate) =
33         ∑ coordinate, initialTheta coordinate := by
34   classical
35   intro n
36   induction n with
37   | zero =>
38       intro history
39       rw [show (∑ coordinate,
40           historyParameter initialTheta eta 0 history coordinate) =
41           ∑ coordinate,
42             (initialTheta coordinate + eta *
43               sourceIncrement (softmaxProbability initialTheta)
44                 (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).2
45                 (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 coordinate) by
46         apply Finset.sum_congr rfl
47         intro coordinate _
48         rw [historyParameter_zero]]
49       rw [Finset.sum_add_distrib, ← Finset.mul_sum,
50         sum_sourceIncrement]
51       · ring
52       · exact softmaxProbability_sum initialTheta
53   | succ n ih =>
54       intro history
55       rw [show (∑ coordinate,
56           historyParameter initialTheta eta (n + 1) history coordinate) =
57           ∑ coordinate,
58             (historyParameter initialTheta eta n
59                 (Exp3.previousPairHistory history) coordinate +
60               eta * sourceIncrement
61                 (softmaxProbability
62                   (historyParameter initialTheta eta n
63                     (Exp3.previousPairHistory history)))
64                 (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).2
65                 (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).1 coordinate) by
66         apply Finset.sum_congr rfl
67         intro coordinate _
68         rw [historyParameter_succ]]
69       rw [Finset.sum_add_distrib, ← Finset.mul_sum,
70         sum_sourceIncrement]
71       · simpa using ih (Exp3.previousPairHistory history)
72       · exact softmaxProbability_sum
73           (historyParameter initialTheta eta n
74             (Exp3.previousPairHistory history))

76 /-- The source initialization `theta = 0` therefore stays zero-sum pathwise. -/
77 theorem historyParameter_zeroInitialization_sum
78     {Action : Type u} [Fintype Action] [DecidableEq Action] [Nonempty Action]
79     (eta : Real) (n : Nat)
80     (history : History.FinitePairHistory Action Real n) :
81     ∑ coordinate,
82       historyParameter (fun _ : Action => 0) eta n history coordinate = 0 := by
83   rw [historyParameter_sum_eq_initial]
84   simp

86 /-- Source-time parameter adapter for one infinite two-arm action/reward trace.
87 Lean time `0` is the pre-action source parameter `theta_{.,1} = 0`; Lean time
88 `n + 1` is the parameter after consuming trace pair `n`, namely source
89 `theta_{.,n+2}`. -/
90 noncomputable def twoArmParameterAt (eta : Real)
91     (trace : Nat -> Fin 2 × Real) : Nat -> Fin 2 -> Real
92   | 0 => fun _ => 0
93   | n + 1 =>
94       historyParameter (fun _ : Fin 2 => 0) eta n
95         (Preorder.frestrictLe n trace)
96 
97 /-- The two-arm softmax law generated from the source-time parameter adapter. -/
98 def twoArmProbabilityAt (eta : Real) (trace : Nat -> Fin 2 × Real)
99     (time : Nat) : Fin 2 -> Real :=
100   softmaxProbability (twoArmParameterAt eta trace time)
101 
102 @[simp]
103 theorem twoArmParameterAt_zero (eta : Real)
104     (trace : Nat -> Fin 2 × Real) (arm : Fin 2) :
105     twoArmParameterAt eta trace 0 arm = 0 := rfl

107 @[simp]
108 theorem twoArmParameterAt_succ (eta : Real)
109     (trace : Nat -> Fin 2 × Real) (n : Nat) (arm : Fin 2) :
110     twoArmParameterAt eta trace (n + 1) arm =
111       historyParameter (fun _ : Fin 2 => 0) eta n
112         (Preorder.frestrictLe n trace) arm := rfl

114 /-- The source-time two-arm parameter stays zero-sum on every path. -/
115 theorem twoArmParameterAt_sum_eq_zero (eta : Real)
116     (trace : Nat -> Fin 2 × Real) (time : Nat) :
117     ∑ arm, twoArmParameterAt eta trace time arm = 0 := by
118   cases time with
119   | zero => simp
120   | succ n =>
121       simpa using historyParameter_zeroInitialization_sum eta n
122         (Preorder.frestrictLe n trace)
123 
124 /-- Hence source arm `2` has the negative parameter of source arm `1`. -/
125 theorem twoArmParameterAt_one_eq_neg_zero (eta : Real)
126     (trace : Nat -> Fin 2 × Real) (time : Nat) :
127     twoArmParameterAt eta trace time 1 =
128       -twoArmParameterAt eta trace time 0 := by
129   have hsum := twoArmParameterAt_sum_eq_zero eta trace time
130   simp only [Fin.sum_univ_two] at hsum
131   linarith

133 /-- Algorithm 1's zero initialization gives the uniform two-arm law before
134 the first action. -/
135 @[simp]
136 theorem twoArmProbabilityAt_zero (eta : Real)
137     (trace : Nat -> Fin 2 × Real) (arm : Fin 2) :
138     twoArmProbabilityAt eta trace 0 arm = 1 / 2 := by
139   fin_cases arm <;>
140     norm_num [twoArmProbabilityAt, softmaxProbability,
141       softmaxDenominator, Fin.sum_univ_two]
142 
143 /-- On two arms, normalization identifies the second softmax probability with
144 the failure probability of the first arm. -/
145 theorem softmaxProbability_one_eq_one_sub_zero (theta : Fin 2 -> Real) :
146     softmaxProbability theta 1 = 1 - softmaxProbability theta 0 := by
147   have hsum := softmaxProbability_sum theta
148   simp only [Fin.sum_univ_two] at hsum
149   linarith
150 
151 /-- The exact two-arm softmax odds before using the zero-sum invariant. -/
152 theorem softmaxProbability_zero_div_one (theta : Fin 2 -> Real) :
153     softmaxProbability theta 0 / softmaxProbability theta 1 =
154       Real.exp (theta 0 - theta 1) := by
155   have hden : softmaxDenominator theta ≠ 0 :=
156     ne_of_gt (softmaxDenominator_pos theta)
157   have hexpOne : Real.exp (theta 1) ≠ 0 := ne_of_gt (Real.exp_pos _)
158   calc
159     softmaxProbability theta 0 / softmaxProbability theta 1 =
160         Real.exp (theta 0) / Real.exp (theta 1) := by
161       simp only [softmaxProbability]
162       field_simp
163     _ = Real.exp (theta 0 - theta 1) := by rw [Real.exp_sub]
164 
165 /-- A zero-sum two-arm parameter has opposite coordinates. -/
166 theorem finTwo_one_eq_neg_zero_of_sum_eq_zero
167     (theta : Fin 2 -> Real) (hsum : ∑ coordinate, theta coordinate = 0) :
168     theta 1 = -theta 0 := by
169   simp only [Fin.sum_univ_two] at hsum
170   linarith

172 /-- The printed division form of Equation (11). Lean arm `0` is source arm
173 `1`, and strict softmax positivity makes the denominator nonzero. -/
174 theorem softmaxProbability_zero_div_one_sub_zero_eq_exp_two_mul
175     (theta : Fin 2 -> Real) (hsum : ∑ coordinate, theta coordinate = 0) :
176     softmaxProbability theta 0 / (1 - softmaxProbability theta 0) =
177       Real.exp (2 * theta 0) := by
178   rw [← softmaxProbability_one_eq_one_sub_zero]
179   rw [softmaxProbability_zero_div_one,
180     finTwo_one_eq_neg_zero_of_sum_eq_zero theta hsum, sub_neg_eq_add,
181     ← two_mul]
182 
183 /-- The multiplication form of Equation (11) used in the source Theorem 1
184 proof: `exp(2 theta_1) (1 - p_1) = p_1`. Lean arm `0` is source arm `1`. -/
185 theorem exp_two_mul_zero_mul_one_sub_softmaxProbability_zero
186     (theta : Fin 2 -> Real) (hsum : ∑ coordinate, theta coordinate = 0) :
187     Real.exp (2 * theta 0) * (1 - softmaxProbability theta 0) =
188       softmaxProbability theta 0 := by
189   rw [← softmaxProbability_one_eq_one_sub_zero]
190   have hprobOne : softmaxProbability theta 1 ≠ 0 :=
191     ne_of_gt (softmaxProbability_pos theta 1)
192   have hodds := softmaxProbability_zero_div_one theta
193   rw [finTwo_one_eq_neg_zero_of_sum_eq_zero theta hsum] at hodds
194   rw [sub_neg_eq_add, ← two_mul] at hodds
195   exact ((div_eq_iff hprobOne).mp hodds).symm
196 
197 /-- The inverse-odds form used for the failure-mass telescoping potential in
198 the second half of the source Theorem 1 proof. -/
199 theorem exp_neg_two_mul_zero_mul_softmaxProbability_zero
200     (theta : Fin 2 -> Real) (hsum : ∑ coordinate, theta coordinate = 0) :
201     Real.exp (-2 * theta 0) * softmaxProbability theta 0 =
202       1 - softmaxProbability theta 0 := by
203   rw [← softmaxProbability_one_eq_one_sub_zero]
204   have heq :=
205     exp_two_mul_zero_mul_one_sub_softmaxProbability_zero theta hsum
206   rw [← softmaxProbability_one_eq_one_sub_zero] at heq
207   rw [← heq, ← mul_assoc, ← Real.exp_add]
208   ring_nf
209   simp

211 /-- Source-time Equation (11) on every infinite action/reward trace. -/
212 theorem twoArmProbabilityAt_exp_two_mul_failure_eq_success
213     (eta : Real) (trace : Nat -> Fin 2 × Real) (time : Nat) :
214     Real.exp (2 * twoArmParameterAt eta trace time 0) *
215         (1 - twoArmProbabilityAt eta trace time 0) =
216       twoArmProbabilityAt eta trace time 0 := by
217   exact exp_two_mul_zero_mul_one_sub_softmaxProbability_zero
218     (twoArmParameterAt eta trace time)
219     (twoArmParameterAt_sum_eq_zero eta trace time)
220 
221 /-- The printed Equation (11) on the explicitly fenced source-time trace. -/
222 theorem twoArmProbabilityAt_zero_div_failure_eq_exp_two_mul
223     (eta : Real) (trace : Nat -> Fin 2 × Real) (time : Nat) :
224     twoArmProbabilityAt eta trace time 0 /
225         (1 - twoArmProbabilityAt eta trace time 0) =
226       Real.exp (2 * twoArmParameterAt eta trace time 0) := by
227   exact softmaxProbability_zero_div_one_sub_zero_eq_exp_two_mul
228     (twoArmParameterAt eta trace time)

231 /-- Finite-history multiplication form of Equation (11) under Algorithm 1's
232 source initialization. -/
233 theorem historyParameter_exp_two_mul_zero_eq_odds
234     (eta : Real) (n : Nat)
235     (history : History.FinitePairHistory (Fin 2) Real n) :
236     Real.exp
237         (2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) *
238         (1 - softmaxProbability
239           (historyParameter (fun _ : Fin 2 => 0) eta n history) 0) =
240       softmaxProbability
241         (historyParameter (fun _ : Fin 2 => 0) eta n history) 0 := by
242   apply exp_two_mul_zero_mul_one_sub_softmaxProbability_zero
