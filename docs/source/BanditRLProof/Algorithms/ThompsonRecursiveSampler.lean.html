5 # Globally coupled Thompson trajectories
6 
7 This module closes the gap between a separately adjoined one-step posterior
8 sampler and the action coordinate of one recursive trajectory.  The Thompson
9 policy is defined non-circularly from a fixed reference trajectory, following
10 the uniform-reference design of LML's `TS.policy`.
11 -/

34 /-- Every measure on a finite space is dominated by its uniform measure. -/
35 theorem absolutelyContinuous_uniformActionMeasure
36     {Action : Type u} [Fintype Action] [Nonempty Action]
37     [MeasurableSpace Action] (mu : Measure Action) :
38     mu ≪ uniformActionMeasure Action := by
39   apply Measure.AbsolutelyContinuous.mk
40   intro event hevent huniform
41   by_cases hempty : event = ∅
42   · simp [hempty]
43   letI : Fintype event := Fintype.ofFinite event
44   have heventNonempty : event.Nonempty := Set.nonempty_iff_ne_empty.mpr hempty
45   have hcard : (Fintype.card event : ENNReal) ≠ 0 := by
46     exact_mod_cast (Fintype.card_pos_iff.mpr
47       (Set.nonempty_coe_sort.mpr heventNonempty)).ne'
48   have hpositive :
49       0 < (Fintype.card event : ENNReal) / Fintype.card Action :=
50     ENNReal.div_pos hcard (ENNReal.natCast_ne_top _)
51   rw [← PMF.toMeasure_uniformOfFintype_apply event hevent] at hpositive
52   unfold uniformActionMeasure at huniform
53   rw [huniform] at hpositive
54   exact (lt_irrefl 0 hpositive).elim

80 /--
81 Mixing environment-indexed trajectory laws preserves a common conditional
82 action kernel.  This is the generic measure transport needed to turn
83 pointwise trajectory laws into one global recursive process law.
84 -/
85 theorem trajectoryMixture_map_history_action_eq_compProd
86     {Env : Type u} {Omega : Type v} {History : Type w} {Action : Type x}
87     [MeasurableSpace Env] [MeasurableSpace Omega]
88     [MeasurableSpace History] [MeasurableSpace Action]
89     (prior : Measure Env) [IsFiniteMeasure prior]
90     (trajectory : ProbabilityTheory.Kernel Env Omega)
91     [ProbabilityTheory.IsMarkovKernel trajectory]
92     (history : Omega -> History) (hhistory : Measurable history)
93     (action : Omega -> Action) (haction : Measurable action)
94     (policy : ProbabilityTheory.Kernel History Action)
95     [ProbabilityTheory.IsMarkovKernel policy]
96     (hlaw : forall env,
97       (trajectory env).map (fun omega => (history omega, action omega)) =
98         (trajectory env).map history ⊗ₘ policy) :
99     (prior ⊗ₘ trajectory).map
100         (fun sample => (history sample.2, action sample.2)) =
101       (prior ⊗ₘ trajectory).map (history ∘ Prod.snd) ⊗ₘ policy := by
102   apply Measure.ext_prod
103   intro historyEvent actionEvent hhistoryEvent hactionEvent
104   let pairMap : Omega -> History × Action :=
105     fun omega => (history omega, action omega)
106   let globalPairMap : Env × Omega -> History × Action :=
107     fun sample => pairMap sample.2
108   let globalHistory : Env × Omega -> History := history ∘ Prod.snd
109   have hpairMap : Measurable pairMap := hhistory.prodMk haction
110   have hglobalPairMap : Measurable globalPairMap :=
111     hpairMap.comp measurable_snd
112   have hglobalHistory : Measurable globalHistory :=
113     hhistory.comp measurable_snd
114   have hpairEvent :
115       MeasurableSet (pairMap ⁻¹' (historyEvent ×ˢ actionEvent)) :=
116     hpairMap (hhistoryEvent.prod hactionEvent)
117   have hpreimage :
118       globalPairMap ⁻¹' (historyEvent ×ˢ actionEvent) =
119         Set.univ ×ˢ (pairMap ⁻¹' (historyEvent ×ˢ actionEvent)) := by
120     ext sample
121     simp [globalPairMap, pairMap]
122   have hpointwise (env : Env) :
123       trajectory env (pairMap ⁻¹' (historyEvent ×ˢ actionEvent)) =
124         ∫⁻ h in historyEvent, policy h actionEvent
125           ∂(trajectory env).map history := by
126     rw [← Measure.map_apply hpairMap (hhistoryEvent.prod hactionEvent),
127       hlaw env, Measure.compProd_apply_prod hhistoryEvent hactionEvent]
128   rw [Measure.map_apply hglobalPairMap (hhistoryEvent.prod hactionEvent),
129     hpreimage, Measure.compProd_apply_prod MeasurableSet.univ hpairEvent]
130   simp_rw [hpointwise]
131   simp_rw [MeasureTheory.setLIntegral_map hhistoryEvent
132     (policy.measurable_coe hactionEvent) hhistory]
133   rw [Measure.compProd_apply_prod hhistoryEvent hactionEvent,
134     MeasureTheory.setLIntegral_map hhistoryEvent
135       (policy.measurable_coe hactionEvent) hglobalHistory]
136   have hhistoryPreimage :
137       globalHistory ⁻¹' historyEvent =
138         Set.univ ×ˢ (history ⁻¹' historyEvent) := by
139     ext sample
140     simp [globalHistory]
141   rw [hhistoryPreimage]
142   simpa only [globalHistory, Function.comp_apply] using
143     (Measure.setLIntegral_compProd
144       (((policy.measurable_coe hactionEvent).comp hhistory).comp measurable_snd)
145       MeasurableSet.univ (hhistory hhistoryEvent)).symm
