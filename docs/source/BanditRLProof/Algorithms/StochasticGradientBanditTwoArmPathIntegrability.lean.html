4 /-!
5 # Path-level integrability for the two-arm SGB exponential recurrences
6 
7 This module closes the finite-time regularity boundary needed before the
8 conditional-distribution recurrences can be used in a tower argument.
9 
10 Probability / process / filtration ledger:
11 
12 * the ambient measure is `twoArmTrajectoryMeasure prior eta environment`, a
13   finite (not necessarily probability) measure unless the caller separately
14   assumes `IsProbabilityMeasure prior`;
15 * the random process is the forward and inverse exponential potential after
16   consuming trace coordinates `0,...,n+1`;
17 * `twoArmPrefixFiltration` retains the latent environment and the inclusive
18   action/reward prefix; no independence assertion is introduced;
19 * `TwoArmBoundedFixedMeanEnvironmentContract` supplies `|reward| <= 1` on each
20   initial/successor reward fiber.  The mixed initial-pair and prefix/next-pair
21   laws transport those fiber statements to the canonical full trajectory;
22 * the finite-history parameter is then bounded pathwise by the number of
23   consumed rewards times `|eta|`, which gives an integrable deterministic
24   exponential envelope on every fixed finite horizon;
25 * no gap positivity, conditional MGF, summability, or all-time uniform bound
26   is used here.  These declarations establish fixed-`n` successor
27   integrability and one-step conditional recurrences only; they do not supply
28   the initial unconditional trajectory step, recurrence-bound integrability,
29   tower iteration, telescoping, or a regret endpoint.
30 
31 The support transport is explicit: a reward-fiber a.e. statement is never
32 treated as a full-trajectory a.e. statement without first using the exact
33 canonical mixed law and `ae_map_iff`.
34 -/

95 theorem twoArmTrajectoryMeasure_reward_zero_abs_le_one_ae
96     {Env : Type v} [MeasurableSpace Env]
97     (prior : Measure Env) [IsFiniteMeasure prior]
98     (eta : Real)
99     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
100     (mean : Fin 2 -> Real)
101     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean) :
102     ∀ᵐ sample ∂twoArmTrajectoryMeasure prior eta environment,
103       |(sample.2 0).2| <= 1 := by
104   let mu := twoArmTrajectoryMeasure prior eta environment
105   let joinEnvironmentInitial :=
106     fun sample : Env × ((k : Nat) -> Fin 2 × Real) =>
107       (sample.1, sample.2 0)
108   have hjoin : Measurable joinEnvironmentInitial :=
109     measurable_fst.prodMk ((measurable_pi_apply 0).comp measurable_snd)
110   have hp : MeasurableSet
111       {sample : Env × (Fin 2 × Real) | |sample.2.2| <= 1} := by
112     exact measurableSet_le
113       ((measurable_snd.comp measurable_snd).abs) measurable_const
114   have hkernel :
115       mu.map joinEnvironmentInitial =
116         prior ⊗ₘ
117           Thompson.measurableEnvironmentInitialPairKernel
118             (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment := by
119     simpa [mu, twoArmTrajectoryMeasure, joinEnvironmentInitial] using
120       (Exp3.canonicalMeasurableEnvironmentTrajectoryMeasure_map_environment_eval_zero
121         prior (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment)
122   have hjoint : ∀ᵐ sample ∂
123       prior ⊗ₘ
124         Thompson.measurableEnvironmentInitialPairKernel
125           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment,
126       |sample.2.2| <= 1 := by
127     apply Measure.ae_compProd_of_ae_ae hp
128     filter_upwards [] with env
129     have hpPair : MeasurableSet
130         {pair : Fin 2 × Real | |pair.2| <= 1} := by
131       exact measurableSet_le measurable_snd.abs measurable_const
132     rw [Thompson.measurableEnvironmentInitialPairKernel_apply]
133     apply Measure.ae_compProd_of_ae_ae hpPair
134     filter_upwards [] with selected
135     simpa [Thompson.MeasurableHistoryEnvironment.at,
136       Kernel.comap_apply] using
137       contract.initial_reward_bound env selected
138   rw [← hkernel] at hjoint
139   exact (ae_map_iff hjoin.aemeasurable hp).mp hjoint

234 /-- One Algorithm-1 coordinate update is no larger than the observed reward
235 when the coordinate probability lies in `[0,1]`. -/
236 theorem abs_sourceIncrement_le_abs_reward_of_mem_Icc
237     {Action : Type*} [DecidableEq Action]
238     (p : Action -> Real) (reward : Real)
239     (selected coordinate : Action)
240     (hp_nonneg : 0 <= p coordinate) (hp_le_one : p coordinate <= 1) :
241     |sourceIncrement p reward selected coordinate| <= |reward| := by
242   by_cases hselected : selected = coordinate
243   · rw [sourceIncrement, if_pos hselected, abs_mul,
244       abs_of_nonneg (sub_nonneg.mpr hp_le_one)]
245     exact mul_le_of_le_one_left (abs_nonneg reward) (by linarith)
246   · rw [sourceIncrement, if_neg hselected, abs_neg, abs_mul,
247       abs_of_nonneg hp_nonneg]
248     exact mul_le_of_le_one_left (abs_nonneg reward) hp_le_one

250 /-- Softmax coordinates satisfy the probability premises of the preceding

252 theorem abs_sourceIncrement_softmax_le_abs_reward
253     {Action : Type*} [Fintype Action] [DecidableEq Action] [Nonempty Action]
254     (theta : Action -> Real) (reward : Real) (selected coordinate : Action) :
255     |sourceIncrement (softmaxProbability theta) reward selected coordinate| <=

259       (softmaxProbability_nonneg theta coordinate)
