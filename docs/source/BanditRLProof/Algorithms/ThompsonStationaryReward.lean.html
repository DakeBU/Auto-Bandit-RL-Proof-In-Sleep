114 /-- At a fixed environment, the sampled latent table has the canonical product arm-stream law. -/
115 theorem stationaryArmStreamKernel_apply
116     {Env : Type u} {K : Nat} [MeasurableSpace Env]
117     (rewardKernel : Kernel (Env × Fin K) Real) [IsMarkovKernel rewardKernel]
118     (env : Env) :
119     stationaryArmStreamKernel rewardKernel env =
120       UCB.armStreamMeasure (stationaryRewardKernelAt rewardKernel env) := by
121   rw [stationaryArmStreamKernel, Kernel.map_apply _
122     (measurable_rewardStreamOfUnitArmStream rewardKernel)]
123   have hbase :
124       (Kernel.id ×ₖ Kernel.const Env (uniformUnitArmStreamMeasure K)) env =
125         Measure.map (Prod.mk env) (uniformUnitArmStreamMeasure K) := by
126     ext s hs
127     rw [Kernel.id_prod_apply' _ _ hs, Kernel.const_apply]
128     exact (Measure.map_apply
129       (measurable_const.prodMk measurable_id) hs).symm
130   rw [hbase, Measure.map_map]
131   · change Measure.map
132         (fun stream : UnitArmStream K =>
133           fun i arm => stationaryRewardSampler rewardKernel (env, arm) (stream i arm))
134         (uniformUnitArmStreamMeasure K) = _
135     have hcoord (arm : Fin K) : Measurable
136         (fun u : Set.Icc (0 : Real) 1 =>
137           stationaryRewardSampler rewardKernel (env, arm) u) :=
138       (measurable_uncurry_stationaryRewardSampler rewardKernel).comp
139         (measurable_const.prodMk measurable_id)
140     have hrow : Measurable
141         (fun row : Fin K -> Set.Icc (0 : Real) 1 =>
142           fun arm => stationaryRewardSampler rewardKernel (env, arm) (row arm)) :=
143       measurable_pi_lambda _ fun arm =>
144         (hcoord arm).comp (measurable_pi_apply arm)
145     calc
146       _ = Measure.infinitePi (fun _ : Nat =>
147           (Measure.infinitePi fun _ : Fin K => volume).map
148             (fun row arm =>
149               stationaryRewardSampler rewardKernel (env, arm) (row arm))) := by
150         exact Measure.infinitePi_map_pi
151           (fun _ : Nat => Measure.infinitePi fun _ : Fin K => volume)
152           (f := fun _ row arm =>
153             stationaryRewardSampler rewardKernel (env, arm) (row arm))
154           (fun _ => hrow)
155       _ = Measure.infinitePi (fun _ : Nat =>
156           Measure.infinitePi fun arm : Fin K => rewardKernel (env, arm)) := by
157         congr 1
158         funext i
159         calc
160           _ = Measure.infinitePi (fun arm : Fin K =>
161                 volume.map (stationaryRewardSampler rewardKernel (env, arm))) := by
162               exact Measure.infinitePi_map_pi
163                 (fun _ : Fin K => volume)
164                 (f := fun arm u =>
165                   stationaryRewardSampler rewardKernel (env, arm) u)
166                 hcoord
167           _ = _ := by
168             simp_rw [stationaryRewardSampler_map_volume rewardKernel]
169       _ = UCB.armStreamMeasure
170           (stationaryRewardKernelAt rewardKernel env) := by
171         simp [UCB.armStreamMeasure, stationaryRewardKernelAt_apply]
172   · exact measurable_rewardStreamOfUnitArmStream rewardKernel
173   · exact measurable_const.prodMk measurable_id

380     (env : Env) (stream : UCB.ArmRewardStream K) (n : Nat) :

382         canonicalMeasurableEnvironmentTrajectoryKernel algorithm
383           (latentArmStreamMeasurableHistoryEnvironment
384             (Env := Env) (K := K)) (env, stream),
385       (trajectory (n + 1)).2 =
386         stream
387           (pullCount
388             (canonicalHistoryTrajectoryAction (Reward := Real) trajectory)
389             (trajectory (n + 1)).1 (n + 1))

392     (Env := Env) (K := K)

394     canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment

416       mu.map (Preorder.frestrictLe n) ⊗ₘ
417         historyStepKernel algorithm (environment.at (env, stream)) n,
418       sample.2.2 =
419         stream (ETC.realHistoryPullCount n sample.1 sample.2.1)
420           sample.2.1 := by
421     apply Measure.ae_compProd_of_ae_ae hp
422     filter_upwards [] with history
423     have hpPair : MeasurableSet
424         {pair : Fin K × Real |
425           pair.2 = stream (ETC.realHistoryPullCount n history pair.1) pair.1} :=

427         (hnextFixed.comp (measurable_const.prodMk measurable_fst))
428     rw [historyStepKernel,
429       Kernel.compProd_apply_eq_compProd_sectR]

431     filter_upwards [] with arm
432     rw [Kernel.sectR_apply,
433       latentArmStreamMeasurableHistoryEnvironment_at_feedback_apply]

435   rw [← hkernel] at hjoint

437     (ae_map_iff

439         (measurable_pi_apply (n + 1))).aemeasurable hp).mp hjoint
