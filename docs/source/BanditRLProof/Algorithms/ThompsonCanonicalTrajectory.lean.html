212 /-- The combined successor pair law determines the successor action policy. -/
213 theorem policy_condDistrib_of_historyAlgorithmEnvironmentSequence
214     {Omega : Type w} {Action : Type u} {Reward : Type v}
215     [MeasurableSpace Omega]
216     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
217     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
218     (mu : Measure Omega) [IsFiniteMeasure mu]
219     (action : Omega -> ActionTrace Action)
220     (reward : Omega -> RewardTrace Reward)
221     (algorithm : HistoryAlgorithm Action Reward)
222     (environment : HistoryEnvironment Action Reward)
223     (source : IsHistoryAlgorithmEnvironmentSequence
224       mu action reward algorithm environment) (n : Nat) :
225     ProbabilityTheory.condDistrib (fun omega => action omega (n + 1))
226         (fun omega => History.finitePairHistoryOfTrace
227           (action omega) (reward omega) n) mu =ᵐ[
228       mu.map (fun omega => History.finitePairHistoryOfTrace
229         (action omega) (reward omega) n)] algorithm.policy n := by
230   let history := fun omega => History.finitePairHistoryOfTrace
231     (action omega) (reward omega) n
232   let nextPair := fun omega => (action omega (n + 1), reward omega (n + 1))
233   have hcomp :
234       ProbabilityTheory.condDistrib (Prod.fst ∘ nextPair) history mu =ᵐ[
235         mu.map history]
236         (ProbabilityTheory.condDistrib nextPair history mu).map Prod.fst :=
237     ProbabilityTheory.condDistrib_comp
238       (μ := mu) (mβ := inferInstance) (Y := nextPair) history
239       ((source.measurable_action (n + 1)).prod
240         (source.measurable_reward (n + 1))).aemeasurable
241       (f := Prod.fst) measurable_fst
242   have hresult :
243       ProbabilityTheory.condDistrib (Prod.fst ∘ nextPair) history mu =ᵐ[
244         mu.map history] algorithm.policy n := by
245     filter_upwards [hcomp, source.step_condDistrib n] with h hcompAt hpairAt
246     calc
247       ProbabilityTheory.condDistrib (Prod.fst ∘ nextPair)
248           history mu h =
249           ((ProbabilityTheory.condDistrib nextPair history mu).map Prod.fst) h :=
250         hcompAt
251       _ = ((historyStepKernel algorithm environment n).map Prod.fst) h := by
252         rw [ProbabilityTheory.Kernel.map_apply _ measurable_fst,
253           ProbabilityTheory.Kernel.map_apply _ measurable_fst, hpairAt]
254       _ = algorithm.policy n h := by
255         rw [historyStepKernel_map_fst]
256   simpa [history, nextPair, Function.comp_def] using hresult

259 The combined successor pair law also determines the feedback conditional law

262 theorem feedback_condDistrib_of_historyAlgorithmEnvironmentSequence
263     {Omega : Type w} {Action : Type u} {Reward : Type v}
264     [MeasurableSpace Omega]
265     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
266     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
267     (mu : Measure Omega) [IsFiniteMeasure mu]
268     (action : Omega -> ActionTrace Action)
269     (reward : Omega -> RewardTrace Reward)
270     (algorithm : HistoryAlgorithm Action Reward)
271     (environment : HistoryEnvironment Action Reward)
272     (source : IsHistoryAlgorithmEnvironmentSequence
273       mu action reward algorithm environment) (n : Nat) :
274     ProbabilityTheory.condDistrib (fun omega => reward omega (n + 1))
275         (fun omega =>
276           (History.finitePairHistoryOfTrace
277             (action omega) (reward omega) n, action omega (n + 1))) mu =ᵐ[
278       mu.map (fun omega =>
279         (History.finitePairHistoryOfTrace
280           (action omega) (reward omega) n, action omega (n + 1)))]
281       environment.feedback n := by
282   let history := fun omega => History.finitePairHistoryOfTrace
283     (action omega) (reward omega) n
284   let nextAction := fun omega => action omega (n + 1)
285   let nextReward := fun omega => reward omega (n + 1)
286   let nextPair := fun omega => (nextAction omega, nextReward omega)
287   have hhistory : Measurable history :=
288     History.measurable_finitePairHistoryOfTrace
289       action reward source.measurable_action source.measurable_reward n
290   have hnextAction : Measurable nextAction := source.measurable_action (n + 1)
291   have hnextReward : Measurable nextReward := source.measurable_reward (n + 1)
292   have hpolicy := policy_condDistrib_of_historyAlgorithmEnvironmentSequence
293     mu action reward algorithm environment source n
294   have hpolicyJoint :
295       mu.map (fun omega => (history omega, nextAction omega)) =
296         mu.map history ⊗ₘ algorithm.policy n :=
297     (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
298       history hnextAction.aemeasurable (algorithm.policy n)).mp <| by
299         simpa [history, nextAction] using hpolicy
300   have hpairJoint :
301       mu.map (fun omega => (history omega, nextPair omega)) =
302         mu.map history ⊗ₘ historyStepKernel algorithm environment n := by
303     simpa [history, nextPair, nextAction, nextReward] using
304       nextPairJointLaw_eq_compProd
305         mu action reward algorithm environment source n
306   have hfeedbackJoint :
307       mu.map (fun omega =>
308           ((history omega, nextAction omega), nextReward omega)) =
309         mu.map (fun omega => (history omega, nextAction omega)) ⊗ₘ
310           environment.feedback n := by
311     calc
312       mu.map (fun omega =>
313           ((history omega, nextAction omega), nextReward omega)) =
314           (mu.map (fun omega => (history omega, nextPair omega))).map
315             MeasurableEquiv.prodAssoc.symm := by
316         rw [Measure.map_map MeasurableEquiv.prodAssoc.symm.measurable
317           (hhistory.prod (hnextAction.prod hnextReward))]
318         exact Measure.map_congr <| Filter.Eventually.of_forall <| fun _ => rfl
319       _ = (mu.map history ⊗ₘ
320           historyStepKernel algorithm environment n).map
321             MeasurableEquiv.prodAssoc.symm := by
322         rw [hpairJoint]
323       _ = (mu.map history ⊗ₘ
324           (algorithm.policy n ⊗ₖ environment.feedback n)).map
325             MeasurableEquiv.prodAssoc.symm := by
326         rfl
327       _ = (mu.map history ⊗ₘ algorithm.policy n) ⊗ₘ
328           environment.feedback n := Measure.compProd_assoc
329       _ = mu.map (fun omega => (history omega, nextAction omega)) ⊗ₘ
330           environment.feedback n := by
331         rw [hpolicyJoint]
332   have hresult :=
333     ProbabilityTheory.condDistrib_ae_eq_of_measure_eq_compProd
334       (fun omega => (history omega, nextAction omega))
335       hnextReward.aemeasurable hfeedbackJoint
336   simpa [history, nextAction, nextReward] using hresult

482 /--
483 For a composition-product sample, conditioning the complete sample on its
484 first coordinate keeps that coordinate and uses the supplied second-coordinate
485 kernel.
486 -/
487 theorem condDistrib_id_fst_compProd_ae_eq_kernelWithInput
488     {Input : Type u} {Output : Type v}
489     [MeasurableSpace Input] [StandardBorelSpace Input] [Nonempty Input]
490     [MeasurableSpace Output] [StandardBorelSpace Output] [Nonempty Output]
491     (prior : Measure Input) [IsFiniteMeasure prior]
492     (kernel : ProbabilityTheory.Kernel Input Output)
493     [ProbabilityTheory.IsMarkovKernel kernel] :
494     ProbabilityTheory.condDistrib id Prod.fst (prior ⊗ₘ kernel) =ᵐ[prior]
495       kernelWithInput kernel := by
496   let sampleMeasure := prior ⊗ₘ kernel
497   have hfirst : sampleMeasure.map Prod.fst = prior := by
498     change (prior ⊗ₘ kernel).fst = prior
499     exact Measure.fst_compProd prior kernel
500   have hjoint :
501       sampleMeasure.map (fun sample => (sample.1, sample)) =
502         sampleMeasure.map Prod.fst ⊗ₘ kernelWithInput kernel := by
503     rw [hfirst]
504     ext event hevent
505     rw [Measure.map_apply
506       (f := fun sample : Input × Output => (sample.1, sample))
507       (measurable_fst.prodMk measurable_id) hevent]
508     change (prior ⊗ₘ kernel)
509         ((fun sample : Input × Output => (sample.1, sample)) ⁻¹' event) =
510       (prior ⊗ₘ kernelWithInput kernel) event
511     rw [Measure.compProd_apply
512       (μ := prior) (κ := kernel)
513       (s := (fun sample : Input × Output => (sample.1, sample)) ⁻¹' event)
