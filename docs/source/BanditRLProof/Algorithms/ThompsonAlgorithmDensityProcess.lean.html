6 # Recursive algorithm-density transport for finite pair histories
7 
8 This module ports the process-facing core of LML's algorithm-density theorem.
9 Two stochastic history policies interact with the same feedback environment.
10 If every actual action law is absolutely continuous with respect to the
11 reference action law, the actual finite action/reward history law is the
12 reference history law weighted by the recursive product of policy
13 Radon-Nikodym derivatives.

160 /-- Recursive product of the initial and policy action likelihood ratios. -/
161 noncomputable def historyDensity
162     {Action : Type u} {Reward : Type v}
163     [MeasurableSpace Action] [MeasurableSpace Reward]
164     [MeasurableSpace.CountablyGenerated Action]
165     (algorithm referenceAlgorithm : HistoryAlgorithm Action Reward) :
166     (n : Nat) -> History.FinitePairHistory Action Reward n -> ENNReal
167   | 0, history =>
168       algorithm.initialAction.rnDeriv referenceAlgorithm.initialAction
169         (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1
170   | n + 1, history =>
171       historyDensity algorithm referenceAlgorithm n (pairHistoryPrefix history) *
172         (algorithm.policy n).rnDeriv (referenceAlgorithm.policy n)
173           (pairHistoryPrefix history) (pairHistoryLast history).1

488         funext i

490           Nat.eq_zero_of_le_zero (Finset.mem_Iic.mp i.2)
491         simp [pairRef, singletonPairHistory,
492           History.finitePairHistoryOfTrace, hi]

494           algorithm.initialAction =
495             referenceAlgorithm.initialAction.withDensity
496               (algorithm.initialAction.rnDeriv
497                 referenceAlgorithm.initialAction) :=
498         (Measure.withDensity_rnDeriv_eq _ _

510                   (algorithm.initialAction.rnDeriv
511                     referenceAlgorithm.initialAction) ⊗ₘ
512                 environment.initialFeedback :=

514               initial ⊗ₘ environment.initialFeedback) hinitialDensity
515           _ = _ := compProd_withDensity_left
516             referenceAlgorithm.initialAction environment.initialFeedback
517             (algorithm.initialAction.rnDeriv
518               referenceAlgorithm.initialAction)
519             (Measure.measurable_rnDeriv _ _)
