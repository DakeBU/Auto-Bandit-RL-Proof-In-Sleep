11 def centeredIndex (X : ℕ → Ω → ℝ) (δ : ℝ) (s : ℕ) (ω : Ω) : ℝ :=
12   peelingSum X s ω / (s : ℝ) + sqrt (4 / (s : ℝ) * logPlus (1 / ((s : ℝ)*δ)))

14 /-- Positive part of the worst centered index through sample count n. -/
15 def optimismDeficit (X : ℕ → Ω → ℝ) (δ : ℝ) : ℕ → Ω → ℝ
16   | 0, _ => 0
17   | n+1, ω => max (optimismDeficit X δ n ω) (-centeredIndex X δ (n+1) ω)
18 
19 theorem optimismDeficit_nonneg (X : ℕ → Ω → ℝ) (δ : ℝ) (n : ℕ) (ω : Ω) :
20     0 ≤ optimismDeficit X δ n ω := by
21   induction n with
22   | zero => exact le_rfl
23   | succ n ih => exact ih.trans (le_max_left _ _)
24 
25 theorem le_optimismDeficit_iff (X : ℕ → Ω → ℝ) (δ gap : ℝ) (hg : 0 < gap)
26     (n : ℕ) (ω : Ω) : gap ≤ optimismDeficit X δ n ω ↔
27       ∃ s : ℕ, 0 < s ∧ s ≤ n ∧ centeredIndex X δ s ω + gap ≤ 0 := by
28   induction n with
29   | zero => simp [optimismDeficit, not_le.mpr hg]
30   | succ n ih =>
31     rw [optimismDeficit, le_max_iff, ih]
32     constructor
33     · rintro (⟨s, hs, hsn, hb⟩ | hb)
34       · exact ⟨s, hs, Nat.le_succ_of_le hsn, hb⟩
35       · exact ⟨n+1, by omega, le_rfl, by linarith⟩
36     · rintro ⟨s, hs, hsn, hb⟩
37       by_cases h : s ≤ n
38       · exact Or.inl ⟨s, hs, h, hb⟩
39       · have heq : s = n+1 := by omega
40         subst s
41         exact Or.inr (by linarith)

43 theorem stronglyMeasurable_centeredIndex (X : ℕ → Ω → ℝ)
44     (hX : ∀ i, StronglyMeasurable (X i)) (δ : ℝ) (s : ℕ) :
45     StronglyMeasurable (centeredIndex X δ s) := by
46   unfold centeredIndex peelingSum
47   have hsum : StronglyMeasurable (fun ω => ∑ j ∈ range s, X (j+1) ω) := by
48     convert (Finset.stronglyMeasurable_sum (range s) (fun i _ => hX (i+1))) using 1
49     ext ω
50     simp
51   exact (hsum.div stronglyMeasurable_const).add stronglyMeasurable_const

53 theorem integrable_centeredIndex [IsFiniteMeasure μ] (X : ℕ → Ω → ℝ)
54     (hX : ∀ i, Integrable (X i) μ) (δ : ℝ) (s : ℕ) :
55     Integrable (centeredIndex X δ s) μ := by
56   unfold centeredIndex peelingSum
57   exact ((integrable_finset_sum _ (fun i _ => hX (i+1))).div_const _).add
58     (integrable_const _)

60 theorem stronglyMeasurable_optimismDeficit (X : ℕ → Ω → ℝ)
61     (hX : ∀ i, StronglyMeasurable (X i)) (δ : ℝ) (n : ℕ) :
62     StronglyMeasurable (optimismDeficit X δ n) := by
63   induction n with
64   | zero => exact stronglyMeasurable_const
65   | succ n ih => exact ih.sup (stronglyMeasurable_centeredIndex X hX δ (n+1)).neg

67 theorem integrable_optimismDeficit [IsFiniteMeasure μ] (X : ℕ → Ω → ℝ)
68     (hX : ∀ i, Integrable (X i) μ) (δ : ℝ) (n : ℕ) :
69     Integrable (optimismDeficit X δ n) μ := by
70   induction n with
71   | zero => exact integrable_const _
72   | succ n ih => exact ih.sup (integrable_centeredIndex X hX δ (n+1)).neg

74 theorem measure_optimismDeficit_ge_le [IsProbabilityMeasure μ] (X : ℕ → Ω → ℝ)
75     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
76     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
77     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
78     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) (n : ℕ) :
79     μ {ω | gap ≤ optimismDeficit X δ n ω} ≤ ENNReal.ofReal (15*δ/gap^2) := by
80   apply (measure_mono (show {ω | gap ≤ optimismDeficit X δ n ω} ⊆
81     meanBadEvent X δ gap from ?_)).trans
82     (measure_meanBadEvent_le_fifteen X hXm hind hmean hsubG δ gap hδ hg)
83   intro ω hω
84   obtain ⟨s, hs, _, hb⟩ := (le_optimismDeficit_iff X δ gap hg n ω).mp hω
85   exact ⟨s, hs, hb⟩

87 /-- Layer-cake identity with integrability derived from the coordinate MGF contracts. -/
88 theorem integral_optimismDeficit_eq_integral_tail [IsProbabilityMeasure μ]
89     (X : ℕ → Ω → ℝ) (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
90     (δ : ℝ) (n : ℕ) :
91     ∫ ω, optimismDeficit X δ n ω ∂μ =
92       ∫ gap in Set.Ioi 0, μ.real {ω | gap ≤ optimismDeficit X δ n ω} :=
93   (integrable_optimismDeficit X (fun i => (hsubG i).integrable) δ n).integral_eq_integral_meas_le
94     (Filter.Eventually.of_forall (optimismDeficit_nonneg X δ n))

96 /-- Source expected optimism-deficit bound, derived from the uniform tail. -/
97 theorem integral_optimismDeficit_le_two_sqrt [IsProbabilityMeasure μ]
98     (X : ℕ → Ω → ℝ) (hXm : ∀ i, StronglyMeasurable (X i))
99     (hind : iIndepFun X μ) (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
100     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
101     (δ : ℝ) (hδ : 0 < δ) (n : ℕ) :
102     ∫ ω, optimismDeficit X δ n ω ∂μ ≤ 2*sqrt (15*δ) := by
103   rw [integral_optimismDeficit_eq_integral_tail X hsubG δ n]
104   have hm : Antitone (fun gap => μ.real {ω | gap ≤ optimismDeficit X δ n ω}) := by
105     intro a b hab
106     exact measureReal_mono (fun ω hω => hab.trans hω)
107   apply Concentration.integral_positive_tail_le_two_sqrt _ hm.measurable
108     (fun _ => measureReal_nonneg) (fun _ => measureReal_le_one) (15*δ) (by positivity)
109   intro gap hg
110   have h := ENNReal.toReal_mono ENNReal.ofReal_ne_top
111     (measure_optimismDeficit_ge_le X hXm hind hmean hsubG δ gap hδ hg n)
112   simpa only [ENNReal.toReal_ofReal (by positivity : 0 ≤ 15*δ/gap^2)] using h

114 /-- The printed 16*sqrt(n*k) optimism contribution in Theorem 9.1. -/
115 theorem twice_horizon_mul_integral_optimismDeficit_le [IsProbabilityMeasure μ]
116     (X : ℕ → Ω → ℝ) (hXm : ∀ i, StronglyMeasurable (X i))
117     (hind : iIndepFun X μ) (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
118     (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
119     (n k : ℕ) (hn : 0 < n) (hk : 0 < k) :
120     2*(n : ℝ)*(∫ ω, optimismDeficit X ((k : ℝ)/n) n ω ∂μ) ≤
121       16*sqrt ((n : ℝ)*k) := by
122   have hnR : 0 < (n : ℝ) := Nat.cast_pos.mpr hn
123   have hkR : 0 < (k : ℝ) := Nat.cast_pos.mpr hk
124   have h := mul_le_mul_of_nonneg_left
125     (integral_optimismDeficit_le_two_sqrt X hXm hind hmean hsubG
126       ((k : ℝ)/n) (div_pos hkR hnR) n) (by positivity : 0 ≤ 2*(n : ℝ))
127   have heq : (n : ℝ)*sqrt (15*((k : ℝ)/n)) = sqrt 15*sqrt ((n : ℝ)*k) := by
128     calc
129       _ = sqrt ((n : ℝ)^2)*sqrt (15*((k : ℝ)/n)) := by rw [sqrt_sq hnR.le]
130       _ = sqrt ((n : ℝ)^2*(15*((k : ℝ)/n))) := (sqrt_mul (sq_nonneg _) _).symm
131       _ = sqrt (15*((n : ℝ)*k)) := by congr 1; field_simp
132       _ = _ := sqrt_mul (by norm_num) _
133   have h15 : sqrt (15 : ℝ) ≤ 4 := (sqrt_le_iff).mpr ⟨by norm_num, by norm_num⟩
134   nlinarith [sqrt_nonneg ((n : ℝ)*k),
135     mul_le_mul_of_nonneg_right h15 (sqrt_nonneg ((n : ℝ)*k))]
