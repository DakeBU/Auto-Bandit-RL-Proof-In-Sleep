4 /-!
5 # Practical finite-arm reward laws for one-policy arm-stream UCB
6 
7 This module packages stationary Real-valued arm laws as a Mathlib kernel and
8 instantiates the canonical arm-stream expected-consistency theorem. The final
9 bounded-law endpoint keeps one recursive policy and one product measure fixed
10 across all horizons.
11 -/

20 /-- A finite family of Real reward laws, viewed as an arm-indexed kernel. -/
21 noncomputable def finiteArmRealRewardKernel {K : Nat}
22     (armLaw : Fin K -> Measure Real) : Kernel (Fin K) Real :=
23   Kernel.ofFunOfCountable armLaw
24 
25 @[simp]
26 theorem finiteArmRealRewardKernel_apply {K : Nat}
27     (armLaw : Fin K -> Measure Real) (arm : Fin K) :
28     finiteArmRealRewardKernel armLaw arm = armLaw arm :=
29   rfl

31 /-- Pointwise probability laws make the finite-arm kernel Markov. -/
32 theorem finiteArmRealRewardKernel_isMarkov {K : Nat}
33     (armLaw : Fin K -> Measure Real)
34     (hprob : forall arm, IsProbabilityMeasure (armLaw arm)) :
35     IsMarkovKernel (finiteArmRealRewardKernel armLaw) where
36   isProbabilityMeasure arm := by
37     simpa using hprob arm

39 /--
40 Expected regret of the canonical one-policy arm-stream process built from
41 direct finite-arm sub-Gaussian laws. The common tuning proxy is the padded
42 finite maximum of the genuine armwise proxies.
43 -/
44 noncomputable def armStreamFiniteArmSubgaussianExpectedRegret
45     {K : Nat} (hK : 0 < K)
46     (armLaw : Fin K -> Measure Real)
47     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
48     (varianceProxy : Fin K -> NNReal) (n : Nat) : Real :=
49   let nu := finiteArmRealRewardKernel armLaw
50   let sigma2 := Concentration.finiteArmPositiveVarianceProxy varianceProxy
51   letI : IsMarkovKernel nu := finiteArmRealRewardKernel_isMarkov armLaw hprob
52   armStreamExpectedRegret hK sigma2 nu n

54 /-- Direct finite-arm sub-Gaussian laws satisfy the fixed logarithmic envelope. -/
55 theorem armStreamFiniteArmSubgaussianExpectedRegret_nonneg_and_le
56     {K : Nat} (hK : 0 < K)
57     (armLaw : Fin K -> Measure Real)
58     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
59     (varianceProxy : Fin K -> NNReal)
60     (hsubG : forall arm,
61       HasSubgaussianMGF
62         (fun reward : Real => reward - integral (armLaw arm) id)
63         (varianceProxy arm) (armLaw arm))
64     (n : Nat) :
65     let nu := finiteArmRealRewardKernel armLaw
66     let sigma2 := Concentration.finiteArmPositiveVarianceProxy varianceProxy
67     0 <= armStreamFiniteArmSubgaussianExpectedRegret
68         hK armLaw hprob varianceProxy n /\
69       armStreamFiniteArmSubgaussianExpectedRegret
70           hK armLaw hprob varianceProxy n <=
71         armStreamAsymptoticModelCoefficient nu sigma2 *
72           (1 + Real.log ((n + 1 : Nat) : Real)) := by
73   let nu := finiteArmRealRewardKernel armLaw
74   let sigma2 := Concentration.finiteArmPositiveVarianceProxy varianceProxy
75   letI : IsMarkovKernel nu := finiteArmRealRewardKernel_isMarkov armLaw hprob
76   have hsigma2 : sigma2 ≠ 0 := by
77     have hpos :=
78       Concentration.finiteArmPositiveVarianceProxy_pos varianceProxy
79     exact_mod_cast (ne_of_gt hpos)
80   have hsubGCommon : forall arm : Fin K,
81       HasSubgaussianMGF
82         (fun reward => reward - realKernelMean nu arm)
83         sigma2 (nu arm) := by
84     intro arm
85     simpa [nu, realKernelMean] using
86       (ConditionalExpectationReward.hasSubgaussianMGF_mono_varianceProxy
87         (Concentration.varianceProxy_le_finiteArmPositiveVarianceProxy
88           varianceProxy arm)
89         (hsubG arm))
90   simpa [armStreamFiniteArmSubgaussianExpectedRegret, nu, sigma2] using
91     (armStreamExpectedRegret_nonneg_and_le
92       hK sigma2 nu hsigma2 hsubGCommon n)

153 /-- Common-bounded finite-arm laws satisfy the fixed logarithmic envelope. -/
154 theorem armStreamBoundedFiniteArmExpectedRegret_nonneg_and_le
155     {K : Nat} (hK : 0 < K)
156     (armLaw : Fin K -> Measure Real)
157     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
158     (lo hi : Real)
159     (hbound : forall arm,
160       Filter.Eventually
161         (fun reward : Real => Set.Icc lo hi reward) (ae (armLaw arm)))
162     (n : Nat) :
163     let nu := finiteArmRealRewardKernel armLaw
164     let sigma2 := Concentration.finiteArmPositiveVarianceProxy
165       (fun _ : Fin K => Concentration.intervalVarianceProxy lo hi)
166     0 <= armStreamBoundedFiniteArmExpectedRegret
167         hK armLaw hprob lo hi n /\
168       armStreamBoundedFiniteArmExpectedRegret hK armLaw hprob lo hi n <=
169         armStreamAsymptoticModelCoefficient nu sigma2 *
170           (1 + Real.log ((n + 1 : Nat) : Real)) := by
171   have hsubG : forall arm : Fin K,
172       HasSubgaussianMGF
173         (fun reward : Real => reward - integral (armLaw arm) id)
174         (Concentration.intervalVarianceProxy lo hi) (armLaw arm) := by
175     intro arm
176     letI : IsProbabilityMeasure (armLaw arm) := hprob arm
177     simpa using
178       (Concentration.boundedCentered_hasSubgaussianMGF_of_mem_Icc_integral_eq
