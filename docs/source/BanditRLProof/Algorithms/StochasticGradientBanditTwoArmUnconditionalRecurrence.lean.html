3 /-!
4 # Unconditional and iterated two-arm SGB recurrences
5 
6 This module integrates the tower-ready conditional-expectation recurrences on
7 the canonical trajectory and performs the finite scalar iterations used by the
8 two-arm Theorem-1 proof.
9 
10 The source-time fence is explicit.  For an inclusive Lean prefix index `n`,
11 `twoArmTrajectoryParameterZero eta n` is the source parameter
12 `theta_{1,n+2}`: trace coordinate zero has already been consumed.
13 Consequently, `twoArmSuccessProbability eta n` and
14 `twoArmFailureMass eta n` are source `p_{1,n+2}` and
15 `1-p_{1,n+2}`.  A tail horizon `N` therefore covers source rounds
16 `t = 2,...,N+1`; the normalized initial bridge adds the exact source-round
17 `t = 1` failure square `(1-1/2)^2 = 1/4`.
18 
19 No reward moment, independence, or conditional-MGF hypothesis is introduced.
20 The one-step and tail recurrences require only the existing finite prior and
21 bounded fixed-mean environment contract.  Normalizing the initial environment
22 mixture requires `IsProbabilityMeasure prior`; the paper-exact fixed-IID
23 specialization uses a Dirac prior and satisfies this automatically.  The final
24 failure-mass bound uses precisely `0 < eta` and
25 `eta * sourceC eta < Delta`.
26 
27 This module does not assemble Equation (7), control the expected terminal
28 parameter by the logarithmic forward potential, or state Theorem 1.
29 -/

109 theorem integrable_twoArmForwardPotential
110     {Env : Type v} [MeasurableSpace Env] [StandardBorelSpace Env]
111     (prior : Measure Env) [IsFiniteMeasure prior]
112     (eta : Real)
113     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
114     (mean : Fin 2 -> Real)
115     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean)
116     (n : Nat) :
117     Integrable (twoArmForwardPotential (Env := Env) eta n)
118       (twoArmTrajectoryMeasure prior eta environment) := by
119   apply Integrable.of_bound
120     (measurable_twoArmForwardPotential eta n).aestronglyMeasurable
121     (Real.exp (2 * ((n + 1 : Nat) : Real) * |eta|))
122   filter_upwards
123       [twoArmTrajectoryMeasure_prefix_rewards_abs_le_one_ae
124         prior eta environment mean contract n]
125     with sample hreward
126   rw [twoArmForwardPotential, Real.norm_eq_abs,
127     abs_of_pos (Real.exp_pos _)]
128   apply Real.exp_le_exp.mpr
129   have htheta := abs_historyParameter_zeroInitialization_le eta n
130     (twoArmEnvironmentPrefix n sample).2 0 hreward
131   calc
132     2 * twoArmTrajectoryParameterZero eta n sample <=
133         2 * |twoArmTrajectoryParameterZero eta n sample| :=
134       mul_le_mul_of_nonneg_left (le_abs_self _) (by norm_num)
135     _ <= 2 * (((n + 1 : Nat) : Real) * |eta|) :=
136       mul_le_mul_of_nonneg_left htheta (by norm_num)
137     _ = 2 * ((n + 1 : Nat) : Real) * |eta| := by ring

288 theorem twoArmForwardUnconditionalRecurrence
289     {Env : Type v} [MeasurableSpace Env] [StandardBorelSpace Env]
290     (prior : Measure Env) [IsFiniteMeasure prior]
291     (eta Delta : Real) (heta : 0 <= eta)
292     (environment : Thompson.MeasurableHistoryEnvironment Env (Fin 2) Real)
293     (mean : Fin 2 -> Real)
294     (contract : TwoArmBoundedFixedMeanEnvironmentContract environment mean)
295     (hgap : mean 0 - mean 1 = Delta) (n : Nat) :
296     integral (twoArmTrajectoryMeasure prior eta environment)
297         (twoArmForwardPotential (Env := Env) eta (n + 1)) <=
298       integral (twoArmTrajectoryMeasure prior eta environment)
299         (twoArmForwardPotential (Env := Env) eta n) +
300       2 * integral (twoArmTrajectoryMeasure prior eta environment)
301           (fun sample => twoArmSuccessProbability
302             (Env := Env) eta n sample ^ 2) *
303         (eta * Delta + eta ^ 2 * sourceC eta) := by
304   let mu := twoArmTrajectoryMeasure prior eta environment
305   have hrec :=
306     twoArmForwardTrajectorySuccessor_condExp_le_recurrenceBound
307       prior eta Delta heta environment mean contract hgap n
308   have hmono := integral_mono_ae
309     (integrable_condExp : Integrable
310       (mu[fun sample =>
311         twoArmForwardSuccessorPotential eta
312           (twoArmEnvironmentPrefix n sample).2 (twoArmNextPair n sample) |
313         twoArmPrefixSigma (Env := Env) n]) mu)
314     (integrable_twoArmForwardRecurrenceBound
315       prior eta Delta environment mean contract n) hrec
316   have htower := integral_condExp
317     (m := twoArmPrefixSigma (Env := Env) n)
318     (μ := mu)
319     (f := fun sample : Env × ((k : Nat) -> Fin 2 × Real) =>
320       twoArmForwardSuccessorPotential eta
321         (twoArmEnvironmentPrefix n sample).2 (twoArmNextPair n sample))
322     ((twoArmPrefixFiltration (Env := Env)).le n)
323   rw [htower] at hmono
324   rw [twoArmForwardSuccessor_eq_nextPotential eta n] at hmono
325   have hpotential := integrable_twoArmForwardPotential
326     prior eta environment mean contract n
327   have hprob := integrable_twoArmSuccessProbability_sq
328     prior eta environment n
329   change _ <= integral mu
330     (fun sample => twoArmForwardPotential eta n sample +
331       2 * twoArmSuccessProbability eta n sample ^ 2 *
332         (eta * Delta + eta ^ 2 * sourceC eta)) at hmono
333   rw [integral_add hpotential
334       ((hprob.const_mul 2).mul_const
335         (eta * Delta + eta ^ 2 * sourceC eta)),
336     integral_mul_const, integral_const_mul] at hmono
337   exact hmono

392 theorem twoArmScalarForwardIterate
393     (value increment : Nat -> Real)
394     (hstep : forall n, value (n + 1) <= value n + increment n) :
395     forall horizon,
396       value horizon <= value 0 + (Finset.range horizon).sum increment := by
397   intro horizon
398   induction horizon with
399   | zero => simp
400   | succ n ih =>
401       calc
402         value (n + 1) <= value n + increment n := hstep n
403         _ <= (value 0 + (Finset.range n).sum increment) + increment n :=
404           by linarith
405         _ = value 0 + (Finset.range (n + 1)).sum increment := by
406           rw [Finset.sum_range_succ]
407           ring

409 theorem twoArmScalarInverseTelescope
410     (value failure : Nat -> Real) (coefficient : Real)
411     (hstep : forall n,
412       value (n + 1) <= value n - coefficient * failure n) :
413     forall horizon,
414       coefficient * (Finset.range horizon).sum failure <=
415         value 0 - value horizon := by
416   intro horizon
417   induction horizon with
418   | zero => simp
419   | succ n ih =>
420       have hlast : coefficient * failure n <= value n - value (n + 1) := by
421         linarith [hstep n]
422       rw [Finset.sum_range_succ]
423       calc
424         coefficient * ((Finset.range n).sum failure + failure n) =
425             coefficient * (Finset.range n).sum failure +
426               coefficient * failure n := by ring
427         _ <= (value 0 - value n) + (value n - value (n + 1)) :=
428           add_le_add ih hlast
429         _ = value 0 - value (n + 1) := by ring
