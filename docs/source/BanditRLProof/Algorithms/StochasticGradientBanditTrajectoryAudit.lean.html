5 # Generated stochastic-gradient-bandit trajectory audit
6 
7 This module lifts the finite Algorithm-1 / Equation-(5) algebra in
8 `StochasticGradientBanditAudit` to the repository's canonical measurable
9 action/reward trajectory.  It constructs the recursive parameter vector,
10 packages its softmax law as a history algorithm, and identifies the generated
11 successor action and pair conditional laws.
12 
13 The learning-rate thresholds, failure events, and regret-rate conclusions of
14 the source paper remain outside this process-level audit.

29 /-- Coordinate measurability of the source softmax transform. -/
30 theorem measurable_softmaxProbability
31     [Fintype Action]
32     {History : Type*} [MeasurableSpace History]
33     (theta : History -> Action -> Real)
34     (htheta : forall action, Measurable (fun history => theta history action))
35     (action : Action) :
36     Measurable (fun history => softmaxProbability (theta history) action) := by
37   unfold softmaxProbability softmaxDenominator
38   exact (Real.measurable_exp.comp (htheta action)).div
39     (Finset.measurable_sum Finset.univ fun candidate _ =>
40       Real.measurable_exp.comp (htheta candidate))

42 /-- Measurability of one Algorithm-1 coordinate update. -/
43 theorem measurable_sourceIncrement
44     [MeasurableSpace Action] [MeasurableSingletonClass Action]
45     [DecidableEq Action]
46     {History : Type*} [MeasurableSpace History]
47     (prob : History -> Action -> Real)
48     (reward : History -> Real) (selected : History -> Action)
49     (coordinate : Action)
50     (hprob : Measurable (fun history => prob history coordinate))
51     (hreward : Measurable reward) (hselected : Measurable selected) :
52     Measurable (fun history =>
53       sourceIncrement (prob history) (reward history)
54         (selected history) coordinate) := by
55   unfold sourceIncrement
56   refine Measurable.ite ?_ ((measurable_const.sub hprob).mul hreward)
57     ((hprob.mul hreward).neg)
58   simpa only [Set.mem_setOf_eq] using
59     hselected (measurableSet_singleton coordinate)

61 /--
62 The Algorithm-1 parameter vector after an inclusive finite action/reward
63 history.  At round zero the first observed pair updates `initialTheta`; every
64 successor uses the softmax vector generated by the preceding inclusive prefix.
65 -/
66 noncomputable def historyParameter
67     [Fintype Action] [DecidableEq Action]
68     (initialTheta : Action -> Real) (eta : Real) :
69     (n : Nat) -> History.FinitePairHistory Action Real n -> Action -> Real
70   | 0, history, coordinate =>
71       initialTheta coordinate + eta *
72         sourceIncrement (softmaxProbability initialTheta)
73           (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).2
74           (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 coordinate
75   | n + 1, history, coordinate =>
76       let previous := Exp3.previousPairHistory history
77       historyParameter initialTheta eta n previous coordinate + eta *
78         sourceIncrement
79           (softmaxProbability (historyParameter initialTheta eta n previous))
80           (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).2
81           (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).1 coordinate

116 /-- Every recursive parameter coordinate is measurable in the finite history. -/
117 theorem measurable_historyParameter
118     (initialTheta : Action -> Real) (eta : Real) :
119     forall n coordinate,
120       Measurable (fun history : History.FinitePairHistory Action Real n =>
121         historyParameter initialTheta eta n history coordinate) := by
122   intro n
123   induction n with
124   | zero =>
125       intro coordinate
126       let zeroIndex : Finset.Iic 0 :=
127         ⟨0, Finset.mem_Iic.mpr le_rfl⟩
128       have hselected : Measurable
129           (fun history : History.FinitePairHistory Action Real 0 =>
130             (history zeroIndex).1) :=
131         measurable_fst.comp (measurable_pi_apply zeroIndex)
132       have hreward : Measurable
133           (fun history : History.FinitePairHistory Action Real 0 =>
134             (history zeroIndex).2) :=
135         measurable_snd.comp (measurable_pi_apply zeroIndex)
136       exact measurable_const.add (measurable_const.mul
137         (measurable_sourceIncrement
138           (fun _history : History.FinitePairHistory Action Real 0 =>
139             softmaxProbability initialTheta)
140           (fun history => (history zeroIndex).2)
141           (fun history => (history zeroIndex).1)
142           coordinate measurable_const hreward hselected))
143   | succ n ih =>
144       intro coordinate
145       let lastIndex : Finset.Iic (n + 1) :=
146         ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩
147       have hprevious : Measurable
148           (Exp3.previousPairHistory (Action := Action) (n := n)) :=
149         Exp3.measurable_previousPairHistory
150       have hstate : Measurable
151           (fun history : History.FinitePairHistory Action Real (n + 1) =>
152             historyParameter initialTheta eta n
153               (Exp3.previousPairHistory history) coordinate) :=
154         (ih coordinate).comp hprevious
155       have hprob : Measurable
156           (fun history : History.FinitePairHistory Action Real (n + 1) =>
157             softmaxProbability
158               (historyParameter initialTheta eta n
159                 (Exp3.previousPairHistory history)) coordinate) :=
160         (measurable_softmaxProbability
161           (fun previous : History.FinitePairHistory Action Real n =>
162             historyParameter initialTheta eta n previous)
163           ih coordinate).comp hprevious
164       have hselected : Measurable
165           (fun history : History.FinitePairHistory Action Real (n + 1) =>
166             (history lastIndex).1) :=
167         measurable_fst.comp (measurable_pi_apply lastIndex)
168       have hreward : Measurable
169           (fun history : History.FinitePairHistory Action Real (n + 1) =>
170             (history lastIndex).2) :=
171         measurable_snd.comp (measurable_pi_apply lastIndex)
172       exact hstate.add (measurable_const.mul
173         (measurable_sourceIncrement
174           (fun history => softmaxProbability
175             (historyParameter initialTheta eta n
176               (Exp3.previousPairHistory history)))
177           (fun history => (history lastIndex).2)
178           (fun history => (history lastIndex).1)
179           coordinate hprob hreward hselected))

235 /-- Equation (5) for the generated initial action/reward pair law. -/
236 theorem integral_measurableEnvironmentInitialPairKernel_sourceIncrement_eq_expectedSourceIncrement
237     {Env : Type v} [MeasurableSpace Env]
238     (initialTheta : Action -> Real) (eta : Real)
239     (environment : Thompson.MeasurableHistoryEnvironment Env Action Real)
240     (env : Env) (mean : Action -> Real) (coordinate : Action)
241     (hIntegrable : Integrable
242       (fun pair : Action × Real =>
243         sourceIncrement (softmaxProbability initialTheta)
244           pair.2 pair.1 coordinate)
245       (Thompson.measurableEnvironmentInitialPairKernel
246         (historyAlgorithm initialTheta eta) environment env))
247     (hmean : forall selected,
248       integral (environment.initialFeedback (env, selected)) id =
249         mean selected) :
250     integral
251         (Thompson.measurableEnvironmentInitialPairKernel
252           (historyAlgorithm initialTheta eta) environment env)
253         (fun pair : Action × Real =>
254           sourceIncrement (softmaxProbability initialTheta)
255             pair.2 pair.1 coordinate) =
256       expectedSourceIncrement (softmaxProbability initialTheta)
257         mean coordinate := by
258   rw [Thompson.measurableEnvironmentInitialPairKernel_apply] at hIntegrable ⊢
259   let prob := softmaxProbability initialTheta
260   have hmeanAt (selected : Action) :
261       (∫ reward : Real, reward
262         ∂(environment.at env).initialFeedback selected) = mean selected := by
263     simpa only [id_eq, Thompson.MeasurableHistoryEnvironment.at,
264       ProbabilityTheory.Kernel.comap_apply] using hmean selected
265   have hinner : forall selected,
266       integral ((environment.at env).initialFeedback selected)
267           (fun reward => sourceIncrement prob reward selected coordinate) =
268         sourceIncrement prob (mean selected) selected coordinate := by
269     intro selected
270     by_cases hselected : selected = coordinate
271     · subst selected
272       simp only [sourceIncrement, ite_true]
273       rw [integral_const_mul, hmeanAt]
274     · simp only [sourceIncrement, if_neg hselected]
275       rw [integral_neg, integral_const_mul, hmeanAt]
276   calc
277     integral
278         ((historyAlgorithm initialTheta eta).initialAction ⊗ₘ
279           (environment.at env).initialFeedback)
280         (fun pair : Action × Real =>
281           sourceIncrement prob pair.2 pair.1 coordinate) =
282         integral (historyAlgorithm initialTheta eta).initialAction
283           (fun selected =>
284             integral ((environment.at env).initialFeedback selected)
285               (fun reward =>
286                 sourceIncrement prob reward selected coordinate)) :=
287       Measure.integral_compProd hIntegrable
288     _ = integral (historyAlgorithm initialTheta eta).initialAction
289           (fun selected =>
290             sourceIncrement prob (mean selected) selected coordinate) := by
291       apply integral_congr_ae
292       exact Filter.Eventually.of_forall hinner
293     _ = ∑ selected,
294           prob selected *
295             sourceIncrement prob (mean selected) selected coordinate := by
296       change integral
297           (Exp3.finiteActionMeasure (Finset.univ : Finset Action) prob)
298           (fun selected =>
299             sourceIncrement prob (mean selected) selected coordinate) = _
300       rw [Exp3.integral_finiteActionMeasure_eq_sum
301         (Finset.univ : Finset Action) prob
302         (softmaxFiniteActionDistribution initialTheta)]
303     _ = expectedSourceIncrement prob mean coordinate := by
304       rfl
