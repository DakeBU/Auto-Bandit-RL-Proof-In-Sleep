6 # Canonical UCB expected regret for context-dependent sub-Gaussian reward kernels
7 
8 The reward distribution and its pointwise sub-Gaussian proxy may vary with
9 context and action. Arm means remain stationary, and a positive uniform proxy
10 ceiling is supplied for the UCB confidence width.
11 -/

20 /--
21 Canonical Real expected pseudo-regret bound for a context-dependent Markov
22 reward kernel with stationary arm means and direct centered sub-Gaussian MGF
23 witnesses.
24 
25 The centered kernel law, selected reward law, trajectory law, and finite-horizon
26 integrability are constructed internally. The caller supplies a positive common
27 ceiling because an arbitrary measurable context space has no finite maximum
28 operation for the pointwise variance proxies.
29 -/
30 theorem integral_real_pseudoRegret_selectedPolicySuccessorGeneratedUCBRegretAction_le_textbookGapSum_trajMeasure_contextDependentSubgaussianRewardKernel
31     {Context : Type} [MeasurableSpace Context]
32     {K : Nat}
33     (model : FiniteBanditModel K)
34     (mu0 : Measure Rat) [IsProbabilityMeasure mu0]
35     (rewardKernel : RewardKernel.MarkovRewardKernel
36       (Context × Fin K) Rat)
37     (context : (n : Nat) -> History.FiniteRewardHistory Rat n -> Context)
38     (hcontext : forall n, Measurable (context n))
39     (varianceProxy : Context -> Fin K -> NNReal)
40     (sigma2 : NNReal)
41     (hsigma2 : 0 < ((sigma2 : NNReal) : Real))
42     (hvariance : forall ctx arm, varianceProxy ctx arm <= sigma2)
43     (hmean : forall ctx arm,
44       integral (RewardKernel.selectedMeasure rewardKernel ctx arm)
45           (fun reward : Rat => ((reward : Rat) : Real)) =
46         ((model.mean arm : Rat) : Real))
47     (hsubG : forall ctx arm,
48       HasSubgaussianMGF
49         (fun reward : Rat =>
50           (((reward - model.mean arm : Rat) : Real)))
51         (varianceProxy ctx arm)
52         (RewardKernel.selectedMeasure rewardKernel ctx arm))
53     (defaultAction : Fin K)
54     (T : Nat) (hT : 0 < T)
55     (delta : Real) (hdelta : 0 < delta) :
56     MeasureTheory.integral
57         (selectedPolicySuccessorRewardTrajMeasure model.hK
58           mu0 rewardKernel context hcontext sigma2 T delta defaultAction)
59         (fun trajectory : RewardTrace Rat =>
60           ((pseudoRegret model
61             (selectedPolicySuccessorGeneratedUCBRegretAction
62               model.hK sigma2 T delta defaultAction
63               (fun y : RewardTrace Rat => y) trajectory)
64             T : Rat) : Real)) <=
65       ((Finset.univ : Finset (Fin K)).filter (fun arm =>
66         0 < (((model.gap arm : Rat) : Real)))).sum (fun arm =>
67           selectedPolicySuccessorTextbookGapBudget K sigma2 T delta
68               (((model.gap arm : Rat) : Real)) +
69             (((model.gap arm : Rat) : Real)) * ((T : Real) * delta)) := by
70   have hmeanMeas :
71       Measurable (fun pair : Context × Fin K => model.mean pair.2) :=
72     (measurable_of_countable model.mean).comp measurable_snd
73   have hkernel :
74       RewardKernel.CenteredRewardKernelLaw rewardKernel
75         (fun _ arm => model.mean arm) varianceProxy := by
76     exact RewardKernel.centeredRewardKernelLaw_of_hasSubgaussianMGF
77       rewardKernel (fun _ arm => model.mean arm) varianceProxy hmean hsubG
78   exact
79     integral_real_pseudoRegret_selectedPolicySuccessorGeneratedUCBRegretAction_le_textbookGapSum_trajMeasure_centeredKernel
80       (model := model)
81       (mu0 := mu0)
82       (rewardKernel := rewardKernel)
83       (context := context)
84       (mean := fun _ arm => model.mean arm)
85       (varianceProxy := varianceProxy)
86       (defaultAction := defaultAction)
87       (sigma2 := sigma2)
88       (hcontext := hcontext)
89       (hmean := hmeanMeas)
90       (hkernel := hkernel)
91       (T := T)
92       (hT := hT)
93       (hsigma2 := hsigma2)
94       (delta := delta)
95       (hdelta := hdelta)

97       (harmMean := fun _ _ _ => rfl)
