6 # ETC phase-switching trace count lemmas
7 
8 This module contains deterministic pull-count facts for the fixed-commit ETC
9 trace.  It stays below regret, empirical commit selection, probability, and
10 concentration.
11 -/

15 /--
16 On any prefix contained inside the configured exploration horizon, the
17 fixed-commit ETC trace has the same pull counts as the pure round-robin
18 exploration trace.
19 
20 This is the `ETC-ACTION-WITH-COMMIT-EXPLORE-PREFIX-PULLCOUNT` project-local
21 trace/count transfer leaf.
22 -/
23 theorem ETC.pullCount_actionWithCommit_eq_pullCount_exploreArm_of_le
24     {K : Nat} (spec : ETC.Spec K) (commitArm a : Fin K)
25     (n : Nat) (hn : n <= spec.explorationPulls * K) :
26     pullCount (ETC.actionWithCommit spec commitArm) a n =
27       pullCount (ETC.exploreArm spec) a n := by
28   revert hn
29   induction n with
30   | zero =>
31       intro _hn
32       simp [pullCount]
33   | succ n ih =>
34       intro hn
35       have hn_le : n <= spec.explorationPulls * K :=
36         Nat.le_trans (Nat.le_succ n) hn
37       have hn_lt : n < spec.explorationPulls * K :=
38         Nat.lt_of_succ_le hn
39       have hact :
40           ETC.actionWithCommit spec commitArm n = ETC.exploreArm spec n :=
41         ETC.actionWithCommit_eq_exploreArm_of_lt spec commitArm hn_lt
42       rw [pullCount_succ, pullCount_succ]
43       rw [ih hn_le]
44       rw [hact]

46 /--
47 At the configured exploration horizon, the fixed-commit ETC trace has pulled
48 each arm exactly `spec.explorationPulls` times.
49 
50 This is the `ETC-ACTION-WITH-COMMIT-EXPLORATION-HORIZON-COUNT` project-local
51 trace/count adapter.
52 -/
53 theorem ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq
54     {K : Nat} (spec : ETC.Spec K) (commitArm a : Fin K) :
55     pullCount (ETC.actionWithCommit spec commitArm) a
56         (spec.explorationPulls * K) =
57       spec.explorationPulls := by
58   calc
59     pullCount (ETC.actionWithCommit spec commitArm) a
60         (spec.explorationPulls * K)
61         = pullCount (ETC.exploreArm spec) a (spec.explorationPulls * K) := by
62             exact
63               ETC.pullCount_actionWithCommit_eq_pullCount_exploreArm_of_le
64                 spec commitArm a (spec.explorationPulls * K) (Nat.le_refl _)
65     _ = spec.explorationPulls := by
66             exact ETC.pullCount_exploreArm_explorationPulls_mul_K_eq spec a

69 At the configured exploration horizon, every arm in the fixed-commit ETC trace
70 has a positive pull count whenever the configured number of exploration pulls
71 is positive.
72 
73 This is the first denominator-positivity leaf for future empirical-mean
74 construction.  It stays purely deterministic and Nat-valued.

77     {K : Nat} (spec : ETC.Spec K) (commitArm a : Fin K)
78     (hexplorationPulls_pos : 0 < spec.explorationPulls) :
79     0 < pullCount (ETC.actionWithCommit spec commitArm) a
80       (spec.explorationPulls * K) := by
81   have hcount :
82       pullCount (ETC.actionWithCommit spec commitArm) a

128 After the configured exploration horizon, one step of the fixed-commit ETC
129 trace updates pull counts according to whether the queried arm is the commit
130 arm.
131 
132 This is the `ETC-ACTION-WITH-COMMIT-POST-COMMIT-SUCC-COUNT` project-local

135 theorem ETC.pullCount_actionWithCommit_succ_eq_add_if_commitArm_of_ge
136     {K : Nat} (spec : ETC.Spec K) (commitArm a : Fin K) {t : Nat}
137     (ht : spec.explorationPulls * K <= t) :
138     pullCount (ETC.actionWithCommit spec commitArm) a (Nat.succ t) =
139       pullCount (ETC.actionWithCommit spec commitArm) a t +
140         if commitArm = a then 1 else 0 := by
141   rw [pullCount_succ]
142   have hact :
143       ETC.actionWithCommit spec commitArm t = commitArm :=
144     ETC.actionWithCommit_eq_commitArm_of_ge spec commitArm ht
145   simp [hact]

147 /--
148 After the configured exploration horizon, the fixed-commit ETC trace has a
149 closed-form pull count: the commit arm receives every suffix pull, while all
150 other arms keep their exploration-horizon count.
151 
152 This is the `ETC-ACTION-WITH-COMMIT-SUFFIX-COUNT` project-local trace/count
153 closed-form leaf.
154 -/
155 theorem ETC.pullCount_actionWithCommit_explorationPulls_mul_K_add_eq
156     {K : Nat} (spec : ETC.Spec K) (commitArm a : Fin K) (r : Nat) :
157     pullCount (ETC.actionWithCommit spec commitArm) a
158         (spec.explorationPulls * K + r) =
159       spec.explorationPulls + (if commitArm = a then r else 0) := by
160   induction r with
161   | zero =>
162       simp [ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq]
163   | succ r ih =>
164       have hge :
165           spec.explorationPulls * K <= spec.explorationPulls * K + r :=
166         Nat.le_add_right (spec.explorationPulls * K) r
167       have hstep :
168           pullCount (ETC.actionWithCommit spec commitArm) a
169               (Nat.succ (spec.explorationPulls * K + r)) =
170             pullCount (ETC.actionWithCommit spec commitArm) a
171                 (spec.explorationPulls * K + r) +
172               if commitArm = a then 1 else 0 :=
173         ETC.pullCount_actionWithCommit_succ_eq_add_if_commitArm_of_ge
174           (spec := spec)
175           (commitArm := commitArm)
176           (a := a)
177           (t := spec.explorationPulls * K + r)
178           hge
179       rw [Nat.add_succ]
180       rw [hstep]
181       rw [ih]
182       by_cases h : commitArm = a
183       · simp [h, Nat.add_assoc]
184       · simp [h]

194     {K : Nat} (spec : ETC.Spec K) {commitArm a : Fin K}
195     (hne : commitArm ≠ a) (r : Nat) :
196     pullCount (ETC.actionWithCommit spec commitArm) a
