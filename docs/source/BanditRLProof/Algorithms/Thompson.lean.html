40 /--
41 Source contract for the Thompson probability-matching identity.
42 
43 The ledger records the exact law expected from a Thompson action sampler: its
44 action kernel at a history agrees on every measurable action event with the
45 posterior distribution pushed forward by the environment-to-best-action map.
46 It is a contract surface, not a Bayes-rule proof or posterior-sampler
47 construction.
48 -/
49 structure PosteriorActionIdentityLedger
50     (History : Type u) (Env : Type v) (Action : Type w)
51     [MeasurableSpace History] [MeasurableSpace Env] [MeasurableSpace Action]
52     where
53   posterior : PosteriorKernel.MarkovPosteriorKernel History Env
54   actionKernel : ProbabilityTheory.Kernel History Action
55   actionKernel_isMarkov : ProbabilityTheory.IsMarkovKernel actionKernel
56   bestAction : Env -> Action
57   bestAction_measurable : Measurable bestAction
58   actionKernel_eq_posteriorBest :
59     forall (history : History) {event : Set Action},
60       MeasurableSet event ->
61         actionKernel history event =
62           Measure.map bestAction (posterior.kernel history) event
63 
64 /--
65 Any best-action selector out of a countable singleton-measurable environment
66 space is measurable.
67 
68 This is the regularity wrapper needed by finite or countable posterior model
69 spaces before constructing a Thompson posterior-action identity ledger.
70 -/
71 theorem bestAction_measurable_of_countable_env
72     {Env : Type v} {Action : Type w}
73     [MeasurableSpace Env] [MeasurableSingletonClass Env] [Countable Env]
74     [MeasurableSpace Action]
75     (bestAction : Env -> Action) :
76     Measurable bestAction :=
77   measurable_of_countable bestAction

112 /--
113 Build the Thompson action ledger directly by mapping a posterior kernel through
114 a measurable best-action selector.
115 -/
116 noncomputable def ofPosteriorMap
117     (posterior : PosteriorKernel.MarkovPosteriorKernel History Env)
118     (bestAction : Env -> Action) (hbestAction : Measurable bestAction) :
119     PosteriorActionIdentityLedger History Env Action where
120   posterior := posterior
121   actionKernel := posterior.kernel.map bestAction
122   actionKernel_isMarkov :=
123     ProbabilityTheory.Kernel.IsMarkovKernel.map posterior.kernel hbestAction
124   bestAction := bestAction
125   bestAction_measurable := hbestAction
126   actionKernel_eq_posteriorBest := by
127     intro history event hevent
128     rw [ProbabilityTheory.Kernel.map_apply _ hbestAction]
129 
130 /-- The event-level ledger identity is equality of the two Markov kernels. -/
131 theorem actionKernel_eq_posterior_map
132     (ledger : PosteriorActionIdentityLedger History Env Action) :
133     ledger.actionKernel = ledger.posterior.kernel.map ledger.bestAction := by
134   ext history event hevent
135   rw [ProbabilityTheory.Kernel.map_apply _ ledger.bestAction_measurable]
136   exact ledger.actionKernel_eq_posteriorBest history hevent
137 
138 /--
139 Event-level Thompson probability matching from the packaged ledger.
140 -/
141 theorem actionKernel_apply_eq_posteriorBest_map
142     (ledger : PosteriorActionIdentityLedger History Env Action)
143     (history : History) {event : Set Action}
144     (hevent : MeasurableSet event) :
145     ledger.actionKernel history event =
146       Measure.map ledger.bestAction (ledger.posterior.kernel history) event :=
147   ledger.actionKernel_eq_posteriorBest history hevent

149 /--
150 Singleton form of the posterior action identity.
151 
152 For discrete or singleton-measurable action spaces, the Thompson probability of
153 choosing `action` is the posterior probability that `action` is the best action.
154 -/
155 theorem actionKernel_apply_singleton_eq_posteriorBest_preimage
156     [MeasurableSingletonClass Action]
157     (ledger : PosteriorActionIdentityLedger History Env Action)
158     (history : History) (action : Action) :
159     ledger.actionKernel history ({action} : Set Action) =
160       ledger.posterior.kernel history {env : Env | ledger.bestAction env = action} := by
161   rw [actionKernel_apply_eq_posteriorBest_map ledger history
162     (MeasurableSet.singleton action)]
163   rw [Measure.map_apply ledger.bestAction_measurable
164     (MeasurableSet.singleton action)]
165   rfl

169 /--
170 Source fields needed for the Thompson posterior-action conditional-law theorem.
171 
172 The first law says the process samples its next action from the ledger action
173 kernel. The second identifies the ledger posterior with the conditional law of
174 the latent environment given the observed history. These are the two law
175 surfaces used by the pinned LML proof.
176 -/
177 structure BayesianPosteriorActionSource
178     {Omega : Type u} (History : Type v) (Env : Type w) (Action : Type*)
179     [MeasurableSpace Omega] [MeasurableSpace History]
180     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
181     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
182     (mu : Measure Omega) [IsFiniteMeasure mu]
183     (env : Omega -> Env) (history : Omega -> History)
184     (nextAction : Omega -> Action)
185     (ledger : PosteriorActionIdentityLedger History Env Action) : Prop where
186   measurable_env : Measurable env
187   measurable_history : Measurable history
188   measurable_nextAction : Measurable nextAction
189   hasCondDistrib_actionKernel :
190     ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
191       ledger.actionKernel
192   posterior_eq_condDistrib_env :
193     ledger.posterior.kernel =ᵐ[mu.map history]
194       ProbabilityTheory.condDistrib env history mu
195 
196 /--
197 Thompson probability matching in Mathlib `condDistrib` form.
198 
199 Mapping the posterior-kernel equality through `bestAction` identifies the
200 ledger action kernel with the mapped environment conditional law. Mathlib's
201 `condDistrib_comp` then identifies that map with the conditional law of the
202 random best action itself.
203 -/
204 theorem condDistrib_action_ae_eq_bestAction_of_bayesianPosteriorActionSource
205     {Omega : Type u} {History : Type v} {Env : Type w} {Action : Type*}
206     [MeasurableSpace Omega] [MeasurableSpace History]
207     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
208     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
209     (mu : Measure Omega) [IsFiniteMeasure mu]
210     (env : Omega -> Env) (history : Omega -> History)
211     (nextAction : Omega -> Action)
212     (ledger : PosteriorActionIdentityLedger History Env Action)
213     (source : BayesianPosteriorActionSource
214       History Env Action mu env history nextAction ledger) :
215     ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
216       ProbabilityTheory.condDistrib (ledger.bestAction ∘ env) history mu := by
217   have haction :
218       ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
219         ledger.posterior.kernel.map ledger.bestAction := by
220     simpa only [ledger.actionKernel_eq_posterior_map] using
221       source.hasCondDistrib_actionKernel
222   have hposteriorMap :
223       ledger.posterior.kernel.map ledger.bestAction =ᵐ[mu.map history]
224         (ProbabilityTheory.condDistrib env history mu).map ledger.bestAction := by
225     filter_upwards [source.posterior_eq_condDistrib_env] with observed hposterior
226     rw [ProbabilityTheory.Kernel.map_apply _ ledger.bestAction_measurable,
227       ProbabilityTheory.Kernel.map_apply _ ledger.bestAction_measurable]
228     exact congrArg (Measure.map ledger.bestAction) hposterior
229   exact haction.trans <| hposteriorMap.trans <|
230     (ProbabilityTheory.condDistrib_comp
231       history source.measurable_env.aemeasurable
232       ledger.bestAction_measurable).symm

234 /--
235 Direct posterior-map form of Thompson probability matching.
236 
237 This is the local Mathlib-facing counterpart of pinned LML
238 `Bandits.TS.hasCondDistrib_action`: it avoids a local `HasCondDistrib` wrapper
239 and states the resulting regular conditional-kernel equality directly.
240 -/
241 theorem condDistrib_action_ae_eq_bestAction_of_posteriorMap
242     {Omega : Type u} {History : Type v} {Env : Type w} {Action : Type*}
243     [MeasurableSpace Omega] [MeasurableSpace History]
244     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
245     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
246     (mu : Measure Omega) [IsFiniteMeasure mu]
247     (env : Omega -> Env) (history : Omega -> History)
248     (nextAction : Omega -> Action)
249     (posterior : PosteriorKernel.MarkovPosteriorKernel History Env)
250     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
251     (henv : Measurable env) (hhistory : Measurable history)
252     (hnextAction : Measurable nextAction)
253     (haction :
254       ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
255         posterior.kernel.map bestAction)
256     (hposterior :
257       posterior.kernel =ᵐ[mu.map history]
258         ProbabilityTheory.condDistrib env history mu) :
259     ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
260       ProbabilityTheory.condDistrib (bestAction ∘ env) history mu := by
261   let ledger := PosteriorActionIdentityLedger.ofPosteriorMap
262     posterior bestAction hbestAction
263   let source : BayesianPosteriorActionSource
264       History Env Action mu env history nextAction ledger :=
265     { measurable_env := henv
266       measurable_history := hhistory
267       measurable_nextAction := hnextAction
268       hasCondDistrib_actionKernel := by
269         simpa only [ledger, PosteriorActionIdentityLedger.ofPosteriorMap] using haction
270       posterior_eq_condDistrib_env := by
271         simpa only [ledger, PosteriorActionIdentityLedger.ofPosteriorMap] using hposterior }
272   simpa only [ledger, PosteriorActionIdentityLedger.ofPosteriorMap] using
273     condDistrib_action_ae_eq_bestAction_of_bayesianPosteriorActionSource
274       mu env history nextAction ledger source

276 /--
277 Thompson probability matching from a Bayesian environment/history pair law.
278 
279 Unlike `condDistrib_action_ae_eq_bestAction_of_posteriorMap`, this theorem does
280 not assume the posterior/environment conditional-law equality.  It constructs
281 that equality from the source pair law using Mathlib's canonical posterior.
282 -/
283 theorem condDistrib_action_ae_eq_bestAction_of_bayesianPairMap
284     {Omega : Type u} {History : Type v} {Env : Type w} {Action : Type*}
285     [MeasurableSpace Omega] [MeasurableSpace History]
286     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
287     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
288     (mu : Measure Omega) [IsFiniteMeasure mu]
289     (env : Omega -> Env) (history : Omega -> History)
290     (nextAction : Omega -> Action)
291     (prior : Measure Env) [IsProbabilityMeasure prior]
292     (likelihood : ProbabilityTheory.Kernel Env History)
293     [ProbabilityTheory.IsMarkovKernel likelihood]
294     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
295     (henv : Measurable env) (hhistory : Measurable history)
296     (hnextAction : Measurable nextAction)
297     (hpair :
298       mu.map (fun omega => (env omega, history omega)) =
299         PosteriorKernel.canonicalJointMeasure prior likelihood)
300     (haction :
301       ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
302         (PosteriorKernel.canonicalPosterior prior likelihood).kernel.map
303           bestAction) :
304     ProbabilityTheory.condDistrib nextAction history mu =ᵐ[mu.map history]
305       ProbabilityTheory.condDistrib (bestAction ∘ env) history mu := by
306   exact condDistrib_action_ae_eq_bestAction_of_posteriorMap
307     mu env history nextAction
308     (PosteriorKernel.canonicalPosterior prior likelihood)
309     bestAction hbestAction henv hhistory hnextAction haction
310     (PosteriorKernel.canonicalPosterior_kernel_ae_eq_condDistrib_of_pair_map_eq
311       mu env history henv hhistory prior likelihood hpair)

313 /--
314 Canonical-product specialization of the Thompson posterior-action law.
315 
316 The source space is `Env × History` with law `prior ⊗ₘ likelihood`, so the
317 posterior conditional-law premise is discharged entirely by the canonical
318 Bayesian construction.  The remaining law premise is exactly the Thompson
319 action sampler's conditional law.
320 -/
321 theorem condDistrib_action_ae_eq_bestAction_of_canonicalPriorLikelihood
322     {History : Type u} {Env : Type v} {Action : Type w}
323     [MeasurableSpace History]
324     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
325     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
326     (prior : Measure Env) [IsProbabilityMeasure prior]
327     (likelihood : ProbabilityTheory.Kernel Env History)
328     [ProbabilityTheory.IsMarkovKernel likelihood]
329     (nextAction : Env × History -> Action)
330     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
331     (hnextAction : Measurable nextAction)
332     (haction :
333       ProbabilityTheory.condDistrib nextAction Prod.snd
334           (PosteriorKernel.canonicalJointMeasure prior likelihood) =ᵐ[
335         (PosteriorKernel.canonicalJointMeasure prior likelihood).map Prod.snd]
336           (PosteriorKernel.canonicalPosterior prior likelihood).kernel.map
337             bestAction) :
338     ProbabilityTheory.condDistrib nextAction Prod.snd
339         (PosteriorKernel.canonicalJointMeasure prior likelihood) =ᵐ[
340       (PosteriorKernel.canonicalJointMeasure prior likelihood).map Prod.snd]
341         ProbabilityTheory.condDistrib (bestAction ∘ Prod.fst) Prod.snd
342           (PosteriorKernel.canonicalJointMeasure prior likelihood) := by
343   apply condDistrib_action_ae_eq_bestAction_of_bayesianPairMap
344     (PosteriorKernel.canonicalJointMeasure prior likelihood)
345     Prod.fst Prod.snd nextAction prior likelihood bestAction hbestAction
346     measurable_fst measurable_snd hnextAction
347   · simp [PosteriorKernel.canonicalJointMeasure]
348   · exact haction
