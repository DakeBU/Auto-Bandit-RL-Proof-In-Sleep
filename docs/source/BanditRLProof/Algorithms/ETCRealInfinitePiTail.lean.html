10 /-!
11 # Native Real ETC concentration under the canonical infinite product law
12 
13 This module proves the exact single-arm ETC wrong-commit tail for a native
14 Real reward kernel under the canonical independent-coordinate product law.
15 It then consumes that tail in the expected pull-count and finite-sum kernel
16 regret identities. Transport from an arbitrary external algorithm/environment
17 sequence remains a downstream law-identification obligation.
18 -/

26 /-- A finite maximizer of the identity-integral means of a Real reward kernel. -/
27 noncomputable def realKernelBestArm {K : Nat} (hK : 0 < K)
28     (nu : ProbabilityTheory.Kernel (Fin K) Real) : Fin K :=
29   ETC.realArgmaxCommit hK (realKernelMean nu)

31 /-- The selected kernel best arm dominates every arm mean. -/
32 theorem realKernelMean_le_realKernelBestArm {K : Nat} (hK : 0 < K)
33     (nu : ProbabilityTheory.Kernel (Fin K) Real) (a : Fin K) :
34     realKernelMean nu a <=
35       realKernelMean nu (ETC.realKernelBestArm hK nu) := by
36   exact ETC.realArgmaxCommit_spec hK (realKernelMean nu) a

38 /-- The finite supremum of kernel means is attained at `realKernelBestArm`. -/
39 theorem ciSup_realKernelMean_eq_realKernelBestArm {K : Nat} (hK : 0 < K)
40     (nu : ProbabilityTheory.Kernel (Fin K) Real) :
41     (⨆ a : Fin K, realKernelMean nu a) =
42       realKernelMean nu (ETC.realKernelBestArm hK nu) := by
43   letI : Nonempty (Fin K) := ⟨Fin.mk 0 hK⟩
44   apply le_antisymm
45   · exact ciSup_le (fun a => ETC.realKernelMean_le_realKernelBestArm hK nu a)

58 /-- Native Real centered candidate-minus-best reward difference. -/
59 noncomputable def realCenteredPairwiseRewardDiff
60     {Omega : Type u} {K : Nat}
61     (spec : ETC.Spec K) (mean : Fin K -> Real) (best commitArm : Fin K)
62     (reward : Omega -> RewardTrace Real) (a : Fin K)
63     (t : Nat) (omega : Omega) : Real :=
64   (if ETC.actionWithCommit spec commitArm t = a then
65       reward omega t - mean a else 0) +
66     (if ETC.actionWithCommit spec commitArm t = best then
67       mean best - reward omega t else 0)
68 
69 /-- Native Real threshold of the centered pairwise exploration event. -/
70 noncomputable def realCenteredPairwiseGapThreshold
71     {K : Nat} (spec : ETC.Spec K) (mean : Fin K -> Real)
72     (best a : Fin K) : Real :=
73   (spec.explorationPulls : Real) * (mean best - mean a)

75 /-- Variance proxy charged at candidate and best-arm exploration coordinates. -/
76 noncomputable def realCenteredPairwiseRewardDiffVarianceProxy
77     {K : Nat} (spec : ETC.Spec K) (best commitArm : Fin K)
78     (cReward : Fin K -> Nat -> NNReal) (a : Fin K) (t : Nat) : NNReal :=
79   if ETC.actionWithCommit spec commitArm t = a then cReward a t
80   else if ETC.actionWithCommit spec commitArm t = best then cReward best t
81   else 0

83 /-- Selected centered Real rewards equal reward sum minus count times mean. -/
84 theorem real_selectedSubMean_sum_eq_sumRewards_sub_pullCount_mul
85     {Action : Type} [DecidableEq Action]
86     (action : ActionTrace Action) (reward : RewardTrace Real)
87     (a : Action) (n : Nat) (mu : Real) :
88     (Finset.range n).sum
89         (fun t => if action t = a then reward t - mu else 0) =
90       sumRewards action reward a n - (pullCount action a n : Real) * mu := by
91   classical
92   calc
93     (Finset.range n).sum
94         (fun t => if action t = a then reward t - mu else 0) =
95       ((Finset.range n).filter (fun t => action t = a)).sum
96         (fun t => reward t - mu) := by rw [Finset.sum_filter]
97     _ = ((Finset.range n).filter (fun t => action t = a)).sum reward -
98         ((Finset.range n).filter (fun t => action t = a)).sum
99           (fun _t => mu) := by rw [Finset.sum_sub_distrib]
100     _ = sumRewards action reward a n -
101         (((Finset.range n).filter (fun t => action t = a)).card : Real) * mu := by
102           rw [← sumRewards_eq_finset_filter_sum]
103           rw [Finset.sum_const]
104           simp [nsmul_eq_mul, mul_comm]
105     _ = sumRewards action reward a n - (pullCount action a n : Real) * mu := by
106           rw [pullCount_eq_finset_filter_card]

108 /-- Selected negative centered Real rewards equal count times mean minus sum. -/
109 theorem real_meanSubSelected_sum_eq_pullCount_mul_sub_sumRewards
110     {Action : Type} [DecidableEq Action]
111     (action : ActionTrace Action) (reward : RewardTrace Real)
112     (a : Action) (n : Nat) (mu : Real) :
113     (Finset.range n).sum
114         (fun t => if action t = a then mu - reward t else 0) =
115       (pullCount action a n : Real) * mu - sumRewards action reward a n := by
116   classical
117   calc
118     (Finset.range n).sum
119         (fun t => if action t = a then mu - reward t else 0) =
120       ((Finset.range n).filter (fun t => action t = a)).sum
121         (fun t => mu - reward t) := by rw [Finset.sum_filter]
122     _ = ((Finset.range n).filter (fun t => action t = a)).sum
123           (fun _t => mu) -
124         ((Finset.range n).filter (fun t => action t = a)).sum reward := by
125           rw [Finset.sum_sub_distrib]
126     _ = (((Finset.range n).filter (fun t => action t = a)).card : Real) * mu -
127         sumRewards action reward a n := by
128           have hsum :
129               ((Finset.range n).filter (fun t => action t = a)).sum reward =
130                 sumRewards action reward a n :=
131             (sumRewards_eq_finset_filter_sum
132               (action := action) (reward := reward) (a := a) (t := n)).symm
133           rw [hsum]
134           rw [Finset.sum_const]
135           simp [nsmul_eq_mul, mul_comm]
136     _ = (pullCount action a n : Real) * mu -
137         sumRewards action reward a n := by
138           rw [pullCount_eq_finset_filter_card]

140 /-- Equal exploration counts turn a raw reward comparison into a centered sum. -/
141 theorem real_sumRewards_le_imp_centered_pairwise_sum_ge
142     {Action : Type} [DecidableEq Action]
143     (action : ActionTrace Action) (reward : RewardTrace Real)
144     (a b : Action) (n m : Nat) (muA muB : Real)
145     (hcount_a : pullCount action a n = m)
146     (hcount_b : pullCount action b n = m)
147     (hraw : sumRewards action reward b n <= sumRewards action reward a n) :
148     (m : Real) * (muB - muA) <=
149       (Finset.range n).sum (fun t =>
150         (if action t = a then reward t - muA else 0) +
151         (if action t = b then muB - reward t else 0)) := by
152   classical
153   rw [Finset.sum_add_distrib]
154   rw [ETC.real_selectedSubMean_sum_eq_sumRewards_sub_pullCount_mul]
155   rw [ETC.real_meanSubSelected_sum_eq_pullCount_mul_sub_sumRewards]
156   rw [hcount_a, hcount_b]
157   have hdiff : 0 <=
158       sumRewards action reward a n - sumRewards action reward b n :=
159     sub_nonneg.mpr hraw
160   calc
161     (m : Real) * (muB - muA) = 0 + (m : Real) * (muB - muA) := by ring
162     _ <= (sumRewards action reward a n - sumRewards action reward b n) +
163         (m : Real) * (muB - muA) := by gcongr
164     _ = (sumRewards action reward a n - (m : Real) * muA) +
165         ((m : Real) * muB - sumRewards action reward b n) := by ring

167 /-- A native Real commit fiber is contained in its centered pairwise tail event. -/

169     {Omega : Type u} {K : Nat}

171     (reward : Omega -> RewardTrace Real) (a : Fin K)
172     (hm : 0 < spec.explorationPulls) :
173     Set.Subset
174       {omega | ETC.realExplorationArgmaxCommit spec baseCommitArm
175           (reward omega) = a}

177         (Finset.range (spec.explorationPulls * K)).sum (fun t =>
178           ETC.realCenteredPairwiseRewardDiff
179             spec mean best baseCommitArm reward a t omega)} := by
180   intro omega hcommit
181   have hscore := ETC.realArgmaxCommit_spec spec.hK
182     (fun b => ETC.realEmpMeanAtExploration spec baseCommitArm (reward omega) b)

185       (fun b => ETC.realEmpMeanAtExploration spec baseCommitArm
186         (reward omega) b) = a by exact hcommit] at hscore
187   rw [ETC.realEmpMeanAtExploration_eq_sumRewards_div_explorationPulls,
188     ETC.realEmpMeanAtExploration_eq_sumRewards_div_explorationPulls] at hscore

190   have hraw :

193         sumRewards (ETC.actionWithCommit spec baseCommitArm) (reward omega) a
194           (spec.explorationPulls * K) :=
195     (div_le_div_iff_of_pos_right hmReal).mp hscore
196   exact ETC.real_sumRewards_le_imp_centered_pairwise_sum_ge
197     (ETC.actionWithCommit spec baseCommitArm) (reward omega) a best
198     (spec.explorationPulls * K) spec.explorationPulls (mean a) (mean best)
199     (ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq
200       spec baseCommitArm a)
201     (ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq
202       spec baseCommitArm best)
203     hraw

205 /-- Coordinate independence survives the native Real pairwise transform. -/
206 theorem iIndepFun_realCenteredPairwiseRewardDiff_of_iIndepFun_reward
207     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
208     (mu : Measure Omega) (spec : ETC.Spec K) (mean : Fin K -> Real)
209     (best commitArm : Fin K) (reward : Omega -> RewardTrace Real)

215   let transform : Nat -> Real -> Real := fun t r =>
216     (if ETC.actionWithCommit spec commitArm t = a then r - mean a else 0) +

219     intro t

225 /-- Per-coordinate centered sub-Gaussianity transfers to the pairwise summand. -/
226 theorem realCenteredPairwiseRewardDiff_hasSubgaussianMGF_of_centeredReward
227     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
228     (mu : Measure Omega) [IsZeroOrProbabilityMeasure mu]
229     (spec : ETC.Spec K) (mean : Fin K -> Real) (best commitArm : Fin K)
230     (reward : Omega -> RewardTrace Real) (cReward : Fin K -> Nat -> NNReal)
231     (a : Fin K) (t : Nat) (hne : a ≠ best)
232     (h_subG : forall b, ETC.actionWithCommit spec commitArm t = b ->
233       ProbabilityTheory.HasSubgaussianMGF
234         (fun omega => reward omega t - mean b) (cReward b t) mu) :
235     ProbabilityTheory.HasSubgaussianMGF
236       (fun omega => ETC.realCenteredPairwiseRewardDiff
237         spec mean best commitArm reward a t omega)
238       (ETC.realCenteredPairwiseRewardDiffVarianceProxy
239         spec best commitArm cReward a t) mu := by
240   classical
241   by_cases ha : ETC.actionWithCommit spec commitArm t = a
242   · have hnot_best : ETC.actionWithCommit spec commitArm t ≠ best := by
243       intro hbest
244       exact hne (ha.symm.trans hbest)
245     simpa [ETC.realCenteredPairwiseRewardDiffVarianceProxy,
246       ETC.realCenteredPairwiseRewardDiff, ha, hnot_best, hne] using h_subG a ha
247   · by_cases hbest : ETC.actionWithCommit spec commitArm t = best
248     · have hbest_ne_a : best ≠ a := Ne.symm hne
249       have h := (h_subG best hbest).neg
250       have htarget : ProbabilityTheory.HasSubgaussianMGF
251           (fun omega => mean best - reward omega t) (cReward best t) mu := by
252         refine h.congr (Filter.Eventually.of_forall (fun omega => ?_))
253         simp [Pi.neg_apply, sub_eq_add_neg]
254       simpa [ETC.realCenteredPairwiseRewardDiffVarianceProxy,
255         ETC.realCenteredPairwiseRewardDiff, ha, hbest, hbest_ne_a] using htarget
256     · have hfun :
257           (fun omega => ETC.realCenteredPairwiseRewardDiff
258             spec mean best commitArm reward a t omega) =
259           (fun _omega => (0 : Real)) := by
260         funext omega
261         simp [ETC.realCenteredPairwiseRewardDiff, ha, hbest]
262       rw [hfun]
263       simp [ETC.realCenteredPairwiseRewardDiffVarianceProxy, ha, hbest]

265 /-- The constant pairwise proxy sums to exactly `2 * m * sigma2`. -/
266 theorem sum_realCenteredPairwiseRewardDiffVarianceProxy_const_eq_two_mul
267     {K : Nat} (spec : ETC.Spec K) (best : Fin K)
268     (sigma2 : NNReal) (a : Fin K) (hne : a ≠ best) :

270         ETC.realCenteredPairwiseRewardDiffVarianceProxy
271           spec best best (fun _ _ => sigma2) a t) =
272       (2 : NNReal) * (spec.explorationPulls : NNReal) * sigma2 := by
273   classical
274   let action : ActionTrace (Fin K) := ETC.actionWithCommit spec best
275   have hpoint : forall t,
276       ETC.realCenteredPairwiseRewardDiffVarianceProxy
277           spec best best (fun _ _ => sigma2) a t =
278         (if action t = a then sigma2 else 0) +
279           (if action t = best then sigma2 else 0) := by
280     intro t

283     · by_cases hbest : action t = best

285           hbest, Ne.symm hne]
286       · simp [ETC.realCenteredPairwiseRewardDiffVarianceProxy, action, ha, hbest]
287   have hselected : forall b : Fin K,
288       (Finset.range (spec.explorationPulls * K)).sum
289           (fun t => if action t = b then sigma2 else 0) =
290         (spec.explorationPulls : NNReal) * sigma2 := by
291     intro b
292     calc
293       (Finset.range (spec.explorationPulls * K)).sum
294           (fun t => if action t = b then sigma2 else 0) =
295         ((Finset.range (spec.explorationPulls * K)).filter
296           (fun t => action t = b)).sum (fun _t => sigma2) := by
297             rw [Finset.sum_filter]
298       _ = (((Finset.range (spec.explorationPulls * K)).filter
299           (fun t => action t = b)).card : NNReal) * sigma2 := by
300             simp [nsmul_eq_mul]
301       _ = (spec.explorationPulls : NNReal) * sigma2 := by
302             rw [← pullCount_eq_finset_filter_card]
303             simp [action,
304               ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq]
305   rw [Finset.sum_congr rfl (fun t _ht => hpoint t)]
306   rw [Finset.sum_add_distrib, hselected a, hselected best]

317     (sigma2 : NNReal)

357       simpa [mu] using Measure.infinitePi_map_eval coordLaw t

369       simpa [coordLaw] using hsubG (ETC.exploreArm spec t)

376             (fun trajectory : RewardTrace Real => trajectory) a t trajectory)
377         (ETC.realCenteredPairwiseRewardDiffVarianceProxy
378           spec best best (fun _ _ => sigma2) a t) mu := by
379     intro t ht

381       mu spec (realKernelMean nu) best best
382       (fun trajectory : RewardTrace Real => trajectory)
383       (fun _ _ => sigma2) a t hne
384       (fun b hb => by simpa [hb] using hcoord_subG t (Finset.mem_range.mp ht))
385   have hgap_nonneg : 0 <= realKernelGap nu a := by
386     rw [ETC.realKernelGap_eq_realKernelBestArm_sub spec.hK nu]
387     exact sub_nonneg.mpr (ETC.realKernelMean_le_realKernelBestArm spec.hK nu a)
388   have hthreshold_nonneg : 0 <=
389       ETC.realCenteredPairwiseGapThreshold
390         spec (realKernelMean nu) best a := by
391     rw [ETC.realCenteredPairwiseGapThreshold]
392     exact mul_nonneg (Nat.cast_nonneg _)
393       (sub_nonneg.mpr (by
394         simpa [best] using ETC.realKernelMean_le_realKernelBestArm spec.hK nu a))
395   have htail := Concentration.subGaussian_sum_tail_of_iIndepFun
396     mu hind hpair_subG hthreshold_nonneg
397   have hsubset :=
398     ETC.realExplorationArgmaxCommit_eq_arm_event_subset_centeredPairwise_sum_event
399       spec (realKernelMean nu) best best

406           (Finset.range (spec.explorationPulls * K)).sum (fun t =>
