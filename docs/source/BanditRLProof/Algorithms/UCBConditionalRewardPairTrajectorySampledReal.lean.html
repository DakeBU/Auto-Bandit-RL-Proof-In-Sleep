10 /-- Shift sampled pair-trajectory actions at coordinates `1, 2, ...` to times `0, 1, ...`. -/
11 def actionRewardTrajectorySuccessorAction
12     {Action Reward : Type} (trajectory : Nat -> Prod Action Reward) :
13     ActionTrace Action :=
14   fun t => (trajectory (t + 1)).1
