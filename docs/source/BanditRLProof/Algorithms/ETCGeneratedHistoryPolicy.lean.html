6 # Generated finite-history ETC policy
7 
8 This module constructs the canonical ETC action trace as a policy generated
9 from finite reward histories. It closes the deterministic action-alignment
10 layer needed before an adaptive reward law can be transported through the
11 existing generated-action conditional-expectation surfaces.
12 
13 It does not prove a reward law, conditional mean-zero, concentration bound, or
14 the final LML ETC regret theorem.
15 -/

20 /--
21 The state reconstructed from a finite reward history for the ETC commit rule.
22 
23 The prefix through `t` is retained exactly and the unobserved suffix is zero.
24 The defaulted suffix is only an implementation device: score reconstruction
25 requires the exploration horizon to be contained in the retained prefix.
26 -/
27 def explorationArgmaxHistoryState
28     (t : Nat) (history : History.FiniteRewardHistory Rat t) :
29     RewardTrace Rat :=
30   History.completeRewardTrace t history (0 : Rat)
31 
32 /-- The finite-history ETC state reconstruction is measurable as a trace. -/
33 theorem measurable_explorationArgmaxHistoryState (t : Nat) :
34     Measurable (explorationArgmaxHistoryState t) := by
35   refine measurable_pi_lambda _ ?_
36   intro s
37   by_cases hs : s <= t
38   · let i : Finset.Iic t := ⟨s, Finset.mem_Iic.mpr hs⟩
39     have h_eval : Measurable
40         (fun history : History.FiniteRewardHistory Rat t => history i) :=
41       measurable_pi_apply i
42     simpa [explorationArgmaxHistoryState, History.completeRewardTrace, hs, i]
43       using h_eval
44   · simp [explorationArgmaxHistoryState, History.completeRewardTrace, hs]

46 /--
47 At policy time `t`, explore at action time `t + 1` while that action lies in
48 the exploration prefix; otherwise choose the empirical-mean argmax computed
49 from the completed finite-history state.
50 -/
51 noncomputable def explorationArgmaxHistoryPolicy
52     {K : Nat} (spec : ETC.Spec K) (model : FiniteBanditModel K)
53     (t : Nat) : Policy.MeasurablePolicy (RewardTrace Rat) (Fin K) where
54   action := fun completedReward =>
55     if t + 1 < spec.explorationPulls * K then
56       ETC.exploreArm spec (t + 1)
57     else
58       (ETC.argmaxCommitOracle model.hK).choose
59         (fun a : Fin K =>
60           ETC.empMeanAtExploration spec model.bestArm completedReward a)
61   measurable_action := by
62     by_cases hphase : t + 1 < spec.explorationPulls * K
63     · simp [hphase]
64     · simpa [hphase] using
65         (ETC.measurable_commitOracle_choose_of_forall_measurable_empMean
66           (oracle := ETC.argmaxCommitOracle model.hK)
67           (empMean := fun omega : RewardTrace Rat => fun a : Fin K =>
68             ETC.empMeanAtExploration spec model.bestArm omega a)
69           (ETC.measurable_empMeanAtExploration_coordinates
70             spec model.bestArm (fun omega : RewardTrace Rat => omega)
71             (fun i => measurable_pi_apply i)))

73 /--
74 The canonical ETC action generator obtained by running the finite-history
75 policy against the identity reward trace.
76 -/
77 noncomputable def explorationArgmaxGeneratedAction
78     {K : Nat} (spec : ETC.Spec K) (model : FiniteBanditModel K) :
79     RewardTrace Rat -> ActionTrace (Fin K) :=
80   ConditionalExpectationReward.generatedActionFromRewardHistory
81     (fun t => ETC.explorationArgmaxHistoryPolicy spec model t)
82     (fun t history => ETC.explorationArgmaxHistoryState t history)
83     (ETC.exploreArm spec 0)
84     (fun reward : RewardTrace Rat => reward)

86 /--
87 The canonical empirical-mean ETC trace is generated by the measurable policy
88 over finite reward histories.
89 
90 Positive exploration pulls are essential at time zero: the shifted generated
91 trace starts with `ETC.exploreArm spec 0`, which matches the ETC trace only
92 when time zero belongs to the exploration prefix. At later commit times, the
93 finite-history score reconstruction theorem supplies the argmax equality.
94 -/
95 theorem explorationArgmaxGeneratedAction_eq_explorationArgmaxAction
96     {K : Nat} (spec : ETC.Spec K) (model : FiniteBanditModel K)
97     (hexplorationPulls_pos : 0 < spec.explorationPulls) :
98     ETC.explorationArgmaxGeneratedAction spec model =
99       ETC.explorationArgmaxAction spec model := by
100   funext reward time
101   cases time with
102   | zero =>
103       have hzero : 0 < spec.explorationPulls * K :=
104         Nat.mul_pos hexplorationPulls_pos spec.hK
105       simp [ETC.explorationArgmaxGeneratedAction,
106         ConditionalExpectationReward.generatedActionFromRewardHistory,
107         Policy.generatedActionTraceSucc, ETC.explorationArgmaxAction,
108         ETC.fixedProductArgmaxAction, ETC.actionWithCommit, hzero]
109   | succ t =>
110       by_cases hphase : t + 1 < spec.explorationPulls * K
111       · simp [ETC.explorationArgmaxGeneratedAction,
112           ConditionalExpectationReward.generatedActionFromRewardHistory,
113           Policy.generatedActionTraceSucc, ETC.explorationArgmaxHistoryPolicy,
114           ETC.explorationArgmaxAction, ETC.fixedProductArgmaxAction,
115           ETC.actionWithCommit, hphase]
116       · have hhorizon : spec.explorationPulls * K <= t + 1 :=
117           Nat.le_of_not_gt hphase
118         have hscores :
119             (fun a : Fin K =>
120               ETC.empMeanAtExploration spec model.bestArm
121                 (ETC.explorationArgmaxHistoryState t
122                   (History.finiteRewardHistoryOfTrace reward t)) a) =
123               (fun a : Fin K =>
124                 ETC.empMeanAtExploration spec model.bestArm reward a) := by
125           funext a
126           exact
127             ETC.empMeanAtExploration_completeRewardTrace_eq_of_explorationHorizon_le
128               spec model.bestArm reward t hhorizon a
129         simp only [ETC.explorationArgmaxGeneratedAction,
130           ConditionalExpectationReward.generatedActionFromRewardHistory,
131           Policy.generatedActionTraceSucc, ETC.explorationArgmaxHistoryPolicy,
132           ETC.explorationArgmaxAction,
133           ETC.fixedProductArgmaxAction, ETC.fixedProductArgmaxCommit,
134           ETC.actionWithCommit, hphase, hscores]

136 /--
137 During the exploration prefix, the generated finite-history ETC policy agrees
138 with every fixed-commit ETC trace because both choose the same round-robin arm.
139 
140 The post-exploration commit arm is deliberately arbitrary in this statement;
141 only the strict exploration-horizon hypothesis is used.
142 -/
143 theorem explorationArgmaxGeneratedAction_eq_actionWithCommit_of_lt
144     {K : Nat} (spec : ETC.Spec K) (model : FiniteBanditModel K)
145     (commitArm : Fin K) (reward : RewardTrace Rat) {t : Nat}
146     (hexplorationPulls_pos : 0 < spec.explorationPulls)
147     (ht : t < spec.explorationPulls * K) :
148     ETC.explorationArgmaxGeneratedAction spec model reward t =
149       ETC.actionWithCommit spec commitArm t := by
150   have hgenerated := congrFun
151     (congrFun
152       (ETC.explorationArgmaxGeneratedAction_eq_explorationArgmaxAction
153         spec model hexplorationPulls_pos)
154       reward)
155     t
156   calc
157     ETC.explorationArgmaxGeneratedAction spec model reward t =
158         ETC.explorationArgmaxAction spec model reward t := hgenerated
159     _ = ETC.exploreArm spec t := by
160       simp [ETC.explorationArgmaxAction, ETC.fixedProductArgmaxAction,
161         ETC.actionWithCommit, ht]
162     _ = ETC.actionWithCommit spec commitArm t :=
163       (ETC.actionWithCommit_eq_exploreArm_of_lt spec commitArm ht).symm

165 /--
166 Canonical kernel-trajectory law source for the finite-history ETC policy.
167 
168 The reward process is the identity coordinate process under the
169 `RewardKernel.historyStepKernelFamily` trajectory measure. The existing
170 canonical `trajMeasure` construction proves the full finite-pair
171 `partialTraj` law for the generated policy action without assuming an ambient
172 selected-reward law. Combined with
173 `explorationArgmaxGeneratedAction_eq_explorationArgmaxAction`, this law is the
174 action-dependent probability foundation for the canonical ETC action trace.
175 
176 It remains a kernel-generated trajectory law. Transport to an arbitrary
177 adaptive environment and identification with the fixed product-coordinate
178 source remain separate obligations.
179 -/

215 /--
216 Canonical trajectory conditional sub-Gaussian MGF for the successor reward
217 centered at the finite-bandit model mean selected by the ETC policy.
218 
219 The mean surface is context-independent, `fun _ action => model.mean action`.
220 The kernel centered-reward law supplies conditional mean-zero and the
221 one-step MGF; the remaining regularity contract is the selected finite-history
222 variance ceiling at the requested successor time. This theorem is therefore a
223 concentration foundation for the canonical kernel trajectory, not a transport
224 to the fixed product-coordinate reward source or an arbitrary adaptive bandit
225 environment.
226 -/
