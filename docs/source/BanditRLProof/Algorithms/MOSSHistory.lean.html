6 # MOSS on the common finite-history policy interface
7 
8 An inclusive history at t contains t+1 observations; the successor selector
9 therefore calls the source action at t+1. No reward law or concentration
10 certificate is required to construct this deterministic Markov policy.
11 -/
12 
13 namespace BanditRLProof.MOSS
14 
15 open MeasureTheory ProbabilityTheory
16 
17 /-- Source MOSS action after an inclusive finite action/reward history. -/
18 noncomputable def historyAction {k : ℕ} (hk : 0 < k) (n t : ℕ)
19     (history : History.FinitePairHistory (Fin k) ℝ t) : Fin k :=
20   action hk n (t + 1) (ETC.realHistoryEmpMean t history)
21     (ETC.realHistoryPullCount t history)
22 
23 theorem measurable_historyAction {k : ℕ} (hk : 0 < k) (n t : ℕ) :
24     Measurable (historyAction hk n t) := by
25   unfold historyAction action
26   split_ifs with ht
27   · exact measurable_const
28   · change Measurable (fun history : History.FinitePairHistory (Fin k) ℝ t =>
29       ETC.realArgmaxCommit hk (index n (ETC.realHistoryEmpMean t history)
30         (ETC.realHistoryPullCount t history)))
31     apply ETC.measurable_realArgmaxCommit_of_forall_measurable
32     intro a
33     exact (UCB.measurable_realHistoryEmpMean t a).add
34       ((measurable_of_countable (radius n k)).comp
35         (UCB.measurable_realHistoryPullCount t a))
36 
37 /-- Concrete MOSS policy on the history interface shared by Chapter 13's
38 minimax regret functional. Initial action is arm zero. -/
39 noncomputable def historyAlgorithm {k : ℕ} (hk : 0 < k) (n : ℕ) :
40     Thompson.HistoryAlgorithm (Fin k) ℝ where
41   policy t := Kernel.deterministic (historyAction hk n t)
42     (measurable_historyAction hk n t)
43   initialAction := Measure.dirac ⟨0, hk⟩
44 
45 @[simp] theorem historyAlgorithm_policy_apply {k : ℕ} (hk : 0 < k)
46     (n t : ℕ) (history : History.FinitePairHistory (Fin k) ℝ t) :
47     (historyAlgorithm hk n).policy t history =
48       Measure.dirac (historyAction hk n t history) := by
49   rw [historyAlgorithm, Kernel.deterministic_apply]
50 
51 /-- Subsequent initialization rounds select the corresponding arm. -/
52 theorem historyAction_initialization {k : ℕ} (hk : 0 < k) (n t : ℕ)
53     (history : History.FinitePairHistory (Fin k) ℝ t) (ht : t + 1 < k) :
54     historyAction hk n t history = ⟨t + 1, ht⟩ :=
55   action_of_lt hk n (t + 1) _ _ ht
56 
57 /-- Exact post-initialization source index maximality on visible history. -/
58 theorem historyAction_index_max {k : ℕ} (hk : 0 < k) (n t : ℕ)
59     (history : History.FinitePairHistory (Fin k) ℝ t) (ht : k ≤ t + 1)
60     (a : Fin k) :
61     index n (ETC.realHistoryEmpMean t history) (ETC.realHistoryPullCount t history) a ≤
62       index n (ETC.realHistoryEmpMean t history) (ETC.realHistoryPullCount t history)
63         (historyAction hk n t history) :=
64   action_index_max hk n (t + 1) _ _ ht a
