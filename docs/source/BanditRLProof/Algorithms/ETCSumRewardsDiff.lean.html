9 /-!
10 # ETC reward-difference finite-sum bridge
11 
12 This module bridges the fixed-horizon `sumRewards` comparison produced by the
13 ETC empirical-mean algebra layer to a centered pairwise finite-sum event.  It
14 stays deterministic: no probability measure, independence, sub-Gaussianity,
15 filtration, conditional expectation, or final ETC result is introduced here.
16 -/
17 
18 namespace BanditRLProof
19 namespace ETC

21 /--
22 Real-valued centered pairwise reward-difference summand for arm `a` against
23 the model's selected best arm at one ETC exploration-horizon time index.
24 
25 The expression remains deterministic and pointwise.  Future probability
26 leaves may impose independence or sub-Gaussian contracts on this function.
27 -/
28 noncomputable def centeredPairwiseRewardDiff
29     {Omega : Type u} {K : Nat}
30     (spec : ETC.Spec K)
31     (model : FiniteBanditModel K)
32     (commitArm : Fin K)
33     (reward : Omega -> RewardTrace Rat)
34     (a : Fin K) (t : Nat) (omega : Omega) : Real :=
35   (((if ETC.actionWithCommit spec commitArm t = a then
36       reward omega t - model.mean a else 0) +
37     (if ETC.actionWithCommit spec commitArm t = model.bestArm then
38       model.mean model.bestArm - reward omega t else 0) : Rat) : Real)
39 
40 /--
41 Real-valued threshold corresponding to `explorationPulls` copies of the
42 mean gap between the selected best arm and arm `a`.
43 -/
44 noncomputable def centeredPairwiseGapThreshold
45     {K : Nat}
46     (spec : ETC.Spec K)
47     (model : FiniteBanditModel K)
48     (a : Fin K) : Real :=
49   (((spec.explorationPulls : Rat) *
50     (model.mean model.bestArm - model.mean a) : Rat) : Real)

52 /--
53 The selected centered reward sum for one arm equals its selected reward total
54 minus its pull count times the supplied mean.
55 
56 This helper is part of the `ETC-SUMREWARDS-PAIRWISE-DIFF-FINSET` bridge.  It
57 uses the existing Mathlib-backed `sumRewards` and `pullCount` Finset wrappers.
58 -/
59 theorem selectedSubMean_sum_eq_sumRewards_sub_pullCount_mul
60     {Action : Type} [DecidableEq Action]
61     (action : ActionTrace Action) (reward : RewardTrace Rat)
62     (a : Action) (n : Nat) (mu : Rat) :
63     (Finset.range n).sum
64       (fun t : Nat => if action t = a then reward t - mu else 0) =
65       sumRewards action reward a n - (pullCount action a n : Rat) * mu := by
66   classical
67   calc
68     (Finset.range n).sum
69         (fun t : Nat => if action t = a then reward t - mu else 0)
70         = ((Finset.range n).filter (fun t : Nat => action t = a)).sum
71             (fun t : Nat => reward t - mu) := by
72           rw [Finset.sum_filter]
73     _ = ((Finset.range n).filter (fun t : Nat => action t = a)).sum
74             (fun t : Nat => reward t) -
75           ((Finset.range n).filter (fun t : Nat => action t = a)).sum
76             (fun _t : Nat => mu) := by
77           rw [Finset.sum_sub_distrib]
78     _ = sumRewards action reward a n -
79           (((Finset.range n).filter
80             (fun t : Nat => action t = a)).card : Rat) * mu := by
81           have hsum :
82               ((Finset.range n).filter
83                 (fun t : Nat => action t = a)).sum
84                 (fun t : Nat => reward t) =
85                 sumRewards action reward a n := by
86             exact
87               (sumRewards_eq_finset_filter_sum
88                 (action := action) (reward := reward) (a := a)
89                 (t := n)).symm
90           rw [hsum]
91           rw [Finset.sum_const]
92           simp [nsmul_eq_mul, mul_comm]
93     _ = sumRewards action reward a n - (pullCount action a n : Rat) * mu := by
94           have hcount :=
95             pullCount_eq_finset_filter_card
96               (action := action) (a := a) (t := n)
97           rw [hcount.symm]

100 The selected negative centered reward sum for one arm equals its pull count
101 times the supplied mean minus its selected reward total.
102 
103 This is the best-arm companion to
104 `selectedSubMean_sum_eq_sumRewards_sub_pullCount_mul`.

106 theorem meanSubSelected_sum_eq_pullCount_mul_sub_sumRewards
107     {Action : Type} [DecidableEq Action]
108     (action : ActionTrace Action) (reward : RewardTrace Rat)
109     (a : Action) (n : Nat) (mu : Rat) :
110     (Finset.range n).sum
111       (fun t : Nat => if action t = a then mu - reward t else 0) =
112       (pullCount action a n : Rat) * mu - sumRewards action reward a n := by
113   classical
114   calc
115     (Finset.range n).sum
116         (fun t : Nat => if action t = a then mu - reward t else 0)
117         = ((Finset.range n).filter (fun t : Nat => action t = a)).sum
118             (fun t : Nat => mu - reward t) := by
119           rw [Finset.sum_filter]
120     _ = ((Finset.range n).filter (fun t : Nat => action t = a)).sum
121             (fun _t : Nat => mu) -
122           ((Finset.range n).filter (fun t : Nat => action t = a)).sum
123             (fun t : Nat => reward t) := by
124           rw [Finset.sum_sub_distrib]
125     _ = (((Finset.range n).filter
126             (fun t : Nat => action t = a)).card : Rat) * mu -
127           sumRewards action reward a n := by
128           have hsum :
129               ((Finset.range n).filter
130                 (fun t : Nat => action t = a)).sum
131                 (fun t : Nat => reward t) =
132                 sumRewards action reward a n := by
133             exact
134               (sumRewards_eq_finset_filter_sum
135                 (action := action) (reward := reward) (a := a)
136                 (t := n)).symm
137           rw [hsum]
138           rw [Finset.sum_const]
139           simp [nsmul_eq_mul, mul_comm]
140     _ = (pullCount action a n : Rat) * mu -
141           sumRewards action reward a n := by
142           have hcount :=
143             pullCount_eq_finset_filter_card
144               (action := action) (a := a) (t := n)
145           rw [hcount.symm]

147 /--
148 If two arms have the same pull count by a horizon and the fixed-horizon reward
149 sum of `b` is at most that of `a`, then the centered pairwise reward-difference
150 finite sum is at least `m * (muB - muA)`.
151 
152 This is the deterministic algebra core of
153 `ETC-SUMREWARDS-PAIRWISE-DIFF-FINSET`.  The equal-count assumptions are kept
154 explicit so ETC can discharge them with its exploration-horizon count theorem.
155 -/
156 theorem sumRewards_le_imp_centered_pairwise_sum_ge
157     {Action : Type} [DecidableEq Action]
158     (action : ActionTrace Action) (reward : RewardTrace Rat)
159     (a b : Action) (n m : Nat) (muA muB : Rat)
160     (hcount_a : pullCount action a n = m)
161     (hcount_b : pullCount action b n = m)
162     (hraw : sumRewards action reward b n <=
163       sumRewards action reward a n) :
164     (m : Rat) * (muB - muA) <=
165       (Finset.range n).sum (fun t : Nat =>
166         (if action t = a then reward t - muA else 0) +
167         (if action t = b then muB - reward t else 0)) := by
168   classical
169   rw [Finset.sum_add_distrib]
170   rw [selectedSubMean_sum_eq_sumRewards_sub_pullCount_mul]
171   rw [meanSubSelected_sum_eq_pullCount_mul_sub_sumRewards]
172   rw [hcount_a, hcount_b]
173   have hdiff :
174       (0 : Rat) <=
175         sumRewards action reward a n - sumRewards action reward b n :=
176     sub_nonneg.mpr hraw
177   calc
178     (m : Rat) * (muB - muA)
179         = 0 + (m : Rat) * (muB - muA) := by
180           simp
181     _ <= (sumRewards action reward a n -
182             sumRewards action reward b n) +
183           (m : Rat) * (muB - muA) := by
184           simpa [add_comm, add_left_comm, add_assoc] using
185             add_le_add_right hdiff ((m : Rat) * (muB - muA))
186     _ = (sumRewards action reward a n - (m : Rat) * muA) +
187           ((m : Rat) * muB - sumRewards action reward b n) := by
188           ring

190 /--
191 Concrete ETC event inclusion from the empirical-mean comparison event to the
192 centered pairwise reward-difference finite-sum event over the configured
193 exploration horizon.
194 
195 This leaf instantiates the earlier abstract event-shape adapter with
196 `idx := Finset.range (spec.explorationPulls * K)` and the centered
197 non-best-minus-best reward-difference summands.  It still does not prove
198 independence, sub-Gaussianity, filtration, or final ETC regret.
199 -/
200 theorem empMeanAtExploration_ge_best_event_subset_centered_pairwise_sum_event
201     {Omega : Type u} {K : Nat}
202     (spec : ETC.Spec K)
203     (model : FiniteBanditModel K)
204     (commitArm : Fin K)
205     (reward : Omega -> RewardTrace Rat)
206     (a : Fin K)
207     (hexplorationPulls_pos : 0 < spec.explorationPulls) :
208     Set.Subset
209       {omega : Omega |
210         ETC.empMeanAtExploration spec commitArm (reward omega) a >=
211           ETC.empMeanAtExploration spec commitArm (reward omega)
212             model.bestArm}
213       {omega : Omega |
214         ETC.centeredPairwiseGapThreshold spec model a <=
215           (Finset.range (spec.explorationPulls * K)).sum (fun t : Nat =>
216             ETC.centeredPairwiseRewardDiff
217               spec model commitArm reward a t omega)} := by
218   apply
219     ETC.empMeanAtExploration_ge_best_event_subset_sumRewards_tail_event_of_imp
220       (spec := spec)
221       (model := model)
222       (commitArm := commitArm)
223       (reward := reward)
224       (a := a)
225       (hexplorationPulls_pos := hexplorationPulls_pos)
226       (idx := Finset.range (spec.explorationPulls * K))
227       (X := fun t omega =>
228         ETC.centeredPairwiseRewardDiff
229           spec model commitArm reward a t omega)
230       (eps := ETC.centeredPairwiseGapThreshold spec model a)
231   intro omega hraw
232   let term : Nat -> Rat := fun t : Nat =>
233     (if ETC.actionWithCommit spec commitArm t = a then
234       reward omega t - model.mean a else 0) +
235     (if ETC.actionWithCommit spec commitArm t = model.bestArm then
236       model.mean model.bestArm - reward omega t else 0)
237   have hrat :
238       (spec.explorationPulls : Rat) *
239           (model.mean model.bestArm - model.mean a) <=
240         (Finset.range (spec.explorationPulls * K)).sum term :=
241     ETC.sumRewards_le_imp_centered_pairwise_sum_ge
242       (action := ETC.actionWithCommit spec commitArm)
243       (reward := reward omega)
244       (a := a)
245       (b := model.bestArm)
246       (n := spec.explorationPulls * K)
247       (m := spec.explorationPulls)
248       (muA := model.mean a)
249       (muB := model.mean model.bestArm)
250       (ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq
251         spec commitArm a)
252       (ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq
253         spec commitArm model.bestArm)
254       hraw
255   have hreal :
256       ETC.centeredPairwiseGapThreshold spec model a <=
257         (((Finset.range (spec.explorationPulls * K)).sum term : Rat) :
258           Real) := by
259     exact (Rat.cast_le (K := Real)).2 hrat
260   have hcast :
261       (((Finset.range (spec.explorationPulls * K)).sum term : Rat) :
262           Real) =
263         (Finset.range (spec.explorationPulls * K)).sum
264           (fun t : Nat => (term t : Real)) := by
265     exact
266       map_sum (Rat.castHom Real) term
267         (Finset.range (spec.explorationPulls * K))
268   simpa [term, hcast, ETC.centeredPairwiseRewardDiff,
269     ETC.centeredPairwiseGapThreshold] using hreal
