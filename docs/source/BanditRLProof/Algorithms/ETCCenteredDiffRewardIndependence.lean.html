6 # ETC centered reward-difference independence transfer
7 
8 This module transfers time-coordinate independence of a stochastic reward trace
9 through the deterministic centered pairwise reward-difference transform used by
10 the ETC wrong-commit tail route.
11 -/
12 
13 namespace BanditRLProof
14 namespace ETC
15 
16 /--
17 If the reward trace coordinates are independent across time, then the centered
18 pairwise reward-difference summands are independent across time for every
19 non-best arm.
20 
21 This is the `ETC-CENTERED-DIFF-INDEPENDENCE-WITNESS` leaf. It only proves the
22 deterministic-transform part of the reward-law independence obligation. It
23 does not prove the reward trace coordinates are independent from a kernel or
24 environment model, and it does not prove any sub-Gaussian witness.
25 -/
26 theorem iIndepFun_centeredPairwiseRewardDiff_of_iIndepFun_reward
27     {Omega : Type u} {K : Nat}
28     [MeasurableSpace Omega]
29     (mu : MeasureTheory.Measure Omega)
30     (spec : ETC.Spec K)
31     (model : FiniteBanditModel K)
32     (commitArm : Fin K)
33     (reward : Omega -> RewardTrace Rat)
34     (h_reward_indep :
35       ProbabilityTheory.iIndepFun
36         (fun t omega => reward omega t) mu) :
37     forall a : Fin K, (a = model.bestArm -> False) ->
38       ProbabilityTheory.iIndepFun
39         (fun t omega =>
40           ETC.centeredPairwiseRewardDiff spec model commitArm reward a t omega)
41         mu := by
42   intro a _hne
43   let transform : Nat -> Rat -> Real := fun t r =>
44     (((if ETC.actionWithCommit spec commitArm t = a then
45         r - model.mean a else 0) +
46       (if ETC.actionWithCommit spec commitArm t = model.bestArm then
47         model.mean model.bestArm - r else 0) : Rat) : Real)
48   have hcomp :
49       ProbabilityTheory.iIndepFun
50         (fun t omega => transform t (reward omega t)) mu :=
51     h_reward_indep.comp transform (fun _t => measurable_of_countable _)
52   simpa [transform, ETC.centeredPairwiseRewardDiff] using hcomp
53 
54 end ETC
55 end BanditRLProof
