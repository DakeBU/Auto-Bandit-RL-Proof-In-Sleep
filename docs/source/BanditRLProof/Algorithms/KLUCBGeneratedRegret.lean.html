5 # Generated finite-arm KL-UCB
6 
7 This module defines one horizon-free KL-UCB policy on the canonical generated
8 Rat action/reward trajectory.  The confidence budget is calibrated from the
9 accepted telescoping empirical-mean radius and an explicit common interior
10 margin.  The selected score is nevertheless the supremum of the genuine
11 Bernoulli-KL confidence set; it is not the ordinary additive UCB score.
12 -/

21 /-- KL exploration budget on a realized generated prefix.  The margin is a
22 known regularity contract, while the empirical count and radius are computed
23 from the actual action/reward trace. -/
24 def generatedBudgetAt
25     {Omega : Type} {K : Nat}
26     (action : Omega -> ActionTrace (Fin K)) (sigma2 : NNReal)
27     (delta margin : Real) (omega : Omega) (t : Nat) (arm : Fin K) : Real :=
28   let count := ConditionalExpectationReward.successorArmPullCount
29     (action omega) arm (t + 1)
30   let radius := UCB.selectedPolicySuccessorTelescopingRadiusAt
31     action sigma2 delta omega t arm
32   (count : Real) * radius ^ 2 / (margin * (1 - margin))

345 /-- A bounded empirical-mean deviation on the actual prefix makes the true
346 interior arm mean feasible for the KL confidence set used by the policy. -/
347 theorem armMean_mem_confidenceSet_of_abs_lt_radius
348     {Omega : Type} {K : Nat}
349     (action : Omega -> ActionTrace (Fin K))
350     (reward : Omega -> RewardTrace Rat)
351     (sigma2 : NNReal) (delta margin : Real)
352     (hmargin0 : 0 < margin) (hmarginHalf : margin <= 1 / 2)
353     (armMean : Fin K -> Rat) (omega : Omega) (t : Nat) (arm : Fin K)
354     (hemp : IsBernoulliParameter
355       (UCB.selectedPolicySuccessorEmpiricalMeanAt action reward omega t arm))
356     (hmean : (armMean arm : Real) ∈ Set.Icc margin (1 - margin))
357     (hdev :
358       |UCB.selectedPolicySuccessorEmpiricalMeanAt action reward omega t arm -
359           (armMean arm : Real)| <
360         UCB.selectedPolicySuccessorTelescopingRadiusAt
361           action sigma2 delta omega t arm) :
362     (armMean arm : Real) ∈ confidenceSet
363       (UCB.selectedPolicySuccessorEmpiricalMeanAt action reward omega t arm)
364       (ConditionalExpectationReward.successorArmPullCount
365         (action omega) arm (t + 1))
366       (generatedBudgetAt action sigma2 delta margin omega t arm) := by
367   let empirical :=
368     UCB.selectedPolicySuccessorEmpiricalMeanAt action reward omega t arm
369   let mean := (armMean arm : Real)
370   let count := ConditionalExpectationReward.successorArmPullCount
371     (action omega) arm (t + 1)
372   let radius := UCB.selectedPolicySuccessorTelescopingRadiusAt
373     action sigma2 delta omega t arm
374   have hmargin1 : margin < 1 :=
375     lt_of_le_of_lt hmarginHalf (by norm_num : (1 / 2 : Real) < 1)
376   have hmean0 : 0 < mean := hmargin0.trans_le hmean.1
377   have hmean1 : mean < 1 := hmean.2.trans_lt (sub_lt_self 1 hmargin0)
378   have hmarginDen : 0 < margin * (1 - margin) :=
379     mul_pos hmargin0 (sub_pos.mpr hmargin1)
380   have hmeanDen : 0 < mean * (1 - mean) :=
381     mul_pos hmean0 (sub_pos.mpr hmean1)
382   have hdenLe : margin * (1 - margin) <= mean * (1 - mean) := by
383     have hprod := mul_nonneg (sub_nonneg.mpr hmean.1)
384       (sub_nonneg.mpr hmean.2)
385     nlinarith
386   have hradius0 : 0 < radius := lt_of_le_of_lt (abs_nonneg _) hdev
387   have hradiusNonneg : 0 <= radius := hradius0.le
388   have hsq : (empirical - mean) ^ 2 < radius ^ 2 := by
389     rw [sq_lt_sq, abs_of_nonneg hradiusNonneg]
390     simpa [empirical, mean, radius] using hdev
391   have hdiv : (empirical - mean) ^ 2 / (mean * (1 - mean)) <
392       radius ^ 2 / (margin * (1 - margin)) := by
393     apply (div_lt_div_iff₀ hmeanDen hmarginDen).2
394     calc
395       (empirical - mean) ^ 2 * (margin * (1 - margin)) <
396           radius ^ 2 * (margin * (1 - margin)) := by gcongr
397       _ <= radius ^ 2 * (mean * (1 - mean)) := by gcongr
398   apply mem_confidenceSet_of_natCast_mul_core_le hemp hmean0 hmean1
399   unfold generatedBudgetAt
400   change (count : Real) * bernoulliKLCore empirical mean <=
401     (count : Real) * radius ^ 2 / (margin * (1 - margin))
402   have hcore := bernoulliKLCore_le_sq_div hemp hmean0 hmean1
403   calc
404     (count : Real) * bernoulliKLCore empirical mean <=
405         (count : Real) * ((empirical - mean) ^ 2 /
406           (mean * (1 - mean))) := by gcongr
407     _ <= (count : Real) * (radius ^ 2 /
408           (margin * (1 - margin))) := by
409         gcongr
410     _ = (count : Real) * radius ^ 2 /
411           (margin * (1 - margin)) := by ring
