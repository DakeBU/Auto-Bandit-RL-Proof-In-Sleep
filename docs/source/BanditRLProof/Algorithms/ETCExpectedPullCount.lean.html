6 /-!
7 # ETC expected pull counts
8 
9 This module integrates the deterministic pull-count formula for an
10 `Omega`-indexed ETC commit selector.  It isolates the exact interface between
11 ETC counting and concentration: a later tail argument only has to bound the
12 commit-fiber probability `mu.real {omega | commit omega = a}`.
13 -/
14 
15 universe u
16 
17 open MeasureTheory
18 
19 namespace BanditRLProof
20 namespace ETC

22 /--
23 The Real cast of a finite-horizon ETC pull count is integrable when the commit
24 selector is measurable and the ambient measure is finite.
25 
26 This regularity adapter is independent of reward laws and concentration.  Its
27 proof uses timewise measurability of the finite-valued ETC action and the
28 deterministic bound `pullCount <= n`.
29 -/
30 theorem integrable_real_pullCount_actionWithCommit_choice_of_measurable_commit
31     {Omega : Type u} {K : Nat}
32     [MeasurableSpace Omega]
33     (mu : Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
34     (spec : ETC.Spec K) (commit : Omega -> Fin K)
35     (a : Fin K) (n : Nat)
36     (hmeas_commit : Measurable commit) :
37     Integrable
38       (fun omega : Omega =>
39         ((pullCount (ETC.actionWithCommit spec (commit omega)) a n : Nat) :
40           Real)) mu := by
41   let action : Omega -> ActionTrace (Fin K) :=
42     fun omega => ETC.actionWithCommit spec (commit omega)
43   have haction : forall t : Nat,
44       Measurable (fun omega : Omega => action omega t) := by
45     intro t
46     exact
47       (measurable_of_countable
48         (fun commitArm : Fin K => ETC.actionWithCommit spec commitArm t)).comp
49         hmeas_commit
50   have hmeas :
51       Measurable
52         (fun omega : Omega =>
53           ((pullCount (action omega) a n : Nat) : Real)) :=
54     measurable_natCast_pullCount action haction a n
55   refine MeasureTheory.Integrable.of_bound
56     hmeas.aestronglyMeasurable (n : Real) ?_
57   filter_upwards [] with omega
58   rw [Real.norm_eq_abs, abs_of_nonneg (Nat.cast_nonneg _)]
59   exact_mod_cast pullCount_le_time (action omega) a n

61 /--
62 At horizon `spec.explorationPulls * K + r`, the expected Real pull count of arm
63 `a` is exactly the exploration count plus `r` times the probability of
64 committing to `a`.
65 
66 This is the direct Bochner/indicator integration of
67 `ETC.pullCount_actionWithCommit_explorationPulls_mul_K_add_eq`.
68 -/
69 theorem integral_real_pullCount_actionWithCommit_choice_eq_exploration_add_suffix_mul_commit_prob
70     {Omega : Type u} {K : Nat}
71     [MeasurableSpace Omega]
72     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
73     (spec : ETC.Spec K) (commit : Omega -> Fin K)
74     (a : Fin K) (r : Nat)
75     (hmeas_commit : Measurable commit) :
76     MeasureTheory.integral mu
77       (fun omega : Omega =>
78         ((pullCount (ETC.actionWithCommit spec (commit omega)) a
79           (spec.explorationPulls * K + r) : Nat) : Real)) =
80     (spec.explorationPulls : Real) +
81       (r : Real) * mu.real {omega : Omega | commit omega = a} := by
82   let commitSet : Set Omega := {omega : Omega | commit omega = a}
83   let suffix : Omega -> Real :=
84     commitSet.indicator (fun _omega : Omega => (r : Real))
85   have hcommitSet : MeasurableSet commitSet := by
86     change MeasurableSet (commit ⁻¹' {a})
87     exact hmeas_commit (measurableSet_singleton a)
88   have hsuffix : Integrable suffix mu :=
89     (integrable_const (r : Real)).indicator hcommitSet
90   have hpoint :
91       (fun omega : Omega =>
92         ((pullCount (ETC.actionWithCommit spec (commit omega)) a
93           (spec.explorationPulls * K + r) : Nat) : Real)) =
94       (fun omega : Omega => (spec.explorationPulls : Real) + suffix omega) := by
95     funext omega
96     rw [ETC.pullCount_actionWithCommit_explorationPulls_mul_K_add_eq]
97     by_cases h : commit omega = a
98     · simp [suffix, commitSet, h]
99     · simp [suffix, commitSet, h]
100   rw [hpoint]
101   rw [MeasureTheory.integral_add (integrable_const _) hsuffix]
102   rw [MeasureTheory.integral_const]
103   rw [show suffix = commitSet.indicator
104       (fun _omega : Omega => (r : Real)) by rfl]
105   rw [MeasureTheory.integral_indicator hcommitSet]
106   rw [MeasureTheory.setIntegral_const]
107   simp [commitSet, MeasureTheory.probReal_univ, smul_eq_mul, mul_comm]
108 
109 /--
110 LML-shaped horizon form of the exact ETC expected pull-count identity.
111 
112 The exploration condition is written as `K * explorationPulls <= n`, and the
113 suffix is `n - K * explorationPulls`, matching the exact ETC theorem-card
114 surface.  No concentration or reward-law assumption is used.
115 -/
116 theorem integral_real_pullCount_actionWithCommit_choice_eq_exploration_add_remaining_mul_commit_prob
117     {Omega : Type u} {K : Nat}
118     [MeasurableSpace Omega]
119     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
120     (spec : ETC.Spec K) (commit : Omega -> Fin K)
121     (a : Fin K) (n : Nat)
122     (hn : K * spec.explorationPulls <= n)
123     (hmeas_commit : Measurable commit) :
124     MeasureTheory.integral mu
125       (fun omega : Omega =>
126         ((pullCount (ETC.actionWithCommit spec (commit omega)) a n : Nat) :
127           Real)) =
128     (spec.explorationPulls : Real) +
129       ((n - K * spec.explorationPulls : Nat) : Real) *
130         mu.real {omega : Omega | commit omega = a} := by
131   have hsplit :
132       spec.explorationPulls * K + (n - K * spec.explorationPulls) = n := by
133     rw [Nat.mul_comm spec.explorationPulls K]
134     exact Nat.add_sub_of_le hn
135   calc
136     MeasureTheory.integral mu
137         (fun omega : Omega =>
138           ((pullCount (ETC.actionWithCommit spec (commit omega)) a n : Nat) :
139             Real)) =
140       MeasureTheory.integral mu
141         (fun omega : Omega =>
142           ((pullCount (ETC.actionWithCommit spec (commit omega)) a
143             (spec.explorationPulls * K +
144               (n - K * spec.explorationPulls)) : Nat) : Real)) := by
145         congr 1
146         funext omega
147         rw [hsplit]
148     _ = (spec.explorationPulls : Real) +
149         ((n - K * spec.explorationPulls : Nat) : Real) *
150           mu.real {omega : Omega | commit omega = a} := by
151       exact
152         integral_real_pullCount_actionWithCommit_choice_eq_exploration_add_suffix_mul_commit_prob
153           mu spec commit a (n - K * spec.explorationPulls) hmeas_commit

155 /--
156 Per-arm ETC expected pull-count bound from a commit-fiber probability bound.
157 
158 This is the concentration consumer needed by the exact LML route: a later
159 empirical-mean tail theorem supplies only
160 `mu.real {omega | commit omega = a} <= p`; the counting and integration steps
161 are discharged here.
162 -/
163 theorem integral_real_pullCount_actionWithCommit_choice_le_exploration_add_remaining_mul_of_commit_prob_le
164     {Omega : Type u} {K : Nat}
165     [MeasurableSpace Omega]
166     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
167     (spec : ETC.Spec K) (commit : Omega -> Fin K)
168     (a : Fin K) (n : Nat) (p : Real)
169     (hn : K * spec.explorationPulls <= n)
170     (hmeas_commit : Measurable commit)
171     (hprob : mu.real {omega : Omega | commit omega = a} <= p) :
172     MeasureTheory.integral mu
173       (fun omega : Omega =>
174         ((pullCount (ETC.actionWithCommit spec (commit omega)) a n : Nat) :
175           Real)) <=
176     (spec.explorationPulls : Real) +
177       ((n - K * spec.explorationPulls : Nat) : Real) * p := by
178   rw [integral_real_pullCount_actionWithCommit_choice_eq_exploration_add_remaining_mul_commit_prob
179     mu spec commit a n hn hmeas_commit]
180   exact add_le_add (le_refl _)
181     (mul_le_mul_of_nonneg_left hprob (Nat.cast_nonneg _))
