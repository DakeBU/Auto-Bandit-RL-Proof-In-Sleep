4 # Two-arm SGB Theorem 2: native stationary environment and one-step prefix step
5 
6 The latent arm-stream coupling draws every reward coordinate before the
7 algorithm runs, whereas the native fixed-IID process draws only the reward of
8 the arm actually selected in each round.  The compiled deterministic-time
9 one-step selected-reward laws describe the coupling; this module states the
10 native process against which those laws must be compared.
11 
12 It records the stationary reward environment whose feedback kernel is the
13 selected-arm law, identifies its history step kernel with the composition of
14 the algorithm policy and that law, proves equality of every finite visible
15 prefix, and promotes those prefix identities to equality of the complete
16 visible trajectory measures.  No selected/stopped IID law,
17 future/no-return statement, or Theorem-2 endpoint is claimed here.
18 -/

45 /-- One-step native extension of the visible prefix under the latent coupling.
46 
47 At every deterministic time `n`, the joint law of the visible prefix through
48 `n` and the next observed action/reward pair is the prefix law composed with the
49 native stationary step kernel.  This is a single-step statement; it does not
50 identify whole prefixes or the full native law. -/
51 theorem latentArmStreamVisiblePrefixNextPair_eq_compProd
52     {Env : Type u} {K : Nat} [MeasurableSpace Env]
53     [StandardBorelSpace Env] [NeZero K]
54     (algorithm : HistoryAlgorithm (Fin K) Real) (env : Env)
55     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] (n : Nat) :
56     Measure.map
57         (fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
58           (Preorder.frestrictLe n sample.2, sample.2 (n + 1)))
59         (latentArmStreamTrajectoryMeasure algorithm env nu) =
60       Measure.map
61           (fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
62             Preorder.frestrictLe n sample.2)
63           (latentArmStreamTrajectoryMeasure algorithm env nu) ⊗ₘ
64         historyStepKernel algorithm (stationaryRewardHistoryEnvironment nu) n := by
65   classical
66   have hmeasPair :
67       Measurable
68         (fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
69           (latentArmStreamVisiblePrefixNextAction n sample.2,
70             latentArmStreamVisibleNextReward n sample.2)) := by
71     exact ((measurable_latentArmStreamVisiblePrefixNextAction n).comp
72       measurable_snd).prodMk
73       ((measurable_latentArmStreamVisibleNextReward n).comp measurable_snd)
74   have hjoint := latentArmStreamVisibleNextReward_joint_eq_compProd
75     algorithm env nu n
76   have haction := latentArmStreamTrajectoryMeasure_map_visiblePrefix_nextAction_eq_compProd
77     algorithm env nu n
78   rw [haction] at hjoint
79   have hassoc := Measure.compProd_assoc'
80     (μ := Measure.map
81       (fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
82         Preorder.frestrictLe n sample.2)
83       (latentArmStreamTrajectoryMeasure algorithm env nu))
84     (κ := algorithm.policy n)
85     (η := UCB.armStreamSelectedRewardKernel n nu)
86   rw [historyStepKernel_stationaryRewardHistoryEnvironment, ← hassoc, ← hjoint,
87     Measure.map_map MeasurableEquiv.prodAssoc.measurable hmeasPair]
88   rfl

93 theorem latentArmStreamVisibleInitialPair_eq_compProd

97     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :

100           sample.2 0)

102       algorithm.initialAction ⊗ₘ nu := by
103   classical
104   have hzero : forall stream : UCB.ArmRewardStream K,
105       (latentArmStreamTrajectoryKernel algorithm env stream).map
106           (fun trajectory : (t : Nat) -> Fin K × Real => trajectory 0) =
107         algorithm.initialAction ⊗ₘ
108           ((latentArmStreamMeasurableHistoryEnvironment
109             (Env := Env) (K := K)).at (env, stream)).initialFeedback := by
110     intro stream

112         ((canonicalMeasurableEnvironmentTrajectoryKernel algorithm
113             (latentArmStreamMeasurableHistoryEnvironment
114               (Env := Env) (K := K))).map
115           (fun trajectory : (t : Nat) -> Fin K × Real => trajectory 0))
116             (env, stream) =

118             (latentArmStreamMeasurableHistoryEnvironment

120       rw [canonicalMeasurableEnvironmentTrajectoryKernel_map_eval_zero]
121     rw [Kernel.map_apply _ (measurable_pi_apply 0),
122       measurableEnvironmentInitialPairKernel_apply] at hkernel
123     simpa [latentArmStreamTrajectoryKernel, Kernel.comap_apply] using hkernel
124   have hdet : forall stream : UCB.ArmRewardStream K,
125       ((latentArmStreamMeasurableHistoryEnvironment
126         (Env := Env) (K := K)).at (env, stream)).initialFeedback =
127         Kernel.deterministic (fun arm : Fin K => stream 0 arm)
128           (measurable_from_top) := by
129     intro stream
130     ext arm S hS
131     rw [Kernel.deterministic_apply]
132     simp [MeasurableHistoryEnvironment.at,
133       latentArmStreamMeasurableHistoryEnvironment_initialFeedback_apply]
134   have hmeasEval :
135       Measurable
136         (fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
137           sample.2 0) := by
138     exact (measurable_pi_apply 0).comp measurable_snd
139   ext S hS
140   rw [Measure.map_apply hmeasEval hS, latentArmStreamTrajectoryMeasure,
141     Measure.compProd_apply (hmeasEval hS)]
142   have hfiber : forall stream : UCB.ArmRewardStream K,
143       (latentArmStreamTrajectoryKernel algorithm env) stream
144           (Prod.mk stream ⁻¹'
145             ((fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
146               sample.2 0) ⁻¹' S)) =
147         algorithm.initialAction ((fun arm : Fin K => (arm, stream 0 arm)) ⁻¹' S) := by
148     intro stream
149     have hpre :
150         (Prod.mk stream ⁻¹'
151           ((fun sample : UCB.ArmRewardStream K × ((t : Nat) -> Fin K × Real) =>
152             sample.2 0) ⁻¹' S)) =
153           (fun trajectory : (t : Nat) -> Fin K × Real => trajectory 0) ⁻¹' S := rfl
154     rw [hpre, ← Measure.map_apply (measurable_pi_apply 0) hS, hzero stream,
155       hdet stream, Measure.compProd_deterministic (measurable_from_top
156         (f := fun arm : Fin K => stream 0 arm))]
157     rw [Measure.map_apply (measurable_from_top
158       (f := fun arm : Fin K => (arm, stream 0 arm))) hS]
159   simp only [hfiber]
160   have hmeasSwap :
161       Measurable
162         (fun p : UCB.ArmRewardStream K × Fin K => (p.2, p.1 0 p.2)) := by
163     refine measurable_from_prod_countable_left ?_
164     intro arm
165     exact measurable_const.prodMk
166       ((measurable_pi_apply arm).comp (measurable_pi_apply 0))
167   have hmeasSet : MeasurableSet
168       ((fun p : UCB.ArmRewardStream K × Fin K => (p.2, p.1 0 p.2)) ⁻¹' S) :=
169     hmeasSwap hS
170   have hswap :
171       ∫⁻ stream : UCB.ArmRewardStream K,
172           algorithm.initialAction
173             ((fun arm : Fin K => (arm, stream 0 arm)) ⁻¹' S)
174           ∂(UCB.armStreamMeasure nu) =
175         ∫⁻ arm : Fin K,
176           UCB.armStreamMeasure nu
177             ((fun stream : UCB.ArmRewardStream K => (arm, stream 0 arm)) ⁻¹' S)
178           ∂algorithm.initialAction :=
179     (Measure.prod_apply (μ := UCB.armStreamMeasure nu)
180         (ν := algorithm.initialAction) hmeasSet).symm.trans
181       (Measure.prod_apply_symm (μ := UCB.armStreamMeasure nu)
182         (ν := algorithm.initialAction) hmeasSet)
183   have hcoord : forall arm : Fin K,
184       UCB.armStreamMeasure nu
185           ((fun stream : UCB.ArmRewardStream K => (arm, stream 0 arm)) ⁻¹' S) =
186         nu arm (Prod.mk arm ⁻¹' S) := by
187     intro arm
188     have hpre :
189         ((fun stream : UCB.ArmRewardStream K => (arm, stream 0 arm)) ⁻¹' S) =
190           (fun stream : UCB.ArmRewardStream K => stream 0 arm) ⁻¹'
191             (Prod.mk arm ⁻¹' S) := rfl
192     have hmeasCoord :
193         Measurable (fun stream : UCB.ArmRewardStream K => stream 0 arm) := by
194       exact (measurable_pi_apply arm).comp (measurable_pi_apply 0)
195     rw [hpre, ← Measure.map_apply hmeasCoord
196       (hS.preimage (measurable_prodMk_left)),

201 /-- The Ionescu-Tulcea trajectory law reproduces its initial law at time zero. -/
202 theorem trajMeasure_map_eval_zero
203     {X : Nat -> Type u} [forall n, MeasurableSpace (X n)]
204     (mu0 : Measure (X 0)) [IsProbabilityMeasure mu0]
205     (kappa : (n : Nat) -> Kernel ((i : Finset.Iic n) -> X i) (X (n + 1)))
206     [forall n, IsMarkovKernel (kappa n)] :
207     (Kernel.trajMeasure mu0 kappa).map (fun x => x 0) = mu0 := by
208   have hcomp :
209       (fun x : (n : Nat) -> X n => x 0) =
210         (fun h : (i : Finset.Iic 0) -> X i =>
211           h ⟨0, Finset.mem_Iic.mpr le_rfl⟩) ∘ Preorder.frestrictLe 0 := rfl
212   rw [hcomp, ← Measure.map_map (by fun_prop) (Preorder.measurable_frestrictLe 0),
213     Kernel.trajMeasure, Measure.map_comp _ _ (Preorder.measurable_frestrictLe 0),
214     Kernel.traj_map_frestrictLe, Kernel.partialTraj_self]
215   simp only [Measure.id_comp]
216   rw [Measure.map_map (measurable_pi_apply
217     (⟨0, Finset.mem_Iic.mpr le_rfl⟩ : Finset.Iic 0)) (by fun_prop)]
218   convert Measure.map_id

220 /-- The visible prefix at `n + 1` is the prefix at `n` extended by the next pair. -/
221 theorem frestrictLe_succ_eq_extendPairHistorySucc
222     {K : Nat} (n : Nat) (x : (t : Nat) -> Fin K × Real) :
223     Preorder.frestrictLe (n + 1) x =
224       History.extendPairHistorySucc (Preorder.frestrictLe n x) (x (n + 1)) := by
225   funext i
226   by_cases hi : i.1 <= n
227   · rw [History.extendPairHistorySucc_apply_of_le _ _ i hi]
228     rfl
229   · have hval : i.1 = n + 1 :=
230       le_antisymm (Finset.mem_Iic.mp i.2) (Nat.succ_le_of_lt (Nat.not_le.mp hi))
231     have hi' : i = ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩ := Subtype.ext hval
232     subst hi'
233     rw [History.extendPairHistorySucc_apply_succ]
234     rfl

252 /-- **Native-prefix identification.**  At every finite horizon `n`, the visible
253 trajectory marginal of the latent arm-stream coupling and the native
254 fixed-i.i.d. process induce the same law on prefixes through `n`.
255 
256 This is a finite-prefix identity.  It does not by itself give the full native
257 visible law, selected- or stopped-reward i.i.d. statements, the stopped-prefix
258 future/no-return law, or Theorem 2. -/
259 theorem latentArmStreamVisibleTrajectoryMeasure_map_frestrictLe_eq_native
260     {Env : Type u} {K : Nat} [MeasurableSpace Env]
261     [StandardBorelSpace Env] [NeZero K]
262     (algorithm : HistoryAlgorithm (Fin K) Real) (env : Env)
263     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] (n : Nat) :
264     ((latentArmStreamTrajectoryMeasure algorithm env nu).map Prod.snd).map
265         (Preorder.frestrictLe n) =
266       (nativeStationaryTrajectoryMeasure algorithm nu).map
267         (Preorder.frestrictLe n) := by
268   classical
269   have hmarginal : forall {gamma : Type} [MeasurableSpace gamma]
270       (g : ((t : Nat) -> Fin K × Real) -> gamma), Measurable g ->
271       ((latentArmStreamTrajectoryMeasure algorithm env nu).map Prod.snd).map g =
272         (latentArmStreamTrajectoryMeasure algorithm env nu).map
273           (fun sample => g sample.2) := by
274     intro gamma _ g hg
275     rw [Measure.map_map hg measurable_snd]
276     rfl
277   induction n with
278   | zero =>
279       have hsingleton : forall x : (t : Nat) -> Fin K × Real,
280           Preorder.frestrictLe 0 x = singletonPairHistory (x 0) := by
281         intro x
282         funext i
283         have : i = ⟨0, Finset.mem_Iic.mpr le_rfl⟩ := Subsingleton.elim _ _
284         subst this
285         rfl
286       have hnative :
287           (nativeStationaryTrajectoryMeasure algorithm nu).map
288               (fun x : (t : Nat) -> Fin K × Real => x 0) =
289             algorithm.initialAction ⊗ₘ nu := by
290         rw [nativeStationaryTrajectoryMeasure, canonicalHistoryTrajectoryMeasure,
291           trajMeasure_map_eval_zero]
292         rfl
293       have hfun :
294           (Preorder.frestrictLe 0 :
295             ((t : Nat) -> Fin K × Real) ->
296               History.FinitePairHistory (Fin K) Real 0) =
297             singletonPairHistory ∘ (fun x => x 0) := funext hsingleton
298       rw [hfun, ← Measure.map_map measurable_singletonPairHistory
299           (measurable_pi_apply 0),
300         ← Measure.map_map measurable_singletonPairHistory
301           (measurable_pi_apply 0),
302         hnative, hmarginal _ (measurable_pi_apply 0),
303         latentArmStreamVisibleInitialPair_eq_compProd]
304   | succ n ih =>
305       have hnativeEq :
306           nativeStationaryTrajectoryMeasure algorithm nu =
307             Kernel.trajMeasure (algorithm.initialAction ⊗ₘ nu)
308               (fun k => historyStepKernel algorithm
309                 (stationaryRewardHistoryEnvironment nu) k) := rfl
310       have hmeasPair :
311           Measurable
312             (fun x : (t : Nat) -> Fin K × Real =>
313               (Preorder.frestrictLe n x, x (n + 1))) :=
314         (Preorder.measurable_frestrictLe n).prodMk (measurable_pi_apply (n + 1))
315       have hstepNative :=
316         Kernel.map_frestrictLe_trajMeasure_compProd_eq_map_trajMeasure
317           (X := fun _ : Nat => Fin K × Real)
318           (μ₀ := algorithm.initialAction ⊗ₘ nu)
319           (κ := fun k => historyStepKernel algorithm
320             (stationaryRewardHistoryEnvironment nu) k)
321           (a := n)
322       have hstepVisible :=
323         latentArmStreamVisiblePrefixNextPair_eq_compProd algorithm env nu n
324       have hjoint :
325           ((latentArmStreamTrajectoryMeasure algorithm env nu).map Prod.snd).map
326               (fun x : (t : Nat) -> Fin K × Real =>
327                 (Preorder.frestrictLe n x, x (n + 1))) =
328             (nativeStationaryTrajectoryMeasure algorithm nu).map
329               (fun x : (t : Nat) -> Fin K × Real =>
330                 (Preorder.frestrictLe n x, x (n + 1))) := by
331         rw [hmarginal _ hmeasPair, hstepVisible,
332           ← hmarginal _ (Preorder.measurable_frestrictLe n), ih, hnativeEq,
333           ← hstepNative]
334       have hfunSucc :
335           (Preorder.frestrictLe (n + 1) :
336             ((t : Nat) -> Fin K × Real) ->
337               History.FinitePairHistory (Fin K) Real (n + 1)) =
338             (fun input :
339                 History.FinitePairHistory (Fin K) Real n × (Fin K × Real) =>
340               History.extendPairHistorySucc input.1 input.2) ∘
341               (fun x => (Preorder.frestrictLe n x, x (n + 1))) :=
342         funext (frestrictLe_succ_eq_extendPairHistorySucc n)
343       rw [hfunSucc,
344         ← Measure.map_map History.measurable_extendPairHistorySucc hmeasPair,
345         ← Measure.map_map History.measurable_extendPairHistorySucc hmeasPair,
346         hjoint]

348 /-- **Full native visible-law identification.**  Forgetting the latent reward
349 stream from the arm-stream coupling gives exactly the native stationary
350 fixed-i.i.d. SGB trajectory law.
351 
352 The proof promotes the compiled equality of every inclusive finite prefix to
353 an equality of complete trajectory measures by projective-limit uniqueness.
354 It does not assert that totalized stopped rewards are i.i.d., identify a
355 random-time future cylinder, or prove the source Theorem 2 endpoint. -/
356 theorem latentArmStreamVisibleTrajectoryMeasure_eq_native
357     {Env : Type u} {K : Nat} [MeasurableSpace Env]
358     [StandardBorelSpace Env] [NeZero K]
359     (algorithm : HistoryAlgorithm (Fin K) Real) (env : Env)
360     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
361     (latentArmStreamTrajectoryMeasure algorithm env nu).map Prod.snd =
362       nativeStationaryTrajectoryMeasure algorithm nu := by
363   let visibleLaw : Measure ((n : Nat) -> Fin K × Real) :=
364     (latentArmStreamTrajectoryMeasure algorithm env nu).map Prod.snd
365   let nativeLaw : Measure ((n : Nat) -> Fin K × Real) :=
366     nativeStationaryTrajectoryMeasure algorithm nu
367   let finiteLaw := fun I : Finset Nat => Measure.map I.restrict nativeLaw
368   have hnative : MeasureTheory.IsProjectiveLimit nativeLaw finiteLaw := by
369     intro I
370     rfl
371   have hvisible : MeasureTheory.IsProjectiveLimit visibleLaw finiteLaw := by
372     intro I
373     let n := I.sup id
374     have hi_le : forall i : I, i.1 <= n := by
375       intro i
376       simpa [n] using (Finset.le_sup (f := id) i.2)
377     let select :
378         History.FinitePairHistory (Fin K) Real n ->
379           ((i : I) -> Fin K × Real) :=
380       fun history i =>
381         history ⟨i.1, Finset.mem_Iic.mpr (hi_le i)⟩
382     have hselect : Measurable select := by
383       exact measurable_pi_lambda _ (fun i => measurable_pi_apply
384         (⟨i.1, Finset.mem_Iic.mpr (hi_le i)⟩ : Finset.Iic n))
385     have hprefixMeasurable :
386         Measurable
387           (Preorder.frestrictLe n :
388             ((t : Nat) -> Fin K × Real) ->
389               History.FinitePairHistory (Fin K) Real n) :=
390       Preorder.measurable_frestrictLe n
391     have hprefix : visibleLaw.map (Preorder.frestrictLe n) =
392         nativeLaw.map (Preorder.frestrictLe n) := by
393       simpa [visibleLaw, nativeLaw] using
394         latentArmStreamVisibleTrajectoryMeasure_map_frestrictLe_eq_native
395           algorithm env nu n
396     have hfinite := congrArg (Measure.map select) hprefix
397     rw [Measure.map_map hselect hprefixMeasurable,
398       Measure.map_map hselect hprefixMeasurable] at hfinite
399     have hselectPrefix :
400         select ∘
401             (Preorder.frestrictLe n :
402               ((t : Nat) -> Fin K × Real) ->
403                 History.FinitePairHistory (Fin K) Real n) =
404           I.restrict := by
405       funext trajectory i
406       rfl
407     simpa [finiteLaw, hselectPrefix] using hfinite
408   simpa [visibleLaw, nativeLaw] using hvisible.unique hnative
