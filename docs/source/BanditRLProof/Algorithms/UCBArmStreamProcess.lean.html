5 # Recursive UCB process on a latent arm-stream space
6 
7 This module mirrors the deterministic part of the pinned LML array model. It
8 recursively builds the inclusive action/reward history, uses round-robin
9 initialization followed by the native Real history-index selector, and reads
10 the selected arm's next unused latent reward coordinate.
11 -/

19 /-- Round-robin arm at an absolute time coordinate. -/
20 def initializationArm {K : Nat} (hK : 0 < K) (t : Nat) : Fin K :=
21   ⟨t % K, Nat.mod_lt t hK⟩
22 
23 @[simp]
24 theorem initializationArm_zero {K : Nat} (hK : 0 < K) :
25     initializationArm hK 0 = Fin.mk 0 hK := by
26   rfl

28 /--
29 LML-shaped next action after an inclusive history through time `n`.
30 
31 The first `K` actions are round robin. Once that prefix is complete, the
32 native Real UCB history index selects a least-encoded maximizing arm.
33 -/
34 noncomputable def realHistoryNextArm
35     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat)
36     (history : History.FinitePairHistory (Fin K) Real n) : Fin K :=
37   if n < K - 1 then
38     initializationArm hK (n + 1)
39   else
40     realHistoryIndexAction hK c n history
41 
42 /-- The next-arm selector is measurable on the finite history space. -/
43 theorem measurable_realHistoryNextArm
44     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat) :
45     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
46       realHistoryNextArm hK c n history) := by
47   by_cases hn : n < K - 1
48   · simp [realHistoryNextArm, hn]
49   · simpa [realHistoryNextArm, hn] using
50       (measurable_realHistoryIndexAction hK c n)

52 /-- Joint evaluation of an arm stream at countable random coordinates. -/
53 theorem measurable_armRewardStream_apply {K : Nat} :
54     Measurable
55       (fun input : ArmRewardStream K × (Nat × Fin K) =>
56         input.1 input.2.1 input.2.2) := by
57   apply measurable_from_prod_countable_left
58   intro index
59   exact (measurable_pi_apply index.2).comp
60     (measurable_pi_apply index.1)

62 /--
63 Inclusive pair history of the deterministic UCB process driven by a latent
64 arm-reward stream.
65 -/
66 noncomputable def armStreamHistory
67     {K : Nat} (hK : 0 < K) (c : Real)
68     (stream : ArmRewardStream K) :
69     (n : Nat) -> History.FinitePairHistory (Fin K) Real n
70   | 0 =>
71       let arm := initializationArm hK 0
72       fun _ => (arm, stream 0 arm)
73   | n + 1 =>
74       let history := armStreamHistory hK c stream n
75       let arm := realHistoryNextArm hK c n history
76       let count := ETC.realHistoryPullCount n history arm
77       History.extendPairHistorySucc history (arm, stream count arm)
78 
79 /-- Action trace extracted from the recursive inclusive histories. -/
80 noncomputable def armStreamAction
81     {K : Nat} (hK : 0 < K) (c : Real) :
82     ArmRewardStream K -> ActionTrace (Fin K) :=
83   fun stream t =>
84     (armStreamHistory hK c stream t
85       ⟨t, Finset.mem_Iic.mpr le_rfl⟩).1

87 /-- Reward trace obtained by reading each selected arm's next unused value. -/
88 noncomputable def armStreamReward
89     {K : Nat} (hK : 0 < K) (c : Real) :
90     ArmRewardStream K -> RewardTrace Real :=
91   rewardFromArmStream (armStreamAction hK c) id

94 theorem armStreamAction_zero
95     {K : Nat} (hK : 0 < K) (c : Real) (stream : ArmRewardStream K) :
96     armStreamAction hK c stream 0 = initializationArm hK 0 := by
97   rfl

99 @[simp]
100 theorem armStreamAction_succ
101     {K : Nat} (hK : 0 < K) (c : Real)
102     (stream : ArmRewardStream K) (n : Nat) :
103     armStreamAction hK c stream (n + 1) =
104       realHistoryNextArm hK c n (armStreamHistory hK c stream n) := by
105   simp [armStreamAction, armStreamHistory]

107 /--
108 The recursively maintained history is exactly the finite pair history of the
109 extracted action and next-unused-coordinate reward traces.
110 -/
111 theorem armStreamHistory_eq_finitePairHistoryOfTrace
112     {K : Nat} (hK : 0 < K) (c : Real)
113     (stream : ArmRewardStream K) (n : Nat) :
114     armStreamHistory hK c stream n =
115       History.finitePairHistoryOfTrace
116         (armStreamAction hK c stream)
117         (armStreamReward hK c stream) n := by
118   induction n with
119   | zero =>
120       funext i
121       have hi : i.1 = 0 := Nat.eq_zero_of_le_zero (Finset.mem_Iic.mp i.2)
122       have hi_subtype : i = ⟨0, Finset.mem_Iic.mpr le_rfl⟩ := Subtype.ext hi
123       subst i
124       simp [armStreamHistory, armStreamAction, armStreamReward,
125         rewardFromArmStream]
126   | succ n ih =>
127       rw [armStreamHistory]
128       rw [History.finitePairHistoryOfTrace_succ]
129       rw [← ih]
130       apply congrArg (History.extendPairHistorySucc (t := n)
131         (armStreamHistory hK c stream n))
132       apply Prod.ext
133       · simp [armStreamAction_succ]
134       · simp only [armStreamReward, rewardFromArmStream]
135         rw [armStreamAction_succ]
136         rw [ih]
137         rw [ETC.realHistoryPullCount_finitePairHistoryOfTrace]
138         rfl

140 /-- Every recursively generated inclusive history is measurable in the stream. -/
141 theorem measurable_armStreamHistory
142     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat) :
143     Measurable (fun stream : ArmRewardStream K =>
144       armStreamHistory hK c stream n) := by
145   induction n with
146   | zero =>
147       have hcoord : Measurable (fun stream : ArmRewardStream K =>
148           stream 0 (initializationArm hK 0)) :=
149         (measurable_pi_apply (initializationArm hK 0)).comp
150           (measurable_pi_apply 0)
151       refine measurable_pi_lambda _ ?_
152       intro _i
153       show Measurable (fun stream : ArmRewardStream K =>
154         (initializationArm hK 0, stream 0 (initializationArm hK 0)))
155       exact measurable_const.prodMk hcoord
156   | succ n ih =>
157       have harm : Measurable (fun stream : ArmRewardStream K =>
158           realHistoryNextArm hK c n (armStreamHistory hK c stream n)) :=
159         (measurable_realHistoryNextArm hK c n).comp ih
160       have hcountEval : Measurable
161           (fun input :
162               History.FinitePairHistory (Fin K) Real n × Fin K =>
163             ETC.realHistoryPullCount n input.1 input.2) := by
164         apply measurable_from_prod_countable_left
165         intro arm
166         exact measurable_realHistoryPullCount n arm
167       have hhistoryArm : Measurable (fun stream : ArmRewardStream K =>
168           (armStreamHistory hK c stream n,
169             realHistoryNextArm hK c n
170               (armStreamHistory hK c stream n))) :=
171         ih.prodMk harm
172       have hcount : Measurable (fun stream : ArmRewardStream K =>
173           ETC.realHistoryPullCount n (armStreamHistory hK c stream n)
174             (realHistoryNextArm hK c n
175               (armStreamHistory hK c stream n))) :=
176         by simpa only [Function.comp_apply] using
177           hcountEval.comp hhistoryArm
178       have hcountArm : Measurable (fun stream : ArmRewardStream K =>
179           (ETC.realHistoryPullCount n (armStreamHistory hK c stream n)
180               (realHistoryNextArm hK c n
181                 (armStreamHistory hK c stream n)),
182             realHistoryNextArm hK c n
183               (armStreamHistory hK c stream n))) :=
184         hcount.prodMk harm
185       have hstreamCountArm : Measurable (fun stream : ArmRewardStream K =>
186           (stream,
187             (ETC.realHistoryPullCount n (armStreamHistory hK c stream n)
188                 (realHistoryNextArm hK c n
189                   (armStreamHistory hK c stream n)),
190               realHistoryNextArm hK c n
191                 (armStreamHistory hK c stream n)))) :=
192         measurable_id.prodMk hcountArm
193       have hreward : Measurable (fun stream : ArmRewardStream K =>
194           stream
195             (ETC.realHistoryPullCount n (armStreamHistory hK c stream n)
196               (realHistoryNextArm hK c n
197                 (armStreamHistory hK c stream n)))
198             (realHistoryNextArm hK c n
199               (armStreamHistory hK c stream n))) :=
200         by simpa only [Function.comp_apply] using
201           measurable_armRewardStream_apply.comp hstreamCountArm
202       have hnext : Measurable (fun stream : ArmRewardStream K =>
203           (realHistoryNextArm hK c n
204               (armStreamHistory hK c stream n),
205             stream
206               (ETC.realHistoryPullCount n (armStreamHistory hK c stream n)
207                 (realHistoryNextArm hK c n
208                   (armStreamHistory hK c stream n)))
209               (realHistoryNextArm hK c n
210                 (armStreamHistory hK c stream n)))) :=
211         harm.prodMk hreward
212       simpa only [armStreamHistory] using
213         History.measurable_extendPairHistorySucc.comp
214           (ih.prodMk hnext)

216 /-- Every action coordinate of the recursive arm-stream process is measurable. -/
217 theorem measurable_armStreamAction
218     {K : Nat} (hK : 0 < K) (c : Real) (t : Nat) :
219     Measurable (fun stream : ArmRewardStream K =>
220       armStreamAction hK c stream t) := by
221   exact (measurable_fst.comp
222     (measurable_pi_apply
223       (⟨t, Finset.mem_Iic.mpr le_rfl⟩ : Finset.Iic t))).comp
224         (measurable_armStreamHistory hK c t)

226 /-- Every reward coordinate of the recursive arm-stream process is measurable. -/
227 theorem measurable_armStreamReward
228     {K : Nat} (hK : 0 < K) (c : Real) (t : Nat) :
229     Measurable (fun stream : ArmRewardStream K =>
230       armStreamReward hK c stream t) := by
231   have heq :
232       (fun stream : ArmRewardStream K => armStreamReward hK c stream t) =
233         (fun stream : ArmRewardStream K =>
234           (armStreamHistory hK c stream t
235             ⟨t, Finset.mem_Iic.mpr le_rfl⟩).2) := by
236     funext stream
237     rw [armStreamHistory_eq_finitePairHistoryOfTrace]
238     rfl
239   rw [heq]
240   exact (measurable_snd.comp
241     (measurable_pi_apply
242       (⟨t, Finset.mem_Iic.mpr le_rfl⟩ : Finset.Iic t))).comp
243         (measurable_armStreamHistory hK c t)

275 /-- Stationary product law: one independent time stream for every arm law. -/
276 noncomputable def armStreamMeasure
277     {K : Nat} (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] :
278     Measure (ArmRewardStream K) :=
279   Measure.infinitePi fun _ : Nat => Measure.infinitePi fun arm : Fin K => nu arm
