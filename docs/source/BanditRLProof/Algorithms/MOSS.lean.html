4 # Source MOSS index and deterministic selection step
5 
6 Algorithm 7 in Lattimore--Szepesvari, *Bandit Algorithms*, uses a horizon
7 dependent, realized-pull-count index. These definitions keep its factor four
8 and log-plus truncation. They do not yet construct a stochastic history law
9 or prove Theorem 9.1's regret bound.
10 -/
11 
12 namespace BanditRLProof.MOSS
13 
14 /-- Source convention `log max {1,x}`. -/
15 noncomputable def logPlus (x : ℝ) : ℝ := Real.log (max 1 x)
16 
17 /-- Source confidence radius at sample count `s`. Real division totalizes
18 `s=0`; the post-initialization algorithm must be used on positive counts. -/
19 noncomputable def radius (n k s : ℕ) : ℝ :=
20   Real.sqrt (4 / (s : ℝ) * logPlus ((n : ℝ) / ((k : ℝ) * (s : ℝ))))
21 
22 /-- Source index for arbitrary current empirical means and pull counts. -/
23 noncomputable def index {k : ℕ} (n : ℕ) (empiricalMean : Fin k → ℝ)
24     (pulls : Fin k → ℕ) (a : Fin k) : ℝ :=
25   empiricalMean a + radius n k (pulls a)
26 
27 /-- Zero-based Algorithm 7 action: first each arm once, then a real argmax.
28 This accepts a state; consistency of that state with past feedback is a
29 separate history-level obligation. -/
30 noncomputable def action {k : ℕ} (hk : 0 < k) (n t : ℕ)
31     (empiricalMean : Fin k → ℝ) (pulls : Fin k → ℕ) : Fin k :=
32   if ht : t < k then ⟨t, ht⟩ else UCB.scoreArgmax hk (index n empiricalMean pulls)
33 
34 theorem logPlus_nonneg (x : ℝ) : 0 ≤ logPlus x :=
35   Real.log_nonneg (le_max_left 1 x)
36 
37 theorem radius_nonneg (n k s : ℕ) : 0 ≤ radius n k s :=
38   Real.sqrt_nonneg _
39 
40 @[simp] theorem radius_zero (n k : ℕ) : radius n k 0 = 0 := by
41   simp [radius]
42 
43 /-- Exact squared source radius, including the totalized zero-count branch. -/
44 theorem radius_sq (n k s : ℕ) :
45     radius n k s ^ 2 =
46       4 / (s : ℝ) * logPlus ((n : ℝ) / ((k : ℝ) * (s : ℝ))) := by
47   apply Real.sq_sqrt
48   exact mul_nonneg (div_nonneg (by norm_num) (Nat.cast_nonneg s))
49     (logPlus_nonneg _)

51 @[simp] theorem action_of_lt {k : ℕ} (hk : 0 < k) (n t : ℕ)
52     (empiricalMean : Fin k → ℝ) (pulls : Fin k → ℕ) (ht : t < k) :
53     action hk n t empiricalMean pulls = ⟨t, ht⟩ := by
54   simp [action, ht]
55 
56 /-- Initialization selects each source arm at its own zero-based time. -/
57 theorem action_initial_arm {k : ℕ} (hk : 0 < k) (n : ℕ)
58     (empiricalMean : Fin k → ℝ) (pulls : Fin k → ℕ) (a : Fin k) :
59     action hk n a.val empiricalMean pulls = a := by
60   simp [action, a.isLt]
61 
62 theorem action_index_max {k : ℕ} (hk : 0 < k) (n t : ℕ)
63     (empiricalMean : Fin k → ℝ) (pulls : Fin k → ℕ)
64     (ht : k ≤ t) (a : Fin k) :
65     index n empiricalMean pulls a ≤
66       index n empiricalMean pulls (action hk n t empiricalMean pulls) := by
67   simpa [action, not_lt.mpr ht] using
68     UCB.scoreArgmax_spec hk (index n empiricalMean pulls) a

70 /-- The large-gap selection implication in the proof of Theorem 9.1.
71 The optimism-deficit bound is explicit and is not a concentration theorem. -/
72 theorem selected_index_gt_mean_add_half_gap {k : ℕ} (hk : 0 < k)
73     (n t : ℕ) (mean empiricalMean : Fin k → ℝ) (pulls : Fin k → ℕ)
74     (best chosen : Fin k) (deficit : ℝ) (ht : k ≤ t)
75     (hselected : action hk n t empiricalMean pulls = chosen)
76     (hoptimism : mean best - deficit ≤ index n empiricalMean pulls best)
77     (hgap : 2 * deficit < mean best - mean chosen) :
78     mean chosen + (mean best - mean chosen) / 2 <
79       index n empiricalMean pulls chosen := by
80   have hmax := action_index_max hk n t empiricalMean pulls ht best
81   rw [hselected] at hmax
82   linarith
