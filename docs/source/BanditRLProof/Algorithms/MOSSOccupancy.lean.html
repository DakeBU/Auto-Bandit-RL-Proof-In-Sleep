8 def fixedLogRadius (δ gap : ℝ) (s : ℕ) : ℝ :=
9   sqrt (4/(s : ℝ)*logPlus (gap^2/δ))
10 
11 theorem sampleRadius_le_fixedLogRadius (δ gap : ℝ) (s : ℕ)
12     (hδ : 0 < δ) (hg : 0 < gap) (hs : 0 < s) (hlarge : 1 ≤ (s : ℝ)*gap^2) :
13     sqrt (4/(s : ℝ)*logPlus (1/((s : ℝ)*δ))) ≤ fixedLogRadius δ gap s := by
14   have hsR : 0 < (s : ℝ) := Nat.cast_pos.mpr hs
15   apply sqrt_le_sqrt
16   apply mul_le_mul_of_nonneg_left (logPlus_mono ?_) (by positivity)
17   apply (div_le_div_iff₀ (mul_pos hsR hδ) hδ).mpr
18   nlinarith [mul_le_mul_of_nonneg_right hlarge hδ.le]

20 def indexExceedanceCount (mean : ℕ → ℝ) (δ gap : ℝ) (n : ℕ) : ℝ :=
21   ∑ s ∈ range n, if gap/2 ≤ mean (s+1) +
22     sqrt (4/((s+1 : ℕ) : ℝ)*logPlus (1/(((s+1 : ℕ) : ℝ)*δ))) then 1 else 0

24 /-- Deterministic count transport; the MOSS policy event is a separate obligation. -/
25 theorem pullCount_le_one_add_indexExceedanceCount {Action : Type*} [DecidableEq Action]
26     (action : ActionTrace Action) (a : Action) (mean : ℕ → ℝ) (δ gap : ℝ) (n : ℕ)
27     (hselected : ∀ t < n, action t = a → 0 < pullCount action a t →
28       gap/2 ≤ mean (pullCount action a t) +
29         sqrt (4/(pullCount action a t : ℝ)*logPlus (1/((pullCount action a t : ℝ)*δ)))) :
30     (pullCount action a n : ℝ) ≤ 1 + indexExceedanceCount mean δ gap n := by
31   classical
32   exact pullCount_le_one_add_eventCount action a
33     (fun s => gap/2 ≤ mean s + sqrt (4/(s : ℝ)*logPlus (1/((s : ℝ)*δ)))) n hselected

35 def fixedLogExceedanceCount (mean : ℕ → ℝ) (δ gap : ℝ) (n : ℕ) : ℝ :=
36   ∑ s ∈ range n, if gap/2 ≤ mean (s+1) + fixedLogRadius δ gap (s+1) then 1 else 0

38 def smallSampleCount (gap : ℝ) (n : ℕ) : ℝ :=
39   ∑ s ∈ range n, if ((s+1 : ℕ) : ℝ)*gap^2 < 1 then 1 else 0

41 /-- Source correction step before applying Lemma 8.2, with no stochastic premise. -/
42 theorem indexExceedanceCount_le_small_add_fixed (mean : ℕ → ℝ)
43     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) (n : ℕ) :
44     indexExceedanceCount mean δ gap n ≤
45       smallSampleCount gap n + fixedLogExceedanceCount mean δ gap n := by
46   unfold indexExceedanceCount smallSampleCount fixedLogExceedanceCount
47   rw [← Finset.sum_add_distrib]
48   apply Finset.sum_le_sum
49   intro s hs
50   by_cases hsmall : ((s+1 : ℕ) : ℝ)*gap^2 < 1
51   · simp only [hsmall, ite_true]
52     split_ifs <;> norm_num
53   · have hr := sampleRadius_le_fixedLogRadius δ gap (s+1) hδ hg (by omega)
54       (le_of_not_gt hsmall)
55     simp only [hsmall, ite_false, zero_add]
56     split_ifs <;> norm_num <;> linarith

58 theorem smallSampleCount_le_horizon (gap : ℝ) (n : ℕ) :
59     smallSampleCount gap n ≤ (n : ℝ) := by
60   unfold smallSampleCount
61   calc
62     _ ≤ ∑ _s ∈ range n, (1 : ℝ) := Finset.sum_le_sum (fun s _ => by split_ifs <;> norm_num)
63     _ = _ := by simp

65 theorem smallSampleCount_le_inv_sq (gap : ℝ) (hg : 0 < gap) (n : ℕ) :
66     smallSampleCount gap n ≤ 1/gap^2 := by
67   have hg2 : 0 < gap^2 := sq_pos_of_pos hg
68   induction n with
69   | zero => simpa only [smallSampleCount, Finset.range_zero, Finset.sum_empty] using
70       (le_of_lt (div_pos zero_lt_one hg2))
71   | succ n ih =>
72     by_cases h : ((n+1 : ℕ) : ℝ)*gap^2 < 1
73     · exact (smallSampleCount_le_horizon gap (n+1)).trans
74         ((le_div_iff₀ hg2).mpr h.le)
75     · unfold smallSampleCount
76       rw [Finset.sum_range_succ, if_neg h, add_zero]
77       exact ih

79 theorem indexExceedanceCount_le_inv_sq_add_fixed (mean : ℕ → ℝ)
80     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) (n : ℕ) :
81     indexExceedanceCount mean δ gap n ≤
82       1/gap^2 + fixedLogExceedanceCount mean δ gap n :=
83   (indexExceedanceCount_le_small_add_fixed mean δ gap hδ hg n).trans
