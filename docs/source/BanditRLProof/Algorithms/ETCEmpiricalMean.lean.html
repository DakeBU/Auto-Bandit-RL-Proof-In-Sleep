7 /-!
8 # ETC empirical means
9 
10 This module defines the deterministic empirical mean for a fixed-commit ETC
11 trace at the configured exploration horizon. It deliberately stays below
12 probability, measurability, and concentration assumptions.
13 -/
14 
15 namespace BanditRLProof
16 namespace ETC
17 
18 /--
19 Empirical mean of arm `a` at the configured ETC exploration horizon for a
20 fixed-commit trace.
21 
22 This is the `ETC-EMP-MEAN-ACTION-WITH-COMMIT-EXPLORATION` project-local leaf.
23 -/
24 def empMeanAtExploration {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K)
25     (reward : RewardTrace Rat) (a : Fin K) : Rat :=
26   sumRewards (ETC.actionWithCommit spec commitArm) reward a
27       (spec.explorationPulls * K) /
28     (pullCount (ETC.actionWithCommit spec commitArm) a
29       (spec.explorationPulls * K) : Rat)
30 
31 /--
32 Exploration empirical means depend only on the reward coordinates observed
33 before the configured exploration horizon.
34 
35 This is the prefix-congruence bridge needed to reconstruct the ETC commit
36 score from a finite reward history after exploration, rather than from an
37 ambient reward trace with future coordinates.
38 -/
39 theorem empMeanAtExploration_eq_of_eq_on_prefix
40     {K : Nat}
41     (spec : ETC.Spec K) (commitArm : Fin K)
42     (reward0 reward1 : RewardTrace Rat)
43     (hprefix :
44       forall t, t < spec.explorationPulls * K -> reward0 t = reward1 t)
45     (a : Fin K) :
46     ETC.empMeanAtExploration spec commitArm reward0 a =
47       ETC.empMeanAtExploration spec commitArm reward1 a := by
48   have hsum :
49       sumRewards (ETC.actionWithCommit spec commitArm) reward0 a
50           (spec.explorationPulls * K) =
51         sumRewards (ETC.actionWithCommit spec commitArm) reward1 a
52           (spec.explorationPulls * K) := by
53     rw [sumRewards_eq_finset_filter_sum, sumRewards_eq_finset_filter_sum]
54     apply Finset.sum_congr rfl
55     intro t ht
56     exact hprefix t (Finset.mem_range.mp (Finset.mem_filter.mp ht).1)
57   simp [ETC.empMeanAtExploration, hsum]
58 
59 /--
60 Once a finite reward history reaches the configured exploration horizon, its
61 default-completed trace gives the same ETC exploration empirical mean as the
62 ambient reward trace.
63 
64 This is the history-reconstruction bridge for a later generated commit policy:
65 at generated action time `t + 1`, the state reads a history through `t`, so the
66 explicit `spec.explorationPulls * K <= t + 1` contract supplies every score
67 coordinate used by the commit rule.
68 -/
69 theorem empMeanAtExploration_completeRewardTrace_eq_of_explorationHorizon_le
70     {K : Nat}
71     (spec : ETC.Spec K) (commitArm : Fin K)
72     (reward : RewardTrace Rat) (t : Nat)
73     (horizon_le : spec.explorationPulls * K <= t + 1)
74     (a : Fin K) :
75     ETC.empMeanAtExploration spec commitArm
76       (History.completeRewardTrace t
77         (History.finiteRewardHistoryOfTrace reward t) (0 : Rat)) a =
78       ETC.empMeanAtExploration spec commitArm reward a := by
79   apply ETC.empMeanAtExploration_eq_of_eq_on_prefix
80   intro s hs
81   apply History.completeRewardTrace_finiteRewardHistoryOfTrace_apply_of_le
82   exact Nat.lt_succ_iff.mp (lt_of_lt_of_le hs horizon_le)
83 
84 /--
85 The empirical-mean denominator at the ETC exploration horizon rewrites to the
86 configured number of exploration pulls per arm.
87 -/
88 theorem empMeanAtExploration_eq_sumRewards_div_explorationPulls
89     {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K)
90     (reward : RewardTrace Rat) (a : Fin K) :
91     ETC.empMeanAtExploration spec commitArm reward a =
92       sumRewards (ETC.actionWithCommit spec commitArm) reward a
93           (spec.explorationPulls * K) /
94         ((spec.explorationPulls : Nat) : Rat) := by
95   simp [ETC.empMeanAtExploration,
96     ETC.pullCount_actionWithCommit_explorationPulls_mul_K_eq]
97 
98 /--
99 At a positive exploration count, comparing two fixed-commit ETC empirical
100 means is equivalent to comparing their fixed-horizon reward sums.
101 
102 This is the `ETC-EMP-MEAN-COMPARISON-AS-FINITE-SUM` deterministic algebra
103 leaf.  It only removes the common positive denominator from two empirical
104 means; it does not introduce probability, concentration, filtration, or final
105 ETC regret.
106 -/
107 theorem empMeanAtExploration_le_iff_sumRewards_le_of_explorationPulls_pos
108     {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K)
109     (reward : RewardTrace Rat) (a b : Fin K)
110     (hexplorationPulls_pos : 0 < spec.explorationPulls) :
111     ETC.empMeanAtExploration spec commitArm reward b <=
112       ETC.empMeanAtExploration spec commitArm reward a ↔
113     sumRewards (ETC.actionWithCommit spec commitArm) reward b
114         (spec.explorationPulls * K) <=
115       sumRewards (ETC.actionWithCommit spec commitArm) reward a
116         (spec.explorationPulls * K) := by
117   have hc : (0 : Rat) < ((spec.explorationPulls : Nat) : Rat) := by
118     exact_mod_cast hexplorationPulls_pos
119   rw [ETC.empMeanAtExploration_eq_sumRewards_div_explorationPulls]
120   rw [ETC.empMeanAtExploration_eq_sumRewards_div_explorationPulls]
121   exact div_le_div_iff_of_pos_right hc
122 
123 /--
124 At a positive exploration count, a pointwise implication from the fixed-horizon
125 reward-sum comparison into a real finite-sum tail event yields the matching
126 event inclusion from the non-best empirical-mean comparison event.
127 
128 This is the `ETC-EMPMEAN-EVENT-SUBSET-SUMREWARDS-TAIL-EVENT` bridge.  It is
129 only an event-shape adapter; it does not instantiate centered reward
130 differences, prove sub-Gaussianity, introduce filtrations, or prove final ETC
131 regret.
132 -/
133 theorem empMeanAtExploration_ge_best_event_subset_sumRewards_tail_event_of_imp
134     {Omega : Type u} {K : Nat}
135     (spec : ETC.Spec K)
136     (model : FiniteBanditModel K)
137     (commitArm : Fin K)
138     (reward : Omega -> RewardTrace Rat)
139     (a : Fin K)
140     (hexplorationPulls_pos : 0 < spec.explorationPulls)
141     {Idx : Type v}
142     (idx : Finset Idx)
143     (X : Idx -> Omega -> Real)
144     (eps : Real)
145     (himp :
146       forall omega : Omega,
147         sumRewards (ETC.actionWithCommit spec commitArm) (reward omega)
148             model.bestArm (spec.explorationPulls * K) <=
149           sumRewards (ETC.actionWithCommit spec commitArm) (reward omega)
150             a (spec.explorationPulls * K) ->
151         eps <= idx.sum (fun i => X i omega)) :
152     Set.Subset
153       {omega : Omega |
154         ETC.empMeanAtExploration spec commitArm (reward omega) a >=
155           ETC.empMeanAtExploration spec commitArm (reward omega)
156             model.bestArm}
157       {omega : Omega | eps <= idx.sum (fun i => X i omega)} := by
158   intro omega hmean
159   exact himp omega
160     ((ETC.empMeanAtExploration_le_iff_sumRewards_le_of_explorationPulls_pos
161       spec commitArm (reward omega) a model.bestArm
162       hexplorationPulls_pos).1 hmean)
163 
164 end ETC
