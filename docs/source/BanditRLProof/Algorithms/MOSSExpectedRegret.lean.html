8 Identification with the common bandit history law is a separate theorem. -/
9 theorem integral_streamTrace_regret_le {Ω : Type*} [MeasurableSpace Ω]
10     (μ : Measure Ω) [IsProbabilityMeasure μ] {k : ℕ} (hk : 0 < k)
11     (n : ℕ) (hkn : k ≤ n) (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ)
12     (best : Fin k) (hbest : ∀ a, mean a ≤ mean best)
13     (hXm : ∀ a i, StronglyMeasurable (X a i))
14     (hind : ∀ a, iIndepFun (X a) μ)
15     (hmean : ∀ a i, ∫ ω, X a i ω ∂μ = 0)
16     (hsubG : ∀ a i, HasSubgaussianMGF (X a i) 1 μ) :
17     (∫ ω, realMeanRegret mean (streamTrace hk n mean X ω) n ∂μ) ≤
18       39*sqrt ((n : ℝ)*k) + ∑ a, (mean best-mean a) := by
19   classical
20   have hn : 0 < n := lt_of_lt_of_le hk hkn
21   have hnR : 0 < (n : ℝ) := Nat.cast_pos.mpr hn
22   have hkR : 0 < (k : ℝ) := Nat.cast_pos.mpr hk
23   let δ : ℝ := (k : ℝ)/(n : ℝ)
24   have hδ : 0 < δ := div_pos hkR hnR
25   let Z := optimismDeficit (X best) δ n
26   let f : Fin k → Ω → ℝ := fun a ω => if 8*sqrt δ ≤ mean best-mean a then
27     (mean best-mean a)*(1+indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n) else 0
28   have hi (a : Fin k) : Integrable (f a) μ := by
29     dsimp [f]
30     split_ifs
31     · exact ((integrable_const 1).add (integrable_indexExceedanceCount (X a) (hXm a) δ _ n)).const_mul _
32     · exact integrable_const 0
33   have hsum : Integrable (fun ω => ∑ a, f a ω) μ := integrable_finset_sum _ (fun a _ => hi a)
34   have hZ : Integrable Z μ := integrable_optimismDeficit (X best)
35     (fun i => (hsubG best i).integrable) δ n
36   have hbase : Integrable (fun ω => (8*sqrt δ+2*Z ω)*(n : ℝ)) μ :=
37     ((integrable_const _).add (hZ.const_mul 2)).mul_const _
38   have hb := integral_mono (integrable_streamTrace_regret μ hk n mean X hXm)
39     (hbase.add hsum) (fun ω => streamTrace_realMeanRegret_le hk n hkn mean X ω best hbest)
40   simp only [Pi.add_apply] at hb
41   rw [integral_add hbase hsum, integral_mul_const] at hb
42   rw [integral_add (integrable_const _) (hZ.const_mul 2), integral_const_mul] at hb
43   simp only [integral_const, probReal_univ, one_smul] at hb
44   rw [integral_const_mul] at hb
45   have hf := integral_largeGapCountSum_le μ mean X best hbest hXm hind hmean hsubG δ hδ n
46   have hz := twice_horizon_mul_integral_optimismDeficit_le (X best) (hXm best)
47     (hind best) (hmean best) (hsubG best) n k hn hk
48   have he : (n : ℝ)*sqrt δ = sqrt ((n : ℝ)*k) := by
49     calc
50       _ = sqrt ((n : ℝ)^2)*sqrt δ := by rw [sqrt_sq hnR.le]
51       _ = sqrt ((n : ℝ)^2*δ) := (sqrt_mul (sq_nonneg _) _).symm
52       _ = _ := by congr 1; dsimp [δ]; field_simp
53   have he' : (k : ℝ)/sqrt δ = sqrt ((n : ℝ)*k) := by
54     apply (div_eq_iff (ne_of_gt (sqrt_pos.mpr hδ))).mpr
55     rw [← he, mul_assoc, ← sq, sq_sqrt hδ.le]
56     dsimp [δ]
57     field_simp
58   have hscale : (k : ℝ)*(15/sqrt δ) = 15*sqrt ((n : ℝ)*k) := by
59     calc
60       _ = 15*((k : ℝ)/sqrt δ) := by ring
61       _ = _ := by rw [he']
62   change (∫ ω, ∑ a, f a ω ∂μ) ≤ _ at hf
63   change 2*(n : ℝ)*(∫ ω, Z ω ∂μ) ≤ _ at hz
64   rw [hscale] at hf
65   nlinarith
