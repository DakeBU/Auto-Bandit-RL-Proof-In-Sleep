19 /--
20 Canonical Real expected pseudo-regret bound for stationary finite-arm reward
21 laws with direct centered sub-Gaussian MGF witnesses.
22 
23 The UCB variance parameter is the maximum of the armwise proxies. At least one
24 proxy must be positive because the existing canonical UCB route requires a
25 strictly positive common proxy; zero proxies for other arms are allowed.
26 -/
27 theorem integral_real_pseudoRegret_selectedPolicySuccessorGeneratedUCBRegretAction_le_textbookGapSum_finiteArmSubgaussianLaws
28     {K : Nat}
29     (model : FiniteBanditModel K)
30     (armLaw : Fin K -> Measure Rat)
31     (hprob : forall arm, IsProbabilityMeasure (armLaw arm))
32     (varianceProxy : Fin K -> NNReal)
33     (hvariancePositive : exists arm,
34       0 < ((varianceProxy arm : NNReal) : Real))
35     (hmean : forall arm,
36       integral (armLaw arm) (fun reward : Rat => ((reward : Rat) : Real)) =
37         ((model.mean arm : Rat) : Real))
38     (hsubG : forall arm,
39       HasSubgaussianMGF
40         (fun reward : Rat =>
41           (((reward - model.mean arm : Rat) : Real)))
42         (varianceProxy arm) (armLaw arm))
43     (defaultAction : Fin K)
44     (T : Nat) (hT : 0 < T)
45     (delta : Real) (hdelta : 0 < delta) :
46     let sigma2 := Concentration.finiteArmVarianceProxy varianceProxy
47     let rewardKernel := RewardKernel.contextIndependentOfActionLaws
48       (Context := Unit) armLaw hprob
49     let context : (n : Nat) -> History.FiniteRewardHistory Rat n -> Unit :=
50       fun _ _ => ()
51     MeasureTheory.integral
52         (selectedPolicySuccessorRewardTrajMeasure model.hK
53           (armLaw defaultAction) rewardKernel context
54           (fun _ => measurable_const) sigma2 T delta defaultAction)
55         (fun trajectory : RewardTrace Rat =>
56           ((pseudoRegret model
57             (selectedPolicySuccessorGeneratedUCBRegretAction
58               model.hK sigma2 T delta defaultAction
59               (fun y : RewardTrace Rat => y) trajectory)
60             T : Rat) : Real)) <=
61       ((Finset.univ : Finset (Fin K)).filter (fun arm =>
62         0 < (((model.gap arm : Rat) : Real)))).sum (fun arm =>
63           selectedPolicySuccessorTextbookGapBudget K sigma2 T delta
64               (((model.gap arm : Rat) : Real)) +
65             (((model.gap arm : Rat) : Real)) * ((T : Real) * delta)) := by
66   let sigma2 := Concentration.finiteArmVarianceProxy varianceProxy
67   let rewardKernel := RewardKernel.contextIndependentOfActionLaws
68     (Context := Unit) armLaw hprob
69   let context : (n : Nat) -> History.FiniteRewardHistory Rat n -> Unit :=
70     fun _ _ => ()
71   have hcontext : forall n : Nat, Measurable (context n) := by
72     intro n
73     exact measurable_const
74   have hmeanMeas :
75       Measurable (fun pair : Unit × Fin K => model.mean pair.2) :=
76     (measurable_of_countable model.mean).comp measurable_snd
77   have hkernel :
78       RewardKernel.CenteredRewardKernelLaw rewardKernel
79         (fun _ arm => model.mean arm)
80         (fun _ arm => varianceProxy arm) := by

94       (rewardKernel := rewardKernel)
95       (context := context)

97       (varianceProxy := fun _ arm => varianceProxy arm)
98       (defaultAction := defaultAction)

102       (hkernel := hkernel)

104       (hT := hT)
105       (hsigma2 := hsigma2)
