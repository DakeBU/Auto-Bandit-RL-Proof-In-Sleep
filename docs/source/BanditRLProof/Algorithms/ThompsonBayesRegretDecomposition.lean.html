5 # Bayesian regret decomposition for the recursive Thompson trajectory
6 
7 This module ports the probability-matching algebra behind LML's
8 `TS.integral_regret_eq_add` to the locally generated Thompson trajectory.
9 The confidence score is abstract but must depend only on the visible history
10 and the candidate action.  Clipped-UCB concentration can therefore be attached
11 downstream without restoring an assumed sampler or posterior law.
12 -/

22 /-- A real score whose time-`n + 1` value sees exactly the history through `n`. -/
23 structure HistoryActionScore (Action : Type u) (Reward : Type v)
24     [MeasurableSpace Action] [MeasurableSpace Reward] where
25   initial : Action -> Real
26   successor :
27     (n : Nat) -> History.FinitePairHistory Action Reward n -> Action -> Real
28   measurable_initial : Measurable initial
29   measurable_successor :
30     forall n, Measurable (fun pair :
31       History.FinitePairHistory Action Reward n × Action =>
32         successor n pair.1 pair.2)

36 /-- Evaluate a history score on the action selected by one complete trace. -/
37 def atTrace
38     {Action : Type u} {Reward : Type v}
39     [MeasurableSpace Action] [MeasurableSpace Reward]
40     (score : HistoryActionScore Action Reward)
41     (action : ActionTrace Action) (reward : RewardTrace Reward) :
42     Nat -> Real
43   | 0 => score.initial (action 0)
44   | n + 1 => score.successor n
45       (History.finitePairHistoryOfTrace action reward n) (action (n + 1))

59 theorem measurable_atTrace
60     {Omega : Type w} {Action : Type u} {Reward : Type v}
61     [MeasurableSpace Omega] [MeasurableSpace Action] [MeasurableSpace Reward]
62     (score : HistoryActionScore Action Reward)
63     (action : Omega -> ActionTrace Action)
64     (reward : Omega -> RewardTrace Reward)
65     (haction : forall t, Measurable (fun omega => action omega t))
66     (hreward : forall t, Measurable (fun omega => reward omega t))
67     (t : Nat) :
68     Measurable (fun omega => score.atTrace (action omega) (reward omega) t) := by
69   cases t with
70   | zero =>
71       exact score.measurable_initial.comp (haction 0)
72   | succ n =>
73       exact (score.measurable_successor n).comp
74         ((History.measurable_finitePairHistoryOfTrace
75           action reward haction hreward n).prodMk (haction (n + 1)))

99 /-- Equal pushforwards give equal integrals of every measurable real score. -/
100 theorem integral_comp_eq_of_map_eq
101     {Omega : Type u} {Target : Type v}
102     [MeasurableSpace Omega] [MeasurableSpace Target]
103     (mu : Measure Omega)
104     (left right : Omega -> Target)
105     (hleft : Measurable left) (hright : Measurable right)
106     (hmap : mu.map left = mu.map right)
107     (score : Target -> Real) (hscore : Measurable score) :
108     integral mu (fun omega => score (left omega)) =
109       integral mu (fun omega => score (right omega)) := by
110   calc
111     _ = integral (mu.map left) score := by
112       rw [integral_map hleft.aemeasurable hscore.aestronglyMeasurable]
113     _ = integral (mu.map right) score := by rw [hmap]
114     _ = _ := by
115       rw [integral_map hright.aemeasurable hscore.aestronglyMeasurable]

117 /--
118 If two actions have the same conditional law given a history, every measurable
119 history/action score has the same expectation under those actions.
120 -/
121 theorem integral_historyAction_eq_of_condDistrib_ae_eq
122     {Omega : Type u} {History : Type v} {Action : Type w}
123     [MeasurableSpace Omega] [MeasurableSpace History]
124     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
125     (mu : Measure Omega) [IsFiniteMeasure mu]
126     (history : Omega -> History) (hhistory : Measurable history)
127     (action bestAction : Omega -> Action)
128     (haction : Measurable action) (hbestAction : Measurable bestAction)
129     (hcond : condDistrib action history mu =ᵐ[mu.map history]
130       condDistrib bestAction history mu)
131     (score : History × Action -> Real) (hscore : Measurable score) :
132     integral mu (fun omega => score (history omega, action omega)) =
133       integral mu (fun omega => score (history omega, bestAction omega)) := by
134   have hpair :
135       mu.map (fun omega => (history omega, action omega)) =
136         mu.map (fun omega => (history omega, bestAction omega)) := by
137     rw [← compProd_map_condDistrib haction.aemeasurable,
138       ← compProd_map_condDistrib hbestAction.aemeasurable]
139     exact Measure.compProd_congr hcond
140   exact integral_comp_eq_of_map_eq mu
141     (fun omega => (history omega, action omega))
142     (fun omega => (history omega, bestAction omega))
143     (hhistory.prodMk haction) (hhistory.prodMk hbestAction) hpair score hscore

145 /-- The environment-indexed trajectory kernel has the algorithm's action law at time zero. -/
146 theorem canonicalMeasurableEnvironmentTrajectoryKernel_map_action_zero
147     {Env : Type u} {Action : Type v} {Reward : Type w}
148     [MeasurableSpace Env] [MeasurableSpace Action] [MeasurableSpace Reward]
149     [Nonempty Action] [Nonempty Reward]
150     (algorithm : HistoryAlgorithm Action Reward)
151     (environment : MeasurableHistoryEnvironment Env Action Reward) :
152     (canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment).map
153         (fun trajectory => (trajectory 0).1) =
154       Kernel.const Env algorithm.initialAction := by
155   change
156     (canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment).map
157         (Prod.fst ∘ fun trajectory => trajectory 0) =
158       Kernel.const Env algorithm.initialAction
159   have hevalZero : Measurable
160       (fun trajectory : (n : Nat) -> Action × Reward => trajectory 0) :=
161     measurable_pi_apply 0
162   rw [Kernel.map_comp_right _ hevalZero measurable_fst,
163     canonicalMeasurableEnvironmentTrajectoryKernel_map_eval_zero,
164     ← Kernel.fst_eq, measurableEnvironmentInitialPairKernel,
165     Kernel.fst_compProd]

167 /-- Mixing the trajectory kernel through a probability prior preserves its initial action law. -/
168 theorem canonicalMeasurableEnvironmentTrajectoryMeasure_map_action_zero
169     {Env : Type u} {Action : Type v} {Reward : Type w}
170     [MeasurableSpace Env] [MeasurableSpace Action] [MeasurableSpace Reward]
171     [Nonempty Action] [Nonempty Reward]
172     (prior : Measure Env) [IsProbabilityMeasure prior]
173     (algorithm : HistoryAlgorithm Action Reward)
174     (environment : MeasurableHistoryEnvironment Env Action Reward) :
175     (prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel
176       algorithm environment).map
177         (fun sample => (sample.2 0).1) =
178       algorithm.initialAction := by
179   let trajectory :=
180     canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment
181   let actionZero := fun trace : (n : Nat) -> Action × Reward => (trace 0).1
182   have hactionZero : Measurable actionZero :=
183     measurable_fst.comp (measurable_pi_apply 0)
184   calc
185     (prior ⊗ₘ trajectory).map (actionZero ∘ Prod.snd) =
186         ((prior ⊗ₘ trajectory).map Prod.snd).map actionZero := by
187       rw [Measure.map_map (g := actionZero) (f := Prod.snd)
188         hactionZero measurable_snd]
189     _ = (trajectory ∘ₘ prior).map actionZero := by
190       change (prior ⊗ₘ trajectory).snd.map actionZero =
191         (trajectory ∘ₘ prior).map actionZero
192       rw [Measure.snd_compProd]
193     _ = (trajectory.map actionZero) ∘ₘ prior := by
194       rw [Measure.map_comp prior trajectory hactionZero]
195     _ = Kernel.const Env algorithm.initialAction ∘ₘ prior := by
196       rw [canonicalMeasurableEnvironmentTrajectoryKernel_map_action_zero]
197     _ = algorithm.initialAction := by
198       rw [Measure.const_comp, measure_univ, one_smul]

200 /-- The initial action and latent best action have the same marginal on the actual TS trajectory. -/
201 theorem uniformReferenceThompsonAlgorithm_map_action_zero_eq_bestAction

203     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]

265 Probability matching on the actual recursive trajectory implies equality of
266 every visible-history score at the selected and latent-best actions.

268 theorem uniformReferenceThompsonAlgorithm_integral_historyScore_eq_bestAction
269     {Env : Type u} {Action : Type v} {Reward : Type w}
270     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
271     [MeasurableSpace Action] [StandardBorelSpace Action]
272     [Fintype Action] [Nonempty Action]
273     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
274     (prior : Measure Env) [IsProbabilityMeasure prior]
275     (environment : MeasurableHistoryEnvironment Env Action Reward)
276     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
277     (score : HistoryActionScore Action Reward) (t : Nat) :
278     let algorithm :=
279       uniformReferenceThompsonAlgorithm prior environment bestAction hbestAction
280     let actualMeasure :=
281       prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment
282     integral actualMeasure (fun sample =>
283         trajectoryHistoryScore score sample t) =
284       integral actualMeasure (fun sample =>
285         trajectoryBestHistoryScore score bestAction sample t) := by
286   dsimp only
287   let algorithm :=
288     uniformReferenceThompsonAlgorithm prior environment bestAction hbestAction
289   let actualMeasure :=
290     prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment
291   cases t with
292   | zero =>
293       exact integral_comp_eq_of_map_eq actualMeasure
294         (fun sample => environmentTrajectoryAction sample 0)
295         (bestAction ∘ Prod.fst)
296         (measurable_environmentTrajectoryAction_apply 0)
297         (hbestAction.comp measurable_fst)
298         (uniformReferenceThompsonAlgorithm_map_action_zero_eq_bestAction
299           prior environment bestAction hbestAction)
300         score.initial score.measurable_initial
301   | succ n =>
302       let history := fun sample :
303           Env × ((n : Nat) -> Action × Reward) =>
304         History.finitePairHistoryOfTrace
305           (environmentTrajectoryAction sample)
306           (environmentTrajectoryReward sample) n
307       have hhistory : Measurable history :=
308         History.measurable_finitePairHistoryOfTrace
309           environmentTrajectoryAction environmentTrajectoryReward
310           measurable_environmentTrajectoryAction_apply
311           measurable_environmentTrajectoryReward_apply n
312       have hnextAction : Measurable
313           (fun sample : Env × ((n : Nat) -> Action × Reward) =>
314             environmentTrajectoryAction sample (n + 1)) :=
315         measurable_environmentTrajectoryAction_apply (n + 1)
316       have hbest : Measurable
317           (fun sample : Env × ((n : Nat) -> Action × Reward) =>
318             bestAction sample.1) :=
319         hbestAction.comp measurable_fst
320       simpa [trajectoryHistoryScore, trajectoryBestHistoryScore,
321         HistoryActionScore.atTrace, HistoryActionScore.atBestTrace, history,
322         actualMeasure, algorithm] using
323         (integral_historyAction_eq_of_condDistrib_ae_eq
324           (Omega := Env × ((n : Nat) -> Action × Reward))
325           (History := History.FinitePairHistory Action Reward n)
326           (Action := Action) actualMeasure
327           history hhistory
328           (fun sample : Env × ((n : Nat) -> Action × Reward) =>
329             environmentTrajectoryAction sample (n + 1))
330           (fun sample : Env × ((n : Nat) -> Action × Reward) =>
331             bestAction sample.1) hnextAction hbest
332           (uniformReferenceThompsonAlgorithm_trajectory_condDistrib_action_ae_eq_bestAction
333             prior environment bestAction hbestAction n)
334           (fun pair : History.FinitePairHistory Action Reward n × Action =>
335             score.successor n pair.1 pair.2)
336           (score.measurable_successor n))

338 /-- A selector is mean-optimal when it maximizes the environment-dependent
339 action mean pointwise.  This contract separates a genuine best-action regret
340 interpretation from the comparator-relative algebra below. -/
341 def IsOptimalMeanSelector
342     {Env : Type u} {Action : Type v}
343     (mean : Env -> Action -> Real) (bestAction : Env -> Action) : Prop :=
344   forall env arm, mean env arm <= mean env (bestAction env)
345 
346 /-- Finite-horizon comparator-relative mean regret.  It is Bayesian
347 pseudo-regret when `bestAction` satisfies `IsOptimalMeanSelector mean
348 bestAction` and the environment is integrated against a prior. -/
349 def trajectoryBayesMeanRegret
350     {Env : Type u} {Action : Type v} {Reward : Type w}
351     (mean : Env -> Action -> Real) (bestAction : Env -> Action)
352     (sample : Env × ((n : Nat) -> Action × Reward)) (horizon : Nat) : Real :=
353   ∑ t ∈ range horizon,
354     (mean sample.1 (bestAction sample.1) -
355       mean sample.1 (environmentTrajectoryAction sample t))

357 /--
358 LML-shaped Thompson Bayesian regret decomposition on the locally generated
359 recursive trajectory.  Integrability is explicit; bounded clipped scores and
360 bounded action means can discharge these four contracts downstream.
361 -/
362 theorem integral_trajectoryBayesMeanRegret_eq_add_historyScore
363     {Env : Type u} {Action : Type v} {Reward : Type w}
364     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
365     [MeasurableSpace Action] [StandardBorelSpace Action]
366     [Fintype Action] [Nonempty Action]
367     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
368     (prior : Measure Env) [IsProbabilityMeasure prior]
369     (environment : MeasurableHistoryEnvironment Env Action Reward)
370     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
371     (mean : Env -> Action -> Real)
372     (score : HistoryActionScore Action Reward)
373     (horizon : Nat)
374     (hmeanBest : Integrable (fun sample :
375       Env × ((n : Nat) -> Action × Reward) =>
376         mean sample.1 (bestAction sample.1))
377       (prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel
378         (uniformReferenceThompsonAlgorithm
379           prior environment bestAction hbestAction) environment))
380     (hmeanAction : forall t : Nat, t < horizon -> Integrable (fun sample :
381       Env × ((n : Nat) -> Action × Reward) =>
382         mean sample.1 (environmentTrajectoryAction sample t))
383       (prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel
384         (uniformReferenceThompsonAlgorithm
385           prior environment bestAction hbestAction) environment))
386     (hscoreAction : forall t : Nat, t < horizon -> Integrable (fun sample :
387       Env × ((n : Nat) -> Action × Reward) =>
388         trajectoryHistoryScore score sample t)
389       (prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel
390         (uniformReferenceThompsonAlgorithm
391           prior environment bestAction hbestAction) environment))
392     (hscoreBest : forall t : Nat, t < horizon -> Integrable (fun sample :
393       Env × ((n : Nat) -> Action × Reward) =>
394         trajectoryBestHistoryScore score bestAction sample t)
395       (prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel
396         (uniformReferenceThompsonAlgorithm
397           prior environment bestAction hbestAction) environment)) :

399       uniformReferenceThompsonAlgorithm prior environment bestAction hbestAction

402     integral actualMeasure (fun sample =>
403         trajectoryBayesMeanRegret mean bestAction sample horizon) =
404       integral actualMeasure (fun sample =>
405         ∑ t ∈ range horizon,

407             trajectoryBestHistoryScore score bestAction sample t)) +
408       integral actualMeasure (fun sample =>
409         ∑ t ∈ range horizon,
410           (trajectoryHistoryScore score sample t -
411             mean sample.1 (environmentTrajectoryAction sample t))) := by

418       integral actualMeasure (fun sample =>
419           trajectoryHistoryScore score sample t) =
420         integral actualMeasure (fun sample =>
421           trajectoryBestHistoryScore score bestAction sample t) := by
422     exact uniformReferenceThompsonAlgorithm_integral_historyScore_eq_bestAction

431   have hbestTerm (t : Nat) (ht : t ∈ range horizon) :

434           mean sample.1 (bestAction sample.1) -

436         actualMeasure :=

441           trajectoryHistoryScore score sample t -

443         actualMeasure :=
444     (hscoreAction t (mem_range.mp ht)).sub
445       (hmeanAction t (mem_range.mp ht))

447     integral actualMeasure (fun sample =>
448         trajectoryBayesMeanRegret mean bestAction sample horizon) =
449       ∑ t ∈ range horizon,

451           mean sample.1 (bestAction sample.1) -

453         simpa only [trajectoryBayesMeanRegret] using
