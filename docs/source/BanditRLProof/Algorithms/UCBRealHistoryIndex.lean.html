6 /-!
7 # Native Real UCB history index
8 
9 This module mirrors the path-dependent score used by the pinned LML UCB
10 route.  In particular, the confidence width divides by the realized pull
11 count.  It is therefore distinct from the earlier UCB surface whose proxy is
12 deterministic in the sample point.
13 -/

18 /-- Real empirical mean of one arm before time `n`. -/
19 noncomputable def realEmpiricalMean
20     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
21     (arm : Fin K) (n : Nat) : Real :=
22   sumRewards action reward arm n / (pullCount action arm n : Real)
23 
24 /-- LML-shaped path-dependent UCB width before time `n`. -/
25 noncomputable def realWidth
26     {K : Nat} (action : ActionTrace (Fin K)) (c : Real)
27     (arm : Fin K) (n : Nat) : Real :=
28   Real.sqrt
29     (2 * c * Real.log ((n + 1 : Nat) : Real) /
30       (pullCount action arm n : Real))
31 
32 /-- Real empirical mean plus the realized pull-count confidence width. -/
33 noncomputable def realIndex
34     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
35     (c : Real) (arm : Fin K) (n : Nat) : Real :=
36   realEmpiricalMean action reward arm n + realWidth action c arm n
37 
38 /-- Inclusive finite-history version of the LML UCB width. -/
39 noncomputable def realHistoryWidth
40     {K : Nat} (c : Real) (n : Nat)
41     (history : History.FinitePairHistory (Fin K) Real n)
42     (arm : Fin K) : Real :=
43   Real.sqrt
44     (2 * c * Real.log ((n + 2 : Nat) : Real) /
45       (ETC.realHistoryPullCount n history arm : Real))
46 
47 /-- Inclusive finite-history Real UCB score. -/
48 noncomputable def realHistoryIndex
49     {K : Nat} (c : Real) (n : Nat)
50     (history : History.FinitePairHistory (Fin K) Real n)
51     (arm : Fin K) : Real :=
52   ETC.realHistoryEmpMean n history arm +
53     realHistoryWidth c n history arm

55 /-- Least-encoded maximizer of the path-dependent Real UCB index. -/
56 noncomputable def realIndexAction
57     {K : Nat} (hK : 0 < K)
58     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
59     (c : Real) (n : Nat) : Fin K :=
60   ETC.realLeastEncodedArgmax hK
61     (fun arm => realIndex action reward c arm n)

63 /-- Least-encoded maximizer on an inclusive finite pair history. -/
64 noncomputable def realHistoryIndexAction
65     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat)
66     (history : History.FinitePairHistory (Fin K) Real n) : Fin K :=
67   ETC.realLeastEncodedArgmax hK
68     (fun arm => realHistoryIndex c n history arm)

70 /-- A fixed-arm pull count is measurable as a function of finite pair history. -/
71 theorem measurable_realHistoryPullCount
72     {K : Nat} (n : Nat) (arm : Fin K) :
73     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
74       ETC.realHistoryPullCount n history arm) := by
75   classical
76   unfold ETC.realHistoryPullCount
77   refine Finset.measurable_sum _ ?_
78   intro i _hi
79   exact Measurable.ite
80     (measurableSet_eq_fun
81       (measurable_fst.comp (measurable_pi_apply i)) measurable_const)
82     measurable_const measurable_const

98 /-- A fixed-arm empirical mean is measurable on finite pair histories. -/
99 theorem measurable_realHistoryEmpMean
100     {K : Nat} (n : Nat) (arm : Fin K) :
101     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
102       ETC.realHistoryEmpMean n history arm) := by
103   unfold ETC.realHistoryEmpMean
104   exact (measurable_realHistorySumRewards n arm).div
105     ((measurable_of_countable (fun count : Nat => (count : Real))).comp
106       (measurable_realHistoryPullCount n arm))

108 /-- The fixed-arm UCB width is measurable on finite pair histories. -/

111     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
112       realHistoryWidth c n history arm) := by
113   unfold realHistoryWidth
114   exact (measurable_const.div

118 /-- Every fixed-arm UCB score is measurable on finite pair histories. -/

120     {K : Nat} (c : Real) (n : Nat) (arm : Fin K) :
121     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
122       realHistoryIndex c n history arm) := by

126 /-- The least-encoded finite-history UCB selector is measurable. -/

137 /-- The trace empirical mean is measurable under timewise measurable data. -/
138 theorem measurable_realEmpiricalMean
139     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
140     [MeasurableSingletonClass (Fin K)]
141     (action : Omega -> ActionTrace (Fin K))
142     (reward : Omega -> RewardTrace Real)
143     (haction : forall t, Measurable (fun omega => action omega t))
144     (hreward : forall t, Measurable (fun omega => reward omega t))
145     (arm : Fin K) (n : Nat) :
146     Measurable (fun omega => realEmpiricalMean (action omega) (reward omega) arm n) := by
147   exact
148     (measurable_sumRewards action reward haction hreward arm n).div
149       (measurable_natCast_pullCount
150         (Beta := Real) action haction arm n)

179 /-- The least-encoded realized UCB index action maximizes every arm score. -/
180 theorem realIndexAction_spec
181     {K : Nat} (hK : 0 < K)
182     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
183     (c : Real) (n : Nat) (arm : Fin K) :
184     realIndex action reward c arm n <=
185       realIndex action reward c (realIndexAction hK action reward c n) n := by
186   exact
187     ETC.realLeastEncodedArgmax_spec hK
188       (fun candidate => realIndex action reward c candidate n) arm

215 /-- Inclusive finite-history widths agree with the trace width at `n + 1`. -/
216 theorem realHistoryWidth_finitePairHistoryOfTrace
217     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
218     (c : Real) (n : Nat) (arm : Fin K) :
219     realHistoryWidth c n
220         (History.finitePairHistoryOfTrace action reward n) arm =
221       realWidth action c arm (n + 1) := by

236 /-- The least-encoded history selector is exactly the trace selector. -/
237 theorem realHistoryIndexAction_finitePairHistoryOfTrace
238     {K : Nat} (hK : 0 < K)
239     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
240     (c : Real) (n : Nat) :
241     realHistoryIndexAction hK c n
242         (History.finitePairHistoryOfTrace action reward n) =
