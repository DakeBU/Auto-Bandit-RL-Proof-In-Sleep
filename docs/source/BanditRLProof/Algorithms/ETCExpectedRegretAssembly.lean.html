9 # ETC lower-integral and Bochner regret assembly
10 
11 This module lifts the pointwise wrong-commit regret bridge to the project's
12 expectation surfaces: an `ENNReal.ofReal` lower-integral surrogate and an
13 ordinary Real-valued Bochner integral wrapper.  It does not introduce
14 concentration, filtrations, or a final ETC theorem.
15 -/

25 /--
26 Lower-integral assembly for an `Omega`-indexed ETC commit selector.
27 
28 The theorem consumes a pointwise non-best gap bound and an abstract upper bound
29 `pWrong` on the wrong-commit event probability.  It is intentionally still an
30 `ENNReal.ofReal` lower-integral statement, matching the existing expectation
31 surface in the project.
32 -/
33 theorem lintegral_ofReal_pseudoRegret_actionWithCommit_choice_le_exploration_add_suffix_badGap_prob
34     {Omega : Type u} {K : Nat}
35     [MeasurableSpace Omega]
36     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
37     (spec : ETC.Spec K) (model : FiniteBanditModel K)
38     (commit : Omega -> Fin K) (r : Nat)
39     (badGapBound : Rat) (pWrong : ENNReal)
40     (hbadGap :
41       forall a : Fin K, (a = model.bestArm -> False) ->
42         model.gap a <= badGapBound)
43     (hmeas_wrong :
44       MeasurableSet {omega : Omega | commit omega = model.bestArm -> False})
45     (hprob_wrong :
46       mu {omega : Omega | commit omega = model.bestArm -> False} <= pWrong) :
47     MeasureTheory.lintegral mu
48       (fun omega : Omega =>
49         ENNReal.ofReal
50           (((pseudoRegret model
51               (ETC.actionWithCommit spec (commit omega))
52               (spec.explorationPulls * K + r) : Rat) : Real))) <=
53     ENNReal.ofReal
54       (((((Finset.univ : Finset (Fin K)).sum
55         (fun a : Fin K => model.gap a)) *
56         (((spec.explorationPulls : Nat) : Rat)) : Rat) : Real)) +
57     ENNReal.ofReal
58       ((((((r : Nat) : Rat) * badGapBound : Rat) : Real))) * pWrong := by
59   let wrongSet : Set Omega :=
60     {omega : Omega | commit omega = model.bestArm -> False}
61   let baseRat : Rat :=
62     ((Finset.univ : Finset (Fin K)).sum
63       (fun a : Fin K => model.gap a)) *
64       (((spec.explorationPulls : Nat) : Rat))
65   let suffixRat : Omega -> Rat :=
66     fun omega : Omega =>
67       (((r : Nat) : Rat) *
68         (if commit omega = model.bestArm then (0 : Rat) else badGapBound))
69   let base : ENNReal :=
70     ENNReal.ofReal (((baseRat : Rat) : Real))
71   let suffix : ENNReal :=
72     ENNReal.ofReal ((((((r : Nat) : Rat) * badGapBound : Rat) : Real)))
73   have hpoint :
74       (fun omega : Omega =>
75         ENNReal.ofReal
76           (((pseudoRegret model
77               (ETC.actionWithCommit spec (commit omega))
78               (spec.explorationPulls * K + r) : Rat) : Real)))
79         <=
80       (fun omega : Omega => base + wrongSet.indicator (fun _ => suffix) omega) := by
81     intro omega
82     have hrat :
83         pseudoRegret model (ETC.actionWithCommit spec (commit omega))
84             (spec.explorationPulls * K + r) <= baseRat + suffixRat omega := by
85       simpa [baseRat, suffixRat] using
86         ETC.pseudoRegret_actionWithCommit_choice_le_sum_gap_mul_explorationPulls_add_suffix_badGap
87           (spec := spec)
88           (model := model)
89           (commit := commit)
90           (r := r)
91           (badGapBound := badGapBound)
92           (hbadGap := hbadGap)
93           (omega := omega)
94     have hreal :
95         (((pseudoRegret model
96             (ETC.actionWithCommit spec (commit omega))
97             (spec.explorationPulls * K + r) : Rat) : Real)) <=
98           (((baseRat + suffixRat omega : Rat) : Real)) := by
99       exact_mod_cast hrat
100     have hmain :
101         ENNReal.ofReal
102           (((pseudoRegret model
103               (ETC.actionWithCommit spec (commit omega))
104               (spec.explorationPulls * K + r) : Rat) : Real)) <=
105         ENNReal.ofReal (((baseRat + suffixRat omega : Rat) : Real)) :=
106       ENNReal.ofReal_le_ofReal hreal
107     have hadd :
108         ENNReal.ofReal (((baseRat + suffixRat omega : Rat) : Real)) <=
109         base +
110           ENNReal.ofReal
111             (((suffixRat omega : Rat) : Real)) := by
112       simpa [base, Rat.cast_add] using
113         (ENNReal.ofReal_add_le
114           (p := (((baseRat : Rat) : Real)))
115           (q := (((suffixRat omega : Rat) : Real))))
116     have hsuffix :
117         ENNReal.ofReal
118           (((suffixRat omega : Rat) : Real)) <=
119         wrongSet.indicator (fun _ => suffix) omega := by
120       by_cases hcommit : commit omega = model.bestArm
121       · simp [wrongSet, suffix, suffixRat, hcommit]
122       · simp [wrongSet, suffix, suffixRat, hcommit]
123     exact le_trans hmain (le_trans hadd (add_le_add (le_refl base) hsuffix))
124   calc
125     MeasureTheory.lintegral mu
126       (fun omega : Omega =>
127         ENNReal.ofReal
128           (((pseudoRegret model
129               (ETC.actionWithCommit spec (commit omega))
130               (spec.explorationPulls * K + r) : Rat) : Real)))
131         <=
132       MeasureTheory.lintegral mu
133         (fun omega : Omega => base + wrongSet.indicator (fun _ => suffix) omega) := by
134           exact MeasureTheory.lintegral_mono hpoint
135     _ =
136       MeasureTheory.lintegral mu (fun _omega : Omega => base) +
137       MeasureTheory.lintegral mu
138         (fun omega : Omega => wrongSet.indicator (fun _ => suffix) omega) := by
139           rw [MeasureTheory.lintegral_add_left measurable_const]
140     _ = base * mu Set.univ + suffix * mu wrongSet := by
141           rw [MeasureTheory.lintegral_const,
142             MeasureTheory.lintegral_indicator_const
143               (by simpa [wrongSet] using hmeas_wrong)]
144     _ = base + suffix * mu wrongSet := by
145           simp [base, suffix, MeasureTheory.IsProbabilityMeasure.measure_univ]
146     _ <= base + suffix * pWrong := by
147           exact
148             add_le_add
149               (le_refl base)
150               (mul_le_mul_right
151                 (by simpa [wrongSet] using hprob_wrong)
152                 suffix)

154 /--
155 Finite-arm `actionWithCommit` pseudo-regret has an integrable Real cast when
156 the selected commit arm is measurable and the ambient measure is finite.
157 
158 The proof uses only the finite range of `commit : Omega -> Fin K`: the
159 integrand is a measurable finite-valued function, hence bounded by the finite
160 sum of the absolute values of its arm-indexed constants.
161 -/
162 theorem integrable_real_pseudoRegret_actionWithCommit_choice_of_measurable_commit
163     {Omega : Type u} {K : Nat}
164     [MeasurableSpace Omega]
165     (mu : Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
166     (spec : ETC.Spec K) (model : FiniteBanditModel K)
167     (commit : Omega -> Fin K) (r : Nat)
168     (hmeas_commit : Measurable commit) :
169     Integrable
170       (fun omega : Omega =>
171         (((pseudoRegret model (ETC.actionWithCommit spec (commit omega))
172           (spec.explorationPulls * K + r) : Rat) : Real))) mu := by
173   let c : Fin K -> Real :=
174     fun a : Fin K =>
175       (((pseudoRegret model (ETC.actionWithCommit spec a)
176         (spec.explorationPulls * K + r) : Rat) : Real))
177   have hc_meas : Measurable c := by
178     exact measurable_of_countable c
179   have hf_meas : Measurable (fun omega : Omega => c (commit omega)) :=
180     hc_meas.comp hmeas_commit
181   have hbound :
182       exists C : Real, forall omega : Omega, norm (c (commit omega)) <= C := by
183     let C : Real :=
184       ((Finset.univ : Finset (Fin K)).sum (fun a : Fin K => norm (c a)))
185     refine Exists.intro C ?_
186     intro omega
187     exact
188       Finset.single_le_sum
189         (fun a _ha => norm_nonneg (c a))
190         (Finset.mem_univ (commit omega))
191   cases hbound with
192   | intro C hC =>
193       change Integrable (fun omega : Omega => c (commit omega)) mu
194       refine MeasureTheory.Integrable.of_bound
195         hf_meas.aestronglyMeasurable C ?_
196       exact Filter.Eventually.of_forall hC

198 /--
199 Bochner/Real expected-regret assembly with a separate probability charge for
200 each possible commit arm.
201 
202 The suffix term is decomposed into the finite family of measurable events
203 `{omega | commit omega = a}`.  Unlike the coarser wrong-commit wrapper below,
204 this theorem preserves every arm gap and therefore exposes the per-arm RHS
205 needed by the LML ETC route.  It does not supply the armwise probability
206 bounds themselves.
207 -/
208 theorem integral_real_pseudoRegret_actionWithCommit_choice_le_exploration_add_suffix_sum_gap_mul_commit_prob
209     {Omega : Type u} {K : Nat}
210     [MeasurableSpace Omega]
211     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
212     (spec : ETC.Spec K) (model : FiniteBanditModel K)
213     (commit : Omega -> Fin K) (r : Nat)
214     (hmeas_commit : Measurable commit) :
215     MeasureTheory.integral mu
216       (fun omega : Omega =>
217         (((pseudoRegret model
218             (ETC.actionWithCommit spec (commit omega))
219             (spec.explorationPulls * K + r) : Rat) : Real))) <=
220     (((((Finset.univ : Finset (Fin K)).sum
221       (fun a : Fin K => model.gap a)) *
222       (((spec.explorationPulls : Nat) : Rat)) : Rat) : Real)) +
223     (Finset.univ : Finset (Fin K)).sum
224       (fun a : Fin K =>
225         ((((((r : Nat) : Rat) * model.gap a : Rat) : Real))) *
226           mu.real {omega : Omega | commit omega = a}) := by
227   let regret : Omega -> Real :=
228     fun omega : Omega =>
229       (((pseudoRegret model
230           (ETC.actionWithCommit spec (commit omega))
231           (spec.explorationPulls * K + r) : Rat) : Real))
232   let baseRat : Rat :=
233     ((Finset.univ : Finset (Fin K)).sum
234       (fun a : Fin K => model.gap a)) *
235       (((spec.explorationPulls : Nat) : Rat))
236   let baseReal : Real := ((baseRat : Rat) : Real)
237   let suffixValue : Fin K -> Real :=
238     fun a : Fin K =>
239       (((((r : Nat) : Rat) * model.gap a : Rat) : Real))
240   let commitSet : Fin K -> Set Omega :=
241     fun a : Fin K => {omega : Omega | commit omega = a}
242   let suffixTerm : Fin K -> Omega -> Real :=
243     fun a : Fin K => (commitSet a).indicator (fun _ => suffixValue a)
244   let suffix : Omega -> Real :=
245     fun omega : Omega =>
246       (Finset.univ : Finset (Fin K)).sum
247         (fun a : Fin K => suffixTerm a omega)
248   let bound : Omega -> Real :=
249     fun omega : Omega => baseReal + suffix omega
250   have hcommitSet : forall a : Fin K, MeasurableSet (commitSet a) := by
251     intro a
252     change MeasurableSet (commit ⁻¹' {a})
253     exact hmeas_commit (measurableSet_singleton a)
254   have hterm : forall a : Fin K, Integrable (suffixTerm a) mu := by
255     intro a
256     exact (integrable_const (suffixValue a)).indicator (hcommitSet a)
257   have hsuffix_integrable : Integrable suffix mu := by
258     exact
259       BanditRLProof.IntegrabilitySums.integrable_univ_sum
260         (mu := mu) (f := suffixTerm) hterm
261   have hbound_integrable : Integrable bound mu := by
262     exact (integrable_const baseReal).add hsuffix_integrable
263   have hregret_integrable : Integrable regret mu := by
264     simpa [regret] using
265       (ETC.integrable_real_pseudoRegret_actionWithCommit_choice_of_measurable_commit
266         (mu := mu) (spec := spec) (model := model)
267         (commit := commit) (r := r) hmeas_commit)
268   have hsuffix_apply : forall omega : Omega,
269       suffix omega = suffixValue (commit omega) := by
270     intro omega
271     rw [show suffix omega =
272         (Finset.univ : Finset (Fin K)).sum
273           (fun a : Fin K => suffixTerm a omega) by rfl]
274     rw [Finset.sum_eq_single (commit omega)]
275     all_goals simp [suffixTerm, commitSet]
276     aesop
277   have hpoint : forall omega : Omega, regret omega <= bound omega := by
278     intro omega
279     have hrat :=
280       ETC.pseudoRegret_actionWithCommit_explorationPulls_mul_K_add_le_sum_gap_mul_explorationPulls_add_suffix_gap
281         (spec := spec) (model := model) (commitArm := commit omega) (r := r)
282     have hreal :
283         regret omega <= baseReal + suffixValue (commit omega) := by
284       unfold regret baseReal baseRat suffixValue
285       exact_mod_cast hrat
286     simpa [bound, hsuffix_apply omega] using hreal
287   have hintegral_suffix :
288       MeasureTheory.integral mu suffix =
289         (Finset.univ : Finset (Fin K)).sum
290           (fun a : Fin K =>
291             suffixValue a * mu.real (commitSet a)) := by
292     rw [show suffix = fun omega : Omega =>
293         (Finset.univ : Finset (Fin K)).sum
294           (fun a : Fin K => suffixTerm a omega) by rfl]
295     rw [BanditRLProof.ExpectationBochnerSums.integral_univ_sum
296       (mu := mu) (f := suffixTerm) hterm]
297     apply Finset.sum_congr rfl
298     intro a _ha
299     rw [show suffixTerm a =
300         (commitSet a).indicator (fun _ : Omega => suffixValue a) by rfl]
301     rw [MeasureTheory.integral_indicator (hcommitSet a)]
302     rw [MeasureTheory.setIntegral_const]
303     simp [Measure.real, mul_comm]
304   calc
305     MeasureTheory.integral mu regret <=
306         MeasureTheory.integral mu bound := by
307       exact MeasureTheory.integral_mono
308         hregret_integrable hbound_integrable hpoint
309     _ = baseReal + MeasureTheory.integral mu suffix := by
310       rw [show bound = fun omega : Omega =>
311           (fun _omega : Omega => baseReal) omega + suffix omega by rfl]
312       rw [MeasureTheory.integral_add
313         (integrable_const baseReal) hsuffix_integrable]
314       rw [MeasureTheory.integral_const]
315       simp [MeasureTheory.probReal_univ]
316     _ = baseReal +
317         (Finset.univ : Finset (Fin K)).sum
318           (fun a : Fin K => suffixValue a * mu.real (commitSet a)) := by
319       rw [hintegral_suffix]
320     _ =
321         (((((Finset.univ : Finset (Fin K)).sum
322           (fun a : Fin K => model.gap a)) *
323           (((spec.explorationPulls : Nat) : Rat)) : Rat) : Real)) +
324         (Finset.univ : Finset (Fin K)).sum
325           (fun a : Fin K =>
326             ((((((r : Nat) : Rat) * model.gap a : Rat) : Real))) *
327               mu.real {omega : Omega | commit omega = a}) := by

330 /--
331 Bochner/Real expected-regret assembly for an `Omega`-indexed ETC commit
332 selector.
333 
334 This is the Real-valued analogue of
335 `lintegral_ofReal_pseudoRegret_actionWithCommit_choice_le_exploration_add_suffix_badGap_prob`.
336 It still consumes an abstract wrong-commit probability bound, but the
337 conclusion is an ordinary Bochner integral of the Real-cast pseudo-regret.
338 -/
339 theorem integral_real_pseudoRegret_actionWithCommit_choice_le_exploration_add_suffix_badGap_prob
340     {Omega : Type u} {K : Nat}
341     [MeasurableSpace Omega]
342     (mu : Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
343     (spec : ETC.Spec K) (model : FiniteBanditModel K)
344     (commit : Omega -> Fin K) (r : Nat)
345     (badGapBound : Rat) (pWrong : Real)
346     (hbadGap :
347       forall a : Fin K, (a = model.bestArm -> False) ->
348         model.gap a <= badGapBound)
349     (hbadGap_nonneg : (0 : Rat) <= badGapBound)
350     (hmeas_wrong :
351       MeasurableSet {omega : Omega | commit omega = model.bestArm -> False})
352     (hprob_wrong :
353       mu.real {omega : Omega | commit omega = model.bestArm -> False} <=
354         pWrong)
355     (hinteg : Integrable
356       (fun omega : Omega =>
357         (((pseudoRegret model (ETC.actionWithCommit spec (commit omega))
358           (spec.explorationPulls * K + r) : Rat) : Real))) mu) :
359     MeasureTheory.integral mu
360       (fun omega : Omega =>
361         (((pseudoRegret model
362             (ETC.actionWithCommit spec (commit omega))
363             (spec.explorationPulls * K + r) : Rat) : Real))) <=
364     (((((Finset.univ : Finset (Fin K)).sum
365       (fun a : Fin K => model.gap a)) *
366       (((spec.explorationPulls : Nat) : Rat)) : Rat) : Real)) +
367     ((((((r : Nat) : Rat) * badGapBound : Rat) : Real)) * pWrong) := by
368   let wrongSet : Set Omega :=
369     {omega : Omega | commit omega = model.bestArm -> False}
370   let baseRat : Rat :=
371     ((Finset.univ : Finset (Fin K)).sum
372       (fun a : Fin K => model.gap a)) *
373       (((spec.explorationPulls : Nat) : Rat))
374   let suffixRat : Omega -> Rat :=
375     fun omega : Omega =>
376       (((r : Nat) : Rat) *
377         (if commit omega = model.bestArm then (0 : Rat) else badGapBound))
378   let baseReal : Real := (((baseRat : Rat) : Real))
379   let suffixReal : Real :=
380     (((((r : Nat) : Rat) * badGapBound : Rat) : Real))
381   let bound : Omega -> Real :=
382     fun omega : Omega =>
383       baseReal + wrongSet.indicator (fun _ => suffixReal) omega
384   have hsuffix_nonneg : 0 <= suffixReal := by
385     have hr_nonneg : (0 : Rat) <= (((r : Nat) : Rat)) := by
386       exact_mod_cast Nat.zero_le r
387     have hsuffix_rat_nonneg :
388         (0 : Rat) <= (((r : Nat) : Rat) * badGapBound) :=
389       mul_nonneg hr_nonneg hbadGap_nonneg
390     have hsuffix_real_nonneg :
391         (0 : Real) <=
392           ((((r : Nat) : Rat) * badGapBound : Rat) : Real) := by
393       exact_mod_cast hsuffix_rat_nonneg
394     simpa [suffixReal] using hsuffix_real_nonneg
395   have hpoint :
396       (fun omega : Omega =>
397         (((pseudoRegret model
398             (ETC.actionWithCommit spec (commit omega))
399             (spec.explorationPulls * K + r) : Rat) : Real))) <= bound := by
400     intro omega
401     have hrat :
402         pseudoRegret model (ETC.actionWithCommit spec (commit omega))
403             (spec.explorationPulls * K + r) <=
404           baseRat + suffixRat omega := by
405       simpa [baseRat, suffixRat] using
406         ETC.pseudoRegret_actionWithCommit_choice_le_sum_gap_mul_explorationPulls_add_suffix_badGap
407           (spec := spec)
408           (model := model)
409           (commit := commit)
410           (r := r)
411           (badGapBound := badGapBound)
412           (hbadGap := hbadGap)
413           (omega := omega)
414     have hreal :
415         (((pseudoRegret model
416             (ETC.actionWithCommit spec (commit omega))
417             (spec.explorationPulls * K + r) : Rat) : Real)) <=
418           (((baseRat + suffixRat omega : Rat) : Real)) := by
419       exact_mod_cast hrat
420     have hrhs :
421         (((baseRat + suffixRat omega : Rat) : Real)) = bound omega := by
422       by_cases hcommit : commit omega = model.bestArm
423       · simp [bound, wrongSet, baseReal, suffixReal, suffixRat, hcommit,
424           Rat.cast]
425       · simp [bound, wrongSet, baseReal, suffixReal, suffixRat, hcommit,
426           Rat.cast]
427     simpa [hrhs]
428       using hreal
429   have hbound_integrable : Integrable bound mu := by
430     have hconst : Integrable (fun _omega : Omega => baseReal) mu :=
431       integrable_const baseReal
432     have hsuffix :
433         Integrable (wrongSet.indicator (fun _ : Omega => suffixReal)) mu :=
434       (integrable_const suffixReal).indicator
435         (by simpa [wrongSet] using hmeas_wrong)
436     simpa [bound] using hconst.add hsuffix
437   calc
438     MeasureTheory.integral mu
439       (fun omega : Omega =>
440         (((pseudoRegret model
441             (ETC.actionWithCommit spec (commit omega))
442             (spec.explorationPulls * K + r) : Rat) : Real)))
443         <= MeasureTheory.integral mu bound := by
444           exact MeasureTheory.integral_mono hinteg hbound_integrable hpoint
445     _ =
446         baseReal + suffixReal * mu.real wrongSet := by
447           have hconst : Integrable (fun _omega : Omega => baseReal) mu :=
448             integrable_const baseReal
449           have hsuffix :
450               Integrable
451                 (wrongSet.indicator (fun _ : Omega => suffixReal)) mu :=
452             (integrable_const suffixReal).indicator
453               (by simpa [wrongSet] using hmeas_wrong)
454           rw [show bound =
455               (fun omega : Omega =>
456                 (fun _omega : Omega => baseReal) omega +
457                   wrongSet.indicator (fun _ : Omega => suffixReal) omega) by
458                 funext omega
459                 rfl]
460           rw [MeasureTheory.integral_add hconst hsuffix]
461           rw [MeasureTheory.integral_const]
462           rw [MeasureTheory.integral_indicator
463             (by simpa [wrongSet] using hmeas_wrong)]
464           rw [MeasureTheory.setIntegral_const]
465           simp [MeasureTheory.probReal_univ, mul_comm]
466     _ <= baseReal + suffixReal * pWrong := by
467           exact
468             add_le_add
469               (le_refl baseReal)
470               (mul_le_mul_of_nonneg_left
471                 (by simpa [wrongSet] using hprob_wrong)
472                 hsuffix_nonneg)
473     _ =
474         (((((Finset.univ : Finset (Fin K)).sum
475           (fun a : Fin K => model.gap a)) *
476           (((spec.explorationPulls : Nat) : Rat)) : Rat) : Real)) +
477         ((((((r : Nat) : Rat) * badGapBound : Rat) : Real)) *
478           pWrong) := by
479           simp [baseReal, suffixReal, baseRat]
