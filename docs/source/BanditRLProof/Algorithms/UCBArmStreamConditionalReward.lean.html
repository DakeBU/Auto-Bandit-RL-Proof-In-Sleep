5 /-!
6 # Conditional laws for stationary UCB arm-stream coordinates
7 
8 This module isolates the product-measure part of the selected-reward route.
9 Every `(pull index, arm)` coordinate is independent of the function containing
10 all other coordinates, so its conditional distribution given that complement
11 is the prescribed stationary arm law.
12 -/

20 /-- A flattened pull-index/arm coordinate of a latent arm-reward stream. -/
21 def armStreamCoordinate {K : Nat} (index : Nat × Fin K) :
22     ArmRewardStream K -> Real :=
23   fun stream => stream index.1 index.2

25 /-- The latent arm stream with one specified coordinate omitted. -/
26 def armStreamWithoutCoordinate {K : Nat} (target : Nat × Fin K) :
27     ArmRewardStream K -> ({index : Nat × Fin K // index ≠ target} -> Real) :=
28   fun stream index => armStreamCoordinate index.1 stream

30 /-- Reconstruct a latent arm stream after supplying one omitted coordinate. -/
31 def armStreamInsertCoordinate {K : Nat} (target : Nat × Fin K) (value : Real) :
32     ({index : Nat × Fin K // index ≠ target} -> Real) -> ArmRewardStream K :=
33   fun rest index arm =>
34     if hindex : (index, arm) = target then value
35     else rest ⟨(index, arm), hindex⟩

37 /-- The history/action condition used by the successor reward kernel. -/
38 noncomputable def armStreamHistoryAction
39     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat) :
40     ArmRewardStream K ->
41       History.FinitePairHistory (Fin K) Real n × Fin K :=
42   fun stream =>
43     let history := armStreamHistory hK c stream n
44     (history, realHistoryNextArm hK c n history)

46 /-- The next unused pull-index/arm coordinate selected after history `n`. -/
47 noncomputable def armStreamNextCoordinate
48     {K : Nat} (hK : 0 < K) (c : Real) (n : Nat) :
49     ArmRewardStream K -> Nat × Fin K :=
50   fun stream =>
51     let condition := armStreamHistoryAction hK c n stream
52     (ETC.realHistoryPullCount n condition.1 condition.2, condition.2)

275 /--
276 The recursive history through `n` only reads coordinates whose pull index is
277 strictly below the corresponding arm count at time `n + 1`.
278 -/
279 theorem armStreamHistory_eq_of_eq_below_pullCount
280     {K : Nat} (hK : 0 < K) (c : Real)
281     (stream stream' : ArmRewardStream K) (n : Nat)
282     (hagrees : ∀ arm index,
283       index < pullCount (armStreamAction hK c stream) arm (n + 1) →
284         stream index arm = stream' index arm) :
285     armStreamHistory hK c stream n = armStreamHistory hK c stream' n := by
286   induction n generalizing stream stream' with
287   | zero =>
288       have hzero : stream 0 (initializationArm hK 0) =
289           stream' 0 (initializationArm hK 0) := by
290         apply hagrees
291         simp
292       simp only [armStreamHistory]
293       funext _i
294       exact Prod.ext rfl hzero
295   | succ n ih =>
296       have hprefix : ∀ arm index,
297           index < pullCount (armStreamAction hK c stream) arm (n + 1) →
298             stream index arm = stream' index arm := by
299         intro arm index hindex
300         apply hagrees arm index
301         exact lt_of_lt_of_le hindex <|
302           pullCount_mono (armStreamAction hK c stream) arm (Nat.le_succ _)
303       have hhistory := ih stream stream' hprefix
304       simp only [armStreamHistory]
305       rw [← hhistory]
306       apply congrArg (History.extendPairHistorySucc
307         (armStreamHistory hK c stream n))
308       apply Prod.ext
309       · rfl
310       · apply hagrees
311         rw [armStreamHistory_eq_finitePairHistoryOfTrace]
312         rw [ETC.realHistoryPullCount_finitePairHistoryOfTrace]
313         have hselected : armStreamAction hK c stream (n + 1) =
314             realHistoryNextArm hK c n
315               (History.finitePairHistoryOfTrace
316                 (armStreamAction hK c stream)
317                 (armStreamReward hK c stream) n) := by
318           rw [← armStreamHistory_eq_finitePairHistoryOfTrace]
319           exact armStreamAction_succ hK c stream n
320         rw [pullCount_succ_of_eq
321           (armStreamAction hK c stream) _ (n + 1) hselected]
322         exact Nat.lt_succ_self _
