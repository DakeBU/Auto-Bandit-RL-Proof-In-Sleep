3 /-!
4 # Two-arm stochastic-gradient-bandit one-step recurrences
5 
6 This module instantiates the generated conditional Equation (8) at the two
7 action-dependent coefficients used in the proof of Theorem 1 of
8 Baudry--Johnson--Vary--Pike-Burke--Rebeschini (NeurIPS 2025).  It proves both
9 one-step exponential recurrences over the actual generated history-step
10 kernel, under the source initialization `theta = 0`.
11 
12 Time fence: a value
13 `historyParameter (fun _ : Fin 2 => 0) eta n history` is the source parameter
14 `theta_{.,n+2}` after the inclusive history through Lean time `n`.  The
15 successor expressions below add the update generated by the next pair, and
16 therefore represent `theta_{.,n+3}`.  They do not consume that pair twice.
17 
18 Probability / Equation-(8) ledger:
19 
20 * the one-step random object is the `(selected action, reward)` pair drawn
21   from `Thompson.historyStepKernel` at the fixed generated history;
22 * no independence premise is used: the reward law is already conditioned on
23   both the history and selected action;
24 * `hmean` fixes the two conditional reward means, and `hgap` identifies their
25   difference with `Delta`;
26 * `hreward` gives the exact source support contract `|reward| <= 1` almost
27   everywhere on each action fiber;
28 * measurability and integrability of the exponential are discharged inside
29   `integral_historyStepKernel_exp_actionReward_le_sourceEqEight_of_mean`;
30 * `sourceC_mono` replaces the branch constants
31   `sourceC (eta * p)` and `sourceC (eta * (1-p))` by the common `sourceC eta`.
32 
33 The final declarations are conditional one-step recurrences.  They do not
34 yet provide the initial/base expectation, tower-property iteration, failure
35 mass summation, or the finite-horizon regret theorem.
36 -/

46 /-- The action-dependent coefficient in the forward exponential potential.
47 For source arm `1` (Lean arm `0`) it is `2 eta (1-p)`; for source arm `2`
48 (Lean arm `1`) it is `-2 eta p`. -/
49 def twoArmForwardQ (eta : Real) (prob : Fin 2 -> Real)
50     (selected : Fin 2) : Real :=
51   if selected = 0 then 2 * eta * (1 - prob 0) else -(2 * eta * prob 0)
52 
53 /-- The coefficient for the inverse-odds exponential potential. -/
54 def twoArmInverseQ (eta : Real) (prob : Fin 2 -> Real)
55     (selected : Fin 2) : Real :=
56   -twoArmForwardQ eta prob selected
57 
58 /-- `q_+ reward` is exactly twice the learning-rate-scaled best-coordinate
59 source update. -/
60 theorem twoArmForwardQ_mul_reward_eq_sourceIncrement
61     (eta reward : Real) (prob : Fin 2 -> Real) (selected : Fin 2) :
62     twoArmForwardQ eta prob selected * reward =
63       2 * eta * sourceIncrement prob reward selected 0 := by
64   fin_cases selected <;> simp [twoArmForwardQ, sourceIncrement] <;> ring
65 
66 /-- `q_- reward` is the negative of twice the learning-rate-scaled
67 best-coordinate source update. -/
68 theorem twoArmInverseQ_mul_reward_eq_sourceIncrement
69     (eta reward : Real) (prob : Fin 2 -> Real) (selected : Fin 2) :
70     twoArmInverseQ eta prob selected * reward =
71       -2 * eta * sourceIncrement prob reward selected 0 := by
72   rw [twoArmInverseQ, neg_mul,
73     twoArmForwardQ_mul_reward_eq_sourceIncrement]
74   ring

76 /-- The two-branch Equation-(8) remainder for the forward potential, after
77 replacing both branch constants by the common source constant `C_eta`. -/
78 theorem twoArmForwardEqEightRemainder_le
79     (eta p meanZero meanOne Delta : Real)
80     (heta : 0 <= eta) (hp_nonneg : 0 <= p) (hp_le_one : p <= 1)
81     (hgap : meanZero - meanOne = Delta) :
82     p *
83           ((2 * eta * (1 - p)) * meanZero +
84             (2 * eta * (1 - p)) ^ 2 / 2 *
85               sourceC (|2 * eta * (1 - p)| / 2)) +
86         (1 - p) *
87           ((-(2 * eta * p)) * meanOne +
88             (-(2 * eta * p)) ^ 2 / 2 *
89               sourceC (|-(2 * eta * p)| / 2)) <=
90       2 * p * (1 - p) * (eta * Delta + eta ^ 2 * sourceC eta) := by
91   have hfail_nonneg : 0 <= 1 - p := sub_nonneg.mpr hp_le_one
92   have hfail_le_one : 1 - p <= 1 := by linarith
93   have hqZero_nonneg : 0 <= 2 * eta * (1 - p) := by positivity
94   have hqOneAbs_nonneg : 0 <= 2 * eta * p := by positivity
95   have habsZero : |2 * eta * (1 - p)| / 2 = eta * (1 - p) := by
96     rw [abs_of_nonneg hqZero_nonneg]
97     ring
98   have habsOne : |-(2 * eta * p)| / 2 = eta * p := by
99     rw [abs_neg, abs_of_nonneg hqOneAbs_nonneg]
100     ring
101   have hCZero : sourceC (eta * (1 - p)) <= sourceC eta :=
102     sourceC_mono (mul_nonneg heta hfail_nonneg)
103       (mul_le_of_le_one_right heta hfail_le_one)
104   have hCOne : sourceC (eta * p) <= sourceC eta :=
105     sourceC_mono (mul_nonneg heta hp_nonneg)
106       (mul_le_of_le_one_right heta hp_le_one)
107   have htailZero :
108       (2 * eta * (1 - p)) ^ 2 / 2 *
109           sourceC (|2 * eta * (1 - p)| / 2) <=
110         (2 * eta * (1 - p)) ^ 2 / 2 * sourceC eta := by
111     rw [habsZero]
112     exact mul_le_mul_of_nonneg_left hCZero (by positivity)
113   have htailOne :
114       (-(2 * eta * p)) ^ 2 / 2 *
115           sourceC (|-(2 * eta * p)| / 2) <=
116         (-(2 * eta * p)) ^ 2 / 2 * sourceC eta := by
117     rw [habsOne]
118     exact mul_le_mul_of_nonneg_left hCOne (by positivity)
119   have hbranchZero :
120       p *
121           ((2 * eta * (1 - p)) * meanZero +
122             (2 * eta * (1 - p)) ^ 2 / 2 *
123               sourceC (|2 * eta * (1 - p)| / 2)) <=
124         p *
125           ((2 * eta * (1 - p)) * meanZero +
126             (2 * eta * (1 - p)) ^ 2 / 2 * sourceC eta) :=
127     mul_le_mul_of_nonneg_left
128       (add_le_add_right htailZero _) hp_nonneg
129   have hbranchOne :
130       (1 - p) *
131           ((-(2 * eta * p)) * meanOne +
132             (-(2 * eta * p)) ^ 2 / 2 *
133               sourceC (|-(2 * eta * p)| / 2)) <=
134         (1 - p) *
135           ((-(2 * eta * p)) * meanOne +
136             (-(2 * eta * p)) ^ 2 / 2 * sourceC eta) :=
137     mul_le_mul_of_nonneg_left
138       (add_le_add_right htailOne _) hfail_nonneg
139   calc
140     p *
141           ((2 * eta * (1 - p)) * meanZero +
142             (2 * eta * (1 - p)) ^ 2 / 2 *
143               sourceC (|2 * eta * (1 - p)| / 2)) +
144         (1 - p) *
145           ((-(2 * eta * p)) * meanOne +
146             (-(2 * eta * p)) ^ 2 / 2 *
147               sourceC (|-(2 * eta * p)| / 2)) <=
148       p *
149           ((2 * eta * (1 - p)) * meanZero +
150             (2 * eta * (1 - p)) ^ 2 / 2 * sourceC eta) +
151         (1 - p) *
152           ((-(2 * eta * p)) * meanOne +
153             (-(2 * eta * p)) ^ 2 / 2 * sourceC eta) :=
154       add_le_add hbranchZero hbranchOne
155     _ = 2 * p * (1 - p) *
156         (eta * Delta + eta ^ 2 * sourceC eta) := by
157       have hmeanZero : meanZero = Delta + meanOne := by linarith
158       rw [hmeanZero]
159       ring

161 /-- The analogous Equation-(8) remainder for the inverse potential. -/
162 theorem twoArmInverseEqEightRemainder_le
163     (eta p meanZero meanOne Delta : Real)
164     (heta : 0 <= eta) (hp_nonneg : 0 <= p) (hp_le_one : p <= 1)
165     (hgap : meanZero - meanOne = Delta) :
166     p *
167           ((-(2 * eta * (1 - p))) * meanZero +
168             (-(2 * eta * (1 - p))) ^ 2 / 2 *
169               sourceC (|-(2 * eta * (1 - p))| / 2)) +
170         (1 - p) *
171           ((2 * eta * p) * meanOne +
172             (2 * eta * p) ^ 2 / 2 *
173               sourceC (|2 * eta * p| / 2)) <=
174       -2 * eta * p * (1 - p) * (Delta - eta * sourceC eta) := by
175   have hfail_nonneg : 0 <= 1 - p := sub_nonneg.mpr hp_le_one
176   have hfail_le_one : 1 - p <= 1 := by linarith
177   have hqZeroAbs_nonneg : 0 <= 2 * eta * (1 - p) := by positivity
178   have hqOne_nonneg : 0 <= 2 * eta * p := by positivity
179   have habsZero : |-(2 * eta * (1 - p))| / 2 = eta * (1 - p) := by
180     rw [abs_neg, abs_of_nonneg hqZeroAbs_nonneg]
181     ring
182   have habsOne : |2 * eta * p| / 2 = eta * p := by
183     rw [abs_of_nonneg hqOne_nonneg]
184     ring
185   have hCZero : sourceC (eta * (1 - p)) <= sourceC eta :=
186     sourceC_mono (mul_nonneg heta hfail_nonneg)
187       (mul_le_of_le_one_right heta hfail_le_one)
188   have hCOne : sourceC (eta * p) <= sourceC eta :=
189     sourceC_mono (mul_nonneg heta hp_nonneg)
190       (mul_le_of_le_one_right heta hp_le_one)
191   have htailZero :
192       (-(2 * eta * (1 - p))) ^ 2 / 2 *
193           sourceC (|-(2 * eta * (1 - p))| / 2) <=
194         (-(2 * eta * (1 - p))) ^ 2 / 2 * sourceC eta := by
195     rw [habsZero]
196     exact mul_le_mul_of_nonneg_left hCZero (by positivity)
197   have htailOne :
198       (2 * eta * p) ^ 2 / 2 * sourceC (|2 * eta * p| / 2) <=
199         (2 * eta * p) ^ 2 / 2 * sourceC eta := by
200     rw [habsOne]
201     exact mul_le_mul_of_nonneg_left hCOne (by positivity)
202   have hbranchZero :
203       p *
204           ((-(2 * eta * (1 - p))) * meanZero +
205             (-(2 * eta * (1 - p))) ^ 2 / 2 *
206               sourceC (|-(2 * eta * (1 - p))| / 2)) <=
207         p *
208           ((-(2 * eta * (1 - p))) * meanZero +
209             (-(2 * eta * (1 - p))) ^ 2 / 2 * sourceC eta) :=
210     mul_le_mul_of_nonneg_left
211       (add_le_add_right htailZero _) hp_nonneg
212   have hbranchOne :
213       (1 - p) *
214           ((2 * eta * p) * meanOne +
215             (2 * eta * p) ^ 2 / 2 * sourceC (|2 * eta * p| / 2)) <=
216         (1 - p) *
217           ((2 * eta * p) * meanOne +
218             (2 * eta * p) ^ 2 / 2 * sourceC eta) :=
219     mul_le_mul_of_nonneg_left
220       (add_le_add_right htailOne _) hfail_nonneg
221   calc
222     p *
223           ((-(2 * eta * (1 - p))) * meanZero +
224             (-(2 * eta * (1 - p))) ^ 2 / 2 *
225               sourceC (|-(2 * eta * (1 - p))| / 2)) +
226         (1 - p) *
227           ((2 * eta * p) * meanOne +
228             (2 * eta * p) ^ 2 / 2 * sourceC (|2 * eta * p| / 2)) <=
229       p *
230           ((-(2 * eta * (1 - p))) * meanZero +
231             (-(2 * eta * (1 - p))) ^ 2 / 2 * sourceC eta) +
232         (1 - p) *
233           ((2 * eta * p) * meanOne +
234             (2 * eta * p) ^ 2 / 2 * sourceC eta) :=
235       add_le_add hbranchZero hbranchOne
236     _ = -2 * eta * p * (1 - p) *
237         (Delta - eta * sourceC eta) := by
238       have hmeanZero : meanZero = Delta + meanOne := by linarith
239       rw [hmeanZero]
240       ring

242 /-- Multiplicative forward one-step recurrence at a fixed generated history.
243 The current parameter is source `theta_{.,n+2}` and the integrand is the
244 source successor `theta_{1,n+3}`. -/
245 theorem integral_twoArmHistoryStepKernel_exp_forwardSuccessor_le
246     (eta Delta : Real) (heta : 0 <= eta)
247     (environment : Thompson.HistoryEnvironment (Fin 2) Real)
248     (n : Nat) (history : History.FinitePairHistory (Fin 2) Real n)
249     (mean : Fin 2 -> Real)
250     (hreward : forall selected,
251       ∀ᵐ reward ∂environment.feedback n (history, selected),
252         |reward| <= 1)
253     (hmean : forall selected,
254       integral (environment.feedback n (history, selected)) id =
255         mean selected)
256     (hgap : mean 0 - mean 1 = Delta) :
257     integral
258         (Thompson.historyStepKernel
259           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment n history)
260         (fun pair : Fin 2 × Real =>
261           Real.exp
262             (2 *
263               (historyParameter (fun _ : Fin 2 => 0) eta n history 0 +
264                 eta * sourceIncrement
265                   (softmaxProbability
266                     (historyParameter (fun _ : Fin 2 => 0) eta n history))
267                   pair.2 pair.1 0))) <=
268       Real.exp
269           (2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) *
270         (1 + 2 *
271           softmaxProbability
272               (historyParameter (fun _ : Fin 2 => 0) eta n history) 0 *
273           (1 - softmaxProbability
274               (historyParameter (fun _ : Fin 2 => 0) eta n history) 0) *
275           (eta * Delta + eta ^ 2 * sourceC eta)) := by
276   let theta := historyParameter (fun _ : Fin 2 => 0) eta n history
277   let prob := softmaxProbability theta
278   let q := twoArmForwardQ eta prob
279   have hp_nonneg : 0 <= prob 0 := softmaxProbability_nonneg theta 0
280   have hp_le_one : prob 0 <= 1 := softmaxProbability_le_one theta 0
281   have hEqEight :=
282     integral_historyStepKernel_exp_actionReward_le_sourceEqEight_of_mean
283       (fun _ : Fin 2 => 0) eta environment n history q mean hreward hmean
284   have hrem :
285       (∑ selected, prob selected *
286         (q selected * mean selected +
287           q selected ^ 2 / 2 * sourceC (|q selected| / 2))) <=
288         2 * prob 0 * (1 - prob 0) *
289           (eta * Delta + eta ^ 2 * sourceC eta) := by
290     have hprobOne : prob 1 = 1 - prob 0 := by
291       simpa [prob] using softmaxProbability_one_eq_one_sub_zero theta
292     simp only [Fin.sum_univ_two]
293     rw [hprobOne]
294     simpa [q, twoArmForwardQ] using
295       twoArmForwardEqEightRemainder_le eta (prob 0)
296         (mean 0) (mean 1) Delta heta hp_nonneg hp_le_one hgap
297   have hbase :
298       integral
299           (Thompson.historyStepKernel
300             (historyAlgorithm (fun _ : Fin 2 => 0) eta)
301               environment n history)
302           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
303         1 + 2 * prob 0 * (1 - prob 0) *
304           (eta * Delta + eta ^ 2 * sourceC eta) := by
305     have hEqEight' :
306         integral
307             (Thompson.historyStepKernel
308               (historyAlgorithm (fun _ : Fin 2 => 0) eta)
309                 environment n history)
310             (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
311           1 + ∑ selected, prob selected *
312             (q selected * mean selected +
313               q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := by
314       simpa [theta, prob] using hEqEight
315     exact hEqEight'.trans (add_le_add_right hrem 1)
316   have hexp :
317       (fun pair : Fin 2 × Real =>
318           Real.exp
319             (2 *
320               (theta 0 + eta * sourceIncrement prob pair.2 pair.1 0))) =
321         (fun pair : Fin 2 × Real =>
322           Real.exp (2 * theta 0) * Real.exp (q pair.1 * pair.2)) := by
323     funext pair
324     rw [← Real.exp_add]
325     apply congrArg Real.exp
326     rw [twoArmForwardQ_mul_reward_eq_sourceIncrement]
327     ring
328   change integral
329       (Thompson.historyStepKernel
330         (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment n history)
331       (fun pair : Fin 2 × Real =>
332         Real.exp
333           (2 * (theta 0 + eta * sourceIncrement prob pair.2 pair.1 0))) <= _
334   rw [hexp, integral_const_mul]
335   exact mul_le_mul_of_nonneg_left hbase (Real.exp_nonneg _)

337 /-- Forward recurrence in the additive source form
338 `E_t exp(2 theta_{1,t+1}) <= exp(2 theta_{1,t})
339   + 2 p_t^2 (eta Delta + eta^2 C_eta)`. -/
340 theorem integral_twoArmHistoryStepKernel_exp_forwardSuccessor_le_add_success_sq
341     (eta Delta : Real) (heta : 0 <= eta)
342     (environment : Thompson.HistoryEnvironment (Fin 2) Real)
343     (n : Nat) (history : History.FinitePairHistory (Fin 2) Real n)
344     (mean : Fin 2 -> Real)
345     (hreward : forall selected,
346       ∀ᵐ reward ∂environment.feedback n (history, selected),
347         |reward| <= 1)
348     (hmean : forall selected,
349       integral (environment.feedback n (history, selected)) id =
350         mean selected)
351     (hgap : mean 0 - mean 1 = Delta) :
352     integral
353         (Thompson.historyStepKernel
354           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment n history)
355         (fun pair : Fin 2 × Real =>
356           Real.exp
357             (2 *
358               (historyParameter (fun _ : Fin 2 => 0) eta n history 0 +
359                 eta * sourceIncrement
360                   (softmaxProbability
361                     (historyParameter (fun _ : Fin 2 => 0) eta n history))
362                   pair.2 pair.1 0))) <=
363       Real.exp
364           (2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) +
365         2 *
366           softmaxProbability
367               (historyParameter (fun _ : Fin 2 => 0) eta n history) 0 ^ 2 *
368           (eta * Delta + eta ^ 2 * sourceC eta) := by
369   let theta := historyParameter (fun _ : Fin 2 => 0) eta n history
370   let p := softmaxProbability theta 0
371   have hmul :=
372     integral_twoArmHistoryStepKernel_exp_forwardSuccessor_le

378     2 * p ^ 2 * (eta * Delta + eta ^ 2 * sourceC eta)
379   calc
380     _ <= Real.exp (2 * theta 0) *
381         (1 + 2 * p * (1 - p) *
382           (eta * Delta + eta ^ 2 * sourceC eta)) := by
383       simpa [theta, p] using hmul
384     _ = Real.exp (2 * theta 0) +
385         2 * p * (Real.exp (2 * theta 0) * (1 - p)) *
386           (eta * Delta + eta ^ 2 * sourceC eta) := by ring
387     _ = Real.exp (2 * theta 0) +
388         2 * p ^ 2 * (eta * Delta + eta ^ 2 * sourceC eta) := by
389       rw [hodds]

392 /-- Multiplicative inverse-potential one-step recurrence at the same time
393 fence. -/
394 theorem integral_twoArmHistoryStepKernel_exp_inverseSuccessor_le
395     (eta Delta : Real) (heta : 0 <= eta)
396     (environment : Thompson.HistoryEnvironment (Fin 2) Real)
397     (n : Nat) (history : History.FinitePairHistory (Fin 2) Real n)
398     (mean : Fin 2 -> Real)
399     (hreward : forall selected,
400       ∀ᵐ reward ∂environment.feedback n (history, selected),
401         |reward| <= 1)
402     (hmean : forall selected,
403       integral (environment.feedback n (history, selected)) id =
404         mean selected)
405     (hgap : mean 0 - mean 1 = Delta) :
406     integral
407         (Thompson.historyStepKernel
408           (historyAlgorithm (fun _ : Fin 2 => 0) eta) environment n history)
409         (fun pair : Fin 2 × Real =>
410           Real.exp
411             (-2 *
412               (historyParameter (fun _ : Fin 2 => 0) eta n history 0 +
413                 eta * sourceIncrement
414                   (softmaxProbability
415                     (historyParameter (fun _ : Fin 2 => 0) eta n history))
416                   pair.2 pair.1 0))) <=
417       Real.exp
418           (-2 * historyParameter (fun _ : Fin 2 => 0) eta n history 0) *
419         (1 - 2 * eta *
420           softmaxProbability
421               (historyParameter (fun _ : Fin 2 => 0) eta n history) 0 *
422           (1 - softmaxProbability
423               (historyParameter (fun _ : Fin 2 => 0) eta n history) 0) *
424           (Delta - eta * sourceC eta)) := by
425   let theta := historyParameter (fun _ : Fin 2 => 0) eta n history
426   let prob := softmaxProbability theta
427   let q := twoArmInverseQ eta prob
428   have hp_nonneg : 0 <= prob 0 := softmaxProbability_nonneg theta 0
429   have hp_le_one : prob 0 <= 1 := softmaxProbability_le_one theta 0
430   have hEqEight :=
431     integral_historyStepKernel_exp_actionReward_le_sourceEqEight_of_mean
432       (fun _ : Fin 2 => 0) eta environment n history q mean hreward hmean
433   have hrem :
434       (∑ selected, prob selected *
435         (q selected * mean selected +
436           q selected ^ 2 / 2 * sourceC (|q selected| / 2))) <=
437         -2 * eta * prob 0 * (1 - prob 0) *
438           (Delta - eta * sourceC eta) := by
439     have hprobOne : prob 1 = 1 - prob 0 := by
440       simpa [prob] using softmaxProbability_one_eq_one_sub_zero theta
441     simp only [Fin.sum_univ_two]
442     rw [hprobOne]
443     simpa [q, twoArmInverseQ, twoArmForwardQ] using
444       twoArmInverseEqEightRemainder_le eta (prob 0)
445         (mean 0) (mean 1) Delta heta hp_nonneg hp_le_one hgap
446   have hbase :
447       integral
448           (Thompson.historyStepKernel
449             (historyAlgorithm (fun _ : Fin 2 => 0) eta)
450               environment n history)
451           (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
452         1 - 2 * eta * prob 0 * (1 - prob 0) *
453           (Delta - eta * sourceC eta) := by
454     have hEqEight' :
455         integral
456             (Thompson.historyStepKernel
457               (historyAlgorithm (fun _ : Fin 2 => 0) eta)
458                 environment n history)
459             (fun pair : Fin 2 × Real => Real.exp (q pair.1 * pair.2)) <=
460           1 + ∑ selected, prob selected *
461             (q selected * mean selected +
462               q selected ^ 2 / 2 * sourceC (|q selected| / 2)) := by
463       simpa [theta, prob] using hEqEight
464     have := hEqEight'.trans (add_le_add_right hrem 1)
465     calc
466       _ <= 1 + -2 * eta * prob 0 * (1 - prob 0) *
467           (Delta - eta * sourceC eta) := this
468       _ = 1 - 2 * eta * prob 0 * (1 - prob 0) *
469           (Delta - eta * sourceC eta) := by ring
470   have hexp :
471       (fun pair : Fin 2 × Real =>
472           Real.exp
473             (-2 *
474               (theta 0 + eta * sourceIncrement prob pair.2 pair.1 0))) =
475         (fun pair : Fin 2 × Real =>
476           Real.exp (-2 * theta 0) * Real.exp (q pair.1 * pair.2)) := by
477     funext pair
478     rw [← Real.exp_add]
479     apply congrArg Real.exp
480     rw [twoArmInverseQ_mul_reward_eq_sourceIncrement]
481     ring
