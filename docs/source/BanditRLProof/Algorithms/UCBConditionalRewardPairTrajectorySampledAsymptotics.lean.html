6 # Asymptotic sampled-successor UCB pseudo-regret
7 
8 This module runs the canonical pair-trajectory sampled-successor UCB theorem
9 with confidence budget `1 / (T + 1)`, proves logarithmic expected
10 pseudo-regret for fixed model data, and derives vanishing expected average
11 pseudo-regret for the resulting horizon-indexed policy family.
12 -/

54 /-- Eventually, the scheduled finite-arm/time peeling budget is at most `4 log(T+1)`. -/
55 theorem selectedPolicySuccessorFiniteArmTimeLogBudget_asymptoticDelta_le
56     (K T : Nat) (hK : 0 < K) (hT : 0 < T) (hlarge : 2 * K <= T + 1) :
57     selectedPolicySuccessorFiniteArmTimeLogBudget K T T
58         (selectedPolicySuccessorAsymptoticDelta T) <=
59       4 * Real.log (((T + 1 : Nat) : Real)) := by
60   rw [selectedPolicySuccessorFiniteArmTimeLogBudget_asymptoticDelta_eq
61     K T hK hT]
62   let n : Real := (((T + 1 : Nat) : Real))
63   let arg : Real := 2 * (K : Real) * (T : Real) * (T : Real) * n
64   have hn_one : 1 <= n := by
65     dsimp [n]
66     exact_mod_cast Nat.succ_le_succ (Nat.zero_le T)
67   have hn_pos : 0 < n := lt_of_lt_of_le zero_lt_one hn_one
68   have htwoK : 2 * (K : Real) <= n := by
69     dsimp [n]
70     exact_mod_cast hlarge
71   have hTn : (T : Real) <= n := by
72     dsimp [n]
73     exact_mod_cast Nat.le_succ T
74   have harg_pos : 0 < arg := by
75     dsimp [arg]
76     positivity
77   have harg_le : arg <= n ^ 4 := by
78     calc
79       arg <= n * n * n * n := by
80         dsimp [arg]
81         gcongr
82       _ = n ^ 4 := by ring
83   have hlog_le : Real.log arg <= 4 * Real.log n := by
84     calc
85       Real.log arg <= Real.log (n ^ 4) :=
86         Real.log_le_log harg_pos harg_le
87       _ = 4 * Real.log n := by
88         rw [Real.log_pow]
89         norm_num
90   have hlog_nonneg : 0 <= 4 * Real.log n := by
91     exact mul_nonneg (by norm_num) (Real.log_nonneg hn_one)
92   simpa [arg, n] using max_le hlog_le hlog_nonneg

99 theorem selectedPolicySuccessorTextbookGapBudget_add_failure_asymptoticDelta_le
100     (K : Nat) (sigma2 : NNReal) (T : Nat) (gap : Real)
101     (hK : 0 < K) (hT : 0 < T) (hlarge : 2 * K <= T + 1)
102     (hgap : 0 < gap) :
103     selectedPolicySuccessorTextbookGapBudget K sigma2 T
104           (selectedPolicySuccessorAsymptoticDelta T) gap +
105         gap * ((T : Real) * selectedPolicySuccessorAsymptoticDelta T) <=
106       selectedPolicySuccessorAsymptoticGapCoefficient sigma2 gap *
107         (1 + Real.log (((T + 1 : Nat) : Real))) := by
108   let variance : Real := (((sigma2 : NNReal) : Real))
109   let budget := selectedPolicySuccessorFiniteArmTimeLogBudget K T T
110     (selectedPolicySuccessorAsymptoticDelta T)
111   let logN := Real.log (((T + 1 : Nat) : Real))
112   let A := 128 * variance / gap + 16
113   let C := A + 3 * gap
114   have hlogN : 0 <= logN := by
115     dsimp [logN]
116     exact Real.log_nonneg (by
117       exact_mod_cast Nat.succ_le_succ (Nat.zero_le T))
118   have hbudget : budget <= 4 * logN := by
119     simpa [budget, logN] using
120       selectedPolicySuccessorFiniteArmTimeLogBudget_asymptoticDelta_le
121         K T hK hT hlarge
122   have hvariance : 0 <= variance := by
123     dsimp [variance]
124     positivity
125   have hquad :
126       32 * variance * budget / gap <=
127         (128 * variance / gap) * logN := by
128     calc
129       32 * variance * budget / gap <=
130           32 * variance * (4 * logN) / gap := by
131         gcongr
132       _ = (128 * variance / gap) * logN := by ring
133   have hlinear : 4 * budget <= 16 * logN := by
134     linarith
135   have hfailure :
136       gap * ((T : Real) * selectedPolicySuccessorAsymptoticDelta T) <= gap :=
137     by
138       simpa using mul_le_mul_of_nonneg_left
139         (horizon_mul_selectedPolicySuccessorAsymptoticDelta_le_one T) hgap.le
140   have hA : 0 <= A := by
141     dsimp [A]
142     positivity
143   have hC : 0 <= C := by
144     dsimp [C]
145     positivity
146   have hmain :
147       selectedPolicySuccessorTextbookGapBudget K sigma2 T
148             (selectedPolicySuccessorAsymptoticDelta T) gap +
149           gap * ((T : Real) * selectedPolicySuccessorAsymptoticDelta T) <=
150         A * logN + 3 * gap := by
151     unfold selectedPolicySuccessorTextbookGapBudget
152     dsimp [budget, variance] at hquad hlinear
153     dsimp [A, logN]
154     linarith
155   have hAL : A * logN <= C * logN := by
156     apply mul_le_mul_of_nonneg_right _ hlogN
157     dsimp [C]
158     linarith
159   have hconst : 3 * gap <= C := by
160     dsimp [C]
161     linarith
162   calc
163     selectedPolicySuccessorTextbookGapBudget K sigma2 T
164           (selectedPolicySuccessorAsymptoticDelta T) gap +
165         gap * ((T : Real) * selectedPolicySuccessorAsymptoticDelta T) <=
166       A * logN + 3 * gap := hmain
167     _ <= C * logN + C := add_le_add hAL hconst
168     _ = selectedPolicySuccessorAsymptoticGapCoefficient sigma2 gap *
169         (1 + Real.log (((T + 1 : Nat) : Real))) := by
170       dsimp [C, A, variance, logN,
171         selectedPolicySuccessorAsymptoticGapCoefficient]
172       ring
