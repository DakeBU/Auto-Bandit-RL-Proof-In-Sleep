4 /-!
5 # ETC phase-switching traces
6 
7 This module introduces the first deterministic boundary for phase-switching
8 Explore-Then-Commit traces.  The commit arm is supplied explicitly; empirical
9 mean selection, probability, concentration, and regret facts live in later
10 leaves.
11 -/
12 
13 namespace BanditRLProof
14 namespace ETC
15 
16 /-- Explore by round-robin until the configured horizon, then play `commitArm`. -/
17 def actionWithCommit {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K) :
18     ActionTrace (Fin K) :=
19   fun t : Nat =>
20     if t < spec.explorationPulls * K then
21       ETC.exploreArm spec t
22     else
23       commitArm
24 
25 /--
26 During the configured exploration prefix, the phase-switching ETC trace agrees
27 with the pure round-robin exploration trace.
28 
29 This is the `ETC-ACTION-WITH-COMMIT-EXPLORE-PHASE` project-local trace-boundary
30 leaf.
31 -/
32 @[simp] theorem actionWithCommit_eq_exploreArm_of_lt
33     {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K) {t : Nat}
34     (h : t < spec.explorationPulls * K) :
35     ETC.actionWithCommit spec commitArm t = ETC.exploreArm spec t := by
36   simp [ETC.actionWithCommit, h]
37 
38 /--
39 After the configured exploration prefix, the phase-switching ETC trace plays
40 the supplied commit arm.
41 
42 This is the `ETC-ACTION-WITH-COMMIT-COMMIT-PHASE` project-local trace-boundary
43 leaf.
44 -/
45 @[simp] theorem actionWithCommit_eq_commitArm_of_ge
46     {K : Nat} (spec : ETC.Spec K) (commitArm : Fin K) {t : Nat}
47     (h : spec.explorationPulls * K <= t) :
48     ETC.actionWithCommit spec commitArm t = commitArm := by
49   have hnot : ¬ t < spec.explorationPulls * K := Nat.not_lt_of_ge h
50   simp [ETC.actionWithCommit, hnot]
51 
52 /--
53 After the configured exploration prefix, if the supplied commit arm is the
54 model's selected best arm, the phase-switching ETC trace plays that best arm.
55 
56 This is the `ETC-ACTION-WITH-COMMIT-BESTARM-COMMIT-PHASE` project-local
57 trace-boundary leaf.
58 -/
59 theorem actionWithCommit_eq_bestArm_of_commitArm_eq_bestArm_of_explorationPulls_mul_K_le
60     {K : Nat}
61     (spec : ETC.Spec K) (model : FiniteBanditModel K)
62     (commitArm : Fin K) (t : Nat)
63     (hcommit : commitArm = model.bestArm)
64     (ht : spec.explorationPulls * K <= t) :
65     ETC.actionWithCommit spec commitArm t = model.bestArm := by
66   rw [ETC.actionWithCommit_eq_commitArm_of_ge
67     (spec := spec)
68     (commitArm := commitArm)
69     (t := t)
70     ht]
71   exact hcommit
72 
73 end ETC
74 end BanditRLProof
