9 /-- Pathwise regret split with a deterministic large-gap filter for integration. -/
10 theorem streamTrace_gapSum_le {Ω : Type*} [MeasurableSpace Ω]
11     {k : ℕ} (hk : 0 < k) (n : ℕ) (hkn : k ≤ n)
12     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) (best : Fin k)
13     (hbest : ∀ a, mean a ≤ mean best) :
14     (∑ a, (mean best - mean a) * (pullCount (streamTrace hk n mean X ω) a n : ℝ)) ≤
15       (8*sqrt ((k : ℝ)/(n : ℝ)) + 2*optimismDeficit (X best) ((k : ℝ)/(n : ℝ)) n ω)*(n : ℝ) +
16       ∑ a, if 8*sqrt ((k : ℝ)/(n : ℝ)) ≤ mean best - mean a then
17         (mean best - mean a) * (1 + indexExceedanceCount (streamMean (X a) ω)
18           ((k : ℝ)/(n : ℝ)) (mean best - mean a) n) else 0 := by
19   classical
20   let δ : ℝ := (k : ℝ)/(n : ℝ)
21   let Z := optimismDeficit (X best) δ n ω
22   have hZ : 0 ≤ Z := optimismDeficit_nonneg _ _ _ _
23   have hbase : 0 ≤ 8*sqrt δ + 2*Z := by positivity
24   have hterm (a : Fin k) :
25       (mean best - mean a) * (pullCount (streamTrace hk n mean X ω) a n : ℝ) ≤
26       (8*sqrt δ+2*Z)*(pullCount (streamTrace hk n mean X ω) a n : ℝ) +
27       (if 8*sqrt δ ≤ mean best - mean a then
28         (mean best - mean a)*(1+indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n)
29         else 0) := by
30     have hg : 0 ≤ mean best - mean a := sub_nonneg.mpr (hbest a)
31     have hc : 0 ≤ indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n := by
32       unfold indexExceedanceCount
33       apply sum_nonneg
34       intro s _
35       split_ifs <;> norm_num
36     by_cases hl : 8*sqrt δ ≤ mean best - mean a
37     · rw [if_pos hl]
38       by_cases hz : 2*Z < mean best - mean a
39       · have ht := mul_le_mul_of_nonneg_left
40           (streamTrace_pullCount_le hk n hkn mean X ω best a hz) hg
41         have hb := mul_nonneg hbase (Nat.cast_nonneg (α := ℝ) (pullCount (streamTrace hk n mean X ω) a n))
42         linarith
43       · have hsmall : mean best - mean a ≤ 8*sqrt δ+2*Z := by
44           nlinarith [sqrt_nonneg δ]
45         have ht := mul_le_mul_of_nonneg_right hsmall
46           (Nat.cast_nonneg (α := ℝ) (pullCount (streamTrace hk n mean X ω) a n))
47         have hr := mul_nonneg hg (show 0 ≤ 1+indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n by linarith)
48         linarith
49     · rw [if_neg hl, add_zero]
50       apply mul_le_mul_of_nonneg_right _ (Nat.cast_nonneg _)
51       linarith
52   have hsum := sum_le_sum (fun a (_ : a ∈ (univ : Finset (Fin k))) => hterm a)
53   rw [sum_add_distrib, ← mul_sum] at hsum
54   have hcount : (∑ a, (pullCount (streamTrace hk n mean X ω) a n : ℝ)) = n := by
55     exact_mod_cast finset_sum_pullCount_eq_time (streamTrace hk n mean X ω) n
56   rw [hcount] at hsum
57   exact hsum

59 theorem streamTrace_realMeanRegret_le {Ω : Type*} [MeasurableSpace Ω]
60     {k : ℕ} (hk : 0 < k) (n : ℕ) (hkn : k ≤ n)
61     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (ω : Ω) (best : Fin k)
62     (hbest : ∀ a, mean a ≤ mean best) :
63     realMeanRegret mean (streamTrace hk n mean X ω) n ≤
64       (8*sqrt ((k : ℝ)/(n : ℝ)) + 2*optimismDeficit (X best) ((k : ℝ)/(n : ℝ)) n ω)*(n : ℝ) +
65       ∑ a, if 8*sqrt ((k : ℝ)/(n : ℝ)) ≤ mean best - mean a then
66         (mean best - mean a) * (1 + indexExceedanceCount (streamMean (X a) ω)
67           ((k : ℝ)/(n : ℝ)) (mean best - mean a) n) else 0 := by
68   letI : Nonempty (Fin k) := ⟨best⟩
69   have hsup : (⨆ a, mean a) = mean best :=
70     le_antisymm (ciSup_le hbest) (le_ciSup (Finite.bddAbove_range mean) best)
71   rw [realMeanRegret_eq_sum_gap_mul_pullCount]
72   simp only [realMeanGap, hsup]
73   exact streamTrace_gapSum_le hk n hkn mean X ω best hbest

75 /-- Integrated large-gap contribution, with the single initialization gap term. -/
76 theorem integral_largeGapCountSum_le {Ω : Type*} [MeasurableSpace Ω]
77     (μ : Measure Ω) [IsProbabilityMeasure μ] {k : ℕ}
78     (mean : Fin k → ℝ) (X : Fin k → ℕ → Ω → ℝ) (best : Fin k)
79     (hbest : ∀ a, mean a ≤ mean best)
80     (hXm : ∀ a i, StronglyMeasurable (X a i))
81     (hind : ∀ a, iIndepFun (X a) μ)
82     (hmean : ∀ a i, ∫ ω, X a i ω ∂μ = 0)
83     (hsubG : ∀ a i, HasSubgaussianMGF (X a i) 1 μ)
84     (δ : ℝ) (hδ : 0 < δ) (n : ℕ) :
85     (∫ ω, ∑ a, if 8*sqrt δ ≤ mean best - mean a then
86       (mean best - mean a)*(1+indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n)
87       else 0 ∂μ) ≤ (∑ a, (mean best-mean a)) + (k : ℝ)*(15/sqrt δ) := by
88   classical
89   let f : Fin k → Ω → ℝ := fun a ω => if 8*sqrt δ ≤ mean best-mean a then
90     (mean best-mean a)*(1+indexExceedanceCount (streamMean (X a) ω) δ (mean best-mean a) n) else 0
91   have hi (a : Fin k) : Integrable (f a) μ := by
92     dsimp [f]
93     split_ifs
94     · exact ((integrable_const 1).add (integrable_indexExceedanceCount (X a) (hXm a) δ _ n)).const_mul _
95     · exact integrable_const 0
96   have hb (a : Fin k) : (∫ ω, f a ω ∂μ) ≤ mean best-mean a + 15/sqrt δ := by
97     have hg : 0 ≤ mean best-mean a := sub_nonneg.mpr (hbest a)
98     dsimp [f]
99     split_ifs with hl
100     · have hgp : 0 < mean best-mean a := lt_of_lt_of_le (by positivity) hl
101       have hc := gap_mul_integral_indexExceedanceCount_le_sharp (X a) (hXm a) (hind a)
102         (hmean a) (hsubG a) δ (mean best-mean a) hδ hgp hl n
103       rw [integral_const_mul]
104       rw [integral_add (integrable_const 1) (integrable_indexExceedanceCount (X a) (hXm a) δ _ n)]
105       simp only [integral_const, probReal_univ, one_smul]
106       nlinarith
107     · simp only [integral_zero]
108       positivity
109   change (∫ ω, ∑ a, f a ω ∂μ) ≤ _
110   rw [integral_finset_sum _ (fun a _ => hi a)]
111   calc
112     _ ≤ ∑ a, (mean best-mean a+15/sqrt δ) := sum_le_sum (fun a _ => hb a)
113     _ = _ := by simp [sum_add_distrib, mul_comm]
