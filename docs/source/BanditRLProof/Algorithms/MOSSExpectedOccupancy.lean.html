12 theorem fixedLogExceedanceCount_eq_fixedRadiusCount (X : ℕ → Ω → ℝ)
13     (δ gap : ℝ) (n : ℕ) (ω : Ω) :
14     fixedLogExceedanceCount (streamMean X ω) δ gap n =
15       Concentration.fixedRadiusCount X (2*logPlus (gap^2/δ)) (gap/2) n ω := by
16   unfold fixedLogExceedanceCount Concentration.fixedRadiusCount
17   apply Finset.sum_congr rfl
18   intro s hs
19   have hr : 4/((s+1 : ℕ) : ℝ)*logPlus (gap^2/δ) =
20       2*(2*logPlus (gap^2/δ))/((s+1 : ℕ) : ℝ) := by ring
21   simp only [Set.indicator_apply, Concentration.fixedRadiusMeanEvent, Set.mem_setOf_eq,
22     streamMean, peelingSum, fixedLogRadius, hr]

37 theorem integral_indexExceedanceCount_le_sharp (X : ℕ → Ω → ℝ)
38     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
39     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0) (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
40     (δ gap : ℝ) (hδ : 0 < δ) (hg : 0 < gap) (hlarge : δ < gap^2) (n : ℕ) :
41     (∫ ω, indexExceedanceCount (streamMean X ω) δ gap n ∂μ) ≤
42       1/gap^2 + (8/gap^2)*(2*logPlus (gap^2/δ)+sqrt (Real.pi*(2*logPlus (gap^2/δ)))+1) := by
43   have hlog : 0 < logPlus (gap^2/δ) := by
44     unfold logPlus
45     apply log_pos
46     exact lt_of_lt_of_le ((one_lt_div hδ).mpr hlarge) (le_max_right _ _)
47   have hi := Concentration.integrable_fixedRadiusCount (μ := μ) X hXm
48     (2*logPlus (gap^2/δ)) (gap/2) n
49   have hle := integral_mono_of_nonneg
50     (Filter.Eventually.of_forall (fun ω => show 0 ≤ indexExceedanceCount (streamMean X ω) δ gap n from by
51       unfold indexExceedanceCount
52       apply Finset.sum_nonneg

65   rw [he] at hb
