24 /--
25 If the commit arm is measurable, then the event that it is not the selected
26 best arm is measurable.
27 
28 This is the first wrong-commit event canary after the deterministic fixed-commit
29 ETC layer.  It is intentionally only an event measurability fact.
30 -/
31 theorem measurableSet_commitArm_ne_bestArm
32     {Omega : Type u} {K : Nat}
33     [MeasurableSpace Omega]
34     [MeasurableSpace (Fin K)] [MeasurableSingletonClass (Fin K)]
35     (model : FiniteBanditModel K)
36     (commitArm : Omega -> Fin K)
37     (hmeas_commit : Measurable commitArm) :
38     MeasurableSet {omega : Omega | commitArm omega = model.bestArm -> False} := by
39   have hEq : MeasurableSet {omega : Omega | commitArm omega = model.bestArm} := by
40     simpa [Set.preimage] using
41       (hmeas_commit (MeasurableSet.singleton model.bestArm))
42   simpa [Set.compl_setOf] using hEq.compl

194 /--
195 Pairwise empirical-mean comparison events are measurable when each empirical
196 mean coordinate is measurable.
197 
198 This is an ordered-event regularity canary for the wrong-mean event.  It does
199 not use measures, probability, commit-arm argmax, finite unions, concentration,

202 theorem measurableSet_empMean_ge_empMean
203     {Omega : Type u} {K : Nat}
204     [MeasurableSpace Omega]
205     (empMean : Omega -> Fin K -> Rat)
206     (hmeas_empMean :
207       forall a : Fin K, Measurable (fun omega : Omega => empMean omega a))
208     (a b : Fin K) :
209     MeasurableSet {omega : Omega | empMean omega a >= empMean omega b} := by
210   change MeasurableSet {omega : Omega | empMean omega b <= empMean omega a}
211   exact measurableSet_le (hmeas_empMean b) (hmeas_empMean a)

213 /--
214 The finite existential wrong-mean event is measurable when each empirical mean
215 coordinate is measurable.
216 
217 This packages the pairwise empirical-mean comparison canary into the exact
218 finite event shape used by the wrong-commit probability bridge.  It does not
219 use measures, probability, commit-arm argmax, concentration, filtration, or an

222 theorem measurableSet_exists_ne_bestArm_empMean_ge_bestArm
223     {Omega : Type u} {K : Nat}
224     [MeasurableSpace Omega]
225     (model : FiniteBanditModel K)
226     (empMean : Omega -> Fin K -> Rat)
227     (hmeas_empMean :
228       forall a : Fin K, Measurable (fun omega : Omega => empMean omega a)) :
229     MeasurableSet {omega : Omega |
230       exists a : Fin K, (a = model.bestArm -> False) /\
231         empMean omega a >= empMean omega model.bestArm} := by
232   classical
233   have hset :
234       {omega : Omega |
235         exists a : Fin K, (a = model.bestArm -> False) /\
236           empMean omega a >= empMean omega model.bestArm}
237       =
238       (⋃ a ∈ (Finset.univ : Finset (Fin K)),
239         {omega : Omega | (a = model.bestArm -> False) /\
240           empMean omega a >= empMean omega model.bestArm}) := by
241     ext omega
242     simp
243   rw [hset]
244   refine Finset.measurableSet_biUnion _ ?_
245   intro a _ha
246   by_cases h : a = model.bestArm
247   · have hempty :
248         {omega : Omega | (a = model.bestArm -> False) /\
249           empMean omega a >= empMean omega model.bestArm}
250         = (∅ : Set Omega) := by
251         ext omega
252         simp [h]
253     rw [hempty]
254     exact MeasurableSet.empty
255   · have hpair :
256         MeasurableSet {omega : Omega |
257           empMean omega a >= empMean omega model.bestArm} :=
258         measurableSet_empMean_ge_empMean
259           empMean hmeas_empMean a model.bestArm
260     simpa [h] using hpair

262 /--
263 The wrong-commit event is contained in the event that some non-best arm's
264 empirical mean beats or ties the selected best arm's empirical mean, assuming
265 the commit arm is an empirical-mean argmax.
266

270 theorem wrong_commit_subset_exists_empMean_ge_bestArm
271     {Omega : Type u} {K : Nat}
272     (model : FiniteBanditModel K)
273     (commitArm : Omega -> Fin K)
274     (empMean : Omega -> Fin K -> Rat)
275     (hcommit_argmax :
276       forall omega : Omega, forall a : Fin K,
277         empMean omega a <= empMean omega (commitArm omega)) :
278     Set.Subset
279       {omega : Omega | commitArm omega = model.bestArm -> False}
280       {omega : Omega |
281         exists a : Fin K, (a = model.bestArm -> False) /\
282           empMean omega a >= empMean omega model.bestArm} := by
283   intro omega hwrong
284   refine Exists.intro (commitArm omega) ?_
285   exact And.intro hwrong (by
286     change empMean omega model.bestArm <= empMean omega (commitArm omega)
287     exact hcommit_argmax omega model.bestArm)

357 theorem prob_exists_ne_bestArm_empMean_ge_bestArm_le_sum
358     {Omega : Type u} {K : Nat}

361     (model : FiniteBanditModel K)
362     (empMean : Omega -> Fin K -> Rat) :
363     mu {omega : Omega |
364       exists a : Fin K, (a = model.bestArm -> False) /\
365         empMean omega a >= empMean omega model.bestArm} <=

368         mu {omega : Omega | (a = model.bestArm -> False) /\

371   let E : Fin K -> Set Omega := fun a =>
372     {omega : Omega | (a = model.bestArm -> False) /\
373       empMean omega a >= empMean omega model.bestArm}

377           empMean omega a >= empMean omega model.bestArm}

379       (⋃ a ∈ (Finset.univ : Finset (Fin K)), E a) := by

385       (μ := mu)
386       (I := (Finset.univ : Finset (Fin K)))
387       (s := E))
