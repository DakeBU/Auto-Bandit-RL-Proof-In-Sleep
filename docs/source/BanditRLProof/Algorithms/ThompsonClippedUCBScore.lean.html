6 # Clipped UCB score for the Thompson Bayesian-regret route
7 
8 This module instantiates `HistoryActionScore` with the clipped upper-confidence
9 score used by the pinned LML Thompson regret proof.  The finite-history score
10 is measurable and lies in `[l, u]`; those bounds discharge every score
11 integrability premise in the compiled Bayesian-regret decomposition.

23 /-- The clipped UCB score on a complete action/reward trace. -/
24 def clippedUCB
25     {K : Nat} (l u sigma2 delta : Real)
26     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
27     (arm : Fin K) (t : Nat) : Real :=
28   if pullCount action arm t = 0 then u
29   else
30     max l (min u
31       (UCB.realEmpiricalMean action reward arm t +
32         Real.sqrt
33           (2 * sigma2 * Real.log (1 / delta) /
34             (pullCount action arm t : Real))))
35 
36 /-- The same clipped score on the inclusive history through time `n`. -/
37 def clippedUCBHistory
38     {K : Nat} (l u sigma2 delta : Real) (n : Nat)
39     (history : History.FinitePairHistory (Fin K) Real n)
40     (arm : Fin K) : Real :=
41   if ETC.realHistoryPullCount n history arm = 0 then u
42   else
43     max l (min u
44       (ETC.realHistoryEmpMean n history arm +
45         Real.sqrt
46           (2 * sigma2 * Real.log (1 / delta) /
47             (ETC.realHistoryPullCount n history arm : Real))))

68 private theorem finset_sum_sqrt_le
69     {ι : Type*} (s : Finset ι) (c : ι -> Real)
70     (hc : forall i, 0 <= c i) :
71     ∑ i ∈ s, Real.sqrt (c i) <=
72       Real.sqrt (s.card * ∑ i ∈ s, c i) := by
73   have h := Real.sum_sqrt_mul_sqrt_le s hc (fun _ => zero_le_one)
74   simp only [Real.sqrt_one, mul_one, Finset.sum_const, nsmul_eq_mul] at h
75   rwa [Real.sqrt_mul (by positivity), mul_comm]

77 private theorem finset_sum_one_div_sqrt_le
78     {n : Nat} (hn : 0 < n) :
79     ∑ k ∈ Finset.range (n + 1), 1 / Real.sqrt k <=
80       2 * Real.sqrt n - 1 := by
81   induction n with
82   | zero => simp at hn
83   | succ n ih =>
84       rw [Finset.sum_range_succ]
85       by_cases hzero : n = 0
86       · rw [hzero]
87         simp
88         norm_num
89       · have hih := ih (Nat.pos_of_ne_zero hzero)
90         suffices 1 / Real.sqrt (n + 1) <=
91             2 * (Real.sqrt (n + 1) - Real.sqrt n) by
92           norm_num [Nat.cast_add, Nat.cast_one] at *
93           linarith
94         field_simp
95         have hsqrtSucc : Real.sqrt (n + 1) * Real.sqrt (n + 1) = n + 1 :=
96           Real.mul_self_sqrt (by positivity)
97         have hsqrt : Real.sqrt n * Real.sqrt n = n :=
98           Real.mul_self_sqrt (by positivity)
99         nlinarith

101 /--
102 Pathwise selected-action clipped-UCB excess bound from the pinned LML
103 Thompson route. The only hypothesis is that every positive-count empirical
104 mean is strictly below its arm mean plus the confidence width.
105 -/
106 theorem sum_clippedUCB_action_sub_mean_le
107     {K : Nat} [NeZero K]
108     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
109     (mean : Fin K -> Real) (l u sigma2 delta : Real)
110     (hmeanMem : forall arm, mean arm ∈ Set.Icc l u) (hlu : l <= u)
111     (hgood : forall s, s < n ->
112       pullCount action (action s) s ≠ 0 ->
113         UCB.realEmpiricalMean action reward (action s) s - mean (action s) <
114           Real.sqrt
115             (2 * sigma2 * Real.log (1 / delta) /
116               (pullCount action (action s) s : Real))) :
117     ∑ s ∈ Finset.range n,
118         (clippedUCB l u sigma2 delta action reward (action s) s -
119           mean (action s)) <=
120       (u - l) * K +
121         4 * Real.sqrt (2 * sigma2 * Real.log (1 / delta) * K * n) := by
122   classical
123   let S0 := (Finset.range n).filter
124     (fun s => pullCount action (action s) s = 0)
125   let S1 := (Finset.range n).filter
126     (fun s => pullCount action (action s) s ≠ 0)
127   have hunion : S0 ∪ S1 = Finset.range n := by
128     exact Finset.filter_union_filter_not_eq _ _
129   have hdisjoint : Disjoint S0 S1 :=
130     Finset.disjoint_filter_filter_not _ _ _
131   rw [← hunion, Finset.sum_union hdisjoint]
132   gcongr
133   · calc
134       ∑ s ∈ S0,
135           (clippedUCB l u sigma2 delta action reward (action s) s -
136             mean (action s)) <=
137           ∑ _s ∈ S0, (u - l) := by
138             apply Finset.sum_le_sum
139             intro s hs
140             have hscore := clippedUCB_mem_Icc
141               l u sigma2 delta hlu action reward (action s) s
142             have hmean := hmeanMem (action s)
143             nlinarith [hscore.2, hmean.1]
144       _ = ∑ s ∈ Finset.range n,
145           if pullCount action (action s) s = 0 then (u - l) else 0 := by
146             dsimp only [S0]
147             rw [Finset.sum_filter]
148       _ = ∑ arm : Fin K,
149           ∑ j ∈ Finset.range (pullCount action arm n),
150             if j = 0 then (u - l) else 0 := by
151             exact finset_sum_comp_pullCount action n
152               (fun j => if j = 0 then (u - l) else 0)
153       _ <= ∑ _arm : Fin K, (u - l) := by
154             apply Finset.sum_le_sum
155             intro arm _harm
156             rw [Finset.sum_ite_eq']
157             split_ifs
158             · rfl
159             · exact sub_nonneg.mpr hlu
160       _ = (u - l) * K := by
161             rw [Fin.sum_const, nsmul_eq_mul, mul_comm]
162   · calc
163       ∑ s ∈ S1,
164           (clippedUCB l u sigma2 delta action reward (action s) s -
165             mean (action s)) <=
166           ∑ s ∈ S1,
167             2 * Real.sqrt
168               (2 * sigma2 * Real.log (1 / delta) /
169                 (pullCount action (action s) s : Real)) := by
170             apply Finset.sum_le_sum
171             intro s hs
172             have hsRange : s ∈ Finset.range n := (Finset.mem_filter.mp hs).1
173             have hsCount : pullCount action (action s) s ≠ 0 :=
174               (Finset.mem_filter.mp hs).2
175             have hwidth : 0 <= Real.sqrt
176                 (2 * sigma2 * Real.log (1 / delta) /
177                   (pullCount action (action s) s : Real)) :=
178               Real.sqrt_nonneg _
179             unfold clippedUCB
180             rw [if_neg hsCount]
181             have hraw := hgood s (Finset.mem_range.mp hsRange) hsCount
182             have hmean := hmeanMem (action s)
183             grind
184       _ <= ∑ s ∈ Finset.range n,
185           2 * Real.sqrt
186             (2 * sigma2 * Real.log (1 / delta) /
187               (pullCount action (action s) s : Real)) := by
188             apply Finset.sum_le_sum_of_subset_of_nonneg (Finset.filter_subset _ _)
189             intro s _hs _hnot
190             positivity
191       _ = 2 * Real.sqrt (2 * sigma2 * Real.log (1 / delta)) *
192           ∑ s ∈ Finset.range n,
193             (1 / Real.sqrt (pullCount action (action s) s)) := by
194             rw [Finset.mul_sum]
195             apply Finset.sum_congr rfl
196             intro s _hs
197             rw [Real.sqrt_div' _ (by positivity)]
198             ring
199       _ = 2 * Real.sqrt (2 * sigma2 * Real.log (1 / delta)) *
200           ∑ arm : Fin K,
201             ∑ j ∈ Finset.range (pullCount action arm n),
202               (1 / Real.sqrt j) := by
203             congr 1
204             exact finset_sum_comp_pullCount action n
205               (fun j : Nat => 1 / Real.sqrt (j : Real))
206       _ <= 2 * Real.sqrt (2 * sigma2 * Real.log (1 / delta)) *
207           (2 * ∑ arm : Fin K, Real.sqrt (pullCount action arm n)) := by
208             rw [Finset.mul_sum _ _ 2]
209             gcongr with arm
210             by_cases hcount : pullCount action arm n = 0
211             · simp [hcount]
212             · have hbound := finset_sum_one_div_sqrt_le
213                 (Nat.pos_of_ne_zero hcount)
214               rw [Finset.sum_range_succ] at hbound
215               have hnonneg : 0 <=
216                   1 / Real.sqrt (pullCount action arm n) := by
217                 positivity
218               linarith
219       _ <= 2 * Real.sqrt (2 * sigma2 * Real.log (1 / delta)) *
220           (2 * Real.sqrt (K *
221             ∑ arm : Fin K, (pullCount action arm n : Real))) := by
222             gcongr
223             have hsqrt := finset_sum_sqrt_le
224               (Finset.univ : Finset (Fin K))
225               (fun arm => (pullCount action arm n : Real))
226               (fun arm => Nat.cast_nonneg _)
227             simpa using hsqrt
228       _ = 2 * Real.sqrt (2 * sigma2 * Real.log (1 / delta)) *
229           (2 * Real.sqrt (K * n)) := by
230             rw [← Nat.cast_sum, finset_sum_pullCount_eq_time action n]
231       _ = 4 * Real.sqrt
232           (2 * sigma2 * Real.log (1 / delta) * K * n) := by
233             ring_nf
234             rw [← Real.sqrt_mul' _ (by positivity)]
235             ring_nf

248 /-- A fixed-arm clipped score is measurable on inclusive pair histories. -/
249 theorem measurable_clippedUCBHistory
250     {K : Nat} (l u sigma2 delta : Real) (n : Nat) (arm : Fin K) :
251     Measurable (fun history : History.FinitePairHistory (Fin K) Real n =>
252       clippedUCBHistory l u sigma2 delta n history arm) := by
253   let hcount : Measurable
254       (fun history : History.FinitePairHistory (Fin K) Real n =>
255         ETC.realHistoryPullCount n history arm) :=
256     UCB.measurable_realHistoryPullCount n arm
257   let hcountReal : Measurable
258       (fun history : History.FinitePairHistory (Fin K) Real n =>
259         (ETC.realHistoryPullCount n history arm : Real)) :=
260     (measurable_of_countable (fun count : Nat => (count : Real))).comp hcount
261   let hraw : Measurable
262       (fun history : History.FinitePairHistory (Fin K) Real n =>
263         ETC.realHistoryEmpMean n history arm +
264           Real.sqrt
265             (2 * sigma2 * Real.log (1 / delta) /
266               (ETC.realHistoryPullCount n history arm : Real))) :=
267     (UCB.measurable_realHistoryEmpMean n arm).add
268       ((measurable_const.div hcountReal).sqrt)
269   exact Measurable.ite
270     (measurableSet_eq_fun hcount measurable_const)
271     measurable_const
272     (measurable_const.max (measurable_const.min hraw))

333 /-- A measurable real function with pointwise range in `[l, u]` is integrable. -/
334 theorem integrable_of_measurable_mem_Icc
335     {Omega : Type u} [MeasurableSpace Omega]
336     (mu : Measure Omega) [IsFiniteMeasure mu]
337     (f : Omega -> Real) (hf : Measurable f)
338     {l u : Real} (hmem : forall omega, f omega ∈ Set.Icc l u) :
339     Integrable f mu := by
340   refine Integrable.of_bound hf.aestronglyMeasurable (max |l| |u|) ?_
341   filter_upwards with omega
342   rw [Real.norm_eq_abs]
343   have h := hmem omega
344   rcases h with ⟨hl, hu⟩
345   exact abs_le_max_abs_abs hl hu

428 theorem integral_trajectoryBayesMeanRegret_eq_add_clippedUCB
429     {Env : Type u} {K : Nat}
430     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
431     [Nonempty (Fin K)]
432     (prior : Measure Env) [IsProbabilityMeasure prior]
433     (environment : MeasurableHistoryEnvironment Env (Fin K) Real)
434     (bestAction : Env -> Fin K) (hbestAction : Measurable bestAction)
435     (mean : Env -> Fin K -> Real)
436     (hmean : Measurable (fun pair : Env × Fin K => mean pair.1 pair.2))
437     (l u sigma2 delta : Real) (hlu : l <= u)
438     (hmeanMem : forall env arm, mean env arm ∈ Set.Icc l u)
439     (horizon : Nat) :
440     let algorithm :=
441       uniformReferenceThompsonAlgorithm prior environment bestAction hbestAction
442     let actualMeasure :=
443       prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment
444     integral actualMeasure (fun sample =>
445         trajectoryBayesMeanRegret mean bestAction sample horizon) =
446       integral actualMeasure (fun sample =>
447         ∑ t ∈ range horizon,
448           (mean sample.1 (bestAction sample.1) -
449             clippedUCB l u sigma2 delta
450               (environmentTrajectoryAction sample)
451               (environmentTrajectoryReward sample)
452               (bestAction sample.1) t)) +
453       integral actualMeasure (fun sample =>
454         ∑ t ∈ range horizon,
455           (clippedUCB l u sigma2 delta
456               (environmentTrajectoryAction sample)
457               (environmentTrajectoryReward sample)
458               (environmentTrajectoryAction sample t) t -
459             mean sample.1 (environmentTrajectoryAction sample t))) := by
460   dsimp only
461   let algorithm :=
462     uniformReferenceThompsonAlgorithm prior environment bestAction hbestAction
463   let actualMeasure :=
464     prior ⊗ₘ canonicalMeasurableEnvironmentTrajectoryKernel algorithm environment
465   have hmeanBest : Integrable (fun sample :
466       Env × ((n : Nat) -> Fin K × Real) =>
467         mean sample.1 (bestAction sample.1)) actualMeasure :=
468     integrable_trajectoryMean_bestAction actualMeasure mean hmean hmeanMem
469       bestAction hbestAction
470   have hmeanAction : forall t : Nat, t < horizon -> Integrable (fun sample :
471       Env × ((n : Nat) -> Fin K × Real) =>
472         mean sample.1 (environmentTrajectoryAction sample t)) actualMeasure :=
473     fun t _ => integrable_trajectoryMean_action
474       actualMeasure mean hmean hmeanMem t
475   have hscoreAction : forall t : Nat, t < horizon -> Integrable (fun sample :
476       Env × ((n : Nat) -> Fin K × Real) =>
477         trajectoryHistoryScore
478           (clippedUCBHistoryScore l u sigma2 delta) sample t) actualMeasure :=
479     fun t _ => integrable_trajectoryHistoryScore_clippedUCB
480       actualMeasure l u sigma2 delta hlu t
481   have hscoreBest : forall t : Nat, t < horizon -> Integrable (fun sample :
482       Env × ((n : Nat) -> Fin K × Real) =>
483         trajectoryBestHistoryScore
484           (clippedUCBHistoryScore l u sigma2 delta)
485           bestAction sample t) actualMeasure :=
486     fun t _ => integrable_trajectoryBestHistoryScore_clippedUCB
487       actualMeasure bestAction hbestAction l u sigma2 delta hlu t
488   simpa only [actualMeasure, algorithm,
489     trajectoryHistoryScore, trajectoryBestHistoryScore,
490     clippedUCBHistoryScore_atTrace,
491     clippedUCBHistoryScore_atBestTrace] using
492     (integral_trajectoryBayesMeanRegret_eq_add_historyScore
493       prior environment bestAction hbestAction mean
494       (clippedUCBHistoryScore l u sigma2 delta) horizon
495       hmeanBest hmeanAction hscoreAction hscoreBest)
