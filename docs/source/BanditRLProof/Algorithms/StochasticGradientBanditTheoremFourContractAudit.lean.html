4 /-!
5 # Stochastic-gradient bandit Theorem 4: source-contract audit
6 
7 This module isolates finite scalar obligations from Appendix E, Steps 1, 3,
8 and 4, of Baudry, Johnson, Vary, Pike-Burke, and Rebeschini,
9 *Does Stochastic Gradient really succeed for Bandits?* (NeurIPS 2025).
10 
11 The camera-ready proof uses a transient-phase argument to claim a positive
12 probability of never returning below best-arm probability `1 / 2`.  Its Step 4
13 switches between the conditions `q >= c` and `q < c`; as printed, the displayed
14 total-probability direction does not provide the lower bound later consumed by
15 Step 3.  The declarations below therefore record the conservative finite
16 contract that would suffice, while leaving the source-contract mismatch open:
17 
18 * the printed learning-rate condition gives a positive drift margin;
19 * a buffered event of mass at least `pPrime`, together with conditional
20   survival mass at least `1 - 2*c`, gives unconditional survival mass at
21   least `pPrime * (1 - 2*c)`; and
22 * for `0 < rho <= 1`, a finite phase-mass sequence already known to be
23   dominated by `(1 - rho)^j` has the usual `1 / rho` geometric envelope.
24 
25 These are dependency-checked source-audit leaves.  They do **not** construct
26 the general-`K` generated SGB process, a stopped supermartingale, a Doob
27 maximal bound, the uniform buffered-event producer, or Theorem 4.
28 -/
29

36 /-- The positive scalar margin used in Appendix E after Equation (22). -/
37 def theoremFourStepOneMargin (K : Nat) (eta Delta : Real) : Real :=
38   2 * Delta - eta * sourceC eta * ((K : Real) + 2)
39 
40 /-- The unconditional survival mass required by the audited Step-4 event
41 composition. -/
42 def theoremFourStepFourSurvivalLowerBound (pPrime c : Real) : Real :=
43   pPrime * (1 - 2 * c)
44 
45 /-- The source learning-rate condition implies that the Equation-(22) drift
46 margin is strictly positive. -/
47 theorem theoremFourStepOneMargin_pos
48     (K : Nat) (eta Delta : Real)
49     (hmargin : eta * sourceC eta < 2 * Delta / ((K : Real) + 2)) :
50     0 < theoremFourStepOneMargin K eta Delta := by
51   have hden : 0 < (K : Real) + 2 := by positivity
52   have hscaled := (lt_div_iff₀ hden).mp hmargin
53   unfold theoremFourStepOneMargin
54   linarith
55 
56 /-- If `pPrime > 0` and `c < 1/2`, the audited Step-4 survival lower bound
57 is positive. -/
58 theorem theoremFourStepFourSurvivalLowerBound_pos
59     (pPrime c : Real) (hpPrime : 0 < pPrime)
60     (hc_half : c < 1 / 2) :
61     0 < theoremFourStepFourSurvivalLowerBound pPrime c := by
62   unfold theoremFourStepFourSurvivalLowerBound
63   have hfactor : 0 < 1 - 2 * c := by linarith
64   positivity
65 
66 /-- Finite total-probability contract for Appendix E, Step 4.
67 
68 `bufferedMass` represents the probability of entering the strict buffer
69 `q_{s+1} < c`; `jointSurvivalMass` represents the probability of both entering
70 that buffer and not returning before the audited finite horizon.  The two
71 middle hypotheses are the multiplication-free form of
72 `P(buffer) >= pPrime` and
73 `P(survival | buffer) >= 1 - 2*c`.
74 -/
75 theorem theoremFourStepFour_survivalMass_ge
76     (pPrime c bufferedMass jointSurvivalMass survivalMass : Real)
77     (hc_half : c < 1 / 2)
78     (hbuffer : pPrime <= bufferedMass)
79     (hconditional : (1 - 2 * c) * bufferedMass <= jointSurvivalMass)
80     (hsubset : jointSurvivalMass <= survivalMass) :
81     theoremFourStepFourSurvivalLowerBound pPrime c <= survivalMass := by
82   have hfactor : 0 <= 1 - 2 * c := by linarith
83   have hscaled := mul_le_mul_of_nonneg_left hbuffer hfactor
84   unfold theoremFourStepFourSurvivalLowerBound
85   calc
86     pPrime * (1 - 2 * c) = (1 - 2 * c) * pPrime := by ring
87     _ <= (1 - 2 * c) * bufferedMass := hscaled
88     _ <= jointSurvivalMass := hconditional
89     _ <= survivalMass := hsubset
90 
91 /-- The audited finite event contract yields a strictly positive survival
92 mass when its buffered event has positive mass. -/
93 theorem theoremFourStepFour_survivalMass_pos
94     (pPrime c bufferedMass jointSurvivalMass survivalMass : Real)
95     (hpPrime : 0 < pPrime) (hc_half : c < 1 / 2)
96     (hbuffer : pPrime <= bufferedMass)
97     (hconditional : (1 - 2 * c) * bufferedMass <= jointSurvivalMass)
98     (hsubset : jointSurvivalMass <= survivalMass) :
99     0 < survivalMass := by
100   exact (theoremFourStepFourSurvivalLowerBound_pos pPrime c hpPrime
101     hc_half).trans_le
102       (theoremFourStepFour_survivalMass_ge pPrime c bufferedMass
103         jointSurvivalMass survivalMass hc_half hbuffer hconditional
104         hsubset)
105 
106 /-- For `0 < rho <= 1`, the finite geometric phase envelope is at most
107 `1 / rho`.  This is the finite statement needed before any infinite
108 expected-phase claim. -/
109 theorem theoremFourFiniteGeometricPhaseMass_le_inv
110     (rho : Real) (hrho_pos : 0 < rho) (hrho_le_one : rho <= 1)
111     (phaseCount : Nat) :
112     (Finset.range phaseCount).sum (fun phase => (1 - rho) ^ phase) <=
113       1 / rho := by
114   have hbase_nonneg : 0 <= 1 - rho := by linarith
115   have hbase_lt_one : 1 - rho < 1 := by linarith
116   have hsummable := summable_geometric_of_lt_one hbase_nonneg hbase_lt_one
117   have hfinite := hsummable.sum_le_tsum (Finset.range phaseCount)
118     (fun phase _ => pow_nonneg hbase_nonneg phase)
119   rw [tsum_geometric_of_lt_one hbase_nonneg hbase_lt_one] at hfinite
120   simpa [one_div] using hfinite
121 
122 /-- Any finite transient-phase mass dominated termwise by the geometric
123 return envelope inherits the same `1 / rho` bound. -/
124 theorem theoremFourFiniteTransientMass_le_inv
125     (rho : Real) (hrho_pos : 0 < rho) (hrho_le_one : rho <= 1)
126     (phaseMass : Nat -> Real)
127     (hphase : forall phase, phaseMass phase <= (1 - rho) ^ phase)
128     (phaseCount : Nat) :
129     (Finset.range phaseCount).sum phaseMass <= 1 / rho := by
130   calc
131     (Finset.range phaseCount).sum phaseMass <=
132         (Finset.range phaseCount).sum (fun phase => (1 - rho) ^ phase) := by
133       exact Finset.sum_le_sum (fun phase _ => hphase phase)
134     _ <= 1 / rho :=
135       theoremFourFiniteGeometricPhaseMass_le_inv rho hrho_pos hrho_le_one
136         phaseCount
