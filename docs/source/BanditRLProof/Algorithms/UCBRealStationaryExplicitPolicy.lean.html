4 # Explicit policy semantics for the canonical-kernel stationary UCB trajectory
5 
6 This module identifies the canonical arm-stream successor action kernel with
7 the deterministic `realHistoryNextArm` kernel. It then transports the complete
8 explicit-policy graph to the independently generated canonical pair trajectory
9 and pairs that graph with the existing expected-average consistency theorem.
10 -/

19 /-- The explicit finite-history selector encoded by the canonical arm-stream policy. -/
20 noncomputable def canonicalRealUCBHistorySelector
21     {K : Nat} (hK : 0 < K) (c : Real) (sigma2 : NNReal) (n : Nat) :
22     History.FinitePairHistory (Fin K) Real n -> Fin K :=

197 /-- Every generated successor action follows the explicit UCB selector a.s. -/
198 theorem canonicalKernelTrajectoryAction_succ_ae_eq_realHistoryNextArm
199     {K : Nat} [NeZero K]
200     (hK : 0 < K) (c : Real) (sigma2 : NNReal)
201     (nu : Kernel (Fin K) Real) [IsMarkovKernel nu] (n : Nat) :
202     Filter.Eventually
203       (fun trajectory =>
204         canonicalKernelTrajectoryAction trajectory (n + 1) =
205           canonicalRealUCBHistorySelector hK c sigma2 n
206             (History.finitePairHistoryOfTrace
207               (canonicalKernelTrajectoryAction trajectory)
208               (canonicalKernelTrajectoryReward trajectory) n))
209       (ae (canonicalKernelTrajectoryMeasure hK c sigma2 nu)) := by
210   let pairTrace := fun trajectory : (t : Nat) -> Prod (Fin K) Real =>
211     fun t => (canonicalKernelTrajectoryAction trajectory t,
212       canonicalKernelTrajectoryReward trajectory t)
213   let canonicalPairTrace := fun stream : ArmRewardStream K =>
214     fun t => (armStreamAction hK (c * (sigma2 : Real)) stream t,
215       armStreamReward hK (c * (sigma2 : Real)) stream t)
216   let policyGraph := {trace : (t : Nat) -> Prod (Fin K) Real |
217     (trace (n + 1)).1 =
218       canonicalRealUCBHistorySelector hK c sigma2 n
219         (History.finitePairHistoryOfTrace
220           (fun t => (trace t).1) (fun t => (trace t).2) n)}
221   have hhistory : Measurable
222       (fun trace : (t : Nat) -> Prod (Fin K) Real =>
223         History.finitePairHistoryOfTrace
224           (fun t => (trace t).1) (fun t => (trace t).2) n) :=
225     History.measurable_finitePairHistoryOfTrace
226       (fun trace : (t : Nat) -> Prod (Fin K) Real =>
227         fun t => (trace t).1)
228       (fun trace : (t : Nat) -> Prod (Fin K) Real =>
229         fun t => (trace t).2)
230       (fun t => measurable_fst.comp (measurable_pi_apply t))
231       (fun t => measurable_snd.comp (measurable_pi_apply t)) n
232   have hpolicyGraph : MeasurableSet policyGraph := by
233     exact measurableSet_eq_fun
234       (measurable_fst.comp (measurable_pi_apply (n + 1)))
235       ((measurable_canonicalRealUCBHistorySelector hK c sigma2 n).comp
236         hhistory)
237   have hcanonical :
238       Filter.Eventually
239         (fun stream : ArmRewardStream K => canonicalPairTrace stream ∈ policyGraph)
240         (ae (armStreamMeasure nu)) := by
241     filter_upwards [] with stream
242     exact armStreamAction_succ_eq_realHistoryNextArm_actualHistory
243       hK (c * (sigma2 : Real)) stream n
244   have htrajectory :=
245     identDistrib_actionRewardTrace_of_realStationaryUCBSequence
246       (canonicalKernelTrajectoryMeasure hK c sigma2 nu)
247       hK c sigma2 nu canonicalKernelTrajectoryAction
248       canonicalKernelTrajectoryReward
249       (realStationaryUCBSequence_canonicalKernelTrajectory
250         hK c sigma2 nu)
251   have hgenerated := htrajectory.symm.ae_mem_snd hpolicyGraph hcanonical
252   simpa [pairTrace, canonicalPairTrace, policyGraph] using hgenerated
