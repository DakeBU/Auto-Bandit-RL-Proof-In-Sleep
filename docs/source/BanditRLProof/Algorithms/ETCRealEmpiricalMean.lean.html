21 /-- The exploration empirical mean formed directly from a Real reward trace. -/
22 noncomputable def realEmpMeanAtExploration {K : Nat} (spec : ETC.Spec K)
23     (baseCommitArm : Fin K) (reward : RewardTrace Real) (a : Fin K) : Real :=
24   sumRewards (ETC.actionWithCommit spec baseCommitArm) reward a
25       (spec.explorationPulls * K) /
26     (pullCount (ETC.actionWithCommit spec baseCommitArm) a
27       (spec.explorationPulls * K) : Real)

32     (reward : RewardTrace Real) (a : Fin K) :

34       sumRewards (ETC.actionWithCommit spec baseCommitArm) reward a

36         (spec.explorationPulls : Real) := by

40 /-- Real exploration empirical means are measurable from measurable rewards. -/
41 theorem measurable_realEmpMeanAtExploration
42     {Omega : Type u} {K : Nat}
43     [MeasurableSpace Omega]
44     (spec : ETC.Spec K) (baseCommitArm a : Fin K)
45     (reward : Omega -> RewardTrace Real)
46     (hreward : forall t : Nat,
47       Measurable (fun omega : Omega => reward omega t)) :
48     Measurable (fun omega : Omega =>
49       ETC.realEmpMeanAtExploration spec baseCommitArm (reward omega) a) := by
50   have hsum : Measurable (fun omega : Omega =>
51       sumRewards (ETC.actionWithCommit spec baseCommitArm) (reward omega) a
52         (spec.explorationPulls * K)) := by
53     simpa using
54       (measurable_sumRewards
55         (action := fun _ : Omega => ETC.actionWithCommit spec baseCommitArm)
56         (reward := reward)
57         (fun _t => measurable_const) hreward a
58         (spec.explorationPulls * K))
59   simpa [ETC.realEmpMeanAtExploration] using
60     hsum.div_const
61       (pullCount (ETC.actionWithCommit spec baseCommitArm) a
62         (spec.explorationPulls * K) : Real)

64 private theorem real_score_le_foldl_select
65     {K : Nat} (scores : Fin K -> Real) (init : Fin K) :
66     forall l : List (Fin K),
67       (forall a : Fin K, List.Mem a l ->
68         scores a <=
69           scores
70             (l.foldl
71               (fun best arm : Fin K =>
72                 if scores best < scores arm then arm else best)
73               init)) /\
74       scores init <=
75         scores
76           (l.foldl
77             (fun best arm : Fin K =>
78               if scores best < scores arm then arm else best)
79             init)
80   | [] => by
81       exact And.intro (by intro _ ha; cases ha) (by simp)
82   | arm :: rest => by
83       let select := fun best arm : Fin K =>
84         if scores best < scores arm then arm else best
85       let next := select init arm
86       have ih := real_score_le_foldl_select (scores := scores) next rest
87       have harm_next : scores arm <= scores next := by
88         exact if hlt : scores init < scores arm then
89           by simp [next, select, hlt]
90         else
91           by simp [next, select, hlt, le_of_not_gt hlt]
92       exact And.intro
93         (by
94           intro a ha
95           cases ha with
96           | head => exact le_trans harm_next ih.2
97           | tail _ ha => exact ih.1 a ha)
98         (by
99           have hinit_next : scores init <= scores next := by
100             exact if hlt : scores init < scores arm then
101               by simpa [next, select, hlt] using le_of_lt hlt
102             else
103               by simp [next, select, hlt]
104           exact le_trans hinit_next ih.2)

106 /-- A deterministic finite argmax for Real-valued arm scores. -/
107 noncomputable def realArgmaxCommit {K : Nat} (hK : 0 < K)
108     (scores : Fin K -> Real) : Fin K :=
109   (List.finRange K).foldl
110     (fun best arm : Fin K =>
111       if scores best < scores arm then arm else best)
112     (Fin.mk 0 hK)
113 
114 /-- The Real finite argmax dominates every arm score. -/
115 theorem realArgmaxCommit_spec {K : Nat} (hK : 0 < K)
116     (scores : Fin K -> Real) (a : Fin K) :
117     scores a <= scores (ETC.realArgmaxCommit hK scores) := by
118   exact
119     (real_score_le_foldl_select
120       (scores := scores) (init := Fin.mk 0 hK) (List.finRange K)).1
121       a (List.mem_finRange a)

123 /-- On a constant score vector, the tie rule keeps the initial arm `0`. -/

125     ETC.realArgmaxCommit hK (fun _a : Fin K => c) = Fin.mk 0 hK := by

128 private theorem measurable_selected_real_score
129     {Omega : Type u} {K : Nat}
130     [MeasurableSpace Omega]
131     (scores : Omega -> Fin K -> Real)
132     (hscores : forall a : Fin K,
133       Measurable (fun omega : Omega => scores omega a))
134     (best : Omega -> Fin K) (hbest : Measurable best) :
135     Measurable (fun omega : Omega => scores omega (best omega)) := by
136   classical
137   have hsum : Measurable (fun omega : Omega =>
138       (Finset.univ : Finset (Fin K)).sum (fun a =>
139         if best omega = a then scores omega a else 0)) := by
140     refine Finset.measurable_sum _ ?_
141     intro a _ha
142     exact Measurable.ite
143       (measurableSet_eq_fun hbest measurable_const)
144       (hscores a) measurable_const
145   convert hsum using 1
146   funext omega
147   simp

149 private theorem measurable_foldl_real_select
150     {Omega : Type u} {K : Nat}
151     [MeasurableSpace Omega]
152     (scores : Omega -> Fin K -> Real)
153     (hscores : forall a : Fin K,
154       Measurable (fun omega : Omega => scores omega a))
155     (best : Omega -> Fin K) (hbest : Measurable best) :
156     forall l : List (Fin K),
157       Measurable (fun omega : Omega =>
158         l.foldl
159           (fun best arm : Fin K =>
160             if scores omega best < scores omega arm then arm else best)
161           (best omega))
162   | [] => by simpa using hbest
163   | arm :: rest => by
164       have hselected : Measurable (fun omega : Omega =>
165           scores omega (best omega)) :=
166         measurable_selected_real_score scores hscores best hbest
167       have hnext : Measurable (fun omega : Omega =>
168           if scores omega (best omega) < scores omega arm then
169             arm
170           else
171             best omega) :=
172         Measurable.ite
173           (measurableSet_lt hselected (hscores arm))
174           measurable_const hbest
175       simpa using
176         (measurable_foldl_real_select scores hscores
177           (fun omega : Omega =>
178             if scores omega (best omega) < scores omega arm then
179               arm
180             else
181               best omega)
182           hnext rest)

184 /-- The finite Real argmax is measurable when every score coordinate is. -/
185 theorem measurable_realArgmaxCommit_of_forall_measurable
186     {Omega : Type u} {K : Nat}
187     [MeasurableSpace Omega]
188     (hK : 0 < K) (scores : Omega -> Fin K -> Real)
189     (hscores : forall a : Fin K,
190       Measurable (fun omega : Omega => scores omega a)) :
191     Measurable (fun omega : Omega => ETC.realArgmaxCommit hK (scores omega)) := by
192   unfold ETC.realArgmaxCommit
193   exact measurable_foldl_real_select scores hscores
194     (fun _omega : Omega => Fin.mk 0 hK) measurable_const (List.finRange K)
