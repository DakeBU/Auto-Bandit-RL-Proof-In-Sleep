7 def conditionCoordinate {k : ℕ} (t : ℕ)
8     (c : History.FinitePairHistory (Fin k) ℝ t × Fin k) : ℕ × Fin k :=
9   (ETC.realHistoryPullCount t c.1 c.2+1, c.2)

11 def conditionBranch {k : ℕ} (t : ℕ) (target : ℕ × Fin k) :=
12   {c : History.FinitePairHistory (Fin k) ℝ t × Fin k | conditionCoordinate t c = target}

14 theorem measurable_conditionCoordinate {k : ℕ} (t : ℕ) : Measurable (conditionCoordinate (k := k) t) := by
15   have hc : Measurable (fun c : History.FinitePairHistory (Fin k) ℝ t × Fin k =>
16       ETC.realHistoryPullCount t c.1 c.2) := by
17     apply measurable_from_prod_countable_left
18     intro a
19     exact UCB.measurable_realHistoryPullCount t a
20   exact (hc.add_const 1).prodMk measurable_snd

44 theorem map_rebuilt_restrict_conditionBranch {k : ℕ} (hk : 0 < k) (n : ℕ)
45     (mean : Fin k → ℝ) (t : ℕ) (target : ℕ × Fin k) (v : ℝ) (μ : Measure (UCB.ArmRewardStream k)) :
46     (Measure.map (fun table => canonicalConditionWithout hk n mean t target v
47       (UCB.armStreamWithoutCoordinate target table)) μ).restrict (conditionBranch t target) =
48     (Measure.map (canonicalCondition hk n mean t) μ).restrict (conditionBranch t target) := by
49   have hr := (measurable_canonicalConditionWithout hk n mean t target v).comp
50     (UCB.measurable_armStreamWithoutCoordinate target)
51   have hc := measurable_canonicalCondition hk n mean t
52   simp only [Function.comp_def] at hr
53   have hb := measurableSet_conditionBranch t target
54   rw [Measure.restrict_map hr hb, Measure.restrict_map hc hb]
55   have he : (fun table => canonicalConditionWithout hk n mean t target v
56       (UCB.armStreamWithoutCoordinate target table)) ⁻¹' conditionBranch t target =
57       {table | canonicalNextCoordinate hk n mean t table = target} := by
58     ext table
59     exact rebuilt_mem_conditionBranch_iff hk n mean t target v table
60   rw [he]
61   apply Measure.map_congr
62   filter_upwards [ae_restrict_mem ((measurable_canonicalNextCoordinate hk n mean t)
63     (measurableSet_singleton target))] with table hnext
64   exact (canonicalCondition_eq_without hk n mean t target v table hnext).symm

66 theorem map_condition_reward_restrict_branch {k : ℕ} (hk : 0 < k) (n : ℕ)
67     (mean : Fin k → ℝ) (t : ℕ) (ν : Kernel (Fin k) ℝ) [IsMarkovKernel ν]
68     (target : ℕ × Fin k) (v : ℝ) :
69     Measure.map (fun table => (canonicalCondition hk n mean t table, canonicalReward hk n mean table (t+1)))
70       ((UCB.armStreamMeasure ν).restrict {table | canonicalNextCoordinate hk n mean t table = target}) =
71     ((Measure.map (canonicalCondition hk n mean t) (UCB.armStreamMeasure ν)).restrict
72       (conditionBranch t target)).prod (ν target.2) := by
73   let R := fun table => canonicalConditionWithout hk n mean t target v (UCB.armStreamWithoutCoordinate target table)
74   let P := fun table => (R table, UCB.armStreamCoordinate target table)
75   let B := {table | canonicalNextCoordinate hk n mean t table = target}
76   have hB : MeasurableSet B := (measurable_canonicalNextCoordinate hk n mean t) (measurableSet_singleton target)
77   have hC := measurableSet_conditionBranch t target
78   have hR : Measurable R := (measurable_canonicalConditionWithout hk n mean t target v).comp
79     (UCB.measurable_armStreamWithoutCoordinate target)
80   have hP : Measurable P := hR.prodMk (UCB.measurable_armStreamCoordinate target)
81   have he : P ⁻¹' (conditionBranch t target ×ˢ Set.univ) = B := by
82     ext table
83     change (R table ∈ conditionBranch t target ∧ UCB.armStreamCoordinate target table ∈ Set.univ) ↔ _
84     simp only [Set.mem_univ, and_true]
85     exact rebuilt_mem_conditionBranch_iff hk n mean t target v table
86   calc
87     _ = Measure.map P ((UCB.armStreamMeasure ν).restrict B) := by
88       apply Measure.map_congr
89       filter_upwards [ae_restrict_mem hB] with table hnext
90       apply Prod.ext
91       · exact canonicalCondition_eq_without hk n mean t target v table hnext
92       · rw [canonicalReward_succ_eq_coordinate, hnext]
93     _ = (Measure.map P (UCB.armStreamMeasure ν)).restrict (conditionBranch t target ×ˢ Set.univ) := by
94       rw [Measure.restrict_map hP (hC.prod MeasurableSet.univ), he]
95     _ = ((Measure.map R (UCB.armStreamMeasure ν)).prod (ν target.2)).restrict
96         (conditionBranch t target ×ˢ Set.univ) := by
97       rw [map_canonicalConditionWithout_coordinate hk n mean t ν target v]
98     _ = ((Measure.map R (UCB.armStreamMeasure ν)).restrict (conditionBranch t target)).prod (ν target.2) := by
99       rw [Measure.restrict_prod_eq_prod_univ]
100     _ = _ := by rw [map_rebuilt_restrict_conditionBranch hk n mean t target v]
