6 # ETC centered reward-difference sub-Gaussian tail producer
7 
8 This module specializes the abstract ETC pairwise sub-Gaussian producer to the
9 compiled centered reward-difference finite-sum event.  It keeps the actual
10 probabilistic reward-law work explicit: independence and sub-Gaussian witnesses
11 for the concrete summands remain hypotheses.
12 -/

15 namespace ETC

17 /--
18 Build the fixed-commit ETC pairwise empirical-mean tail contract from concrete
19 centered pairwise reward-difference sub-Gaussian witnesses over the ETC
20 exploration horizon.
21 
22 This is the `ETC-PAIRWISE-TAIL-PRODUCER-CENTERED-DIFF` bridge.  It instantiates
23 the abstract producer with
24 `idx := Finset.range (spec.explorationPulls * K)`,
25 `X := ETC.centeredPairwiseRewardDiff`, and
26 `eps := ETC.centeredPairwiseGapThreshold`.  It still does not prove
27 independence, sub-Gaussianity of reward differences, filtration, or final ETC
28 regret.
29 -/
30 theorem pairwiseEmpMeanTailContract_of_centered_subGaussian_event_bounds
31     {Omega : Type u} {K : Nat}
32     [MeasurableSpace Omega]
33     (mu : MeasureTheory.Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
34     (spec : ETC.Spec K)
35     (model : FiniteBanditModel K)
36     (commitArm : Fin K)
37     (reward : Omega -> RewardTrace Rat)
38     (tail : Fin K -> ENNReal)
39     (c : Fin K -> Nat -> NNReal)
40     (hexplorationPulls_pos : 0 < spec.explorationPulls)
41     (h_indep :
42       forall a : Fin K, (a = model.bestArm -> False) ->
43         ProbabilityTheory.iIndepFun
44           (fun t omega =>
45             ETC.centeredPairwiseRewardDiff
46               spec model commitArm reward a t omega)
47           mu)
48     (h_subG :
49       forall a : Fin K, (a = model.bestArm -> False) ->
50         forall t, t ∈ Finset.range (spec.explorationPulls * K) ->
51           ProbabilityTheory.HasSubgaussianMGF
52             (fun omega =>
53               ETC.centeredPairwiseRewardDiff
54                 spec model commitArm reward a t omega)
55             (c a t) mu)
56     (htail :
57       forall a : Fin K, (a = model.bestArm -> False) ->
58         ENNReal.ofReal
59           (Real.exp
60             (-(ETC.centeredPairwiseGapThreshold spec model a) ^ 2 /
61               (2 *
62                 (((Finset.range (spec.explorationPulls * K)).sum
63                   (fun t => c a t) : NNReal) : Real)))) <= tail a) :
64     ETC.PairwiseEmpMeanTailContract
65       mu spec model commitArm reward tail := by
66   refine
67     ETC.pairwiseEmpMeanTailContract_of_subGaussian_event_bounds
68       (mu := mu)
69       (spec := spec)
70       (model := model)
71       (commitArm := commitArm)
72       (reward := reward)
73       (tail := tail)
74       (idx := Finset.range (spec.explorationPulls * K))
75       (X := fun a t omega =>
76         ETC.centeredPairwiseRewardDiff
77           spec model commitArm reward a t omega)
78       (c := c)
79       (eps := fun a => ETC.centeredPairwiseGapThreshold spec model a)
80       ?h_indep ?h_subG ?heps ?hsubset ?htail
81   · exact h_indep
82   · exact h_subG
83   · intro a _hne
84     unfold ETC.centeredPairwiseGapThreshold
85     exact
86       (Rat.cast_nonneg (K := Real)).2
87         (mul_nonneg
88           (Nat.cast_nonneg spec.explorationPulls)
89           (sub_nonneg.mpr
90             (FiniteBanditModel.mean_le_bestArm_mean model a)))
91   · intro a _hne
92     exact
93       ETC.empMeanAtExploration_ge_best_event_subset_centered_pairwise_sum_event
94         spec model commitArm reward a hexplorationPulls_pos
95   · exact htail
