5 # ETC bounded reward sub-Gaussian source
6 
7 This module connects Mathlib's bounded-variable Hoeffding lemma to the ETC
8 reward-coordinate wrong-commit route. It still leaves the stochastic source of
9 reward-coordinate independence, boundedness, and mean identities explicit.
10 -/

15 /--
16 Sub-Gaussian variance proxy induced by an almost-sure interval bound on one
17 raw reward coordinate.
18 -/
19 noncomputable def centeredRewardBoundVarianceProxy
20     {K : Nat}
21     (lo hi : Fin K -> Nat -> Real)
22     (a : Fin K) (t : Nat) : NNReal :=
23   ((nnnorm (hi a t - lo a t) / 2) ^ 2)

25 /--
26 An a.e. bounded raw reward coordinate is integrable.

40       (fun omega : Omega => (((reward omega t : Rat) : Real))) mu)

49 /--
50 An exact raw-reward mean identity gives zero integral for the centered reward.
51 
52 This is the zero-integral source needed by the conditional mean-zero wrapper.
53 It still keeps raw reward integrability explicit; boundedness or a concrete
54 reward law can discharge that assumption in later leaves.
55 -/
56 theorem centeredReward_integral_eq_zero_of_integral_eq_mean
57     {Omega : Type u} {K : Nat}
58     [MeasurableSpace Omega]
59     (mu : MeasureTheory.Measure Omega) [MeasureTheory.IsProbabilityMeasure mu]
60     (model : FiniteBanditModel K)
61     (reward : Omega -> RewardTrace Rat)
62     (a : Fin K) (t : Nat)
63     (hint : MeasureTheory.Integrable
64       (fun omega : Omega => (((reward omega t : Rat) : Real))) mu)
65     (hmean :
66       MeasureTheory.integral mu
67         (fun omega : Omega => (((reward omega t : Rat) : Real))) =
68       (((model.mean a : Rat) : Real))) :
69     MeasureTheory.integral mu
70       (fun omega : Omega =>
71         (((reward omega t - model.mean a : Rat) : Real))) = 0 := by
72   have hconst :
73       MeasureTheory.Integrable
74         (fun _omega : Omega => (((model.mean a : Rat) : Real))) mu :=
75     MeasureTheory.integrable_const _
76   calc
77     MeasureTheory.integral mu
78         (fun omega : Omega =>
79           (((reward omega t - model.mean a : Rat) : Real))) =
80       MeasureTheory.integral mu
81         (fun omega : Omega =>
82           (((reward omega t : Rat) : Real) -
83             (((model.mean a : Rat) : Real)))) := by
84         congr 1
85         funext omega
86         simp [Rat.cast_sub]
87     _ =
88       MeasureTheory.integral mu
89         (fun omega : Omega => (((reward omega t : Rat) : Real))) -
90       MeasureTheory.integral mu
91         (fun _omega : Omega => (((model.mean a : Rat) : Real))) := by
92         rw [MeasureTheory.integral_sub hint hconst]
93     _ = 0 := by
94         simp [hmean]
