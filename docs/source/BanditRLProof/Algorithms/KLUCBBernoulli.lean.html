8 # Bernoulli relative entropy and KL confidence indices
9 
10 This file owns the project-local binary relative entropy used by KL-UCB.  The
11 codomain is `ENNReal`: singular Bernoulli comparisons are genuinely `top`, not
12 the accidental finite value obtained from Mathlib's totalized `Real.log 0`.

22 /-- The closed unit interval predicate used by every KL-UCB parameter. -/
23 def IsBernoulliParameter (p : Real) : Prop := p ∈ Set.Icc (0 : Real) 1
24 
25 local instance (p : Real) : Decidable (IsBernoulliParameter p) :=
26   Classical.propDecidable _
27 
28 local instance : DecidableEq Real := Classical.decEq _
29 
30 /-- The finite analytic Bernoulli relative-entropy expression.  It is used
31 only when the second parameter is strictly between zero and one. -/
32 noncomputable def bernoulliKLCore (p q : Real) : Real :=
33   p * Real.log (p / q) +
34     (1 - p) * Real.log ((1 - p) / (1 - q))
35 
36 /-- Bernoulli relative entropy with exact support/endpoint conventions.
37 
38 * parameters outside `[0,1]` have value `top`;
39 * `d(0,0)=d(1,1)=0`;
40 * `d(p,0)=top` for `p>0` and `d(p,1)=top` for `p<1`;
41 * otherwise the ordinary finite logarithmic expression is used.
42 -/
43 noncomputable def bernoulliKL (p q : Real) : ENNReal :=
44   if hp : IsBernoulliParameter p then
45     if hq : IsBernoulliParameter q then
46       if hq0 : q = 0 then
47         if hp0 : p = 0 then 0 else ⊤
48       else if hq1 : q = 1 then
49         if hp1 : p = 1 then 0 else ⊤
50       else
51         ENNReal.ofReal (bernoulliKLCore p q)
52     else ⊤
53   else ⊤
54 
55 theorem bernoulliKL_nonneg (p q : Real) :
56     0 <= bernoulliKL p q := bot_le

134 /-- The analytic binary KL expression is the two-atom `klFun` integral. -/
135 theorem bernoulliKLCore_eq_klFun
136     {p q : Real} (hq0 : q ≠ 0) (hq1 : q ≠ 1) :
137     bernoulliKLCore p q =
138       q * InformationTheory.klFun (p / q) +
139         (1 - q) * InformationTheory.klFun ((1 - p) / (1 - q)) := by
140   rw [InformationTheory.klFun_apply, InformationTheory.klFun_apply]
141   unfold bernoulliKLCore
142   have h1q : 1 - q ≠ 0 := sub_ne_zero.mpr (Ne.symm hq1)
143   field_simp
144   ring

146 /-- Nontrivial nonnegativity of the finite logarithmic expression.  This is
147 stronger than the order-theoretic nonnegativity of its `ENNReal` wrapper. -/
148 theorem bernoulliKLCore_nonneg
149     {p q : Real} (hp : IsBernoulliParameter p)
150     (hq0 : 0 < q) (hq1 : q < 1) :
151     0 <= bernoulliKLCore p q := by
152   rw [bernoulliKLCore_eq_klFun (ne_of_gt hq0) (ne_of_lt hq1)]
153   have hpq : 0 <= p / q := div_nonneg hp.1 hq0.le
154   have hcomp : 0 <= (1 - p) / (1 - q) := by
155     exact div_nonneg (sub_nonneg.mpr hp.2) (sub_nonneg.mpr hq1.le)
156   exact add_nonneg
157     (mul_nonneg hq0.le (InformationTheory.klFun_nonneg hpq))
158     (mul_nonneg (sub_nonneg.mpr hq1.le)
159       (InformationTheory.klFun_nonneg hcomp))

172 /-- Algebraically expanded finite KL, convenient for differentiation in the
173 second parameter. -/
174 noncomputable def bernoulliKLExpanded (p q : Real) : Real :=
175   p * Real.log p - p * Real.log q +
176     (1 - p) * Real.log (1 - p) -
177       (1 - p) * Real.log (1 - q)
178 
179 theorem bernoulliKLCore_eq_expanded
180     {p q : Real} (hp0 : p ≠ 0) (hp1 : p ≠ 1)
181     (hq0 : q ≠ 0) (hq1 : q ≠ 1) :
182     bernoulliKLCore p q = bernoulliKLExpanded p q := by
183   unfold bernoulliKLCore bernoulliKLExpanded
184   rw [Real.log_div hp0 hq0,
185     Real.log_div (sub_ne_zero.mpr (Ne.symm hp1))
186       (sub_ne_zero.mpr (Ne.symm hq1))]
187   ring

189 theorem hasDerivAt_bernoulliKLExpanded_right
190     (p q : Real) (hq0 : q ≠ 0) (hq1 : q ≠ 1) :
191     HasDerivAt (fun r => bernoulliKLExpanded p r)
192       ((q - p) / (q * (1 - q))) q := by
193   unfold bernoulliKLExpanded
194   have hlogq := Real.hasDerivAt_log hq0
195   have h1q : 1 - q ≠ 0 := sub_ne_zero.mpr (Ne.symm hq1)
196   have honeSub : HasDerivAt (fun r : Real => 1 - r) (-1) q := by
197     convert (hasDerivAt_const q (1 : Real)).sub (hasDerivAt_id q) using 1 <;> ring
198   have hlogOneSub := honeSub.log (sub_ne_zero.mpr (Ne.symm hq1))
199   convert
200     (((hasDerivAt_const q (p * Real.log p)).sub
201         (hlogq.const_mul p)).add
202       (hasDerivAt_const q ((1 - p) * Real.log (1 - p)))).sub
203         (hlogOneSub.const_mul (1 - p)) using 1 <;>
204       field_simp [hq0, h1q] <;> ring

206 /-- A conservative binary Pinsker inequality.  The constant `1/2` is weaker
207 than the sharp natural-log constant `2`, but is sufficient to invert every
208 KL-UCB confidence set without changing the KL score itself. -/
209 theorem half_sq_sub_le_bernoulliKLCore
210     {p q : Real} (hp : IsBernoulliParameter p)
211     (hq0 : 0 < q) (hq1 : q < 1) :
212     (1 / 2 : Real) * (p - q) ^ 2 <= bernoulliKLCore p q := by
213   by_cases hp0 : p = 0
214   · subst p
215     simp only [zero_sub, neg_sq]
216     have hlog := Real.log_le_sub_one_of_pos (sub_pos.mpr hq1)
217     have hqle : (1 / 2 : Real) * q ^ 2 <= q := by
218       nlinarith [hq0.le, hq1.le]
219     simpa [bernoulliKLCore] using hqle.trans (by linarith : q <= -Real.log (1 - q))
220   by_cases hp1 : p = 1
221   · subst p
222     have hlog := Real.log_le_sub_one_of_pos hq0
223     have hcomp : 0 <= 1 - q := sub_nonneg.mpr hq1.le
224     have hcomple : 1 - q <= 1 := by linarith
225     have hsq : (1 / 2 : Real) * (1 - q) ^ 2 <= 1 - q := by
226       nlinarith
227     simpa [bernoulliKLCore] using hsq.trans (by linarith : 1 - q <= -Real.log q)
228   have hp0' : 0 < p := lt_of_le_of_ne hp.1 (Ne.symm hp0)
229   have hp1' : p < 1 := lt_of_le_of_ne hp.2 hp1
230   let f : Real -> Real := fun r =>
231     bernoulliKLExpanded p r - (1 / 2 : Real) * (p - r) ^ 2
232   have hfderiv (r : Real) (hr0 : 0 < r) (hr1 : r < 1) :
233       HasDerivAt f
234         ((r - p) * (1 / (r * (1 - r)) - 1)) r := by
235     have hkl := hasDerivAt_bernoulliKLExpanded_right p r
236       (ne_of_gt hr0) (ne_of_lt hr1)
237     have hdiff : HasDerivAt (fun x : Real => p - x) (-1) r := by
238       convert (hasDerivAt_const r p).sub (hasDerivAt_id r) using 1 <;> ring
239     have hsq : HasDerivAt
240         (fun x : Real => (1 / 2 : Real) * (p - x) ^ 2) (r - p) r := by
241       convert (hdiff.pow 2).const_mul (1 / 2 : Real) using 1 <;> ring
242     convert hkl.sub hsq using 1
243     field_simp [ne_of_gt hr0, sub_ne_zero.mpr (Ne.symm (ne_of_lt hr1))]
244   have hcoefficient (r : Real) (hr0 : 0 < r) (hr1 : r < 1) :
245       0 <= 1 / (r * (1 - r)) - 1 := by
246     have hprod : 0 < r * (1 - r) := mul_pos hr0 (sub_pos.mpr hr1)
247     have hprod_le : r * (1 - r) <= 1 := by nlinarith [sq_nonneg r]
248     have hone : (1 : Real) <= 1 / (r * (1 - r)) := by
249       exact (le_div_iff₀ hprod).2 (by simpa using hprod_le)
250     linarith
251   have hfp : f p = 0 := by
252     simp [f, bernoulliKLExpanded]
253   have hresult : f p <= f q := by
254     rcases le_total p q with hpq | hqp
255     · let D : Set Real := Set.Icc p q
256       have hinside (r : Real) (hr : r ∈ D) : 0 < r ∧ r < 1 :=
257         ⟨hp0'.trans_le hr.1, hr.2.trans_lt hq1⟩
258       have hcontinuous : ContinuousOn f D := by
259         intro r hr
260         exact (hfderiv r (hinside r hr).1 (hinside r hr).2).continuousAt.continuousWithinAt
261       have hmono : MonotoneOn f D :=
262         monotoneOn_of_deriv_nonneg (convex_Icc p q) hcontinuous
263           (fun r hr => (hfderiv r
264             (hinside r (interior_subset hr)).1
265             (hinside r (interior_subset hr)).2).differentiableAt.differentiableWithinAt)
266           (fun r hr => by
267             rw [(hfderiv r
268               (hinside r (interior_subset hr)).1
269               (hinside r (interior_subset hr)).2).deriv]
270             exact mul_nonneg (sub_nonneg.mpr (interior_subset hr).1)
271               (hcoefficient r
272                 (hinside r (interior_subset hr)).1
273                 (hinside r (interior_subset hr)).2))
274       exact hmono ⟨le_rfl, hpq⟩ ⟨hpq, le_rfl⟩ hpq
275     · let D : Set Real := Set.Icc q p
276       have hinside (r : Real) (hr : r ∈ D) : 0 < r ∧ r < 1 :=
277         ⟨hq0.trans_le hr.1, hr.2.trans_lt hp1'⟩
278       have hcontinuous : ContinuousOn f D := by
279         intro r hr
280         exact (hfderiv r (hinside r hr).1 (hinside r hr).2).continuousAt.continuousWithinAt
281       have hanti : AntitoneOn f D :=
282         antitoneOn_of_deriv_nonpos (convex_Icc q p) hcontinuous
283           (fun r hr => (hfderiv r
284             (hinside r (interior_subset hr)).1
285             (hinside r (interior_subset hr)).2).differentiableAt.differentiableWithinAt)
286           (fun r hr => by
287             rw [(hfderiv r
288               (hinside r (interior_subset hr)).1
289               (hinside r (interior_subset hr)).2).deriv]
290             exact mul_nonpos_of_nonpos_of_nonneg
291               (sub_nonpos.mpr (interior_subset hr).2)
292               (hcoefficient r
293                 (hinside r (interior_subset hr)).1
294                 (hinside r (interior_subset hr)).2))
295       exact hanti ⟨le_rfl, hqp⟩ ⟨hqp, le_rfl⟩ hqp
296   rw [hfp] at hresult
297   have hcore := bernoulliKLCore_eq_expanded hp0 hp1
298     (ne_of_gt hq0) (ne_of_lt hq1)
299   simpa [f, hcore] using hresult

301 /-- On an interior reference mean, binary KL is controlled by the squared
302 deviation divided by the Bernoulli variance denominator.  This is the bridge
303 from the repository's bounded-reward empirical-mean tails to a genuine KL
304 confidence event. -/
305 theorem bernoulliKLCore_le_sq_div
306     {p q : Real} (hp : IsBernoulliParameter p)
307     (hq0 : 0 < q) (hq1 : q < 1) :
308     bernoulliKLCore p q <= (p - q) ^ 2 / (q * (1 - q)) := by
309   have hqcomp : 0 < 1 - q := sub_pos.mpr hq1
310   by_cases hp0 : p = 0
311   · subst p
312     have hlog := Real.one_sub_inv_le_log_of_pos hqcomp
313     have hcore : bernoulliKLCore 0 q = -Real.log (1 - q) := by
314       simp [bernoulliKLCore]
315     have hden : q * (1 - q) ≠ 0 := mul_ne_zero (ne_of_gt hq0) (ne_of_gt hqcomp)
316     rw [hcore]
317     calc
318       -Real.log (1 - q) <= (1 - q)⁻¹ - 1 := by linarith
319       _ = (0 - q) ^ 2 / (q * (1 - q)) := by
320         field_simp [hden, ne_of_gt hqcomp]
321         ring
322   by_cases hp1 : p = 1
323   · subst p
324     have hlog := Real.one_sub_inv_le_log_of_pos hq0
325     have hcore : bernoulliKLCore 1 q = -Real.log q := by
326       simp [bernoulliKLCore]
327     have hden : q * (1 - q) ≠ 0 := mul_ne_zero (ne_of_gt hq0) (ne_of_gt hqcomp)
328     rw [hcore]
329     calc
330       -Real.log q <= q⁻¹ - 1 := by linarith
331       _ = (1 - q) ^ 2 / (q * (1 - q)) := by
332         field_simp [hden, ne_of_gt hq0, ne_of_gt hqcomp]
333   · have hp0' : 0 < p := lt_of_le_of_ne hp.1 (Ne.symm hp0)
334     have hp1' : p < 1 := lt_of_le_of_ne hp.2 hp1
335     have hfirst := Real.log_le_sub_one_of_pos (div_pos hp0' hq0)
336     have hsecond := Real.log_le_sub_one_of_pos
337       (div_pos (sub_pos.mpr hp1') hqcomp)
338     have hfirstMul :
339         p * Real.log (p / q) <= p * (p / q - 1) :=
340       mul_le_mul_of_nonneg_left hfirst hp.1
341     have hsecondMul :
342         (1 - p) * Real.log ((1 - p) / (1 - q)) <=
343           (1 - p) * ((1 - p) / (1 - q) - 1) :=
344       mul_le_mul_of_nonneg_left hsecond (sub_nonneg.mpr hp.2)
345     unfold bernoulliKLCore
346     calc
347       p * Real.log (p / q) +
348           (1 - p) * Real.log ((1 - p) / (1 - q)) <=
349         p * (p / q - 1) +
350           (1 - p) * ((1 - p) / (1 - q) - 1) :=
351             add_le_add hfirstMul hsecondMul
352       _ = (p - q) ^ 2 / (q * (1 - q)) := by
353         field_simp [ne_of_gt hq0, ne_of_gt hqcomp]
354         ring

386 leaves; endpoint singularities remain in `ENNReal`. -/
387 theorem continuousAt_bernoulliKLCore_right
388     (p q : Real) (hq0 : q ≠ 0) (hq1 : q ≠ 1) :

406 /-- KL confidence set at one empirical mean, pull count, and exploration
407 budget. -/
408 def confidenceSet (empiricalMean : Real) (count : Nat) (budget : Real) :
409     Set Real :=
410   {q | IsBernoulliParameter q ∧
411     (count : ENNReal) * bernoulliKL empiricalMean q <=
412       ENNReal.ofReal budget}

414 /-- The KL-UCB index is the supremum of its confidence set. -/
415 noncomputable def index
416     (empiricalMean : Real) (count : Nat) (budget : Real) : Real :=
417   sSup (confidenceSet empiricalMean count budget)
