3 /-!
4 # Native Real ETC LML field compatibility surface
5 
6 The pinned LML source currently uses a newer Lean/mathlib toolchain, so ABRL
7 cannot import its `IsAlgEnvSeq` declaration directly. This module packages the
8 exact measurable-action, measurable-feedback, action-behavior, and stationary
9 feedback-law consequences consumed by the local ETC theorem. It is a local
10 compatibility structure, not an imported LML proof.
11 -/

18 /--
19 The exact consequences of a stationary Real ETC algorithm-environment sequence
20 used by the local regret route.
21 
22 The fields correspond to the pinned source's `IsAlgEnvSeq` measurability and
23 feedback fields together with `ETC.arm_of_lt`, `ETC.arm_mul`, and
24 `ETC.arm_of_ge`. Conditional laws are stated as `condDistrib` equalities, which
25 is the Mathlib-facing form consumed by ABRL.
26 -/
27 structure RealStationaryETCSequence
28     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
29     (mu : Measure Omega) [IsFiniteMeasure mu] (spec : ETC.Spec K)
30     (nu : ProbabilityTheory.Kernel (Fin K) Real)
31     [ProbabilityTheory.IsMarkovKernel nu]
32     (action : Omega -> ActionTrace (Fin K))
33     (reward : Omega -> RewardTrace Real) : Prop where
34   measurable_action : forall t, Measurable (fun omega => action omega t)
35   measurable_reward : forall t, Measurable (fun omega => reward omega t)
36   arm_of_lt : forall t, t < K * spec.explorationPulls ->
37     Filter.EventuallyEq (ae mu)
38       (fun omega => action omega t)
39       (fun _omega => ETC.exploreArm spec t)
40   arm_mul : Filter.EventuallyEq (ae mu)
41     (fun omega => action omega (K * spec.explorationPulls))
42     (fun omega => ETC.realLeastEncodedArgmax spec.hK
43       (fun arm => ETC.realHistoryEmpMean
44         (K * spec.explorationPulls - 1)
45         (History.finitePairHistoryOfTrace (action omega) (reward omega)
46           (K * spec.explorationPulls - 1)) arm))
47   arm_of_ge : forall t, K * spec.explorationPulls <= t ->
48     Filter.EventuallyEq (ae mu)
49       (fun omega => action omega t)
50       (fun omega => action omega (K * spec.explorationPulls))
51   hasCondDistrib_feedback_zero : ProbabilityTheory.condDistrib
52       (fun omega => reward omega 0)
53       (fun omega => action omega 0) mu =ᵐ[mu.map (fun omega => action omega 0)]
54     ProbabilityTheory.Kernel.ofFunOfCountable (fun arm : Fin K => nu arm)
55   hasCondDistrib_feedback : forall i,
56     let fullCondition := fun omega : Omega =>
57       (History.finitePairHistoryOfTrace
58           (action omega) (reward omega) i,
59         action omega (i + 1))
60     ProbabilityTheory.condDistrib
61         (fun omega => reward omega (i + 1))
62         fullCondition mu =ᵐ[mu.map fullCondition]
63       (RewardKernel.contextIndependentOfActionLaws
64         (Context := History.FinitePairHistory (Fin K) Real i)
65         (fun arm : Fin K => nu arm)
66         (fun _arm => inferInstance)).kernel

68 /--
69 Exact native Real ETC regret from the bundled stationary sequence fields.
70 
71 This is the local theorem corresponding to the mathematical statement of the
72 pinned LML `Bandits.ETC.regret_le`. A direct theorem about the imported LML
73 `IsAlgEnvSeq` symbol still requires a common Lean/mathlib toolchain.
74 -/
75 theorem regret_le_of_realStationaryETCSequence
76     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
77     (mu : Measure Omega) [IsProbabilityMeasure mu]
78     (spec : ETC.Spec K)
79     (nu : ProbabilityTheory.Kernel (Fin K) Real)
80     [ProbabilityTheory.IsMarkovKernel nu]
81     (action : Omega -> ActionTrace (Fin K))
82     (reward : Omega -> RewardTrace Real)
83     (h : ETC.RealStationaryETCSequence mu spec nu action reward)
84     (sigma2 : NNReal)
85     (hsubG : forall arm, ProbabilityTheory.HasSubgaussianMGF
86       (fun x => x - realKernelMean nu arm) sigma2 (nu arm))
87     (hm : 0 < spec.explorationPulls) (n : Nat)
88     (hn : K * spec.explorationPulls <= n) :
89     integral mu (fun omega => realKernelRegret nu (action omega) n) <=
90       (Finset.univ : Finset (Fin K)).sum (fun arm =>
91         realKernelGap nu arm *
92           ((spec.explorationPulls : Real) +
93             ((n - K * spec.explorationPulls : Nat) : Real) *
94               Real.exp
95                 (-(spec.explorationPulls : Real) *
96                   (realKernelGap nu arm) ^ 2 /
97                     (4 * (sigma2 : Real))))) := by
98   exact
99     ETC.integral_realKernelRegret_externalAction_le_exact_sum_of_actionDependent_actionRewardHistory_condDistrib_of_historyLeastEncodedCommit_persist
100       mu spec nu sigma2 hsubG hm n hn action reward
101         h.measurable_action h.measurable_reward h.arm_of_lt h.arm_mul
102         h.arm_of_ge h.hasCondDistrib_feedback_zero
103         (fun i _hi => h.hasCondDistrib_feedback i)
