7 def canonicalAction {k : ℕ} (hk : 0 < k) (n : ℕ) (mean : Fin k → ℝ) :=
8   streamTrace hk n mean (centeredRewardTable mean)

10 /-- Consume the next unused one-based raw reward of the chosen arm. -/
11 def canonicalReward {k : ℕ} (hk : 0 < k) (n : ℕ) (mean : Fin k → ℝ) :=
12   UCB.rewardFromArmStream (canonicalAction hk n mean)
13     (fun table : UCB.ArmRewardStream k => fun i a => table (i+1) a)

16     (table : UCB.ArmRewardStream k) (t : ℕ) :=
17   History.finitePairHistoryOfTrace (canonicalAction hk n mean table)

26 theorem canonicalHistory_empiricalMean {k : ℕ} (hk : 0 < k) (n : ℕ) (mean : Fin k → ℝ)
27     (table : UCB.ArmRewardStream k) (t : ℕ) (a : Fin k) :
28     ETC.realHistoryEmpMean t (canonicalHistory hk n mean table t) a =
29       (∑ j ∈ Finset.range (pullCount (canonicalAction hk n mean table) a (t+1)), table (j+1) a) /
30         (pullCount (canonicalAction hk n mean table) a (t+1) : ℝ) := by
31   rw [canonicalHistory, ETC.realHistoryEmpMean_finitePairHistoryOfTrace]
32   rw [canonicalReward, UCB.sumRewards_rewardFromArmStream_eq_armPrefixSum]
33   rfl

35 /-- The realized next action is exactly the common finite-history MOSS selector. -/
36 theorem canonicalAction_succ_eq_historyAction {k : ℕ} (hk : 0 < k) (n : ℕ)
37     (mean : Fin k → ℝ) (table : UCB.ArmRewardStream k) (t : ℕ) :
38     canonicalAction hk n mean table (t+1) =
39       historyAction hk n t (canonicalHistory hk n mean table t) := by
40   unfold historyAction
41   have hc : ETC.realHistoryPullCount t (canonicalHistory hk n mean table t) =
42       fun a => pullCount (canonicalAction hk n mean table) a (t+1) :=
43     funext (canonicalHistory_pullCount hk n mean table t)
44   have he : ETC.realHistoryEmpMean t (canonicalHistory hk n mean table t) =
45       fun a => (∑ j ∈ Finset.range (pullCount (canonicalAction hk n mean table) a (t+1)), table (j+1) a) /
46         (pullCount (canonicalAction hk n mean table) a (t+1) : ℝ) :=
47     funext (canonicalHistory_empiricalMean hk n mean table t)
48   rw [hc, he]
49   exact canonicalReward_action_eq_raw hk n (t+1) mean table

81 theorem canonicalHistory_succ {k : ℕ} (hk : 0 < k) (n : ℕ)
82     (mean : Fin k → ℝ) (table : UCB.ArmRewardStream k) (t : ℕ) :
83     canonicalHistory hk n mean table (t+1) = History.extendPairHistorySucc
84       (canonicalHistory hk n mean table t)
85       (historyAction hk n t (canonicalHistory hk n mean table t),
86         table (ETC.realHistoryPullCount t (canonicalHistory hk n mean table t)
87           (historyAction hk n t (canonicalHistory hk n mean table t))+1)
88           (historyAction hk n t (canonicalHistory hk n mean table t))) := by
89   conv_lhs => unfold canonicalHistory
90   rw [History.finitePairHistoryOfTrace_succ]
91   change History.extendPairHistorySucc (canonicalHistory hk n mean table t) _ = _
92   congr 1
93   rw [← canonicalAction_succ_eq_historyAction]
94   rw [canonicalHistory_pullCount]
95   rfl

97 /-- Changing rewards that have not been consumed cannot change the observed history. -/
98 theorem canonicalHistory_eq_of_eq_consumed {k : ℕ} (hk : 0 < k) (n : ℕ)
99     (mean : Fin k → ℝ) (table table' : UCB.ArmRewardStream k) (t : ℕ)
100     (hagrees : ∀ a j, j < pullCount (canonicalAction hk n mean table) a (t+1) →
101       table (j+1) a = table' (j+1) a) :
102     canonicalHistory hk n mean table t = canonicalHistory hk n mean table' t := by
103   induction t with
104   | zero =>
105       have he := hagrees ⟨0, hk⟩ 0 (by simp [pullCount_succ, canonicalAction_zero])
106       funext i
107       have hi : i.val = 0 := by have := Finset.mem_Iic.mp i.property; omega
108       simp [canonicalHistory, History.finitePairHistoryOfTrace, hi,
109         canonicalReward, UCB.rewardFromArmStream, canonicalAction_zero, he]
110   | succ t ih =>
111       have hh := ih (fun a j hj => hagrees a j (lt_of_lt_of_le hj
112         (pullCount_mono (canonicalAction hk n mean table) a (Nat.le_succ _))))
113       rw [canonicalHistory_succ, canonicalHistory_succ, ← hh]
114       congr 2
115       apply hagrees
116       rw [canonicalHistory_pullCount]
117       have hs := canonicalAction_succ_eq_historyAction hk n mean table t
118       rw [pullCount_succ_of_eq _ _ _ hs]
119       exact Nat.lt_succ_self _
