4 /-!
5 # Thompson sampling from a reference posterior policy
6 
7 LML defines Thompson sampling from the posterior under a fixed reference
8 algorithm and then transports that posterior to the actual process by an
9 algorithm-density theorem.  This module isolates the corresponding local
10 Mathlib boundary.
11 
12 The next action is genuinely sampled by a composition-product measure, so no
13 action conditional-law premise remains.  The only process-level premise in the
14 final theorem is posterior invariance between the reference and actual history
15 laws.
16 -/

29     [MeasurableSpace OmegaRef] [MeasurableSpace History]

89 /-- Extend a base process by sampling an action from a history-indexed policy. -/
90 noncomputable def policySamplerMeasure
91     {Omega : Type u} {History : Type v} {Action : Type w}
92     [MeasurableSpace Omega] [MeasurableSpace History] [MeasurableSpace Action]
93     (mu : Measure Omega) [IsFiniteMeasure mu]
94     (history : Omega -> History) (hhistory : Measurable history)
95     (policy : ProbabilityTheory.Kernel History Action)
96     [ProbabilityTheory.IsMarkovKernel policy] :
97     Measure (Omega × Action) :=
98   mu ⊗ₘ policy.comap history hhistory

147 /--
148 History/action projection of a policy sampler.
149 
150 This is the arbitrary-history-map version of `map_compProd_comap_snd` and is a
151 generic Mathlib candidate.
152 -/
153 theorem map_compProd_comap_history
154     {Omega History Action : Type*}
155     [MeasurableSpace Omega] [MeasurableSpace History] [MeasurableSpace Action]
156     (mu : Measure Omega) [IsFiniteMeasure mu]
157     (history : Omega -> History) (hhistory : Measurable history)
158     (policy : ProbabilityTheory.Kernel History Action)
159     [ProbabilityTheory.IsMarkovKernel policy] :
160     (mu ⊗ₘ policy.comap history hhistory).map
161         (fun sample => (history sample.1, sample.2)) =
162       mu.map history ⊗ₘ policy := by
163   apply Measure.ext_prod
164   intro historyEvent actionEvent hhistoryEvent hactionEvent
165   have hmap : Measurable
166       (fun sample : Omega × Action => (history sample.1, sample.2)) :=
167     (hhistory.comp measurable_fst).prodMk measurable_snd
168   rw [Measure.map_apply hmap (hhistoryEvent.prod hactionEvent)]
169   have hpreimage :
170       (fun sample : Omega × Action => (history sample.1, sample.2)) ⁻¹'
171           (historyEvent ×ˢ actionEvent) =
172         (history ⁻¹' historyEvent) ×ˢ actionEvent := by
173     ext sample
174     simp
175   rw [hpreimage,
176     Measure.compProd_apply_prod (hhistory hhistoryEvent) hactionEvent,
177     Measure.compProd_apply_prod hhistoryEvent hactionEvent]
178   simp_rw [ProbabilityTheory.Kernel.comap_apply]
179   rw [MeasureTheory.setLIntegral_map hhistoryEvent
180     (policy.measurable_coe hactionEvent) hhistory]

182 /-- Adjoining a Markov-policy sample preserves every measurable base map. -/
183 theorem policySampler_base_map_eq
184     {Omega History Action Target : Type*}
185     [MeasurableSpace Omega] [MeasurableSpace History]
186     [MeasurableSpace Action] [MeasurableSpace Target]
187     (mu : Measure Omega) [IsFiniteMeasure mu]
188     (history : Omega -> History) (hhistory : Measurable history)
189     (policy : ProbabilityTheory.Kernel History Action)
190     [ProbabilityTheory.IsMarkovKernel policy]
191     (f : Omega -> Target) (hf : Measurable f) :
192     (policySamplerMeasure mu history hhistory policy).map
193         (fun sample => f sample.1) =
194       mu.map f := by
195   calc
196     (policySamplerMeasure mu history hhistory policy).map
197         (fun sample => f sample.1) =
198         ((policySamplerMeasure mu history hhistory policy).map Prod.fst).map f := by
199           rw [Measure.map_map hf measurable_fst]
200           rfl
201     _ = mu.map f := by
202       congr 1
203       change (mu ⊗ₘ policy.comap history hhistory).fst = mu
204       rw [Measure.fst_compProd]

206 /-- The constructed sampler's history/action joint law is `historyLaw ⊗ policy`. -/
207 theorem policySampler_history_action_map_eq
208     {Omega History Action : Type*}
209     [MeasurableSpace Omega] [MeasurableSpace History] [MeasurableSpace Action]
210     (mu : Measure Omega) [IsFiniteMeasure mu]
211     (history : Omega -> History) (hhistory : Measurable history)
212     (policy : ProbabilityTheory.Kernel History Action)
213     [ProbabilityTheory.IsMarkovKernel policy] :
214     (policySamplerMeasure mu history hhistory policy).map
215         (fun sample =>
216           (policySamplerHistory history sample, policySamplerAction sample)) =
217       (policySamplerMeasure mu history hhistory policy).map
218           (policySamplerHistory history) ⊗ₘ policy := by
219   have hhistoryLaw :
220       (policySamplerMeasure mu history hhistory policy).map
221           (policySamplerHistory history) = mu.map history := by
222     simpa only [policySamplerHistory] using
223       policySampler_base_map_eq mu history hhistory policy history hhistory
224   rw [hhistoryLaw]
225   simpa only [policySamplerMeasure, policySamplerHistory, policySamplerAction] using
226     map_compProd_comap_history mu history hhistory policy

228 /-- The sampled action has the policy as its conditional law given history. -/
229 theorem policySampler_condDistrib_action_ae_eq_policy
230     {Omega History Action : Type*}
231     [MeasurableSpace Omega] [MeasurableSpace History]
232     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
233     (mu : Measure Omega) [IsFiniteMeasure mu]
234     (history : Omega -> History) (hhistory : Measurable history)
235     (policy : ProbabilityTheory.Kernel History Action)
236     [ProbabilityTheory.IsMarkovKernel policy] :
237     ProbabilityTheory.condDistrib policySamplerAction
238         (policySamplerHistory history)
239         (policySamplerMeasure mu history hhistory policy) =ᵐ[
240       (policySamplerMeasure mu history hhistory policy).map
241         (policySamplerHistory history)] policy := by
242   exact (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
243     (μ := policySamplerMeasure mu history hhistory policy)
244     (policySamplerHistory history) policySamplerAction_measurable.aemeasurable
245     policy).2 (policySampler_history_action_map_eq
246       mu history hhistory policy)

248 /--
249 Adjoining a history-dependent action sample preserves an environment posterior.
250 -/
251 theorem policySampler_condDistrib_env_ae_eq_of_base
252     {Omega History Env Action : Type*}
253     [MeasurableSpace Omega] [MeasurableSpace History]
254     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
255     [MeasurableSpace Action]
256     (mu : Measure Omega) [IsFiniteMeasure mu]
257     (env : Omega -> Env) (history : Omega -> History)
258     (henv : Measurable env) (hhistory : Measurable history)
259     (policy : ProbabilityTheory.Kernel History Action)
260     [ProbabilityTheory.IsMarkovKernel policy]
261     (posterior : ProbabilityTheory.Kernel History Env)
262     [ProbabilityTheory.IsMarkovKernel posterior]
263     (hbase :
264       ProbabilityTheory.condDistrib env history mu =ᵐ[mu.map history]
265         posterior) :
266     ProbabilityTheory.condDistrib (policySamplerEnv env)
267         (policySamplerHistory history)
268         (policySamplerMeasure mu history hhistory policy) =ᵐ[
269       (policySamplerMeasure mu history hhistory policy).map
270         (policySamplerHistory history)] posterior := by
271   have hbaseJoint :
272       mu.map (fun omega => (history omega, env omega)) =
273         mu.map history ⊗ₘ posterior :=
274     (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
275       (μ := mu) history henv.aemeasurable posterior).1 hbase
276   have hhistoryLaw :
277       (policySamplerMeasure mu history hhistory policy).map
278           (policySamplerHistory history) = mu.map history := by
279     simpa only [policySamplerHistory] using
280       policySampler_base_map_eq mu history hhistory policy history hhistory
281   have hjointLaw :
282       (policySamplerMeasure mu history hhistory policy).map
283           (fun sample =>
284             (policySamplerHistory history sample, policySamplerEnv env sample)) =
285         (policySamplerMeasure mu history hhistory policy).map
286             (policySamplerHistory history) ⊗ₘ posterior := by
287     calc
288       _ = mu.map (fun omega => (history omega, env omega)) := by
289         simpa only [policySamplerHistory, policySamplerEnv] using
290           policySampler_base_map_eq mu history hhistory policy
291             (fun omega => (history omega, env omega)) (hhistory.prodMk henv)
292       _ = mu.map history ⊗ₘ posterior := hbaseJoint
293       _ = _ := by rw [hhistoryLaw]
294   exact (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
295     (μ := policySamplerMeasure mu history hhistory policy)
296     (policySamplerHistory history)
297     (policySamplerEnv_measurable env henv).aemeasurable posterior).2 hjointLaw

299 /--
300 Reference-posterior Thompson probability matching with a constructed sampler.
301 
302 The action law is generated by `policySamplerMeasure`.  The sole law transport
303 premise is that the reference posterior agrees with the actual environment
304 posterior at the actual history law; this is the conclusion supplied by LML's
305 algorithm-density route.
306 -/
307 theorem referencePolicySampler_condDistrib_action_ae_eq_bestAction_of_posterior_invariance
308     {Omega : Type u} {OmegaRef : Type v}
309     {History : Type w} {Env : Type x} {Action : Type y}
310     [MeasurableSpace Omega] [MeasurableSpace OmegaRef]
311     [MeasurableSpace History]
312     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
313     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
314     (mu : Measure Omega) [IsFiniteMeasure mu]
315     (env : Omega -> Env) (history : Omega -> History)
316     (henv : Measurable env) (hhistory : Measurable history)
317     (referenceMu : Measure OmegaRef) [IsFiniteMeasure referenceMu]
318     (referenceEnv : OmegaRef -> Env)
319     (referenceHistory : OmegaRef -> History)
320     (hreferenceEnv : Measurable referenceEnv)
321     (hreferenceHistory : Measurable referenceHistory)
322     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
323     (hposteriorInvariance :
324       (referencePosterior referenceMu referenceEnv referenceHistory
325         hreferenceEnv hreferenceHistory).kernel =ᵐ[mu.map history]
326         ProbabilityTheory.condDistrib env history mu) :
327     let policy := referenceActionKernel referenceMu referenceEnv referenceHistory
328       hreferenceEnv hreferenceHistory bestAction hbestAction
329     let sampler := policySamplerMeasure mu history hhistory policy
330     ProbabilityTheory.condDistrib policySamplerAction
331         (policySamplerHistory history) sampler =ᵐ[
332       sampler.map (policySamplerHistory history)]
333       ProbabilityTheory.condDistrib (bestAction ∘ policySamplerEnv env)
334         (policySamplerHistory history) sampler := by
335   dsimp only
336   let posterior := referencePosterior referenceMu referenceEnv referenceHistory
337     hreferenceEnv hreferenceHistory
338   let policy := referenceActionKernel referenceMu referenceEnv referenceHistory
339     hreferenceEnv hreferenceHistory bestAction hbestAction
340   let sampler := policySamplerMeasure mu history hhistory policy
341   have haction :
342       ProbabilityTheory.condDistrib policySamplerAction
343           (policySamplerHistory history) sampler =ᵐ[
344         sampler.map (policySamplerHistory history)]
345         posterior.kernel.map bestAction := by
346     simpa only [sampler, policy, posterior, referenceActionKernel] using
347       policySampler_condDistrib_action_ae_eq_policy
348         mu history hhistory policy
349   have hposterior :
350       posterior.kernel =ᵐ[sampler.map (policySamplerHistory history)]
351         ProbabilityTheory.condDistrib (policySamplerEnv env)
352           (policySamplerHistory history) sampler := by
353     have hbase :
354         ProbabilityTheory.condDistrib env history mu =ᵐ[mu.map history]
355           posterior.kernel := by
356       simpa only [posterior] using hposteriorInvariance.symm
357     exact (policySampler_condDistrib_env_ae_eq_of_base
358       mu env history henv hhistory policy posterior.kernel hbase).symm
359   exact condDistrib_action_ae_eq_bestAction_of_posteriorMap
360     sampler (policySamplerEnv env) (policySamplerHistory history)
361     policySamplerAction posterior bestAction hbestAction
362     (policySamplerEnv_measurable env henv)
363     (policySamplerHistory_measurable history hhistory)
364     policySamplerAction_measurable haction hposterior

366 /--
367 Finite action/reward-prefix specialization of the reference-policy theorem.
368 
369 This is the per-time interface used by the recursive bandit route.  The
370 remaining premise is exactly posterior invariance between the reference and
371 actual finite-pair history laws.
372 -/
373 theorem finitePairReferencePolicySampler_condDistrib_action_ae_eq_bestAction_of_posterior_invariance
374     {Omega : Type u} {OmegaRef : Type v}
375     {Env : Type w} {Action : Type x} {Reward : Type y}
376     [MeasurableSpace Omega] [MeasurableSpace OmegaRef]
377     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
378     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
379     [MeasurableSpace Reward]
380     (mu : Measure Omega) [IsFiniteMeasure mu]
381     (env : Omega -> Env)
382     (action : Omega -> ActionTrace Action)
383     (reward : Omega -> RewardTrace Reward)
384     (henv : Measurable env)
385     (haction : forall t : Nat, Measurable (fun omega => action omega t))
386     (hreward : forall t : Nat, Measurable (fun omega => reward omega t))
387     (referenceMu : Measure OmegaRef) [IsFiniteMeasure referenceMu]
388     (referenceEnv : OmegaRef -> Env)
389     (referenceAction : OmegaRef -> ActionTrace Action)
390     (referenceReward : OmegaRef -> RewardTrace Reward)
391     (hreferenceEnv : Measurable referenceEnv)
392     (hreferenceAction : forall t : Nat,
393       Measurable (fun omega => referenceAction omega t))
394     (hreferenceReward : forall t : Nat,
395       Measurable (fun omega => referenceReward omega t))
396     (n : Nat)
397     (bestAction : Env -> Action) (hbestAction : Measurable bestAction)
398     (hposteriorInvariance :
399       ProbabilityTheory.condDistrib referenceEnv
400           (fun omega => History.finitePairHistoryOfTrace
401             (referenceAction omega) (referenceReward omega) n)
402           referenceMu =ᵐ[
403         mu.map (fun omega => History.finitePairHistoryOfTrace
404           (action omega) (reward omega) n)]
405         ProbabilityTheory.condDistrib env
406           (fun omega => History.finitePairHistoryOfTrace
407             (action omega) (reward omega) n) mu) :
408     let actualHistory := fun omega => History.finitePairHistoryOfTrace
409       (action omega) (reward omega) n
410     let referenceHistory := fun omega => History.finitePairHistoryOfTrace
411       (referenceAction omega) (referenceReward omega) n
412     let policy := referenceActionKernel referenceMu referenceEnv referenceHistory
413       hreferenceEnv
414       (History.measurable_finitePairHistoryOfTrace
415         referenceAction referenceReward hreferenceAction hreferenceReward n)
416       bestAction hbestAction
417     let sampler := policySamplerMeasure mu actualHistory
418       (History.measurable_finitePairHistoryOfTrace
419         action reward haction hreward n) policy
420     ProbabilityTheory.condDistrib policySamplerAction
421         (policySamplerHistory actualHistory) sampler =ᵐ[
422       sampler.map (policySamplerHistory actualHistory)]
423       ProbabilityTheory.condDistrib (bestAction ∘ policySamplerEnv env)
424         (policySamplerHistory actualHistory) sampler := by
425   dsimp only
426   let actualHistory := fun omega => History.finitePairHistoryOfTrace
427     (action omega) (reward omega) n
428   let referenceHistory := fun omega => History.finitePairHistoryOfTrace
429     (referenceAction omega) (referenceReward omega) n
430   have hactualHistory : Measurable actualHistory :=
431     History.measurable_finitePairHistoryOfTrace
432       action reward haction hreward n
433   have hreferenceHistory : Measurable referenceHistory :=
434     History.measurable_finitePairHistoryOfTrace
435       referenceAction referenceReward hreferenceAction hreferenceReward n
436   have hinvariance :
437       (referencePosterior referenceMu referenceEnv referenceHistory
438         hreferenceEnv hreferenceHistory).kernel =ᵐ[mu.map actualHistory]
439         ProbabilityTheory.condDistrib env actualHistory mu := by
440     simpa only [referencePosterior_kernel, actualHistory, referenceHistory] using
441       hposteriorInvariance
442   have hresult :=
443     referencePolicySampler_condDistrib_action_ae_eq_bestAction_of_posterior_invariance
444       mu env actualHistory henv hactualHistory
445       referenceMu referenceEnv referenceHistory hreferenceEnv hreferenceHistory
446       bestAction hbestAction hinvariance
447   simpa only [actualHistory, referenceHistory] using hresult
