3 /-!
4 # Two-arm SGB Theorem 2: missing-pull-aware selected-block transport
5 
6 Appendix C of Baudry--Johnson--Vary--Pike-Burke--Rebeschini works with rewards
7 in optimal-arm pull order.  Adaptive selection makes a naive stopped-value
8 IID statement false as a proof interface: a requested pull can be absent, and
9 the event that a block of pulls occurs can itself depend on earlier rewards.
10 
11 This module therefore transports the exact observable block that the native
12 process exposes.  Every coordinate records both the `WithTop Nat` nth-pull
13 time and Mathlib's stopped reward.  On the latent coupling, the reward is
14 replaced by the corresponding arm-stream coordinate only when the pull time
15 is finite; at `top` the original stopped-value fallback is retained.  The
16 result is an exact finite-block law on the native stationary process while
17 keeping the missing-pull boundary explicit.
18 
19 The unmasked latent prefix still has the separately compiled product law
20 `twoArmFixedIIDLatentTrajectoryMeasure_map_optimalPrefix_eq_pi`.  The masked
21 block below is deliberately not claimed to have that product law, either
22 unconditionally after totalization or after conditioning on all pulls being
23 present.
24 -/

66 /-- The latent comparison block.  A finite pull reads its corresponding
67 arm-`0` stream coordinate.  A missing pull retains the stopped-value fallback
68 rather than silently turning an absent observation into an IID reward. -/
69 def twoArmLatentMaskedOptimalPullBlock (m : Nat) :
70     UCB.ArmRewardStream 2 × ((t : Nat) -> Fin 2 × Real) ->
71       ((i : Fin m) -> WithTop Nat × Real) :=
72   fun sample i =>
73     let trajectorySample := ((), sample.2)
74     let pullTime :=
75       twoArmNthOptimalPullTime (Env := Unit) (i : Nat) trajectorySample
76     let stoppedReward :=
77       twoArmNthOptimalPullReward (Env := Unit) (i : Nat) trajectorySample
78     (pullTime,
79       if pullTime = (⊤ : WithTop Nat) then
80         stoppedReward
81       else
82         sample.1 (i : Nat) 0)

378 /-- The running reward sum inside Appendix C's recovery phase.
379 
380 The index `k : Fin (n1 + 1)` permits every prefix length from `0` through
381 `n1`.  The ambient reward block contains the unlucky phase of length `n0`
382 followed by the recovery phase of length `n1`. -/
383 def twoArmAppendixCPhaseOnePrefixSum (n0 n1 : Nat)
384     (rewardBlock : Fin (n0 + n1) -> Real) (k : Fin (n1 + 1)) : Real :=
385   ∑ i : Fin (k : Nat),
386     rewardBlock ⟨n0 + (i : Nat), by omega⟩

392   refine Finset.measurable_sum Finset.univ fun i _ => ?_
393   let j : Fin (n0 + n1) := ⟨n0 + (i : Nat), by omega⟩
394   exact measurable_pi_apply j

396 /-- The exact finite reward event used by the two phases in Appendix C.
397 
398 Phase `S0` consists of `n0` rewards equal to `-1`.  Phase `S1` consists only
399 of `{-1, 1}` rewards, has the specified exact terminal sum, and has running
400 sum at most zero at every prefix.  The later arithmetic layer will instantiate
401 `phaseOneTotal` with the rounded Rademacher count selected by the source; this
402 definition itself contains no probability or IID premise. -/
403 def twoArmAppendixCRewardPhaseEvent (n0 n1 : Nat)
404     (phaseOneTotal : Real) : Set (Fin (n0 + n1) -> Real) :=
405   {rewardBlock |
406     (forall i : Fin n0,
407       rewardBlock (Fin.castAdd n1 i) = -1) /\
408     (forall i : Fin n1,
409       rewardBlock (Fin.natAdd n0 i) = -1 \/
410         rewardBlock (Fin.natAdd n0 i) = 1) /\
411     twoArmAppendixCPhaseOnePrefixSum n0 n1 rewardBlock (Fin.last n1) =
412       phaseOneTotal /\
413     (forall k : Fin (n1 + 1),
414       twoArmAppendixCPhaseOnePrefixSum n0 n1 rewardBlock k <= 0)}
415

418     MeasurableSet

422         forall i : Fin n0,
423           rewardBlock (Fin.castAdd n1 i) = -1} := by
424     rw [Set.setOf_forall]
425     exact MeasurableSet.iInter fun i =>
426       measurableSet_eq_fun (measurable_pi_apply (Fin.castAdd n1 i))
427         measurable_const
428   have hphaseOneSupport : MeasurableSet
429       {rewardBlock : Fin (n0 + n1) -> Real |
430         forall i : Fin n1,
431           rewardBlock (Fin.natAdd n0 i) = -1 \/
432             rewardBlock (Fin.natAdd n0 i) = 1} := by
433     rw [Set.setOf_forall]
434     refine MeasurableSet.iInter fun i => ?_
435     simpa only [Set.setOf_or] using
436       (measurableSet_eq_fun (measurable_pi_apply (Fin.natAdd n0 i))
437         measurable_const).union
438       (measurableSet_eq_fun (measurable_pi_apply (Fin.natAdd n0 i))
439         measurable_const)
440   have hphaseOneTotal : MeasurableSet
441       {rewardBlock : Fin (n0 + n1) -> Real |
442         twoArmAppendixCPhaseOnePrefixSum n0 n1 rewardBlock (Fin.last n1) =
443           phaseOneTotal} :=
444     measurableSet_eq_fun
445       (measurable_twoArmAppendixCPhaseOnePrefixSum n0 n1 (Fin.last n1))
446       measurable_const
447   have hballot : MeasurableSet
448       {rewardBlock : Fin (n0 + n1) -> Real |
449         forall k : Fin (n1 + 1),
450           twoArmAppendixCPhaseOnePrefixSum n0 n1 rewardBlock k <= 0} := by
451     rw [Set.setOf_forall]

453       measurableSet_le
454         (measurable_twoArmAppendixCPhaseOnePrefixSum n0 n1 k)
455         measurable_const
456   simpa only [twoArmAppendixCRewardPhaseEvent, Set.setOf_and] using
457     hphaseZero.inter
458       (hphaseOneSupport.inter (hphaseOneTotal.inter hballot))
