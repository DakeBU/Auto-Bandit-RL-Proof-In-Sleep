4 # Native Real ETC finite-history score bridge
5 
6 This module mirrors the finite-pair-history `pullCount'`, `sumRewards'`, and
7 `empMean'` score surface used by the pinned LML ETC source. It identifies that
8 history score with the existing native Real exploration score, then feeds the
9 history-shaped commit law into the exact source adapter.
10 -/

17 /-- Number of occurrences of an arm in an inclusive finite pair history. -/
18 noncomputable def realHistoryPullCount {K : Nat} (n : Nat)
19     (history : History.FinitePairHistory (Fin K) Real n)
20     (arm : Fin K) : Nat :=
21   Finset.univ.sum (fun i => if (history i).1 = arm then 1 else 0)
22 
23 /-- Sum of rewards of an arm in an inclusive finite pair history. -/
24 noncomputable def realHistorySumRewards {K : Nat} (n : Nat)
25     (history : History.FinitePairHistory (Fin K) Real n)
26     (arm : Fin K) : Real :=
27   Finset.univ.sum (fun i => if (history i).1 = arm then (history i).2 else 0)

29 /-- Empirical mean of an arm in an inclusive finite pair history. -/
30 noncomputable def realHistoryEmpMean {K : Nat} (n : Nat)
31     (history : History.FinitePairHistory (Fin K) Real n)
32     (arm : Fin K) : Real :=
33   ETC.realHistorySumRewards n history arm /
34     (ETC.realHistoryPullCount n history arm : Real)

36 /-- Inclusive history pull counts are exclusive trace pull counts at `n + 1`. -/
37 theorem realHistoryPullCount_finitePairHistoryOfTrace
38     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
39     (n : Nat) (arm : Fin K) :
40     ETC.realHistoryPullCount n
41         (History.finitePairHistoryOfTrace action reward n) arm =
42       pullCount action arm (n + 1) := by
43   rw [pullCount_eq_finset_filter_card]
44   simp only [ETC.realHistoryPullCount,
45     History.finitePairHistoryOfTrace_apply]
46   calc
47     (∑ i : Finset.Iic n, if action i.1 = arm then 1 else 0) =
48         (Finset.Iic n).sum
49           (fun i : Nat => if action i = arm then 1 else 0) :=
50       Finset.sum_coe_sort (Finset.Iic n)
51         (fun i : Nat => if action i = arm then 1 else 0)
52     _ = (Finset.range (n + 1)).sum
53           (fun i : Nat => if action i = arm then 1 else 0) := by
54       congr 1
55       ext i
56       simp [Finset.mem_Iic, Finset.mem_range]
57     _ = ((Finset.range (n + 1)).filter
58           (fun i : Nat => action i = arm)).card := by
59       classical
60       induction Finset.range (n + 1) using Finset.induction_on with
61       | empty => simp
62       | @insert i s hi ih =>
63           by_cases hia : action i = arm
64           · simp_all
65           · simp_all

67 /-- Inclusive history reward sums are exclusive trace sums at `n + 1`. -/
68 theorem realHistorySumRewards_finitePairHistoryOfTrace
69     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
70     (n : Nat) (arm : Fin K) :
71     ETC.realHistorySumRewards n
72         (History.finitePairHistoryOfTrace action reward n) arm =
73       sumRewards action reward arm (n + 1) := by
74   simp only [ETC.realHistorySumRewards,
75     History.finitePairHistoryOfTrace_apply]
76   rw [Finset.sum_coe_sort (s := Finset.Iic n)
77     (f := fun i : Nat => if action i = arm then reward i else 0)]
78   rw [sumRewards_eq_finset_filter_sum]
79   rw [Finset.sum_filter]
80   apply Finset.sum_congr
81   · ext i
82     simp [Finset.mem_Iic, Finset.mem_range]
83   · intro i hi
84     rfl

86 /-- The source-shaped history mean is the trace empirical mean at `n + 1`. -/
87 theorem realHistoryEmpMean_finitePairHistoryOfTrace
88     {K : Nat} (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
89     (n : Nat) (arm : Fin K) :
90     ETC.realHistoryEmpMean n
91         (History.finitePairHistoryOfTrace action reward n) arm =
92       sumRewards action reward arm (n + 1) /
93         (pullCount action arm (n + 1) : Real) := by
94   rw [ETC.realHistoryEmpMean,
95     ETC.realHistorySumRewards_finitePairHistoryOfTrace,
96     ETC.realHistoryPullCount_finitePairHistoryOfTrace]

98 private theorem pullCount_eq_of_eq_on_lt
99     {Action : Type} [DecidableEq Action]
100     (action action' : ActionTrace Action) (arm : Action) (n : Nat)
101     (haction : forall t, t < n -> action t = action' t) :
102     pullCount action arm n = pullCount action' arm n := by
103   rw [pullCount_eq_finset_filter_card, pullCount_eq_finset_filter_card]
104   congr 1
105   ext t
106   simp only [Finset.mem_filter, Finset.mem_range]
107   by_cases ht : t < n
108   · simp [ht, haction t ht]
109   · simp [ht]

111 private theorem sumRewards_eq_of_action_eq_on_lt
112     {Action Reward : Type} [DecidableEq Action] [AddCommMonoid Reward]
113     (action action' : ActionTrace Action) (reward : RewardTrace Reward)
114     (arm : Action) (n : Nat)
115     (haction : forall t, t < n -> action t = action' t) :
116     sumRewards action reward arm n = sumRewards action' reward arm n := by
117   rw [sumRewards_eq_finset_filter_sum, sumRewards_eq_finset_filter_sum]
118   apply Finset.sum_congr
119   · ext t
120     simp only [Finset.mem_filter, Finset.mem_range]
121     by_cases ht : t < n
122     · simp [ht, haction t ht]
123     · simp [ht]
124   · intro t ht

127 /--
128 At the exploration boundary, the pinned-source finite-history score equals the
129 native Real exploration score whenever the observed actions are round robin.
130 -/
131 theorem realHistoryEmpMean_exploration_eq_realEmpMeanAtExploration
132     {K : Nat} (spec : ETC.Spec K) (baseCommitArm : Fin K)
133     (action : ActionTrace (Fin K)) (reward : RewardTrace Real)
134     (hm : 0 < spec.explorationPulls)
135     (hactionExplore : forall t, t < K * spec.explorationPulls ->
136       action t = ETC.exploreArm spec t)
137     (arm : Fin K) :
138     ETC.realHistoryEmpMean (K * spec.explorationPulls - 1)
139         (History.finitePairHistoryOfTrace action reward
140           (K * spec.explorationPulls - 1)) arm =
141       ETC.realEmpMeanAtExploration spec baseCommitArm reward arm := by
142   have hpos : 0 < K * spec.explorationPulls :=
143     Nat.mul_pos spec.hK hm
144   have hsucc : K * spec.explorationPulls - 1 + 1 =
145       K * spec.explorationPulls := by omega
146   rw [ETC.realHistoryEmpMean_finitePairHistoryOfTrace]
147   rw [hsucc]
148   rw [ETC.realEmpMeanAtExploration]
149   have hboundary : K * spec.explorationPulls =
150       spec.explorationPulls * K := by simp [Nat.mul_comm]
151   have hactionFixed : forall t, t < K * spec.explorationPulls ->
152       action t = ETC.actionWithCommit spec baseCommitArm t := by
153     intro t ht
154     rw [ETC.actionWithCommit_eq_exploreArm_of_lt spec baseCommitArm
155       (by simpa [Nat.mul_comm] using ht)]
156     exact hactionExplore t ht
157   rw [hboundary]
158   rw [show (pullCount action arm (spec.explorationPulls * K) : Real) =
159       (pullCount (ETC.actionWithCommit spec baseCommitArm) arm
160         (spec.explorationPulls * K) : Real) by
161     norm_num
162     exact ETC.pullCount_eq_of_eq_on_lt
163       action (ETC.actionWithCommit spec baseCommitArm) arm
164         (spec.explorationPulls * K)
165         (by
166           intro t ht
167           exact hactionFixed t (by simpa [Nat.mul_comm] using ht))]
168   congr 1
169   · exact ETC.sumRewards_eq_of_action_eq_on_lt
170       action (ETC.actionWithCommit spec baseCommitArm) reward arm
171         (spec.explorationPulls * K)
172         (by
173           intro t ht
174           exact hactionFixed t (by simpa [Nat.mul_comm] using ht))

176 /--
177 Exact native Real ETC regret from a source-shaped finite-history commit score.
178 The history score is rewritten locally; callers no longer provide a commit law
179 already phrased with `realEmpMeanAtExploration`.
180 -/
181 theorem integral_realKernelRegret_externalAction_le_exact_sum_of_actionDependent_actionRewardHistory_condDistrib_of_historyLeastEncodedCommit_persist
182     {Omega : Type u} {K : Nat} [MeasurableSpace Omega]
183     (mu : Measure Omega) [IsProbabilityMeasure mu]
184     (spec : ETC.Spec K)
185     (nu : ProbabilityTheory.Kernel (Fin K) Real)
186     [ProbabilityTheory.IsMarkovKernel nu]
187     (sigma2 : NNReal)
188     (hsubG : forall arm, ProbabilityTheory.HasSubgaussianMGF
189       (fun reward => reward - realKernelMean nu arm) sigma2 (nu arm))
190     (hm : 0 < spec.explorationPulls) (n : Nat)
191     (hn : K * spec.explorationPulls <= n)
192     (action : Omega -> ActionTrace (Fin K))
193     (reward : Omega -> RewardTrace Real)
194     (haction : forall t, Measurable (fun omega => action omega t))
195     (hreward : forall t, Measurable (fun omega => reward omega t))
196     (hactionExplore : forall t, t < K * spec.explorationPulls ->
197       Filter.EventuallyEq (ae mu)
198         (fun omega => action omega t)
199         (fun _omega => ETC.exploreArm spec t))
200     (hactionCommit : Filter.EventuallyEq (ae mu)
201       (fun omega => action omega (K * spec.explorationPulls))
202       (fun omega => ETC.realLeastEncodedArgmax spec.hK
203         (fun arm => ETC.realHistoryEmpMean
204           (K * spec.explorationPulls - 1)
205           (History.finitePairHistoryOfTrace (action omega) (reward omega)
206             (K * spec.explorationPulls - 1)) arm)))
207     (hactionPersist : forall t, K * spec.explorationPulls <= t ->
208       Filter.EventuallyEq (ae mu)
209         (fun omega => action omega t)
210         (fun omega => action omega (K * spec.explorationPulls)))
211     (hzero : ProbabilityTheory.condDistrib
212         (fun omega => reward omega 0)
213         (fun omega => action omega 0) mu =ᵐ[mu.map (fun omega => action omega 0)]
214       ProbabilityTheory.Kernel.ofFunOfCountable (fun arm : Fin K => nu arm))
215     (hcond : forall i, i < spec.explorationPulls * K - 1 ->
216       let fullCondition := fun omega : Omega =>
217         (History.finitePairHistoryOfTrace
218             (action omega) (reward omega) i,
219           action omega (i + 1))
220       ProbabilityTheory.condDistrib
221           (fun omega => reward omega (i + 1))
222           fullCondition mu =ᵐ[mu.map fullCondition]
223         (RewardKernel.contextIndependentOfActionLaws
224           (Context := History.FinitePairHistory (Fin K) Real i)
225           (fun arm : Fin K => nu arm)
226           (fun _arm => inferInstance)).kernel) :
227     integral mu (fun omega => realKernelRegret nu (action omega) n) <=
228       (Finset.univ : Finset (Fin K)).sum (fun arm =>
229         realKernelGap nu arm *
230           ((spec.explorationPulls : Real) +
231             ((n - K * spec.explorationPulls : Nat) : Real) *
232               Real.exp
233                 (-(spec.explorationPulls : Real) *
234                   (realKernelGap nu arm) ^ 2 /
235                     (4 * (sigma2 : Real))))) := by
236   have hExploreAll : Filter.Eventually
237       (fun omega => forall t, t < K * spec.explorationPulls ->
238         action omega t = ETC.exploreArm spec t) (ae mu) := by
239     rw [ae_all_iff]
240     intro t
241     by_cases ht : t < K * spec.explorationPulls
242     · filter_upwards [hactionExplore t ht] with omega homega
243       intro _ht
244       exact homega
245     · filter_upwards [] with omega
246       intro hfalse
247       exact (ht hfalse).elim
248   have hactionCommitLocal : Filter.EventuallyEq (ae mu)
249       (fun omega => action omega (K * spec.explorationPulls))
250       (fun omega => ETC.realLeastEncodedArgmax spec.hK
251         (fun arm => ETC.realEmpMeanAtExploration spec
252           (ETC.realKernelBestArm spec.hK nu) (reward omega) arm)) := by
253     filter_upwards [hactionCommit, hExploreAll] with omega hcommit hexplore
254     rw [hcommit]
255     congr 1
256     funext arm
257     exact ETC.realHistoryEmpMean_exploration_eq_realEmpMeanAtExploration
258       spec (ETC.realKernelBestArm spec.hK nu) (action omega) (reward omega)
259         hm hexplore arm
260   exact
