5 # Predictable EXP3 regret with variance-sensitive mixed-square confidence
6 
7 This module consumes the generated mixed-square Bernstein tail in the existing
8 three-event predictable-regret assembly.  The square-event radius now uses the
9 second-moment coefficient `K / epsilon`; the pure-cross and fixed-comparator
10 events retain their compiled Bernstein radii.
11 -/

19 /-- Predictable regret budget using the variance-sensitive mixed-square
20 Bernstein radius. -/
21 noncomputable def sampledPredictableBernsteinSquareHighProbabilityRegretBudget
22     {Action : Type v} (arms : Finset Action) (eta gamma : Real)
23     (horizon : Nat) (deltaSquare deltaConfidence : Real) : Real :=
24   Real.log arms.card / eta +
25     (eta * (1 / (1 - gamma))) *
26       ((arms.card : Real) * (horizon : Real) +
27         sampledMixedSquaredBernsteinConfidenceRadius
28           arms gamma horizon deltaSquare) +
29     gamma * (horizon : Real) +
30     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
31       arms gamma horizon deltaConfidence +
32     sampledComparatorEstimatorBernsteinConfidenceRadius
33       arms gamma horizon deltaConfidence

35 /-- Generated predictable EXP3 regret with the mixed-square, pure-cross, and
36 fixed-comparator Bernstein events. -/
37 theorem sampledPredictable_bernsteinSquareHighProbabilityRegret_tail
38     {Env : Type u} {Action : Type v}
39     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
40     [MeasurableSpace Action] [MeasurableSingletonClass Action]
41     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
42     (prior : Measure Env) [IsProbabilityMeasure prior]
43     (arms : Finset Action) (harms : arms.Nonempty)
44     (eta gamma : Real) (heta : 0 < eta)
45     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
46     (loss : PredictableLossVector Env Action)
47     (comparator : Action) (hcomparator : comparator ∈ arms)
48     (horizon : Nat)
49     (deltaSquare deltaConfidence : Real)
50     (hdeltaSquare : 0 < deltaSquare)
51     (hdeltaConfidence : 0 < deltaConfidence) :
52     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
53       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
54     mu {sample |
55         sampledPredictableBernsteinSquareHighProbabilityRegretBudget
56             arms eta gamma horizon deltaSquare deltaConfidence <=
57           (Finset.range horizon).sum (fun t =>
58               sampledTrajectoryExploredPredictableLossAt
59                 arms eta gamma loss t sample) -
60             (Finset.range horizon).sum (fun t =>
61               predictableLossAt loss t sample comparator)} <=
62       (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
63         ENNReal.ofReal deltaConfidence := by
64   dsimp only
65   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
66     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
67   let purePredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
68     (Finset.range horizon).sum (fun t =>
69       sampledTrajectoryPurePredictableLossAt arms eta gamma loss t sample)
70   let pureObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
71     (Finset.range horizon).sum (fun t =>
72       sampledTrajectoryPureObservedLossAt arms eta gamma t sample)
73   let comparatorObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
74     (Finset.range horizon).sum (fun t =>
75       observedImportanceWeightedLossAt arms eta gamma t sample comparator)
76   let comparatorTrue := fun sample : Env × ((k : Nat) -> Action × Real) =>
77     (Finset.range horizon).sum (fun t =>
78       predictableLossAt loss t sample comparator)
79   let exploredPredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
80     (Finset.range horizon).sum (fun t =>
81       sampledTrajectoryExploredPredictableLossAt arms eta gamma loss t sample)
82   let secondMoment := sampledObservedMixedSquaredSum
83     (Env := Env) arms eta gamma horizon
84   let secondMomentBudget :=
85     (arms.card : Real) * (horizon : Real) +
86       sampledMixedSquaredBernsteinConfidenceRadius
87         arms gamma horizon deltaSquare
88   let pureRadius :=
89     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
90       arms gamma horizon deltaConfidence
91   let comparatorRadius := sampledComparatorEstimatorBernsteinConfidenceRadius
92     arms gamma horizon deltaConfidence
93   let secondMomentBad : Set (Env × ((k : Nat) -> Action × Real)) :=
94     {sample | secondMomentBudget <= secondMoment sample}
95   let pureBad : Set (Env × ((k : Nat) -> Action × Real)) :=
96     {sample | pureRadius <= purePredictable sample - pureObserved sample}
97   let comparatorBad : Set (Env × ((k : Nat) -> Action × Real)) :=
98     {sample | comparatorRadius <= comparatorObserved sample - comparatorTrue sample}
99   have hsecondTail : mu secondMomentBad <= ENNReal.ofReal deltaSquare := by
100     have h := sampledPredictableObservedMixedSquared_sum_tail_bernstein_delta
101       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss horizon
102         deltaSquare hdeltaSquare
103     dsimp only at h
104     simpa [mu, secondMomentBad, secondMomentBudget, secondMoment] using h
105   have hpureTail : mu pureBad <= ENNReal.ofReal deltaConfidence := by
106     have h := sampledPurePredictableMinusObserved_sum_tail_bernstein_delta
107       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss
108         horizon deltaConfidence hdeltaConfidence
109     dsimp only at h
110     simpa [mu, pureBad, pureRadius, purePredictable, pureObserved] using h
111   have hcomparatorTail : mu comparatorBad <= ENNReal.ofReal deltaConfidence := by
112     have h := sampledObservedComparatorEstimatorDeviation_sum_tail_bernstein_delta
113       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss comparator
114         hcomparator horizon deltaConfidence hdeltaConfidence
115     dsimp only at h
116     simpa [mu, comparatorBad, comparatorRadius, comparatorObserved,
117       comparatorTrue,
118       sampledTrajectoryObservedComparatorEstimatorDeviationAt] using h
119   have hhedge :=
120     sampledPredictableTrajectoryMeasure_hedge_exploredSecondMoment_le_ae
121       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss horizon
122         comparator hcomparator
123   dsimp only at hhedge
124   have hsubset : ∀ᵐ sample ∂mu,
125       sampledPredictableBernsteinSquareHighProbabilityRegretBudget
126           arms eta gamma horizon deltaSquare deltaConfidence <=
127         exploredPredictable sample - comparatorTrue sample ->
128       sample ∈ (secondMomentBad ∪ pureBad) ∪ comparatorBad := by
129     filter_upwards [hhedge] with sample hsampleHedge
130     intro hregret
131     by_cases hsecond : sample ∈ secondMomentBad
132     · exact Or.inl (Or.inl hsecond)
133     by_cases hpure : sample ∈ pureBad
134     · exact Or.inl (Or.inr hpure)
135     by_cases hcomparator : sample ∈ comparatorBad
136     · exact Or.inr hcomparator
137     exfalso
138     have hsecondGood : secondMoment sample < secondMomentBudget := by
139       change ¬ secondMomentBudget <= secondMoment sample at hsecond
140       exact lt_of_not_ge hsecond
141     have hpureGood :
142         purePredictable sample - pureObserved sample < pureRadius := by
143       change ¬ pureRadius <= purePredictable sample - pureObserved sample at hpure
144       exact lt_of_not_ge hpure
145     have hcomparatorGood :
146         comparatorObserved sample - comparatorTrue sample <
147           comparatorRadius := by
148       change ¬ comparatorRadius <=
149         comparatorObserved sample - comparatorTrue sample at hcomparator
150       exact lt_of_not_ge hcomparator
151     have hexploration :
152         exploredPredictable sample <=
153           purePredictable sample + gamma * (horizon : Real) := by
154       have h := (sampledTrajectory_finiteHorizon_explorationBias_secondMoment
155         arms harms eta gamma hgamma_pos.le hgamma_lt_one loss horizon sample).1
156       simpa [exploredPredictable, purePredictable,
157         sampledTrajectoryExploredPredictableLossAt,
158         sampledTrajectoryPurePredictableLossAt,
159         sampledTrajectoryPureProbabilityAt] using h
160     have hcoef_pos : 0 < eta * (1 / (1 - gamma)) :=
161       mul_pos heta (one_div_pos.mpr (sub_pos.mpr hgamma_lt_one))
162     have hweightedSecond :
163         (eta * (1 / (1 - gamma))) * secondMoment sample <
164           (eta * (1 / (1 - gamma))) * secondMomentBudget :=
165       mul_lt_mul_of_pos_left hsecondGood hcoef_pos
166     have hhedge' :
167         pureObserved sample - comparatorObserved sample <=
168           Real.log arms.card / eta +
169             (eta * (1 / (1 - gamma))) * secondMoment sample := by
170       simpa [pureObserved, comparatorObserved, secondMoment,
171         sampledObservedMixedSquaredSum] using hsampleHedge
172     have hfinal : exploredPredictable sample - comparatorTrue sample <
173         sampledPredictableBernsteinSquareHighProbabilityRegretBudget
174           arms eta gamma horizon deltaSquare deltaConfidence := by
175       dsimp [sampledPredictableBernsteinSquareHighProbabilityRegretBudget,
176         secondMomentBudget, pureRadius, comparatorRadius]
177       linarith
178     exact (not_lt_of_ge hregret) hfinal
179   calc
180     mu {sample |
181         sampledPredictableBernsteinSquareHighProbabilityRegretBudget
182             arms eta gamma horizon deltaSquare deltaConfidence <=
183           (Finset.range horizon).sum (fun t =>
184               sampledTrajectoryExploredPredictableLossAt
185                 arms eta gamma loss t sample) -
186             (Finset.range horizon).sum (fun t =>
187               predictableLossAt loss t sample comparator)} <=
188         mu ((secondMomentBad ∪ pureBad) ∪ comparatorBad) := by
189       apply measure_mono_ae
190       simpa [exploredPredictable, comparatorTrue, mu] using hsubset
191     _ <= (mu secondMomentBad + mu pureBad) + mu comparatorBad := by
192       exact (measure_union_le _ _).trans
193         (add_le_add (measure_union_le _ _) le_rfl)
194     _ <= (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
195         ENNReal.ofReal deltaConfidence :=
196       add_le_add (add_le_add hsecondTail hpureTail) hcomparatorTail

198 /-- Total-failure form with all three Bernstein events allocated `delta / 3`. -/
199 theorem sampledPredictable_bernsteinSquareHighProbabilityRegret_tail_total_delta
200     {Env : Type u} {Action : Type v}
201     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
202     [MeasurableSpace Action] [MeasurableSingletonClass Action]
203     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
204     (prior : Measure Env) [IsProbabilityMeasure prior]
205     (arms : Finset Action) (harms : arms.Nonempty)
206     (eta gamma : Real) (heta : 0 < eta)
207     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
208     (loss : PredictableLossVector Env Action)
209     (comparator : Action) (hcomparator : comparator ∈ arms)
210     (horizon : Nat)
211     (delta : Real) (hdelta : 0 < delta) :
212     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
213       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
214     mu {sample |
215         sampledPredictableBernsteinSquareHighProbabilityRegretBudget
216             arms eta gamma horizon (delta / 3) (delta / 3) <=
217           (Finset.range horizon).sum (fun t =>
218               sampledTrajectoryExploredPredictableLossAt
219                 arms eta gamma loss t sample) -
220             (Finset.range horizon).sum (fun t =>
221               predictableLossAt loss t sample comparator)} <=
222       ENNReal.ofReal delta := by
223   have hthird_pos : 0 < delta / 3 := div_pos hdelta (by norm_num)
224   have h :=
225     sampledPredictable_bernsteinSquareHighProbabilityRegret_tail
226       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
227         hcomparator horizon (delta / 3) (delta / 3) hthird_pos hthird_pos
228   dsimp only at h ⊢
229   have hthird_nonneg : 0 <= delta / 3 := hthird_pos.le
230   have hprob :
231       (ENNReal.ofReal (delta / 3) + ENNReal.ofReal (delta / 3)) +
232           ENNReal.ofReal (delta / 3) = ENNReal.ofReal delta := by
233     rw [← ENNReal.ofReal_add hthird_nonneg hthird_nonneg]
234     rw [← ENNReal.ofReal_add (add_nonneg hthird_nonneg hthird_nonneg)
235       hthird_nonneg]
236     congr 1
237     ring
238   exact h.trans_eq hprob
