5 /-!
6 # Joint measurability for the OFUL Gaussian mixture
7 
8 This file packages the random-score/random-Gram quadratic exponential on the
9 product of a sample space and a finite-dimensional Gaussian parameter space.
10 It proves the joint measurable surface needed by Tonelli and exposes both a
11 generic product-measure identity and the `N(0, V0⁻¹)` specialization.
12 -/
13 
14 namespace BanditRLProof.OFUL
15 
16 open MeasureTheory ProbabilityTheory Real Matrix WithLp
17 open scoped ENNReal NNReal ProbabilityTheory InnerProductSpace MatrixOrder
18 
19 variable {Omega Feature : Type*}
20   [MeasurableSpace Omega]
21   [Fintype Feature] [DecidableEq Feature]
22 
23 /-- The quadratic exponential mixed over the Gaussian direction in OFUL. -/
24 noncomputable def gaussianQuadraticExponential
25     (score : Omega -> EuclideanSpace Real Feature)
26     (gram : Omega -> Matrix Feature Feature Real)
27     (p : Omega × EuclideanSpace Real Feature) : Real :=
28   Real.exp
29     (⟪score p.1, p.2⟫_ℝ -
30       ⟪p.2, Matrix.toEuclideanCLM (𝕜 := Real) (gram p.1) p.2⟫_ℝ / 2)
31 
32 omit [DecidableEq Feature] in
33 private theorem measurable_gaussianQuadraticExponential_dot
34     (score : Omega -> EuclideanSpace Real Feature)
35     (gram : Omega -> Matrix Feature Feature Real)
36     (hscore : Measurable score)
37     (hgram : forall i j, Measurable (fun omega => gram omega i j)) :
38     Measurable
39       (fun p : Omega × EuclideanSpace Real Feature =>
40         Real.exp
41           ((WithLp.ofLp (score p.1)) ⬝ᵥ (WithLp.ofLp p.2) -
42             (WithLp.ofLp p.2) ⬝ᵥ
43                 (gram p.1).mulVec (WithLp.ofLp p.2) / 2)) := by
44   simp only [dotProduct, Matrix.mulVec]
45   fun_prop
46 
47 /--
48 Joint measurability of the quadratic exponential from measurable random
49 scores and coordinatewise measurable random Gram matrices.
50 -/
51 theorem measurable_gaussianQuadraticExponential
52     (score : Omega -> EuclideanSpace Real Feature)
53     (gram : Omega -> Matrix Feature Feature Real)
54     (hscore : Measurable score)
55     (hgram : forall i j, Measurable (fun omega => gram omega i j)) :
56     Measurable (gaussianQuadraticExponential score gram) := by
57   change Measurable
58     (fun p : Omega × EuclideanSpace Real Feature =>
59       Real.exp
60         (⟪score p.1, p.2⟫_ℝ -
61           ⟪p.2, Matrix.toEuclideanCLM (𝕜 := Real) (gram p.1) p.2⟫_ℝ / 2))
62   have hdot :=
63     measurable_gaussianQuadraticExponential_dot score gram hscore hgram
64   simpa [EuclideanSpace.inner_eq_star_dotProduct,
65     Matrix.inner_toEuclideanCLM, dotProduct_comm] using hdot
66 
67 /-- The nonnegative extended-real surface used by Tonelli. -/
68 noncomputable def gaussianQuadraticExponentialENNReal
69     (score : Omega -> EuclideanSpace Real Feature)
70     (gram : Omega -> Matrix Feature Feature Real)
71     (p : Omega × EuclideanSpace Real Feature) : ENNReal :=
72   ENNReal.ofReal (gaussianQuadraticExponential score gram p)
73 
74 /-- Joint measurability of the `ENNReal` Tonelli surface. -/
75 theorem measurable_gaussianQuadraticExponentialENNReal
76     (score : Omega -> EuclideanSpace Real Feature)
77     (gram : Omega -> Matrix Feature Feature Real)
78     (hscore : Measurable score)
79     (hgram : forall i j, Measurable (fun omega => gram omega i j)) :
80     Measurable (gaussianQuadraticExponentialENNReal score gram) :=
81   (measurable_gaussianQuadraticExponential score gram hscore hgram).ennreal_ofReal
82 
83 /-- Tonelli for the quadratic exponential under an arbitrary `SFinite` parameter law. -/
84 theorem lintegral_gaussianQuadraticExponentialENNReal_prod
85     (mu : Measure Omega) (nu : Measure (EuclideanSpace Real Feature))
86     [SFinite nu]
87     (score : Omega -> EuclideanSpace Real Feature)
88     (gram : Omega -> Matrix Feature Feature Real)
89     (hscore : Measurable score)
90     (hgram : forall i j, Measurable (fun omega => gram omega i j)) :
91     ∫⁻ p, gaussianQuadraticExponentialENNReal score gram p ∂mu.prod nu =
92       ∫⁻ omega, ∫⁻ theta,
93         gaussianQuadraticExponentialENNReal score gram (omega, theta) ∂nu ∂mu := by
94   exact MeasureTheory.lintegral_prod _
95     (measurable_gaussianQuadraticExponentialENNReal
96       score gram hscore hgram).aemeasurable
97 
98 /-- Tonelli specialized to the `N(0, V0⁻¹)` parameter law used by OFUL. -/
99 theorem lintegral_gaussianQuadraticExponentialENNReal_prod_multivariateGaussian_zero_inv
100     (mu : Measure Omega) (V0 : Matrix Feature Feature Real)
101     (score : Omega -> EuclideanSpace Real Feature)
102     (gram : Omega -> Matrix Feature Feature Real)
103     (hscore : Measurable score)
104     (hgram : forall i j, Measurable (fun omega => gram omega i j)) :
105     ∫⁻ p, gaussianQuadraticExponentialENNReal score gram p
106         ∂mu.prod (ProbabilityTheory.multivariateGaussian 0 V0⁻¹) =
