4 # Scheduled OFUL expected regret under positive action costs
5 
6 This module specializes the cumulative positive-cost budget-exhaustion route
7 to a deterministic Nat-valued cost on the finite action set. At round `t`, the
8 cost process reads the current action from the canonical generated trajectory.
9 -/

20 /-- The deterministic cost of the current canonical trajectory action. -/
21 def canonicalActionCostProcess
22     {K : Nat}
23     (actionCost : Fin K -> Nat)
24     (t : Nat)
25     (trajectory : Nat -> Fin K × Real) : Nat :=
26   actionCost (trajectory t).1
27 
28 /--
29 The current-action cost is adapted to the canonical all-round filtration:
30 the current trajectory coordinate is measurable, as are all maps between
31 countable measurable spaces.
32 -/
33 theorem adapted_canonicalActionCostProcess
34     {K : Nat}
35     (actionCost : Fin K -> Nat) :
36     Adapted
37       (OFUL.canonicalHistoryTrajectoryAllRoundFiltration (K := K))
38       (canonicalActionCostProcess actionCost) := by
39   intro t
40   exact
41     (measurable_of_countable actionCost).comp
42       (measurable_fst.comp
43         (OFUL.measurable_canonicalHistoryTrajectory_coordinate_allRound
44           (K := K) (t := t) (horizon := t) le_rfl))
45 
46 /-- Positive arm costs give a pointwise positive trajectory cost process. -/
47 theorem canonicalActionCostProcess_one_le
48     {K : Nat}
49     (actionCost : Fin K -> Nat)
50     (hpositive : forall action, 1 <= actionCost action) :
51     forall t trajectory, 1 <= canonicalActionCostProcess actionCost t trajectory := by
52   intro t trajectory
53   exact hpositive (trajectory t).1
54 
55 /--
56 Action cost spent after `t` completed rounds, using the half-open index set
57 `{0, ..., t - 1}`.
58 -/
59 def cumulativeActionCost
60     {K : Nat}
61     (actionCost : Fin K -> Nat)
62     (t : Nat)
63     (trajectory : Nat -> Fin K × Real) : Nat :=
64   cumulativeSpent (canonicalActionCostProcess actionCost) t trajectory
65 
66 @[simp]
67 theorem cumulativeActionCost_apply
68     {K : Nat}
69     (actionCost : Fin K -> Nat)
70     (t : Nat)
71     (trajectory : Nat -> Fin K × Real) :
72     cumulativeActionCost actionCost t trajectory =
73       (Finset.range t).sum (fun s => actionCost (trajectory s).1) :=
74   rfl
75 
76 /-- The half-open cumulative action-cost process remains adapted. -/
77 theorem adapted_cumulativeActionCost
78     {K : Nat}
79     (actionCost : Fin K -> Nat) :
80     Adapted
81       (OFUL.canonicalHistoryTrajectoryAllRoundFiltration (K := K))
82       (cumulativeActionCost actionCost) := by
83   simpa only [cumulativeActionCost] using
84     adapted_cumulativeSpent
85       (adapted_canonicalActionCostProcess actionCost)
86 
87 /-- Positive arm costs make cumulative action cost grow by at least one. -/
88 theorem cumulativeActionCost_unitGrowth_of_one_le
89     {K : Nat}
90     (actionCost : Fin K -> Nat)
91     (hpositive : forall action, 1 <= actionCost action) :
92     forall t trajectory,
93       cumulativeActionCost actionCost t trajectory + 1 <=
94         cumulativeActionCost actionCost (t + 1) trajectory := by
95   simpa only [cumulativeActionCost] using
96     cumulativeSpent_unitGrowth_of_one_le
97       (canonicalActionCostProcess actionCost)
98       (canonicalActionCostProcess_one_le actionCost hpositive)
