5 # Oracle restart at global population-mean change points
6 
7 This module builds a deterministic oracle schedule that restarts exactly after
8 each supplied change point.  It then specializes the change predicate to
9 population-mean changes of an independent finite-arm reward law.
10 -/

19 /-- Start of the most recent epoch when a restart occurs after every `change`
20 boundary.  A true value at `s` starts the new epoch at `s + 1`. -/
21 def oracleChangePointRestartStart
22     (change : Nat -> Prop) [DecidablePred change] : Nat -> Nat
23   | 0 => 0
24   | n + 1 =>
25       if change n then n + 1 else oracleChangePointRestartStart change n
26 
27 /-- A valid restart schedule generated by an arbitrary decidable boundary
28 predicate. -/
29 def oracleChangePointRestartSchedule
30     (change : Nat -> Prop) [DecidablePred change] : OracleRestartSchedule where
31   start := oracleChangePointRestartStart change
32   start_zero := rfl
33   start_le := by
34     intro t
35     induction t with
36     | zero => simp [oracleChangePointRestartStart]
37     | succ n ih =>
38         simp only [oracleChangePointRestartStart]
39         split <;> omega
40   start_succ := by
41     intro n
42     by_cases hchange : change n
43     · exact Or.inr (by simp [oracleChangePointRestartStart, hchange])
44     · exact Or.inl (by simp [oracleChangePointRestartStart, hchange])
45 
46 @[simp]
47 theorem oracleChangePointRestartSchedule_start_zero
48     (change : Nat -> Prop) [DecidablePred change] :
49     (oracleChangePointRestartSchedule change).start 0 = 0 := rfl
50 
51 @[simp]
52 theorem oracleChangePointRestartSchedule_start_succ
53     (change : Nat -> Prop) [DecidablePred change] (n : Nat) :
54     (oracleChangePointRestartSchedule change).start (n + 1) =
55       if change n then n + 1
56       else (oracleChangePointRestartSchedule change).start n := rfl

58 /-- The registered epochs are zero together with every successor of a change
59 point before the inclusive horizon. -/
60 theorem oracleRestartScheduleEpochs_oracleChangePointRestartSchedule
61     (change : Nat -> Prop) [DecidablePred change] (horizon : Nat) :
62     oracleRestartScheduleEpochs
63         (oracleChangePointRestartSchedule change) horizon =
64       insert 0
65         (((Finset.range horizon).filter change).image (fun s => s + 1)) := by
66   induction horizon with
67   | zero =>
68       simp [oracleRestartScheduleEpochs,
69         oracleChangePointRestartSchedule, oracleChangePointRestartStart]
70   | succ horizon ih =>
71       rw [oracleRestartScheduleEpochs]
72       rw [show horizon + 1 + 1 = (horizon + 1) + 1 by omega]
73       rw [Finset.range_add_one, Finset.image_insert]
74       change
75         insert
76             ((oracleChangePointRestartSchedule change).start (horizon + 1))
77             (oracleRestartScheduleEpochs
78               (oracleChangePointRestartSchedule change) horizon) =
79           _
80       have hstartMem :
81           (oracleChangePointRestartSchedule change).start horizon ∈
82             insert 0
83               (((Finset.range horizon).filter change).image
84                 (fun s => s + 1)) := by
85         rw [← ih]
86         exact oracleRestartSchedule_start_mem_epochs
87           (oracleChangePointRestartSchedule change) horizon horizon (by simp)
88       rw [ih]
89       by_cases hchange : change horizon
90       · simp [Finset.range_add_one, Finset.filter_insert,
91           Finset.image_insert,
92           oracleChangePointRestartSchedule_start_succ, hchange,
93           Finset.insert_comm]
94       · simp [Finset.range_add_one, Finset.filter_insert,
95           oracleChangePointRestartSchedule_start_succ, hchange, hstartMem]

97 /-- A change-point restart schedule has exactly one more visited epoch than
98 the number of change boundaries before the inclusive horizon. -/
99 theorem oracleRestartScheduleEpochs_card_oracleChangePointRestartSchedule
100     (change : Nat -> Prop) [DecidablePred change] (horizon : Nat) :
101     (oracleRestartScheduleEpochs
102         (oracleChangePointRestartSchedule change) horizon).card =
103       ((Finset.range horizon).filter change).card + 1 := by
104   rw [oracleRestartScheduleEpochs_oracleChangePointRestartSchedule]
105   rw [Finset.card_insert_of_notMem]
106   · rw [Finset.card_image_of_injective]
107     intro s t hst
108     exact Nat.add_right_cancel hst
109   · simp

111 /-- A boundary has a global population-mean change when at least one arm
112 changes mean between its two adjacent reward laws. -/
113 def finiteArmIndependentGlobalMeanChangeAt {K : Nat}
114     (armLaw : Nat -> Fin K -> Measure Rat) (s : Nat) : Prop :=
115   ∃ arm : Fin K,
116     finiteArmIndependentRewardMean armLaw (s + 1) arm ≠
117       finiteArmIndependentRewardMean armLaw s arm
118 
119 /-- Global population-mean change boundaries strictly before `horizon`. -/
120 noncomputable def finiteArmIndependentGlobalMeanChangeTimes {K : Nat}
121     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) : Finset Nat := by
122   classical
123   exact (Finset.range horizon).filter
124     (finiteArmIndependentGlobalMeanChangeAt armLaw)
125 
126 /-- Population-mean oracle schedule: restart immediately after every global
127 mean-change boundary. -/
128 noncomputable def finiteArmIndependentGlobalMeanChangeRestartSchedule
129     {K : Nat} (armLaw : Nat -> Fin K -> Measure Rat) :
130     OracleRestartSchedule := by
131   classical
132   exact oracleChangePointRestartSchedule
133     (finiteArmIndependentGlobalMeanChangeAt armLaw)

135 /-- The population-mean restart schedule has exactly one initial epoch plus
136 one epoch for every global mean-change boundary. -/
137 theorem
138     oracleRestartScheduleEpochs_card_finiteArmIndependentGlobalMeanChangeRestartSchedule
139     {K : Nat} (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) :
140     (oracleRestartScheduleEpochs
141         (finiteArmIndependentGlobalMeanChangeRestartSchedule armLaw)
142         horizon).card =
143       (finiteArmIndependentGlobalMeanChangeTimes armLaw horizon).card + 1 := by
144   classical
145   simpa [finiteArmIndependentGlobalMeanChangeRestartSchedule,
146     finiteArmIndependentGlobalMeanChangeTimes,
147     finiteArmIndependentGlobalMeanChangeAt] using
148     (oracleRestartScheduleEpochs_card_oracleChangePointRestartSchedule
149       (finiteArmIndependentGlobalMeanChangeAt armLaw) horizon)

151 /-- The existing real-valued global switch count is the coercion of the named
152 change-time finset cardinality. -/
153 theorem finiteArmIndependentCumulativeGlobalMeanSwitchCount_eq_changeTimes_card
154     {K : Nat} (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) :
155     finiteArmIndependentCumulativeGlobalMeanSwitchCount armLaw horizon =
156       ((finiteArmIndependentGlobalMeanChangeTimes armLaw horizon).card : Real) := by
157   classical
158   have htimes :
159       finiteArmIndependentGlobalMeanChangeTimes armLaw horizon =
160         (Finset.range horizon).filter (fun s =>
161           ∃ arm : Fin K,
162             finiteArmIndependentRewardMean armLaw (s + 1) arm ≠
163               finiteArmIndependentRewardMean armLaw s arm) := by
164     ext s
165     simp [finiteArmIndependentGlobalMeanChangeTimes,
166       finiteArmIndependentGlobalMeanChangeAt]

184 /-- Since every global mean change starts a new epoch, each arm's population
185 mean is constant from the current epoch start through the current round. -/
186 theorem
187     finiteArmIndependentRewardMean_eq_globalMeanChangeRestartStart
188     {K : Nat} (armLaw : Nat -> Fin K -> Measure Rat)
189     (t : Nat) (arm : Fin K) :
190     finiteArmIndependentRewardMean armLaw t arm =
191       finiteArmIndependentRewardMean armLaw
192         ((finiteArmIndependentGlobalMeanChangeRestartSchedule armLaw).start t)
193         arm := by
194   classical
195   induction t with
196   | zero =>
197       simp [finiteArmIndependentGlobalMeanChangeRestartSchedule,
198         oracleChangePointRestartSchedule, oracleChangePointRestartStart]
199   | succ n ih =>
200       by_cases hchange :
201           finiteArmIndependentGlobalMeanChangeAt armLaw n
202       · simp [finiteArmIndependentGlobalMeanChangeRestartSchedule,
203           oracleChangePointRestartSchedule, oracleChangePointRestartStart,
204           hchange]
205       · have hsame :
206             finiteArmIndependentRewardMean armLaw (n + 1) arm =
207               finiteArmIndependentRewardMean armLaw n arm := by
208           apply not_ne_iff.mp
209           intro hne
210           exact hchange ⟨arm, hne⟩
211         rw [hsame, ih]
212         simp [finiteArmIndependentGlobalMeanChangeRestartSchedule,
213           oracleChangePointRestartSchedule, oracleChangePointRestartStart,
214           hchange]
