5 # Conditional-mean producer for scheduled half-Tsallis expected gaps
6 
7 This module exposes the sampling-before-action sigma-algebra at each scheduled
8 time.  The scheduled action probability is measurable with respect to that
9 sigma-algebra, so Mathlib's conditional-expectation pull-out theorem turns a
10 constant conditional loss-gap law into `HasScheduledExpectedGapLaw`.
11 -/

20 /-- Information in the generated action/reward trace strictly before the
21 scheduled action at time `t`.  At time zero there is no trace information; at
22 time `n + 1` this is the sigma-algebra generated by the prefix through `n`. -/
23 @[reducible] def sampledScheduledHalfTsallisPastSigma
24     {Env : Type u} {Action : Type v} [MeasurableSpace Action] (t : Nat) :
25     MeasurableSpace (Env × ((k : Nat) -> Action × Real)) :=
26   match t with
27   | 0 => ⊥
28   | n + 1 =>
29       (inferInstance : MeasurableSpace
30         (History.FinitePairHistory Action Real n)).comap
31           (fun sample => Preorder.frestrictLe n sample.2)

78 available before the scheduled action at each time. -/

80     {Env : Type u} {Action : Type v}
81     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
82     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
83     (arms : Finset Action) (loss : Exp3.PredictableLossVector Env Action)
84     (best : Action) (gap : Action -> Real) (horizon : Nat) : Prop :=
85   ∀ t, t <= horizon -> ∀ action, action ∈ arms.erase best ->
86     condExp
87         (sampledScheduledHalfTsallisPastSigma
88           (Env := Env) (Action := Action) t)
89         mu
90         (fun sample =>
91           Exp3.predictableLossAt loss t sample action -
