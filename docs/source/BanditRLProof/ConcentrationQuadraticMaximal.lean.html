5 # Finite maximal quadratic fixed-MGF tails
6 
7 This module adds a finite-index maximal surface to the quadratic fixed-MGF
8 route.  It uses equal confidence shares and the finite outer-measure union
9 bound; it is not a Ville, Doob, or infinite-horizon maximal inequality.
10 -/
11 
12 namespace BanditRLProof.Concentration
13 
14 open MeasureTheory ProbabilityTheory Real
15 
16 open scoped ENNReal
17 
18 /-- Equal-share quadratic radius for a finite family of events. -/
19 noncomputable def quadraticFixedMGFMaximalRadius
20     {Idx : Type*} [DecidableEq Idx] (times : Finset Idx)
21     (varianceScale varianceBudget tiltCap delta : Real) : Real :=
22   quadraticFixedMGFRadius varianceScale varianceBudget tiltCap
23     (delta / (times.card : Real))
24 
25 /-- Finite maximal delta tail obtained by optimizing every fixed-tilt event
26 at confidence `delta / times.card` and taking the finite union. -/
27 theorem measure_biUnion_deviation_ge_inter_variance_le_delta_of_fixedTilt_quadratic_tail
28     {Omega Idx : Type*} [MeasurableSpace Omega] [DecidableEq Idx]
29     (mu : Measure Omega) (times : Finset Idx) (htimes : times.Nonempty)
30     (deviation predictableVariance : Idx -> Omega -> Real)
31     (varianceScale varianceBudget tiltCap delta : Real)
32     (hvarianceScale : 0 < varianceScale)
33     (hvarianceBudget : 0 < varianceBudget) (htiltCap : 0 < tiltCap)
34     (hdelta : 0 < delta)
35     (hfixed : forall i, i ∈ times -> forall tilt,
36       0 <= tilt -> tilt <= tiltCap ->
37         mu {omega |
38             quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
39                 tiltCap delta <= deviation i omega ∧
40               predictableVariance i omega <= varianceBudget} <=
41           ENNReal.ofReal (Real.exp
42             (-tilt * quadraticFixedMGFMaximalRadius times varianceScale
43                 varianceBudget tiltCap delta +
44               varianceScale * (tilt ^ 2 * varianceBudget)))) :
45     mu (⋃ i ∈ times, {omega |
46         quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
47             tiltCap delta <= deviation i omega ∧
48           predictableVariance i omega <= varianceBudget}) <=
49       ENNReal.ofReal delta := by
50   let deltaShare := delta / (times.card : Real)
51   have hcardNat : 0 < times.card := Finset.card_pos.mpr htimes
52   have hcardReal : 0 < (times.card : Real) := Nat.cast_pos.mpr hcardNat
53   have hdeltaShare : 0 < deltaShare := div_pos hdelta hcardReal
54   have htail : ∀ i ∈ times,
55       mu {omega |
56           quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
57               tiltCap delta <= deviation i omega ∧
58             predictableVariance i omega <= varianceBudget} <=
59         ENNReal.ofReal deltaShare := by
60     intro i hi
61     have hiTail :=
62       measure_deviation_ge_inter_variance_le_delta_of_fixedTilt_quadratic_tail
63         mu (deviation i) (predictableVariance i) varianceScale varianceBudget
64           tiltCap deltaShare hvarianceScale hvarianceBudget htiltCap hdeltaShare
65           (by
66             intro tilt htilt_nonneg htilt_le
67             simpa [quadraticFixedMGFMaximalRadius, deltaShare] using
68               hfixed i hi tilt htilt_nonneg htilt_le)
69     simpa [quadraticFixedMGFMaximalRadius, deltaShare] using hiTail
70   have hcardENN_ne_zero : (times.card : ENNReal) ≠ 0 := by
71     simp [Finset.card_ne_zero.mpr htimes]
72   have hcardENN_ne_top : (times.card : ENNReal) ≠ ⊤ := by simp
73   calc
74     mu (⋃ i ∈ times, {omega |
75         quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
76             tiltCap delta <= deviation i omega ∧
77           predictableVariance i omega <= varianceBudget}) <=
78         ∑ i ∈ times, mu {omega |
79           quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
80               tiltCap delta <= deviation i omega ∧
81             predictableVariance i omega <= varianceBudget} :=
82       ProbabilityUnionBound.measure_biUnion_finset_le mu times fun i =>
83         {omega |
84           quadraticFixedMGFMaximalRadius times varianceScale varianceBudget
85               tiltCap delta <= deviation i omega ∧
86             predictableVariance i omega <= varianceBudget}
87     _ <= ∑ _i ∈ times, ENNReal.ofReal deltaShare := by
88       exact Finset.sum_le_sum fun i hi => htail i hi
89     _ = (times.card : ENNReal) * ENNReal.ofReal deltaShare := by
90       simp [nsmul_eq_mul]
91     _ = ENNReal.ofReal delta := by
92       dsimp [deltaShare]
93       rw [ENNReal.ofReal_div_of_pos hcardReal]
94       simp only [ENNReal.ofReal_natCast]
95       exact ENNReal.mul_div_cancel hcardENN_ne_zero hcardENN_ne_top
