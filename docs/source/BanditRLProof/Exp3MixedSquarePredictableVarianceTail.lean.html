3 /-!
4 # Predictable-variance tail for the mixed-square EXP3 process
5 
6 This module keeps the exact conditional second moment random. It first
7 compensates each centered mixed-square increment by its finite-law variance,
8 then iterates the resulting zero-budget conditional MGF. The main endpoint is
9 a fixed-tilt tail on the event that the cumulative predictable variance is at
10 most a caller-supplied budget.
11 -/

19 /-- A centered mixed-square score is bounded by the reciprocal probability
20 floor on the support of the finite sampling distribution. -/
21 theorem abs_mixedSquaredEstimatorDeviation_le_inv_floor
22     {History : Type u} {Action : Type v}
23     [MeasurableSpace History] [MeasurableSpace Action] [DecidableEq Action]
24     (arms : Finset Action) (prob loss : History -> Action -> Real)
25     (history : History) (hdist : FiniteActionDistribution arms (prob history))
26     (epsilon : Real)
27     (regularity : BoundedMeasurableLossWithProbabilityFloor
28       arms prob loss epsilon)
29     (chosen : Action) (hchosen : chosen ∈ arms) :
30     |mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
31         chosen - arms.sum (fun action => (loss history action) ^ 2)| <=
32       1 / epsilon := by
33   let raw := fun selected =>
34     mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
35       selected
36   let mean := arms.sum (fun action => (loss history action) ^ 2)
37   have hprob_pos (action : Action) (haction : action ∈ arms) :
38       0 < prob history action :=
39     regularity.epsilon_pos.trans_le (regularity.prob_floor history action haction)
40   have hmean_nonneg : 0 <= mean :=
41     Finset.sum_nonneg fun action _haction => sq_nonneg (loss history action)
42   have hmean_le_card : mean <= (arms.card : Real) := by
43     dsimp only [mean]
44     calc
45       arms.sum (fun action => (loss history action) ^ 2) <=
46           arms.sum (fun _action => (1 : Real)) := by
47             apply Finset.sum_le_sum
48             intro action haction
49             simpa using
50               (sq_le_sq₀ (regularity.loss_mem_Icc history action haction).1
51                 zero_le_one).2
52                 (regularity.loss_mem_Icc history action haction).2
53       _ = (arms.card : Real) := by simp
54   have hcard_le_inv : (arms.card : Real) <= 1 / epsilon := by
55     rw [le_div_iff₀ regularity.epsilon_pos]
56     have hsum : arms.sum (fun _action => epsilon) <=
57         arms.sum (prob history) :=
58       Finset.sum_le_sum fun action haction =>
59         regularity.prob_floor history action haction
60     simpa [hdist.sum_eq_one, mul_comm] using hsum
61   have hraw_nonneg : 0 <= raw chosen := by
62     unfold raw mixedSquaredImportanceWeightedLoss
63     exact Finset.sum_nonneg fun action haction =>
64       mul_nonneg (hdist.nonneg action haction) (sq_nonneg _)
65   have hraw_le : raw chosen <= 1 / epsilon := by
66     dsimp only [raw]
67     rw [mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
68       arms (prob history) (loss history) chosen hchosen
69         (hprob_pos chosen hchosen).ne']
70     have hsq : (loss history chosen) ^ 2 <= 1 := by
71       simpa using
72         (sq_le_sq₀ (regularity.loss_mem_Icc history chosen hchosen).1
73           zero_le_one).2
74           (regularity.loss_mem_Icc history chosen hchosen).2
75     calc
76       (loss history chosen) ^ 2 / prob history chosen <=
77           1 / prob history chosen :=
78         div_le_div_of_nonneg_right hsq (hprob_pos chosen hchosen).le
79       _ <= 1 / epsilon := one_div_le_one_div_of_le regularity.epsilon_pos
80         (regularity.prob_floor history chosen hchosen)
81   change |raw chosen - mean| <= 1 / epsilon
82   rw [abs_le]
83   constructor <;> nlinarith

85 /-- Exact finite-law fixed-tilt MGF bound. Unlike the deterministic Bernstein
86 wrapper, its exponent retains the actual centered second moment. -/
87 theorem finiteActionMixedSquaredEstimator_hasMGFUpperBoundAt_variance
88     {History : Type u} {Action : Type v}
89     [MeasurableSpace History]
90     [MeasurableSpace Action] [MeasurableSingletonClass Action]
91     [DecidableEq Action]
92     (arms : Finset Action) (prob loss : History -> Action -> Real)
93     (history : History) (hdist : FiniteActionDistribution arms (prob history))
94     (epsilon : Real)
95     (regularity : BoundedMeasurableLossWithProbabilityFloor
96       arms prob loss epsilon)
97     (tilt : Real) (htilt_nonneg : 0 <= tilt) (htilt_le : tilt <= epsilon) :
98     Concentration.HasMGFUpperBoundAt
99       (fun chosen =>
100         mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
101             chosen -
102           arms.sum (fun action => (loss history action) ^ 2))
103       tilt
104       (tilt ^ 2 *
105         mixedSquaredEstimatorCenteredSecondMoment arms prob loss history)
106       (finiteActionMeasure arms (prob history)) := by
107   let raw := fun chosen =>
108     mixedSquaredImportanceWeightedLoss arms (prob history) (loss history) chosen
109   let mean := arms.sum (fun action => (loss history action) ^ 2)
110   let X := fun chosen => raw chosen - mean
111   let variance := mixedSquaredEstimatorCenteredSecondMoment arms prob loss history
112   letI : IsProbabilityMeasure (finiteActionMeasure arms (prob history)) :=
113     finiteActionMeasure_isProbabilityMeasure arms (prob history) hdist
114   have hprob_pos (action : Action) (haction : action ∈ arms) :
115       0 < prob history action :=
116     regularity.epsilon_pos.trans_le (regularity.prob_floor history action haction)
117   have hinv_nonneg : 0 <= 1 / epsilon :=
118     (one_div_pos.mpr regularity.epsilon_pos).le
119   have hX_bound (chosen : Action) (hchosen : chosen ∈ arms) :
120       |X chosen| <= 1 / epsilon := by
121     simpa [X, raw, mean] using
122       abs_mixedSquaredEstimatorDeviation_le_inv_floor
123         arms prob loss history hdist epsilon regularity chosen hchosen
124   constructor
125   · intro s
126     rw [finiteActionMeasure]
127     refine integrable_finset_sum_measure.2 fun action haction => ?_
128     exact (integrable_dirac (by simp)).smul_measure ENNReal.ofReal_ne_top
129   · rw [ProbabilityTheory.mgf,
130       integral_finiteActionMeasure_eq_sum arms (prob history) hdist]
131     have hexp (chosen : Action) (hchosen : chosen ∈ arms) :
132         Real.exp (tilt * X chosen) <=
133           1 + tilt * X chosen + (tilt * X chosen) ^ 2 := by
134       apply Concentration.exp_le_one_add_self_add_sq_of_abs_le_one
135       rw [abs_mul, abs_of_nonneg htilt_nonneg]
136       calc
137         tilt * |X chosen| <= tilt * (1 / epsilon) :=
138           mul_le_mul_of_nonneg_left (hX_bound chosen hchosen) htilt_nonneg
139         _ <= epsilon * (1 / epsilon) :=
140           mul_le_mul_of_nonneg_right htilt_le hinv_nonneg
141         _ = 1 := by field_simp [ne_of_gt regularity.epsilon_pos]
142     calc
143       arms.sum (fun chosen =>
144           prob history chosen * Real.exp (tilt * X chosen)) <=
145           arms.sum (fun chosen => prob history chosen *
146             (1 + tilt * X chosen + (tilt * X chosen) ^ 2)) := by
147         apply Finset.sum_le_sum
148         intro chosen hchosen
149         exact mul_le_mul_of_nonneg_left (hexp chosen hchosen)
150           (hdist.nonneg chosen hchosen)
151       _ = 1 + tilt * arms.sum (fun chosen => prob history chosen * X chosen) +
152           tilt ^ 2 * arms.sum (fun chosen =>
153             prob history chosen * (X chosen) ^ 2) := by
154         have hterm (chosen : Action) :
155             prob history chosen *
156                 (1 + tilt * X chosen + (tilt * X chosen) ^ 2) =
157               (prob history chosen +
158                 tilt * (prob history chosen * X chosen)) +
159                 tilt ^ 2 * (prob history chosen * (X chosen) ^ 2) := by ring
160         simp_rw [hterm, Finset.sum_add_distrib, ← Finset.mul_sum,
161           hdist.sum_eq_one]
162       _ = 1 + tilt ^ 2 * variance := by
163         have hmean : arms.sum (fun chosen => prob history chosen * X chosen) = 0 := by
164           dsimp only [X]
165           simp_rw [mul_sub]
166           rw [Finset.sum_sub_distrib,
167             sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
168               arms (prob history) (loss history)
169                 (fun action haction => (hprob_pos action haction).ne'),
170             ← Finset.sum_mul, hdist.sum_eq_one]
171           simp [mean]
172         rw [hmean, mul_zero, add_zero]
173         rfl
174       _ <= Real.exp (tilt ^ 2 * variance) := by
175         simpa [add_comm] using Real.add_one_le_exp (tilt ^ 2 * variance)

713 /-- Optimized radius for the mixed-square predictable-variance event. -/
714 noncomputable def sampledMixedSquaredPredictableVarianceRadius
715     {Action : Type v} [DecidableEq Action]
716     (arms : Finset Action) (gamma varianceBudget delta : Real) : Real :=
