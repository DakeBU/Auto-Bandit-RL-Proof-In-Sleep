19 /-- The finite-horizon martingale score `sum_{i<n} x_i eta_i`. -/
20 noncomputable def finiteHorizonNoiseScore
21     [Fintype Feature]
22     (feature : Nat -> Omega -> Feature -> Real)
23     (noise : Nat -> Omega -> Real)
24     (n : Nat) (omega : Omega) : EuclideanSpace Real Feature :=
25   WithLp.toLp 2 (fun j =>
26     (Finset.range n).sum (fun i => feature i omega j * noise i omega))
27 
28 /-- Feature rescaling whose ordinary Gram is weighted by the variance proxy. -/
29 noncomputable def varianceWeightedFeature
30     (feature : Nat -> Omega -> Feature -> Real)
31     (varianceProxy : Nat -> NNReal)
32     (i : Nat) (omega : Omega) (j : Feature) : Real :=
33   Real.sqrt (((varianceProxy i : NNReal) : Real)) * feature i omega j

35 /-- The finite-horizon variance-weighted Gram `sum_{i<n} c_i x_i x_i^T`. -/

37     [Fintype Feature]
38     (feature : Nat -> Omega -> Feature -> Real)
39     (varianceProxy : Nat -> NNReal)
40     (n : Nat) (omega : Omega) : Matrix Feature Feature Real :=
41   prefixFeatureGram
42     (fun i => varianceWeightedFeature feature varianceProxy i omega) n

44 /-- The score inner product is the finite sum of projected noise increments. -/
45 theorem inner_finiteHorizonNoiseScore_eq_sum_projection_mul_noise
46     [Fintype Feature]
47     (feature : Nat -> Omega -> Feature -> Real)
48     (noise : Nat -> Omega -> Real)
49     (n : Nat) (omega : Omega)
50     (theta : EuclideanSpace Real Feature) :
51     ⟪finiteHorizonNoiseScore feature noise n omega, theta⟫_Real =
52       (Finset.range n).sum (fun i =>
53         dotProduct (WithLp.ofLp theta) (feature i omega) * noise i omega) := by
54   change
55     dotProduct (WithLp.ofLp theta)
56         (fun j =>
57           (Finset.range n).sum
58             (fun i => feature i omega j * noise i omega)) =
59       (Finset.range n).sum (fun i =>
60         dotProduct (WithLp.ofLp theta) (feature i omega) * noise i omega)
61   simp only [dotProduct, Finset.mul_sum]
62   rw [Finset.sum_comm]
63   apply Finset.sum_congr rfl
64   intro i hi
65   rw [Finset.sum_mul]
66   apply Finset.sum_congr rfl
67   intro j hj
68   ring

70 /-- The weighted Gram quadratic form is the variance-weighted projection sum. -/
71 theorem finiteHorizonVarianceGram_quadraticForm_eq_sum
72     [Fintype Feature]
73     (feature : Nat -> Omega -> Feature -> Real)
74     (varianceProxy : Nat -> NNReal)
75     (n : Nat) (omega : Omega) (theta : Feature -> Real) :
76     quadraticForm
77         (finiteHorizonVarianceGram feature varianceProxy n omega) theta =
78       (Finset.range n).sum (fun i =>
79         (((varianceProxy i : NNReal) : Real)) *
80           (dotProduct theta (feature i omega)) ^ 2) := by
81   rw [finiteHorizonVarianceGram,
82     prefixFeatureGram_quadraticForm_eq_sum_sq]
83   apply Finset.sum_congr rfl
84   intro i hi
85   have hnonneg :
86       0 <= (((varianceProxy i : NNReal) : Real)) := by positivity
87   rw [show
88       (Finset.univ : Finset Feature).sum (fun j =>
89           varianceWeightedFeature feature varianceProxy i omega j * theta j) =
90         Real.sqrt (((varianceProxy i : NNReal) : Real)) *
91           dotProduct theta (feature i omega) by
92     simp only [varianceWeightedFeature, dotProduct]
93     rw [Finset.mul_sum]
94     apply Finset.sum_congr rfl
95     intro j hj
96     ring]
97   rw [mul_pow, Real.sq_sqrt hnonneg]

99 /-- The finite-horizon variance Gram is positive semidefinite. -/
100 theorem finiteHorizonVarianceGram_posSemidef
101     [Fintype Feature]
102     (feature : Nat -> Omega -> Feature -> Real)
103     (varianceProxy : Nat -> NNReal)
104     (n : Nat) (omega : Omega) :
105     (finiteHorizonVarianceGram feature varianceProxy n omega).PosSemidef := by
106   refine Matrix.PosSemidef.of_dotProduct_mulVec_nonneg
107     (prefixFeatureGram_isHermitian
108       (fun i => varianceWeightedFeature feature varianceProxy i omega) n) ?_
109   intro theta
110   have hstar : star theta = theta := by
111     funext i
112     simp
113   rw [hstar, dotProduct_mulVec_eq_quadraticForm,
114     finiteHorizonVarianceGram_quadraticForm_eq_sum]

116     mul_nonneg (by positivity) (sq_nonneg _))

146 /--
147 Pointwise identification of the fixed-direction compensated finite sum with
148 the random score/random-Gram quadratic exponent.
149 -/
150 theorem compensatedScore_eq_inner_sub_varianceGram
151     [Fintype Feature] [DecidableEq Feature]
152     (feature : Nat -> Omega -> Feature -> Real)
153     (noise : Nat -> Omega -> Real)
154     (varianceProxy : Nat -> NNReal)
155     (n : Nat) (omega : Omega)
156     (theta : EuclideanSpace Real Feature) :
157     (Finset.range (n + 1)).sum (fun t =>
158       match t with
159       | 0 => 0
160       | i + 1 =>
161           dotProduct (WithLp.ofLp theta) (feature i omega) * noise i omega -
162             (((varianceProxy i : NNReal) : Real) *
163               (dotProduct (WithLp.ofLp theta) (feature i omega)) ^ 2 / 2)) =
164       ⟪finiteHorizonNoiseScore feature noise n omega, theta⟫_Real -
165         ⟪theta,

167             (finiteHorizonVarianceGram feature varianceProxy n omega)

169   rw [Finset.sum_range_succ']
170   simp only [inner_finiteHorizonNoiseScore_eq_sum_projection_mul_noise,
171     Matrix.inner_toEuclideanCLM, dotProduct_mulVec_eq_quadraticForm,
