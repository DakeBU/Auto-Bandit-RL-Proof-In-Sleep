3 /-!
4 # All-horizon realized EXP3 Bernstein route
5 
6 The explicit Bernstein schedule gives its `11 * gamma * T` threshold when the
7 three large-horizon inequalities make clipping inactive. This module closes
8 the complementary branch honestly: generated realized losses are at most one
9 almost surely, comparator predictable losses are nonnegative pointwise, and
10 therefore realized regret is at most `T` almost surely. A branch threshold uses
11 `T + 1` outside the Bernstein regime, yielding one theorem for every positive
12 horizon without pretending that the clipped branch satisfies cubic dominance.
13 -/

21 /-- Every predictable comparator coordinate is nonnegative. -/
22 theorem predictableLossAt_nonneg
23     {Env : Type u} {Action : Type v}
24     [MeasurableSpace Env] [MeasurableSpace Action]
25     (loss : PredictableLossVector Env Action) (t : Nat)
26     (sample : Env × ((k : Nat) -> Action × Real)) (action : Action) :
27     0 <= predictableLossAt loss t sample action := by
28   cases t with
29   | zero => exact loss.initial_nonneg sample.1 action
30   | succ n =>
31       exact loss.successor_nonneg n sample.1
32         (Preorder.frestrictLe n sample.2) action

58 before a finite horizon. -/
59 theorem sampledPredictableTrajectoryMeasure_finiteHorizon_realizedLoss_le_one_ae
60     {Env : Type u} {Action : Type v}
61     [MeasurableSpace Env] [StandardBorelSpace Env]
62     [MeasurableSpace Action] [MeasurableSingletonClass Action]
63     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
64     (prior : Measure Env) [IsFiniteMeasure prior]
65     (arms : Finset Action) (harms : arms.Nonempty)
66     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
67     (hgamma_le_one : gamma <= 1)
68     (loss : PredictableLossVector Env Action) (horizon : Nat) :
69     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
70       eta gamma hgamma_nonneg hgamma_le_one loss.environment
71     ∀ᵐ sample ∂mu, ∀ t, t < horizon ->
72       sampledTrajectoryRealizedLossAt t sample <= 1 := by

83 /-- Generated selected-loss regret against any comparator is at most the
84 horizon almost surely. -/
85 theorem sampledPredictable_realizedRegret_le_horizon_ae
86     {Env : Type u} {Action : Type v}
87     [MeasurableSpace Env] [StandardBorelSpace Env]
88     [MeasurableSpace Action] [MeasurableSingletonClass Action]
89     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
90     (prior : Measure Env) [IsFiniteMeasure prior]
91     (arms : Finset Action) (harms : arms.Nonempty)
92     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
93     (hgamma_le_one : gamma <= 1)
94     (loss : PredictableLossVector Env Action) (comparator : Action)
95     (horizon : Nat) :
96     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
97       eta gamma hgamma_nonneg hgamma_le_one loss.environment
98     ∀ᵐ sample ∂mu,
99       (Finset.range horizon).sum (fun t =>
100           sampledTrajectoryRealizedLossAt t sample) -
101         (Finset.range horizon).sum (fun t =>
102           predictableLossAt loss t sample comparator) <=
103         (horizon : Real) := by
104   dsimp only
105   have hreward :=
106     sampledPredictableTrajectoryMeasure_finiteHorizon_realizedLoss_le_one_ae
107       prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon
108   dsimp only at hreward
109   filter_upwards [hreward] with sample hsample
110   have hrealized :
111       (Finset.range horizon).sum (fun t =>
112           sampledTrajectoryRealizedLossAt t sample) <= (horizon : Real) := by
113     calc
114       (Finset.range horizon).sum (fun t =>
115           sampledTrajectoryRealizedLossAt t sample) <=
116           (Finset.range horizon).sum (fun _t => (1 : Real)) := by
117         apply Finset.sum_le_sum
118         intro t ht
119         exact hsample t (Finset.mem_range.mp ht)
120       _ = (horizon : Real) := by simp
121   have hcomparator :
122       0 <= (Finset.range horizon).sum (fun t =>
123         predictableLossAt loss t sample comparator) := by
124     exact Finset.sum_nonneg fun t _ht =>
125       predictableLossAt_nonneg loss t sample comparator
126   linarith

170 /-- The regime in which the clipped explicit schedule satisfies the Bernstein
171 dominance contracts without activating its clip. -/
172 def bernsteinLargeHorizonCondition (K T delta : Real) : Prop :=
173   8 * (K * Real.log K) <= T ∧
174     8 * (K * Real.log (3 / delta)) <= T ∧
175     8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
176       Real.log (3 / delta) <= T
177 
178 /-- All-horizon threshold: use the explicit Bernstein rate in its valid regime
179 and the strict pathwise horizon fallback otherwise. -/
180 noncomputable def bernsteinAllHorizonRegretThreshold
181     (K T delta : Real) : Real := by
182   classical
183   exact if bernsteinLargeHorizonCondition K T delta then
184     11 * bernsteinClippedExplorationRate K T delta * T
185   else
186     T + 1
