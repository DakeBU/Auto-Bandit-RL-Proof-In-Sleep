4 # Spectral Gaussian mixture identity for OFUL
5 
6 This module transports the compiled independent-coordinate Gaussian
7 quadratic-exponential identity through the orthonormal eigenbasis of a real
8 positive-semidefinite matrix. It then collects the spectral product and
9 coordinate quadratic form as `det (1 + A)` and `(1 + A)⁻¹`.
10 -/

39 private theorem inner_toEuclideanCLM_sum_eigenvectorBasis
40     (A : Matrix Feature Feature Real) (hA : A.PosSemidef)
41     (x : Feature -> Real) :
42     let b := hA.isHermitian.eigenvectorBasis
43     let v : EuclideanSpace Real Feature := ∑ i, x i • b i
44     ⟪v, Matrix.toEuclideanCLM (𝕜 := Real) A v⟫_ℝ =
45       ∑ i, hA.isHermitian.eigenvalues i * x i ^ 2 := by
46   dsimp only
47   let b := hA.isHermitian.eigenvectorBasis
48   let v : EuclideanSpace Real Feature := ∑ i, x i • b i
49   have hsumcoord (y : Feature → Real) (i : Feature) :
50       ⟪b i, ∑ j, y j • b j⟫_ℝ = y i := by
51     calc
52       ⟪b i, ∑ j, y j • b j⟫_ℝ =
53           b.repr (∑ j, y j • b j) i :=
54         (b.repr_apply_apply (∑ j, y j • b j) i).symm
55       _ = y i := by
56         rw [b.sum_repr_symm]
57         simp
58   have hcoord (i : Feature) : ⟪b i, v⟫_ℝ = x i := by
59     exact hsumcoord x i
60   have heigen (i : Feature) :
61       Matrix.toEuclideanCLM (𝕜 := Real) A (b i) =
62         hA.isHermitian.eigenvalues i • b i := by
63     apply WithLp.ofLp_injective
64     simpa only [Matrix.ofLp_toEuclideanCLM, WithLp.ofLp_smul] using
65       hA.isHermitian.mulVec_eigenvectorBasis i
66   have hAv :
67       Matrix.toEuclideanCLM (𝕜 := Real) A v =
68         ∑ i, (hA.isHermitian.eigenvalues i * x i) • b i := by
69     dsimp [v]
70     rw [map_sum]
71     apply Finset.sum_congr rfl
72     intro i hi
73     rw [map_smul, heigen]
74     simp only [smul_smul]
75     ring_nf
76   have hcoordLeft (i : Feature) : ⟪v, b i⟫_ℝ = x i := by
77     rw [real_inner_comm, hcoord]
78   have hcoordA :
79       ∀ i : Feature,
80         ⟪b i, Matrix.toEuclideanCLM (𝕜 := Real) A v⟫_ℝ =
81           hA.isHermitian.eigenvalues i * x i := by
82     intro i
83     rw [hAv]
84     exact hsumcoord (fun j => hA.isHermitian.eigenvalues j * x j) i
85   change
86     ⟪v, Matrix.toEuclideanCLM (𝕜 := Real) A v⟫_ℝ =
87       ∑ i, hA.isHermitian.eigenvalues i * x i ^ 2
88   calc
89     ⟪v, Matrix.toEuclideanCLM (𝕜 := Real) A v⟫_ℝ =
90         ∑ i, ⟪v, b i⟫_ℝ *
91           ⟪b i, Matrix.toEuclideanCLM (𝕜 := Real) A v⟫_ℝ :=
92       (OrthonormalBasis.sum_inner_mul_inner b v
93         (Matrix.toEuclideanCLM (𝕜 := Real) A v)).symm
94     _ = ∑ i, hA.isHermitian.eigenvalues i * x i ^ 2 := by
95       apply Finset.sum_congr rfl
96       intro i hi
97       rw [hcoordLeft, hcoordA]
98       ring

100 /--
101 The standard-Gaussian quadratic-exponential integral in the orthonormal
102 eigenbasis of a positive-semidefinite matrix.
103 
104 This is the explicit spectral-coordinate transport bridge from the diagonal
105 product identity. The following lemmas collect its right-hand side into
106 matrix determinant and inverse-quadratic notation.
107 -/
108 theorem integral_exp_inner_sub_quadratic_stdGaussian_eigenvalues
109     (A : Matrix Feature Feature Real) (hA : A.PosSemidef)
110     (score : EuclideanSpace Real Feature) :
111     integral
112         (ProbabilityTheory.stdGaussian
113           (EuclideanSpace Real Feature))
114         (fun z : EuclideanSpace Real Feature =>
115           Real.exp
116             (⟪score, z⟫_ℝ -
117               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) A z⟫_ℝ / 2)) =
118       (Finset.univ.prod (fun i =>
119         Real.sqrt (1 + hA.isHermitian.eigenvalues i)))⁻¹ *
120         Real.exp
121           (Finset.univ.sum (fun i =>
122             ⟪score, hA.isHermitian.eigenvectorBasis i⟫_ℝ ^ 2 /
123               (2 * (1 + hA.isHermitian.eigenvalues i)))) := by
124   let b := hA.isHermitian.eigenvectorBasis
125   have hgauss :
126       ProbabilityTheory.stdGaussian (EuclideanSpace Real Feature) =
127         (Measure.pi (fun _ : Feature =>
128           ProbabilityTheory.gaussianReal 0 1)).map
129             (fun x : Feature → Real => ∑ i, x i • b i) :=
130     ProbabilityTheory.stdGaussian_eq_map_pi_orthonormalBasis b
131   rw [hgauss]
132   rw [MeasureTheory.integral_map]
133   · have hpoint :
134         (fun x : Feature → Real =>
135           Real.exp
136             (⟪score, ∑ i, x i • b i⟫_ℝ -
137               ⟪∑ i, x i • b i,
138                 Matrix.toEuclideanCLM (𝕜 := Real) A
139                   (∑ i, x i • b i)⟫_ℝ / 2)) =
140           (fun x : Feature → Real =>
141             Real.exp
142               (Finset.univ.sum (fun i =>
143                 ⟪score, b i⟫_ℝ * x i -
144                   hA.isHermitian.eigenvalues i * (x i) ^ 2 / 2))) := by
145         funext x
146         congr 1
147         calc
148           ⟪score, ∑ i, x i • b i⟫_ℝ -
149                 ⟪∑ i, x i • b i,
150                   Matrix.toEuclideanCLM (𝕜 := Real) A
151                     (∑ i, x i • b i)⟫_ℝ / 2 =
152               (∑ i, ⟪score, b i⟫_ℝ * x i) -
153                 (∑ i, hA.isHermitian.eigenvalues i * x i ^ 2) / 2 :=
154             congrArg₂ (fun linear quadratic => linear - quadratic / 2)
155               (inner_sum_smul_orthonormalBasis_left b score x)
156               (inner_toEuclideanCLM_sum_eigenvectorBasis A hA x)
157           _ = Finset.univ.sum (fun i =>
158                 ⟪score, b i⟫_ℝ * x i -
159                   hA.isHermitian.eigenvalues i * (x i) ^ 2 / 2) := by
160             rw [Finset.sum_sub_distrib, Finset.sum_div]
161     rw [hpoint]
162     exact
163       integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal_eq
164         (fun i => ⟪score, b i⟫_ℝ)
165         hA.isHermitian.eigenvalues
166         hA.eigenvalues_nonneg
167   · exact (by fun_prop :
168       Measurable (fun x : Feature → Real => ∑ i, x i • b i)).aemeasurable
169   · fun_prop

171 /--
172 The determinant of `1 + A` is the product of one plus the eigenvalues of a
173 real positive-semidefinite matrix.
174 -/
175 theorem det_one_add_posSemidef_eq_prod_eigenvalues
176     (A : Matrix Feature Feature Real) (hA : A.PosSemidef) :
177     Matrix.det (1 + A) =
178       Finset.univ.prod (fun i =>
179         1 + hA.isHermitian.eigenvalues i) := by
180   let eigenvalues := hA.isHermitian.eigenvalues
181   let U := hA.isHermitian.eigenvectorUnitary
182   change Matrix.det (1 + A) =
183     Finset.univ.prod (fun i => 1 + eigenvalues i)
184   have hspec :
185       A = Unitary.conjStarAlgAut Real (Matrix Feature Feature Real) U
186         (Matrix.diagonal eigenvalues) := by
187     exact hA.isHermitian.spectral_theorem
188   calc
189     Matrix.det (1 + A) =
190         Matrix.det
191           (1 + Unitary.conjStarAlgAut Real
192             (Matrix Feature Feature Real) U
193             (Matrix.diagonal eigenvalues)) := by rw [hspec]
194     _ = Matrix.det
195           (Unitary.conjStarAlgAut Real
196             (Matrix Feature Feature Real) U
197             (1 + Matrix.diagonal eigenvalues)) := by
198       rw [map_add, map_one]
199     _ = Finset.univ.prod (fun i => 1 + eigenvalues i) := by
200       rw [Unitary.conjStarAlgAut_apply, Matrix.det_mul,
201         Matrix.det_mul]
202       have hone_diag :
203           1 + Matrix.diagonal eigenvalues =
204             Matrix.diagonal (fun i => 1 + eigenvalues i) := by
205         ext i j
206         by_cases hij : i = j
207         · subst j
208           simp
209         · simp [hij]
210       rw [hone_diag, Matrix.det_diagonal]
211       have hdetU :
212           Matrix.det (U : Matrix Feature Feature Real) *
213               Matrix.det (star (U : Matrix Feature Feature Real)) =
214             1 := by
215         rw [← Matrix.det_mul]
216         have hunit :
217             (U : Matrix Feature Feature Real) *
218                 star (U : Matrix Feature Feature Real) =
219               1 :=
220           Unitary.coe_mul_star_self U
221         rw [hunit, Matrix.det_one]
222       calc
223         Matrix.det (U : Matrix Feature Feature Real) *
224               (∏ i, (1 + eigenvalues i)) *
225               Matrix.det (star (U : Matrix Feature Feature Real)) =
226             (∏ i, (1 + eigenvalues i)) *
227               (Matrix.det (U : Matrix Feature Feature Real) *
228                 Matrix.det (star (U : Matrix Feature Feature Real))) := by
229           ring
230         _ = ∏ i, (1 + eigenvalues i) := by
231           rw [hdetU, mul_one]

233 private theorem toEuclideanCLM_one_add_posSemidef_inv_eigenvectorBasis
234     (A : Matrix Feature Feature Real) (hA : A.PosSemidef)
235     (i : Feature) :
236     Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹
237         (hA.isHermitian.eigenvectorBasis i) =
238       (1 + hA.isHermitian.eigenvalues i)⁻¹ •
239         hA.isHermitian.eigenvectorBasis i := by
240   let b := hA.isHermitian.eigenvectorBasis
241   let eigenvalue : Real := 1 + hA.isHermitian.eigenvalues i
242   have heigenvalue_pos : 0 < eigenvalue := by
243     dsimp [eigenvalue]
244     linarith [hA.eigenvalues_nonneg i]
245   have hBpos : (1 + A).PosDef :=
246     Matrix.PosDef.one.add_posSemidef hA
247   have hBaction :
248       (1 + A).mulVec (b i) = eigenvalue • b i := by
249     dsimp [b, eigenvalue]
250     rw [Matrix.add_mulVec, Matrix.one_mulVec,
251       hA.isHermitian.mulVec_eigenvectorBasis]
252     ext j
253     simp [Pi.add_apply, Pi.smul_apply]
254     ring
255   have hb :
256       (b i : Feature → Real) =
257         eigenvalue⁻¹ • (1 + A).mulVec (b i) := by
258     rw [hBaction, smul_smul]
259     field_simp
260     simp
261   apply WithLp.ofLp_injective
262   simpa only [Matrix.ofLp_toEuclideanCLM, WithLp.ofLp_smul] using
263     (show
264       (1 + A)⁻¹ *ᵥ (b i : Feature → Real) =
265         eigenvalue⁻¹ • (b i : Feature → Real) by
266       calc
267         (1 + A)⁻¹ *ᵥ (b i : Feature → Real) =
268             (1 + A)⁻¹ *ᵥ
269               (eigenvalue⁻¹ • (1 + A).mulVec (b i)) := by
270           rw [← hb]
271         _ = eigenvalue⁻¹ •
272               ((1 + A)⁻¹ *ᵥ ((1 + A).mulVec (b i))) := by
273           rw [Matrix.mulVec_smul]
274         _ = eigenvalue⁻¹ •
275               (((1 + A)⁻¹ * (1 + A)).mulVec (b i)) := by
276           rw [Matrix.mulVec_mulVec]
277         _ = eigenvalue⁻¹ • (b i : Feature → Real) := by
278           rw [Matrix.nonsing_inv_mul (1 + A)
279               ((1 + A).isUnit_iff_isUnit_det.mp hBpos.isUnit),
280             Matrix.one_mulVec])

282 /--
283 The inverse quadratic form of `1 + A` equals its spectral-coordinate sum.
284 -/
285 theorem dotProduct_one_add_posSemidef_inv_mulVec_eq_sum_eigenvalues
286     (A : Matrix Feature Feature Real) (hA : A.PosSemidef)
287     (score : EuclideanSpace Real Feature) :
288     score ⬝ᵥ (1 + A)⁻¹.mulVec score =
289       Finset.univ.sum (fun i =>
290         ⟪score, hA.isHermitian.eigenvectorBasis i⟫_ℝ ^ 2 /
291           (1 + hA.isHermitian.eigenvalues i)) := by
292   let b := hA.isHermitian.eigenvectorBasis
293   let coefficient : Feature → Real :=
294     fun i => ⟪score, b i⟫_ℝ
295   let inverseEigenvalue : Feature → Real :=
296     fun i => (1 + hA.isHermitian.eigenvalues i)⁻¹
297   have hscore :
298       score = ∑ i, coefficient i • b i := by
299     calc
300       score = ∑ i, ⟪b i, score⟫_ℝ • b i :=
301         (b.sum_repr' score).symm
302       _ = ∑ i, coefficient i • b i := by
303         apply Finset.sum_congr rfl
304         intro i hi
305         congr 1
306         dsimp [coefficient]
307         rw [real_inner_comm]
308   have hinvScore :
309       Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹ score =
310         ∑ i, (coefficient i * inverseEigenvalue i) • b i := by
311     rw [hscore, map_sum]
312     apply Finset.sum_congr rfl
313     intro i hi
314     rw [map_smul,
315       toEuclideanCLM_one_add_posSemidef_inv_eigenvectorBasis A hA i]
316     simp only [smul_smul]
317     dsimp [inverseEigenvalue]
318   have hcoordInv (i : Feature) :
319       ⟪b i,
320         Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹ score⟫_ℝ =
321           coefficient i * inverseEigenvalue i := by
322     rw [hinvScore]
323     exact inner_sum_smul_orthonormalBasis b
324       (fun j => coefficient j * inverseEigenvalue j) i
325   rw [← Matrix.inner_toEuclideanCLM]
326   calc
327     ⟪score,
328         Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹ score⟫_ℝ =
329       ∑ i, ⟪score, b i⟫_ℝ *
330         ⟪b i,
331           Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹ score⟫_ℝ :=
332       (OrthonormalBasis.sum_inner_mul_inner b score
333         (Matrix.toEuclideanCLM (𝕜 := Real) (1 + A)⁻¹ score)).symm
334     _ = Finset.univ.sum (fun i =>
335         ⟪score, hA.isHermitian.eigenvectorBasis i⟫_ℝ ^ 2 /
336           (1 + hA.isHermitian.eigenvalues i)) := by
337       apply Finset.sum_congr rfl
338       intro i hi
339       rw [hcoordInv]
340       dsimp [b, coefficient, inverseEigenvalue]
341       ring

343 /--
344 Exact standard-Gaussian quadratic-exponential identity for an arbitrary real
345 positive-semidefinite matrix.
346 
347 This closes the orthonormal spectral transport from the diagonal-coordinate
348 Gaussian mixture leaf. Transport from a nonstandard initial covariance and
349 the stochastic Tonelli/Markov assembly remain separate obligations.
350 -/
351 theorem integral_exp_inner_sub_quadratic_stdGaussian_det
352     (A : Matrix Feature Feature Real) (hA : A.PosSemidef)
353     (score : EuclideanSpace Real Feature) :
354     integral
355         (ProbabilityTheory.stdGaussian
356           (EuclideanSpace Real Feature))
357         (fun z : EuclideanSpace Real Feature =>
358           Real.exp
359             (⟪score, z⟫_ℝ -
360               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) A z⟫_ℝ / 2)) =
361       (Real.sqrt (Matrix.det (1 + A)))⁻¹ *
362         Real.exp
363           (score ⬝ᵥ (1 + A)⁻¹.mulVec score / 2) := by
364   rw [integral_exp_inner_sub_quadratic_stdGaussian_eigenvalues
365     A hA score]
366   have heigenvalue_nonneg :
367       ∀ i : Feature, 0 ≤ 1 + hA.isHermitian.eigenvalues i := by
368     intro i
369     linarith [hA.eigenvalues_nonneg i]
370   have heigenvalue_ne :
371       ∀ i : Feature, 1 + hA.isHermitian.eigenvalues i ≠ 0 := by
372     intro i
373     exact ne_of_gt (by linarith [hA.eigenvalues_nonneg i])
374   have hsqrtDet :
375       Real.sqrt (Matrix.det (1 + A)) =
376         Finset.univ.prod (fun i =>
377           Real.sqrt (1 + hA.isHermitian.eigenvalues i)) := by
378     rw [det_one_add_posSemidef_eq_prod_eigenvalues A hA]
379     exact Real.sqrt_prod Finset.univ
380       (fun i hi => heigenvalue_nonneg i)
381   have hquadratic :=
382     dotProduct_one_add_posSemidef_inv_mulVec_eq_sum_eigenvalues
383       A hA score
384   have hsum :
385       Finset.univ.sum (fun i =>
386           ⟪score, hA.isHermitian.eigenvectorBasis i⟫_ℝ ^ 2 /
387             (2 * (1 + hA.isHermitian.eigenvalues i))) =
388         Finset.univ.sum (fun i =>
389           ⟪score, hA.isHermitian.eigenvectorBasis i⟫_ℝ ^ 2 /
390             (1 + hA.isHermitian.eigenvalues i)) / 2 := by
391     rw [Finset.sum_div]
392     apply Finset.sum_congr rfl
393     intro i hi
394     field_simp [heigenvalue_ne i]
395   rw [hsqrtDet, hquadratic, hsum]
