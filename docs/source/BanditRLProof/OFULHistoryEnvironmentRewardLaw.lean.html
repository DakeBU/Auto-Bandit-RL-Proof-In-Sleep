20 /--
21 For any canonical history algorithm, the time-zero reward marginal is the
22 environment feedback law at that algorithm's initial action.
23 -/
24 theorem
25     canonicalHistoryTrajectory_initialReward_map_eq_historyAlgorithmInitialFeedback
26     {K : Nat}
27     (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
28     (initialAction : Fin K)
29     (hinitial : algorithm.initialAction = Measure.dirac initialAction)
30     (environment : Thompson.HistoryEnvironment (Fin K) Real) :
31     Measure.map (fun trajectory : Nat -> Fin K × Real => trajectory 0 |>.2)
32         (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment) =
33       environment.initialFeedback initialAction := by
34   calc
35     Measure.map (fun trajectory : Nat -> Fin K × Real => trajectory 0 |>.2)
36         (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment) =
37         Measure.snd
38           (Measure.map (fun trajectory : Nat -> Fin K × Real => trajectory 0)
39             (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment)) := by
40           rw [Measure.snd]
41           symm
42           simpa [Function.comp_def] using
43             (Measure.map_map measurable_snd (measurable_pi_apply 0))
44     _ = Measure.snd (algorithm.initialAction ⊗ₘ environment.initialFeedback) := by
45       congr 1
46       simpa [Thompson.canonicalHistoryTrajectoryAction,
47         Thompson.canonicalHistoryTrajectoryReward] using
48         (Thompson.canonicalHistoryTrajectory_initialPair_map_eq
49           algorithm environment)
50     _ = environment.initialFeedback initialAction := by
51       rw [hinitial]
52       rw [Measure.snd_compProd]
53       ext s hs
54       rw [Measure.bind_apply hs environment.initialFeedback.aemeasurable,
55         lintegral_dirac]

57 /--
58 For any canonical history algorithm, the successor reward conditioned on its
59 finite pair prefix has the reward marginal of the history-step kernel.
60 -/
61 theorem canonicalHistoryTrajectory_reward_succ_condDistrib_eq_historyStepReward
62     {K : Nat}
63     (hK : 0 < K)
64     (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
65     (environment : Thompson.HistoryEnvironment (Fin K) Real)
66     (n : Nat) :
67     ProbabilityTheory.condDistrib
68         (fun trajectory : Nat -> Fin K × Real => (trajectory (n + 1)).2)
69         (Preorder.frestrictLe n)
70         (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment) =ᵐ[
71       (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment).map
72         (Preorder.frestrictLe n)]
73       (Thompson.historyStepKernel algorithm environment n).map Prod.snd := by
74   letI : Nonempty (Fin K) := ⟨⟨0, hK⟩⟩
75   let mu := Thompson.canonicalHistoryTrajectoryMeasure algorithm environment
76   let nextPair : (Nat -> Fin K × Real) -> Fin K × Real :=
77     fun trajectory => trajectory (n + 1)
78   let history : (Nat -> Fin K × Real) ->
79       History.FinitePairHistory (Fin K) Real n :=
80     Preorder.frestrictLe n
81   have hcomp :
82       ProbabilityTheory.condDistrib (Prod.snd ∘ nextPair) history mu =ᵐ[
83         mu.map history]
84         (ProbabilityTheory.condDistrib nextPair history mu).map Prod.snd :=
85     ProbabilityTheory.condDistrib_comp
86       (μ := mu) (mβ := inferInstance) (Y := nextPair) history
87       (measurable_pi_apply (n + 1)).aemeasurable
88       (f := Prod.snd) measurable_snd
89   have hpair :
90       ProbabilityTheory.condDistrib nextPair history mu =ᵐ[mu.map history]
91         Thompson.historyStepKernel algorithm environment n := by
92     simpa [mu, nextPair, history,
93       Thompson.canonicalHistoryTrajectoryAction,
94       Thompson.canonicalHistoryTrajectoryReward] using
95       (Thompson.canonicalHistoryTrajectory_step_condDistrib
96         algorithm environment n)
97   have hresult :
98       ProbabilityTheory.condDistrib (Prod.snd ∘ nextPair) history mu =ᵐ[
99         mu.map history]
100         (Thompson.historyStepKernel algorithm environment n).map Prod.snd := by
101     filter_upwards [hcomp, hpair] with historyValue hcompAt hpairAt
102     calc
103       ProbabilityTheory.condDistrib (Prod.snd ∘ nextPair) history mu
104           historyValue =
105           ((ProbabilityTheory.condDistrib nextPair history mu).map Prod.snd)
106             historyValue := hcompAt
107       _ = ((Thompson.historyStepKernel algorithm environment n).map Prod.snd)
108           historyValue := by
109             simpa [ProbabilityTheory.Kernel.map_apply _ measurable_snd] using
110               congrArg (Measure.map Prod.snd) hpairAt
111   simpa [mu, nextPair, history, Function.comp_def] using hresult

172     {K : Nat}

181               (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment)
182               ((inferInstance : MeasurableSpace Unit).comap
183                 (fun _trajectory : Nat -> Fin K × Real => ()))
184               trajectory) =
185           environment.initialFeedback initialAction)
186       (ae ((Thompson.canonicalHistoryTrajectoryMeasure algorithm environment).trim
187         ((show Measurable
188             (fun _trajectory : Nat -> Fin K × Real => ())

198     dsimp [Q]
199     infer_instance
200   letI : ProbabilityTheory.IsMarkovKernel kernel := by
201     dsimp [kernel]
202     infer_instance

204     simpa [mu, reward0, Q] using
205       canonicalHistoryTrajectory_initialReward_map_eq_historyAlgorithmInitialFeedback
206         algorithm initialAction hinitial environment
207   have hcondition_map : Measure.map condition mu = Measure.dirac () := by
208     simp [condition]
209   have hjoint :
210       Measure.map (fun trajectory => (condition trajectory, reward0 trajectory)) mu =
211         Measure.map condition mu ⊗ₘ kernel := by
212     calc
213       Measure.map (fun trajectory => (condition trajectory, reward0 trajectory)) mu =
214           Measure.map (Prod.mk ())
215             (Measure.map reward0 mu) := by
216               symm
217               simpa [condition, reward0, Function.comp_def] using
218                 (Measure.map_map measurable_prodMk_left
219                   (measurable_snd.comp (measurable_pi_apply 0)))
220       _ = Measure.map (Prod.mk ()) Q := by rw [hreward]
221       _ = Measure.dirac () ⊗ₘ kernel := by
222         simpa [kernel] using (Measure.dirac_unit_compProd_const Q).symm
223       _ = Measure.map condition mu ⊗ₘ kernel := by rw [hcondition_map]
224   have hcond :
225       ProbabilityTheory.condDistrib reward0 condition mu =ᵐ[mu.map condition]
226         kernel :=
227     ProbabilityTheory.condDistrib_ae_eq_of_measure_eq_compProd
228       condition
229       (measurable_snd.comp (measurable_pi_apply 0)).aemeasurable
230       hjoint
231   have hbridge :=
232     ConditionalExpectationReward.condExpKernel_map_eq_of_condDistrib_ae_eq_real_trim
233       mu reward0 condition
234       (measurable_snd.comp (measurable_pi_apply 0))
235       measurable_const kernel hcond
236   simpa only [mu, reward0, condition, kernel, Q,
237     ProbabilityTheory.Kernel.const_apply] using hbridge

239 /--
240 For the deterministic finite-history OFUL policy, the reward marginal of the
241 next pair kernel is exactly the environment feedback kernel selected by the
242 current history.
243 -/
244 theorem finiteHistoryScalarRidge_historyStepKernel_map_snd
245     {K : Nat} {Feature : Type u}
246     [Fintype Feature] [DecidableEq Feature]
247     (hK : 0 < K)
248     (lambda : Real)
249     (actionFeature : Fin K -> Feature -> Real)
250     (R algorithmDelta S : Real)
251     (environment : Thompson.HistoryEnvironment (Fin K) Real)
252     (n : Nat) (history : History.FinitePairHistory (Fin K) Real n) :
253     (Thompson.historyStepKernel
254         (finiteHistoryScalarRidgeOptimisticAlgorithm
255           hK lambda actionFeature R algorithmDelta S)
256         environment n).map Prod.snd history =
257       environment.feedback n
258         (history,
259           finiteHistoryScalarRidgeOptimisticAction
260             hK lambda actionFeature R algorithmDelta S n history) := by
261   ext s hs
262   rw [ProbabilityTheory.Kernel.map_apply _ measurable_snd,
263     Measure.map_apply measurable_snd hs]
264   unfold Thompson.historyStepKernel
265   rw [
266     ProbabilityTheory.Kernel.compProd_apply (measurable_snd hs)]
267   simp only [finiteHistoryScalarRidgeOptimisticAlgorithm,
268     finiteHistoryScalarRidgeOptimisticAction,
269     finiteHistoryOptimisticAlgorithm_policy_apply]
270   rw [lintegral_dirac]
271   let selected :=
272     finiteHistoryOptimisticAction hK
273       (finiteHistoryScalarRidgeEstimate lambda actionFeature)
274       (finiteHistoryScalarRidgeDesign lambda actionFeature)
275       (finiteHistoryScalarRidgeRadius
276         actionFeature R algorithmDelta lambda S)
277       (finiteHistoryFixedActionFeature actionFeature) n history
278   change environment.feedback n (history, selected)
279       (Prod.mk selected ⁻¹' (Prod.snd ⁻¹' s)) =
280     environment.feedback n (history, selected) s
281   have hpreimage :
282       Prod.mk selected ⁻¹' (Prod.snd ⁻¹' s) = s := by
283     ext reward
284     rfl
285   rw [hpreimage]
