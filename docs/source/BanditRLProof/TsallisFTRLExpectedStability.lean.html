5 # Expected finite-horizon half-Tsallis stability
6 
7 This module sums the one-round conditional half-Tsallis stability theorem over
8 a finite horizon under a common ambient trajectory measure.  Conditional-law
9 identification is used both for the one-round inequality and for transporting
10 product-law integrability back to each realized history/action score.
11 
12 The theorem is stated for a finite measure as an integral inequality.  Under
13 a probability measure, this is the usual expected stability bound.
14 
15 The canonical endpoint still exposes the Markov policy/law identification and
16 measurability/integrability of the updated stability score.  No measurability
17 property of `Classical.choose` is inferred.
18 -/

29 theorem integrable_importanceWeightedStabilityScore_comp_history_action_of_condDistrib
30     {Omega : Type u} {History : Type v} {Action : Type w}
31     [MeasurableSpace Omega] [MeasurableSpace History]
32     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
33     (mu : Measure Omega) [IsFiniteMeasure mu]
34     (history : Omega -> History) (hhistory : Measurable history)
35     (action : Omega -> Action) (haction : Measurable action)
36     (arms : Finset Action)
37     (prob loss : History -> Action -> Real)
38     (next : History -> Action -> Action -> Real)
39     (policy : Kernel History Action) [IsMarkovKernel policy]
40     (hcond : condDistrib action history mu =ᵐ[mu.map history] policy)
41     (hIntegrable : Integrable
42       (importanceWeightedStabilityScore arms prob loss next)
43       (mu.map history ⊗ₘ policy)) :
44     Integrable (fun omega =>
45       importanceWeightedStabilityScore arms prob loss next
46         (history omega, action omega)) mu := by
47   have hpair :
48       mu.map (fun omega => (history omega, action omega)) =
49         mu.map history ⊗ₘ policy :=
50     (condDistrib_ae_eq_iff_measure_eq_compProd history
51       haction.aemeasurable policy).mp hcond
52   have hmap : Integrable
53       (importanceWeightedStabilityScore arms prob loss next)
54       (mu.map (fun omega => (history omega, action omega))) := by
55     rwa [hpair]
56   simpa [Function.comp_def] using
57     hmap.comp_measurable (hhistory.prodMk haction)

73 /--
74 Expected finite-horizon half-Tsallis stability under identified conditional
75 action laws and explicit current/update minimizer certificates.
76 
77 All rounds live on one ambient measure `mu`.  This is the theorem-level bridge
78 from the one-round sampling-law average to the expected finite stability sum.
79 -/
80 theorem integral_sum_importanceWeightedStabilityScore_le_integral_sum_halfPowerStabilityBound_of_condDistrib_of_minimizers
81     {Omega : Type u} {History : Type v} {Action : Type w}
82     [MeasurableSpace Omega] [MeasurableSpace History]
83     [MeasurableSpace Action] [MeasurableSingletonClass Action]
84     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
85     (mu : Measure Omega) [IsFiniteMeasure mu]
86     (horizon : Nat) (arms : Finset Action) (eta : Real)
87     (history : Nat -> Omega -> History)
88     (action : Nat -> Omega -> Action)
89     (score prob loss : Nat -> History -> Action -> Real)
90     (next : Nat -> History -> Action -> Action -> Real)
91     (policy : Nat -> Kernel History Action)
92     (hmarkov : forall t, IsMarkovKernel (policy t))
93     (hhistory : forall t, Measurable (history t))
94     (haction : forall t, Measurable (action t))
95     (hpolicy : forall t, policy t =ᵐ[mu.map (history t)]
96       fun h => Exp3.finiteActionMeasure arms (prob t h))
97     (hcond : forall t,
98       condDistrib (action t) (history t) mu =ᵐ[mu.map (history t)] policy t)
99     (heta : 0 < eta)
100     (hprobMin : forall t h,
101       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
102         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
103         (score t h) (prob t h))
104     (hnextMin : forall t h chosen, chosen ∈ arms ->
105       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
106         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
107         (fun candidate => score t h candidate +
108           Exp3.importanceWeightedLoss
109             (prob t h) (loss t h) chosen candidate)
110         (next t h chosen))
111     (hloss : forall t h candidate, candidate ∈ arms ->
112       0 <= loss t h candidate ∧ loss t h candidate <= 1)
113     (hscore : forall t, Measurable
114       (importanceWeightedStabilityScore arms (prob t) (loss t) (next t)))
115     (hIntegrable : forall t, Integrable
116       (importanceWeightedStabilityScore arms (prob t) (loss t) (next t))
117       (mu.map (history t) ⊗ₘ policy t))
118     (hboundIntegrable : forall t, Integrable
119       (halfPowerStabilityBound arms eta (prob t)) (mu.map (history t))) :
120     integral mu (fun omega => (Finset.range horizon).sum (fun t =>
121         importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
122           (history t omega, action t omega))) <=
123       integral mu (fun omega => (Finset.range horizon).sum (fun t =>
124         halfPowerStabilityBound arms eta (prob t) (history t omega))) := by
125   have hscoreComp : forall t, Integrable (fun omega =>
126       importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
127         (history t omega, action t omega)) mu := by
128     intro t
129     letI : IsMarkovKernel (policy t) := hmarkov t
130     exact
131       integrable_importanceWeightedStabilityScore_comp_history_action_of_condDistrib
132         mu (history t) (hhistory t) (action t) (haction t)
133         arms (prob t) (loss t) (next t) (policy t) (hcond t)
134         (hIntegrable t)
135   have hboundComp : forall t, Integrable (fun omega =>
136       halfPowerStabilityBound arms eta (prob t) (history t omega)) mu := by
137     intro t
138     exact integrable_halfPowerStabilityBound_comp_history
139       mu (history t) (hhistory t) arms eta (prob t) (hboundIntegrable t)
140   have hround : forall t,
141       integral mu (fun omega =>
142           importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
143             (history t omega, action t omega)) <=
144         integral mu (fun omega =>
145           halfPowerStabilityBound arms eta (prob t) (history t omega)) := by
146     intro t
147     letI : IsMarkovKernel (policy t) := hmarkov t
148     have h :=
149       integral_importanceWeightedStabilityScore_le_integral_halfPowerStabilityBound_of_condDistrib_of_minimizers
150         mu (history t) (hhistory t) (action t) (haction t)
151         arms eta (score t) (prob t) (loss t) (next t) (policy t)
152         (hpolicy t) (hcond t) heta (hprobMin t) (hnextMin t)
153         (hloss t) (hscore t) (hIntegrable t) (hboundIntegrable t)
154     calc
155       integral mu (fun omega =>
156           importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
157             (history t omega, action t omega)) <=
158           integral (mu.map (history t))
159             (halfPowerStabilityBound arms eta (prob t)) := h
160       _ = integral mu (fun omega =>
161           halfPowerStabilityBound arms eta (prob t) (history t omega)) := by
162         rw [integral_map (hhistory t).aemeasurable
163           (hboundIntegrable t).aestronglyMeasurable]
164   calc
165     integral mu (fun omega => (Finset.range horizon).sum (fun t =>
166         importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
167           (history t omega, action t omega))) =
168       (Finset.range horizon).sum (fun t => integral mu (fun omega =>
169         importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
170           (history t omega, action t omega))) := by
171         exact ExpectationBochnerSums.integral_finset_sum mu
172           (Finset.range horizon) (fun t omega =>
173             importanceWeightedStabilityScore arms (prob t) (loss t) (next t)
174               (history t omega, action t omega))
175           (fun t _ => hscoreComp t)
176     _ <= (Finset.range horizon).sum (fun t => integral mu (fun omega =>
177         halfPowerStabilityBound arms eta (prob t) (history t omega))) := by
178       exact Finset.sum_le_sum fun t _ => hround t
179     _ = integral mu (fun omega => (Finset.range horizon).sum (fun t =>
180         halfPowerStabilityBound arms eta (prob t) (history t omega)) ) := by
181       symm
182       exact ExpectationBochnerSums.integral_finset_sum mu
183         (Finset.range horizon) (fun t omega =>
184           halfPowerStabilityBound arms eta (prob t) (history t omega))
185         (fun t _ => hboundComp t)

187 /-- Canonical half-Tsallis finite-horizon expected stability theorem.  The
188 current and updated minimizer certificates are internal; selector and score
189 regularity remain explicit. -/
190 theorem integral_sum_halfTsallisHistoryStability_le_integral_sum_halfPowerStabilityBound
191     {Omega : Type u} {History : Type v} {Action : Type w}
192     [MeasurableSpace Omega] [MeasurableSpace History]
193     [MeasurableSpace Action] [MeasurableSingletonClass Action]
194     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
195     (mu : Measure Omega) [IsFiniteMeasure mu]
196     (horizon : Nat) (arms : Finset Action) (harms : arms.Nonempty)
197     (eta : Real) (history : Nat -> Omega -> History)
198     (action : Nat -> Omega -> Action)
199     (score loss : Nat -> History -> Action -> Real)
200     (policy : Nat -> Kernel History Action)
201     (hmarkov : forall t, IsMarkovKernel (policy t))
202     (hhistory : forall t, Measurable (history t))
203     (haction : forall t, Measurable (action t))
204     (hpolicy : forall t, policy t =ᵐ[mu.map (history t)] fun h =>
205       Exp3.finiteActionMeasure arms
206         (halfTsallisHistoryMinimizer arms harms eta (score t) h))
207     (hcond : forall t,
208       condDistrib (action t) (history t) mu =ᵐ[mu.map (history t)] policy t)
209     (heta : 0 < eta)
210     (hloss : forall t h candidate, candidate ∈ arms ->
211       0 <= loss t h candidate ∧ loss t h candidate <= 1)
212     (hscore : forall t, Measurable
213       (importanceWeightedStabilityScore arms
214         (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
215         (halfTsallisHistoryUpdatedMinimizer
216           arms harms eta (score t) (loss t))))
217     (hIntegrable : forall t, Integrable
218       (importanceWeightedStabilityScore arms
219         (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
220         (halfTsallisHistoryUpdatedMinimizer
221           arms harms eta (score t) (loss t)))
222       (mu.map (history t) ⊗ₘ policy t))
223     (hboundIntegrable : forall t, Integrable
224       (halfPowerStabilityBound arms eta
225         (halfTsallisHistoryMinimizer arms harms eta (score t)))
226       (mu.map (history t))) :
227     integral mu (fun omega => (Finset.range horizon).sum (fun t =>
228         importanceWeightedStabilityScore arms
229           (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
230           (halfTsallisHistoryUpdatedMinimizer
231             arms harms eta (score t) (loss t))
232           (history t omega, action t omega))) <=
233       integral mu (fun omega => (Finset.range horizon).sum (fun t =>
234         halfPowerStabilityBound arms eta
235           (halfTsallisHistoryMinimizer arms harms eta (score t))
236           (history t omega))) := by
237   exact
238     integral_sum_importanceWeightedStabilityScore_le_integral_sum_halfPowerStabilityBound_of_condDistrib_of_minimizers
239       mu horizon arms eta history action score
240       (fun t => halfTsallisHistoryMinimizer arms harms eta (score t))
241       loss
242       (fun t => halfTsallisHistoryUpdatedMinimizer
243         arms harms eta (score t) (loss t))
244       policy hmarkov hhistory haction hpolicy hcond heta
245       (fun t h => halfTsallisMinimizer_isRegularizedMinimizer
246         arms harms eta (score t h))
247       (fun t h chosen _ =>
248         halfTsallisUpdatedMinimizer_isRegularizedMinimizer
249           arms harms eta (score t h) (loss t h) chosen)
250       hloss hscore hIntegrable hboundIntegrable

252 /-- The realized FTRL stability term using the next round's current
253 half-Tsallis selector. -/
254 noncomputable def halfTsallisSuccessorStabilityScore
255     {Omega : Type u} {History : Type v} {Action : Type w}
256     [DecidableEq Action]
257     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
258     (history : Nat -> Omega -> History) (action : Nat -> Omega -> Action)
259     (score loss : Nat -> History -> Action -> Real)
260     (t : Nat) (omega : Omega) : Real :=
261   let prob := halfTsallisHistoryMinimizer
262     arms harms eta (score t) (history t omega)
263   let estimatedLoss := Exp3.importanceWeightedLoss prob
264     (loss t (history t omega)) (action t omega)
265   FTRL.linearLoss arms prob estimatedLoss -
266     FTRL.linearLoss arms
267       (halfTsallisHistoryMinimizer
268         arms harms eta (score (t + 1)) (history (t + 1) omega))
269       estimatedLoss

271 /--
272 Expected finite-horizon bound for the actual successor stability sum.
273 
274 The score recursion identifies the next round's current selector with the
275 importance-weighted updated selector from the current round.  This closes the
276 action-dependent successor alignment without claiming it pathwise absent the
277 explicit recursion contract.
278 -/
279 theorem integral_sum_halfTsallisSuccessorStability_le_integral_sum_halfPowerStabilityBound_of_score_succ
280     {Omega : Type u} {History : Type v} {Action : Type w}
281     [MeasurableSpace Omega] [MeasurableSpace History]
282     [MeasurableSpace Action] [MeasurableSingletonClass Action]
283     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
284     (mu : Measure Omega) [IsFiniteMeasure mu]
285     (horizon : Nat) (arms : Finset Action) (harms : arms.Nonempty)
286     (eta : Real) (history : Nat -> Omega -> History)
287     (action : Nat -> Omega -> Action)
288     (score loss : Nat -> History -> Action -> Real)
289     (policy : Nat -> Kernel History Action)
290     (hmarkov : forall t, IsMarkovKernel (policy t))
291     (hhistory : forall t, Measurable (history t))
292     (haction : forall t, Measurable (action t))
293     (hpolicy : forall t, policy t =ᵐ[mu.map (history t)] fun h =>
294       Exp3.finiteActionMeasure arms
295         (halfTsallisHistoryMinimizer arms harms eta (score t) h))
296     (hcond : forall t,
297       condDistrib (action t) (history t) mu =ᵐ[mu.map (history t)] policy t)
298     (heta : 0 < eta)
299     (hloss : forall t h candidate, candidate ∈ arms ->
300       0 <= loss t h candidate ∧ loss t h candidate <= 1)
301     (hscoreSucc : forall t omega,
302       score (t + 1) (history (t + 1) omega) = fun candidate =>
303         score t (history t omega) candidate +
304           Exp3.importanceWeightedLoss
305             (halfTsallisHistoryMinimizer
306               arms harms eta (score t) (history t omega))
307             (loss t (history t omega)) (action t omega) candidate)
308     (hscore : forall t, Measurable
309       (importanceWeightedStabilityScore arms
310         (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
311         (halfTsallisHistoryUpdatedMinimizer
312           arms harms eta (score t) (loss t))))
313     (hIntegrable : forall t, Integrable
314       (importanceWeightedStabilityScore arms
315         (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
316         (halfTsallisHistoryUpdatedMinimizer
317           arms harms eta (score t) (loss t)))
318       (mu.map (history t) ⊗ₘ policy t))
319     (hboundIntegrable : forall t, Integrable
320       (halfPowerStabilityBound arms eta
321         (halfTsallisHistoryMinimizer arms harms eta (score t)))
322       (mu.map (history t))) :
323     integral mu (fun omega => (Finset.range horizon).sum (fun t =>
324         halfTsallisSuccessorStabilityScore
325           arms harms eta history action score loss t omega)) <=
326       integral mu (fun omega => (Finset.range horizon).sum (fun t =>
327         halfPowerStabilityBound arms eta
328           (halfTsallisHistoryMinimizer arms harms eta (score t))
329           (history t omega))) := by
330   have hbase :=
331     integral_sum_halfTsallisHistoryStability_le_integral_sum_halfPowerStabilityBound
332       mu horizon arms harms eta history action score loss policy
333       hmarkov hhistory haction hpolicy hcond heta hloss
334       hscore hIntegrable hboundIntegrable
335   calc
336     integral mu (fun omega => (Finset.range horizon).sum (fun t =>
337         halfTsallisSuccessorStabilityScore
338           arms harms eta history action score loss t omega)) =
339       integral mu (fun omega => (Finset.range horizon).sum (fun t =>
340         importanceWeightedStabilityScore arms
341           (halfTsallisHistoryMinimizer arms harms eta (score t)) (loss t)
342           (halfTsallisHistoryUpdatedMinimizer
343             arms harms eta (score t) (loss t))
344           (history t omega, action t omega))) := by
345       apply integral_congr_ae
346       filter_upwards [] with omega
347       apply Finset.sum_congr rfl
348       intro t _ht
349       have hnext :
350           halfTsallisHistoryMinimizer
351               arms harms eta (score (t + 1)) (history (t + 1) omega) =
352             halfTsallisHistoryUpdatedMinimizer
353               arms harms eta (score t) (loss t)
354                 (history t omega) (action t omega) := by
355         simp only [halfTsallisHistoryMinimizer,
356           halfTsallisHistoryUpdatedMinimizer, halfTsallisUpdatedMinimizer]
357         rw [hscoreSucc t omega]
358         simp only [halfTsallisHistoryMinimizer]
359       simp only [halfTsallisSuccessorStabilityScore,
360         importanceWeightedStabilityScore]
361       rw [hnext]
362     _ <= integral mu (fun omega => (Finset.range horizon).sum (fun t =>
363         halfPowerStabilityBound arms eta
364           (halfTsallisHistoryMinimizer arms harms eta (score t))
365           (history t omega))) := hbase
