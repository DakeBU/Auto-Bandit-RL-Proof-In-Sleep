9 # Refined half-Tsallis importance-weighted moments
10 
11 This module isolates the finite-sum calculation behind the ordinary
12 importance-weighted part of the refined half-Tsallis stability route.  The
13 baseline is the sampled raw loss.  The quadratic moment contracts to
14 `sum_a sqrt (p a) * (1 - p a)`, while the positive cubic remainder is at
15 most one.
16 
17 No potential, Hessian-comparison, conditional-expectation, or trajectory-law
18 statement is proved here.
19 -/

26 /-- Inverse-half-Tsallis-Hessian quadratic moment after subtracting a baseline. -/
27 noncomputable def shiftedHalfPowerImportanceWeightedMoment
28     {Action : Type u}
29     (arms : Finset Action) (prob loss : Action -> Real)
30     (chosen : Action) : Real :=
31   arms.sum (fun action =>
32     Real.sqrt (prob action) * prob action *
33       (Exp3.importanceWeightedLoss prob loss chosen action - loss chosen) ^ 2)
34 
35 /-- Positive cubic remainder used by the refined Taylor bound. -/
36 noncomputable def shiftedPositiveCubicImportanceWeightedMoment
37     {Action : Type u}
38     (arms : Finset Action) (prob loss : Action -> Real)
39     (chosen : Action) : Real :=
40   arms.sum (fun action =>
41     (prob action) ^ 2 *
42       (max (loss chosen -
43         Exp3.importanceWeightedLoss prob loss chosen action) 0) ^ 3)

45 /-- Reindex a weighted sum over complements using simplex normalization. -/
46 theorem sum_mul_sum_erase_eq_sum_mul_one_sub
47     {Action : Type u} [DecidableEq Action]
48     (arms : Finset Action) (prob weight : Action -> Real)
49     (hsum : arms.sum prob = 1) :
50     arms.sum (fun chosen =>
51         prob chosen * (arms.erase chosen).sum weight) =
52       arms.sum (fun action => weight action * (1 - prob action)) := by
53   calc
54     arms.sum (fun chosen =>
55         prob chosen * (arms.erase chosen).sum weight) =
56         arms.sum (fun chosen =>
57           prob chosen * (arms.sum weight - weight chosen)) := by
58       apply Finset.sum_congr rfl
59       intro chosen hchosen
60       rw [show (arms.erase chosen).sum weight =
61         arms.sum weight - weight chosen by
62           rw [← Finset.sum_erase_add _ _ hchosen]
63           ring]
64     _ = arms.sum (fun chosen =>
65         prob chosen * arms.sum weight - prob chosen * weight chosen) := by
66       apply Finset.sum_congr rfl
67       intro chosen _hchosen
68       ring
69     _ = arms.sum prob * arms.sum weight -
70         arms.sum (fun action => prob action * weight action) := by
71       rw [Finset.sum_sub_distrib, Finset.sum_mul]
72     _ = arms.sum weight -
73         arms.sum (fun action => weight action * prob action) := by
74       rw [hsum]
75       simp only [one_mul]
76       congr 1
77       apply Finset.sum_congr rfl
78       intro action _haction
79       ring
80     _ = arms.sum (fun action =>
81         weight action - weight action * prob action) := by
82       rw [Finset.sum_sub_distrib]
83     _ = arms.sum (fun action => weight action * (1 - prob action)) := by
84       apply Finset.sum_congr rfl
85       intro action _haction
86       ring

88 /-- Exact sampled-action expansion of the shifted quadratic moment. -/
89 theorem prob_mul_shiftedHalfPowerImportanceWeightedMoment_eq
90     {Action : Type u} [DecidableEq Action]
91     (arms : Finset Action) (prob loss : Action -> Real)
92     {chosen : Action} (hchosen : chosen ∈ arms)
93     (hprob : 0 < prob chosen) :
94     prob chosen *
95         shiftedHalfPowerImportanceWeightedMoment arms prob loss chosen =
96       (loss chosen) ^ 2 *
97         (Real.sqrt (prob chosen) * (1 - prob chosen) ^ 2 +
98           prob chosen * (arms.erase chosen).sum (fun action =>
99             Real.sqrt (prob action) * prob action)) := by
100   unfold shiftedHalfPowerImportanceWeightedMoment
101   rw [Finset.mul_sum]
102   rw [← Finset.sum_erase_add _ _ hchosen]
103   have hchosenTerm :
104       prob chosen *
105           (Real.sqrt (prob chosen) * prob chosen *
106             (Exp3.importanceWeightedLoss prob loss chosen chosen -
107               loss chosen) ^ 2) =
108         (loss chosen) ^ 2 *
109           (Real.sqrt (prob chosen) * (1 - prob chosen) ^ 2) := by
110     simp only [Exp3.importanceWeightedLoss, if_true]
111     field_simp [ne_of_gt hprob]
112   rw [hchosenTerm]
113   have herase :
114       (arms.erase chosen).sum (fun action =>
115           prob chosen *
116             (Real.sqrt (prob action) * prob action *
117               (Exp3.importanceWeightedLoss prob loss chosen action -
118                 loss chosen) ^ 2)) =
119         (loss chosen) ^ 2 *
120           (prob chosen * (arms.erase chosen).sum (fun action =>
121             Real.sqrt (prob action) * prob action)) := by
122     rw [Finset.mul_sum, Finset.mul_sum]
123     apply Finset.sum_congr rfl
124     intro action haction
125     have hne : chosen ≠ action :=
126       Ne.symm (Finset.ne_of_mem_erase haction)
127     simp [Exp3.importanceWeightedLoss, hne]
128     ring
129   rw [herase]
130   ring

132 /--
133 The sampled shifted quadratic IW moment is bounded by the refined all-arm
134 half-power mass.
135 -/
136 theorem sum_prob_mul_shiftedHalfPowerImportanceWeightedMoment_le
137     {Action : Type u} [DecidableEq Action]
138     (arms : Finset Action) (prob loss : Action -> Real)
139     (hprobability : FTRL.finiteSimplex arms prob)
140     (hprob : forall action, action ∈ arms -> 0 < prob action)
141     (hloss : forall action, action ∈ arms ->
142       0 <= loss action ∧ loss action <= 1) :
143     arms.sum (fun chosen =>
144         prob chosen *
145           shiftedHalfPowerImportanceWeightedMoment arms prob loss chosen) <=
146       arms.sum (fun action =>
147         Real.sqrt (prob action) * (1 - prob action)) := by
148   rw [Finset.sum_congr rfl (fun chosen hchosen =>
149     prob_mul_shiftedHalfPowerImportanceWeightedMoment_eq
150       arms prob loss hchosen (hprob chosen hchosen))]
151   calc
152     arms.sum (fun chosen =>
153         (loss chosen) ^ 2 *
154           (Real.sqrt (prob chosen) * (1 - prob chosen) ^ 2 +
155             prob chosen * (arms.erase chosen).sum (fun action =>
156               Real.sqrt (prob action) * prob action))) <=
157         arms.sum (fun chosen =>
158           Real.sqrt (prob chosen) * (1 - prob chosen) ^ 2 +
159             prob chosen * (arms.erase chosen).sum (fun action =>
160               Real.sqrt (prob action) * prob action)) := by
161       apply Finset.sum_le_sum
162       intro chosen hchosen
163       have hl := hloss chosen hchosen
164       have hpNonneg := hprobability.1 chosen hchosen
165       have hpLeOne :=
166         finiteSimplex_apply_le_one hprobability hchosen
167       have heraseNonneg :
168           0 <= (arms.erase chosen).sum (fun action =>
169             Real.sqrt (prob action) * prob action) := by
170         apply Finset.sum_nonneg
171         intro action haction
172         exact mul_nonneg (Real.sqrt_nonneg _)
173           (hprobability.1 action (Finset.mem_of_mem_erase haction))
174       have hinnerNonneg :
175           0 <= Real.sqrt (prob chosen) * (1 - prob chosen) ^ 2 +
176             prob chosen * (arms.erase chosen).sum (fun action =>
177               Real.sqrt (prob action) * prob action) := by
178         exact add_nonneg
179           (mul_nonneg (Real.sqrt_nonneg _) (sq_nonneg _))
180           (mul_nonneg hpNonneg heraseNonneg)
181       nlinarith [sq_nonneg (loss chosen),
182         mul_self_le_mul_self hl.1 hl.2]
183     _ = arms.sum (fun action =>
184         Real.sqrt (prob action) * (1 - prob action)) := by
185       rw [Finset.sum_add_distrib]
186       rw [sum_mul_sum_erase_eq_sum_mul_one_sub
187         arms prob (fun action => Real.sqrt (prob action) * prob action)
188         hprobability.2]
189       rw [← Finset.sum_add_distrib]
190       apply Finset.sum_congr rfl
191       intro action _haction
192       ring

194 /-- Exact sampled-action expansion of the positive cubic remainder. -/
195 theorem prob_mul_shiftedPositiveCubicImportanceWeightedMoment_eq
196     {Action : Type u} [DecidableEq Action]
197     (arms : Finset Action) (prob loss : Action -> Real)
198     {chosen : Action} (hchosen : chosen ∈ arms)
199     (hprob : 0 < prob chosen) (hprobLeOne : prob chosen <= 1)
200     (hlossNonneg : 0 <= loss chosen) :
201     prob chosen *
202         shiftedPositiveCubicImportanceWeightedMoment arms prob loss chosen =
203       prob chosen * (loss chosen) ^ 3 *
204         (arms.erase chosen).sum (fun action => (prob action) ^ 2) := by
205   unfold shiftedPositiveCubicImportanceWeightedMoment
206   rw [Finset.mul_sum]
207   rw [← Finset.sum_erase_add _ _ hchosen]
208   have hselected :
209       max (loss chosen -
210         Exp3.importanceWeightedLoss prob loss chosen chosen) 0 = 0 := by
211     simp only [Exp3.importanceWeightedLoss, if_true]
212     rw [max_eq_right]
213     apply sub_nonpos.2
214     apply (le_div_iff₀ hprob).2
215     nlinarith
216   simp only [hselected, zero_pow (by norm_num : 3 ≠ 0), mul_zero, add_zero]
217   rw [Finset.mul_sum]
218   apply Finset.sum_congr rfl
219   intro action haction
220   have hne : chosen ≠ action :=
221     Ne.symm (Finset.ne_of_mem_erase haction)
222   simp [Exp3.importanceWeightedLoss, hne, hlossNonneg]
223   ring

225 /-- The sampled positive cubic IW remainder is at most one. -/
226 theorem sum_prob_mul_shiftedPositiveCubicImportanceWeightedMoment_le_one
227     {Action : Type u} [DecidableEq Action]
228     (arms : Finset Action) (prob loss : Action -> Real)
229     (hprobability : FTRL.finiteSimplex arms prob)
230     (hprob : forall action, action ∈ arms -> 0 < prob action)
231     (hloss : forall action, action ∈ arms ->
232       0 <= loss action ∧ loss action <= 1) :
233     arms.sum (fun chosen =>
234         prob chosen *
235           shiftedPositiveCubicImportanceWeightedMoment
236             arms prob loss chosen) <= 1 := by
237   rw [Finset.sum_congr rfl (fun chosen hchosen =>
238     prob_mul_shiftedPositiveCubicImportanceWeightedMoment_eq
239       arms prob loss hchosen (hprob chosen hchosen)
240       (finiteSimplex_apply_le_one hprobability hchosen)
241       (hloss chosen hchosen).1)]
242   calc
243     arms.sum (fun chosen =>
244         prob chosen * (loss chosen) ^ 3 *
245           (arms.erase chosen).sum (fun action => (prob action) ^ 2)) <=
246         arms.sum (fun chosen =>
247           prob chosen *
248             (arms.erase chosen).sum (fun action => (prob action) ^ 2)) := by
249       apply Finset.sum_le_sum
250       intro chosen hchosen
251       have hp := hprobability.1 chosen hchosen
252       have hl := hloss chosen hchosen
253       have heraseNonneg :
254           0 <= (arms.erase chosen).sum (fun action => (prob action) ^ 2) := by
255         positivity
256       have hlcube : (loss chosen) ^ 3 <= 1 := by
257         nlinarith [sq_nonneg (loss chosen),
258           mul_self_le_mul_self hl.1 hl.2]
259       calc
260         prob chosen * (loss chosen) ^ 3 *
261             (arms.erase chosen).sum (fun action => (prob action) ^ 2) =
262             (prob chosen *
263               (arms.erase chosen).sum (fun action => (prob action) ^ 2)) *
264                 (loss chosen) ^ 3 := by ring
265         _ <= (prob chosen *
266               (arms.erase chosen).sum (fun action => (prob action) ^ 2)) *
267                 1 :=
268           mul_le_mul_of_nonneg_left hlcube
269             (mul_nonneg hp heraseNonneg)
270         _ = prob chosen *
271             (arms.erase chosen).sum (fun action => (prob action) ^ 2) := by
272           ring
273     _ = arms.sum (fun action =>
274         (prob action) ^ 2 * (1 - prob action)) := by
275       exact sum_mul_sum_erase_eq_sum_mul_one_sub
276         arms prob (fun action => (prob action) ^ 2) hprobability.2
277     _ <= arms.sum (fun action => (prob action) ^ 2) := by
278       apply Finset.sum_le_sum
279       intro action haction
280       have hp := hprobability.1 action haction
281       have hpLeOne :=
282         finiteSimplex_apply_le_one hprobability haction
283       nlinarith [sq_nonneg (prob action)]
284     _ <= arms.sum prob := by
285       apply Finset.sum_le_sum
286       intro action haction
287       have hp := hprobability.1 action haction
288       have hpLeOne :=
289         finiteSimplex_apply_le_one hprobability haction
290       nlinarith [sq_nonneg (prob action)]
291     _ = 1 := hprobability.2

293 /--
294 Paper-shaped finite-sum consumer for a shifted Taylor/Hessian stability bound.
295 
296 The remaining hypothesis is deliberately pointwise: a producer must compare
297 its instantaneous stability quantity with the shifted quadratic and positive
298 cubic moments.  This theorem performs all sampled-action averaging.
299 -/
300 theorem sum_prob_mul_stability_le_refinedHalfPower_add_square
301     {Action : Type u} [DecidableEq Action]
302     (arms : Finset Action) (eta : Real)
303     (prob loss stability : Action -> Real)
304     (heta : 0 <= eta)
305     (hprobability : FTRL.finiteSimplex arms prob)
306     (hprob : forall action, action ∈ arms -> 0 < prob action)
307     (hloss : forall action, action ∈ arms ->
308       0 <= loss action ∧ loss action <= 1)
309     (hpointwise : forall chosen, chosen ∈ arms ->
310       stability chosen <=
311         eta / 2 *
312             shiftedHalfPowerImportanceWeightedMoment
313               arms prob loss chosen +
314           eta ^ 2 / 2 *
315             shiftedPositiveCubicImportanceWeightedMoment
316               arms prob loss chosen) :
317     arms.sum (fun chosen => prob chosen * stability chosen) <=
318       eta / 2 * arms.sum (fun action =>
319         Real.sqrt (prob action) * (1 - prob action)) +
320         eta ^ 2 / 2 := by
321   have hquadratic :=
322     sum_prob_mul_shiftedHalfPowerImportanceWeightedMoment_le
323       arms prob loss hprobability hprob hloss
324   have hcubic :=
325     sum_prob_mul_shiftedPositiveCubicImportanceWeightedMoment_le_one
326       arms prob loss hprobability hprob hloss
327   calc
328     arms.sum (fun chosen => prob chosen * stability chosen) <=
329         arms.sum (fun chosen =>
330           prob chosen *
331             (eta / 2 *
332                 shiftedHalfPowerImportanceWeightedMoment
333                   arms prob loss chosen +
334               eta ^ 2 / 2 *
335                 shiftedPositiveCubicImportanceWeightedMoment
336                   arms prob loss chosen)) := by
337       apply Finset.sum_le_sum
338       intro chosen hchosen
339       exact mul_le_mul_of_nonneg_left (hpointwise chosen hchosen)
340         (hprobability.1 chosen hchosen)
341     _ = eta / 2 *
342           arms.sum (fun chosen =>
343             prob chosen *
344               shiftedHalfPowerImportanceWeightedMoment
345                 arms prob loss chosen) +
346         eta ^ 2 / 2 *
347           arms.sum (fun chosen =>
348             prob chosen *
349               shiftedPositiveCubicImportanceWeightedMoment
350                 arms prob loss chosen) := by
351       calc
352         arms.sum (fun chosen =>
353             prob chosen *
354               (eta / 2 *
355                   shiftedHalfPowerImportanceWeightedMoment
356                     arms prob loss chosen +
357                 eta ^ 2 / 2 *
358                   shiftedPositiveCubicImportanceWeightedMoment
359                     arms prob loss chosen)) =
360             arms.sum (fun chosen =>
361               eta / 2 *
362                   (prob chosen *
363                     shiftedHalfPowerImportanceWeightedMoment
364                       arms prob loss chosen) +
365                 eta ^ 2 / 2 *
366                   (prob chosen *
367                     shiftedPositiveCubicImportanceWeightedMoment
368                       arms prob loss chosen)) := by
369           apply Finset.sum_congr rfl
370           intro chosen _hchosen
371           ring
372         _ = eta / 2 *
373               arms.sum (fun chosen =>
374                 prob chosen *
375                   shiftedHalfPowerImportanceWeightedMoment
376                     arms prob loss chosen) +
377             eta ^ 2 / 2 *
378               arms.sum (fun chosen =>
379                 prob chosen *
380                   shiftedPositiveCubicImportanceWeightedMoment
381                     arms prob loss chosen) := by
382           rw [Finset.sum_add_distrib, Finset.mul_sum, Finset.mul_sum]
383     _ <= eta / 2 * arms.sum (fun action =>
384           Real.sqrt (prob action) * (1 - prob action)) +
385         eta ^ 2 / 2 * 1 := by
386       exact add_le_add
387         (mul_le_mul_of_nonneg_left hquadratic (div_nonneg heta (by norm_num)))
388         (mul_le_mul_of_nonneg_left hcubic (div_nonneg (sq_nonneg eta) (by norm_num)))
389     _ = eta / 2 * arms.sum (fun action =>
390           Real.sqrt (prob action) * (1 - prob action)) +
391         eta ^ 2 / 2 := by ring

393 /--
394 Exact current-FTRL-expression wrapper around the shifted-moment consumer.
395 
396 The sole unresolved premise is the deterministic shifted Taylor/Hessian
397 comparison for each sampled-action update.
398 -/
399 theorem sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_refined_of_shiftedTaylor
400     {Action : Type u} [DecidableEq Action]
401     (arms : Finset Action) (eta : Real)
402     (prob loss : Action -> Real) (next : Action -> Action -> Real)
403     (heta : 0 <= eta)
404     (hprobability : FTRL.finiteSimplex arms prob)
405     (hprob : forall action, action ∈ arms -> 0 < prob action)
406     (hloss : forall action, action ∈ arms ->
407       0 <= loss action ∧ loss action <= 1)
408     (hshiftedTaylor : forall chosen, chosen ∈ arms ->
409       FTRL.linearLoss arms prob
410             (Exp3.importanceWeightedLoss prob loss chosen) -
411           FTRL.linearLoss arms (next chosen)
412             (Exp3.importanceWeightedLoss prob loss chosen) <=
413         eta / 2 *
414             shiftedHalfPowerImportanceWeightedMoment
415               arms prob loss chosen +
416           eta ^ 2 / 2 *
417             shiftedPositiveCubicImportanceWeightedMoment
418               arms prob loss chosen) :
419     arms.sum (fun chosen =>
420         prob chosen *
421           (FTRL.linearLoss arms prob
422               (Exp3.importanceWeightedLoss prob loss chosen) -
423             FTRL.linearLoss arms (next chosen)
424               (Exp3.importanceWeightedLoss prob loss chosen))) <=
425       eta / 2 * arms.sum (fun action =>
426         Real.sqrt (prob action) * (1 - prob action)) +
427         eta ^ 2 / 2 := by
428   exact sum_prob_mul_stability_le_refinedHalfPower_add_square
429     arms eta prob loss
430     (fun chosen =>
431       FTRL.linearLoss arms prob
432           (Exp3.importanceWeightedLoss prob loss chosen) -
433         FTRL.linearLoss arms (next chosen)
434           (Exp3.importanceWeightedLoss prob loss chosen))
435     heta hprobability hprob hloss hshiftedTaylor
