3 /-!
4 # Coefficient-aware refined self-bounding scalar tuning
5 
6 The local refined stability envelope has coefficient `5`.  After the change
7 of variables used by the square-root schedule, its scalar objective therefore
8 has a beta equation with offset `2`, rather than the paper's idealized offset
9 `1`.  This module records the corrected root interval and an elementary root
10 estimate that does not require Lambert W.
11 -/

18 /-- The beta equation produced by the local coefficient-five envelope. -/
19 noncomputable def refinedLocalBetaEquation
20     (scale reciprocalGap corruption beta : Real) : Real :=
21   corruption * reciprocalGap / scale * beta - Real.log beta - 2
22 
23 /-- The local beta equation is continuous on every interval bounded below by
24 one. -/
25 theorem continuousOn_refinedLocalBetaEquation
26     (scale reciprocalGap corruption upper : Real) :
27     ContinuousOn (refinedLocalBetaEquation scale reciprocalGap corruption)
28       (Icc 1 upper) := by
29   unfold refinedLocalBetaEquation
30   apply ContinuousOn.sub
31   · apply ContinuousOn.sub
32     · fun_prop
33     · exact Real.continuousOn_log.mono (by
34         intro beta hbeta
35         exact ne_of_gt (zero_lt_one.trans_le hbeta.1))
36   · exact continuousOn_const

38 /-- The elementary inequality behind the Lambert-free beta estimate. -/
39 theorem sq_sqrt_sub_one_le_sub_log_sub_one
40     (weight : Real) (hweight : 1 <= weight) :
41     (Real.sqrt weight - 1) ^ 2 <=
42       weight - Real.log weight - 1 := by
43   have hweightNonneg : 0 <= weight := zero_le_one.trans hweight
44   have hsqrtPos : 0 < Real.sqrt weight :=
45     Real.sqrt_pos.2 (zero_lt_one.trans_le hweight)
46   have hlogSqrt := Real.log_le_sub_one_of_pos hsqrtPos
47   have hlogSqrtEq :
48       Real.log (Real.sqrt weight) = Real.log weight / 2 :=
49     Real.log_sqrt hweightNonneg
50   rw [hlogSqrtEq] at hlogSqrt
51   nlinarith [Real.sq_sqrt hweightNonneg]

53 /-- A root of the coefficient-five equation admits the same kind of elementary
54 upper estimate as the paper's Lambert-W expression, with the offset correction
55 appearing as `+1` under the outer square root. -/
56 theorem refinedLocalBetaWeight_bounds_of_eq_zero
57     (scale reciprocalGap corruption beta : Real)
58     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
59     (hcorruption : 0 < corruption) (hbeta : 1 <= beta)
60     (hcorruptionUpper : corruption * reciprocalGap <= scale)
61     (hroot : refinedLocalBetaEquation
62       scale reciprocalGap corruption beta = 0) :
63     let weight := corruption * reciprocalGap / scale * beta
64     1 <= weight ∧
65       weight <=
66         (1 + Real.sqrt
67           (Real.log (scale / (corruption * reciprocalGap)) + 1)) ^ 2 := by
68   dsimp only
69   let weight := corruption * reciprocalGap / scale * beta
70   let ratio := scale / (corruption * reciprocalGap)
71   have hproductPos : 0 < corruption * reciprocalGap :=
72     mul_pos hcorruption hreciprocalGap
73   have hratioPos : 0 < ratio := by
74     dsimp [ratio]
75     positivity
76   have hratioOne : 1 <= ratio := by
77     dsimp [ratio]
78     apply (le_div_iff₀ hproductPos).2
79     simpa using hcorruptionUpper
80   have hlogBetaNonneg : 0 <= Real.log beta := Real.log_nonneg hbeta
81   have hweightEq : weight = Real.log beta + 2 := by
82     dsimp [weight]
83     unfold refinedLocalBetaEquation at hroot
84     linarith
85   have hweightOne : 1 <= weight := by linarith
86   have hweightPos : 0 < weight := zero_lt_one.trans_le hweightOne
87   have hbetaEq : beta = ratio * weight := by
88     dsimp [ratio, weight]
89     field_simp
90   have hlogBetaEq :
91       Real.log beta = Real.log ratio + Real.log weight := by
92     rw [hbetaEq, Real.log_mul (ne_of_gt hratioPos) (ne_of_gt hweightPos)]
93   have hidentity :
94       weight - Real.log weight - 2 = Real.log ratio := by
95     calc
96       weight - Real.log weight - 2 =
97           Real.log beta - Real.log weight := by linarith
98       _ = Real.log ratio := by rw [hlogBetaEq]; ring
99   have hlower :=
100     sq_sqrt_sub_one_le_sub_log_sub_one weight hweightOne
101   have hsquare :
102       (Real.sqrt weight - 1) ^ 2 <= Real.log ratio + 1 := by
103     linarith
104   have hlogRatioNonneg : 0 <= Real.log ratio := Real.log_nonneg hratioOne
105   have hsqrtWeightOne : 1 <= Real.sqrt weight := by
106     exact Real.one_le_sqrt.mpr hweightOne
107   have hsqrtBound :
108       Real.sqrt weight - 1 <= Real.sqrt (Real.log ratio + 1) := by
109     have hrightNonneg : 0 <= Real.log ratio + 1 := by linarith
110     have hsqrtRightSq := Real.sq_sqrt hrightNonneg
111     have hleftNonneg : 0 <= Real.sqrt weight - 1 := by linarith
112     have hsqrtRightNonneg := Real.sqrt_nonneg (Real.log ratio + 1)
113     nlinarith
114   have hweightNonneg : 0 <= weight := zero_le_one.trans hweightOne
115   have hsqrtWeightSq := Real.sq_sqrt hweightNonneg
116   have hfinal :
117       weight <=
118         (1 + Real.sqrt (Real.log ratio + 1)) ^ 2 := by
119     nlinarith [Real.sqrt_nonneg (Real.log ratio + 1)]
120   exact ⟨hweightOne, by simpa [ratio] using hfinal⟩

122 /-- Under the coefficient-aware corruption window, the corrected beta equation
123 has a root in the interval that corresponds to `alpha` between the local
124 horizon threshold and one. -/
125 theorem exists_refinedLocalBetaEquation_eq_zero
126     (scale reciprocalGap corruption : Real)
127     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
128     (hupper : 2 <= scale / (25 * reciprocalGap ^ 2))
129     (hcorruptionUpper : 2 * (corruption * reciprocalGap) <= scale)
130     (hcorruptionLower :
131       25 * reciprocalGap *
132           (Real.log (scale / (25 * reciprocalGap ^ 2)) + 2) <=
133         corruption) :
134     exists beta,
135       beta ∈ Icc 2 (scale / (25 * reciprocalGap ^ 2)) ∧
136         refinedLocalBetaEquation
137           scale reciprocalGap corruption beta = 0 := by
138   let upper := scale / (25 * reciprocalGap ^ 2)
139   let g := refinedLocalBetaEquation scale reciprocalGap corruption
140   have hgTwo : g 2 <= 0 := by
141     have hdiv :
142         corruption * reciprocalGap / scale <= 1 / 2 := by
143       rw [div_le_iff₀ hscale]
144       nlinarith
145     have hlogTwo : 0 <= Real.log 2 := Real.log_nonneg (by norm_num)
146     dsimp [g, refinedLocalBetaEquation]
147     nlinarith
148   have hgUpper : 0 <= g upper := by
149     have hreciprocalGapNe : reciprocalGap ≠ 0 := ne_of_gt hreciprocalGap
150     have hscaleNe : scale ≠ 0 := ne_of_gt hscale
151     have hnormalize :
152         corruption * reciprocalGap / scale * upper =
153           corruption / (25 * reciprocalGap) := by
154       dsimp [upper]
155       field_simp
156     have hdenomPos : 0 < 25 * reciprocalGap := by positivity
157     have hdiv :
158         Real.log upper + 2 <= corruption / (25 * reciprocalGap) := by
159       apply (le_div_iff₀ hdenomPos).2
160       simpa [upper, mul_assoc, mul_left_comm, mul_comm] using
161         hcorruptionLower
162     dsimp [g, refinedLocalBetaEquation]
163     rw [hnormalize]
164     linarith
165   have hcontinuous : ContinuousOn g (Icc 2 upper) := by
166     exact (continuousOn_refinedLocalBetaEquation
167       scale reciprocalGap corruption upper).mono (by
168         intro beta hbeta
169         exact ⟨one_le_two.trans hbeta.1, hbeta.2⟩)
170   have hzero : (0 : Real) ∈ Icc (g 2) (g upper) := ⟨hgTwo, hgUpper⟩
171   rcases intermediate_value_Icc hupper hcontinuous hzero with
172     ⟨beta, hbeta, hbetaZero⟩
173   exact ⟨beta, hbeta, hbetaZero⟩

178 theorem exists_refinedLocalBetaEquation_eq_zero_and_weight_bounds
179     (scale reciprocalGap corruption : Real)
180     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
181     (hcorruption : 0 < corruption)
182     (hupper : 2 <= scale / (25 * reciprocalGap ^ 2))

207 /-- The coefficient-aware change of variables from beta to alpha. -/
208 noncomputable def refinedLocalAlpha
209     (scale reciprocalGap beta : Real) : Real :=
210   Real.sqrt (25 * reciprocalGap ^ 2 * beta / scale)
211 
212 /-- The inverse of `alpha = 2 * lambda / (1 + lambda)`. -/
213 noncomputable def refinedLocalLambda
214     (scale reciprocalGap beta : Real) : Real :=
215   let alpha := refinedLocalAlpha scale reciprocalGap beta
216   alpha / (2 - alpha)
217 
218 theorem refinedLocalAlpha_sq
219     (scale reciprocalGap beta : Real)
220     (hscale : 0 < scale) (hbeta : 0 <= beta) :
221     refinedLocalAlpha scale reciprocalGap beta ^ 2 =
222       25 * reciprocalGap ^ 2 * beta / scale := by
223   unfold refinedLocalAlpha

225   positivity

236 theorem refinedLocalAlpha_le_one
237     (scale reciprocalGap beta : Real)
238     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
239     (hbeta : beta <= scale / (25 * reciprocalGap ^ 2)) :
240     refinedLocalAlpha scale reciprocalGap beta <= 1 := by

242   have hmul : 25 * reciprocalGap ^ 2 * beta <= scale := by
243     have := (le_div_iff₀ hdenom).1 hbeta

249 theorem refinedLocalLambda_mem_Ioc
250     (scale reciprocalGap beta : Real)
251     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
252     (hbetaPos : 0 < beta)
253     (hbetaUpper : beta <= scale / (25 * reciprocalGap ^ 2)) :
254     refinedLocalLambda scale reciprocalGap beta ∈ Set.Ioc (0 : Real) 1 := by
255   let alpha := refinedLocalAlpha scale reciprocalGap beta
256   have halphaPos : 0 < alpha :=
257     refinedLocalAlpha_pos scale reciprocalGap beta
258       hscale hreciprocalGap hbetaPos
259   have halphaLe : alpha <= 1 :=
260     refinedLocalAlpha_le_one scale reciprocalGap beta
261       hscale hreciprocalGap hbetaUpper
262   have hdenom : 0 < 2 - alpha := by linarith
263   constructor
264   · dsimp [refinedLocalLambda, alpha]
265     positivity
266   · dsimp [refinedLocalLambda, alpha]
267     exact (div_le_one hdenom).2 (by linarith)
