4 # Scheduled recursive half-Tsallis trajectories
5 
6 This module lifts the recursive pure half-Tsallis trajectory from a fixed
7 learning rate to a deterministic schedule.  The initial action uses `eta 0`;
8 after the visible prefix through round `n`, the successor policy uses
9 `eta (n + 1)`.  This indexing matches the scheduled FTRL penalty route.
10 
11 The module constructs only the selector, score, policy, trajectory, and
12 conditional action law.  It does not prove a refined stability or regret
13 bound.
14 -/

40 /-- Cumulative importance-weighted score through an inclusive history under
41 the scheduled policies. -/
42 noncomputable def sampledScheduledHalfTsallisHistoryScore
43     {Action : Type u} [DecidableEq Action]
44     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real) :
45     (n : Nat) -> History.FinitePairHistory Action Real n -> Action -> Real
46   | 0, history, action =>
47       Exp3.importanceWeightedLoss
48         (initialHalfTsallisDistribution arms harms (eta 0))
49         (fun _ => (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).2)
50         (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 action
51   | n + 1, history, action =>
52       let previous := Exp3.previousPairHistory history
53       sampledScheduledHalfTsallisHistoryScore arms harms eta n previous action +
54         Exp3.importanceWeightedLoss
55           (halfTsallisMinimizer arms harms (eta (n + 1))
56             (sampledScheduledHalfTsallisHistoryScore
57               arms harms eta n previous))
58           (fun _ => (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).2)
59           (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).1 action

91 /-- Scheduled selector regularity makes every supported score coordinate
92 measurable. -/
93 theorem measurable_sampledScheduledHalfTsallisHistoryScore
94     {Action : Type u}
95     [MeasurableSpace Action] [MeasurableSingletonClass Action]
96     [DecidableEq Action]
97     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
98     (selector : HalfTsallisScheduleFiniteHistorySelectorMeasurability
99       arms harms eta) :
100     forall n action, action ∈ arms ->
101       Measurable (fun history : History.FinitePairHistory Action Real n =>
102         sampledScheduledHalfTsallisHistoryScore
103           arms harms eta n history action) := by
104   intro n
105   induction n with
106   | zero =>
107       intro action _haction
108       let zeroIndex : Finset.Iic 0 := ⟨0, Finset.mem_Iic.mpr le_rfl⟩
109       have hchosen : Measurable
110           (fun history : History.FinitePairHistory Action Real 0 =>
111             (history zeroIndex).1) :=
112         measurable_fst.comp (measurable_pi_apply zeroIndex)
113       have hloss : Measurable
114           (fun history : History.FinitePairHistory Action Real 0 =>
115             (history zeroIndex).2) :=
116         measurable_snd.comp (measurable_pi_apply zeroIndex)
117       simpa [sampledScheduledHalfTsallisHistoryScore, zeroIndex] using
118         (Exp3.measurable_observedImportanceWeightedLoss
119           (fun _history : History.FinitePairHistory Action Real 0 =>
120             initialHalfTsallisDistribution arms harms (eta 0))
121           (fun history => (history zeroIndex).1)
122           (fun history => (history zeroIndex).2)
123           action measurable_const hchosen hloss)
124   | succ n ih =>
125       intro action haction
126       let lastIndex : Finset.Iic (n + 1) :=
127         ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩
128       have hprevious : Measurable
129           (Exp3.previousPairHistory (Action := Action) (n := n)) :=
130         Exp3.measurable_previousPairHistory
131       have hscore : Measurable
132           (fun history : History.FinitePairHistory Action Real (n + 1) =>
133             sampledScheduledHalfTsallisHistoryScore arms harms eta n
134               (Exp3.previousPairHistory history) action) :=
135         (ih action haction).comp hprevious
136       have hprob : Measurable
137           (fun history : History.FinitePairHistory Action Real (n + 1) =>
138             halfTsallisMinimizer arms harms (eta (n + 1))
139               (sampledScheduledHalfTsallisHistoryScore arms harms eta n
140                 (Exp3.previousPairHistory history)) action) :=
141         ((selector.round (n + 1)).measurable_selector n
142           (sampledScheduledHalfTsallisHistoryScore arms harms eta n)
143           (fun selected hselected => ih selected hselected)
144           action haction).comp hprevious
145       have hchosen : Measurable
146           (fun history : History.FinitePairHistory Action Real (n + 1) =>
147             (history lastIndex).1) :=
148         measurable_fst.comp (measurable_pi_apply lastIndex)
149       have hloss : Measurable
150           (fun history : History.FinitePairHistory Action Real (n + 1) =>
151             (history lastIndex).2) :=
152         measurable_snd.comp (measurable_pi_apply lastIndex)
153       have hincrement : Measurable
154           (fun history : History.FinitePairHistory Action Real (n + 1) =>
155             Exp3.importanceWeightedLoss
156               (halfTsallisMinimizer arms harms (eta (n + 1))
157                 (sampledScheduledHalfTsallisHistoryScore arms harms eta n
158                   (Exp3.previousPairHistory history)))
159               (fun _ => (history lastIndex).2)
160               (history lastIndex).1 action) :=
161         Exp3.measurable_observedImportanceWeightedLoss
162           (fun history =>
163             halfTsallisMinimizer arms harms (eta (n + 1))
164               (sampledScheduledHalfTsallisHistoryScore arms harms eta n
165                 (Exp3.previousPairHistory history)))
166           (fun history => (history lastIndex).1)
167           (fun history => (history lastIndex).2)
168           action hprob hchosen hloss
169       simpa [sampledScheduledHalfTsallisHistoryScore, lastIndex] using
170         hscore.add hincrement
