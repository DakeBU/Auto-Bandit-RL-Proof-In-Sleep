4 /-!
5 # Variance-sensitive fixed-comparator EXP3 concentration
6 
7 This module replaces the range-squared Hoeffding proxy for one fixed comparator estimator by a
8 fixed-tilt second-moment bound.  The scalar input is the quadratic exponential remainder on
9 `[-1, 1]`; the probabilistic input is the exact second moment of the importance-weighted
10 estimator under its finite sampling law.
11 -/

23   have hle : Real.exp x - 1 - x <= |Real.exp x - 1 - x| := le_abs_self _

26 /-- Optimize a quadratic fixed-tilt MGF budget under the hard constraint `tilt <= epsilon`. -/
27 theorem exists_tilt_fixedMGF_exponent_le_neg
28     (horizon epsilon budget : Real)
29     (hhorizon : 0 <= horizon) (hepsilon : 0 < epsilon)
30     (hbudget : 0 <= budget) :
31     exists tilt : Real, 0 <= tilt ∧ tilt <= epsilon ∧
32       -tilt *
33           (2 * Real.sqrt (horizon * budget / epsilon) + budget / epsilon) +
34         horizon * (tilt ^ 2 / epsilon) <= -budget := by
35   by_cases hbudget_zero : budget = 0
36   · refine ⟨0, le_rfl, hepsilon.le, ?_⟩
37     simp [hbudget_zero]
38   have hbudget_pos : 0 < budget :=
39     lt_of_le_of_ne hbudget (Ne.symm hbudget_zero)
40   by_cases hsmall : budget <= epsilon * horizon
41   · have hhorizon_pos : 0 < horizon := by nlinarith
42     let tilt := Real.sqrt (epsilon * budget / horizon)
43     let scale := Real.sqrt (horizon * budget / epsilon)
44     have htilt_nonneg : 0 <= tilt := Real.sqrt_nonneg _
45     have hscale_nonneg : 0 <= scale := Real.sqrt_nonneg _
46     have htilt_sq : tilt ^ 2 = epsilon * budget / horizon := by
47       dsimp only [tilt]
48       rw [Real.sq_sqrt]
49       positivity
50     have hscale_sq : scale ^ 2 = horizon * budget / epsilon := by
51       dsimp only [scale]
52       rw [Real.sq_sqrt]
53       positivity
54     have hprod_sq : (tilt * scale) ^ 2 = budget ^ 2 := by
55       rw [mul_pow, htilt_sq, hscale_sq]
56       field_simp
57     have hprod : tilt * scale = budget := by
58       nlinarith [mul_nonneg htilt_nonneg hscale_nonneg]
59     have htilt_sq_le : tilt ^ 2 <= epsilon ^ 2 := by
60       rw [htilt_sq, div_le_iff₀ hhorizon_pos]
61       nlinarith
62     have htilt_le : tilt <= epsilon :=
63       (sq_le_sq₀ htilt_nonneg hepsilon.le).mp htilt_sq_le
64     refine ⟨tilt, htilt_nonneg, htilt_le, ?_⟩
65     change -tilt * (2 * scale + budget / epsilon) +
66       horizon * (tilt ^ 2 / epsilon) <= -budget
67     have hbudget_term : horizon * (tilt ^ 2 / epsilon) = budget := by
68       rw [htilt_sq]
69       field_simp
70     have hextra : 0 <= tilt * (budget / epsilon) :=
71       mul_nonneg htilt_nonneg (div_nonneg hbudget hepsilon.le)
72     rw [hbudget_term]
73     nlinarith
74   · refine ⟨epsilon, hepsilon.le, le_rfl, ?_⟩
75     let scale := Real.sqrt (horizon * budget / epsilon)
76     have hscale_nonneg : 0 <= scale :=
77       Real.sqrt_nonneg _
78     have hscale_sq : scale ^ 2 = horizon * budget / epsilon := by
79       dsimp only [scale]
80       rw [Real.sq_sqrt]
81       positivity
82     have hlarge : epsilon * horizon < budget := lt_of_not_ge hsmall
83     have hhorizon_sq_le : horizon ^ 2 <= scale ^ 2 := by
84       rw [hscale_sq, le_div_iff₀ hepsilon]
85       have hmul := mul_le_mul_of_nonneg_right hlarge.le hhorizon
86       simpa [pow_two, mul_assoc, mul_comm, mul_left_comm] using hmul
87     have hhorizon_le_scale : horizon <= scale :=
88       (sq_le_sq₀ hhorizon hscale_nonneg).mp hhorizon_sq_le
89     have hcover : horizon * epsilon <= 2 * epsilon * scale := by
90       calc
91         horizon * epsilon = epsilon * horizon := by ring
92         _ <= epsilon * scale :=
93           mul_le_mul_of_nonneg_left hhorizon_le_scale hepsilon.le
94         _ <= 2 * epsilon * scale := by
95           nlinarith [mul_nonneg hepsilon.le hscale_nonneg]
96     calc
97       -epsilon *
98             (2 * Real.sqrt (horizon * budget / epsilon) + budget / epsilon) +
99           horizon * (epsilon ^ 2 / epsilon) =
100         -2 * epsilon * Real.sqrt (horizon * budget / epsilon) - budget +
101           horizon * epsilon := by
102             field_simp
103             ring
104       _ <= -budget := by
105         change -2 * epsilon * scale - budget + horizon * epsilon <= -budget
106         nlinarith

114 /-- Exact centered second moment of one fixed-arm importance-weighted estimator. -/
115 theorem sum_prob_mul_sq_comparatorEstimatorDeviation_eq
116     {Action : Type u} [DecidableEq Action]
117     (arms : Finset Action) (prob loss : Action -> Real)
118     (hdist : FiniteActionDistribution arms prob)
119     (comparator : Action) (hcomparator : comparator ∈ arms)
120     (hprob : prob comparator ≠ 0) :
121     arms.sum (fun chosen =>
122         prob chosen *
123           (importanceWeightedLoss prob loss chosen comparator - loss comparator) ^ 2) =
124       (loss comparator) ^ 2 / prob comparator - (loss comparator) ^ 2 := by
125   let raw := fun chosen => importanceWeightedLoss prob loss chosen comparator
126   have hfirst : arms.sum (fun chosen => prob chosen * raw chosen) = loss comparator := by
127     simpa [raw] using sum_prob_mul_importanceWeightedLoss_eq_loss
128       arms prob loss comparator hcomparator hprob
129   have hsecond : arms.sum (fun chosen => prob chosen * (raw chosen) ^ 2) =
130       (loss comparator) ^ 2 / prob comparator := by
131     rw [Finset.sum_eq_single comparator]
132     · simp [raw, importanceWeightedLoss]
133       field_simp
134     · intro chosen hchosen hne
135       simp [raw, importanceWeightedLoss, hne]
136     · exact fun hnotmem => (hnotmem hcomparator).elim
137   have hcenter (chosen : Action) :
138       prob chosen * (raw chosen - loss comparator) ^ 2 =
139         (prob chosen * (raw chosen) ^ 2 -
140           2 * loss comparator * (prob chosen * raw chosen)) +
141           (loss comparator) ^ 2 * prob chosen := by
142     ring
143   change arms.sum (fun chosen =>
144     prob chosen * (raw chosen - loss comparator) ^ 2) = _
145   simp_rw [hcenter]
146   rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
147     ← Finset.mul_sum, ← Finset.mul_sum, hsecond, hfirst, hdist.sum_eq_one]
148   ring

150 /-- The centered comparator estimator has second moment at most the reciprocal probability floor. -/
151 theorem sum_prob_mul_sq_comparatorEstimatorDeviation_le_inv_floor
152     {Action : Type u} [DecidableEq Action]
153     (arms : Finset Action) (prob loss : Action -> Real)
154     (hdist : FiniteActionDistribution arms prob)
155     (epsilon : Real) (hepsilon : 0 < epsilon)
156     (comparator : Action) (hcomparator : comparator ∈ arms)
157     (hfloor : epsilon <= prob comparator)
158     (hloss : loss comparator ∈ Set.Icc (0 : Real) 1) :
159     arms.sum (fun chosen =>
160         prob chosen *
161           (importanceWeightedLoss prob loss chosen comparator - loss comparator) ^ 2) <=
162       1 / epsilon := by
163   have hprob_pos : 0 < prob comparator := hepsilon.trans_le hfloor
164   rw [sum_prob_mul_sq_comparatorEstimatorDeviation_eq arms prob loss hdist
165     comparator hcomparator hprob_pos.ne']
166   have hloss_sq : (loss comparator) ^ 2 <= 1 := by
167     simpa using (sq_le_sq₀ hloss.1 zero_le_one).2 hloss.2
168   have hdiv : (loss comparator) ^ 2 / prob comparator <= 1 / epsilon := by
169     calc
170       (loss comparator) ^ 2 / prob comparator <= 1 / prob comparator :=
171         div_le_div_of_nonneg_right hloss_sq hprob_pos.le
172       _ <= 1 / epsilon := one_div_le_one_div_of_le hepsilon hfloor
173   nlinarith [sq_nonneg (loss comparator)]

175 /-- Fixed-tilt MGF budget for a finite-law fixed-comparator estimator. -/
176 theorem finiteActionComparatorEstimator_hasMGFUpperBoundAt
177     {Action : Type u} [MeasurableSpace Action] [MeasurableSingletonClass Action]
178     [DecidableEq Action]
179     (arms : Finset Action) (prob loss : Action -> Real)
180     (hdist : FiniteActionDistribution arms prob)
181     (epsilon : Real) (hepsilon : 0 < epsilon)
182     (comparator : Action) (hcomparator : comparator ∈ arms)
183     (hfloor : epsilon <= prob comparator)
184     (hloss : loss comparator ∈ Set.Icc (0 : Real) 1)
185     (tilt : Real) (htilt_nonneg : 0 <= tilt) (htilt_le : tilt <= epsilon) :
186     Concentration.HasMGFUpperBoundAt
187       (fun chosen =>
188         importanceWeightedLoss prob loss chosen comparator - loss comparator)
189       tilt (tilt ^ 2 / epsilon) (finiteActionMeasure arms prob) := by
190   let X := fun chosen =>
191     importanceWeightedLoss prob loss chosen comparator - loss comparator
192   letI : IsProbabilityMeasure (finiteActionMeasure arms prob) :=
193     finiteActionMeasure_isProbabilityMeasure arms prob hdist

195   have hprob_le_one : prob comparator <= 1 := by
196     calc
197       prob comparator <= arms.sum prob := by
198         exact Finset.single_le_sum (fun action haction => hdist.nonneg action haction)
199           hcomparator
200       _ = 1 := hdist.sum_eq_one
201   have hepsilon_le_one : epsilon <= 1 := hfloor.trans hprob_le_one
202   have hinv_nonneg : 0 <= 1 / epsilon := (one_div_pos.mpr hepsilon).le
203   have hone_le_inv : 1 <= 1 / epsilon := by
204     rw [le_div_iff₀ hepsilon]
205     simpa using hepsilon_le_one
206   have hX_bound (chosen : Action) : |X chosen| <= 1 / epsilon := by
207     have hraw_nonneg : 0 <= importanceWeightedLoss prob loss chosen comparator :=
208       importanceWeightedLoss_nonneg (chosen := chosen)
209         (hdist.nonneg comparator hcomparator) hloss.1
210     have hraw_le : importanceWeightedLoss prob loss chosen comparator <= 1 / epsilon := by
211       unfold importanceWeightedLoss
212       split_ifs
213       · exact (div_le_div_of_nonneg_right hloss.2 hprob_pos.le).trans
214           (one_div_le_one_div_of_le hepsilon hfloor)
215       · exact hinv_nonneg
216     dsimp only [X]
217     rw [abs_le]
218     constructor
219     · calc
220         -(1 / epsilon) <= -loss comparator := by
221           exact neg_le_neg (hloss.2.trans hone_le_inv)
222         _ <= importanceWeightedLoss prob loss chosen comparator - loss comparator := by
223           linarith
224     · calc
225         importanceWeightedLoss prob loss chosen comparator - loss comparator <=
226             importanceWeightedLoss prob loss chosen comparator := by
227           linarith [hloss.1]
228         _ <= 1 / epsilon := hraw_le
229   constructor
230   · intro s
231     rw [finiteActionMeasure]
232     refine integrable_finset_sum_measure.2 fun action haction => ?_
233     exact (integrable_dirac (by simp)).smul_measure ENNReal.ofReal_ne_top
234   · rw [ProbabilityTheory.mgf, integral_finiteActionMeasure_eq_sum arms prob hdist]
235     have hexp (chosen : Action) (hchosen : chosen ∈ arms) :
236         Real.exp (tilt * X chosen) <=
237           1 + tilt * X chosen + (tilt * X chosen) ^ 2 := by
238       apply Concentration.exp_le_one_add_self_add_sq_of_abs_le_one
239       rw [abs_mul, abs_of_nonneg htilt_nonneg]
240       calc
241         tilt * |X chosen| <= tilt * (1 / epsilon) :=
242           mul_le_mul_of_nonneg_left (hX_bound chosen) htilt_nonneg
243         _ <= epsilon * (1 / epsilon) :=
244           mul_le_mul_of_nonneg_right htilt_le hinv_nonneg
245         _ = 1 := by field_simp
246     calc
247       arms.sum (fun chosen => prob chosen * Real.exp (tilt * X chosen)) <=
248           arms.sum (fun chosen =>
249             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2)) := by
250         apply Finset.sum_le_sum
251         intro chosen hchosen
252         exact mul_le_mul_of_nonneg_left (hexp chosen hchosen)
253           (hdist.nonneg chosen hchosen)
254       _ = 1 + tilt * arms.sum (fun chosen => prob chosen * X chosen) +
255           tilt ^ 2 * arms.sum (fun chosen => prob chosen * (X chosen) ^ 2) := by
256         have hterm (chosen : Action) :
257             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2) =
258               (prob chosen + tilt * (prob chosen * X chosen)) +
259                 tilt ^ 2 * (prob chosen * (X chosen) ^ 2) := by ring
260         simp_rw [hterm, Finset.sum_add_distrib, ← Finset.mul_sum, hdist.sum_eq_one]
261       _ <= 1 + tilt ^ 2 * (1 / epsilon) := by
262         have hmean : arms.sum (fun chosen => prob chosen * X chosen) = 0 := by
263           dsimp only [X]
264           simp_rw [mul_sub]
265           rw [Finset.sum_sub_distrib,
266             sum_prob_mul_importanceWeightedLoss_eq_loss arms prob loss comparator
267               hcomparator hprob_pos.ne', ← Finset.sum_mul, hdist.sum_eq_one]
268           ring
269         rw [hmean, mul_zero, add_zero]
270         have hsecond := mul_le_mul_of_nonneg_left
271           (by
272             simpa [X] using
273               sum_prob_mul_sq_comparatorEstimatorDeviation_le_inv_floor
274                 arms prob loss hdist epsilon hepsilon comparator hcomparator hfloor hloss)
275           (sq_nonneg tilt)
276         simpa only [X, one_div, add_comm] using add_le_add_left hsecond 1
277       _ <= Real.exp (tilt ^ 2 / epsilon) := by
278         rw [show tilt ^ 2 * (1 / epsilon) = tilt ^ 2 / epsilon by ring]
279         simpa [add_comm] using Real.add_one_le_exp (tilt ^ 2 / epsilon)
