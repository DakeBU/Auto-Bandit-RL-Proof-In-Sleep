4 # Scheduled OFUL expected regret under unit resource growth
5 
6 This module derives the pathwise reach-by-budget premise from the more
7 primitive contract that the accumulated Nat-valued resource grows by at least
8 one at every round, then reuses the compiled budget-exhaustion OFUL theorem.
9 -/

20 /--
21 If a Nat-valued resource process grows by at least one at every step, then its
22 value at every index dominates that index.
23 -/
24 theorem index_le_spent_of_unitGrowth
25     {Omega : Type v}
26     (spent : Nat -> Omega -> Nat)
27     (hunit : forall t omega,
28       spent t omega + 1 <= spent (t + 1) omega) :
29     forall t omega, t <= spent t omega := by
30   intro t
31   induction t with
32   | zero =>
33       intro omega
34       exact Nat.zero_le _
35   | succ t ih =>
36       intro omega
37       exact le_trans (Nat.succ_le_succ (ih omega)) (hunit t omega)
38 
39 /--
40 Unit pathwise resource growth makes the budget-exhaustion time at most the
41 budget index.
42 -/
43 theorem budgetExhaustionTime_le_budget_of_unitGrowth
44     {Omega : Type v}
45     (spent : Nat -> Omega -> Nat)
46     (budget : Nat)
47     (hunit : forall t omega,
48       spent t omega + 1 <= spent (t + 1) omega) :
49     forall omega,
50       budgetExhaustionTime spent budget omega <= (budget : WithTop Nat) := by
51   exact
52     budgetExhaustionTime_le_budget_of_spent_budget
53       spent budget (index_le_spent_of_unitGrowth spent hunit budget)
