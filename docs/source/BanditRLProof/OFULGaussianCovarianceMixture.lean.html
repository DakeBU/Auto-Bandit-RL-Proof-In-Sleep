5 # Positive-definite covariance Gaussian mixture for OFUL
6 
7 This file transports the standard-Gaussian quadratic-exponential identity through
8 the square root of `V0⁻¹`.  For a positive-definite prior precision `V0` and a
9 positive-semidefinite Gram matrix `G`, it collects the transformed determinant and
10 quadratic form into the OFUL-facing matrices `V0 + G` and `(V0 + G)⁻¹`.
11 -/

57 private theorem integral_exp_inner_sub_quadratic_multivariateGaussian_zero_inv_transformed
58     (V0 G : Matrix Feature Feature Real) (hG : G.PosSemidef)
59     (score : EuclideanSpace Real Feature) :
60     let C := CFC.sqrt V0⁻¹
61     integral
62         (ProbabilityTheory.multivariateGaussian 0 V0⁻¹)
63         (fun z : EuclideanSpace Real Feature =>
64           Real.exp
65             (⟪score, z⟫_ℝ -
66               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2)) =
67       (Real.sqrt (Matrix.det (1 + C * G * C)))⁻¹ *
68         Real.exp
69           ((Matrix.toEuclideanCLM (𝕜 := Real) C score) ⬝ᵥ
70             (1 + C * G * C)⁻¹.mulVec
71               (Matrix.toEuclideanCLM (𝕜 := Real) C score) / 2) := by
72   dsimp only
73   let C := CFC.sqrt V0⁻¹
74   have hC : C.PosSemidef := by
75     exact (CFC.sqrt_nonneg V0⁻¹).posSemidef
76   have hCGC : (C * G * C).PosSemidef := by
77     have hcongr := hG.conjTranspose_mul_mul_same C
78     simpa only [hC.isHermitian.eq] using hcongr
79   rw [ProbabilityTheory.multivariateGaussian, MeasureTheory.integral_map]
80   · simp only [zero_add]
81     have hpoint :
82         (fun z : EuclideanSpace Real Feature =>
83           Real.exp
84             (⟪score, Matrix.toEuclideanCLM (𝕜 := Real) C z⟫_ℝ -
85               ⟪Matrix.toEuclideanCLM (𝕜 := Real) C z,
86                 Matrix.toEuclideanCLM (𝕜 := Real) G
87                   (Matrix.toEuclideanCLM (𝕜 := Real) C z)⟫_ℝ / 2)) =
88           (fun z : EuclideanSpace Real Feature =>
89             Real.exp
90               (⟪Matrix.toEuclideanCLM (𝕜 := Real) C score, z⟫_ℝ -
91                 ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) (C * G * C) z⟫_ℝ /
92                   2)) := by
93       funext z
94       rw [inner_toEuclideanCLM_selfAdjoint C hC.isHermitian,
95         inner_toEuclideanCLM_congruence C G hC.isHermitian]
96     rw [hpoint]
97     exact integral_exp_inner_sub_quadratic_stdGaussian_det
98       (C * G * C) hCGC
99         (Matrix.toEuclideanCLM (𝕜 := Real) C score)

101   · fun_prop

103 private theorem sqrt_inv_congruence_factorization
104     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef) :
105     let D := CFC.sqrt V0
106     let C := CFC.sqrt V0⁻¹
107     D * (1 + C * G * C) * D = V0 + G := by
108   dsimp only
109   let D := CFC.sqrt V0
110   let C := CFC.sqrt V0⁻¹
111   have hDpos : D.PosDef := hV0.isStrictlyPositive.sqrt.posDef
112   have hdetD : IsUnit D.det :=
113     D.isUnit_iff_isUnit_det.mp hDpos.isUnit
114   have hCeq : C = D⁻¹ := by
115     exact hV0.posSemidef.inv_sqrt.symm
116   have hDC : D * C = 1 := by
117     rw [hCeq, Matrix.mul_nonsing_inv D hdetD]
118   have hCD : C * D = 1 := by
119     rw [hCeq, Matrix.nonsing_inv_mul D hdetD]
120   have hDD : D * D = V0 := by
121     exact CFC.sqrt_mul_sqrt_self V0 hV0.posSemidef.nonneg
122   calc
123     D * (1 + C * G * C) * D =
124         D * D + (D * C) * G * (C * D) := by
125       noncomm_ring
126     _ = V0 + G := by
127       rw [hDD, hDC, hCD]
128       simp

130 theorem det_one_add_sqrt_inv_congruence_eq_ratio
131     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef) :
132     let C := CFC.sqrt V0⁻¹
133     Matrix.det (1 + C * G * C) =
134       Matrix.det (V0 + G) / Matrix.det V0 := by
135   dsimp only
136   let D := CFC.sqrt V0
137   let C := CFC.sqrt V0⁻¹
138   have hfactor :
139       D * (1 + C * G * C) * D = V0 + G :=
140     sqrt_inv_congruence_factorization V0 G hV0
141   have hDD : D * D = V0 := by
142     exact CFC.sqrt_mul_sqrt_self V0 hV0.posSemidef.nonneg
143   have hdetV0 : Matrix.det V0 ≠ 0 :=
144     ne_of_gt hV0.det_pos
145   have hdetDD :
146       Matrix.det V0 = Matrix.det D * Matrix.det D := by
147     calc
148       Matrix.det V0 = Matrix.det (D * D) := by rw [hDD]
149       _ = Matrix.det D * Matrix.det D := by rw [Matrix.det_mul]
150   have hdetFactor :
151       Matrix.det (V0 + G) =
152         Matrix.det D * Matrix.det (1 + C * G * C) * Matrix.det D := by
153     calc
154       Matrix.det (V0 + G) =
155           Matrix.det (D * (1 + C * G * C) * D) := by rw [hfactor]
156       _ = Matrix.det D * Matrix.det (1 + C * G * C) * Matrix.det D := by
157         rw [Matrix.det_mul, Matrix.det_mul]
158   apply (eq_div_iff hdetV0).2
159   rw [hdetDD, hdetFactor]
160   ring

162 theorem sqrt_inv_congruence_inverse
163     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef) :
164     let C := CFC.sqrt V0⁻¹
165     C * (1 + C * G * C)⁻¹ * C = (V0 + G)⁻¹ := by
166   dsimp only
167   let D := CFC.sqrt V0
168   let C := CFC.sqrt V0⁻¹
169   let B := 1 + C * G * C
170   have hCeq : C = D⁻¹ := by
171     exact hV0.posSemidef.inv_sqrt.symm
172   have hfactor : D * B * D = V0 + G := by
173     exact sqrt_inv_congruence_factorization V0 G hV0
174   calc
175     C * B⁻¹ * C = D⁻¹ * B⁻¹ * D⁻¹ := by rw [hCeq]
176     _ = (D * B * D)⁻¹ := by
177       symm
178       calc
179         (D * B * D)⁻¹ = D⁻¹ * (D * B)⁻¹ := by
180           exact Matrix.mul_inv_rev (D * B) D
181         _ = D⁻¹ * (B⁻¹ * D⁻¹) := by
182           rw [Matrix.mul_inv_rev]
183         _ = D⁻¹ * B⁻¹ * D⁻¹ := by
184           rw [Matrix.mul_assoc]
185     _ = (V0 + G)⁻¹ := by rw [hfactor]

187 theorem dotProduct_sqrt_inv_congruence_inverse
188     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef)
189     (score : EuclideanSpace Real Feature) :
190     let C := CFC.sqrt V0⁻¹
191     (Matrix.toEuclideanCLM (𝕜 := Real) C score) ⬝ᵥ
192         (1 + C * G * C)⁻¹.mulVec
193           (Matrix.toEuclideanCLM (𝕜 := Real) C score) =
194       score ⬝ᵥ (V0 + G)⁻¹.mulVec score := by
195   dsimp only
196   let C := CFC.sqrt V0⁻¹
197   let B := 1 + C * G * C
198   have hC : C.PosSemidef := by
199     exact (CFC.sqrt_nonneg V0⁻¹).posSemidef
200   have hinv : C * B⁻¹ * C = (V0 + G)⁻¹ := by
201     exact sqrt_inv_congruence_inverse V0 G hV0
202   rw [← Matrix.inner_toEuclideanCLM, ← Matrix.inner_toEuclideanCLM]
203   calc
204     ⟪Matrix.toEuclideanCLM (𝕜 := Real) C score,
205         Matrix.toEuclideanCLM (𝕜 := Real) B⁻¹
206           (Matrix.toEuclideanCLM (𝕜 := Real) C score)⟫_ℝ =
207       ⟪score, Matrix.toEuclideanCLM (𝕜 := Real) C
208         (Matrix.toEuclideanCLM (𝕜 := Real) B⁻¹
209           (Matrix.toEuclideanCLM (𝕜 := Real) C score))⟫_ℝ := by
210       rw [real_inner_comm,
211         inner_toEuclideanCLM_selfAdjoint C hC.isHermitian, real_inner_comm]
212     _ = ⟪score,
213         Matrix.toEuclideanCLM (𝕜 := Real) (C * B⁻¹ * C) score⟫_ℝ := by
214       congr 1
215       apply WithLp.ofLp_injective
216       simp only [Matrix.ofLp_toEuclideanCLM, Matrix.mulVec_mulVec]
217       rw [Matrix.mul_assoc]
218     _ = ⟪score,
219         Matrix.toEuclideanCLM (𝕜 := Real) (V0 + G)⁻¹ score⟫_ℝ := by
220       rw [hinv]

222 theorem integral_exp_inner_sub_quadratic_multivariateGaussian_zero_inv_detRatio
223     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef)
224     (hG : G.PosSemidef) (score : EuclideanSpace Real Feature) :
225     integral
226         (ProbabilityTheory.multivariateGaussian 0 V0⁻¹)
227         (fun z : EuclideanSpace Real Feature =>
228           Real.exp

230               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2)) =
231       (Real.sqrt (Matrix.det (V0 + G) / Matrix.det V0))⁻¹ *
232         Real.exp
233           (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2) := by

240 theorem integral_exp_inner_sub_quadratic_multivariateGaussian_zero_inv
241     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef)
242     (hG : G.PosSemidef) (score : EuclideanSpace Real Feature) :
243     integral
244         (ProbabilityTheory.multivariateGaussian 0 V0⁻¹)
245         (fun z : EuclideanSpace Real Feature =>
246           Real.exp
247             (⟪score, z⟫_ℝ -
248               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2)) =
249       Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
250         Real.exp
251           (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2) := by
252   rw [
253     integral_exp_inner_sub_quadratic_multivariateGaussian_zero_inv_detRatio
254       V0 G hV0 hG score]
255   congr 1
256   have hV0det : 0 < Matrix.det V0 := hV0.det_pos
257   have hsumdet : 0 < Matrix.det (V0 + G) :=
258     (hV0.add_posSemidef hG).det_pos
259   rw [Real.sqrt_div (le_of_lt hsumdet),
260     Real.sqrt_div (le_of_lt hV0det)]
261   field_simp [ne_of_gt (Real.sqrt_pos.2 hV0det),
262     ne_of_gt (Real.sqrt_pos.2 hsumdet)]
