21 /-- The complete scalar budget displayed by the power-of-two forced tail. -/
22 noncomputable def powerOfTwoForcedScalarHighProbabilityPseudoRegretBound
23     {Feature : Type u} [Fintype Feature]

30 /--
31 Increasing the outer confidence budget decreases the explicit telescoping
32 confidence logarithm.
33 -/
34 theorem telescopingHighProbabilityRegretLogBudget_anti_delta
35     {Feature : Type u} [Fintype Feature]
36     (lambda L2 : Real)
37     {deltaSmall deltaLarge : Real}
38     (hdeltaSmall : 0 < deltaSmall)
39     (hdelta : deltaSmall <= deltaLarge)
40     (horizon : Nat) :
41     telescopingHighProbabilityRegretLogBudget
42         (Feature := Feature) lambda deltaLarge horizon L2 <=
43       telescopingHighProbabilityRegretLogBudget
44         (Feature := Feature) lambda deltaSmall horizon L2 := by
45   have hdeltaLarge : 0 < deltaLarge := hdeltaSmall.trans_le hdelta
46   have hnum_nonneg :
47       0 <=
48         (((horizon + 1 : Nat) : Real) *
49           ((horizon + 2 : Nat) : Real)) := by
50     positivity
51   have hnum_pos :
52       0 <
53         (((horizon + 1 : Nat) : Real) *
54           ((horizon + 2 : Nat) : Real)) := by
55     positivity
56   have hratio :
57       ((((horizon + 1 : Nat) : Real) *
58             ((horizon + 2 : Nat) : Real)) / deltaLarge) <=
59         ((((horizon + 1 : Nat) : Real) *
60             ((horizon + 2 : Nat) : Real)) / deltaSmall) :=
61     div_le_div_of_nonneg_left hnum_nonneg hdeltaSmall hdelta
62   have hlog :
63       Real.log
64           ((((horizon + 1 : Nat) : Real) *
65               ((horizon + 2 : Nat) : Real)) / deltaLarge) <=
66         Real.log
67           ((((horizon + 1 : Nat) : Real) *
68               ((horizon + 2 : Nat) : Real)) / deltaSmall) :=
69     Real.log_le_log (div_pos hnum_pos hdeltaLarge) hratio
70   unfold telescopingHighProbabilityRegretLogBudget
71   exact add_le_add le_rfl
72     (mul_le_mul_of_nonneg_left hlog (by norm_num : (0 : Real) <= 2))

85       telescopingHighProbabilityRegretLogBudget

88         Real.log (((horizon + 1 : Nat) : Real))) := by

91         telescopingStandardExpectedRegretLogBudget

95     telescopingStandardExpectedRegretLogBudget_isBigO_log

98       (fun horizon : Nat =>
99         telescopingHighProbabilityRegretLogBudget
100           (Feature := Feature) lambda delta horizon L2) =O[atTop]
101         (fun horizon : Nat =>
102           telescopingStandardExpectedRegretLogBudget
103             (Feature := Feature) lambda horizon L2) := by

105     refine ⟨1, ?_⟩

110     have hthreshold_le :
111         (threshold : Real) <= (((horizon + 1 : Nat) : Real)) := by
112       exact_mod_cast le_trans hhorizon (Nat.le_add_right horizon 1)

114         1 / delta < (((horizon + 1 : Nat) : Real)) :=
115       ((le_max_left (1 / delta) delta).trans_lt hthreshold).trans_le
116         hthreshold_le
117     have hdelta_succ :
118         delta < (((horizon + 1 : Nat) : Real)) :=
119       ((le_max_right (1 / delta) delta).trans_lt hthreshold).trans_le
120         hthreshold_le
121     have hshare :
122         standardExpectedRegretDelta horizon <= delta := by
123       unfold standardExpectedRegretDelta
124       rw [div_le_iff₀ (by positivity :
125         0 < (((horizon + 1 : Nat) : Real)))]
126       have hmul :
127           1 < (((horizon + 1 : Nat) : Real)) * delta :=
128         (div_lt_iff₀ hdelta).mp hscale
129       nlinarith
130     have hfixed_nonneg :
131         0 <=
132           telescopingHighProbabilityRegretLogBudget
133             (Feature := Feature) lambda delta horizon L2 := by
134       have hdet :
135           0 <= standardScalarLogDetBudget
136             (Feature := Feature) lambda (horizon + 1) L2 :=
137         standardScalarLogDetBudget_nonneg
138           (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
139       have hfirst_nonneg :

144       have hdelta_prod :
145           delta <=
146             (((horizon + 1 : Nat) : Real) *
147               ((horizon + 2 : Nat) : Real)) :=
148         (le_of_lt hdelta_succ).trans
149           (le_mul_of_one_le_right hfirst_nonneg hsecond_one)
150       have hratio_one :
151           (1 : Real) <=
152             ((((horizon + 1 : Nat) : Real) *
153                 ((horizon + 2 : Nat) : Real)) / delta) := by
154         rw [le_div_iff₀ hdelta]
155         simpa using hdelta_prod
156       unfold telescopingHighProbabilityRegretLogBudget
157       exact add_nonneg hdet
158         (mul_nonneg (by norm_num) (Real.log_nonneg hratio_one))
159     have hscheduled_nonneg :
160         0 <=
161           telescopingStandardExpectedRegretLogBudget
162             (Feature := Feature) lambda horizon L2 := by
163       rw [←
164         telescopingHighProbabilityRegretLogBudget_standardExpectedRegretDelta
165           (Feature := Feature) lambda horizon L2]
166       exact
167         telescopingHighProbabilityRegretLogBudget_nonneg
168           (Feature := Feature) lambda hlambda
169           (standardExpectedRegretDelta horizon)
170           (standardExpectedRegretDelta_pos horizon)
171           (standardExpectedRegretDelta_le_one horizon)
172           horizon L2 hL2
173     have hbudget :
174         telescopingHighProbabilityRegretLogBudget
175             (Feature := Feature) lambda delta horizon L2 <=
176           telescopingStandardExpectedRegretLogBudget
177             (Feature := Feature) lambda horizon L2 := by
178       rw [←
179         telescopingHighProbabilityRegretLogBudget_standardExpectedRegretDelta
180           (Feature := Feature) lambda horizon L2]
181       exact
182         telescopingHighProbabilityRegretLogBudget_anti_delta
183           (Feature := Feature) lambda L2
184           (standardExpectedRegretDelta_pos horizon) hshare horizon
185     rw [Real.norm_eq_abs, abs_of_nonneg hfixed_nonneg,
186       Real.norm_eq_abs, abs_of_nonneg hscheduled_nonneg, one_mul]

202     (fun horizon : Nat =>
203       telescopingHighProbabilityPseudoRegretBound
204         (Feature := Feature) R delta lambda S horizon L2) =O[atTop]
205       (fun horizon : Nat =>
206         Real.sqrt (((horizon + 1 : Nat) : Real)) *
207           Real.log (((horizon + 1 : Nat) : Real))) := by
208   let n : Nat -> Real := fun horizon => (((horizon + 1 : Nat) : Real))
209   let logN : Nat -> Real := fun horizon => Real.log (n horizon)
210   let sqrtN : Nat -> Real := fun horizon => Real.sqrt (n horizon)
211   let B : Nat -> Real := fun horizon =>
212     standardScalarLogDetBudget

215     telescopingHighProbabilityRegretLogBudget
216       (Feature := Feature) lambda delta horizon L2
217   have hlog_nonneg :
218       ∀ᶠ horizon in atTop, 0 <= logN horizon := by

220       dsimp [logN, n]
221       exact Real.log_nonneg (by
222         exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon))
223   have hB : B =O[atTop] logN := by
224     simpa [B, logN, n] using
225       standardScalarLogDetBudget_isBigO_log
226         (Feature := Feature) lambda hlambda L2 hL2
227   have hC : C =O[atTop] logN := by
228     simpa [C, logN, n] using
229       telescopingHighProbabilityRegretLogBudget_isBigO_log_succ
230         (Feature := Feature) lambda hlambda
231         delta hdelta L2 hL2
232   have hsqrtC :
233       (fun horizon => Real.sqrt (C horizon)) =O[atTop]
234         (fun horizon => Real.sqrt (logN horizon)) :=
235     hC.sqrt hlog_nonneg
236   have hsqrt2B :
237       (fun horizon => Real.sqrt (2 * B horizon)) =O[atTop]
238         (fun horizon => Real.sqrt (logN horizon)) :=
239     (hB.const_mul_left 2).sqrt hlog_nonneg
240   have hsqrtLogLog :
241       (fun horizon => Real.sqrt (logN horizon)) =O[atTop] logN := by
242     simpa [logN, n] using sqrt_log_succ_isBigO_log_succ
243   have hsqrtN : sqrtN =O[atTop] sqrtN :=
244     Asymptotics.isBigO_refl _ _
245   have hsqrtN_sqrtLog_scale :
246       (fun horizon => sqrtN horizon * Real.sqrt (logN horizon)) =O[atTop]
247         (fun horizon => sqrtN horizon * logN horizon) :=
248     hsqrtN.mul hsqrtLogLog
249   have hvariance :
250       (fun horizon => sqrtN horizon * Real.sqrt (2 * B horizon)) =O[atTop]
251         (fun horizon => sqrtN horizon * logN horizon) :=
252     (hsqrtN.mul hsqrt2B).trans hsqrtN_sqrtLog_scale
253   have hconfidenceVariance :
254       (fun horizon =>
255         Real.sqrt (C horizon) *
256           (sqrtN horizon * Real.sqrt (2 * B horizon))) =O[atTop]
257         (fun horizon => sqrtN horizon * logN horizon) := by
258     have hproduct := hsqrtC.mul hsqrt2B
259     have hproductLog :
260         (fun horizon =>
261           Real.sqrt (C horizon) * Real.sqrt (2 * B horizon)) =O[atTop]
262           logN := by
263       refine hproduct.congr' Filter.EventuallyEq.rfl ?_
264       filter_upwards [hlog_nonneg] with horizon hlog
265       exact Real.mul_self_sqrt hlog
266     have hmul := hsqrtN.mul hproductLog
267     simpa [mul_assoc, mul_left_comm, mul_comm] using hmul
268   have hn_tendsto : Tendsto n atTop atTop := by
269     dsimp [n]
270     exact (tendsto_add_atTop_iff_nat 1).2 tendsto_natCast_atTop_atTop
271   have hconstantLog :
272       (fun _horizon : Nat => 2 * S * Real.sqrt L2) =o[atTop] logN := by
273     change
274       (fun _horizon : Nat => 2 * S * Real.sqrt L2) =o[atTop]
275         (fun horizon => Real.log (n horizon))
276     simpa only [Function.comp_apply] using
277       (Real.isLittleO_const_log_atTop
278         (c := 2 * S * Real.sqrt L2)).comp_tendsto hn_tendsto
279   have hlogScale :
280       logN =O[atTop] (fun horizon => sqrtN horizon * logN horizon) := by
281     refine (Asymptotics.isBigO_refl logN atTop).trans_le ?_
282     intro horizon
283     have hn_one : 1 <= n horizon := by
284       dsimp [n]
285       exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
286     have hsqrt_one : 1 <= sqrtN horizon := by
287       dsimp [sqrtN]
288       rw [← Real.sqrt_one]
289       exact Real.sqrt_le_sqrt hn_one
290     have hlog : 0 <= logN horizon := by
291       dsimp [logN]
292       exact Real.log_nonneg hn_one
293     rw [Real.norm_eq_abs, abs_of_nonneg hlog,
294       Real.norm_eq_abs, abs_of_nonneg
295         (mul_nonneg (Real.sqrt_nonneg _) hlog)]
296     nlinarith
297   have hconstant :
298       (fun _horizon : Nat => 2 * S * Real.sqrt L2) =O[atTop]
299         (fun horizon => sqrtN horizon * logN horizon) :=
300     hconstantLog.isBigO.trans hlogScale
301   have hconfidenceTerm :
302       (fun horizon =>
303         2 * R *
304           (Real.sqrt (C horizon) *
305             (sqrtN horizon * Real.sqrt (2 * B horizon)))) =O[atTop]
306         (fun horizon => sqrtN horizon * logN horizon) :=
307     hconfidenceVariance.const_mul_left (2 * R)
308   have hbiasTerm :
309       (fun horizon =>
310         (2 * (Real.sqrt lambda * S)) *
311           (sqrtN horizon * Real.sqrt (2 * B horizon))) =O[atTop]
312         (fun horizon => sqrtN horizon * logN horizon) :=
313     hvariance.const_mul_left (2 * (Real.sqrt lambda * S))
314   have hsum := hconstant.add (hconfidenceTerm.add hbiasTerm)
315   refine hsum.congr (fun horizon => ?_) (fun _ => rfl)
316   simp only [telescopingHighProbabilityPseudoRegretBound]
317   dsimp [B, C, sqrtN, logN, n]
318   ring
