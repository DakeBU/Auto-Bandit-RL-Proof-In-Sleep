18 /-- Ordinary finite-dimensional Cauchy--Schwarz on the local linear-value surface. -/
19 theorem abs_linearValue_le_euclideanLength_mul_euclideanLength
20     {Feature : Type u} [Fintype Feature]
21     (theta x : Feature -> Real) :
22     |linearValue theta x| <= euclideanLength theta * euclideanLength x := by
23   have hupper :
24       linearValue theta x <=
25         euclideanLength theta * euclideanLength x := by
26     have h :=
27       Real.sum_mul_le_sqrt_mul_sqrt
28         (Finset.univ : Finset Feature) theta x
29     simpa [linearValue, euclideanLength, dotProduct, pow_two] using h
30   have hnegative :
31       -linearValue theta x <=
32         euclideanLength theta * euclideanLength x := by
33     have h :=
34       Real.sum_mul_le_sqrt_mul_sqrt
35         (Finset.univ : Finset Feature) (fun i => -theta i) x
36     simpa [linearValue, euclideanLength, dotProduct, pow_two] using h
37   exact abs_le.2 ⟨by linarith, hupper⟩

39 /-- A common parameter/feature envelope bounds the absolute linear value. -/
40 theorem abs_linearValue_le_parameterFeatureBound
41     {Feature : Type u} [Fintype Feature]
42     (theta x : Feature -> Real)
43     (S L2 : Real) (hS : 0 <= S)
44     (htheta : euclideanLength theta <= S)
45     (hx : dotProduct x x <= L2) :
46     |linearValue theta x| <= S * Real.sqrt L2 := by
47   have hxLength : euclideanLength x <= Real.sqrt L2 := by
48     exact Real.sqrt_le_sqrt hx
49   exact
50     (abs_linearValue_le_euclideanLength_mul_euclideanLength theta x).trans
51       (mul_le_mul htheta hxLength (Real.sqrt_nonneg _) hS)

53 /-- Two arms sharing the same feature envelope differ by at most `2*S*sqrt L2`. -/
54 theorem linearValue_sub_linearValue_le_two_mul_parameterFeatureBound
55     {Feature : Type u} [Fintype Feature]
56     (theta x y : Feature -> Real)
57     (S L2 : Real) (hS : 0 <= S)
58     (htheta : euclideanLength theta <= S)
59     (hx : dotProduct x x <= L2)
60     (hy : dotProduct y y <= L2) :
61     linearValue theta x - linearValue theta y <=
62       2 * S * Real.sqrt L2 := by
63   have hxAbs :=
64     abs_linearValue_le_parameterFeatureBound theta x S L2 hS htheta hx
65   have hyAbs :=
66     abs_linearValue_le_parameterFeatureBound theta y S L2 hS htheta hy
67   have hxUpper := (abs_le.mp hxAbs).2
68   have hyLower := (abs_le.mp hyAbs).1
69   linarith

71 /-- Deterministic charge used for the canonical time-zero arm. -/
72 noncomputable def standardScalarInitialGapBound (S L2 : Real) : Real :=
73   2 * S * Real.sqrt L2

116 /-- Standard finite-window cumulative-gap budget including time zero. -/
117 noncomputable def standardScalarAllRoundGapBound
118     {Feature : Type u} [Fintype Feature]
119     (R delta lambda S : Real) (horizon : Nat) (L2 : Real) : Real :=
120   standardScalarInitialGapBound S L2 +
121     standardScalarRadiusWidthBound
122       (Feature := Feature)
123       R (delta / ((horizon + 1 : Nat) : Real))
124       lambda S (horizon + 1) L2
