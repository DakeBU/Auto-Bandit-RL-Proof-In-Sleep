4 /-!
5 # All-time realized EXP3 regret
6 
7 This module combines the accepted same-process predictable-regret and pure
8 realized-deviation all-time events. The total confidence budget is split
9 equally between those two event families. The result is an outer-measure
10 bound for realized selected-loss regret at every positive prefix, with fixed
11 process parameters and one fixed supported comparator.
12 -/

21 /-- Exact finite-prefix decomposition of realized selected-loss regret into
22 exploration-mixed predictable regret and realized deviation. -/
23 theorem sampledTrajectoryRealizedRegret_eq_predictableRegret_add_realizedDeviation
24     {Env : Type u} {Action : Type v}
25     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
26     (arms : Finset Action) (eta gamma : Real)
27     (loss : PredictableLossVector Env Action) (comparator : Action)
28     (horizon : Nat) (sample : Env × ((k : Nat) -> Action × Real)) :
29     (Finset.range horizon).sum (fun t =>
30         sampledTrajectoryRealizedLossAt t sample) -
31       (Finset.range horizon).sum (fun t =>
32         predictableLossAt loss t sample comparator) =
33       ((Finset.range horizon).sum (fun t =>
34           sampledTrajectoryExploredPredictableLossAt
35             arms eta gamma loss t sample) -
36         (Finset.range horizon).sum (fun t =>
37           predictableLossAt loss t sample comparator)) +
38       (Finset.range horizon).sum (fun t =>
39         sampledTrajectoryRealizedDeviationAt
40           arms eta gamma loss t sample) := by
41   simp only [sampledTrajectoryRealizedDeviationAt, Finset.sum_sub_distrib]
42   ring

44 /-- At prefix `n+1`, add the predictable-regret and pure realized-deviation
45 schedules after assigning half of the total confidence budget to each event
46 family. -/
47 noncomputable def sampledRealizedRegretGeometricAllTimeBudget
48     {Action : Type v} (arms : Finset Action) (eta gamma delta : Real)
49     (n : Nat) : Real :=
50   sampledPredictableRegretGeometricAllTimeBudget
51       arms eta gamma (delta / 2) n +
52     sampledRealizedDeviationGeometricAllTimeRadius (delta / 2) n
53 
54 /-- Countable realized selected-loss regret failure event over every positive
55 prefix of one generated EXP3 trajectory. -/
56 noncomputable def sampledRealizedRegretGeometricAllTimeFailureSet
57     {Env : Type u} {Action : Type v}
58     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
59     (arms : Finset Action) (eta gamma : Real)
60     (loss : PredictableLossVector Env Action) (comparator : Action)
61     (delta : Real) : Set (Env × ((k : Nat) -> Action × Real)) :=
62   ⋃ n, {sample |
63     sampledRealizedRegretGeometricAllTimeBudget
64           arms eta gamma delta n <=
65         (Finset.range (n + 1)).sum (fun t =>
66             sampledTrajectoryRealizedLossAt t sample) -
67           (Finset.range (n + 1)).sum (fun t =>
68             predictableLossAt loss t sample comparator)}
69 
70 /-- Membership is realized selected-loss regret failure at at least one
71 positive prefix. -/
72 theorem mem_sampledRealizedRegretGeometricAllTimeFailureSet_iff
73     {Env : Type u} {Action : Type v}
74     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
75     (arms : Finset Action) (eta gamma : Real)
76     (loss : PredictableLossVector Env Action) (comparator : Action)
77     (delta : Real) (sample : Env × ((k : Nat) -> Action × Real)) :
78     sample ∈ sampledRealizedRegretGeometricAllTimeFailureSet
79         arms eta gamma loss comparator delta ↔
80       ∃ n,
81         sampledRealizedRegretGeometricAllTimeBudget
82               arms eta gamma delta n <=
83             (Finset.range (n + 1)).sum (fun t =>
84                 sampledTrajectoryRealizedLossAt t sample) -
85               (Finset.range (n + 1)).sum (fun t =>
86                 predictableLossAt loss t sample comparator) := by
87   simp [sampledRealizedRegretGeometricAllTimeFailureSet]

89 /-- A combined realized-regret crossing forces either a predictable-regret
90 crossing or a pure realized-deviation crossing at the same prefix. -/
91 theorem sampledRealizedRegretGeometricAllTimeFailureSet_subset
92     {Env : Type u} {Action : Type v}
93     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
94     (arms : Finset Action) (eta gamma : Real)
95     (loss : PredictableLossVector Env Action) (comparator : Action)
96     (delta : Real) :
97     sampledRealizedRegretGeometricAllTimeFailureSet
98         arms eta gamma loss comparator delta ⊆
99       sampledPredictableRegretGeometricAllTimeFailureSet
100           arms eta gamma loss comparator (delta / 2) ∪
101         sampledRealizedDeviationGeometricAllTimeFailureSet
102           arms eta gamma loss (delta / 2) := by
103   intro sample hsample
104   rw [mem_sampledRealizedRegretGeometricAllTimeFailureSet_iff] at hsample
105   obtain ⟨n, hregret⟩ := hsample
106   by_cases hpredictable :
107       sample ∈ sampledPredictableRegretGeometricAllTimeFailureSet
108         arms eta gamma loss comparator (delta / 2)
109   · exact Or.inl hpredictable
110   by_cases hdeviation :
111       sample ∈ sampledRealizedDeviationGeometricAllTimeFailureSet
112         arms eta gamma loss (delta / 2)
113   · exact Or.inr hdeviation
114   exfalso
115   rw [mem_sampledPredictableRegretGeometricAllTimeFailureSet_iff] at hpredictable
116   rw [mem_sampledRealizedDeviationGeometricAllTimeFailureSet_iff] at hdeviation
117   have hpredictableGood :
118       (Finset.range (n + 1)).sum (fun t =>
119           sampledTrajectoryExploredPredictableLossAt
120             arms eta gamma loss t sample) -
121         (Finset.range (n + 1)).sum (fun t =>
122           predictableLossAt loss t sample comparator) <
123       sampledPredictableRegretGeometricAllTimeBudget
124         arms eta gamma (delta / 2) n := by
125     apply lt_of_not_ge
126     intro hcross
127     exact hpredictable ⟨n, hcross⟩
128   have hdeviationGood :
129       (Finset.range (n + 1)).sum (fun t =>
130           sampledTrajectoryRealizedDeviationAt
131             arms eta gamma loss t sample) <
132         sampledRealizedDeviationGeometricAllTimeRadius (delta / 2) n := by
133     apply lt_of_not_ge
134     intro hcross
135     exact hdeviation ⟨n, hcross⟩
136   rw [sampledTrajectoryRealizedRegret_eq_predictableRegret_add_realizedDeviation
137     arms eta gamma loss comparator (n + 1) sample] at hregret
138   unfold sampledRealizedRegretGeometricAllTimeBudget at hregret
139   linarith

141 /-- On one fixed generated EXP3 process and against one fixed supported
142 comparator, realized selected-loss regret stays below the sum of the scheduled
143 predictable-regret and realized-deviation budgets at every positive prefix,
144 outside a set of outer measure at most the total confidence budget. -/
145 theorem measure_sampledRealizedRegretGeometricAllTimeFailureSet_le
146     {Env : Type u} {Action : Type v}
147     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
148     [MeasurableSpace Action] [MeasurableSingletonClass Action]
149     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
150     (prior : Measure Env) [IsProbabilityMeasure prior]
151     (arms : Finset Action) (harms : arms.Nonempty)
152     (eta gamma : Real) (heta : 0 < eta)
153     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
154     (loss : PredictableLossVector Env Action)
155     (comparator : Action) (hcomparator : comparator ∈ arms)
156     (delta : Real) (hdelta : 0 < delta) :
157     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
158       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
159     mu (sampledRealizedRegretGeometricAllTimeFailureSet
160         arms eta gamma loss comparator delta) <= ENNReal.ofReal delta := by
161   dsimp only
162   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
163     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
164   have hpredictable :=
165     measure_sampledPredictableRegretGeometricAllTimeFailureSet_le
166       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
167         hcomparator (delta / 2) (half_pos hdelta)
168   have hdeviation :=
169     measure_sampledRealizedDeviationGeometricAllTimeFailureSet_le
170       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss
171         (delta / 2) (half_pos hdelta)
172   dsimp only at hpredictable hdeviation
173   calc
174     mu (sampledRealizedRegretGeometricAllTimeFailureSet
175         arms eta gamma loss comparator delta) <=
176         mu (sampledPredictableRegretGeometricAllTimeFailureSet
177               arms eta gamma loss comparator (delta / 2) ∪
178             sampledRealizedDeviationGeometricAllTimeFailureSet
179               arms eta gamma loss (delta / 2)) := by
180       exact measure_mono
181         (sampledRealizedRegretGeometricAllTimeFailureSet_subset
182           arms eta gamma loss comparator delta)
183     _ <=
184         mu (sampledPredictableRegretGeometricAllTimeFailureSet
185               arms eta gamma loss comparator (delta / 2)) +
186           mu (sampledRealizedDeviationGeometricAllTimeFailureSet
187               arms eta gamma loss (delta / 2)) := measure_union_le _ _
188     _ <= ENNReal.ofReal (delta / 2) + ENNReal.ofReal (delta / 2) :=
189       add_le_add hpredictable hdeviation
190     _ = ENNReal.ofReal delta := by
191       rw [← ENNReal.ofReal_add (half_pos hdelta).le (half_pos hdelta).le,
192         add_halves]
