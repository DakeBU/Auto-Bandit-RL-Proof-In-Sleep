3 /-!
4 # Exploration bias for concrete sampled EXP3
5 
6 This module compares the pure exponential-weights distribution `q_t` used by
7 the Hedge potential with the exploration-mixed sampling distribution
8 `p_t = (1 - gamma) q_t + gamma / |arms|`.
9 
10 The final finite-horizon theorem gives both inequalities needed before the
11 almost-sure Hedge bound can be integrated: actual predictable loss is at most
12 pure-distribution predictable loss plus `gamma` per round, and the pure
13 estimator square is controlled by the actual probability-mixed estimator
14 square with factor `1 / (1 - gamma)`.
15 -/

22 /-- A pure Hedge coordinate is at most the corresponding explored probability
23 divided by `1 - gamma`. -/
24 theorem distribution_le_sampledTrajectoryProbabilityAt_div_one_sub_gamma
25     {Env : Type u} {Action : Type v} [DecidableEq Action]
26     (arms : Finset Action)
27     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
28     (hgamma_lt_one : gamma < 1) (t : Nat)
29     (sample : Env × ((k : Nat) -> Action × Real)) (action : Action) :
30     distribution arms eta
31         (sampledTrajectoryObservedLoss arms eta gamma sample) t action <=
32       sampledTrajectoryProbabilityAt arms eta gamma t sample action /
33         (1 - gamma) := by
34   have hmix := sampledTrajectoryProbabilityAt_eq_mix_distribution
35     arms eta gamma t sample action
36   have hfloor_nonneg : 0 <= gamma / (arms.card : Real) :=
37     div_nonneg hgamma_nonneg (Nat.cast_nonneg _)
38   have hscaled :
39       (1 - gamma) *
40           distribution arms eta
41             (sampledTrajectoryObservedLoss arms eta gamma sample) t action <=
42         sampledTrajectoryProbabilityAt arms eta gamma t sample action := by
43     rw [hmix]
44     exact le_add_of_nonneg_right hfloor_nonneg
45   apply (le_div_iff₀ (sub_pos.mpr hgamma_lt_one)).2
46   simpa [mul_comm] using hscaled

48 /-- The pure-Hedge estimator square is bounded by the explored-probability
49 mixed square with the standard `1 / (1 - gamma)` factor. -/
50 theorem mixedSquaredLoss_sampledTrajectoryObservedLoss_le_inv_one_sub_gamma
51     {Env : Type u} {Action : Type v} [DecidableEq Action]
52     (arms : Finset Action)
53     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
54     (hgamma_lt_one : gamma < 1) (t : Nat)
55     (sample : Env × ((k : Nat) -> Action × Real)) :
56     mixedSquaredLoss arms eta
57         (sampledTrajectoryObservedLoss arms eta gamma sample) t <=
58       (1 / (1 - gamma)) *
59         observedMixedSquaredImportanceWeightedLossAt
60           arms eta gamma t sample := by
61   unfold mixedSquaredLoss
62   calc
63     arms.sum (fun action =>
64         distribution arms eta
65             (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
66           (sampledTrajectoryObservedLoss arms eta gamma sample t action) ^ 2) <=
67       arms.sum (fun action =>
68         (sampledTrajectoryProbabilityAt arms eta gamma t sample action /
69             (1 - gamma)) *
70           (sampledTrajectoryObservedLoss arms eta gamma sample t action) ^ 2) := by
71         apply Finset.sum_le_sum
72         intro action _haction
73         exact mul_le_mul_of_nonneg_right
74           (distribution_le_sampledTrajectoryProbabilityAt_div_one_sub_gamma
75             arms eta gamma hgamma_nonneg hgamma_lt_one t sample action)
76           (sq_nonneg _)
77     _ = (1 / (1 - gamma)) *
78         observedMixedSquaredImportanceWeightedLossAt
79           arms eta gamma t sample := by
80       unfold observedMixedSquaredImportanceWeightedLossAt
81         mixedSquaredImportanceWeightedLoss sampledTrajectoryObservedLoss
82         observedImportanceWeightedLossAt
83       rw [Finset.mul_sum]
84       apply Finset.sum_congr rfl
85       intro action _haction
86       ring

88 /-- Exploration changes one predictable `[0,1]` mixed loss by at most
89 `gamma` relative to the pure Hedge distribution. -/
90 theorem sampledTrajectoryPredictableMixedLoss_le_pure_add_gamma
91     {Env : Type u} {Action : Type v}
92     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
93     (arms : Finset Action) (harms : arms.Nonempty)
94     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
95     (loss : PredictableLossVector Env Action) (t : Nat)
96     (sample : Env × ((k : Nat) -> Action × Real)) :
97     arms.sum (fun action =>
98         sampledTrajectoryProbabilityAt arms eta gamma t sample action *
99           predictableLossAt loss t sample action) <=
100       arms.sum (fun action =>
101         distribution arms eta
102             (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
103           predictableLossAt loss t sample action) + gamma := by
104   have hloss (action : Action) :
105       0 <= predictableLossAt loss t sample action ∧
106         predictableLossAt loss t sample action <= 1 := by
107     cases t with
108     | zero =>
109         exact loss.initial_mem_unitInterval sample.1 action
110     | succ n =>
111         exact loss.successor_mem_unitInterval n sample.1
112           (Preorder.frestrictLe n sample.2) action
113   calc
114     arms.sum (fun action =>
115         sampledTrajectoryProbabilityAt arms eta gamma t sample action *
116           predictableLossAt loss t sample action) =
117       arms.sum (fun action =>
118         ((1 - gamma) *
119             distribution arms eta
120               (sampledTrajectoryObservedLoss arms eta gamma sample) t action +
121           gamma / (arms.card : Real)) *
122             predictableLossAt loss t sample action) := by
123         apply Finset.sum_congr rfl
124         intro action _haction
125         rw [sampledTrajectoryProbabilityAt_eq_mix_distribution]
126     _ <= arms.sum (fun action =>
127         distribution arms eta
128             (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
129           predictableLossAt loss t sample action +
130         gamma / (arms.card : Real)) := by
131       apply Finset.sum_le_sum
132       intro action _haction
133       have hq_nonneg := distribution_nonneg arms harms eta
134         (sampledTrajectoryObservedLoss arms eta gamma sample) t action
135       have hq_loss_nonneg := mul_nonneg hq_nonneg (hloss action).1
136       have hgamma_q_loss_nonneg := mul_nonneg hgamma_nonneg hq_loss_nonneg
137       have hfirst :
138           (1 - gamma) *
139               (distribution arms eta
140                 (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
141                 predictableLossAt loss t sample action) <=
142             distribution arms eta
143                 (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
144               predictableLossAt loss t sample action := by
145         nlinarith
146       have huniform_nonneg : 0 <= gamma / (arms.card : Real) :=
147         div_nonneg hgamma_nonneg (Nat.cast_nonneg _)
148       have hsecond :
149           gamma / (arms.card : Real) * predictableLossAt loss t sample action <=
150             gamma / (arms.card : Real) := by
151         exact mul_le_of_le_one_right huniform_nonneg (hloss action).2
152       calc
153         ((1 - gamma) *
154               distribution arms eta
155                 (sampledTrajectoryObservedLoss arms eta gamma sample) t action +
156             gamma / (arms.card : Real)) *
157             predictableLossAt loss t sample action =
158           (1 - gamma) *
159               (distribution arms eta
160                 (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
161                 predictableLossAt loss t sample action) +
162             gamma / (arms.card : Real) *
163               predictableLossAt loss t sample action := by ring
164         _ <= distribution arms eta
165               (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
166               predictableLossAt loss t sample action +
167             gamma / (arms.card : Real) := add_le_add hfirst hsecond
168     _ = arms.sum (fun action =>
169           distribution arms eta
170               (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
171             predictableLossAt loss t sample action) + gamma := by
172       rw [Finset.sum_add_distrib]
173       simp only [Finset.sum_const, nsmul_eq_mul]
174       have hcard_pos : 0 < (arms.card : Real) := by
175         exact_mod_cast harms.card_pos
176       field_simp [ne_of_gt hcard_pos]

178 /-- Finite-horizon exploration bias and second-moment comparison on one
179 concrete sampled trajectory. -/
180 theorem sampledTrajectory_finiteHorizon_explorationBias_secondMoment
181     {Env : Type u} {Action : Type v}
182     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
183     (arms : Finset Action) (harms : arms.Nonempty)
184     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
185     (hgamma_lt_one : gamma < 1)
186     (loss : PredictableLossVector Env Action) (horizon : Nat)
187     (sample : Env × ((k : Nat) -> Action × Real)) :
188     ((Finset.range horizon).sum (fun t =>
189         arms.sum (fun action =>
190           sampledTrajectoryProbabilityAt arms eta gamma t sample action *
191             predictableLossAt loss t sample action)) <=
192       (Finset.range horizon).sum (fun t =>
193         arms.sum (fun action =>
194           distribution arms eta
195               (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
196             predictableLossAt loss t sample action)) +
197         gamma * (horizon : Real)) ∧
198     ((Finset.range horizon).sum (fun t =>
199         mixedSquaredLoss arms eta
200           (sampledTrajectoryObservedLoss arms eta gamma sample) t) <=
201       (1 / (1 - gamma)) * (Finset.range horizon).sum (fun t =>
202         observedMixedSquaredImportanceWeightedLossAt
203           arms eta gamma t sample)) := by
204   constructor
205   · calc
206       (Finset.range horizon).sum (fun t =>
207           arms.sum (fun action =>
208             sampledTrajectoryProbabilityAt arms eta gamma t sample action *
209               predictableLossAt loss t sample action)) <=
210         (Finset.range horizon).sum (fun t =>
211           (arms.sum (fun action =>
212             distribution arms eta
213                 (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
214               predictableLossAt loss t sample action) + gamma)) := by
215           apply Finset.sum_le_sum
216           intro t _ht
217           exact sampledTrajectoryPredictableMixedLoss_le_pure_add_gamma
218             arms harms eta gamma hgamma_nonneg loss t sample
219       _ = (Finset.range horizon).sum (fun t =>
220             arms.sum (fun action =>
221               distribution arms eta
222                   (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
223                 predictableLossAt loss t sample action)) +
224           gamma * (horizon : Real) := by
225         rw [Finset.sum_add_distrib]
226         simp [mul_comm]
227   · calc
228       (Finset.range horizon).sum (fun t =>
229           mixedSquaredLoss arms eta
230             (sampledTrajectoryObservedLoss arms eta gamma sample) t) <=
231         (Finset.range horizon).sum (fun t =>
232           (1 / (1 - gamma)) *
233             observedMixedSquaredImportanceWeightedLossAt
234               arms eta gamma t sample) := by
235           apply Finset.sum_le_sum
236           intro t _ht
237           exact
238             mixedSquaredLoss_sampledTrajectoryObservedLoss_le_inv_one_sub_gamma
239               arms eta gamma hgamma_nonneg hgamma_lt_one t sample
240       _ = (1 / (1 - gamma)) * (Finset.range horizon).sum (fun t =>
241           observedMixedSquaredImportanceWeightedLossAt
242             arms eta gamma t sample) := by
243         rw [Finset.mul_sum]
