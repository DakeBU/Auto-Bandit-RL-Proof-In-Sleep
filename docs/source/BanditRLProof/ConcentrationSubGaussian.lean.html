29 /--
30 A conditionally sub-Gaussian real random variable is integrable under the
31 ambient measure.
32 
33 Mathlib's conditional MGF contract already includes exponential
34 integrability for every real tilt.  Applying it at `1` and `-1` gives
35 integrability of the first absolute moment.
36 -/
37 theorem HasCondSubgaussianMGF.integrable
38     {Omega : Type u} {m mOmega : MeasurableSpace Omega}
39     [StandardBorelSpace Omega]
40     {mu : MeasureTheory.Measure Omega}
41     [MeasureTheory.IsFiniteMeasure mu]
42     {X : Omega -> Real} {c : NNReal}
43     (hm : m <= mOmega)
44     (hX : HasCondSubgaussianMGF m hm X c mu) :
45     MeasureTheory.Integrable X mu := by
46   have h_exp_pos :
47       MeasureTheory.Integrable (fun omega => Real.exp (1 * X omega)) mu :=
48     hX.integrable_exp_mul 1
49   have h_exp_neg :
50       MeasureTheory.Integrable (fun omega => Real.exp (-1 * X omega)) mu :=
51     hX.integrable_exp_mul (-1)
52   have h_abs :
53       MeasureTheory.Integrable (fun omega => |X omega| ^ (1 : Nat)) mu := by

55       (t := (1 : Real)) one_ne_zero h_exp_pos (by simpa using h_exp_neg) 1

57     have h_exp := (hX.integrable_exp_mul 1).aemeasurable

63 /--
64 A conditionally sub-Gaussian variable remains conditionally sub-Gaussian with
65 the same proxy after restriction to an event measurable in the conditioning
66 sigma-algebra.
67 
68 The proof uses one common exceptional set for every exponential tilt: the
69 conditional-expectation kernel is supported on the current side of a
70 conditioning-measurable event, so the indicator-masked variable is kernel-a.e.
71 equal either to the original variable or to zero.
72 -/

363 /--
364 A sub-Gaussian MGF controls the first absolute moment at the natural
365 square-root scale.
366 
367 For a positive proxy, evaluate the MGF at the dimensionless tilt
368 `t = 1 / sqrt c` and use `exp |x| <= exp x + exp (-x)`.  The zero-proxy case
369 is Mathlib's a.e.-zero theorem.  The constant is intentionally simple; the
370 route needs the `sqrt c` scaling rather than the sharp Gaussian constant.
371 -/
372 theorem integral_abs_le_two_mul_sqrt_mul_exp_half_of_hasSubgaussianMGF
373     {Omega : Type u} [MeasurableSpace Omega]
374     (mu : Measure Omega) [IsProbabilityMeasure mu]
375     (X : Omega -> Real) (c : NNReal)
376     (hX : ProbabilityTheory.HasSubgaussianMGF X c mu) :
377     integral mu (fun omega => |X omega|) <=
378       2 * Real.sqrt (c : Real) * Real.exp (1 / 2 : Real) := by
379   by_cases hc : c = 0
380   · subst c
381     have hzero := hX.ae_eq_zero_of_hasSubgaussianMGF_zero
382     have habs : (fun omega => |X omega|) =ᵐ[mu] 0 := by
383       filter_upwards [hzero] with omega homega
384       simp [homega]
385     rw [integral_congr_ae habs]
386     simp
387   · let s : Real := Real.sqrt (c : Real)
388     let t : Real := s⁻¹
389     have hcpos : 0 < (c : Real) := by positivity
390     have hspos : 0 < s := Real.sqrt_pos.2 hcpos
391     have htpos : 0 < t := inv_pos.2 hspos
392     have hst : s * t = 1 := by
393       simp [t, ne_of_gt hspos]
394     have hs_sq : s ^ 2 = (c : Real) := by
395       simp [s, Real.sq_sqrt (le_of_lt hcpos)]
396     have hpoint : forall omega,
397         |X omega| <=
398           s * (Real.exp (t * X omega) + Real.exp (-t * X omega)) := by
399       intro omega
400       have habs_le_exp (x : Real) : |x| <= Real.exp |x| := by
401         calc
402           |x| <= 1 + |x| := by linarith
403           _ = |x| + 1 := by ring
404           _ <= Real.exp |x| := Real.add_one_le_exp |x|
405       calc
406         |X omega| = s * (t * |X omega|) := by rw [← mul_assoc, hst, one_mul]
407         _ = s * |t * X omega| := by
408           rw [abs_mul, abs_of_pos htpos]
409         _ <= s * Real.exp |t * X omega| :=
410           mul_le_mul_of_nonneg_left (habs_le_exp _) hspos.le
411         _ <= s * (Real.exp (t * X omega) + Real.exp (-t * X omega)) :=
412           mul_le_mul_of_nonneg_left (by
413             simpa [neg_mul] using Real.exp_abs_le (t * X omega)) hspos.le
414     have hpos : Integrable (fun omega => Real.exp (t * X omega)) mu :=
415       hX.integrable_exp_mul t
416     have hneg : Integrable (fun omega => Real.exp (-t * X omega)) mu := by
417       simpa [neg_mul] using hX.integrable_exp_mul (-t)
418     have hrhs : Integrable
419         (fun omega => s *
420           (Real.exp (t * X omega) + Real.exp (-t * X omega))) mu :=
421       (hpos.add hneg).const_mul s
422     have hmgf :
423         ProbabilityTheory.mgf X mu t + ProbabilityTheory.mgf X mu (-t) <=
424           Real.exp ((c : Real) * t ^ 2 / 2) +
425             Real.exp ((c : Real) * (-t) ^ 2 / 2) :=
426       add_le_add (hX.mgf_le t) (hX.mgf_le (-t))
427     have hct : (c : Real) * t ^ 2 = 1 := by
428       rw [← hs_sq]
429       calc
430         s ^ 2 * t ^ 2 = (s * t) ^ 2 := by ring
431         _ = 1 := by rw [hst]; norm_num
432     calc
433       integral mu (fun omega => |X omega|) <=
434           integral mu (fun omega => s *
435             (Real.exp (t * X omega) + Real.exp (-t * X omega))) :=
436         integral_mono hX.integrable.abs hrhs hpoint
437       _ = s * (ProbabilityTheory.mgf X mu t +
438             ProbabilityTheory.mgf X mu (-t)) := by
439         simp_rw [integral_const_mul, integral_add hpos hneg]
440         simp [ProbabilityTheory.mgf, neg_mul]
441       _ <= s * (Real.exp ((c : Real) * t ^ 2 / 2) +
442             Real.exp ((c : Real) * (-t) ^ 2 / 2)) :=
443         mul_le_mul_of_nonneg_left hmgf hspos.le
444       _ = 2 * Real.sqrt (c : Real) * Real.exp (1 / 2 : Real) := by
445         rw [neg_sq, hct]
446         simp [s]
447         ring

456 theorem integral_sq_le_four_mul_proxy_mul_exp_half_of_hasSubgaussianMGF
457     {Omega : Type u} [MeasurableSpace Omega]
458     (mu : Measure Omega) [IsProbabilityMeasure mu]
459     (X : Omega -> Real) (c : NNReal)

466     have hsq : (fun omega => X omega ^ 2) =ᵐ[mu] 0 := by

468       simp [homega]

470     simp

472     let t : Real := s⁻¹

474     have hspos : 0 < s := Real.sqrt_pos.2 hcpos
475     have hst : s * t = 1 := by
476       simp [t, ne_of_gt hspos]
477     have hs_sq : s ^ 2 = (c : Real) := by
478       simp [s, Real.sq_sqrt (le_of_lt hcpos)]
479     have hct : (c : Real) * t ^ 2 = 1 := by
480       rw [← hs_sq]

483         _ = 1 := by rw [hst]; norm_num
484     have hquadratic (x : Real) : |x| ^ 2 <= 2 * Real.exp |x| := by

486       nlinarith [abs_nonneg x]
487     have hpoint : forall omega,
488         X omega ^ 2 <=
489           2 * (c : Real) *
490             (Real.exp (t * X omega) + Real.exp (-t * X omega)) := by
491       intro omega
492       calc
493         X omega ^ 2 =
494             (c : Real) * (t * X omega) ^ 2 := by
