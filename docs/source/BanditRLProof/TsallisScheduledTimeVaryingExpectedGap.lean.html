4 /-!
5 # Time-varying expected-gap laws for scheduled half-Tsallis FTRL
6 
7 This module lets the conditional or independent mean loss gap vary with the
8 round.  It is the law surface needed by deterministic predictable corruption
9 schedules: the baseline stochastic gap remains fixed, while clipping a
10 round-dependent reward shift perturbs the actual gap by a known per-round
11 amount.
12 -/

21 /-- Per-time first-moment law for probability-weighted predictable loss gaps. -/
22 def HasScheduledTimeVaryingExpectedGapLaw
23     {Env : Type u} {Action : Type v}
24     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
25     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
26     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
27     (loss : Exp3.PredictableLossVector Env Action)
28     (best : Action) (gap : Nat -> Action -> Real) (horizon : Nat) : Prop :=
29   ∀ t, t <= horizon -> ∀ action, action ∈ arms.erase best ->
30     integral mu (fun sample =>
31         sampledScheduledHalfTsallisProbabilityAtTime
32             arms harms eta t sample action *
33           (Exp3.predictableLossAt loss t sample action -
34             Exp3.predictableLossAt loss t sample best)) =
35       gap t action * sampledScheduledHalfTsallisExpectedProbabilityAt
36         mu arms harms eta t action

73 conditional mean. -/
74 theorem hasScheduledTimeVaryingConditionalMeanGapLaw_of_independentMeanGapLaw
75     {Env : Type u} {Action : Type v}

78     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
79     [IsProbabilityMeasure mu]

88     Exp3.predictableLossAt loss t sample action -
89       Exp3.predictableLossAt loss t sample best
90   let mLoss : MeasurableSpace (Env × ((k : Nat) -> Action × Real)) :=
91     MeasurableSpace.comap lossDiff inferInstance
92   let mPast : MeasurableSpace (Env × ((k : Nat) -> Action × Real)) :=
93     sampledScheduledHalfTsallisPastSigma
94       (Env := Env) (Action := Action) t
95   have hmeasLoss : @Measurable
96       (Env × ((k : Nat) -> Action × Real)) Real
97       Prod.instMeasurableSpace inferInstance lossDiff := by
98     simpa only [lossDiff] using
99       (Exp3.measurable_predictableLossAt loss t action).sub
100         (Exp3.measurable_predictableLossAt loss t best)
101   have hleLoss : mLoss <=
102       (Prod.instMeasurableSpace : MeasurableSpace
103         (Env × ((k : Nat) -> Action × Real))) :=
104     Measurable.comap_le hmeasLoss
105   have hlePast : mPast <=
106       (Prod.instMeasurableSpace : MeasurableSpace
107         (Env × ((k : Nat) -> Action × Real))) := by
108     exact sampledScheduledHalfTsallisPastSigma_le t
109   have hstrongLoss : @StronglyMeasurable
110       (Env × ((k : Nat) -> Action × Real)) Real _ mLoss lossDiff :=
111     (comap_measurable lossDiff).stronglyMeasurable
112   rcases hgap t ht action haction with ⟨hindep, hmean⟩
113   have hcond :=
114     @MeasureTheory.condExp_indep_eq
115       (Env × ((k : Nat) -> Action × Real)) Real _ _ _
116       mLoss mPast Prod.instMeasurableSpace mu lossDiff
117       hleLoss hlePast inferInstance hstrongLoss hindep
118   change condExp mPast mu lossDiff =ᵐ[mu] fun _sample => gap t action
119   exact hcond.mono (fun _sample hsample => by rw [hsample, hmean])

165           (sampledScheduledHalfTsallisPastSigma
166             (Env := Env) (Action := Action) t)

173             (Env := Env) (Action := Action) t)

183             (Env := Env) (Action := Action) t)
184           mu (probability * lossDiff)) := by

188         (sampledScheduledHalfTsallisPastSigma
