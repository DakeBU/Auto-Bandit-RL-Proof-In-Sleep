4 # Scheduled OFUL expected regret under cumulative positive costs
5 
6 This module constructs an adapted Nat-valued resource process by summing
7 per-round costs over completed rounds. Pointwise positive costs give the
8 unit-growth contract required by the compiled budget-exhaustion OFUL theorem.
9 -/

20 /--
21 Resource spent after `t` completed rounds, using the half-open index set
22 `{0, ..., t - 1}`.
23 -/
24 def cumulativeSpent
25     {Omega : Type v}
26     (cost : Nat -> Omega -> Nat)
27     (t : Nat)
28     (omega : Omega) : Nat :=
29   (Finset.range t).sum fun s => cost s omega
30 
31 @[simp]
32 theorem cumulativeSpent_zero
33     {Omega : Type v}
34     (cost : Nat -> Omega -> Nat)
35     (omega : Omega) :
36     cumulativeSpent cost 0 omega = 0 := by
37   simp [cumulativeSpent]
38 
39 @[simp]
40 theorem cumulativeSpent_succ
41     {Omega : Type v}
42     (cost : Nat -> Omega -> Nat)
43     (t : Nat)
44     (omega : Omega) :
45     cumulativeSpent cost (t + 1) omega =
46       cumulativeSpent cost t omega + cost t omega := by
47   simp [cumulativeSpent, Finset.sum_range_succ]
48 
49 /--
50 Finite prefix sums of an adapted per-round Nat-valued cost process remain
51 adapted. A cost observed at `s < t` is promoted from `F s` to `F t`.
52 -/
53 theorem adapted_cumulativeSpent
54     {Omega : Type v} [mOmega : MeasurableSpace Omega]
55     {F : Filtration Nat mOmega}
56     {cost : Nat -> Omega -> Nat}
57     (hcost : Adapted F cost) :
58     Adapted F (cumulativeSpent cost) := by
59   intro t
60   unfold cumulativeSpent
61   exact Finset.measurable_sum (Finset.range t) fun s hs =>
62     hcost.measurable_le
63       (Nat.le_of_lt (Finset.mem_range.mp hs))
64 
65 /--
66 If every completed round costs at least one, cumulative spend grows by at
67 least one at each step.
68 -/
69 theorem cumulativeSpent_unitGrowth_of_one_le
70     {Omega : Type v}
71     (cost : Nat -> Omega -> Nat)
72     (hcost : forall t omega, 1 <= cost t omega) :
73     forall t omega,
74       cumulativeSpent cost t omega + 1 <=
75         cumulativeSpent cost (t + 1) omega := by
76   intro t omega
77   rw [cumulativeSpent_succ]
78   exact Nat.add_le_add_left (hcost t omega) _
