4 /-!
5 # Time-varying loss evaluators on IID states for scheduled half-Tsallis
6 
7 Fresh states remain IID, but the deterministic evaluator may depend on the
8 round.  This is the process-law layer for an oblivious predictable corruption
9 schedule.  The generated trajectory still factors through each finite state
10 prefix, so the current coordinate is independent of the pre-action trace.
11 -/

28     (hvalue_le_one : ∀ t state action, value t state action <= 1) :

30   initial environment action := value 0 (environment 0) action
31   successor n environment _history action :=
32     value (n + 1) (environment (n + 1)) action

44         (Nat -> LossState) ×

46         input.2.2) := measurable_snd.comp measurable_snd
47     exact (hvalue (n + 1)).comp (hstate.prodMk haction)
48   initial_nonneg environment action :=
49     hvalue_nonneg 0 (environment 0) action
50   initial_le_one environment action :=
51     hvalue_le_one 0 (environment 0) action
52   successor_nonneg n environment _history action :=
53     hvalue_nonneg (n + 1) (environment (n + 1)) action
