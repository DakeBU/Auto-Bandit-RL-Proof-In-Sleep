3 /-!
4 # Square-integrable unbounded stopping-time expected OFUL pseudo-regret rate
5 
6 This module controls the random bad-event overflow from the exact unbounded
7 stopping-time decomposition by an `L2` Cauchy-Schwarz bound. It yields a valid
8 `sqrt delta` overflow rate from the compiled stopped-event tail. This remains
9 an event decomposition and does not invoke optional stopping.
10 -/

20 /--
21 Semantic finiteness and second-moment regularity for a `WithTop Nat` stopping
22 time. Under a finite measure this contract implies
23 `IntegrableFiniteStoppingTime`.
24 -/
25 structure SquareIntegrableFiniteStoppingTime
26     {Omega : Type v} [MeasurableSpace Omega]
27     (mu : Measure Omega) (tau : Omega -> WithTop Nat) : Prop where
28   finite_ae : ∀ᵐ omega ∂mu, tau omega ≠ ⊤
29   memLp_rounds :
30     MemLp (fun omega => (((tau omega).untopA + 1 : Nat) : Real)) 2 mu
31 
32 /-- An `L2` finite stopping-time contract supplies the earlier `L1` contract. -/
33 theorem SquareIntegrableFiniteStoppingTime.toIntegrableFiniteStoppingTime
34     {Omega : Type v} [MeasurableSpace Omega]
35     (mu : Measure Omega) [IsFiniteMeasure mu]
36     (tau : Omega -> WithTop Nat)
37     (hstop : SquareIntegrableFiniteStoppingTime mu tau) :
38     IntegrableFiniteStoppingTime mu tau := by
39   refine ⟨hstop.finite_ae, ?_⟩
40   exact hstop.memLp_rounds.integrable (by norm_num)

42 /--
43 Project-local `L2` indicator bound. This is the nonnegative `2,2` Holder
44 specialization needed by the random-horizon overflow consumer.
45 -/
46 theorem integral_indicator_le_sqrt_secondMoment_mul_sqrt_real_measure
47     {Omega : Type v} [MeasurableSpace Omega]
48     (mu : Measure Omega) [IsFiniteMeasure mu]
49     (f : Omega -> Real)
50     (hf_nonneg : forall omega, 0 <= f omega)
51     (hf : MemLp f 2 mu)
52     (bad : Set Omega) (hbad : MeasurableSet bad) :
53     integral mu (bad.indicator f) <=
54       Real.sqrt (integral mu (fun omega => f omega ^ 2)) *
55         Real.sqrt (mu.real bad) := by
56   let indicatorOne : Omega -> Real := bad.indicator (fun _ => 1)
57   have hpq : (2 : Real).HolderConjugate 2 := by
58     rw [Real.holderConjugate_iff]
59     norm_num
60   have hindicatorOne_nonneg : forall omega, 0 <= indicatorOne omega := by
61     intro omega
62     by_cases homega : omega ∈ bad <;> simp [indicatorOne, homega]
63   have hindicatorOne : MemLp indicatorOne 2 mu := by
64     exact memLp_indicator_const 2 hbad 1
65       (Or.inr (measure_lt_top mu bad).ne)
66   have hf_realTwo : MemLp f (ENNReal.ofReal (2 : Real)) mu := by
67     norm_num
68     exact hf
69   have hindicatorOne_realTwo :
70       MemLp indicatorOne (ENNReal.ofReal (2 : Real)) mu := by
71     norm_num
72     exact hindicatorOne
73   have hholder :=
74     integral_mul_le_Lp_mul_Lq_of_nonneg hpq
75       (Filter.Eventually.of_forall hf_nonneg)
76       (Filter.Eventually.of_forall hindicatorOne_nonneg)
77       hf_realTwo hindicatorOne_realTwo
78   have hmul :
79       (fun omega => f omega * indicatorOne omega) = bad.indicator f := by
80     funext omega
81     by_cases homega : omega ∈ bad <;> simp [indicatorOne, homega]
82   have hpow :
83       (fun omega => indicatorOne omega ^ (2 : Real)) =
84         bad.indicator (fun _ => (1 : Real)) := by
85     funext omega
86     by_cases homega : omega ∈ bad <;> simp [indicatorOne, homega]
87   have hintegralIndicatorOne :
88       integral mu (bad.indicator (fun _ => (1 : Real))) = mu.real bad := by
89     simpa using (integral_indicator_one (μ := mu) hbad)
90   rw [hmul, hpow, hintegralIndicatorOne] at hholder
91   simp_rw [Real.rpow_two] at hholder
92   simpa only [← Real.sqrt_eq_rpow] using hholder

96 second moment and the square root of the event probability.

111         roundSecondMoment)

113     integral mu

115           (fun trajectory =>
116             standardScalarAllRoundGapEnvelope
117               S (tau trajectory).untopA L2)) <=
118       standardScalarInitialGapBound S L2 *
119         Real.sqrt roundSecondMoment *
120           Real.sqrt (mu.real bad) := by
