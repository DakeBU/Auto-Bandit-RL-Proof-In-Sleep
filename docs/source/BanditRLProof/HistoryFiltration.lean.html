5 # History filtrations
6 
7 This module gives a narrow project-local filtration canary: the sigma-algebras
8 generated by past action/reward singleton events form a Mathlib filtration.
9 It does not construct kernels, policies, conditional expectations, conditional
10 MGF witnesses, or adaptive regret theorems.
11 -/

20 /-- Finite action histories through index `t`, represented on Mathlib's
21 `Finset.Iic t` finite-prefix index type. -/
22 abbrev FiniteActionHistory (Action : Type v) (t : Nat) :=
23   (i : Finset.Iic t) -> Action
24 
25 /-- Finite reward histories through index `t`, represented on Mathlib's
26 `Finset.Iic t` finite-prefix index type. -/
27 abbrev FiniteRewardHistory (Reward : Type w) (t : Nat) :=
28   (i : Finset.Iic t) -> Reward
29 
30 /-- Pair-coordinate finite action/reward histories through index `t`. -/
31 abbrev FinitePairHistory (Action : Type v) (Reward : Type w) (t : Nat) :=
32   (i : Finset.Iic t) -> Prod Action Reward
33 
34 /-- The paired finite action/reward history object at a finite prefix. -/
35 abbrev FiniteHistory (Action : Type v) (Reward : Type w) (t : Nat) :=
36   FiniteActionHistory Action t × FiniteRewardHistory Reward t
37 
38 /-- Restrict an infinite action trace to the finite prefix indexed by
39 `Finset.Iic t`. -/
40 def finiteActionHistoryOfTrace
41     {Action : Type v}
42     (action : ActionTrace Action) (t : Nat) :
43     FiniteActionHistory Action t :=
44   fun i => action i.1
45 
46 /-- Restrict an infinite reward trace to the finite prefix indexed by
47 `Finset.Iic t`. -/
48 def finiteRewardHistoryOfTrace
49     {Reward : Type w}
50     (reward : RewardTrace Reward) (t : Nat) :
51     FiniteRewardHistory Reward t :=
52   fun i => reward i.1
53 
54 /--
55 Complete a finite reward history to an infinite trace, using `default` outside
56 the observed prefix.
57 
58 The completion agrees with `history` at every coordinate through `t`.  It is a
59 deterministic history-reconstruction helper; it does not assert that the
60 defaulted future coordinates agree with an ambient reward process.
61 -/
62 def completeRewardTrace
63     {Reward : Type w}
64     (t : Nat) (history : FiniteRewardHistory Reward t) (default : Reward) :
65     RewardTrace Reward :=
66   fun s => if h : s <= t then history ⟨s, Finset.mem_Iic.mpr h⟩ else default
67 
68 /-- Restrict infinite action and reward traces to a paired finite history. -/
69 def finiteHistoryOfTrace
70     {Action : Type v} {Reward : Type w}
71     (action : ActionTrace Action)
72     (reward : RewardTrace Reward)
73     (t : Nat) :
74     FiniteHistory Action Reward t :=
75   (finiteActionHistoryOfTrace action t, finiteRewardHistoryOfTrace reward t)

77 /-- Restrict infinite action and reward traces to pair-coordinate history. -/
78 def finitePairHistoryOfTrace
79     {Action : Type v} {Reward : Type w}
80     (action : ActionTrace Action)
81     (reward : RewardTrace Reward)
82     (t : Nat) :
83     FinitePairHistory Action Reward t :=
84   fun i => (action i.1, reward i.1)

86 /-- Extend a finite pair history from `t` to `t + 1` by appending the next pair. -/
87 def extendPairHistorySucc
88     {Action : Type v} {Reward : Type w} {t : Nat}
89     (history : FinitePairHistory Action Reward t)
90     (next : Prod Action Reward) :
91     FinitePairHistory Action Reward (t + 1) :=
92   fun i =>
93     if h : i.1 <= t then
94       history ⟨i.1, Finset.mem_Iic.mpr h⟩
95     else
96       next

170 /--
171 The trace prefix at `t + 1` is the old pair prefix extended by the next
172 action/reward pair.

174 @[simp]
175 theorem finitePairHistoryOfTrace_succ
176     {Action : Type v} {Reward : Type w}
177     (action : ActionTrace Action)
178     (reward : RewardTrace Reward)
179     (t : Nat) :
180     finitePairHistoryOfTrace action reward (t + 1) =
181       extendPairHistorySucc
182         (finitePairHistoryOfTrace action reward t)
183         (action (t + 1), reward (t + 1)) := by
184   funext i
185   by_cases hi : i.1 <= t
186   · simp [extendPairHistorySucc, hi]
187   · have hi_le : i.1 <= t + 1 := Finset.mem_Iic.mp i.2
188     have h_succ_le : t + 1 <= i.1 := Nat.succ_le_of_lt (lt_of_not_ge hi)
189     have hi_eq : i.1 = t + 1 := le_antisymm hi_le h_succ_le
190     simp [extendPairHistorySucc, hi_eq]

284 /-- The successor-extension map for pair histories is measurable. -/
285 theorem measurable_extendPairHistorySucc
286     {Action : Type v} {Reward : Type w} {t : Nat}
287     [MeasurableSpace Action] [MeasurableSpace Reward] :

289       (fun input :
290           Prod (FinitePairHistory Action Reward t) (Prod Action Reward) =>

360 /--
361 The singleton-valued action/reward events visible before time `t`.
362 
363 This is deliberately discrete: it is shaped for finite arms and discrete local
364 reward traces such as `Rat`, where singleton events are the current compiled
365 measurability surface.
366 -/
367 def historyGenerators
368     {Omega : Type u} {Action : Type v} {Reward : Type w}
369     (action : Omega -> ActionTrace Action)
370     (reward : Omega -> RewardTrace Reward)
371     (t : Nat) : Set (Set Omega) :=
372   Set.union
373     {s | Exists (fun i => And (i < t)
374       (Exists (fun a =>
375         s = Set.preimage (fun omega => action omega i) (Set.singleton a))))}
376     {s | Exists (fun i => And (i < t)
377       (Exists (fun r =>
378         s = Set.preimage (fun omega => reward omega i) (Set.singleton r))))}

380 /-- Past history generators are monotone in the horizon. -/
381 theorem historyGenerators_mono
382     {Omega : Type u} {Action : Type v} {Reward : Type w}
383     (action : Omega -> ActionTrace Action)
384     (reward : Omega -> RewardTrace Reward)
385     {t u : Nat} (htu : t <= u) :
386     Set.Subset (historyGenerators action reward t)
387       (historyGenerators action reward u) := by
388   intro s hs
389   rcases hs with hs | hs
390   · cases hs with
391     | intro i hi =>
392       cases hi with
393       | intro hit hrest =>
394         cases hrest with
395         | intro a hs_eq =>
396           exact Or.inl
397             (Exists.intro i
398               (And.intro (lt_of_lt_of_le hit htu)
399                 (Exists.intro a hs_eq)))
400   · cases hs with
401     | intro i hi =>
402       cases hi with
403       | intro hit hrest =>
404         cases hrest with
405         | intro r hs_eq =>
406           exact Or.inr
407             (Exists.intro i
408               (And.intro (lt_of_lt_of_le hit htu)
409                 (Exists.intro r hs_eq)))

469 /--
470 The local history filtration generated by past action/reward singleton events.
471 
472 This is the compiled `FILTRATION-HISTORY` canary.  It is intentionally only a
473 filtration construction, not a policy/predictability or conditional-expectation
474 theorem.
475 -/
476 def historyFiltration
477     {Omega : Type u} {Action : Type v} {Reward : Type w}
478     [mOmega : MeasurableSpace Omega] [MeasurableSpace Action]
479     [MeasurableSingletonClass Action]
480     [MeasurableSpace Reward] [MeasurableSingletonClass Reward]
481     (action : Omega -> ActionTrace Action)
482     (reward : Omega -> RewardTrace Reward)
483     (haction : forall t : Nat,
484       Measurable (fun omega : Omega => action omega t))
485     (hreward : forall t : Nat,
486       Measurable (fun omega : Omega => reward omega t)) :
487     Filtration Nat mOmega where
488   seq t := historyMeasurableSpace action reward t
489   mono' := fun _ _ htu => historyMeasurableSpace_mono action reward htu
490   le' := historyMeasurableSpace_le action reward haction hreward

499     (reward : Omega -> RewardTrace Reward)

501       Measurable (fun omega : Omega => action omega t))
