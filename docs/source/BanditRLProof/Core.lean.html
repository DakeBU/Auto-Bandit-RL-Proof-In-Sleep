6 This file keeps the first project layer dependency-light.  It provides a
7 small executable language for finite action traces, pull counts, reward sums,
8 and finite-arm mean models.  Strong probabilistic statements can later import
9 Mathlib or external libraries without changing this public surface.
10 -/

16 /-- A sequence of actions chosen by a bandit algorithm. -/
17 abbrev ActionTrace (Action : Type u) := Nat → Action
18 
19 /-- A sequence of observed rewards. -/
20 abbrev RewardTrace (Reward : Type v) := Nat → Reward
21 
22 /-- Number of pulls of action `a` before time `t`. -/
23 def pullCount [DecidableEq Action] (action : ActionTrace Action) (a : Action) : Nat → Nat
24   | 0 => 0
25   | t + 1 => pullCount action a t + if action t = a then 1 else 0
26 
27 @[simp] theorem pullCount_zero [DecidableEq Action]
28     (action : ActionTrace Action) (a : Action) :
29     pullCount action a 0 = 0 := rfl
30 
31 @[simp] theorem pullCount_succ [DecidableEq Action]
32     (action : ActionTrace Action) (a : Action) (t : Nat) :
33     pullCount action a (t + 1) =
34       pullCount action a t + if action t = a then 1 else 0 := rfl
35 
36 /-- Sum of rewards obtained from action `a` before time `t`. -/
37 def sumRewards [DecidableEq Action] [OfNat Reward 0] [HAdd Reward Reward Reward]
38     (action : ActionTrace Action) (reward : RewardTrace Reward) (a : Action) :
39     Nat → Reward
40   | 0 => 0
41   | t + 1 =>
42       sumRewards action reward a t + if action t = a then reward t else 0
43 
44 @[simp] theorem sumRewards_zero [DecidableEq Action] [OfNat Reward 0]
45     [HAdd Reward Reward Reward]
46     (action : ActionTrace Action) (reward : RewardTrace Reward) (a : Action) :
47     sumRewards action reward a 0 = 0 := rfl
48 
49 @[simp] theorem sumRewards_succ [DecidableEq Action] [OfNat Reward 0]
50     [HAdd Reward Reward Reward]
51     (action : ActionTrace Action) (reward : RewardTrace Reward) (a : Action) (t : Nat) :
52     sumRewards action reward a (t + 1) =
53       sumRewards action reward a t + if action t = a then reward t else 0 := rfl

55 /-- A finite stochastic bandit model represented by the mean reward of each arm. -/
56 structure FiniteBanditModel (K : Nat) where
57   hK : 0 < K
58   mean : Fin K → Rat
59 
60 namespace FiniteBanditModel
61 
62 /-- A computable argmax-style selector for the best arm. -/
63 noncomputable def bestArm (model : FiniteBanditModel K) : Fin K :=
64   (List.finRange K).foldl
65     (fun best arm => if model.mean best < model.mean arm then arm else best)
66     ⟨0, model.hK⟩
67 
68 /-- The mean reward of `bestArm`. -/
69 noncomputable def bestMean (model : FiniteBanditModel K) : Rat :=
70   model.mean model.bestArm
71 
72 /-- The gap of an arm relative to the selected best arm. -/
73 noncomputable def gap (model : FiniteBanditModel K) (arm : Fin K) : Rat :=
74   if arm = model.bestArm then 0 else model.bestMean - model.mean arm
75 
76 @[simp] theorem gap_bestArm (model : FiniteBanditModel K) :
77     model.gap model.bestArm = 0 := by
78   simp [gap]
79 
80 end FiniteBanditModel
