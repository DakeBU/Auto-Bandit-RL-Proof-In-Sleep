5 /-!
6 # OFUL self-normalized confidence route
7 
8 This module starts the probabilistic OFUL route after the deterministic
9 elliptical-potential theorem.  It first formalizes the predictable-projection
10 conditional exponential inequality used by the method of mixtures.  The
11 multivariate Gaussian mixture identity and final self-normalized event bound
12 remain separate until they are compiled locally.
13 -/

23 /--
24 Freeze a conditioning-measurable multiplier inside the conditional law and
25 compensate its conditionally sub-Gaussian MGF.
26 
27 The explicit exponential-integrability premise is the regularity required by
28 the local fixed-tilt composition API. A bounded-predictable-multiplier wrapper
29 will discharge it for the OFUL feature process.
30 -/
31 theorem HasCondSubgaussianMGF.predictable_mul_compensated_hasCondMGFUpperBoundAt

33     [StandardBorelSpace Omega]

36     (hm : m <= mOmega)
37     (hX : HasCondSubgaussianMGF m hm X c mu)

39     (hintegrable : forall s : Real,
40       Integrable
41         (fun omega =>
42           Real.exp
43             (s * (A omega * X omega -
44               (((c : NNReal) : Real) * A omega ^ 2 / 2))))
45         mu) :

105         funext y

108               (((c : NNReal) : Real) * A omega ^ 2 / 2) =
109             -(((c : NNReal) : Real) * A omega ^ 2 / 2) +

130 /--
131 Uniform boundedness of a predictable multiplier discharges the exponential
132 integrability contract of the compensated increment.
133 -/
134 theorem HasCondSubgaussianMGF.integrable_exp_mul_predictable_mul_compensated_of_abs_le
135     {Omega : Type u} {m mOmega : MeasurableSpace Omega}
136     [StandardBorelSpace Omega]
137     {mu : Measure Omega} [IsProbabilityMeasure mu]
138     {X A : Omega -> Real} {c : NNReal}
139     (hm : m <= mOmega)
140     (hX : HasCondSubgaussianMGF m hm X c mu)
141     (hA : @Measurable Omega Real m inferInstance A)
142     (B : Real) (hB : 0 <= B)
143     (hAbound : forall omega, |A omega| <= B) :
144     forall s : Real,
145       Integrable
146         (fun omega =>
147           Real.exp
148             (s * (A omega * X omega -
149               (((c : NNReal) : Real) * A omega ^ 2 / 2))))
150         mu := by
151   intro s
152   let q : Real := ((c : NNReal) : Real) / 2
153   let D : Real := |s| * B
154   let C : Real := |s| * q * B ^ 2
155   have hD : 0 <= D := mul_nonneg (abs_nonneg s) hB
156   have hdom0 :
157       Integrable (fun omega => Real.exp (D * |X omega|)) mu := by
158     exact ProbabilityTheory.integrable_exp_mul_abs
159       (hX.integrable_exp_mul D)
160       (hX.integrable_exp_mul (-D))
161   have hdom :
162       Integrable
163         (fun omega => Real.exp C * Real.exp (D * |X omega|)) mu :=
164     hdom0.const_mul (Real.exp C)
165   have hXmeas : AEStronglyMeasurable X mu :=
166     (hX.integrable hm).1
167   have htargetMeas :
168       AEStronglyMeasurable
169         (fun omega =>
170           Real.exp
171             (s * (A omega * X omega -
172               (((c : NNReal) : Real) * A omega ^ 2 / 2))))
173         mu := by
174     have hAambient : Measurable A := hA.mono hm le_rfl
175     fun_prop
176   refine Integrable.mono' hdom htargetMeas (ae_of_all mu fun omega => ?_)
177   have hq : 0 <= q := by
178     dsimp [q]
179     positivity
180   have hAsq : A omega ^ 2 <= B ^ 2 := by
181     rw [sq_le_sq]
182     simpa [abs_of_nonneg hB] using hAbound omega
183   have hAX :
184       |A omega * X omega| <= B * |X omega| := by
185     rw [abs_mul]
186     exact mul_le_mul_of_nonneg_right (hAbound omega) (abs_nonneg (X omega))
187   have hqAsq :
188       |q * A omega ^ 2| <= q * B ^ 2 := by
189     rw [abs_of_nonneg (mul_nonneg hq (sq_nonneg (A omega)))]
190     exact mul_le_mul_of_nonneg_left hAsq hq
191   have hzabs :
192       |A omega * X omega - q * A omega ^ 2| <=
193         B * |X omega| + q * B ^ 2 := by
194     exact (abs_sub _ _).trans (add_le_add hAX hqAsq)
195   have hlog :
196       s * (A omega * X omega - q * A omega ^ 2) <=
197         C + D * |X omega| := by
198     calc
199       s * (A omega * X omega - q * A omega ^ 2) <=
200           |s * (A omega * X omega - q * A omega ^ 2)| :=
201         le_abs_self _
202       _ = |s| * |A omega * X omega - q * A omega ^ 2| := by
203         rw [abs_mul]
204       _ <= |s| * (B * |X omega| + q * B ^ 2) :=
205         mul_le_mul_of_nonneg_left hzabs (abs_nonneg s)
206       _ = C + D * |X omega| := by
207         dsimp [C, D]
208         ring
209   rw [Real.norm_eq_abs, abs_of_pos (Real.exp_pos _)]
210   rw [← Real.exp_add]
211   apply Real.exp_le_exp.mpr
212   convert hlog using 1
213   all_goals
214     dsimp [q]
215     ring

251 /--
252 Finite-horizon fixed-direction exponential-supermartingale endpoint for
253 predictable vector features and conditionally sub-Gaussian scalar noise.
254 
255 This is the deterministic-horizon local form of Lemma 1 in
256 Abbasi-Yadkori, Pal, and Szepesvari (2011). It is the input to the Gaussian
257 mixture step, not yet the vector self-normalized determinant-ratio theorem.
258 -/
259 theorem fixedDirectionCompensatedScore_hasMGFUpperBoundAt
260     {Omega : Type v} {Feature : Type w}
261     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
262     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
263     (mu : Measure Omega) [IsProbabilityMeasure mu]
264     (F : Filtration Nat mOmega)
265     (feature : Nat -> Omega -> Feature -> Real)
266     (noise : Nat -> Omega -> Real)
267     (varianceProxy : Nat -> NNReal)
268     (theta : Feature -> Real)
269     (projectionBound : Nat -> Real)
270     (hprojection : forall i,
271       StronglyMeasurable[F i]
272         (fun omega => dotProduct theta (feature i omega)))
273     (hnoise : StronglyAdapted F (fun t omega =>
274       match t with
275       | 0 => 0
276       | i + 1 => noise i omega))
277     (hprojectionBound_nonneg : forall i, 0 <= projectionBound i)
278     (hprojectionBound : forall i omega,
279       |dotProduct theta (feature i omega)| <= projectionBound i)
280     (n : Nat)
281     (hsubGaussian : forall i, i < n ->
282       HasCondSubgaussianMGF
283         (F i) (F.le i) (noise i) (varianceProxy i) mu) :
284     Concentration.HasMGFUpperBoundAt
285       (fun omega =>
286         (Finset.range (n + 1)).sum (fun t =>
287           match t with
288           | 0 => 0
289           | i + 1 =>
290               dotProduct theta (feature i omega) * noise i omega -
291                 (((varianceProxy i : NNReal) : Real) *
292                   (dotProduct theta (feature i omega)) ^ 2 / 2)))
293       1 0 mu := by
294   let Z : Nat -> Omega -> Real := fun t omega =>
295     match t with
296     | 0 => 0
297     | i + 1 =>
298         dotProduct theta (feature i omega) * noise i omega -
299           (((varianceProxy i : NNReal) : Real) *
300             (dotProduct theta (feature i omega)) ^ 2 / 2)
301   have hZ : StronglyAdapted F Z := by
302     intro t
303     cases t with
304     | zero =>
305         exact stronglyMeasurable_const
306     | succ i =>
307         have hA :
308             StronglyMeasurable[F (i + 1)]
309               (fun omega => dotProduct theta (feature i omega)) :=
310           (hprojection i).mono (F.mono (Nat.le_succ i))
311         have hX :
312             StronglyMeasurable[F (i + 1)] (noise i) := by
313           simpa using hnoise (i + 1)
314         have hprod := hA.measurable.mul hX.measurable
315         have hsquare := hA.measurable.pow_const 2
316         have hcoeff :
317             Measurable[F (i + 1)]
318               (fun _ : Omega => (((varianceProxy i : NNReal) : Real))) :=
319           measurable_const
320         have hcompensator :=
321           (hcoeff.mul hsquare).div_const (2 : Real)
322         simpa [Z] using (hprod.sub hcompensator).stronglyMeasurable
323   have hzero : Concentration.HasMGFUpperBoundAt (Z 0) 1 0 mu := by
324     have hzero' :
325         Concentration.HasMGFUpperBoundAt
326           (fun _ : Omega => (0 : Real)) 1 0 mu := by
327       constructor
328       · intro s
329         simp
330       · simp [ProbabilityTheory.mgf]
331     simpa [Z] using hzero'
332   have hcond : forall i, i < (n + 1) - 1 ->
333       Concentration.HasCondMGFUpperBoundAt
334         (F i) (F.le i) (Z (i + 1)) 1 0 mu := by
335     intro i hi
336     have hsg := hsubGaussian i (by omega)
337     have h :=
338       hsg.predictable_mul_compensated_hasCondMGFUpperBoundAt_of_abs_le
339         (F.le i) (hprojection i).measurable
340         (projectionBound i) (hprojectionBound_nonneg i)
341         (hprojectionBound i)
342     simpa [Z] using h
343   have hsum :=
344     Concentration.HasMGFUpperBoundAt.sum_of_hasCondMGFUpperBoundAt
345       (μ := mu) (ℱ := F) (Y := Z) (ψY := fun _ => 0) (t := 1)
346       hZ hzero (n + 1) hcond
347   simpa [Z] using hsum
