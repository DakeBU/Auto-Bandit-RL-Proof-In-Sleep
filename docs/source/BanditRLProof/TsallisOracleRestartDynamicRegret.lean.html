5 # Oracle-restart dynamic-regret assembly
6 
7 This module proves the finite-sum assembly behind a switch-aligned restart
8 route. It does not construct a restarted selector or trajectory kernel:
9 epoch-local fixed-comparator regret certificates remain explicit inputs.
10 -/

20 /-- Included rounds assigned to one epoch. -/
21 def oracleRestartEpochRounds
22     {Epoch : Type w} [DecidableEq Epoch]
23     (epochOf : Nat -> Epoch) (horizon : Nat) (epoch : Epoch) : Finset Nat :=
24   (Finset.range (horizon + 1)).filter (fun t => epochOf t = epoch)

26 /-- Predictable fixed-comparator regret contributed by one epoch fiber. -/
27 noncomputable def sampledScheduledHalfTsallisPredictableOracleRestartEpochRegret
28     {Env : Type u} {Action : Type v} {Epoch : Type w}
29     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
30     [DecidableEq Epoch]
31     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
32     (loss : Exp3.PredictableLossVector Env Action)
33     (epochOf : Nat -> Epoch) (epochComparator : Epoch -> Action)
34     (horizon : Nat) (epoch : Epoch)
35     (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
36   (oracleRestartEpochRounds epochOf horizon epoch).sum (fun t =>
37     FTRL.linearLoss arms
38         (sampledScheduledHalfTsallisProbabilityAtTime
39           arms harms eta t sample)
40         (Exp3.predictableLossAt loss t sample) -

42         (epochComparator (epochOf t)))

74 /-- Cauchy--Schwarz bound for the square roots of epoch-fiber lengths. -/
75 theorem sum_sqrt_oracleRestartEpochRounds_card_le
76     {Epoch : Type w} [DecidableEq Epoch]
77     (epochs : Finset Epoch) (epochOf : Nat -> Epoch) (horizon : Nat)
78     (hEpochOf : ∀ t ∈ Finset.range (horizon + 1), epochOf t ∈ epochs) :
79     epochs.sum (fun epoch =>
80         Real.sqrt
81           ((oracleRestartEpochRounds epochOf horizon epoch).card : Real)) <=
82       Real.sqrt (epochs.card : Real) *
83         Real.sqrt (((horizon + 1 : Nat) : Real)) := by
84   have hcardNat :
85       (Finset.range (horizon + 1)).card =
86         epochs.sum (fun epoch =>
87           (oracleRestartEpochRounds epochOf horizon epoch).card) := by
88     simpa [oracleRestartEpochRounds] using
89       (Finset.card_eq_sum_card_fiberwise
90         (s := Finset.range (horizon + 1))
91         (t := epochs) (f := epochOf) hEpochOf)
92   have hcardReal :
93       (((horizon + 1 : Nat) : Real)) =
94         epochs.sum (fun epoch =>
95           ((oracleRestartEpochRounds epochOf horizon epoch).card : Real)) := by
96     have hcast := congrArg (fun n : Nat => (n : Real)) hcardNat
97     simpa using hcast
98   have hcs :=
99     Real.sum_sqrt_mul_sqrt_le epochs
100       (f := fun _epoch => (1 : Real))
101       (g := fun epoch =>
102         ((oracleRestartEpochRounds epochOf horizon epoch).card : Real))
103       (fun _epoch => by positivity)
104       (fun _epoch => by positivity)
105   simpa [hcardReal] using hcs

119     (sample : Env × ((k : Nat) -> Action × Real))
120     (hEpochRegret :
121       ∀ epoch ∈ epochs,
122         sampledScheduledHalfTsallisPredictableOracleRestartEpochRegret

124           coefficient *
125             Real.sqrt
126               ((oracleRestartEpochRounds epochOf horizon epoch).card :

159 /-- Switch-count-facing oracle-restart assembly. When the epoch partition has
160 at most one more epoch than switches, the global moving-comparator bound has
161 the standard `sqrt((switches + 1) * (horizon + 1))` product form. -/
162 theorem sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret_le_oracleRestartSwitchCountSqrt
163     {Env : Type u} {Action : Type v} {Epoch : Type w}
164     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
165     [DecidableEq Epoch]
166     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
167     (loss : Exp3.PredictableLossVector Env Action)
168     (epochs : Finset Epoch) (epochOf : Nat -> Epoch)
169     (epochComparator : Epoch -> Action) (horizon switches : Nat)
170     (hEpochOf : ∀ t ∈ Finset.range (horizon + 1), epochOf t ∈ epochs)
171     (hEpochCard : epochs.card <= switches + 1)
172     (coefficient : Real) (hcoefficient : 0 <= coefficient)
173     (sample : Env × ((k : Nat) -> Action × Real))
174     (hEpochRegret :
175       ∀ epoch ∈ epochs,
176         sampledScheduledHalfTsallisPredictableOracleRestartEpochRegret
177             arms harms eta loss epochOf epochComparator horizon epoch sample <=
178           coefficient *
179             Real.sqrt
180               ((oracleRestartEpochRounds epochOf horizon epoch).card :
181                 Real)) :
182     sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret
183         arms harms eta loss (fun t => epochComparator (epochOf t))
184         horizon sample <=
185       coefficient * Real.sqrt (((switches + 1 : Nat) : Real)) *
186         Real.sqrt (((horizon + 1 : Nat) : Real)) := by
187   have hrestart :=
188     sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret_le_oracleRestartSqrt
189       arms harms eta loss epochs epochOf epochComparator horizon hEpochOf
190       coefficient hcoefficient sample hEpochRegret
191   have hcardReal :
192       (epochs.card : Real) <= (((switches + 1 : Nat) : Real)) := by
193     exact_mod_cast hEpochCard
194   have hsqrt :
195       Real.sqrt (epochs.card : Real) <=
196         Real.sqrt (((switches + 1 : Nat) : Real)) :=
197     Real.sqrt_le_sqrt hcardReal

199     _ <= coefficient * Real.sqrt (epochs.card : Real) *
