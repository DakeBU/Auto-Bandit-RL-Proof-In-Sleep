3 /-!
4 # Canonical half-Tsallis finite-horizon minimizers
5 
6 This module applies the fixed half-Tsallis minimizer choice to cumulative loss
7 vectors.  It removes the caller-supplied minimizer certificates from the
8 deterministic finite-horizon FTRL decomposition and records the successor
9 indexing needed by importance-weighted updates.
10 
11 The selected minimizer uses `Classical.choose`.  No measurability of the
12 history-dependent selector, conditional expectation identity, or bound on the
13 finite-horizon stability sum is claimed here.
14 -/
15 
16 namespace BanditRLProof
17 namespace Tsallis
18 
19 universe u

21 /-- The fixed half-Tsallis minimizer for losses accumulated before round `t`. -/
22 noncomputable def halfTsallisCumulativeMinimizer
23     {Action : Type u} [DecidableEq Action]
24     (arms : Finset Action) (harms : arms.Nonempty)
25     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) : Action -> Real :=
26   halfTsallisMinimizer arms harms eta (FTRL.cumulativeLoss loss t)
27 
28 /-- Every canonical cumulative selector carries the minimizer certificate
29 required by the finite-horizon FTRL decomposition. -/
30 theorem halfTsallisCumulativeMinimizer_isRegularizedMinimizer
31     {Action : Type u} [DecidableEq Action]
32     (arms : Finset Action) (harms : arms.Nonempty)
33     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) :
34     FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
35       arms eta (negEntropyRegularizer arms (1 / 2 : Real))
36       (FTRL.cumulativeLoss loss t)
37       (halfTsallisCumulativeMinimizer arms harms eta loss t) :=
38   halfTsallisMinimizer_isRegularizedMinimizer
39     arms harms eta (FTRL.cumulativeLoss loss t)
40 
41 /-- The selector at `t + 1` is the fixed minimizer after appending round `t`'s
42 loss vector. -/
43 theorem halfTsallisCumulativeMinimizer_succ
44     {Action : Type u} [DecidableEq Action]
45     (arms : Finset Action) (harms : arms.Nonempty)
46     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) :
47     halfTsallisCumulativeMinimizer arms harms eta loss (t + 1) =
48       halfTsallisMinimizer arms harms eta (fun action =>
49         FTRL.cumulativeLoss loss t action + loss t action) := by
50   rw [halfTsallisCumulativeMinimizer, FTRL.cumulativeLoss_succ]
51 
52 /-- If the realized round loss is the importance-weighted estimator generated
53 from the current selector, the successor selector is exactly the canonical
54 one-step updated minimizer. -/
55 theorem halfTsallisCumulativeMinimizer_succ_eq_updated
56     {Action : Type u} [DecidableEq Action]
57     (arms : Finset Action) (harms : arms.Nonempty)
58     (eta : Real) (loss : Nat -> Action -> Real)
59     (rawLoss : Action -> Real) (chosen : Action) (t : Nat)
60     (hloss : loss t =
61       Exp3.importanceWeightedLoss
62         (halfTsallisCumulativeMinimizer arms harms eta loss t)
63         rawLoss chosen) :
64     halfTsallisCumulativeMinimizer arms harms eta loss (t + 1) =
65       halfTsallisUpdatedMinimizer arms harms eta
66         (FTRL.cumulativeLoss loss t) rawLoss chosen := by
67   rw [halfTsallisCumulativeMinimizer, FTRL.cumulativeLoss_succ, hloss]
68   rfl

70 /-- Finite-horizon half-Tsallis FTRL decomposition with the cumulative
71 minimizer sequence selected internally.  The remaining first term on the
72 right is the pathwise stability sum. -/
73 theorem cumulativeLinearLoss_sub_comparator_le_stability_add_powerSumPenalty_half_canonical
74     {Action : Type u} [DecidableEq Action]
75     (arms : Finset Action) (harms : arms.Nonempty)
76     (eta : Real) (loss : Nat -> Action -> Real)
77     (q : Action -> Real) (T : Nat)
78     (heta : 0 < eta) (hq : FTRL.finiteSimplex arms q) :
79     let p := halfTsallisCumulativeMinimizer arms harms eta loss
80     (Finset.range T).sum (fun t =>
81         FTRL.linearLoss arms (p t) (loss t) -
82           FTRL.linearLoss arms q (loss t)) <=
83       (Finset.range T).sum (fun t =>
84           FTRL.linearLoss arms (p t) (loss t) -
85             FTRL.linearLoss arms (p (t + 1)) (loss t)) +
86         ((powerSum arms (1 / 2 : Real) (p 0) -
87             powerSum arms (1 / 2 : Real) q) /
88           (1 - (1 / 2 : Real))) / eta := by
89   dsimp only
90   exact cumulativeLinearLoss_sub_comparator_le_stability_add_powerSumPenalty
91     arms (1 / 2 : Real) eta loss
92       (halfTsallisCumulativeMinimizer arms harms eta loss) q T
93       (by norm_num) heta
94       (fun t _ =>
95         halfTsallisCumulativeMinimizer_isRegularizedMinimizer
96           arms harms eta loss t)
97       hq
