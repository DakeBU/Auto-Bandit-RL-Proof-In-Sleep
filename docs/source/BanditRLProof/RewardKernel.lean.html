10 # Reward kernel contract surface
11 
12 This module records a narrow `KERNEL-REWARD` leaf: a reward law is represented
13 as a Mathlib Markov kernel indexed by an arm/context object.  It exposes the
14 kernel and the measurability/probability regularity facts needed by later
15 policy-kernel and trajectory-law leaves, but it does not bind kernels or build
16 a trajectory measure.

568 /--
569 Measure-level action marginal of the one-step action/reward kernel.
570 
571 The action coordinate is deterministic: mapping the `(Action × Reward)` law
572 through `Prod.fst` recovers the Dirac measure at the policy-selected action.
573 -/
574 theorem composePolicyActionReward_action_map
575     {State : Type u} [MeasurableSpace State]
576     (rewardKernel : MarkovRewardKernel (Prod Context Action) Reward)
577     (policy : Policy.MeasurablePolicy State Action)
578     (pair : Prod Context State) :
579     Measure.map Prod.fst
580         ((composePolicyActionReward rewardKernel policy).kernel pair) =
581       Measure.dirac (policy.action pair.2) := by
582   rw [composePolicyActionReward_kernel_apply_eq_map_prod_mk]
583   have hprod :
584       Measurable
585         (Prod.mk (policy.action pair.2) :
586           Reward -> Prod Action Reward) := by
587     fun_prop
588   haveI :
589       IsProbabilityMeasure
590         (selectedMeasure rewardKernel pair.1 (policy.action pair.2)) :=
591     isProbabilityMeasure_selectedMeasure rewardKernel pair.1
592       (policy.action pair.2)
593   refine Measure.ext fun event hevent => ?_
594   rw [Measure.map_apply measurable_fst hevent]
595   rw [Measure.map_apply hprod (measurable_fst hevent)]
596   rw [Measure.dirac_apply' (policy.action pair.2) hevent]
597   by_cases hmem : event (policy.action pair.2)
598   · have hpre :
599         Set.preimage
600             (Prod.mk (policy.action pair.2) :
601               Reward -> Prod Action Reward)
602             (Set.preimage Prod.fst event) =
603           Set.univ := by
604       ext reward
605       change event (policy.action pair.2) ↔ True
606       exact iff_true_intro hmem
607     rw [hpre]
608     simp [Set.indicator_of_mem hmem]
609   · have hpre :
610         Set.preimage
611             (Prod.mk (policy.action pair.2) :
