5 # Independence through a past-measurable kernel extension
6 
7 This module records the semidirect-product transport needed by stochastic
8 bandit trajectory laws.  Sampling from a Markov kernel that only sees a past
9 summary cannot create dependence between its output and a random variable
10 already independent of that summary.
11 -/

19 /-- Restricting both coordinates of a semidirect product is the semidirect
20 product of the restricted base measure and restricted fiber kernel. -/
21 theorem compProd_restrict_prod
22     {A B : Type*} [MeasurableSpace A] [MeasurableSpace B]
23     (mu : Measure A) [SFinite mu]
24     (kernel : Kernel A B) [IsSFiniteKernel kernel]
25     {s : Set A} {t : Set B} (hs : MeasurableSet s) (ht : MeasurableSet t) :
26     (mu ⊗ₘ kernel).restrict (s ×ˢ t) =
27       mu.restrict s ⊗ₘ kernel.restrict ht := by
28   have hprodMk :
29       (Kernel.prodMkLeft Unit kernel).restrict ht =
30         Kernel.prodMkLeft Unit (kernel.restrict ht) := by
31     ext input
32     rw [Kernel.restrict_apply, Kernel.prodMkLeft_apply,
33       Kernel.prodMkLeft_apply, Kernel.restrict_apply]
34   unfold Measure.compProd
35   rw [← Kernel.restrict_const (α := Unit) hs, ← hprodMk]
36   exact congrArg (fun extendedKernel => extendedKernel ())
37     (Kernel.compProd_restrict
38       (κ := Kernel.const Unit mu)
39       (η := Kernel.prodMkLeft Unit kernel) hs ht).symm

41 /-- A semidirect-product law restricted to a measurable safe set depends only
42 on the base law on a measurable base safe set and on each kernel's restriction
43 to the corresponding safe fiber. -/
44 theorem compProd_restrict_eq_of_base_restrict_eq_of_fiber_restrict_eq
45     {A B : Type*} [MeasurableSpace A] [MeasurableSpace B]
46     {mu nu : Measure A} [SFinite mu] [SFinite nu]
47     {kernel eta : Kernel A B}
48     [IsSFiniteKernel kernel] [IsSFiniteKernel eta]
49     {baseSafe : Set A} {safe : Set (A × B)}
50     (hbaseSafe : MeasurableSet baseSafe)
51     (hsafe : MeasurableSet safe)
52     (hsafe_base : safe ⊆ baseSafe ×ˢ Set.univ)
53     (hbase : mu.restrict baseSafe = nu.restrict baseSafe)
54     (hfiber : ∀ a ∈ baseSafe,
55       (kernel a).restrict (Prod.mk a ⁻¹' safe) =
56         (eta a).restrict (Prod.mk a ⁻¹' safe)) :
57     (mu ⊗ₘ kernel).restrict safe = (nu ⊗ₘ eta).restrict safe := by
58   rw [Measure.restrict_congr_meas hsafe]
59   intro event hevent_safe hevent
60   rw [Measure.compProd_apply hevent, Measure.compProd_apply hevent]
61   have hsection_eq : ∀ a ∈ baseSafe,
62       kernel a (Prod.mk a ⁻¹' event) = eta a (Prod.mk a ⁻¹' event) := by
63     intro a ha
64     exact (Measure.restrict_congr_meas (measurable_prodMk_left hsafe)).mp
65       (hfiber a ha) (Prod.mk a ⁻¹' event)
66         (Set.preimage_mono hevent_safe) (measurable_prodMk_left hevent)
67   have hkernel_zero : ∀ a ∉ baseSafe,
68       kernel a (Prod.mk a ⁻¹' event) = 0 := by
69     intro a ha
70     have hsection : Prod.mk a ⁻¹' event = (∅ : Set B) := by
71       ext b
72       simp only [Set.mem_preimage, Set.mem_empty_iff_false, iff_false]
73       intro hab
74       exact ha (hsafe_base (hevent_safe hab)).1
75     rw [hsection]
76     exact measure_empty
77   have heta_zero : ∀ a ∉ baseSafe,
78       eta a (Prod.mk a ⁻¹' event) = 0 := by
79     intro a ha
80     have hsection : Prod.mk a ⁻¹' event = (∅ : Set B) := by
81       ext b
82       simp only [Set.mem_preimage, Set.mem_empty_iff_false, iff_false]
83       intro hab
84       exact ha (hsafe_base (hevent_safe hab)).1
85     rw [hsection]
86     exact measure_empty
87   calc
88     (∫⁻ a, kernel a (Prod.mk a ⁻¹' event) ∂mu) =
89         ∫⁻ a, baseSafe.indicator
90           (fun a => kernel a (Prod.mk a ⁻¹' event)) a ∂mu := by
91       apply lintegral_congr
92       intro a
93       by_cases ha : a ∈ baseSafe
94       · simp [ha]
95       · simp [ha, hkernel_zero a ha]
96     _ = ∫⁻ a, kernel a (Prod.mk a ⁻¹' event) ∂(mu.restrict baseSafe) :=
97       lintegral_indicator hbaseSafe _
98     _ = ∫⁻ a, kernel a (Prod.mk a ⁻¹' event) ∂(nu.restrict baseSafe) := by
99       rw [hbase]
100     _ = ∫⁻ a, eta a (Prod.mk a ⁻¹' event) ∂(nu.restrict baseSafe) := by
101       apply lintegral_congr_ae
102       exact (ae_restrict_mem hbaseSafe).mono hsection_eq
103     _ = ∫⁻ a, baseSafe.indicator
104           (fun a => eta a (Prod.mk a ⁻¹' event)) a ∂nu :=
105       (lintegral_indicator hbaseSafe _).symm
106     _ = ∫⁻ a, eta a (Prod.mk a ⁻¹' event) ∂nu := by
107       apply lintegral_congr
108       intro a
109       by_cases ha : a ∈ baseSafe
110       · simp [ha]
111       · simp [ha, heta_zero a ha]

113 /-- Mapped form of
114 `compProd_restrict_eq_of_base_restrict_eq_of_fiber_restrict_eq`: a measurable
115 successor map preserves the safe-set equality. -/
116 theorem map_compProd_restrict_eq_of_base_restrict_eq_of_fiber_restrict_eq
117     {A B C : Type*}
118     [MeasurableSpace A] [MeasurableSpace B] [MeasurableSpace C]
119     {mu nu : Measure A} [SFinite mu] [SFinite nu]
120     {kernel eta : Kernel A B}
121     [IsSFiniteKernel kernel] [IsSFiniteKernel eta]
122     (successor : A × B → C) (hsuccessor : Measurable successor)
123     {baseSafe : Set A} {successorSafe : Set C}
124     (hbaseSafe : MeasurableSet baseSafe)
125     (hsuccessorSafe : MeasurableSet successorSafe)
126     (hpreimage_base :
127       successor ⁻¹' successorSafe ⊆ baseSafe ×ˢ Set.univ)
128     (hbase : mu.restrict baseSafe = nu.restrict baseSafe)
129     (hfiber : ∀ a ∈ baseSafe,
130       (kernel a).restrict
131           (Prod.mk a ⁻¹' (successor ⁻¹' successorSafe)) =
132         (eta a).restrict
133           (Prod.mk a ⁻¹' (successor ⁻¹' successorSafe))) :
134     ((mu ⊗ₘ kernel).map successor).restrict successorSafe =
135       ((nu ⊗ₘ eta).map successor).restrict successorSafe := by
136   rw [Measure.restrict_map hsuccessor hsuccessorSafe,
137     Measure.restrict_map hsuccessor hsuccessorSafe,
138     compProd_restrict_eq_of_base_restrict_eq_of_fiber_restrict_eq
139       hbaseSafe (hsuccessor hsuccessorSafe) hpreimage_base hbase hfiber]

141 end Measure

143 /-- Pull independence on a pushforward measure back along the measurable map
144 that produced that pushforward. -/
145 theorem IndepFun.comp_of_map
146     {Omega Sample X Y : Type*}
147     [MeasurableSpace Omega] [MeasurableSpace Sample]
148     [MeasurableSpace X] [MeasurableSpace Y]
149     {mu : Measure Omega} {z : Omega -> Sample}
150     {x : Sample -> X} {y : Sample -> Y}
151     (hz : Measurable z) (hx : Measurable x) (hy : Measurable y)
152     (hindep : IndepFun x y (mu.map z)) :
153     IndepFun (x ∘ z) (y ∘ z) mu := by
154   rw [indepFun_iff_measure_inter_preimage_eq_mul] at hindep ⊢
155   intro s t hs ht
156   have hxs : MeasurableSet (x ⁻¹' s) := hx hs
157   have hyt : MeasurableSet (y ⁻¹' t) := hy ht
158   have h := hindep s t hs ht
159   rw [Measure.map_apply hz (hxs.inter hyt),
160     Measure.map_apply hz hxs, Measure.map_apply hz hyt] at h
161   simpa only [Function.comp_apply, Set.preimage_inter] using h

163 /-- If `x` is independent of `past`, adjoining a Markov-kernel output whose
164 law only depends on `past` leaves `x` independent of that output. -/
165 theorem indepFun_fst_snd_compProd_comap_of_indepFun
166     {Omega X Past Output : Type*}
167     [MeasurableSpace Omega] [MeasurableSpace X]
168     [MeasurableSpace Past] [MeasurableSpace Output]
169     (mu : Measure Omega) [IsProbabilityMeasure mu]
170     (x : Omega -> X) (hx : Measurable x)
171     (past : Omega -> Past) (hpast : Measurable past)
172     (kernel : Kernel Past Output) [IsMarkovKernel kernel]
173     (hindep : IndepFun x past mu) :
174     IndepFun (x ∘ Prod.fst) Prod.snd
175       (mu ⊗ₘ kernel.comap past hpast) := by
176   rw [indepFun_iff_measure_inter_preimage_eq_mul]
177   intro s t hs ht
178   let liftedKernel := kernel.comap past hpast
179   let rho := mu ⊗ₘ liftedKernel
180   have hleft : Measurable (fun omega : Omega =>
181       s.indicator (fun _ => (1 : ENNReal)) (x omega)) :=
182     (measurable_const.indicator hs).comp hx
183   have hright : Measurable (fun omega : Omega => kernel (past omega) t) :=
184     (kernel.measurable_coe ht).comp hpast
185   have hindepValues : IndepFun
186       (fun omega : Omega => s.indicator (fun _ => (1 : ENNReal)) (x omega))
187       (fun omega : Omega => kernel (past omega) t) mu :=
188     hindep.comp (measurable_const.indicator hs) (kernel.measurable_coe ht)
189   have hfactor :=
190     lintegral_mul_eq_lintegral_mul_lintegral_of_indepFun
191       hleft hright hindepValues
192   change rho ({sample | x sample.1 ∈ s} ∩ {sample | sample.2 ∈ t}) =
193     rho {sample | x sample.1 ∈ s} * rho {sample | sample.2 ∈ t}
194   rw [Measure.compProd_apply]
195   · rw [Measure.compProd_apply]
196     · rw [Measure.compProd_apply]
197       · simp only [liftedKernel, Kernel.comap_apply]
198         convert hfactor using 1
199         · apply lintegral_congr
200           intro omega
201           by_cases homega : x omega ∈ s <;> simp [homega, Set.indicator]
202         · congr 1
203           apply lintegral_congr
204           intro omega
205           by_cases homega : x omega ∈ s <;> simp [homega, Set.indicator]
206       · exact measurable_snd ht
207     · exact hx.comp measurable_fst hs
208   · exact (hx.comp measurable_fst hs).inter (measurable_snd ht)

210 /-- If `x` is independent of `past`, adjoining an output through a finite
211 kernel that only sees `past` gives the joint output/`x` law as the output
212 marginal times the original `x` marginal.  Unlike the preceding `IndepFun`
213 wrapper, this statement remains valid for subprobability kernels such as a
214 branch-restricted Markov kernel. -/
215 theorem map_snd_x_compProd_comap_eq_prod_map_of_indepFun
216     {Omega X Past Output : Type*}
217     [MeasurableSpace Omega] [MeasurableSpace X]
218     [MeasurableSpace Past] [MeasurableSpace Output]
219     (mu : Measure Omega) [IsProbabilityMeasure mu]
220     (x : Omega -> X) (hx : Measurable x)
221     (past : Omega -> Past) (hpast : Measurable past)
222     (kernel : Kernel Past Output) [IsFiniteKernel kernel]
223     (hindep : IndepFun x past mu) :
224     Measure.map (fun sample : Omega × Output => (sample.2, x sample.1))
225         (mu ⊗ₘ kernel.comap past hpast) =
226       (Measure.map Prod.snd (mu ⊗ₘ kernel.comap past hpast)).prod
227         (Measure.map x mu) := by
228   apply Measure.ext_prod
229   intro outputEvent xEvent houtputEvent hxEvent
230   let liftedKernel := kernel.comap past hpast
231   let rho := mu ⊗ₘ liftedKernel
232   have hleft : Measurable (fun omega : Omega =>
233       xEvent.indicator (fun _ => (1 : ENNReal)) (x omega)) :=
234     (measurable_const.indicator hxEvent).comp hx
235   have hright : Measurable
236       (fun omega : Omega => kernel (past omega) outputEvent) :=
237     (kernel.measurable_coe houtputEvent).comp hpast
238   have hindepValues : IndepFun
239       (fun omega : Omega =>
240         xEvent.indicator (fun _ => (1 : ENNReal)) (x omega))
241       (fun omega : Omega => kernel (past omega) outputEvent) mu :=
242     hindep.comp (measurable_const.indicator hxEvent)
243       (kernel.measurable_coe houtputEvent)
244   have hfactor :=
245     lintegral_mul_eq_lintegral_mul_lintegral_of_indepFun
246       hleft hright hindepValues
247   have hjoint : Measurable
248       (fun sample : Omega × Output => (sample.2, x sample.1)) := by
249     fun_prop
250   rw [Measure.map_apply
251       hjoint (houtputEvent.prod hxEvent),
252     Measure.prod_prod outputEvent xEvent,
253     Measure.map_apply measurable_snd houtputEvent,
254     Measure.map_apply hx hxEvent]
255   change rho
256       {sample : Omega × Output |
257         sample.2 ∈ outputEvent ∧ x sample.1 ∈ xEvent} =
258     rho {sample : Omega × Output | sample.2 ∈ outputEvent} *
259       mu {omega : Omega | x omega ∈ xEvent}
260   rw [Measure.compProd_apply]
261   · rw [Measure.compProd_apply]
262     · simp only [liftedKernel, Kernel.comap_apply]
263       have hjointIntegral :
264           (∫⁻ omega,
265               kernel (past omega)
266                 (Prod.mk omega ⁻¹'
267                   {sample : Omega × Output |
268                     sample.2 ∈ outputEvent ∧ x sample.1 ∈ xEvent}) ∂mu) =
269             ∫⁻ omega,
270               xEvent.indicator (fun _ => (1 : ENNReal)) (x omega) *
271                 kernel (past omega) outputEvent ∂mu := by
272         apply lintegral_congr
273         intro omega
274         by_cases homega : x omega ∈ xEvent <;>
275           simp [homega, Set.indicator]
276       have houtputIntegral :
277           (∫⁻ omega,
278               kernel (past omega)
279                 (Prod.mk omega ⁻¹'
280                   {sample : Omega × Output |
281                     sample.2 ∈ outputEvent}) ∂mu) =
282             ∫⁻ omega, kernel (past omega) outputEvent ∂mu := by
283         apply lintegral_congr
284         intro omega
285         rfl
286       have hindicator :
287           (∫⁻ omega,
288               xEvent.indicator (fun _ => (1 : ENNReal)) (x omega) ∂mu) =
289             mu {omega : Omega | x omega ∈ xEvent} := by
290         have hset : MeasurableSet {omega : Omega | x omega ∈ xEvent} :=
291           hx hxEvent
292         have hfun :
293             (fun omega : Omega =>
294               xEvent.indicator (fun _ => (1 : ENNReal)) (x omega)) =
295               {omega : Omega | x omega ∈ xEvent}.indicator
296                 (fun _ => (1 : ENNReal)) := by
297           funext omega
298           by_cases homega : x omega ∈ xEvent <;>
299             simp [homega, Set.indicator]
300         rw [hfun]
301         simpa only [Pi.one_apply] using
302           (MeasureTheory.lintegral_indicator_one (μ := mu) hset)
303       have hfactor' :
304           (∫⁻ omega,
305               xEvent.indicator (fun _ => (1 : ENNReal)) (x omega) *
306                 kernel (past omega) outputEvent ∂mu) =
307             (∫⁻ omega,
308               xEvent.indicator (fun _ => (1 : ENNReal)) (x omega) ∂mu) *
309               ∫⁻ omega, kernel (past omega) outputEvent ∂mu := by
310         simpa only [Pi.mul_apply] using hfactor
311       rw [hjointIntegral, houtputIntegral, hfactor', hindicator, mul_comm]
312     · exact measurable_snd houtputEvent
313   · exact (measurable_snd houtputEvent).inter
314       (hx.comp measurable_fst hxEvent)
