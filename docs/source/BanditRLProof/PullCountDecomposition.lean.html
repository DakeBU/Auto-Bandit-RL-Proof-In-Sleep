6 # Pull-count decompositions over finite action spaces
7 
8 This module contains deterministic count identities that consume the
9 Mathlib-backed `Finset.range` wrappers.  It stays below the probability and
10 algorithm-specific layers.
11 -/

20 /--
21 The pull counts over a finite action space partition the time horizon.
22 
23 This is the deterministic `PULLCOUNT-SUM-TIME` leaf.  It consumes the compiled
24 `pullCount` Finset wrapper and Mathlib's finite fiber-cardinality theorem.
25 -/
26 theorem finset_sum_pullCount_eq_time :
27     (Finset.univ : Finset Action).sum
28       (fun a : Action => pullCount action a t) = t := by
29   calc
30     (Finset.univ : Finset Action).sum
31         (fun a : Action => pullCount action a t)
32         = (Finset.univ : Finset Action).sum
33             (fun a : Action =>
34               ((Finset.range t).filter (fun s : Nat => action s = a)).card) := by
35           apply Finset.sum_congr rfl
36           intro a _ha
37           rw [pullCount_eq_finset_filter_card]
38     _ = (Finset.range t).card := by
39           exact (Finset.card_eq_sum_card_fiberwise
40             (f := action)
41             (s := Finset.range t)
42             (t := (Finset.univ : Finset Action))
43             (by intro _s _hs; simp)).symm
44     _ = t := by
45           simp

47 /--
48 Reindex a sum over action times by arm and the arm's prior pull count.
49 
50 At a time `s`, the value `pullCount action (action s) s` is the zero-based
51 index of that pull among occurrences of the selected arm. This is the local
52 Mathlib-backed counterpart of LML's `sum_comp_pullCount` bookkeeping lemma.
53 -/
54 theorem finset_sum_comp_pullCount
55     {R : Type v} [AddCommMonoid R] (f : Nat -> R) :
56     ∑ s ∈ Finset.range t, f (pullCount action (action s) s) =
57       ∑ a : Action, ∑ j ∈ Finset.range (pullCount action a t), f j := by
58   induction t with
59   | zero => simp
60   | succ n ih =>
61       have hf : f (pullCount action (action n) n) =
62           ∑ a : Action,
63             if action n = a then f (pullCount action a n) else 0 := by
64         exact (Fintype.sum_ite_eq (action n)
65           (fun a => f (pullCount action a n))).symm
66       simp_rw [Finset.sum_range_succ, ih, hf,
67         ← Finset.sum_add_distrib, pullCount_succ]
68       apply Finset.sum_congr rfl
69       intro a _ha
70       by_cases h : action n = a
71       · simp [h, Finset.sum_range_succ]
72       · simp [h]
73
