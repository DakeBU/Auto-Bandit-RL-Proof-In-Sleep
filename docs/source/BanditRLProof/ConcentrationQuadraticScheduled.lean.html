4 # Countable scheduled quadratic fixed-MGF tails
5 
6 This module lifts the one-event quadratic fixed-MGF delta theorem to a
7 countable family with time-varying parameters and confidence shares. The
8 result uses countable outer-measure subadditivity; it is not a Ville, Doob,
9 mixture, optional-stopping, or general Freedman theorem.
10 -/

18 /-- The optimized quadratic radius evaluated at one index of a parameter
19 schedule. -/
20 noncomputable def quadraticFixedMGFScheduledRadius
21     (varianceScale varianceBudget tiltCap deltaAt : Nat -> Real)
22     (n : Nat) : Real :=
23   quadraticFixedMGFRadius
24     (varianceScale n) (varianceBudget n) (tiltCap n) (deltaAt n)

26 /-- A countable family of quadratic fixed-MGF events is controlled by the
27 sum of its scheduled confidence shares. Event measurability is not required
28 because `Measure.measure_iUnion_le` is an outer-measure inequality. -/
29 theorem measure_iUnion_scheduled_deviation_ge_inter_variance_le_tsum_of_fixedTilt_quadratic_tail
30     {Omega : Type*} [MeasurableSpace Omega]
31     (mu : Measure Omega)
32     (deviation predictableVariance : Nat -> Omega -> Real)
33     (varianceScale varianceBudget tiltCap deltaAt : Nat -> Real)
34     (hvarianceScale : forall n, 0 < varianceScale n)
35     (hvarianceBudget : forall n, 0 < varianceBudget n)
36     (htiltCap : forall n, 0 < tiltCap n)
37     (hdeltaAt : forall n, 0 < deltaAt n)
38     (hfixed : forall n tilt, 0 <= tilt -> tilt <= tiltCap n ->
39       mu {omega |
40           quadraticFixedMGFScheduledRadius
41                 varianceScale varianceBudget tiltCap deltaAt n <=
42               deviation n omega ∧
43             predictableVariance n omega <= varianceBudget n} <=
44         ENNReal.ofReal (Real.exp
45           (-tilt * quadraticFixedMGFScheduledRadius
46               varianceScale varianceBudget tiltCap deltaAt n +
47             varianceScale n * (tilt ^ 2 * varianceBudget n)))) :
48     mu (⋃ n, {omega |
49         quadraticFixedMGFScheduledRadius
50               varianceScale varianceBudget tiltCap deltaAt n <=
51             deviation n omega ∧
52           predictableVariance n omega <= varianceBudget n}) <=
53       ∑' n, ENNReal.ofReal (deltaAt n) := by
54   calc
55     mu (⋃ n, {omega |
56         quadraticFixedMGFScheduledRadius
57               varianceScale varianceBudget tiltCap deltaAt n <=
58             deviation n omega ∧
59           predictableVariance n omega <= varianceBudget n}) <=
60         ∑' n, mu {omega |
61           quadraticFixedMGFScheduledRadius
62                 varianceScale varianceBudget tiltCap deltaAt n <=
63               deviation n omega ∧
64             predictableVariance n omega <= varianceBudget n} :=
65       MeasureTheory.measure_iUnion_le _

67       apply ENNReal.tsum_le_tsum
