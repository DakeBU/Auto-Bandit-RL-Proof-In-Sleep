4 # Scalar-regularization bias for OFUL confidence ellipsoids

7 finite-horizon confidence ellipsoid when the base matrix is `lambda I`.

17 /-- The Euclidean length written on the same finite-coordinate surface as `dotProduct`. -/
18 noncomputable def euclideanLength
19     [Fintype Feature] (theta : Feature -> Real) : Real :=
20   Real.sqrt (dotProduct theta theta)

22 /-- A positive scalar multiple of the identity is positive definite. -/
23 theorem scalarIdentity_posDef
24     [Fintype Feature] [DecidableEq Feature]
25     (lambda : Real) (hlambda : 0 < lambda) :
26     (Matrix.scalar Feature lambda : Matrix Feature Feature Real).PosDef := by
27   rw [Matrix.scalar_apply]
28   exact Matrix.PosDef.diagonal (fun _ => hlambda)

30 /--
31 For `V = lambda I + G` with `G` positive semidefinite, the `V`-norm of the
32 ridge bias `V⁻¹ (lambda theta)` is at most `sqrt lambda * ‖theta‖₂`.
33 -/
34 theorem matrixNorm_nonsingInv_scalar_mulVec_le_sqrt_mul_euclideanLength
35     [Fintype Feature] [DecidableEq Feature]
36     (lambda : Real) (hlambda : 0 < lambda)
37     (G : Matrix Feature Feature Real) (hG : G.PosSemidef)
38     (theta : Feature -> Real) :
39     matrixNorm
40         (Matrix.scalar Feature lambda + G)
41         ((Matrix.scalar Feature lambda + G)⁻¹.mulVec
42           ((Matrix.scalar Feature lambda).mulVec theta)) <=
43       Real.sqrt lambda * euclideanLength theta := by
44   let V : Matrix Feature Feature Real := Matrix.scalar Feature lambda + G
45   let z : Feature -> Real :=
46     V⁻¹.mulVec ((Matrix.scalar Feature lambda).mulVec theta)
47   change matrixNorm V z <=
48     Real.sqrt lambda * Real.sqrt (dotProduct theta theta)
49   have hscalar :
50       (Matrix.scalar Feature lambda : Matrix Feature Feature Real).PosDef :=
51     scalarIdentity_posDef lambda hlambda
52   have hV : V.PosDef := hscalar.add_posSemidef hG
53   have hVz :
54       V.mulVec z = (Matrix.scalar Feature lambda).mulVec theta := by
55     exact posDef_mulVec_nonsingInv_mulVec V hV _
56   have hscalar_theta :
57       (Matrix.scalar Feature lambda).mulVec theta = lambda • theta := by
58     ext i
59     simp [Matrix.mulVec, Matrix.scalar_apply, Matrix.diagonal_apply, dotProduct]
60   have hz_energy :
61       matrixNorm V z ^ 2 = lambda * dotProduct z theta := by
62     rw [matrixNorm_sq V hV, hVz, hscalar_theta]
63     simp [dotProduct_smul, smul_eq_mul]
64   have hV_energy :
65       matrixNorm V z ^ 2 =
66         lambda * dotProduct z z + dotProduct z (G.mulVec z) := by
67     rw [matrixNorm_sq V hV]
68     change dotProduct z ((Matrix.scalar Feature lambda + G).mulVec z) =
69       lambda * dotProduct z z + dotProduct z (G.mulVec z)
70     rw [Matrix.add_mulVec, dotProduct_add]
71     have hscalar_z :
72         (Matrix.scalar Feature lambda).mulVec z = lambda • z := by
73       ext i
74       simp [Matrix.mulVec, Matrix.scalar_apply, Matrix.diagonal_apply, dotProduct]
75     rw [hscalar_z]
76     simp [dotProduct_smul, smul_eq_mul]
77   have hG_nonneg : 0 <= dotProduct z (G.mulVec z) :=
78     hG.dotProduct_mulVec_nonneg z
79   have hbase :
80       lambda * dotProduct z z <= matrixNorm V z ^ 2 := by
81     rw [hV_energy]
82     linarith
83   have hdot_self_z :
84       dotProduct z z =
85         (Finset.univ : Finset Feature).sum (fun i => z i ^ 2) := by
86     simp [dotProduct, pow_two]
87   have hdot_self_theta :
88       dotProduct theta theta =
89         (Finset.univ : Finset Feature).sum (fun i => theta i ^ 2) := by
90     simp [dotProduct, pow_two]
91   have hcs :
92       dotProduct z theta <=
93         Real.sqrt (dotProduct z z) * Real.sqrt (dotProduct theta theta) := by
94     rw [hdot_self_z, hdot_self_theta]
95     simpa [dotProduct] using
96       Real.sum_mul_le_sqrt_mul_sqrt
97         (Finset.univ : Finset Feature) z theta
98   have hzz_nonneg : 0 <= dotProduct z z := by
99     rw [hdot_self_z]
100     exact Finset.sum_nonneg (fun i _hi => sq_nonneg (z i))
101   have htt_nonneg : 0 <= dotProduct theta theta := by
102     rw [hdot_self_theta]
103     exact Finset.sum_nonneg (fun i _hi => sq_nonneg (theta i))
104   have hsqrt_lambda_nonneg : 0 <= Real.sqrt lambda := Real.sqrt_nonneg _
105   have hsqrt_z_nonneg : 0 <= Real.sqrt (dotProduct z z) := Real.sqrt_nonneg _
106   have hsqrt_theta_nonneg :
107       0 <= Real.sqrt (dotProduct theta theta) := Real.sqrt_nonneg _
108   have hmatrix_nonneg : 0 <= matrixNorm V z := Real.sqrt_nonneg _
109   have hsqrt_z_le :
110       Real.sqrt lambda * Real.sqrt (dotProduct z z) <= matrixNorm V z := by
111     apply (sq_le_sq₀ (mul_nonneg hsqrt_lambda_nonneg hsqrt_z_nonneg)
112       hmatrix_nonneg).mp
113     rw [mul_pow, Real.sq_sqrt hlambda.le, Real.sq_sqrt hzz_nonneg]
114     exact hbase
115   have henergy_upper :
116       matrixNorm V z ^ 2 <=
117         lambda * Real.sqrt (dotProduct z z) *
118           Real.sqrt (dotProduct theta theta) := by
119     rw [hz_energy]
120     calc
121       lambda * dotProduct z theta <=
122           lambda *
123             (Real.sqrt (dotProduct z z) *
124               Real.sqrt (dotProduct theta theta)) :=
125         mul_le_mul_of_nonneg_left hcs hlambda.le
126       _ = _ := by ring
127   have hmatrix_sq_le :
128       matrixNorm V z ^ 2 <=
129         (Real.sqrt lambda * Real.sqrt (dotProduct theta theta)) *
130           matrixNorm V z := by
131     calc
132       matrixNorm V z ^ 2 <=
133           lambda * Real.sqrt (dotProduct z z) *
134             Real.sqrt (dotProduct theta theta) := henergy_upper
135       _ =
136           (Real.sqrt lambda * Real.sqrt (dotProduct z z)) *
137             (Real.sqrt lambda * Real.sqrt (dotProduct theta theta)) := by
138         nlinarith [Real.sq_sqrt hlambda.le]
139       _ <=
140           matrixNorm V z *
141             (Real.sqrt lambda * Real.sqrt (dotProduct theta theta)) := by
142         exact mul_le_mul_of_nonneg_right hsqrt_z_le
143           (mul_nonneg hsqrt_lambda_nonneg hsqrt_theta_nonneg)
144       _ = _ := by ring
145   by_cases hz : matrixNorm V z = 0
146   · rw [hz]
147     exact mul_nonneg hsqrt_lambda_nonneg hsqrt_theta_nonneg
148   · have hzpos : 0 < matrixNorm V z := lt_of_le_of_ne hmatrix_nonneg (Ne.symm hz)
149     have hcancel :
150         matrixNorm V z <=
151           Real.sqrt lambda * Real.sqrt (dotProduct theta theta) := by
152       have hmul :
153           matrixNorm V z * matrixNorm V z <=
154             matrixNorm V z *
155               (Real.sqrt lambda * Real.sqrt (dotProduct theta theta)) := by
156         simpa [pow_two, mul_assoc, mul_left_comm, mul_comm] using hmatrix_sq_le
157       exact le_of_mul_le_mul_left hmul hzpos
158     exact hcancel

160 /-- The standard scalar-regularized OFUL radius `noise radius + sqrt lambda * S`. -/
161 noncomputable def finiteHorizonScalarConfidenceRadius
162     [Fintype Feature] [DecidableEq Feature]
163     (feature : Nat -> Omega -> Feature -> Real)
164     (R delta lambda S : Real) (n : Nat) (omega : Omega) : Real :=
165   finiteHorizonConfidenceRadius
166     (Matrix.scalar Feature lambda) feature R delta
167       (Real.sqrt lambda * S) n omega
168 
169 /--
170 The scalar regularization bias is uniformly bounded by `sqrt lambda * S`
171 whenever the true parameter has Euclidean length at most `S`.
172 -/
173 theorem finiteHorizon_scalarRegularizationBias_le
174     [Fintype Feature] [DecidableEq Feature]
175     (lambda : Real) (hlambda : 0 < lambda)
176     (thetaStar : Feature -> Real) (S : Real)
177     (htheta : euclideanLength thetaStar <= S)
178     (feature : Nat -> Omega -> Feature -> Real)
179     (n : Nat) (omega : Omega) :
180     matrixNorm
181         (Matrix.scalar Feature lambda +
182           finiteHorizonFeatureGram feature n omega)
183         ((Matrix.scalar Feature lambda +
184             finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
185           ((Matrix.scalar Feature lambda).mulVec thetaStar)) <=
186       Real.sqrt lambda * S := by
187   exact
188     (matrixNorm_nonsingInv_scalar_mulVec_le_sqrt_mul_euclideanLength
189       lambda hlambda (finiteHorizonFeatureGram feature n omega)
190       (finiteHorizonFeatureGram_posSemidef feature n omega)
191       thetaStar).trans
192       (mul_le_mul_of_nonneg_left htheta (Real.sqrt_nonneg lambda))

194 /--
195 Finite-horizon OFUL confidence ellipsoid with scalar ridge regularization.
196 The explicit bias premise of the general theorem is discharged from
197 `euclideanLength thetaStar <= S`.
198 -/
199 theorem
200     measure_finiteHorizonScalarRidgeEstimate_error_matrixNorm_gt_confidenceRadius_le
201     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
202     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
203     (mu : Measure Omega) [IsProbabilityMeasure mu]
204     (lambda : Real) (hlambda : 0 < lambda)
205     (thetaStar : Feature -> Real) (S : Real)
206     (htheta : euclideanLength thetaStar <= S)
207     (F : Filtration Nat mOmega)
208     (feature : Nat -> Omega -> Feature -> Real)
209     (response noise : Nat -> Omega -> Real)
210     (R : Real) (hR : 0 < R)
211     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
212     (hfeature : forall i j,
213       StronglyMeasurable[F i] (fun omega => feature i omega j))
214     (hnoise : StronglyAdapted F (fun t omega =>
215       match t with
216       | 0 => 0
217       | i + 1 => noise i omega))
218     (hprojectionBound_nonneg : forall theta i,
219       0 <= projectionBound theta i)
220     (hprojectionBound : forall theta i omega,
221       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
222         projectionBound theta i)
223     (n : Nat)
224     (hsubGaussian : forall i, i < n ->
225       HasCondSubgaussianMGF
226         (F i) (F.le i) (noise i)
227         (constantSquaredVarianceProxy R i) mu)
228     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1)
229     (hresponse : forall omega i, i < n ->
230       response i omega =
231         dotProduct thetaStar (feature i omega) + noise i omega) :
232     mu {omega |
233         matrixNorm
234             (Matrix.scalar Feature lambda +
235               finiteHorizonFeatureGram feature n omega)
236             (finiteHorizonRidgeEstimate
237                 (Matrix.scalar Feature lambda) feature response n omega -
238               thetaStar) >
239           finiteHorizonScalarConfidenceRadius
240             feature R delta lambda S n omega} <=
241       ENNReal.ofReal delta := by
242   simpa [finiteHorizonScalarConfidenceRadius] using
243     measure_finiteHorizonRidgeEstimate_error_matrixNorm_gt_confidenceRadius_le
244       mu (Matrix.scalar Feature lambda)
245       (scalarIdentity_posDef lambda hlambda)
246       thetaStar F feature response noise R hR projectionBound hfeature hnoise
247       hprojectionBound_nonneg hprojectionBound n hsubGaussian delta hdelta
248       hdelta_one (Real.sqrt lambda * S) hresponse
249       (fun omega =>
250         finiteHorizon_scalarRegularizationBias_le
251           lambda hlambda thetaStar S htheta feature n omega)
252
