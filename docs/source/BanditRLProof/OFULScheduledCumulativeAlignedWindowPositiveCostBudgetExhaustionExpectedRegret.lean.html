3 /-!
4 # Scheduled OFUL expected regret under aligned-window positive costs
5 
6 This module separates a resource threshold from the deterministic horizon by
7 which it is reached. It then allows individual rounds to have zero cost while
8 requiring every aligned block of a fixed length to spend at least one unit.
9 -/

183 /--
184 If every aligned block of length `window` costs at least one, cumulative spend
185 after `block * window` completed rounds is at least `block`.
186 -/
187 theorem block_le_cumulativeSpent_mul_of_alignedWindowPositive
188     {Omega : Type v}
189     (cost : Nat -> Omega -> Nat)
190     (window : Nat)
191     (haligned : forall block omega,
192       1 <=
193         (Finset.Ico (block * window) ((block + 1) * window)).sum
194           (fun s => cost s omega)) :
195     forall block omega,
196       block <= cumulativeSpent cost (block * window) omega := by
197   intro block
198   induction block with
199   | zero =>
200       intro omega
201       exact Nat.zero_le _
202   | succ block ih =>
203       intro omega
204       have hle :
205           block * window <= (block + 1) * window :=
206         Nat.mul_le_mul_right window (Nat.le_succ block)
207       have hsplit :
208           cumulativeSpent cost (block * window) omega +
209               (Finset.Ico (block * window) ((block + 1) * window)).sum
210                 (fun s => cost s omega) =
211             cumulativeSpent cost ((block + 1) * window) omega := by
212         exact
213           Finset.sum_range_add_sum_Ico
214             (fun s => cost s omega) hle
215       have hresult :
216           block + 1 <=
217             cumulativeSpent cost ((block + 1) * window) omega := by
218         calc
219           block + 1 <=
220               cumulativeSpent cost (block * window) omega + 1 :=
221             Nat.add_le_add_right (ih omega) 1
222           _ <=
223               cumulativeSpent cost (block * window) omega +
224                 (Finset.Ico (block * window) ((block + 1) * window)).sum
225                   (fun s => cost s omega) :=
226             Nat.add_le_add_left (haligned block omega) _
227           _ = cumulativeSpent cost ((block + 1) * window) omega :=
228             hsplit
229       simpa [Nat.succ_eq_add_one] using hresult
