10 /-- A line-15 allocation together with the exact hypotheses needed to make
11 its coordinates a probability distribution.  EAP must eventually construct
12 these fields; this structure does not assume that obligation away. -/
13 structure DelayedSAPOAllocation (K : Nat) where
14   active : Finset (Fin K)
15   inactiveProbability : Fin K → ℝ
16   active_nonempty : active.Nonempty
17   inactive_nonnegative :
18     ∀ i ∈ inactiveArms active, 0 ≤ inactiveProbability i
19   inactive_mass_le_one :
20     (∑ i ∈ inactiveArms active, inactiveProbability i) ≤ 1
21 
22 namespace DelayedSAPOAllocation
23 
24 /-- The Algorithm 5 line-15 probability coordinate associated with a
25 certified allocation. -/
26 noncomputable def probability {K : Nat}
27     (allocation : DelayedSAPOAllocation K) (i : Fin K) : ℝ :=
28   delayedSAPOProbability allocation.active allocation.inactiveProbability i
29 
30 theorem probability_nonnegative {K : Nat}
31     (allocation : DelayedSAPOAllocation K) (i : Fin K) :
32     0 ≤ allocation.probability i := by
33   exact delayedSAPOProbability_nonneg allocation.active
34     allocation.inactiveProbability allocation.inactive_nonnegative
35     allocation.inactive_mass_le_one i
36 
37 theorem sum_probability_eq_one {K : Nat}
38     (allocation : DelayedSAPOAllocation K) :
39     ∑ i, allocation.probability i = 1 := by
40   exact sum_delayedSAPOProbability_eq_one allocation.active
41     allocation.inactiveProbability allocation.active_nonempty
42 
43 /-- Reuse the library's finite-action probability-vector interface instead
44 of introducing a second ad hoc sampling semantics for Delayed SAPO. -/
45 theorem finiteActionDistribution {K : Nat}
46     (allocation : DelayedSAPOAllocation K) :
47     Exp3.FiniteActionDistribution (Finset.univ : Finset (Fin K))
48       allocation.probability where
49   nonneg := by
50     intro i _hi
51     exact allocation.probability_nonnegative i
52   sum_eq_one := allocation.sum_probability_eq_one
53 
54 /-- One-round randomized action law induced by the certified line-15
55 allocation.  A measurable history kernel and recursive generated trajectory
56 remain separate obligations. -/
57 noncomputable def actionMeasure {K : Nat}
58     (allocation : DelayedSAPOAllocation K) : Measure (Fin K) :=
59   Exp3.finiteActionMeasure Finset.univ allocation.probability
60 
61 theorem actionMeasure_isProbabilityMeasure {K : Nat}
62     (allocation : DelayedSAPOAllocation K) :
63     IsProbabilityMeasure allocation.actionMeasure := by
64   exact Exp3.finiteActionMeasure_isProbabilityMeasure Finset.univ
65     allocation.probability allocation.finiteActionDistribution

69 /-- Convert a causal allocation rule into a causal, measure-valued action
70 rule.  The only input remains `ActionTimeView`; hidden delays and unobserved
71 losses are not added to the algorithm interface. -/
72 noncomputable def causalDelayedSAPOActionMeasureRule
73     {K : Nat} {Loss : Type*}
74     (rule : CausalDecisionRule (Fin K) Loss (DelayedSAPOAllocation K)) :
75     CausalDecisionRule (Fin K) Loss (Measure (Fin K)) :=
76   fun t view => (rule t view).actionMeasure
77 
78 /-- Every output of the causal measure-valued rule is a probability measure. -/
79 theorem causalDelayedSAPOActionMeasureRule_isProbabilityMeasure
80     {K : Nat} {Loss : Type*}
81     (rule : CausalDecisionRule (Fin K) Loss (DelayedSAPOAllocation K))
82     (t : Nat) (view : ActionTimeView (Fin K) Loss) :
83     IsProbabilityMeasure (causalDelayedSAPOActionMeasureRule rule t view) := by
84   exact DelayedSAPOAllocation.actionMeasure_isProbabilityMeasure
85     (rule t view)
86 
87 /-- Observation-equivalent hidden worlds induce exactly the same one-round
88 Delayed SAPO action law for every allocation rule typed on the causal view. -/
89 theorem causalDelayedSAPOActionMeasureRule_eq_of_observation_equivalent
90     {K : Nat} {Loss : Type*}
91     (rule : CausalDecisionRule (Fin K) Loss (DelayedSAPOAllocation K))
92     (delay₁ delay₂ : Nat → Nat)
93     (action₁ action₂ : Nat → Fin K) (loss₁ loss₂ : Nat → Loss)
94     (t : Nat)
95     (hvisible : observedBefore delay₁ t = observedBefore delay₂ t)
96     (haction : ∀ s, s < t → action₁ s = action₂ s)
97     (hloss : ∀ s, s ∈ observedBefore delay₁ t → loss₁ s = loss₂ s) :
98     causalDelayedSAPOActionMeasureRule rule t
99         (actionTimeViewAt delay₁ action₁ loss₁ t) =
100       causalDelayedSAPOActionMeasureRule rule t
101         (actionTimeViewAt delay₂ action₂ loss₂ t) := by
102   rw [actionTimeViewAt_eq_of_observation_equivalent
103     delay₁ delay₂ action₁ action₂ loss₁ loss₂ t hvisible haction hloss]
