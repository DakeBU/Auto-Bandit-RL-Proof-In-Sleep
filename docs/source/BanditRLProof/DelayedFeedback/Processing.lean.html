7 /-- Feedback source rounds that are available before action `t` but have not
8 yet been processed.  This is the set-level content of Algorithm 5's
9 `B(t) \ S`; the source sequence order remains a later algorithm-state choice. -/
10 def newlyObservedBefore
11     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat) : Finset Nat :=
12   observedBefore delay t \ processed
13 
14 /-- Strictly available feedback remains available at every later action
15 round. -/
16 theorem observedBefore_mono
17     (delay : Nat → Nat) {t u : Nat} (htu : t ≤ u) :
18     observedBefore delay t ⊆ observedBefore delay u := by
19   intro s hs
20   rw [observedBefore, Finset.mem_filter] at hs ⊢
21   exact ⟨Finset.mem_range.mpr (lt_of_lt_of_le
22     (Finset.mem_range.mp hs.1) htu), lt_of_lt_of_le hs.2 htu⟩
23 
24 /-- A source round is never both already processed and newly observed. -/
25 theorem processed_disjoint_newlyObservedBefore
26     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat) :
27     Disjoint processed (newlyObservedBefore delay processed t) := by
28   rw [Finset.disjoint_left]
29   intro s hprocessed hnew
30   exact (Finset.mem_sdiff.mp hnew).2 hprocessed
31 
32 /-- If all processed rounds were legitimately available, adjoining every new
33 arrival yields exactly the current available set. -/
34 theorem processed_union_newlyObservedBefore
35     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat)
36     (hprocessed : processed ⊆ observedBefore delay t) :
37     processed ∪ newlyObservedBefore delay processed t =
38       observedBefore delay t := by
39   ext s
40   simp only [Finset.mem_union, newlyObservedBefore, Finset.mem_sdiff]
41   constructor
42   · rintro (hs | hs)
43     · exact hprocessed hs
44     · exact hs.1
45   · intro havailable
46     by_cases hs : s ∈ processed
47     · exact Or.inl hs
48     · exact Or.inr ⟨havailable, hs⟩
49 
50 /-- The update obtained by processing every currently new arrival is the
51 current available set. -/
52 def processAllNew
53     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat) : Finset Nat :=
54   processed ∪ newlyObservedBefore delay processed t
55 
56 theorem processAllNew_eq_observedBefore
57     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat)
58     (hprocessed : processed ⊆ observedBefore delay t) :
59     processAllNew delay processed t = observedBefore delay t := by
60   exact processed_union_newlyObservedBefore delay processed t hprocessed
61 
62 /-- A completed earlier available set is a valid processed prefix later. -/
63 theorem previousObservedBefore_subset_current
64     (delay : Nat → Nat) {t u : Nat} (htu : t ≤ u) :
65     observedBefore delay t ⊆ observedBefore delay u :=
66   observedBefore_mono delay htu
67 
68 /-- Starting from the fully processed set at an earlier round and processing
69 all arrivals through a later round yields exactly the later available set. -/
70 theorem processAllNew_from_previous_eq_current
71     (delay : Nat → Nat) {t u : Nat} (htu : t ≤ u) :
72     processAllNew delay (observedBefore delay t) u =
73       observedBefore delay u := by
74   exact processAllNew_eq_observedBefore delay (observedBefore delay t) u
75     (observedBefore_mono delay htu)
76 
77 /-- Outstanding feedback cannot appear in the newly observed batch at the
78 same action time. -/
79 theorem outstandingAt_disjoint_newlyObservedBefore
80     (delay : Nat → Nat) (processed : Finset Nat) (t : Nat) :
81     Disjoint (outstandingAt delay t)
82       (newlyObservedBefore delay processed t) := by
83   rw [Finset.disjoint_left]
84   intro s houtstanding hnew
85   exact (Finset.mem_filter.mp houtstanding).2
86     (Finset.mem_filter.mp (Finset.mem_sdiff.mp hnew).1).2
