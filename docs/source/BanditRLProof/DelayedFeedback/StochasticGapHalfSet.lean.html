3 /-!
4 # A source-faithful deterministic core of Delayed SAPO Lemma D.11
5 
6 The frozen Delayed SAPO source uses a Markov-style count for stochastic loss
7 gaps: at most half of the gaps are greater than `2 * mu`.  This formalization
8 promotes exactly the nonnegative domain used by that application and handles
9 the zero-average case separately.
10 
11 This file records the deterministic statement used by the bandit
12 application.  It treats the empty family and zero average explicitly, and
13 then specializes the result to stochastic loss gaps
14 `mean i - mean optimal`.  It does not construct an Algorithm-5 trajectory,
15 the BSC/EAP branches, a stochastic good event, or the later D.12/regret
16 endpoints.
17 -/

23 /-- Arithmetic mean of a finite family of real gaps.  Lean's total division
24 makes this definition equal to zero when `K = 0`; the counting theorem below
25 handles that empty case before using the denominator. -/
26 noncomputable def finiteAverageGap {K : Nat} (gap : Fin K -> Real) : Real :=
27   (∑ i : Fin K, gap i) / (K : Real)
28 
29 /-- Indices whose gaps are strictly greater than twice the finite average,
30 matching the strict word "greater" in the source statement of Lemma D.11. -/
31 noncomputable def aboveTwiceAverageGap {K : Nat}
32     (gap : Fin K -> Real) : Finset (Fin K) :=
33   Finset.univ.filter fun i => 2 * finiteAverageGap gap < gap i
34 
35 /-- Nonnegative-domain deterministic content used by the source's Lemma D.11.
36 
37 Nonnegativity is explicit rather than extending the promoted contract to an
38 arbitrary signed family.  The theorem includes `K = 0`; for positive `K`, its
39 proof separately closes the zero-average branch before cancelling the
40 positive average in the usual counting/Markov argument. -/
41 theorem two_mul_card_aboveTwiceAverageGap_le
42     {K : Nat} (gap : Fin K -> Real) (hgap : forall i, 0 <= gap i) :
43     2 * (aboveTwiceAverageGap gap).card <= K := by
44   classical
45   by_cases hK : K = 0
46   · subst K
47     simp [aboveTwiceAverageGap]
48   have hKpos : 0 < K := Nat.pos_of_ne_zero hK
49   have hKcastPos : (0 : Real) < K := by exact_mod_cast hKpos
50   have hsumNonneg : 0 <= ∑ i : Fin K, gap i := by
51     exact Finset.sum_nonneg fun i _ => hgap i
52   have haverageNonneg : 0 <= finiteAverageGap gap := by
53     exact div_nonneg hsumNonneg hKcastPos.le
54   by_cases haverageZero : finiteAverageGap gap = 0
55   · have hsumZero : (∑ i : Fin K, gap i) = 0 := by
56       rw [finiteAverageGap, div_eq_zero_iff] at haverageZero
57       exact haverageZero.resolve_right (by positivity)
58     have hgapZero : forall i : Fin K, gap i = 0 := by
59       intro i
60       exact (Finset.sum_eq_zero_iff_of_nonneg
61         (fun j (_hj : j ∈ (Finset.univ : Finset (Fin K))) => hgap j)).mp
62           hsumZero i (Finset.mem_univ i)
63     have hlargeEmpty : aboveTwiceAverageGap gap = ∅ := by
64       ext i
65       simp [aboveTwiceAverageGap, haverageZero, hgapZero i]
66     simp [hlargeEmpty]
67   have haveragePos : 0 < finiteAverageGap gap :=
68     lt_of_le_of_ne haverageNonneg (Ne.symm haverageZero)
69   by_cases hlargeNonempty : (aboveTwiceAverageGap gap).Nonempty
70   · have hstrictSum :
71         (aboveTwiceAverageGap gap).sum
72               (fun _ => 2 * finiteAverageGap gap) <
73             (aboveTwiceAverageGap gap).sum gap := by
74       apply Finset.sum_lt_sum_of_nonempty hlargeNonempty
75       intro i hi
76       exact (Finset.mem_filter.mp hi).2
77     have hsubsetSum :
78         (aboveTwiceAverageGap gap).sum gap <=
79             (Finset.univ : Finset (Fin K)).sum gap := by
80       apply Finset.sum_le_sum_of_subset_of_nonneg
81         (Finset.filter_subset _ _)
82       intro i _hi _hnot
83       exact hgap i
84     have hsumIdentity :
85         (Finset.univ : Finset (Fin K)).sum gap =
86             finiteAverageGap gap * (K : Real) := by
87       rw [finiteAverageGap]
88       field_simp
89     have hcardReal :
90         (2 : Real) * ((aboveTwiceAverageGap gap).card : Real) < (K : Real) := by
91       have hconstantSum :
92           (aboveTwiceAverageGap gap).sum
93                 (fun _ => 2 * finiteAverageGap gap) =
94               ((aboveTwiceAverageGap gap).card : Real) *
95                 (2 * finiteAverageGap gap) := by
96         simp
97       rw [hconstantSum] at hstrictSum
98       rw [hsumIdentity] at hsubsetSum
99       nlinarith
100     have hcardNat : 2 * (aboveTwiceAverageGap gap).card < K := by
101       exact_mod_cast hcardReal
102     exact Nat.le_of_lt hcardNat
103   · have hlargeEmpty : aboveTwiceAverageGap gap = ∅ :=
104       Finset.not_nonempty_iff_eq_empty.mp hlargeNonempty
105     simp [hlargeEmpty]

107 /-- The stochastic loss gap convention used by the delayed-SAPO source:
108 smaller mean loss is better, so the gap of arm `i` is
109 `mean i - mean optimal`. -/
110 def sourceStochasticLossGap {K : Nat}
111     (mean : Fin K -> Real) (optimal i : Fin K) : Real :=
112   mean i - mean optimal
113 
114 /-- An optimal arm makes every source stochastic loss gap nonnegative. -/
115 theorem sourceStochasticLossGap_nonneg
116     {K : Nat} (mean : Fin K -> Real) (optimal : Fin K)
117     (hoptimal : forall i, mean optimal <= mean i) (i : Fin K) :
118     0 <= sourceStochasticLossGap mean optimal i := by
119   unfold sourceStochasticLossGap
120   linarith [hoptimal i]
121 
122 /-- Bandit specialization of the nonnegative-domain D.11 counting statement.
123 
124 Among the `K` stochastic loss gaps from an optimal arm, strictly fewer than
125 half can exceed twice their average (and hence their cardinality is at most
126 half).  This is a deterministic producer; it does not assume or claim any
127 generated delayed-feedback probability law. -/
128 theorem two_mul_card_sourceStochasticLossGap_aboveTwiceAverage_le
129     {K : Nat} (mean : Fin K -> Real) (optimal : Fin K)
130     (hoptimal : forall i, mean optimal <= mean i) :
131     2 * (aboveTwiceAverageGap
132       (sourceStochasticLossGap mean optimal)).card <= K := by
133   apply two_mul_card_aboveTwiceAverageGap_le
134   intro i
135   exact sourceStochasticLossGap_nonneg mean optimal hoptimal i
