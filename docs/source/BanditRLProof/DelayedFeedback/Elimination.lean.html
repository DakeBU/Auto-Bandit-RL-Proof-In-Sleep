9 /-- Inputs read by Algorithm 5 line 7 when it processes one newly available
10 feedback item.  This is deliberately an elimination snapshot rather than a
11 claim that the full Delayed SAPO state machine has already been implemented. -/
12 structure DelayedSAPOEliminationSnapshot (K : Nat) where
13   active : Finset (Fin K)
14   empiricalMean : Fin K → ℝ
15   empiricalWidth : Fin K → ℝ
16   ucbStar : ℝ
17 
18 namespace DelayedSAPOEliminationSnapshot
19 
20 /-- Arms selected by the strict elimination test in Algorithm 5 line 7. -/
21 noncomputable def eliminated {K : Nat}
22     (snapshot : DelayedSAPOEliminationSnapshot K) :
23     Finset (Fin K) :=
24   snapshot.active.filter (fun i =>
25     snapshot.ucbStar <
26       snapshot.empiricalMean i - (9 : ℝ) * snapshot.empiricalWidth i)
27 
28 /-- Active set after Algorithm 5 line 8 removes every arm selected by line 7. -/
29 noncomputable def remainingActive {K : Nat}
30     (snapshot : DelayedSAPOEliminationSnapshot K) :
31     Finset (Fin K) :=
32   snapshot.active \ snapshot.eliminated
33 
34 @[simp]
35 theorem mem_eliminated_iff {K : Nat}
36     (snapshot : DelayedSAPOEliminationSnapshot K) (i : Fin K) :
37     i ∈ snapshot.eliminated ↔
38       i ∈ snapshot.active ∧
39         snapshot.ucbStar <
40           snapshot.empiricalMean i -
41             (9 : ℝ) * snapshot.empiricalWidth i := by
42   simp [eliminated]
43 
44 @[simp]
45 theorem mem_remainingActive_iff {K : Nat}
46     (snapshot : DelayedSAPOEliminationSnapshot K) (i : Fin K) :
47     i ∈ snapshot.remainingActive ↔
48       i ∈ snapshot.active ∧
49         snapshot.empiricalMean i -
50             (9 : ℝ) * snapshot.empiricalWidth i ≤ snapshot.ucbStar := by
51   simp only [remainingActive, Finset.mem_sdiff, mem_eliminated_iff, not_and_or]
52   by_cases hactive : i ∈ snapshot.active
53   · simp [hactive, not_lt]
54   · simp [hactive]

56 /-- The two source conditions used in the deterministic core of Lemma D.9:
57 the empirical mean of the optimal arm lies in its good-event confidence
58 interval, and the source's `ucbStar` remains an upper certificate for the
59 optimal mean.  The event probability establishing these fields is a separate
60 concentration obligation. -/
61 structure OptimalArmSurvivalCertificate {K : Nat}
62     (snapshot : DelayedSAPOEliminationSnapshot K)
63     (mean : Fin K → ℝ) (optimal : Fin K) : Prop where
64   optimal_active : optimal ∈ snapshot.active
65   width_nonnegative : 0 ≤ snapshot.empiricalWidth optimal
66   empirical_confidence :
67     |snapshot.empiricalMean optimal - mean optimal| ≤
68       snapshot.empiricalWidth optimal
69   optimalMean_le_ucbStar : mean optimal ≤ snapshot.ucbStar
70 
71 /-- Deterministic core of source Lemma D.9: on the relevant stochastic
72 good-event projections, Algorithm 5's strict line-7 test cannot eliminate the
73 certified optimal arm. -/
74 theorem optimal_mem_remainingActive_of_certificate {K : Nat}
75     (snapshot : DelayedSAPOEliminationSnapshot K)
76     (mean : Fin K → ℝ) (optimal : Fin K)
77     (certificate : OptimalArmSurvivalCertificate snapshot mean optimal) :
78     optimal ∈ snapshot.remainingActive := by
79   rw [mem_remainingActive_iff]
80   refine ⟨certificate.optimal_active, ?_⟩
81   have hempiricalUpper :
82       snapshot.empiricalMean optimal ≤
83         mean optimal + snapshot.empiricalWidth optimal := by
84     have hupper := (abs_le.mp certificate.empirical_confidence).2
85     linarith
86   linarith [certificate.width_nonnegative,
87     certificate.optimalMean_le_ucbStar]
88 
89 /-- Consequently the post-elimination active set is nonempty.  This closes
90 the exact nonemptiness premise required by Algorithm 5 line 15's residual
91 allocation. -/
92 theorem remainingActive_nonempty_of_certificate {K : Nat}
93     (snapshot : DelayedSAPOEliminationSnapshot K)
94     (mean : Fin K → ℝ) (optimal : Fin K)
95     (certificate : OptimalArmSurvivalCertificate snapshot mean optimal) :
96     snapshot.remainingActive.Nonempty :=
97   ⟨optimal, optimal_mem_remainingActive_of_certificate
98     snapshot mean optimal certificate⟩
99 
100 /-- Under the same optimal-arm survival certificate, the probability vector
101 computed after line 8 and line 15 has total mass exactly one.  EAP still has
102 to establish coordinate nonnegativity and the inactive-mass upper bound. -/
103 theorem sum_delayedSAPOProbability_after_elimination_eq_one {K : Nat}
104     (snapshot : DelayedSAPOEliminationSnapshot K)
105     (mean : Fin K → ℝ) (optimal : Fin K)
106     (certificate : OptimalArmSurvivalCertificate snapshot mean optimal)
107     (inactiveProbability : Fin K → ℝ) :
108     ∑ i, delayedSAPOProbability snapshot.remainingActive
109       inactiveProbability i = 1 := by
110   exact sum_delayedSAPOProbability_eq_one
111     snapshot.remainingActive inactiveProbability
112     (remainingActive_nonempty_of_certificate
113       snapshot mean optimal certificate)
