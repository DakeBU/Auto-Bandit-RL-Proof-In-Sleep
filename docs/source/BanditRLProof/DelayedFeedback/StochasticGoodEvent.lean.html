10 /-- The two armwise upper-confidence surfaces entering the source definition
11 of `ucbStar`.  The recursive construction of those surfaces from a processed
12 history remains outside this snapshot. -/
13 structure DelayedSAPOSourceConfidenceSnapshot (K : Nat) extends
14     DelayedSAPOEliminationSnapshot K where
15   empiricalUpper : Fin K → ℝ
16   importanceUpper : Fin K → ℝ
17 
18 namespace DelayedSAPOSourceConfidenceSnapshot
19 
20 /-- Source-shaped projection of
21 `ucbStar(S) = min_i {ucb_i(S), overline-ucb_i(S)}`. -/
22 noncomputable def sourceUcbStar {K : Nat} [Nonempty (Fin K)]
23     (snapshot : DelayedSAPOSourceConfidenceSnapshot K) : ℝ :=
24   Finset.univ.inf' Finset.univ_nonempty (fun i =>
25     min (snapshot.empiricalUpper i) (snapshot.importanceUpper i))
26 
27 /-- The elimination-relevant projection of the stochastic good event in
28 Definition D.1.  It records both empirical-mean confidence and the two upper
29 confidence surfaces used by `ucbStar`.  Count/phase/error/delay clauses from
30 the full source event are deliberately not included here. -/
31 structure EliminationGoodEvent {K : Nat} [Nonempty (Fin K)]
32     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
33     (mean : Fin K → ℝ) : Prop where
34   width_nonnegative : ∀ i, 0 ≤ snapshot.empiricalWidth i
35   empirical_confidence : ∀ i,
36     |snapshot.empiricalMean i - mean i| ≤ snapshot.empiricalWidth i
37   empiricalUpper_confidence : ∀ i, mean i ≤ snapshot.empiricalUpper i
38   importanceUpper_confidence : ∀ i, mean i ≤ snapshot.importanceUpper i
39   ucbStar_eq_source : snapshot.ucbStar = snapshot.sourceUcbStar
40 
41 /-- On the source-shaped elimination good event, the best loss mean is below
42 the exact minimum of both armwise upper-confidence surfaces. -/
43 theorem optimalMean_le_ucbStar_of_eliminationGoodEvent
44     {K : Nat} [Nonempty (Fin K)]
45     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
46     (mean : Fin K → ℝ) (optimal : Fin K)
47     (hoptimal : ∀ i, mean optimal ≤ mean i)
48     (hgood : EliminationGoodEvent snapshot mean) :
49     mean optimal ≤ snapshot.ucbStar := by
50   rw [hgood.ucbStar_eq_source, sourceUcbStar]
51   apply Finset.le_inf' Finset.univ_nonempty
52   intro i _hi
53   exact le_trans (hoptimal i)
54     (le_min (hgood.empiricalUpper_confidence i)
55       (hgood.importanceUpper_confidence i))
56 
57 /-- The source-shaped elimination projection constructs the exact certificate
58 consumed by the deterministic core of Lemma D.9; no independent
59 `mean optimal <= ucbStar` premise remains. -/
60 theorem optimalArmSurvivalCertificate_of_eliminationGoodEvent
61     {K : Nat} [Nonempty (Fin K)]
62     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
63     (mean : Fin K → ℝ) (optimal : Fin K)
64     (hoptimal : ∀ i, mean optimal ≤ mean i)
65     (hactive : optimal ∈ snapshot.active)
66     (hgood : EliminationGoodEvent snapshot mean) :
67     DelayedSAPOEliminationSnapshot.OptimalArmSurvivalCertificate
68       snapshot.toDelayedSAPOEliminationSnapshot mean optimal where
69   optimal_active := hactive
70   width_nonnegative := hgood.width_nonnegative optimal
71   empirical_confidence := hgood.empirical_confidence optimal
72   optimalMean_le_ucbStar :=
73     optimalMean_le_ucbStar_of_eliminationGoodEvent
74       snapshot mean optimal hoptimal hgood

76 /-- Source Lemma D.9 at one elimination snapshot, conditional only on the
77 elimination projection of Definition D.1 and current activity of the optimal
78 arm.  Establishing the full good-event probability and persistence over the
79 recursive state machine remain separate obligations. -/
80 theorem optimal_mem_remainingActive_of_eliminationGoodEvent
81     {K : Nat} [Nonempty (Fin K)]
82     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
83     (mean : Fin K → ℝ) (optimal : Fin K)
84     (hoptimal : ∀ i, mean optimal ≤ mean i)
85     (hactive : optimal ∈ snapshot.active)
86     (hgood : EliminationGoodEvent snapshot mean) :
87     optimal ∈ snapshot.remainingActive := by
88   exact DelayedSAPOEliminationSnapshot.optimal_mem_remainingActive_of_certificate
89     snapshot.toDelayedSAPOEliminationSnapshot mean optimal
90     (optimalArmSurvivalCertificate_of_eliminationGoodEvent
91       snapshot mean optimal hoptimal hactive hgood)

93 /-- Random-state event on which the elimination projection of Definition D.1
94 holds. -/
95 def eliminationGoodEventSet {Ω : Type*} {K : Nat} [Nonempty (Fin K)]
96     (snapshot : Ω → DelayedSAPOSourceConfidenceSnapshot K)
97     (mean : Fin K → ℝ) : Set Ω :=
98   {ω | EliminationGoodEvent (snapshot ω) mean}
99 
100 /-- Event that the optimal arm survives the current elimination update. -/
101 def optimalSurvivalEventSet {Ω : Type*} {K : Nat}
102     (snapshot : Ω → DelayedSAPOSourceConfidenceSnapshot K)
103     (optimal : Fin K) : Set Ω :=
104   {ω | optimal ∈ (snapshot ω).remainingActive}
105 
106 /-- The elimination good event is contained in the optimal-arm survival event
107 when the optimal arm is active in every input snapshot. -/
108 theorem eliminationGoodEventSet_subset_optimalSurvivalEventSet
109     {Ω : Type*} {K : Nat} [Nonempty (Fin K)]
110     (snapshot : Ω → DelayedSAPOSourceConfidenceSnapshot K)
111     (mean : Fin K → ℝ) (optimal : Fin K)
112     (hoptimal : ∀ i, mean optimal ≤ mean i)
113     (hactive : ∀ ω, optimal ∈ (snapshot ω).active) :
114     eliminationGoodEventSet snapshot mean ⊆
115       optimalSurvivalEventSet snapshot optimal := by
116   intro ω hgood
117   exact optimal_mem_remainingActive_of_eliminationGoodEvent
118     (snapshot ω) mean optimal hoptimal (hactive ω) hgood

120 /-- Any tail bound for the complement of the source-shaped good event
121 immediately controls the probability that the optimal arm is eliminated. -/
122 theorem measure_optimalSurvivalEventSet_compl_le
123     {Ω : Type*} [MeasurableSpace Ω] {K : Nat} [Nonempty (Fin K)]
124     (mu : Measure Ω)
125     (snapshot : Ω → DelayedSAPOSourceConfidenceSnapshot K)
126     (mean : Fin K → ℝ) (optimal : Fin K)
127     (hoptimal : ∀ i, mean optimal ≤ mean i)
128     (hactive : ∀ ω, optimal ∈ (snapshot ω).active) :
129     mu (optimalSurvivalEventSet snapshot optimal)ᶜ ≤
130       mu (eliminationGoodEventSet snapshot mean)ᶜ := by
131   apply measure_mono
132   intro ω hfailure hgood
133   exact hfailure
134     (eliminationGoodEventSet_subset_optimalSurvivalEventSet
135       snapshot mean optimal hoptimal hactive hgood)

137 /-- Failure-budget consumer for the D.9 projection.  The hypothesis must be
138 discharged by the D.2--D.7 concentration/counting development and the
139 Corollary-D.8 union assembly; this theorem does not manufacture that
140 probability estimate. -/
141 theorem measure_optimalSurvivalEventSet_compl_le_of_goodEvent
142     {Ω : Type*} [MeasurableSpace Ω] {K : Nat} [Nonempty (Fin K)]
143     (mu : Measure Ω)
144     (snapshot : Ω → DelayedSAPOSourceConfidenceSnapshot K)
145     (mean : Fin K → ℝ) (optimal : Fin K)
146     (delta : ℝ)
147     (hoptimal : ∀ i, mean optimal ≤ mean i)
148     (hactive : ∀ ω, optimal ∈ (snapshot ω).active)
149     (hgoodProbability :
150       mu (eliminationGoodEventSet snapshot mean)ᶜ ≤ ENNReal.ofReal delta) :
151     mu (optimalSurvivalEventSet snapshot optimal)ᶜ ≤ ENNReal.ofReal delta := by
152   exact (measure_optimalSurvivalEventSet_compl_le
153     mu snapshot mean optimal hoptimal hactive).trans hgoodProbability
