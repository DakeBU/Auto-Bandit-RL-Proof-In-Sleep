5 # Ordered no-switch structural traces for Delayed SAPO
6 
7 This module composes the source-faithful one-item transition from
8 `OrderedProcessingTransition.lean` across an arbitrary finite trace.  A trace
9 may either process one member of `B(t) \ S` through the no-switch structural
10 projection of Algorithm 5 lines 3--4 and 7--8, or close an exhausted inner
11 loop and advance to the next action round.
12 
13 The resulting active-set monotonicity discharges one genuine temporal premise
14 of the repaired D.12 / main-text Lemma-4.2 consumer: an arm eliminated later
15 must still belong to the earlier line-8 remaining set.  The factor-20 theorem
16 below remains conditional on the earlier snapshot's D.4 count clause and
17 elimination-good projection.  No numerical BSC/EAP producer, generated
18 probability law, D.4 probability bound, switch path, or regret endpoint is
19 constructed here.
20 -/

26 /-- Certificate that the no-switch inner loop for one action round is
27 exhausted and that its final intra-round active set is the line-15 active set
28 recorded by the source-round trace.
29 
30 The second field is a structural consistency contract.  It is not a claim
31 that EAP has already constructed a valid probability vector. -/
32 structure DelayedSAPONoSwitchRoundClose {K : Nat}
33     (state : DelayedSAPOStructuralRoundState K) : Prop where
34   noNew :
35     newlyObservedBefore state.delayAt state.processedOrder.toFinset
36       state.currentActionRound = ∅
37   currentActive_eq_sourceRound :
38     state.currentActive =
39       state.activeAtSourceRound state.currentActionRound

43 /-- Exhausting `B(t) \ S` means that the ordered ledger contains exactly the
44 feedback available before action `t`.  The equality is set-level only and does
45 not impose a chronological order on simultaneous arrivals. -/
46 theorem processedOrder_toFinset_eq_observedBefore {K : Nat}
47     {state : DelayedSAPOStructuralRoundState K}
48     (closed : DelayedSAPONoSwitchRoundClose state) :
49     state.processedOrder.toFinset =
50       observedBefore state.delayAt state.currentActionRound := by
51   apply Finset.Subset.antisymm
52   · intro s hs
53     have hsList : s ∈ state.processedOrder := by
54       simpa using hs
55     have havailable := state.processedOrder_available s hsList
56     rw [observedBefore, Finset.mem_filter]
57     exact ⟨Finset.mem_range.mpr (by omega), havailable⟩
58   · intro s hs
59     by_contra hprocessed
60     have hnew :
61         s ∈ newlyObservedBefore state.delayAt
62           state.processedOrder.toFinset state.currentActionRound := by
63       exact Finset.mem_sdiff.mpr ⟨hs, hprocessed⟩
64     rw [closed.noNew] at hnew
65     simp at hnew

67 /-- Structural state at the start of the next action round.  The processed
68 ledger and active set are unchanged; the round-start invariant follows from
69 the close certificate's identification with the just-finished source-round
70 active set. -/
71 def nextRoundState {K : Nat}
72     {state : DelayedSAPOStructuralRoundState K}
73     (closed : DelayedSAPONoSwitchRoundClose state) :
74     DelayedSAPOStructuralRoundState K where
75   currentActionRound := state.currentActionRound + 1
76   currentActionRound_pos := Nat.succ_pos state.currentActionRound
77   delayAt := state.delayAt
78   processedOrder := state.processedOrder
79   processedOrder_nodup := state.processedOrder_nodup
80   processedOrder_available := by
81     intro s hs
82     exact lt_trans (state.processedOrder_available s hs)
83       (Nat.lt_succ_self state.currentActionRound)
84   activeAtSourceRound := state.activeAtSourceRound
85   sourceActive_antitone := state.sourceActive_antitone
86   currentActive := state.currentActive
87   currentActive_subset_roundStart := by
88     have hsource :
89         state.currentActive <=
90           state.activeAtSourceRound state.currentActionRound := by
91       rw [closed.currentActive_eq_sourceRound]
92     simpa using hsource
93   chosenArmAt := state.chosenArmAt
94   inactiveProbabilityAt := state.inactiveProbabilityAt

117 /-- One edge of the deterministic no-switch structural trace.  Processing
118 uses the exact line-8 successor; advancing rounds requires an exhausted-loop
119 certificate. -/
120 inductive DelayedSAPONoSwitchStructuralStep {K : Nat}
121     (horizon : Nat) :
122     DelayedSAPOStructuralRoundState K →
123       DelayedSAPOStructuralRoundState K → Prop
124   | process {state : DelayedSAPOStructuralRoundState K}
125       (step : DelayedSAPONoSwitchProcessOne state) :
126       DelayedSAPONoSwitchStructuralStep horizon state
127         (step.afterLine8 horizon)
128   | nextRound {state : DelayedSAPOStructuralRoundState K}
129       (closed : DelayedSAPONoSwitchRoundClose state) :
130       DelayedSAPONoSwitchStructuralStep horizon state
131         closed.nextRoundState

133 /-- Finite reflexive-transitive no-switch reachability.  It is a structural
134 relation, not a generated stochastic trajectory. -/
135 abbrev DelayedSAPONoSwitchStructuralReachable {K : Nat}
136     (horizon : Nat) :=
137   Relation.ReflTransGen
138     (DelayedSAPONoSwitchStructuralStep (K := K) horizon)
139 
140 /-- Every primitive no-switch structural edge can only remove arms. -/
141 theorem currentActive_subset_of_structuralStep {K horizon : Nat}
142     {initial final : DelayedSAPOStructuralRoundState K}
143     (step :
144       DelayedSAPONoSwitchStructuralStep horizon initial final) :
145     final.currentActive <= initial.currentActive := by
146   cases step with
147   | process processStep =>
148       exact processStep.afterLine8_currentActive_subset_before horizon
149   | nextRound _closed =>
150       exact Finset.Subset.rfl
151 
152 /-- Active sets are antitone along every finite no-switch structural trace. -/
153 theorem currentActive_subset_of_structuralReachable {K horizon : Nat}
154     {initial final : DelayedSAPOStructuralRoundState K}
155     (run :
156       DelayedSAPONoSwitchStructuralReachable horizon initial final) :
157     final.currentActive <= initial.currentActive := by
158   induction run using Relation.ReflTransGen.trans_induction_on with
159   | refl _state =>
160       exact Finset.Subset.rfl
161   | single edge =>
162       exact currentActive_subset_of_structuralStep edge
163   | trans _left _right hleft hright =>
164       exact hright.trans hleft

166 /-- An arm eliminated by a later processing step was still present after an
167 earlier line-8 removal whenever the two steps are connected by a no-switch
168 structural trace.  This is the temporal premise previously left to callers. -/
169 theorem mem_earlierRemainingActive_of_laterEliminated
170     {K horizon : Nat}
171     {initial laterState : DelayedSAPOStructuralRoundState K}
172     (earlierStep : DelayedSAPONoSwitchProcessOne initial)
173     (between :
174       DelayedSAPONoSwitchStructuralReachable horizon
175         (earlierStep.afterLine8 horizon) laterState)
176     (laterStep : DelayedSAPONoSwitchProcessOne laterState)
177     (iLater : Fin K)
178     (hLaterEliminated :
179       iLater ∈
180         (laterStep.toPreEliminationSummary.toConfidenceSnapshot horizon).eliminated) :
181     iLater ∈
182       (earlierStep.toPreEliminationSummary.toConfidenceSnapshot horizon).remainingActive := by
183   have hLaterActive : iLater ∈ laterState.currentActive :=
184     (DelayedSAPOEliminationSnapshot.mem_eliminated_iff
185       (laterStep.toPreEliminationSummary.toConfidenceSnapshot horizon).toDelayedSAPOEliminationSnapshot
186       iLater).mp hLaterEliminated |>.1
187   have hEarlierAfter :
188       iLater ∈ (earlierStep.afterLine8 horizon).currentActive :=
189     currentActive_subset_of_structuralReachable between hLaterActive
190   exact hEarlierAfter

192 /-- Ordered two-elimination factor-20 consumer.  Unlike the one-snapshot
193 consumer, this theorem derives later-arm survival at the earlier snapshot from
194 the exact structural trace.  It remains conditional on the earlier D.4 count
195 clause and elimination-good projection. -/
196 theorem gap_le_twenty_mul_gap_of_ordered_no_switch_eliminations
197     {K : Nat} [Nonempty (Fin K)]
198     {initial laterState : DelayedSAPOStructuralRoundState K}
199     (horizon : Nat) (hhorizon : 1 < horizon)
200     (earlierStep : DelayedSAPONoSwitchProcessOne initial)
201     (between :
202       DelayedSAPONoSwitchStructuralReachable horizon
203         (earlierStep.afterLine8 horizon) laterState)
204     (laterStep : DelayedSAPONoSwitchProcessOne laterState)
205     (mean : Fin K → Real) (optimal iEarlier iLater : Fin K)
206     (hoptimal : ∀ i, mean optimal <= mean i)
207     (hmeanBounds : ∀ i, mean i ∈ Set.Icc (0 : Real) 1)
208     (hD4 :
209       earlierStep.toPreEliminationSummary.D4CountClause horizon)
210     (hgood :
211       (earlierStep.toPreEliminationSummary.toConfidenceSnapshot horizon).EliminationGoodEvent
212         mean)
213     (hoptimalActive : optimal ∈ initial.currentActive)
214     (hEarlierEliminated :
215       iEarlier ∈
216         (earlierStep.toPreEliminationSummary.toConfidenceSnapshot horizon).eliminated)
217     (hLaterEliminated :
218       iLater ∈
219         (laterStep.toPreEliminationSummary.toConfidenceSnapshot horizon).eliminated) :
220     mean iLater - mean optimal <=
221       20 * (mean iEarlier - mean optimal) := by
222   have hLaterRemaining :=
223     mem_earlierRemainingActive_of_laterEliminated earlierStep between laterStep
224       iLater hLaterEliminated
225   exact
226     earlierStep.toPreEliminationSummary.gap_le_twenty_mul_gap_at_earlier_elimination_snapshot_of_traceSummary
227         horizon hD4 hhorizon mean optimal iEarlier iLater hoptimal hmeanBounds
228           hgood hoptimalActive hEarlierEliminated hLaterRemaining
