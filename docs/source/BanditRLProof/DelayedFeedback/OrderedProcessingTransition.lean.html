5 # One ordered no-switch processing step for Delayed SAPO
6 
7 This module formalizes the deterministic structural content of one iteration
8 of Algorithm 5 lines 3--4 and 7--8.  A newly observed source round is appended
9 to the paper sequence before the line-7 confidence snapshot is formed.  The
10 line-8 successor then removes exactly the arms selected by that snapshot.
11 
12 The source leaves the order of simultaneous arrivals unspecified, so the step
13 accepts any member of `B(t) \ S` and never sorts by source time.  It models only
14 the branch on which the line-5 BSC does not switch away from Delayed SAPO.
15 Numerical confidence surfaces remain explicit structural inputs: this module
16 does not implement their recursive update, BSC, EAP, a generated trajectory,
17 the D.4 probability bound, ordered multi-snapshot elimination, or any regret
18 endpoint.
19 -/

25 /-- Structural state while Algorithm 5 is processing newly observed feedback
26 before action `currentActionRound`.
27 
28 `processedOrder` is the paper sequence `S`, not a sorted set of source rounds.
29 The current intra-round active set is contained in the previous action round's
30 line-15 active set.  Together with the antitone source-round trace, this is the
31 primitive invariant from which a line-7 trace summary obtains current-to-source
32 active persistence. -/
33 structure DelayedSAPOStructuralRoundState (K : Nat) where
34   currentActionRound : Nat
35   currentActionRound_pos : 0 < currentActionRound
36   delayAt : Nat -> Nat
37   processedOrder : List Nat
38   processedOrder_nodup : processedOrder.Nodup
39   processedOrder_available : forall s, s ∈ processedOrder ->
40     s + delayAt s < currentActionRound
41   activeAtSourceRound : Nat -> Finset (Fin K)
42   sourceActive_antitone : Antitone activeAtSourceRound
43   currentActive : Finset (Fin K)
44   currentActive_subset_roundStart :
45     currentActive <= activeAtSourceRound (currentActionRound - 1)
46   chosenArmAt : Nat -> Fin K
47   inactiveProbabilityAt : Nat -> Fin K -> Real
48

51 /-- Every source already in the processed order lies no later than the action
52 round immediately preceding the current one. -/
53 theorem source_le_roundStart {K : Nat}
54     (state : DelayedSAPOStructuralRoundState K) {s : Nat}
55     (hs : s ∈ state.processedOrder) :
56     s <= state.currentActionRound - 1 := by
57   have havailable := state.processedOrder_available s hs
58   omega
59 
60 /-- The current intra-round active set is contained in the source-round active
61 set of every previously processed item. -/
62 theorem currentActive_subset_activeAtSourceRound {K : Nat}
63     (state : DelayedSAPOStructuralRoundState K) {s : Nat}
64     (hs : s ∈ state.processedOrder) :
65     state.currentActive <= state.activeAtSourceRound s := by
66   exact state.currentActive_subset_roundStart.trans
67     (state.sourceActive_antitone (state.source_le_roundStart hs))
68 
69 end DelayedSAPOStructuralRoundState

71 /-- One source-faithful no-switch iteration of Algorithm 5's inner processing
72 loop.  `source_new` is exactly line 3's membership in `B(t) \ S`.  The numerical
73 fields are the values read by line 7 after line 4 has appended `sourceRound`;
74 constructing them from observed losses is a separate recursive-state leaf. -/
75 structure DelayedSAPONoSwitchProcessOne {K : Nat}
76     (state : DelayedSAPOStructuralRoundState K) where
77   sourceRound : Nat
78   source_new : sourceRound ∈ newlyObservedBefore state.delayAt
79     state.processedOrder.toFinset state.currentActionRound
80   empiricalMean : Fin K -> Real
81   importanceUpper : Fin K -> Real
82   previousEmpiricalUpper : Fin K -> Real
83   ucbStar : Real

87 /-- Algorithm 5 line 4: append the selected source to the end of the processing
88 sequence. -/
89 def extendedOrder {K : Nat} {state : DelayedSAPOStructuralRoundState K}
90     (step : DelayedSAPONoSwitchProcessOne state) : List Nat :=
91   state.processedOrder.concat step.sourceRound
92 
93 /-- The line-3 source was not already present in the processed sequence. -/
94 theorem sourceRound_not_mem {K : Nat}
95     {state : DelayedSAPOStructuralRoundState K}
96     (step : DelayedSAPONoSwitchProcessOne state) :
97     step.sourceRound ∉ state.processedOrder := by
98   have hnotFinset := (Finset.mem_sdiff.mp step.source_new).2
99   simpa using hnotFinset
100 
101 /-- Appending one genuinely new source preserves duplicate freedom. -/
102 theorem extendedOrder_nodup {K : Nat}
103     {state : DelayedSAPOStructuralRoundState K}
104     (step : DelayedSAPONoSwitchProcessOne state) :
105     step.extendedOrder.Nodup := by
106   rw [extendedOrder, List.nodup_concat]
107   exact ⟨step.sourceRound_not_mem, state.processedOrder_nodup⟩
108 
109 /-- Every source in the extended sequence satisfies the paper's exact strict
110 availability condition at the current action round. -/
111 theorem extendedOrder_available {K : Nat}
112     {state : DelayedSAPOStructuralRoundState K}
113     (step : DelayedSAPONoSwitchProcessOne state) {s : Nat}
114     (hs : s ∈ step.extendedOrder) :
115     s + state.delayAt s < state.currentActionRound := by
116   have hcases : s ∈ state.processedOrder ∨ s = step.sourceRound := by
117     simpa [extendedOrder] using hs
118   rcases hcases with hold | rfl
119   · exact state.processedOrder_available s hold
120   · exact (Finset.mem_filter.mp (Finset.mem_sdiff.mp step.source_new).1).2
121 
122 /-- Every source in the extended sequence is at most the previous action
123 round. -/
124 theorem extendedSource_le_roundStart {K : Nat}
125     {state : DelayedSAPOStructuralRoundState K}
126     (step : DelayedSAPONoSwitchProcessOne state) {s : Nat}
127     (hs : s ∈ step.extendedOrder) :
128     s <= state.currentActionRound - 1 := by
129   have havailable := step.extendedOrder_available hs
130   omega

132 /-- The line-7 active set is contained in every source-time line-15 active set
133 represented by the extended processing sequence. -/
134 theorem currentActive_subset_extendedSourceActive {K : Nat}
135     {state : DelayedSAPOStructuralRoundState K}
136     (step : DelayedSAPONoSwitchProcessOne state) {s : Nat}
137     (hs : s ∈ step.extendedOrder) :
138     state.currentActive <= state.activeAtSourceRound s := by
139   exact state.currentActive_subset_roundStart.trans
140     (state.sourceActive_antitone (step.extendedSource_le_roundStart hs))

142 /-- Line-7 trace summary after, not before, Algorithm 5 line 4 appends the
143 newly observed source.  Its source-index injectivity, strict availability, and
144 current-to-source containment are derived from the ordered transition state. -/
145 def toPreEliminationSummary {K : Nat}
146     {state : DelayedSAPOStructuralRoundState K}
147     (step : DelayedSAPONoSwitchProcessOne state) :
148     DelayedSAPOProcessedTraceSummary K where
149   length := step.extendedOrder.length
150   sourceIndex := step.extendedOrder.get
151   sourceIndex_injective := step.extendedOrder_nodup.injective_get
152   currentActionRound := state.currentActionRound
153   delayAt := state.delayAt
154   source_available := by
155     intro q
156     exact step.extendedOrder_available (List.get_mem step.extendedOrder q)
157   activeAtSourceRound := state.activeAtSourceRound
158   sourceActive_antitone := state.sourceActive_antitone
159   currentActive := state.currentActive
160   currentActive_subset_sourceActive := by
161     intro q
162     exact step.currentActive_subset_extendedSourceActive
163       (List.get_mem step.extendedOrder q)
164   chosenArmAt := state.chosenArmAt
165   inactiveProbabilityAt := state.inactiveProbabilityAt
166   empiricalMean := step.empiricalMean
167   importanceUpper := step.importanceUpper
168   previousEmpiricalUpper := step.previousEmpiricalUpper
169   ucbStar := step.ucbStar

171 /-- The exact Algorithm-5 line-8 removal is contained in the active set read by
172 the line-7 snapshot. -/
173 theorem line8RemainingActive_subset_currentActive {K : Nat}
174     {state : DelayedSAPOStructuralRoundState K}
175     (step : DelayedSAPONoSwitchProcessOne state) (horizon : Nat) :
176     (step.toPreEliminationSummary.toConfidenceSnapshot horizon).remainingActive <=
177       state.currentActive := by
178   intro i hi
179   have hiActive :
180       i ∈ (step.toPreEliminationSummary.toConfidenceSnapshot horizon).active :=
181     (DelayedSAPOEliminationSnapshot.mem_remainingActive_iff
182       (step.toPreEliminationSummary.toConfidenceSnapshot horizon).toDelayedSAPOEliminationSnapshot
183       i).mp hi |>.1
184   exact hiActive

186 /-- Algorithm 5 line 8 successor for the same action round.  It preserves the
187 ordered source sequence and updates only the intra-round active set to the
188 exact line-7 complement. -/
189 noncomputable def afterLine8 {K : Nat}
190     {state : DelayedSAPOStructuralRoundState K}
191     (step : DelayedSAPONoSwitchProcessOne state) (horizon : Nat) :
192     DelayedSAPOStructuralRoundState K where
193   currentActionRound := state.currentActionRound
194   currentActionRound_pos := state.currentActionRound_pos
195   delayAt := state.delayAt
196   processedOrder := step.extendedOrder
197   processedOrder_nodup := step.extendedOrder_nodup
198   processedOrder_available := by
199     intro s hs
200     exact step.extendedOrder_available hs
201   activeAtSourceRound := state.activeAtSourceRound
202   sourceActive_antitone := state.sourceActive_antitone
203   currentActive :=
204     (step.toPreEliminationSummary.toConfidenceSnapshot horizon).remainingActive
205   currentActive_subset_roundStart :=
206     (step.line8RemainingActive_subset_currentActive horizon).trans
207       state.currentActive_subset_roundStart
208   chosenArmAt := state.chosenArmAt
209   inactiveProbabilityAt := state.inactiveProbabilityAt

211 /-- Line 8 can only remove arms from the current intra-round active set. -/
212 theorem afterLine8_currentActive_subset_before {K : Nat}
213     {state : DelayedSAPOStructuralRoundState K}
214     (step : DelayedSAPONoSwitchProcessOne state) (horizon : Nat) :
215     (step.afterLine8 horizon).currentActive <= state.currentActive := by
216   exact step.line8RemainingActive_subset_currentActive horizon
217 
218 /-- The line-8 successor remains below the active set at the start of the
219 current action round, so another arbitrary new arrival can be processed. -/
220 theorem afterLine8_preserves_roundStart {K : Nat}
221     {state : DelayedSAPOStructuralRoundState K}
222     (step : DelayedSAPONoSwitchProcessOne state) (horizon : Nat) :
223     (step.afterLine8 horizon).currentActive <=
224       state.activeAtSourceRound (state.currentActionRound - 1) := by
225   exact (step.afterLine8 horizon).currentActive_subset_roundStart

227 end DelayedSAPONoSwitchProcessOne
