    1 import Mathlib
2 import BanditRLProof.DelayedFeedback.CausalView

6 namespace DelayedFeedback

8 open scoped BigOperators
9 
10 /-- Arms outside the current active set in Algorithm 5. -/
11 def inactiveArms {K : Nat} (active : Finset (Fin K)) : Finset (Fin K) :=
12   Finset.univ \ active
13 
14 /-- Algorithm 5 line 15 assigns the residual mass equally to active arms. -/
15 noncomputable def activeEqualShare {K : Nat}
16     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ) : ℝ :=
17   (1 - ∑ i ∈ inactiveArms active, inactiveProbability i) /
18     (active.card : ℝ)
19 
20 /-- Full probability vector obtained from externally maintained probabilities
21 on eliminated arms and equal residual mass on active arms. -/
22 noncomputable def delayedSAPOProbability {K : Nat}
23     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
24     (i : Fin K) : ℝ :=
25   if i ∈ active then activeEqualShare active inactiveProbability
26   else inactiveProbability i
27 
28 @[simp]
29 theorem delayedSAPOProbability_of_active {K : Nat}
30     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
31     {i : Fin K} (hi : i ∈ active) :
32     delayedSAPOProbability active inactiveProbability i =
33       activeEqualShare active inactiveProbability := by
34   simp [delayedSAPOProbability, hi]
35 
36 @[simp]
37 theorem delayedSAPOProbability_of_inactive {K : Nat}
38     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
39     {i : Fin K} (hi : i ∉ active) :
40     delayedSAPOProbability active inactiveProbability i =
41       inactiveProbability i := by
42   simp [delayedSAPOProbability, hi]
43 
44 /-- The residual active-arm share is nonnegative when eliminated-arm mass does
45 not exceed one. -/
46 theorem activeEqualShare_nonneg {K : Nat}
47     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
48     (hmass : (∑ i ∈ inactiveArms active, inactiveProbability i) ≤ 1) :
49     0 ≤ activeEqualShare active inactiveProbability := by
50   exact div_nonneg (sub_nonneg.mpr hmass) (Nat.cast_nonneg active.card)
51 
52 /-- Every coordinate of the line-15 allocation is nonnegative under the
53 source-side residual-mass and inactive-coordinate hypotheses. -/
54 theorem delayedSAPOProbability_nonneg {K : Nat}
55     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
56     (hinactive : ∀ i ∈ inactiveArms active, 0 ≤ inactiveProbability i)
57     (hmass : (∑ i ∈ inactiveArms active, inactiveProbability i) ≤ 1)
58     (i : Fin K) :
59     0 ≤ delayedSAPOProbability active inactiveProbability i := by
60   by_cases hi : i ∈ active
61   · rw [delayedSAPOProbability_of_active active inactiveProbability hi]
62     exact activeEqualShare_nonneg active inactiveProbability hmass
63   · rw [delayedSAPOProbability_of_inactive active inactiveProbability hi]
64     exact hinactive i (by simp [inactiveArms, hi])
65 
66 /-- With at least one active arm, the equal-residual allocation has total mass
67 exactly one. -/
68 theorem sum_delayedSAPOProbability_eq_one {K : Nat}
69     (active : Finset (Fin K)) (inactiveProbability : Fin K → ℝ)
70     (hactive : active.Nonempty) :
71     ∑ i, delayedSAPOProbability active inactiveProbability i = 1 := by
72   classical
73   have hcardNat : active.card ≠ 0 := Finset.card_ne_zero.mpr hactive
74   have hcardReal : (active.card : ℝ) ≠ 0 := by
75     exact_mod_cast hcardNat
76   rw [← Finset.sum_filter_add_sum_filter_not
77     (s := Finset.univ) (p := fun i : Fin K => i ∈ active)
78     (f := delayedSAPOProbability active inactiveProbability)]
79   have hactiveFilter :
80       Finset.univ.filter (fun i : Fin K => i ∈ active) = active := by
81     ext i
82     simp
83   have hinactiveFilter :
84       Finset.univ.filter (fun i : Fin K => ¬ i ∈ active) =
85         inactiveArms active := by
86     ext i
87     simp [inactiveArms]
88   rw [hactiveFilter, hinactiveFilter]
89   have hactiveSum :
90       ∑ i ∈ active, delayedSAPOProbability active inactiveProbability i =
91         (active.card : ℝ) * activeEqualShare active inactiveProbability := by
92     calc
93       ∑ i ∈ active, delayedSAPOProbability active inactiveProbability i =
94           ∑ _i ∈ active, activeEqualShare active inactiveProbability := by
95             apply Finset.sum_congr rfl
96             intro i hi
97             exact delayedSAPOProbability_of_active active inactiveProbability hi
98       _ = (active.card : ℝ) * activeEqualShare active inactiveProbability := by
99         simp
100   have hinactiveSum :
101       ∑ i ∈ inactiveArms active,
102           delayedSAPOProbability active inactiveProbability i =
103         ∑ i ∈ inactiveArms active, inactiveProbability i := by
104     apply Finset.sum_congr rfl
105     intro i hi
106     apply delayedSAPOProbability_of_inactive
107     simpa [inactiveArms] using hi
108   rw [hactiveSum, hinactiveSum]
109   unfold activeEqualShare
110   field_simp [hcardReal]
111   ring
