4 # Algorithm 5 line-10 initialization for eliminated Delayed-SAPO arms
5 
6 This module records the numerical state created for each arm in the line-7
7 elimination set after line 8 of the frozen Delayed-SAPO algorithm.  It follows
8 physical PDF page 22, Algorithm 5 lines 9--10: the elimination round and
9 processed prefix are frozen, the initial inactive-arm probability is formed
10 from the processed pull count, the surrogate gap is eight empirical widths,
11 and the first EAP phase is initialized.
12 
13 The definitions below do not run EAP, prove BSC, construct a generated
14 trajectory, or establish any stochastic good-event probability.  In
15 particular, totalized real division makes the displayed source formula a Lean
16 term even at a zero surrogate gap; positivity of the first phase target is
17 proved only under the explicit positive-width condition that the mathematical
18 analysis needs.
19 -/

25 /-- Algorithm 5 line 10's initial sampling probability
26 `p_i^1 = 1/(2K) + n_i(S)/(2T)`. -/
27 noncomputable def delayedSAPOInitialEliminatedProbability
28     (armCount horizon pullCount : Nat) : Real :=
29   1 / (2 * (armCount : Real)) +
30     (pullCount : Real) / (2 * (horizon : Real))
31 
32 /-- Algorithm 5 line 10's first EAP phase target
33 `N_i^1 = 1280/(p_i^1 * (Delta-tilde_i)^2)`. -/
34 noncomputable def delayedSAPOInitialPhaseTarget
35     (probability surrogateGap : Real) : Real :=
36   1280 / (probability * surrogateGap ^ 2)
37 
38 /-- A positive source scale makes the capped inverse-square-root width
39 strictly positive, including the zero-count branch where the width is one. -/
40 theorem sourceEmpiricalWidthScale_pos (scale count : Real)
41     (hscale : 0 < scale) :
42     0 < sourceEmpiricalWidthScale scale count := by
43   by_cases hcount : count <= 0
44   · simp [sourceEmpiricalWidthScale, hcount]
45   · rw [sourceEmpiricalWidthScale, if_neg hcount]
46     exact lt_min (by norm_num)
47       (Real.sqrt_pos.2 (div_pos hscale (lt_of_not_ge hcount)))

49 /-- Per-arm data created by Algorithm 5 line 10.
50 
51 `processedAtProbabilityLevel j` represents the source sets
52 `C_i^(p_i^1 * 2^(-j))`; line 10 initializes every represented level to the
53 empty set.  Later EAP calls and their phase transitions remain a separate
54 state-machine leaf. -/
55 structure DelayedSAPOEliminatedArmInitialization (K : Nat) where
56   arm : Fin K
57   eliminationRound : Nat
58   initialProbability : Real
59   eliminationProcessedOrder : List Nat
60   eliminationEmpiricalMean : Real
61   surrogateGap : Real
62   initialPhaseTarget : Real
63   errorCount : Nat
64   phaseIndex : Nat
65   phaseSamples : List Nat
66   processedAtProbabilityLevel : Nat -> Finset Nat
67 
68 /-- Registry of EAP data.  Active arms have not yet entered EAP and are
69 therefore represented by `none`. -/
70 abbrev DelayedSAPOEliminatedArmBank (K : Nat) :=
71   Fin K -> Option (DelayedSAPOEliminatedArmInitialization K)

73 /-- Round-start invariant needed to justify that line 10 initializes rather
74 than resets every arm in the newly eliminated set. -/
75 def ActiveArmsUninitialized {K : Nat}
76     (state : DelayedSAPOStructuralRoundState K)
77     (bank : DelayedSAPOEliminatedArmBank K) : Prop :=
78   forall i, i ∈ state.currentActive -> bank i = none

82 /-- Construct exactly the line-10 data for one candidate arm from the
83 post-line-4 numerical snapshot.  Membership in the line-7 elimination set is
84 enforced by `initializeIfEliminated` below. -/
85 noncomputable def ofProcessOne {K : Nat}
86     {state : DelayedSAPOStructuralRoundState K}
87     (step : DelayedSAPONoSwitchProcessOne state)
88     (horizon : Nat) (i : Fin K) :
89     DelayedSAPOEliminatedArmInitialization K :=
90   let summary := step.toPreEliminationSummary
91   let pullCount := summary.toProcessedPrefix.processedPullCount i
92   let probability :=
93     delayedSAPOInitialEliminatedProbability K horizon pullCount
94   let gap := 8 * summary.empiricalWidthAt horizon i
95   {
96     arm := i
97     eliminationRound := state.currentActionRound
98     initialProbability := probability
99     eliminationProcessedOrder := step.extendedOrder
100     eliminationEmpiricalMean := summary.empiricalMean i
101     surrogateGap := gap
102     initialPhaseTarget := delayedSAPOInitialPhaseTarget probability gap
103     errorCount := 0
104     phaseIndex := 1
105     phaseSamples := []
106     processedAtProbabilityLevel := fun _ => {}
107   }

109 /-- Algorithm 5 lines 9--10 initialize precisely the arms selected by the
110 line-7 elimination set.  Returning `Option` keeps that domain visible instead
111 of silently producing EAP state for an arm that remains active. -/
112 noncomputable def initializeIfEliminated {K : Nat}
113     {state : DelayedSAPOStructuralRoundState K}
114     (step : DelayedSAPONoSwitchProcessOne state)
115     (horizon : Nat) (i : Fin K) :
116     Option (DelayedSAPOEliminatedArmInitialization K) :=
117   if i ∈ (step.toPreEliminationSummary.toConfidenceSnapshot horizon).eliminated
118   then some (ofProcessOne step horizon i)
119   else none

121 /-- Pointwise line-10 update of an existing EAP registry.  Arms in the new
122 line-7 elimination set receive the literal source initializer; every other
123 arm keeps its prior state. -/
124 noncomputable def initializeNewlyEliminated {K : Nat}
125     {state : DelayedSAPOStructuralRoundState K}
126     (step : DelayedSAPONoSwitchProcessOne state)
127     (horizon : Nat)
128     (prior : DelayedSAPOEliminatedArmBank K) :
129     DelayedSAPOEliminatedArmBank K :=
130   fun i =>
131     if i ∈ (step.toPreEliminationSummary.toConfidenceSnapshot horizon).eliminated
132     then some (ofProcessOne step horizon i)
133     else prior i

323 /-- The source initial eliminated-arm probability is strictly positive when
324 there is an arm and the horizon is positive.  No count upper bound is needed
325 for this lower endpoint. -/
326 theorem initialProbability_pos {K : Nat}
327     {state : DelayedSAPOStructuralRoundState K}
328     (step : DelayedSAPONoSwitchProcessOne state)
329     (horizon : Nat) (i : Fin K) :
330     0 < (ofProcessOne step horizon i).initialProbability := by
331   have hK : 0 < (K : Real) := by
332     exact_mod_cast (Nat.zero_lt_of_lt i.isLt)
333   have hfirst :
334       0 < (1 : Real) / (2 * (K : Real)) := by positivity
335   have hsecond :
336       0 <=
337         ((step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i : Nat) : Real) /
338           (2 * (horizon : Real)) := by positivity
339   change 0 < (1 : Real) / (2 * (K : Real)) +
340     ((step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i : Nat) : Real) /
341       (2 * (horizon : Real))
342   linarith

344 /-- With the source-side count bound `n_i(S) <= T`, the initial probability
345 is at most one.  The count bound will be generated by the full Algorithm-5
346 trajectory; it is not manufactured by this numerical initializer. -/
347 theorem initialProbability_le_one {K : Nat}
348     {state : DelayedSAPOStructuralRoundState K}
349     (step : DelayedSAPONoSwitchProcessOne state)
350     (horizon : Nat) (i : Fin K) (hhorizon : 0 < horizon)
351     (hcount :
352       step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i <=
353         horizon) :
354     (ofProcessOne step horizon i).initialProbability <= 1 := by
355   have hK : (1 : Real) <= (K : Real) := by
356     exact_mod_cast (Nat.succ_le_iff.mpr (Nat.zero_lt_of_lt i.isLt))
357   have hT : 0 < (horizon : Real) := by
358     exact_mod_cast hhorizon
359   have hfirst :
360       (1 : Real) / (2 * (K : Real)) <= 1 / 2 := by
361     apply div_le_div_of_nonneg_left (by norm_num) (by positivity)
362     nlinarith
363   have hcountReal :
364       ((step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i : Nat) : Real) <=
365         (horizon : Real) := by exact_mod_cast hcount
366   have hsecond :
367       ((step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i : Nat) : Real) /
368           (2 * (horizon : Real)) <= 1 / 2 := by
369     apply (div_le_iff₀ (by positivity : (0 : Real) < 2 * (horizon : Real))).2
370     nlinarith
371   change (1 : Real) / (2 * (K : Real)) +
372       ((step.toPreEliminationSummary.toProcessedPrefix.processedPullCount i : Nat) : Real) /
373         (2 * (horizon : Real)) <= 1
374   linarith

376 /-- The line-10 surrogate gap is nonnegative because the exact source width
377 is capped below by zero. -/
378 theorem surrogateGap_nonneg {K : Nat}
379     {state : DelayedSAPOStructuralRoundState K}
380     (step : DelayedSAPONoSwitchProcessOne state)
381     (horizon : Nat) (i : Fin K) :
382     0 <= (ofProcessOne step horizon i).surrogateGap := by
383   change 0 <= 8 * step.toPreEliminationSummary.empiricalWidthAt horizon i
384   exact mul_nonneg (by norm_num)

388 /-- For the nontrivial source regime `1 < T`, the frozen line-10 surrogate
389 gap is strictly positive for every processed count. -/
390 theorem surrogateGap_pos {K : Nat}
391     {state : DelayedSAPOStructuralRoundState K}
392     (step : DelayedSAPONoSwitchProcessOne state)
393     (horizon : Nat) (i : Fin K) (hhorizon : 1 < horizon) :
394     0 < (ofProcessOne step horizon i).surrogateGap := by
395   have hhorizonReal : (1 : Real) < (horizon : Real) := by
396     exact_mod_cast hhorizon
397   have hscale : 0 < 2 * Real.log (horizon : Real) := by
398     have hlog := Real.log_pos hhorizonReal
399     linarith
400   change 0 < 8 * step.toPreEliminationSummary.empiricalWidthAt horizon i
401   have hwidth :
402       0 < step.toPreEliminationSummary.empiricalWidthAt horizon i := by
403     exact sourceEmpiricalWidthScale_pos _ _ hscale
404   positivity

406 /-- The first EAP phase target is nonnegative without hiding Lean's
407 totalized-division boundary. -/
408 theorem initialPhaseTarget_nonneg {K : Nat}
409     {state : DelayedSAPOStructuralRoundState K}
410     (step : DelayedSAPONoSwitchProcessOne state)
411     (horizon : Nat) (i : Fin K) :
412     0 <= (ofProcessOne step horizon i).initialPhaseTarget := by
413   change 0 <= 1280 /
414     ((ofProcessOne step horizon i).initialProbability *
415       (ofProcessOne step horizon i).surrogateGap ^ 2)
416   apply div_nonneg (by norm_num)
417   exact mul_nonneg (initialProbability_pos step horizon i).le (sq_nonneg _)

419 /-- Under the explicit positive-width condition used by the source analysis,
420 the first EAP phase target has a genuinely positive denominator and value. -/
421 theorem initialPhaseTarget_pos {K : Nat}
422     {state : DelayedSAPOStructuralRoundState K}
423     (step : DelayedSAPONoSwitchProcessOne state)
424     (horizon : Nat) (i : Fin K) (hhorizon : 1 < horizon) :
425     0 < (ofProcessOne step horizon i).initialPhaseTarget := by
426   have hp := initialProbability_pos step horizon i
427   have hgap := surrogateGap_pos step horizon i hhorizon
428   change 0 < 1280 /
429     ((ofProcessOne step horizon i).initialProbability *
430       (ofProcessOne step horizon i).surrogateGap ^ 2)
431   positivity
