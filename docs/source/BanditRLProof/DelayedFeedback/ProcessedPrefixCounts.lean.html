9 /-- A ledger for one ordered ledger of source rounds whose feedback has been
10 processed by delayed SAPO.  Each entry stores the arm chosen at that source
11 round and the Algorithm-5 allocation that was in force when the action was
12 sampled.  In particular, `inactiveProbabilityAtSource` is source-time data;
13 it must not be reconstructed from the later processing-time state. -/
14 structure DelayedSAPOProcessedPrefix (K : Nat) where
15   length : Nat
16   activeAtSource : Fin length -> Finset (Fin K)
17   inactiveProbabilityAtSource : Fin length -> Fin K -> Real
18   chosenArmAtSource : Fin length -> Fin K
19   previousEmpiricalUpper : Fin K -> Real

23 /-- The source count `n_i(S)` on the processed ledger. -/
24 def processedPullCount {K : Nat}
25     (ledger : DelayedSAPOProcessedPrefix K) (i : Fin K) : Nat :=
26   (Finset.univ.filter fun s => ledger.chosenArmAtSource s = i).card
27 
28 /-- The source conditional mass `sum_{s in S} p_i(s)` on the same ledger. -/
29 noncomputable def expectedPullMass {K : Nat}
30     (ledger : DelayedSAPOProcessedPrefix K) (i : Fin K) : Real :=
31   ∑ s, delayedSAPOProbability (ledger.activeAtSource s)
32     (ledger.inactiveProbabilityAtSource s) i
33 
34 /-- Two arms that were active at every source round represented by the
35 processed ledger receive the same cumulative Algorithm-5 probability mass.
36 This is the deterministic line-15 producer used by the D.1 count clause. -/
37 theorem expectedPullMass_eq_of_active_throughout
38     {K : Nat} (ledger : DelayedSAPOProcessedPrefix K) (i j : Fin K)
39     (hi : forall s, i ∈ ledger.activeAtSource s)
40     (hj : forall s, j ∈ ledger.activeAtSource s) :
41     ledger.expectedPullMass i = ledger.expectedPullMass j := by
42   classical
43   apply Finset.sum_congr rfl
44   intro s _hs
45   rw [delayedSAPOProbability_of_active
46       (ledger.activeAtSource s) (ledger.inactiveProbabilityAtSource s) (hi s)]
47   rw [delayedSAPOProbability_of_active
48       (ledger.activeAtSource s) (ledger.inactiveProbabilityAtSource s) (hj s)]
49 
50 end DelayedSAPOProcessedPrefix

52 /-- The exact deterministic projection needed from Algorithm 5 and the
53 pull-count clause of source Definition D.1 at one processed ledger.
54 
55 The certificate keeps the source-time allocation ledger explicit.  It assumes
56 the D.1 count inequalities and the source definitions of the width and the
57 recursive empirical UCB, but it does not assume either of the width-comparison
58 conclusions that it is designed to prove.  Constructing this projection from
59 the full recursive delayed-SAPO state and proving its probability are separate
60 obligations. -/
61 structure DelayedSAPOProcessedPrefixCountCertificate
62     {K : Nat} [Nonempty (Fin K)]
63     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
64     (ledger : DelayedSAPOProcessedPrefix K) (horizon : Nat) : Prop where
65   active_persistence : forall s,
66     snapshot.active ⊆ ledger.activeAtSource s
67   count_lower : forall i,
68     (1 / 2 : Real) * ledger.expectedPullMass i -
69         3 * Real.log (horizon : Real) <=
70       (ledger.processedPullCount i : Real)
71   count_upper : forall i,
72     (ledger.processedPullCount i : Real) <=
73       2 * ledger.expectedPullMass i + 12 * Real.log (horizon : Real)
74   empiricalWidth_eq_source : forall i,
75     snapshot.empiricalWidth i =
76       sourceEmpiricalWidthScale (2 * Real.log (horizon : Real))
77         (ledger.processedPullCount i : Real)
78   empiricalUpper_eq_source : forall i,
79     snapshot.empiricalUpper i =
80       min (snapshot.empiricalMean i + snapshot.empiricalWidth i)
81         (ledger.previousEmpiricalUpper i)

85 /-- The capped source empirical width is always nonnegative. -/
86 theorem sourceEmpiricalWidthScale_nonneg (scale count : Real) :
87     0 <= sourceEmpiricalWidthScale scale count := by
88   by_cases hcount : count <= 0
89   · simp [sourceEmpiricalWidthScale, hcount]
90   · rw [sourceEmpiricalWidthScale, if_neg hcount]
91     exact le_min (by norm_num) (Real.sqrt_nonneg _)

93 /-- The capped source empirical width is at most one. -/
94 theorem sourceEmpiricalWidthScale_le_one (scale count : Real) :
95     sourceEmpiricalWidthScale scale count <= 1 := by
96   by_cases hcount : count <= 0
97   · simp [sourceEmpiricalWidthScale, hcount]
98   · rw [sourceEmpiricalWidthScale, if_neg hcount]
99     exact min_le_left _ _

101 /-- If one positive count is at most eight times another, then the other arm's
102 inverse-square-root width is at most three times the reference width.  The
103 factor three is the integer relaxation of `sqrt 8` used in source D.10. -/
104 theorem sourceEmpiricalWidthScale_le_three_of_count_le_eight_mul
105     (scale countReference countOther : Real)
106     (hscale : 0 <= scale) (hreference : 0 < countReference)
107     (hcount : countReference <= 8 * countOther) :
108     sourceEmpiricalWidthScale scale countOther <=
109       3 * sourceEmpiricalWidthScale scale countReference := by
110   have hother : 0 < countOther := by nlinarith
111   rw [sourceEmpiricalWidthScale, if_neg (not_le.mpr hother)]
112   rw [sourceEmpiricalWidthScale, if_neg (not_le.mpr hreference)]
113   have hratio :
114       scale / countOther <= 9 * (scale / countReference) := by
115     rw [show 9 * (scale / countReference) =
116       (9 * scale) / countReference by ring]
117     rw [div_le_div_iff₀ hother hreference]
118     have hscaled :
119         scale * countReference <= scale * (8 * countOther) :=
120       mul_le_mul_of_nonneg_left hcount hscale
121     have hscaleOther : 0 <= scale * countOther :=
122       mul_nonneg hscale hother.le
123     nlinarith
124   have hreferenceRatio : 0 <= scale / countReference :=
125     div_nonneg hscale hreference.le
126   have hotherRatio : 0 <= scale / countOther :=
127     div_nonneg hscale hother.le
128   have hsqrt :
129       Real.sqrt (scale / countOther) <=
130         3 * Real.sqrt (scale / countReference) := by
131     have hotherSquare := Real.sq_sqrt hotherRatio
132     have hreferenceSquare := Real.sq_sqrt hreferenceRatio
133     have hotherSqrtNonnegative := Real.sqrt_nonneg (scale / countOther)
134     have hreferenceSqrtNonnegative :=
135       Real.sqrt_nonneg (scale / countReference)
136     nlinarith
137   by_cases hcapped : 1 <= Real.sqrt (scale / countReference)
138   · rw [min_eq_left hcapped]
139     calc
140       min 1 (Real.sqrt (scale / countOther)) <= 1 := min_le_left _ _
141       _ <= 3 * 1 := by norm_num
142   · have huncapped : Real.sqrt (scale / countReference) <= 1 :=
143       (le_of_lt (lt_of_not_ge hcapped))
144     rw [min_eq_right huncapped]
145     exact (min_le_right 1 _).trans hsqrt

263 /-- The recursive empirical UCB is no larger than the current empirical
264 mean-plus-width surface, so the source minimum `ucbStar` has the current-UCB
265 upper edge needed by the active-arm gap proof. -/
266 theorem ucbStar_le_empiricalMean_add_width
267     {K : Nat} [Nonempty (Fin K)]
268     {snapshot : DelayedSAPOSourceConfidenceSnapshot K}
269     {ledger : DelayedSAPOProcessedPrefix K} {horizon : Nat}
270     (certificate : DelayedSAPOProcessedPrefixCountCertificate
271       snapshot ledger horizon)
272     (mean : Fin K -> Real)
273     (hgood : snapshot.EliminationGoodEvent mean) (i : Fin K) :
274     snapshot.ucbStar <=
275       snapshot.empiricalMean i + snapshot.empiricalWidth i := by
276   rw [hgood.ucbStar_eq_source, DelayedSAPOSourceConfidenceSnapshot.sourceUcbStar]
277   calc
278     Finset.univ.inf' Finset.univ_nonempty (fun j =>
279         min (snapshot.empiricalUpper j) (snapshot.importanceUpper j)) <=
280         min (snapshot.empiricalUpper i) (snapshot.importanceUpper i) :=
281       Finset.inf'_le _ (Finset.mem_univ i)
282     _ <= snapshot.empiricalUpper i := min_le_left _ _
283     _ <= snapshot.empiricalMean i + snapshot.empiricalWidth i := by
284       rw [certificate.empiricalUpper_eq_source i]
285       exact min_le_left _ _

287 /-- The exact large/small branch expected by the existing active-arm gap
288 consumer is now produced from the processed-ledger certificate. -/
289 theorem activeArmGapBranch
290     {K : Nat} [Nonempty (Fin K)]
291     {snapshot : DelayedSAPOSourceConfidenceSnapshot K}
292     {ledger : DelayedSAPOProcessedPrefix K} {horizon : Nat}
293     (certificate : DelayedSAPOProcessedPrefixCountCertificate
294       snapshot ledger horizon)
295     (hhorizon : 1 < horizon) (mean : Fin K -> Real)
296     (hgood : snapshot.EliminationGoodEvent mean)
297     {optimal i : Fin K} (hoptimalActive : optimal ∈ snapshot.active)
298     (hiActive : i ∈ snapshot.active) :
299     (snapshot.ucbStar <=
300           snapshot.empiricalMean optimal + snapshot.empiricalWidth optimal /\
301         snapshot.empiricalWidth optimal <=
302           3 * snapshot.empiricalWidth i) \/
303       (exists scale count : Real,
304         0 < scale /\ count <= 96 * scale /\
305         snapshot.empiricalWidth i =
306           sourceEmpiricalWidthScale scale count) := by
307   have hhorizonReal : (1 : Real) < (horizon : Real) := by exact_mod_cast hhorizon
308   have hlog : 0 < Real.log (horizon : Real) := Real.log_pos hhorizonReal
309   by_cases hsmall :
310       (ledger.processedPullCount i : Real) <=
311         192 * Real.log (horizon : Real)
312   · right
313     refine ⟨2 * Real.log (horizon : Real),
314       (ledger.processedPullCount i : Real), ?_, ?_, ?_⟩
315     · linarith
316     · nlinarith
317     · exact certificate.empiricalWidth_eq_source i
318   · left
319     constructor
320     · exact certificate.ucbStar_le_empiricalMean_add_width mean hgood optimal
321     · exact certificate.empiricalWidth_le_three_of_large_count
322         hhorizon hiActive hoptimalActive (lt_of_not_ge hsmall)

324 /-- Repaired same-snapshot D.12 / main-text Lemma-4.2 deterministic slice.
325 The theorem removes the manually supplied branch and pair-width premises
326 from the earlier consumer: both are generated by the actual Algorithm-5
327 source-time allocation ledger and the D.1 count clause.  The full recursive

329 theorem gap_le_twenty_mul_gap_at_earlier_elimination_snapshot_of_countCertificate

331     (snapshot : DelayedSAPOSourceConfidenceSnapshot K)
332     (ledger : DelayedSAPOProcessedPrefix K) (horizon : Nat)
