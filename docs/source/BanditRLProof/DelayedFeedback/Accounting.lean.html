    1 import Mathlib.Data.Finset.Card
2 import Mathlib.Data.Finset.Lattice.Fold
3 
4 namespace BanditRLProof

6 namespace DelayedFeedback

8 /-- Source rounds whose delayed feedback is available before the action at
9 round `t`.  The strict inequality matches the NeurIPS 2025 delayed-SAPO source:
10 feedback generated at `s` arrives at the end of `s + delay s`, hence it can be
11 used for the next action only when `s + delay s < t`. -/
12 def observedBefore (delay : Nat → Nat) (t : Nat) : Finset Nat :=
13   (Finset.range t).filter (fun s => s + delay s < t)
14 
15 /-- Source rounds before `t` whose feedback is not yet available when the
16 action at `t` is chosen.  Future source rounds are excluded by `Finset.range
17 t`. -/
18 def outstandingAt (delay : Nat → Nat) (t : Nat) : Finset Nat :=
19   (Finset.range t).filter (fun s => ¬ s + delay s < t)
20 
21 /-- Available and outstanding feedback form disjoint parts of the past. -/
22 theorem observedBefore_disjoint_outstandingAt (delay : Nat → Nat) (t : Nat) :
23     Disjoint (observedBefore delay t) (outstandingAt delay t) := by
24   rw [Finset.disjoint_left]
25   intro s hsObserved hsOutstanding
26   exact (Finset.mem_filter.mp hsOutstanding).2
27     (Finset.mem_filter.mp hsObserved).2
28 
29 /-- Every source round before `t` is either available or outstanding. -/
30 theorem observedBefore_union_outstandingAt (delay : Nat → Nat) (t : Nat) :
31     observedBefore delay t ∪ outstandingAt delay t = Finset.range t := by
32   ext s
33   simp only [Finset.mem_union, observedBefore, outstandingAt,
34     Finset.mem_filter, Finset.mem_range]
35   constructor
36   · rintro (hs | hs)
37     · exact hs.1
38     · exact hs.1
39   · intro hs
40     by_cases havailable : s + delay s < t
41     · exact Or.inl ⟨hs, havailable⟩
42     · exact Or.inr ⟨hs, havailable⟩
43 
44 /-- The available and outstanding counts add up to the number of past source
45 rounds. -/
46 theorem card_observedBefore_add_card_outstandingAt
47     (delay : Nat → Nat) (t : Nat) :
48     (observedBefore delay t).card + (outstandingAt delay t).card = t := by
49   rw [← Finset.card_union_of_disjoint
50     (observedBefore_disjoint_outstandingAt delay t)]
51   rw [observedBefore_union_outstandingAt]
52   exact Finset.card_range t
53 
54 /-- Number of source rounds before `t` whose feedback cannot yet be used by
55 the action at `t`.  This action-time surface is kept separate from the paper's
56 end-of-round `sigma(t)` until their one-based/zero-based index bridge is
57 proved. -/
58 def outstandingCount (delay : Nat → Nat) (t : Nat) : Nat :=
59   (outstandingAt delay t).card
60 
61 /-- Largest action-time outstanding count through the inclusive horizon. -/
62 def maxOutstandingBeforeThrough (delay : Nat → Nat) (horizon : Nat) : Nat :=
63   (Finset.range (horizon + 1)).sup (outstandingCount delay)
64 
65 /-- An action-time outstanding count never exceeds the number of past source
66 rounds. -/
67 theorem outstandingCount_le_round (delay : Nat → Nat) (t : Nat) :
68     outstandingCount delay t ≤ t := by
69   rw [outstandingCount]
70   exact (Finset.card_le_card (Finset.filter_subset _ _)).trans_eq
71     (Finset.card_range t)
72 
73 /-- Every action-time outstanding count inside the horizon is bounded by the
74 finite maximum surface. -/
75 theorem outstandingCount_le_maxOutstandingBeforeThrough
76     (delay : Nat → Nat) {t horizon : Nat} (ht : t ≤ horizon) :
77     outstandingCount delay t ≤ maxOutstandingBeforeThrough delay horizon := by
78   exact Finset.le_sup
79     (s := Finset.range (horizon + 1))
80     (f := outstandingCount delay)
81     (Finset.mem_range.mpr (Nat.lt_succ_of_le ht))
82 
83 /-- Convert a one-based paper delay sequence into the zero-based source
84 carrier used by `outstandingAt`: zero-based source `s` represents paper source
85 round `s + 1`. -/
86 def oneBasedDelayShift (delay : Nat → Nat) (s : Nat) : Nat :=
87   delay (s + 1)
88 
89 /-- The paper's end-of-round missing-feedback set, represented on a zero-based
90 finite carrier.  An element `s` denotes paper round `s + 1`, and the predicate
91 is exactly `(s + 1) + d_(s+1) > t` for source rounds at most `t`. -/
92 def paperMissingAtEnd (delay : Nat → Nat) (t : Nat) : Finset Nat :=
93   (Finset.range t).filter (fun s => t < (s + 1) + delay (s + 1))
94 
95 /-- The source paper's one-based end-of-round missing set is exactly the
96 action-time outstanding set after reindexing source rounds and delays.  This
97 lemma is the explicit off-by-one bridge; the two surfaces are not identified
98 by notation alone. -/
99 theorem paperMissingAtEnd_eq_outstandingAt_oneBasedDelayShift
100     (delay : Nat → Nat) (t : Nat) :
101     paperMissingAtEnd delay t = outstandingAt (oneBasedDelayShift delay) t := by
102   ext s
103   simp only [paperMissingAtEnd, outstandingAt, oneBasedDelayShift,
104     Finset.mem_filter, Finset.mem_range]
105   constructor
106   · rintro ⟨hs, hmissing⟩
107     exact ⟨hs, by omega⟩
108   · rintro ⟨hs, houtstanding⟩
109     exact ⟨hs, by omega⟩
110 
111 /-- Paper-facing count corresponding to `sigma(t)`, with paper rounds
112 `1, ..., t` represented by zero-based source indices. -/
113 def paperMissingCount (delay : Nat → Nat) (t : Nat) : Nat :=
114   (paperMissingAtEnd delay t).card
115 
116 /-- Cardinal form of the one-based/end-of-round indexing bridge. -/
117 theorem paperMissingCount_eq_outstandingCount_oneBasedDelayShift
118     (delay : Nat → Nat) (t : Nat) :
119     paperMissingCount delay t = outstandingCount (oneBasedDelayShift delay) t := by
120   rw [paperMissingCount, outstandingCount,
121     paperMissingAtEnd_eq_outstandingAt_oneBasedDelayShift]
122 
123 /-- The paper-facing missing count at round `t` is at most `t`. -/
124 theorem paperMissingCount_le_round (delay : Nat → Nat) (t : Nat) :
125     paperMissingCount delay t ≤ t := by
126   rw [paperMissingCount_eq_outstandingCount_oneBasedDelayShift]
127   exact outstandingCount_le_round (oneBasedDelayShift delay) t
128 
129 /-- Finite maximum of the paper-facing missing-count surface through an
130 inclusive horizon. -/
131 def paperSigmaMaxThrough (delay : Nat → Nat) (horizon : Nat) : Nat :=
132   (Finset.range (horizon + 1)).sup (paperMissingCount delay)
133 
134 /-- Each paper-facing missing count is bounded by its finite maximum through
135 the declared horizon. -/
136 theorem paperMissingCount_le_paperSigmaMaxThrough
137     (delay : Nat → Nat) {t horizon : Nat} (ht : t ≤ horizon) :
138     paperMissingCount delay t ≤ paperSigmaMaxThrough delay horizon := by
139   exact Finset.le_sup
140     (s := Finset.range (horizon + 1))
141     (f := paperMissingCount delay)
142     (Finset.mem_range.mpr (Nat.lt_succ_of_le ht))
