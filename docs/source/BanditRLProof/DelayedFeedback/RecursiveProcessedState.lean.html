4 # Processed trace-summary adapter for Delayed SAPO
5 
6 This module makes one deterministic interface preceding the probabilistic count
7 clause in source Definition D.1 / Lemma D.4 explicit.  A processed trace summary
8 stores the source round of every processed item and reads the allocation that
9 was present at that source round.  It never reconstructs a source-time
10 probability from the later processing-time state.  Source indices are unique
11 and carry the strict availability witness `s + d_s < t` used by Algorithm 5.
12 
13 The summary is not constructed from the recursive Algorithm-5 transition
14 system.  Establishing such a generated-trajectory-to-summary producer remains
15 open, so this module must not be reported as a full state-to-ledger projection.
16 
17 The only stochastic input is `D4CountClause`, which records the two displayed
18 count inequalities.  Their probability on a generated Delayed-SAPO trajectory
19 is deliberately not proved here.  In particular, this module does not claim
20 source Lemmas D.4, D.10, D.12, main-text Lemma 4.2, or either regret endpoint.
21 -/

27 /-- Source-shaped deterministic summary of a Delayed-SAPO processed trace.
28 
29 `sourceIndex` is an ordered processing ledger, not a chronological source-round
30 prefix: feedback from a later source round may be processed before feedback
31 from an earlier source round.  Its entries are distinct and satisfy the exact
32 strict availability test at `currentActionRound`.  `activeAtSourceRound`
33 records the line-15 sampling set at past source rounds and is antitone because
34 Algorithm 5 only removes arms.  The possibly intra-round `currentActive` set is
35 stored separately.  Its containment in every ledger source set is an explicit
36 trace-summary invariant that a future Algorithm-5 producer must prove.  The
37 current empirical surfaces are likewise separate from the source-time
38 allocations used by the ledger.
39 
40 This record is an interface summary, not a proof that Algorithm 5 generates
41 the supplied fields. -/
42 structure DelayedSAPOProcessedTraceSummary (K : Nat) where
43   length : Nat
44   sourceIndex : Fin length -> Nat
45   sourceIndex_injective : Function.Injective sourceIndex
46   currentActionRound : Nat
47   delayAt : Nat -> Nat
48   source_available : forall q,
49     sourceIndex q + delayAt (sourceIndex q) < currentActionRound
50   activeAtSourceRound : Nat -> Finset (Fin K)
51   sourceActive_antitone : Antitone activeAtSourceRound
52   currentActive : Finset (Fin K)
53   currentActive_subset_sourceActive : forall q,
54     currentActive <= activeAtSourceRound (sourceIndex q)
55   chosenArmAt : Nat -> Fin K
56   inactiveProbabilityAt : Nat -> Fin K -> Real
57   empiricalMean : Fin K -> Real
58   importanceUpper : Fin K -> Real
59   previousEmpiricalUpper : Fin K -> Real
60   ucbStar : Real

64 /-- Read every processed entry at its recorded source round. -/
65 def toProcessedPrefix {K : Nat}
66     (state : DelayedSAPOProcessedTraceSummary K) :
67     DelayedSAPOProcessedPrefix K where
68   length := state.length
69   activeAtSource q := state.activeAtSourceRound (state.sourceIndex q)
70   inactiveProbabilityAtSource q :=
71     state.inactiveProbabilityAt (state.sourceIndex q)
72   chosenArmAtSource q := state.chosenArmAt (state.sourceIndex q)
73   previousEmpiricalUpper := state.previousEmpiricalUpper
74 
75 /-- The exact source empirical width at the current processed state. -/
76 noncomputable def empiricalWidthAt {K : Nat}
77     (state : DelayedSAPOProcessedTraceSummary K)
78     (horizon : Nat) (i : Fin K) : Real :=
79   sourceEmpiricalWidthScale (2 * Real.log (horizon : Real))
80     (state.toProcessedPrefix.processedPullCount i : Real)
81 
82 /-- The recursive empirical UCB printed by the source, evaluated from the
83 current empirical mean, the current processed count, and the preceding UCB. -/
84 noncomputable def empiricalUpperAt {K : Nat}
85     (state : DelayedSAPOProcessedTraceSummary K)
86     (horizon : Nat) (i : Fin K) : Real :=
87   min (state.empiricalMean i + state.empiricalWidthAt horizon i)
88     (state.previousEmpiricalUpper i)
89 
90 /-- Confidence snapshot projected from the supplied trace summary.  The width
91 and recursive empirical UCB are definitions here, rather than certificate
92 premises. -/
93 noncomputable def toConfidenceSnapshot {K : Nat}
94     (state : DelayedSAPOProcessedTraceSummary K) (horizon : Nat) :
95     DelayedSAPOSourceConfidenceSnapshot K where
96   active := state.currentActive
97   empiricalMean := state.empiricalMean
98   empiricalWidth := state.empiricalWidthAt horizon
99   ucbStar := state.ucbStar
100   empiricalUpper := state.empiricalUpperAt horizon
101   importanceUpper := state.importanceUpper

103 /-- The two armwise count inequalities printed in source Definition D.1 and
104 used by source Lemma D.4.  This is the stochastic boundary of the present
105 module: proving that it holds simultaneously with probability at least
106 `1 - 2 / T` on the generated trajectory remains open. -/
107 structure D4CountClause {K : Nat}
108     (state : DelayedSAPOProcessedTraceSummary K)
109     (horizon : Nat) : Prop where
110   count_lower : forall i,
111     (1 / 2 : Real) * state.toProcessedPrefix.expectedPullMass i -
112         3 * Real.log (horizon : Real) <=
113       (state.toProcessedPrefix.processedPullCount i : Real)
114   count_upper : forall i,
115     (state.toProcessedPrefix.processedPullCount i : Real) <=
116       2 * state.toProcessedPrefix.expectedPullMass i +
117         12 * Real.log (horizon : Real)
118 
119 /-- Expose the trace-summary invariant that every arm in the intra-round current
120 set was active at every source round in the processed ledger.  Producing this
121 invariant from Algorithm 5 is deliberately outside this adapter. -/
122 theorem currentActive_subset_activeAt_sourceIndex {K : Nat}
123     (state : DelayedSAPOProcessedTraceSummary K) (q : Fin state.length) :
124     state.currentActive <=
125       state.activeAtSourceRound (state.sourceIndex q) := by
126   exact state.currentActive_subset_sourceActive q

128 /-- Deterministic trace-summary adapter for the processed-prefix count
129 certificate.  Source-time allocation data come from `toProcessedPrefix`,
130 active persistence is an explicit trace-summary invariant, and the two count
131 bounds are the explicit D.4 boundary.  No width comparison or gap conclusion
132 is assumed. -/
133 theorem toProcessedPrefixCountCertificate
134     {K : Nat} [Nonempty (Fin K)]
135     (state : DelayedSAPOProcessedTraceSummary K)
136     (horizon : Nat) (hD4 : state.D4CountClause horizon) :
137     DelayedSAPOProcessedPrefixCountCertificate
138       (state.toConfidenceSnapshot horizon)
139       state.toProcessedPrefix horizon where
140   active_persistence := by
141     intro q
142     simpa [toConfidenceSnapshot, toProcessedPrefix] using
143       state.currentActive_subset_activeAt_sourceIndex q
144   count_lower := hD4.count_lower
145   count_upper := hD4.count_upper
146   empiricalWidth_eq_source := by
147     intro i
148     rfl
149   empiricalUpper_eq_source := by
150     intro i
151     rfl

153 /-- Downstream same-snapshot factor-twenty consumer reached from a processed
154 trace summary plus the explicit D.4 count clause.  This is still conditional
155 on the elimination projection of the source good event; neither its
156 probability nor an ordered multi-snapshot elimination theorem is claimed. -/
157 theorem gap_le_twenty_mul_gap_at_earlier_elimination_snapshot_of_traceSummary
158     {K : Nat} [Nonempty (Fin K)]
159     (state : DelayedSAPOProcessedTraceSummary K) (horizon : Nat)
160     (hD4 : state.D4CountClause horizon) (hhorizon : 1 < horizon)
161     (mean : Fin K -> Real) (optimal iEarlier iLater : Fin K)
162     (hoptimal : forall j, mean optimal <= mean j)
163     (hmeanBounds : forall j, mean j ∈ Set.Icc (0 : Real) 1)
164     (hgood : (state.toConfidenceSnapshot horizon).EliminationGoodEvent mean)
165     (hoptimalActive :
166       optimal ∈ (state.toConfidenceSnapshot horizon).active)
167     (hEarlierEliminated :
168       iEarlier ∈ (state.toConfidenceSnapshot horizon).eliminated)
169     (hLaterRemaining :
170       iLater ∈ (state.toConfidenceSnapshot horizon).remainingActive) :
171     mean iLater - mean optimal <=
172       20 * (mean iEarlier - mean optimal) := by
173   exact
174     (state.toProcessedPrefixCountCertificate horizon hD4).gap_le_twenty_mul_gap_at_earlier_elimination_snapshot_of_countCertificate
175       (state.toConfidenceSnapshot horizon) state.toProcessedPrefix horizon
176         hhorizon mean optimal iEarlier iLater hoptimal hmeanBounds hgood
177           hoptimalActive hEarlierEliminated hLaterRemaining
