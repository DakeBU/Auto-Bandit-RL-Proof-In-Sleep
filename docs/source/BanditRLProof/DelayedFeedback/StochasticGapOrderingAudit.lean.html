7 /-- The scalar form of the empirical width printed in the delayed-SAPO
8 source, with `scale = 2 * log T` and `count = n_i(S)`.  A nonpositive count
9 uses the capped width `1`; this prevents Lean's totalized real division at
10 zero from manufacturing a zero-width observation.  Keeping the scale explicit
11 isolates the positive-count order issue from logarithmic side conditions. -/
12 noncomputable def sourceEmpiricalWidthScale (scale count : Real) : Real :=
13   if count <= 0 then 1 else min 1 (Real.sqrt (scale / count))
14 
15 /-- For nonnegative scale and positive counts, the source empirical width is
16 antitone in the count.  Thus a later state with at least as many pulls has no
17 larger width than an earlier prefix. -/
18 theorem sourceEmpiricalWidthScale_antitone
19     (scale countEarlier countLater : Real)
20     (hscale : 0 <= scale) (hcountEarlier : 0 < countEarlier)
21     (hcount : countEarlier <= countLater) :
22     sourceEmpiricalWidthScale scale countLater <=
23       sourceEmpiricalWidthScale scale countEarlier := by
24   have hcountLater : 0 < countLater := hcountEarlier.trans_le hcount
25   rw [sourceEmpiricalWidthScale, if_neg (not_le.mpr hcountLater)]
26   rw [sourceEmpiricalWidthScale, if_neg (not_le.mpr hcountEarlier)]
27   apply min_le_min_left
28   apply Real.sqrt_le_sqrt
29   exact div_le_div_of_nonneg_left hscale hcountEarlier hcount

31 /-- The small-count branch used in the source proof of Lemma D.10.  If the
32 count is at most `96 * scale`, then the capped inverse-square-root width is at
33 least one tenth.  For the printed choice `scale = 2 * log T`, this is exactly
34 the implication from `count <= 192 * log T` to `1 <= 10 * width`. -/
35 theorem one_le_ten_mul_sourceEmpiricalWidthScale_of_count_le_96_mul_scale
36     (scale count : Real) (hscale : 0 < scale)
37     (hcount : count <= 96 * scale) :
38     1 <= 10 * sourceEmpiricalWidthScale scale count := by
39   by_cases hcountNonpositive : count <= 0
40   · rw [sourceEmpiricalWidthScale, if_pos hcountNonpositive]
41     norm_num
42   · have hcountPositive : 0 < count := lt_of_not_ge hcountNonpositive
43     rw [sourceEmpiricalWidthScale, if_neg hcountNonpositive]
44     have hratioNonnegative : 0 <= scale / count :=
45       div_nonneg hscale.le hcountPositive.le
46     have hsquare : ((1 / 10 : Real) ^ 2) <= scale / count := by
47       rw [le_div_iff₀ hcountPositive]
48       norm_num at hcount ⊢
49       linarith
50     have hsqrt : (1 / 10 : Real) <= Real.sqrt (scale / count) :=
51       (Real.le_sqrt (by norm_num) hratioNonnegative).2 hsquare
52     have hwidth : (1 / 10 : Real) <=
53         min 1 (Real.sqrt (scale / count)) :=
54       (le_min_iff).2 ⟨by norm_num, hsqrt⟩
55     nlinarith

57 /-- Source-parameter specialization of the preceding small-count lemma. -/
58 theorem one_le_ten_mul_sourceEmpiricalWidthScale_two_log_of_small_count
59     (horizon count : Real) (hhorizon : 1 < horizon)
60     (hcount : count <= 192 * Real.log horizon) :
61     1 <= 10 * sourceEmpiricalWidthScale (2 * Real.log horizon) count := by

63   · have hlog : 0 < Real.log horizon := Real.log_pos hhorizon
64     linarith
65   · nlinarith

90 /-- A literal source-width instance at integer horizon `T = 4`.  Moving from
91 one to four processed pulls strictly decreases the printed capped radius, so
92 the reverse transport fails inside the paper's own parameter domain rather
93 than only for the normalized scale-one diagnostic above. -/
94 theorem not_sourceEmpiricalWidthScale_horizon_four_one_le_four :
95     ¬ (sourceEmpiricalWidthScale (2 * Real.log 4) 1 <=
96       sourceEmpiricalWidthScale (2 * Real.log 4) 4) := by
97   have hlogFour : Real.log 4 = 2 * Real.log 2 := by
98     rw [show (4 : Real) = 2 ^ 2 by norm_num, Real.log_pow]
99     norm_num
100   have hhalfLogTwo : (1 / 2 : Real) <= Real.log 2 := by
101     have h := Real.one_sub_inv_le_log_of_pos (by norm_num : (0 : Real) < 2)
102     norm_num at h ⊢
103     exact h
104   have hscale : (1 : Real) <= 2 * Real.log 4 := by
105     rw [hlogFour]
106     linarith
107   have hone : sourceEmpiricalWidthScale (2 * Real.log 4) 1 = 1 := by
108     rw [sourceEmpiricalWidthScale, if_neg (by norm_num : ¬ ((1 : Real) <= 0))]
109     exact min_eq_left (Real.one_le_sqrt.mpr (by simpa using hscale))
110   have hlogTwoNonnegative : 0 <= Real.log 2 := Real.log_nonneg (by norm_num)
111   have hlogTwoLtOne : Real.log 2 < 1 := by
112     have h := Real.log_lt_sub_one_of_pos
113       (by norm_num : (0 : Real) < 2) (by norm_num : (2 : Real) ≠ 1)
114     norm_num at h ⊢
115     exact h
116   have hratio : (2 * Real.log 4) / 4 = Real.log 2 := by
117     rw [hlogFour]
118     ring
119   have hsqrt : Real.sqrt ((2 * Real.log 4) / 4) < 1 := by
120     rw [hratio, <- Real.sqrt_one]
121     exact Real.sqrt_lt_sqrt hlogTwoNonnegative hlogTwoLtOne
122   have hfour : sourceEmpiricalWidthScale (2 * Real.log 4) 4 < 1 := by
123     rw [sourceEmpiricalWidthScale, if_neg (by norm_num : ¬ ((4 : Real) <= 0))]
124     exact (min_le_right 1 _).trans_lt hsqrt
125   rw [hone]
126   exact not_le.mpr hfour

189 consumer while leaving the recursive count and width-shape producers explicit.

227 comparison premises still have to be produced by the source count event on a

290 /-- Exact deterministic interface needed by the displayed proof of source
291 Lemma D.12 (main-text Lemma 4.2).  Its index is a shared processed-sequence
292 prefix length, not wall-clock action time.  The fields deliberately name the
293 four edges consumed by D.12: D.10 supplies the gap endpoints and cross-arm
294 comparison, while the width definition supplies the time transport. -/
295 structure DelayedSAPOD10D12GapOrderingContract (K : Nat) where
296   eliminationPrefixIndex : Fin K -> Nat
297   widthAt : Fin K -> Nat -> Real
298   gap : Fin K -> Real
299   gap_upper_at_elimination : forall i,
300     gap i <= 16 * widthAt i (eliminationPrefixIndex i)
301   surrogateGap_lower_at_elimination : forall i,
302     8 * widthAt i (eliminationPrefixIndex i) <= gap i
303   width_antitone : forall i {earlier later}, earlier <= later ->
304     widthAt i later <= widthAt i earlier
305   width_comparable_before_both : forall i j t,
306     t <= eliminationPrefixIndex i -> t <= eliminationPrefixIndex j ->
307       widthAt i t <= 10 * widthAt j t

311 /-- The source surrogate gap `Delta-tilde_i = 8 width_i(S-tilde_i)`. -/
312 noncomputable def surrogateGap {K : Nat}

314     (i : Fin K) : Real :=

339     (iEarlier iLater : Fin K)
