7 universe uAction uLoss uDecision
8 
9 /-- The information exposed to an action rule immediately before a delayed
10 bandit action.  Past actions are visible, while a loss is exposed only when
11 its source round belongs to the source-faithful strict-availability set.
12 
13 The view deliberately has no delay field and no total loss trace field.  A
14 future Delayed SAPO implementation must consume this view (or a proved
15 equivalent), rather than the environment's hidden delay/loss functions. -/
16 structure ActionTimeView (Action : Type uAction) (Loss : Type uLoss) where
17   pastAction : Nat → Option Action
18   observedLoss : Nat → Option Loss
19 
20 @[ext]
21 theorem ActionTimeView.ext
22     {Action : Type uAction} {Loss : Type uLoss}
23     {left right : ActionTimeView Action Loss}
24     (hpast : left.pastAction = right.pastAction)
25     (hloss : left.observedLoss = right.observedLoss) :
26     left = right := by
27   cases left
28   cases right
29   cases hpast
30   cases hloss
31   rfl
32 
33 /-- Construct the pre-action view at round `t` from an environment trace.
34 Actions before `t` are known to the learner.  A source loss is known exactly
35 when `s + delay s < t`; future and outstanding losses return `none`. -/
36 def actionTimeViewAt
37     {Action : Type uAction} {Loss : Type uLoss}
38     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
39     (t : Nat) : ActionTimeView Action Loss where
40   pastAction := fun s => if s < t then some (action s) else none
41   observedLoss := fun s =>
42     if s ∈ observedBefore delay t then some (loss s) else none
43 
44 /-- A causal decision rule receives only the round number and its action-time
45 view.  `Decision` can later be instantiated by an action distribution, a
46 kernel, or a deterministic action. -/
47 abbrev CausalDecisionRule
48     (Action : Type uAction) (Loss : Type uLoss) (Decision : Type uDecision) :=
49   (t : Nat) → ActionTimeView Action Loss → Decision
50 
51 @[simp]
52 theorem actionTimeViewAt_pastAction_of_lt
53     {Action : Type uAction} {Loss : Type uLoss}
54     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
55     {s t : Nat} (hs : s < t) :
56     (actionTimeViewAt delay action loss t).pastAction s = some (action s) := by
57   simp [actionTimeViewAt, hs]
58 
59 @[simp]
60 theorem actionTimeViewAt_pastAction_of_not_lt
61     {Action : Type uAction} {Loss : Type uLoss}
62     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
63     {s t : Nat} (hs : ¬ s < t) :
64     (actionTimeViewAt delay action loss t).pastAction s = none := by
65   simp [actionTimeViewAt, hs]
66 
67 @[simp]
68 theorem actionTimeViewAt_observedLoss_of_mem
69     {Action : Type uAction} {Loss : Type uLoss}
70     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
71     {s t : Nat} (hs : s ∈ observedBefore delay t) :
72     (actionTimeViewAt delay action loss t).observedLoss s = some (loss s) := by
73   simp [actionTimeViewAt, hs]
74 
75 @[simp]
76 theorem actionTimeViewAt_observedLoss_of_not_mem
77     {Action : Type uAction} {Loss : Type uLoss}
78     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
79     {s t : Nat} (hs : s ∉ observedBefore delay t) :
80     (actionTimeViewAt delay action loss t).observedLoss s = none := by
81   simp [actionTimeViewAt, hs]
82 
83 /-- Outstanding feedback is absent from the action-time view. -/
84 theorem actionTimeViewAt_outstanding_loss_hidden
85     {Action : Type uAction} {Loss : Type uLoss}
86     (delay : Nat → Nat) (action : Nat → Action) (loss : Nat → Loss)
87     {s t : Nat} (hs : s ∈ outstandingAt delay t) :
88     (actionTimeViewAt delay action loss t).observedLoss s = none := by
89   apply actionTimeViewAt_observedLoss_of_not_mem
90   intro hobserved
91   exact (Finset.mem_filter.mp hs).2 (Finset.mem_filter.mp hobserved).2
92 
93 /-- Two environment traces yield exactly the same pre-action view whenever
94 their visible source sets, past actions, and revealed losses agree.  Hidden
95 delays and unobserved losses may differ. -/
96 theorem actionTimeViewAt_eq_of_observation_equivalent
97     {Action : Type uAction} {Loss : Type uLoss}
98     (delay₁ delay₂ : Nat → Nat)
99     (action₁ action₂ : Nat → Action) (loss₁ loss₂ : Nat → Loss)
100     (t : Nat)
101     (hvisible : observedBefore delay₁ t = observedBefore delay₂ t)
102     (haction : ∀ s, s < t → action₁ s = action₂ s)
103     (hloss : ∀ s, s ∈ observedBefore delay₁ t → loss₁ s = loss₂ s) :
104     actionTimeViewAt delay₁ action₁ loss₁ t =
105       actionTimeViewAt delay₂ action₂ loss₂ t := by
106   apply ActionTimeView.ext
107   · funext s
108     by_cases hs : s < t
109     · simp [actionTimeViewAt, hs, haction s hs]
110     · simp [actionTimeViewAt, hs]
111   · funext s
112     by_cases hs : s ∈ observedBefore delay₁ t
113     · have hs₂ : s ∈ observedBefore delay₂ t := by
114         simpa [← hvisible] using hs
115       simp [actionTimeViewAt, hs, hs₂, hloss s hs]
116     · have hs₂ : s ∉ observedBefore delay₂ t := by
117         simpa [← hvisible] using hs
118       simp [actionTimeViewAt, hs, hs₂]
119 
120 /-- A decision rule consuming only `ActionTimeView` cannot distinguish two
121 worlds that agree on all information visible before the action. -/
122 theorem causalDecision_eq_of_observation_equivalent
123     {Action : Type uAction} {Loss : Type uLoss}
124     {Decision : Type uDecision}
125     (rule : CausalDecisionRule Action Loss Decision)
126     (delay₁ delay₂ : Nat → Nat)
127     (action₁ action₂ : Nat → Action) (loss₁ loss₂ : Nat → Loss)
128     (t : Nat)
129     (hvisible : observedBefore delay₁ t = observedBefore delay₂ t)
130     (haction : ∀ s, s < t → action₁ s = action₂ s)
131     (hloss : ∀ s, s ∈ observedBefore delay₁ t → loss₁ s = loss₂ s) :
132     rule t (actionTimeViewAt delay₁ action₁ loss₁ t) =
133       rule t (actionTimeViewAt delay₂ action₂ loss₂ t) := by
134   rw [actionTimeViewAt_eq_of_observation_equivalent
135     delay₁ delay₂ action₁ action₂ loss₁ loss₂ t hvisible haction hloss]
