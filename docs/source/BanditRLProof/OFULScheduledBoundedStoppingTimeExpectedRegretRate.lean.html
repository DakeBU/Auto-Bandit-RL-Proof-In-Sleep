5 # Explicit bounded-stopping-time expected OFUL pseudo-regret rate
6 
7 This module tunes the outer confidence budget of the scheduled bounded
8 stopping-time expectation theorem to `1 / (T + 1)`. The stopped bad-event
9 charge becomes one initial-gap envelope, while the scheduled confidence
10 logarithm remains explicit.
11 -/

20 /--
21 The explicit scheduled confidence logarithm after choosing the outer
22 expected-regret budget `delta_T = 1 / (T + 1)`.
23 -/
24 noncomputable def telescopingStandardExpectedRegretLogBudget
25     {Feature : Type u} [Fintype Feature]
26     (lambda : Real) (horizon : Nat) (L2 : Real) : Real :=
27   standardScalarLogDetBudget
28       (Feature := Feature) lambda (horizon + 1) L2 +
29     2 * Real.log
30       ((((horizon + 1 : Nat) : Real) ^ 2) *
31         ((horizon + 2 : Nat) : Real))
32 
33 /--
34 Substituting `delta_T = 1 / (T + 1)` into the scheduled confidence logarithm
35 gives the explicit cubic-in-horizon logarithmic scale.
36 -/
37 theorem
38     telescopingHighProbabilityRegretLogBudget_standardExpectedRegretDelta
39     {Feature : Type u} [Fintype Feature]
40     (lambda : Real) (horizon : Nat) (L2 : Real) :
41     telescopingHighProbabilityRegretLogBudget
42         (Feature := Feature)
43         lambda (standardExpectedRegretDelta horizon) horizon L2 =
44       telescopingStandardExpectedRegretLogBudget
45         (Feature := Feature) lambda horizon L2 := by
46   have hscale :
47       ((((horizon + 1 : Nat) : Real) *
48           ((horizon + 2 : Nat) : Real)) /
49           standardExpectedRegretDelta horizon) =
50         (((horizon + 1 : Nat) : Real) ^ 2) *
51           ((horizon + 2 : Nat) : Real) := by
52     unfold standardExpectedRegretDelta
53     field_simp
54   unfold telescopingHighProbabilityRegretLogBudget
55     telescopingStandardExpectedRegretLogBudget
56   rw [hscale]

58 /--
59 Explicit expected pseudo-regret budget for a stopping time bounded by `T`
60 under the single scheduled policy tuned with outer budget `1 / (T + 1)`.
61 -/
62 noncomputable def telescopingStandardExpectedPseudoRegretBound
63     {Feature : Type u} [Fintype Feature]
64     (R lambda S : Real) (horizon : Nat) (L2 : Real) : Real :=
65   4 * S * Real.sqrt L2 +
66     2 *
67       (R * Real.sqrt
68           (telescopingStandardExpectedRegretLogBudget
69             (Feature := Feature) lambda horizon L2) +
70         Real.sqrt lambda * S) *
71       (Real.sqrt (((horizon + 1 : Nat) : Real)) *
72         Real.sqrt
73           (2 *
74             standardScalarLogDetBudget
75               (Feature := Feature) lambda (horizon + 1) L2))
76 
77 /--
78 The endpoint high-probability budget plus the tuned bad-event envelope charge
79 is exactly the explicit scheduled expected pseudo-regret rate.
80 -/
81 theorem
82     telescopingHighProbabilityPseudoRegretBound_add_initial_standardExpectedRegretDelta_eq
83     {Feature : Type u} [Fintype Feature]
84     (R lambda S : Real) (horizon : Nat) (L2 : Real) :
85     telescopingHighProbabilityPseudoRegretBound
86         (Feature := Feature)
87         R (standardExpectedRegretDelta horizon)
88         lambda S horizon L2 +
89         standardScalarInitialGapBound S L2 =
90       telescopingStandardExpectedPseudoRegretBound
91         (Feature := Feature) R lambda S horizon L2 := by
92   unfold telescopingHighProbabilityPseudoRegretBound
93     telescopingStandardExpectedPseudoRegretBound
94     standardScalarInitialGapBound
95   rw [
96     telescopingHighProbabilityRegretLogBudget_standardExpectedRegretDelta
