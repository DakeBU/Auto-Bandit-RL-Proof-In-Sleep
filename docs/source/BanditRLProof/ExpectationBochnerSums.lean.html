4 /-!
5 # Bochner expectation over finite sums
6 
7 Thin Mathlib-backed wrappers for finite-sum linearity of the Bochner integral.
8 These are the expectation-level companion to `IntegrabilitySums`: each summand
9 must be integrable, and then the integral of the finite sum is the finite sum
10 of the integrals.
11 -/
12 
13 namespace BanditRLProof
14 namespace ExpectationBochnerSums
15 
16 open MeasureTheory
17 
18 universe u v w
19 
20 /--
21 The Bochner integral distributes over a finite sum of integrable terms.
22 
23 This is the `EXP-FINITE-SUM` import wrapper.  It is polymorphic in the
24 Bochner codomain; bandit expected-regret applications typically instantiate
25 `E := Real`.
26 -/
27 theorem integral_finset_sum
28     {Omega : Type u} [MeasurableSpace Omega]
29     {Idx : Type v}
30     {E : Type w} [NormedAddCommGroup E] [NormedSpace Real E]
31     (mu : Measure Omega)
32     (s : Finset Idx)
33     (f : Idx -> Omega -> E)
34     (hf : forall i, i ∈ s -> Integrable (f i) mu) :
35     MeasureTheory.integral mu
36       (fun omega : Omega => s.sum (fun i => f i omega)) =
37     s.sum (fun i => MeasureTheory.integral mu (f i)) := by
38   simpa only [Finset.sum_apply] using
39     (MeasureTheory.integral_finset_sum
40       (μ := mu)
41       (s := s)
42       (f := f)
43       hf)
44 
45 /--
46 Finite-type specialization of `integral_finset_sum`.
47 
48 The statement exposes the common finite-arm `(Finset.univ : Finset Idx)`
49 shape used by regret decompositions.
50 -/
51 theorem integral_univ_sum
52     {Omega : Type u} [MeasurableSpace Omega]
53     {Idx : Type v} [Fintype Idx]
54     {E : Type w} [NormedAddCommGroup E] [NormedSpace Real E]
55     (mu : Measure Omega)
56     (f : Idx -> Omega -> E)
57     (hf : forall i : Idx, Integrable (f i) mu) :
58     MeasureTheory.integral mu
59       (fun omega : Omega =>
60         (Finset.univ : Finset Idx).sum (fun i => f i omega)) =
61     (Finset.univ : Finset Idx).sum
62       (fun i => MeasureTheory.integral mu (f i)) := by
63   exact
64     integral_finset_sum
65       (mu := mu)
66       (s := (Finset.univ : Finset Idx))
67       (f := f)
68       (hf := fun i _hi => hf i)
69 
70 end ExpectationBochnerSums
