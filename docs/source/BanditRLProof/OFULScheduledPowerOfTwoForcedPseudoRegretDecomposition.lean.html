3 /-!
4 # Pseudo-regret decomposition for power-of-two forced telescoping OFUL
5 
6 This module separates the complete finite-horizon pseudo-regret of the
7 horizon-independent power-of-two forced policy into the initial action,
8 power-of-two forced successor actions, and ordinary optimistic successor
9 actions. The successor index `n` denotes the history used to select action
10 `n + 1`.
11 -/

128 /-- Successor pseudo-regret charged at power-of-two forced indices. -/
129 noncomputable def powerOfTwoForcedSuccessorPseudoRegret
130     {K : Nat} {Feature : Type u}
131     [Fintype Feature]
132     (thetaStar : Feature -> Real)
133     (actionFeature : Fin K -> Feature -> Real)
134     (best : Fin K)
135     (horizon : Nat)
136     (trajectory : Nat -> Fin K × Real) : Real :=
137   (powerOfTwoForcedIndexSet horizon).sum (fun n =>
138     linearValue thetaStar (actionFeature best) -
139       linearValue thetaStar
140         (actionFeature
141           (Thompson.canonicalHistoryTrajectoryAction trajectory (n + 1))))

173 /--
174 Complete pseudo-regret through action `horizon` is the initial gap plus the
175 power-of-two forced and nonforced optimistic successor charges.
176 -/
177 theorem
178     canonicalStandardHighProbabilityPseudoRegret_eq_initial_add_powerOfTwoForced_add_optimistic
179     {K : Nat} {Feature : Type u}
180     [Fintype Feature]
181     (thetaStar : Feature -> Real)
182     (actionFeature : Fin K -> Feature -> Real)
183     (best : Fin K)
184     (horizon : Nat)
185     (trajectory : Nat -> Fin K × Real) :
186     canonicalStandardHighProbabilityPseudoRegret
187         thetaStar actionFeature best horizon trajectory =
188       (linearValue thetaStar (actionFeature best) -
189           linearValue thetaStar
190             (actionFeature
191               (Thompson.canonicalHistoryTrajectoryAction trajectory 0))) +
192         powerOfTwoForcedSuccessorPseudoRegret
193           thetaStar actionFeature best horizon trajectory +
194         powerOfTwoOptimisticSuccessorPseudoRegret
195           thetaStar actionFeature best horizon trajectory := by
196   unfold canonicalStandardHighProbabilityPseudoRegret
197     canonicalHistoryTrajectorySumRangeAllGap
198     powerOfTwoForcedSuccessorPseudoRegret
199     powerOfTwoOptimisticSuccessorPseudoRegret
200     powerOfTwoForcedIndexSet
201   rw [Finset.sum_range_succ']
202   rw [← Finset.sum_filter_add_sum_filter_not
203     (Finset.range horizon) (fun n => isPowerOfTwoForcedIndex n)]
204   ring

206 /--
207 Under the power-of-two forced policy, its trajectory-valued forced successor
208 charge is almost surely the deterministic prescribed-action charge.
209 -/
210 theorem
211     powerOfTwoForcedSuccessorPseudoRegret_ae_eq_forcedActionCharge
212     {K : Nat} {Feature : Type u}
213     [Fintype Feature] [DecidableEq Feature]
214     (hK : 0 < K)
215     (lambda : Real)
216     (thetaStar : Feature -> Real)
217     (actionFeature : Fin K -> Feature -> Real)
218     (R delta S : Real)
219     (forcedAction : Nat -> Fin K)
220     (environment : Thompson.HistoryEnvironment (Fin K) Real)
221     (horizon : Nat)
222     (best : Fin K) :
223     powerOfTwoForcedSuccessorPseudoRegret
224         thetaStar actionFeature best horizon =ᵐ[
225       Thompson.canonicalHistoryTrajectoryMeasure
226         (finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAlgorithm
227           hK lambda actionFeature R delta S forcedAction)
228         environment]
229       fun _trajectory =>
230         powerOfTwoForcedActionSuccessorPseudoRegret
231           thetaStar actionFeature best forcedAction horizon := by
232   let mu :=
233     Thompson.canonicalHistoryTrajectoryMeasure
234       (finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAlgorithm
235         hK lambda actionFeature R delta S forcedAction)
236       environment
237   have hselectors :
238       ∀ᵐ trajectory ∂mu, ∀ n,
239         n < horizon ->
240         isPowerOfTwoForcedIndex n ->
241         Thompson.canonicalHistoryTrajectoryAction trajectory (n + 1) =
242           forcedAction (Nat.log2 (n + 1)) := by
243     rw [ae_all_iff]
244     intro n
245     by_cases hn : n < horizon
246     · by_cases hforced : isPowerOfTwoForcedIndex n
247       · filter_upwards [
248           canonicalHistoryTrajectory_action_succ_ae_eq_forcedAction_of_powerOfTwoForced
249             hK lambda actionFeature R delta S forcedAction environment n
250             hforced] with trajectory htrajectory
251         intro _hn _hforced
252         simpa only [Thompson.canonicalHistoryTrajectoryAction] using htrajectory
253       · exact Filter.Eventually.of_forall (fun _trajectory _hn hcontra =>
254           (hforced hcontra).elim)
255     · exact Filter.Eventually.of_forall (fun _trajectory hcontra =>
256         (hn hcontra).elim)
257   filter_upwards [hselectors] with trajectory hselector
258   unfold powerOfTwoForcedSuccessorPseudoRegret
259     powerOfTwoForcedActionSuccessorPseudoRegret
260   apply Finset.sum_congr rfl
261   intro n hn
262   have hn_range : n < horizon :=
263     Finset.mem_range.mp (Finset.mem_filter.mp hn).1
264   have hforced : isPowerOfTwoForcedIndex n :=
265     Finset.mem_filter.mp hn |>.2
266   rw [hselector n hn_range hforced]

269 The complete pathwise decomposition with its forced term already replaced by
270 the deterministic prescribed-action charge almost surely.

272 theorem
273     canonicalStandardHighProbabilityPseudoRegret_ae_eq_initial_add_powerOfTwoForcedAction_add_optimistic

275     [Fintype Feature] [DecidableEq Feature]
276     (hK : 0 < K)
277     (lambda : Real)
278     (thetaStar : Feature -> Real)
279     (actionFeature : Fin K -> Feature -> Real)
280     (R delta S : Real)
281     (forcedAction : Nat -> Fin K)
282     (environment : Thompson.HistoryEnvironment (Fin K) Real)
283     (horizon : Nat)
284     (best : Fin K) :
285     ∀ᵐ trajectory ∂
286         Thompson.canonicalHistoryTrajectoryMeasure
287           (finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAlgorithm
288             hK lambda actionFeature R delta S forcedAction)
289           environment,
290       canonicalStandardHighProbabilityPseudoRegret
291           thetaStar actionFeature best horizon trajectory =

302       hK lambda thetaStar actionFeature R delta S forcedAction
