21 theorem standardScalarLogDetBudget_isBigO_log
22     {Feature : Type u}
23     [Fintype Feature] [Nonempty Feature]

25     (L2 : Real) (hL2 : 0 <= L2) :
26     (fun horizon : Nat =>
27       standardScalarLogDetBudget
28         (Feature := Feature) lambda (horizon + 1) L2) =O[atTop]
29       (fun horizon : Nat =>
30         Real.log (((horizon + 1 : Nat) : Real))) := by
31   rw [Asymptotics.isBigO_iff]
32   let d : Real := Fintype.card Feature
33   let c : Real := L2 / (d * lambda)
34   have hd : 0 < d := by
35     dsimp [d]
36     exact_mod_cast (Fintype.card_pos_iff.mpr inferInstance :
37       0 < Fintype.card Feature)
38   have hc : 0 <= c := div_nonneg hL2 (mul_pos hd hlambda).le
39   let n : Nat -> Real := fun horizon => (((horizon + 1 : Nat) : Real))
40   have hn_tendsto : Tendsto n atTop atTop := by
41     exact (tendsto_add_atTop_iff_nat 1).2 tendsto_natCast_atTop_atTop
42   have hlog_tendsto : Tendsto (fun horizon => Real.log (n horizon)) atTop atTop :=
43     Real.tendsto_log_atTop.comp hn_tendsto
44   refine ⟨2 * d, ?_⟩
45   filter_upwards
46       [hlog_tendsto.eventually (eventually_ge_atTop (Real.log (1 + c))),
47        hlog_tendsto.eventually (eventually_ge_atTop 0)]
48       with horizon hconst hlog
49   have hn_pos : 0 < n horizon := by
50     dsimp [n]
51     positivity
52   have honec_pos : 0 < 1 + c := by linarith
53   have harg :
54       1 + n horizon * c <= n horizon * (1 + c) := by
55     nlinarith [show 1 <= n horizon by
56       dsimp [n]
57       exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)]
58   have hlog_arg :
59       Real.log (1 + n horizon * c) <=
60         Real.log (n horizon) + Real.log (1 + c) := by
61     calc
62       Real.log (1 + n horizon * c)
63           <= Real.log (n horizon * (1 + c)) :=
64         Real.log_le_log (by positivity) harg
65       _ = Real.log (n horizon) + Real.log (1 + c) :=
66         Real.log_mul hn_pos.ne' honec_pos.ne'
67   have hbudget_nonneg :
68       0 <= standardScalarLogDetBudget
69         (Feature := Feature) lambda (horizon + 1) L2 :=
70     standardScalarLogDetBudget_nonneg
71       (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
72   rw [Real.norm_eq_abs, abs_of_nonneg hbudget_nonneg,
73     Real.norm_eq_abs, abs_of_nonneg hlog]
74   have hfinal :
75       d * Real.log (1 + n horizon * c) <=
76         (2 * d) * Real.log (n horizon) := by
77     nlinarith [mul_le_mul_of_nonneg_left hlog_arg hd.le]
78   simpa [standardScalarLogDetBudget, d, c, n, div_eq_mul_inv, mul_assoc] using hfinal
