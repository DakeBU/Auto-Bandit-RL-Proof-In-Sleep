3 /-!
4 # Exponential confidence for the EXP3 mixed estimator-square sum
5 
6 This module replaces the Markov-only confidence step for the observed mixed
7 importance-weighted estimator-square sum by a finite-action conditional
8 sub-Gaussian route.  The raw score lies in `[0, 1 / epsilon]` and its
9 conditional mean is the armwise predictable loss-square sum.  Consequently,
10 the centered generated process has an exponential finite-horizon tail.
11 
12 The proxy remains Hoeffding's interval proxy for `[0, |arms| / gamma]`.  This
13 is a genuine exponential tail, but not the sharper variance-sensitive
14 Freedman/EXP3.P rate.
15 -/

23 /-- Under a positive probability floor and unit losses, the mixed estimator
24 square has the sharper reciprocal-floor bound, rather than the generic square
25 of that reciprocal. -/
26 theorem mixedSquaredImportanceWeightedLoss_le_inv_floor
27     {History : Type u} {Action : Type v}
28     [MeasurableSpace History] [MeasurableSpace Action]
29     [MeasurableSingletonClass Action] [DecidableEq Action]
30     (arms : Finset Action) (prob loss : History -> Action -> Real)
31     (epsilon : Real)
32     (regularity : BoundedMeasurableLossWithProbabilityFloor
33       arms prob loss epsilon)
34     (history : History) (chosen : Action) :
35     mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
36         chosen <= 1 / epsilon := by
37   by_cases hchosen : chosen ∈ arms
38   · have hprob_pos := regularity.prob_pos history chosen hchosen
39     have hloss := regularity.loss_mem_Icc history chosen hchosen
40     have hsq : (loss history chosen) ^ 2 <= 1 := by
41       nlinarith [hloss.1, hloss.2]
42     have hone_le : 1 <= (1 / epsilon) * prob history chosen := by
43       rw [one_div_mul_eq_div, le_div_iff₀ regularity.epsilon_pos]
44       simpa using regularity.prob_floor history chosen hchosen
45     rw [mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
46       arms (prob history) (loss history) chosen hchosen hprob_pos.ne']
47     apply (div_le_iff₀ hprob_pos).2
48     exact hsq.trans hone_le
49   · have hzero : mixedSquaredImportanceWeightedLoss arms (prob history)
50         (loss history) chosen = 0 := by
51       unfold mixedSquaredImportanceWeightedLoss
52       apply Finset.sum_eq_zero
53       intro candidate hcandidate
54       have hne : chosen ≠ candidate := by
55         intro heq
56         exact hchosen (heq ▸ hcandidate)
57       simp [importanceWeightedLoss, hne]
58     rw [hzero]
59     exact one_div_nonneg.mpr regularity.epsilon_pos.le
