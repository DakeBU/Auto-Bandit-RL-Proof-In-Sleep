5 # Generated half-Tsallis stability measurability
6 
7 This module derives measurability of the generated one-round stability score
8 from coordinate measurability of the current and updated half-Tsallis
9 selectors.  The updated selector remains noncomputable, so its coordinate
10 measurability is kept as an explicit selector contract rather than inferred
11 from `Classical.choose`.
12 -/

21 /-- Coordinate regularity for the generated current and sampled-action updated
22 half-Tsallis selectors.  The finite-history component constructs the policy;
23 the second component covers the environment/prefix/action parameter space of
24 the updated canonical minimizer. -/
25 structure HalfTsallisGeneratedSelectorMeasurability
26     {Env : Type u} {Action : Type v}
27     [MeasurableSpace Env] [MeasurableSpace Action]
28     [MeasurableSingletonClass Action] [DecidableEq Action]
29     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)

31   finiteHistory :
32     HalfTsallisFiniteHistorySelectorMeasurability arms harms eta
33   measurable_updated : forall n candidate, candidate ∈ arms ->
34     Measurable (fun sample :
35         (Env × History.FinitePairHistory Action Real n) × Action =>
36       sampledHalfTsallisUpdatedAt arms harms eta loss n
37         sample.1 sample.2 candidate)

50 /-- A finite sum of current-minus-updated importance-weighted linear losses is
51 measurable once all current, loss, and updated coordinates are measurable. -/
52 theorem measurable_importanceWeightedStabilityScore
53     {History : Type u} {Action : Type v}
54     [MeasurableSpace History] [MeasurableSpace Action]
55     [MeasurableSingletonClass Action] [DecidableEq Action]
56     (arms : Finset Action)
57     (prob loss : History -> Action -> Real)
58     (next : History -> Action -> Action -> Real)
59     (hprob : forall candidate, candidate ∈ arms ->
60       Measurable (fun history => prob history candidate))
61     (hloss : forall candidate, candidate ∈ arms ->
62       Measurable (fun history => loss history candidate))
63     (hnext : forall candidate, candidate ∈ arms ->
64       Measurable (fun sample : History × Action =>
65         next sample.1 sample.2 candidate)) :
66     Measurable (importanceWeightedStabilityScore arms prob loss next) := by
67   classical
68   unfold importanceWeightedStabilityScore FTRL.linearLoss
69   apply (Finset.measurable_sum arms fun candidate hcandidate => ?_).sub
70     (Finset.measurable_sum arms fun candidate hcandidate => ?_)
71   · have hestimated : Measurable (fun sample : History × Action =>
72         Exp3.importanceWeightedLoss (prob sample.1) (loss sample.1)
73           sample.2 candidate) := by
74       unfold Exp3.importanceWeightedLoss
75       refine Measurable.ite ?_
76         (((hloss candidate hcandidate).comp measurable_fst).div
77           ((hprob candidate hcandidate).comp measurable_fst))
78         measurable_const
79       simpa only [Set.mem_setOf_eq] using
80         measurable_snd (measurableSet_singleton candidate)
81     exact ((hprob candidate hcandidate).comp measurable_fst).mul hestimated
82   · have hestimated : Measurable (fun sample : History × Action =>
83         Exp3.importanceWeightedLoss (prob sample.1) (loss sample.1)
84           sample.2 candidate) := by
85       unfold Exp3.importanceWeightedLoss
86       refine Measurable.ite ?_
87         (((hloss candidate hcandidate).comp measurable_fst).div
88           ((hprob candidate hcandidate).comp measurable_fst))
89         measurable_const
90       simpa only [Set.mem_setOf_eq] using
91         measurable_snd (measurableSet_singleton candidate)
92     exact (hnext candidate hcandidate).mul hestimated

94 /-- Every supported coordinate of the generated predictable loss vector is
95 measurable on the environment/prefix parameter space. -/
96 theorem measurable_sampledHalfTsallisPredictableLossAt
97     {Env : Type u} {Action : Type v}
98     [MeasurableSpace Env] [MeasurableSpace Action]
99     (loss : Exp3.PredictableLossVector Env Action) (n : Nat)
100     (candidate : Action) :
101     Measurable (fun input :
102         Env × History.FinitePairHistory Action Real n =>
103       sampledHalfTsallisPredictableLossAt loss n input candidate) := by
104   simpa only [sampledHalfTsallisPredictableLossAt] using
105     (loss.measurable_successor n).comp
106       (measurable_fst.prodMk (measurable_snd.prodMk measurable_const))

108 /-- Every supported coordinate of the canonical sampled-action update is
109 measurable without an external selector assumption. -/
110 theorem measurable_sampledHalfTsallisUpdatedAt_canonical
111     {Env : Type u} {Action : Type v}
112     [MeasurableSpace Env] [MeasurableSpace Action]
113     [MeasurableSingletonClass Action] [DecidableEq Action]
114     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
115     (loss : Exp3.PredictableLossVector Env Action) (n : Nat)
116     (candidate : Action) (hcandidate : candidate ∈ arms) :
117     Measurable (fun sample :
118         (Env × History.FinitePairHistory Action Real n) × Action =>
119       sampledHalfTsallisUpdatedAt arms harms eta loss n
120         sample.1 sample.2 candidate) := by
121   let finiteSelector : HalfTsallisFiniteHistorySelectorMeasurability
122       arms harms eta :=
123     canonicalHalfTsallisFiniteHistorySelectorMeasurability arms harms eta
124   let updatedScore :
125       ((Env × History.FinitePairHistory Action Real n) × Action) ->
126         Action -> Real :=
127     fun sample selected =>
128       sampledHalfTsallisScoreAt arms harms eta n sample.1 selected +
129         Exp3.importanceWeightedLoss
130           (sampledHalfTsallisProbabilityAt arms harms eta n sample.1)
131           (sampledHalfTsallisPredictableLossAt loss n sample.1)
132           sample.2 selected
133   have hupdatedScore : forall selected, selected ∈ arms ->
134       Measurable (fun sample => updatedScore sample selected) := by
135     intro selected hselected
136     have hscore : Measurable (fun sample :
137         ((Env × History.FinitePairHistory Action Real n) × Action) =>
138         sampledHalfTsallisScoreAt arms harms eta n sample.1 selected) :=
139       (measurable_sampledHalfTsallisHistoryScore
140         arms harms eta finiteSelector n selected hselected).comp
141           (measurable_snd.comp measurable_fst)
142     have hprob : Measurable (fun sample :
143         ((Env × History.FinitePairHistory Action Real n) × Action) =>
144         sampledHalfTsallisProbabilityAt arms harms eta n
145           sample.1 selected) :=
146       ((sampledHalfTsallisEnvironmentHistoryDistributionSource
147         (Env := Env) arms harms eta finiteSelector n).measurable_prob
148           selected hselected).comp measurable_fst
149     have hloss : Measurable (fun sample :
150         ((Env × History.FinitePairHistory Action Real n) × Action) =>
151         sampledHalfTsallisPredictableLossAt loss n
152           sample.1 selected) :=
153       (measurable_sampledHalfTsallisPredictableLossAt loss n selected).comp
154         measurable_fst
155     have hincrement : Measurable (fun sample :
156         ((Env × History.FinitePairHistory Action Real n) × Action) =>
157         Exp3.importanceWeightedLoss
158           (sampledHalfTsallisProbabilityAt arms harms eta n sample.1)
159           (sampledHalfTsallisPredictableLossAt loss n sample.1)
160           sample.2 selected) := by
161       unfold Exp3.importanceWeightedLoss
162       refine Measurable.ite ?_ (hloss.div hprob) measurable_const
163       simpa only [Set.mem_setOf_eq] using
164         measurable_snd (measurableSet_singleton selected)
165     change Measurable (fun sample :
166         ((Env × History.FinitePairHistory Action Real n) × Action) =>
167       sampledHalfTsallisScoreAt arms harms eta n sample.1 selected +
168         Exp3.importanceWeightedLoss
169           (sampledHalfTsallisProbabilityAt arms harms eta n sample.1)
170           (sampledHalfTsallisPredictableLossAt loss n sample.1)
171           sample.2 selected)
172     exact hscore.add hincrement
173   have hcanonical : Measurable (fun sample =>
174       halfTsallisMinimizer arms harms eta
175         (updatedScore sample) candidate) :=
176     measurable_halfTsallisMinimizer_comp
177       arms harms eta updatedScore hupdatedScore candidate hcandidate
178   simpa only [updatedScore, sampledHalfTsallisUpdatedAt,
179     halfTsallisHistoryUpdatedMinimizer, halfTsallisUpdatedMinimizer,
180     sampledHalfTsallisProbabilityAt, halfTsallisHistoryMinimizer] using
181       hcanonical

183 /-- The canonical current and one-step updated minimizers satisfy the full
184 generated-selector regularity contract; callers no longer need to assume
185 measurability of either `Classical.choose` surface. -/
186 noncomputable def canonicalHalfTsallisGeneratedSelectorMeasurability
187     {Env : Type u} {Action : Type v}
188     [MeasurableSpace Env] [MeasurableSpace Action]
189     [MeasurableSingletonClass Action] [DecidableEq Action]
190     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
191     (loss : Exp3.PredictableLossVector Env Action) :
192     HalfTsallisGeneratedSelectorMeasurability arms harms eta loss where
193   finiteHistory :=
194     canonicalHalfTsallisFiniteHistorySelectorMeasurability arms harms eta
195   measurable_updated :=
196     measurable_sampledHalfTsallisUpdatedAt_canonical arms harms eta loss

198 /-- The generated one-round current-minus-updated stability score is
199 measurable under the generated selector coordinate contract. -/
200 theorem measurable_sampledHalfTsallisHistoryActionStabilityAt
201     {Env : Type u} {Action : Type v}
202     [MeasurableSpace Env] [MeasurableSpace Action]
203     [MeasurableSingletonClass Action] [DecidableEq Action]
204     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
205     (loss : Exp3.PredictableLossVector Env Action)
206     (selector : HalfTsallisGeneratedSelectorMeasurability
207       arms harms eta loss) (n : Nat) :
208     Measurable (sampledHalfTsallisHistoryActionStabilityAt
209       arms harms eta loss n) := by
210   unfold sampledHalfTsallisHistoryActionStabilityAt
211   apply measurable_importanceWeightedStabilityScore
212   · intro candidate hcandidate
213     exact (sampledHalfTsallisEnvironmentHistoryDistributionSource
214       (Env := Env) arms harms eta selector.finiteHistory n).measurable_prob
215         candidate hcandidate
216   · intro candidate _hcandidate
217     exact measurable_sampledHalfTsallisPredictableLossAt loss n candidate
218   · intro candidate hcandidate
219     exact selector.measurable_updated n candidate hcandidate

221 /-- Generated finite-horizon actual-successor stability with all score and
222 integrability regularity derived from one generated-selector contract. -/
223 theorem integral_sum_sampledHalfTsallisSuccessorStability_le_integral_sum_halfPowerStabilityBound_of_selector
224     {Env : Type u} {Action : Type v}
225     [MeasurableSpace Env] [StandardBorelSpace Env]
226     [MeasurableSpace Action] [MeasurableSingletonClass Action]
227     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
228     (prior : Measure Env) [IsFiniteMeasure prior]
229     (horizon : Nat) (arms : Finset Action) (harms : arms.Nonempty)
230     (eta : Real) (heta : 0 < eta)
231     (loss : Exp3.PredictableLossVector Env Action)
232     (selector : HalfTsallisGeneratedSelectorMeasurability
233       arms harms eta loss) :
234     let mu := prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
235       arms harms eta selector.finiteHistory loss.environment
236     integral mu (fun sample => (Finset.range horizon).sum (fun n =>
237         sampledHalfTsallisSuccessorStabilityAt
238           arms harms eta loss n sample)) <=
239       integral mu (fun sample => (Finset.range horizon).sum (fun n =>
240         sampledHalfTsallisHalfPowerBoundAt (Env := Env)
241           arms harms eta n (sampledHalfTsallisHistoryAt n sample))) := by
242   exact
243     integral_sum_sampledHalfTsallisSuccessorStability_le_integral_sum_halfPowerStabilityBound_of_measurable
244       prior horizon arms harms eta heta selector.finiteHistory loss
245       (fun n => measurable_sampledHalfTsallisHistoryActionStabilityAt
246         arms harms eta loss selector n)

248 /-- Generated finite-horizon actual-successor stability for the canonical
249 half-Tsallis policy, with selector measurability proved internally. -/
250 theorem integral_sum_sampledHalfTsallisSuccessorStability_le_integral_sum_halfPowerStabilityBound_canonical
251     {Env : Type u} {Action : Type v}
252     [MeasurableSpace Env] [StandardBorelSpace Env]
253     [MeasurableSpace Action] [MeasurableSingletonClass Action]
254     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
255     (prior : Measure Env) [IsFiniteMeasure prior]
256     (horizon : Nat) (arms : Finset Action) (harms : arms.Nonempty)
257     (eta : Real) (heta : 0 < eta)
258     (loss : Exp3.PredictableLossVector Env Action) :
259     let selector := canonicalHalfTsallisGeneratedSelectorMeasurability
260       arms harms eta loss
261     let mu := prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
262       arms harms eta selector.finiteHistory loss.environment
263     integral mu (fun sample => (Finset.range horizon).sum (fun n =>
264         sampledHalfTsallisSuccessorStabilityAt
265           arms harms eta loss n sample)) <=
266       integral mu (fun sample => (Finset.range horizon).sum (fun n =>
267         sampledHalfTsallisHalfPowerBoundAt (Env := Env)

269   dsimp only
270   exact
