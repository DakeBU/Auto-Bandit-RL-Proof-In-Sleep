5 # Realized EXP3 regret from a predictable-variance expectation budget
6 
7 This module discharges the explicit predictable-variance overflow residual by
8 Markov's inequality. The resulting theorem requires a caller-supplied
9 `lintegral` bound for the cumulative predictable mixed-square variance; no
10 such algorithm-specific expectation bound is inferred from predictability.
11 -/

72 /-- Mathlib-backed Markov tail for cumulative predictable mixed-square
73 variance. The measure need not be finite. -/
74 theorem measure_sampledPredictableMixedSquaredVarianceSum_gt_le_lintegral_div
75     {Env : Type u} {Action : Type v}
76     [MeasurableSpace Env]
77     [MeasurableSpace Action] [MeasurableSingletonClass Action]
78     [DecidableEq Action]
79     (mu : Measure (Env × ((k : Nat) → Action × Real)))
80     (arms : Finset Action) (harms : arms.Nonempty)
81     (eta gamma : Real) (hgamma_pos : 0 < gamma)
82     (hgamma_le_one : gamma ≤ 1)
83     (loss : PredictableLossVector Env Action) (horizon : Nat)
84     (varianceBudget : Real) (hvarianceBudget : 0 < varianceBudget) :
85     mu {sample |
86         varianceBudget < sampledPredictableMixedSquaredVarianceSum
87           arms eta gamma loss horizon sample} ≤
88       sampledPredictableMixedSquaredVarianceLIntegral
89           mu arms eta gamma loss horizon /
90         ENNReal.ofReal varianceBudget := by
91   let varianceSum := sampledPredictableMixedSquaredVarianceSum
92     arms eta gamma loss horizon
93   have hvarianceSum : Measurable varianceSum :=
94     measurable_sampledPredictableMixedSquaredVarianceSum
95       arms harms eta gamma hgamma_pos hgamma_le_one loss horizon
96   have hofReal : AEMeasurable (fun sample =>
97       ENNReal.ofReal (varianceSum sample)) mu :=
98     (ENNReal.measurable_ofReal.comp hvarianceSum).aemeasurable
99   have hmarkov := meas_ge_le_lintegral_div hofReal
100     (ENNReal.ofReal_pos.2 hvarianceBudget).ne'
101     ENNReal.ofReal_ne_top
102   calc
103     mu {sample | varianceBudget < varianceSum sample} ≤
104         mu {sample |
105           ENNReal.ofReal varianceBudget ≤ ENNReal.ofReal (varianceSum sample)} := by
106       apply measure_mono
107       intro sample hsample
108       exact ENNReal.ofReal_le_ofReal hsample.le
109     _ ≤ (∫⁻ sample, ENNReal.ofReal (varianceSum sample) ∂mu) /
110           ENNReal.ofReal varianceBudget := hmarkov
111     _ = sampledPredictableMixedSquaredVarianceLIntegral
112           mu arms eta gamma loss horizon /
113         ENNReal.ofReal varianceBudget := by
114       rfl

129     (horizon : Nat) (hhorizon : 0 < horizon)

133     (hdeltaConfidence : 0 < deltaConfidence)
134     (hdeltaRealized : 0 < deltaRealized)

142       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
143     mu {sample |
144         sampledPredictableVarianceSquareRealizedHighProbabilityRegretBudget

148               sampledTrajectoryRealizedLossAt t sample) -
149             (Finset.range horizon).sum (fun t =>

151       (((ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +

153         ENNReal.ofReal varianceMeanBudget /

171             sampledTrajectoryPredictableMixedSquaredVarianceAt
172               arms eta gamma loss i sample)} ≤
173         ENNReal.ofReal varianceMeanBudget /
174           ENNReal.ofReal varianceBudget := by

178             sampledTrajectoryPredictableMixedSquaredVarianceAt
179               arms eta gamma loss i sample)} ≤
180           sampledPredictableMixedSquaredVarianceLIntegral
181               mu arms eta gamma loss horizon /

189 /-- Five-event realized-regret budget: four confidence failures and one
190 Markov predictable-variance overflow failure. -/
191 noncomputable def sampledPredictableVarianceSquareRealizedMarkovHighProbabilityRegretBudget
192     {Action : Type v} [DecidableEq Action]
193     (arms : Finset Action) (eta gamma : Real) (horizon : Nat)

201 theorem sampledPredictable_predictableVarianceSquareRealizedMarkovHighProbabilityRegret_tail_total_delta
202     {Env : Type u} {Action : Type v}
203     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
204     [MeasurableSpace Action] [MeasurableSingletonClass Action]

209     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
210     (loss : PredictableLossVector Env Action)
211     (comparator : Action) (hcomparator : comparator ∈ arms)
212     (horizon : Nat) (hhorizon : 0 < horizon)
213     (varianceMeanBudget delta : Real)
214     (hvarianceMeanBudget : 0 < varianceMeanBudget) (hdelta : 0 < delta)
215     (hvarianceLIntegral :
216       sampledPredictableMixedSquaredVarianceLIntegral
217           (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
218             eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
219           arms eta gamma loss horizon ≤ ENNReal.ofReal varianceMeanBudget) :
220     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
221       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
222     mu {sample |
223         sampledPredictableVarianceSquareRealizedMarkovHighProbabilityRegretBudget
224             arms eta gamma horizon varianceMeanBudget delta ≤
225           (Finset.range horizon).sum (fun t =>
226               sampledTrajectoryRealizedLossAt t sample) -
227             (Finset.range horizon).sum (fun t =>
228               predictableLossAt loss t sample comparator)} ≤
229       ENNReal.ofReal delta := by
230   have hfifth_pos : 0 < delta / 5 := div_pos hdelta (by norm_num)
231   have hvarianceBudget : 0 < varianceMeanBudget / (delta / 5) :=
232     div_pos hvarianceMeanBudget hfifth_pos
233   have h :=

235       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
236         hcomparator horizon hhorizon (varianceMeanBudget / (delta / 5))
237         (delta / 5) (delta / 5) (delta / 5) hvarianceBudget hfifth_pos
238         hfifth_pos hfifth_pos varianceMeanBudget hvarianceLIntegral
239   dsimp only at h ⊢
240   have hmean_ne_zero : ENNReal.ofReal varianceMeanBudget ≠ 0 :=
241     (ENNReal.ofReal_pos.2 hvarianceMeanBudget).ne'
242   have hratio :
243       ENNReal.ofReal varianceMeanBudget /
244           ENNReal.ofReal (varianceMeanBudget / (delta / 5)) =
245         ENNReal.ofReal (delta / 5) := by
246     rw [ENNReal.ofReal_div_of_pos hfifth_pos]
247     exact ENNReal.div_div_cancel hmean_ne_zero ENNReal.ofReal_ne_top
248   have hfifth_nonneg : 0 ≤ delta / 5 := hfifth_pos.le
249   have hprob :
250       (((ENNReal.ofReal (delta / 5) + ENNReal.ofReal (delta / 5)) +
251           ENNReal.ofReal (delta / 5)) + ENNReal.ofReal (delta / 5)) +
252           ENNReal.ofReal (delta / 5) = ENNReal.ofReal delta := by
253     rw [← ENNReal.ofReal_add hfifth_nonneg hfifth_nonneg]
254     rw [← ENNReal.ofReal_add
255       (add_nonneg hfifth_nonneg hfifth_nonneg) hfifth_nonneg]
