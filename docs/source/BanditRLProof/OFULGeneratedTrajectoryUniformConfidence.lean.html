20 /--
21 Regularity source needed to apply scalar-ridge uniform confidence to one
22 canonical history-algorithm trajectory.
23 
24 The selected feature is the actual canonical action feature. Constructing this
25 source from a concrete reward environment therefore remains a genuine
26 predictability and conditional-law obligation.
27 -/
28 structure CanonicalScalarRidgeConfidenceSource
29     {K : Nat} {Feature : Type u}
30     [Fintype Feature] [DecidableEq Feature]
31     (algorithm : Thompson.HistoryAlgorithm (Fin K) Real)
32     (environment : Thompson.HistoryEnvironment (Fin K) Real)
33     (thetaStar : Feature -> Real)
34     (actionFeature : Fin K -> Feature -> Real)
35     (R S : Real)
36     (horizon : Nat) where
37   filtration :
38     Filtration Nat
39       (inferInstance : MeasurableSpace (Nat -> Fin K × Real))
40   noise : Nat -> (Nat -> Fin K × Real) -> Real
41   projectionBound : EuclideanSpace Real Feature -> Nat -> Real
42   theta_norm_le : euclideanLength thetaStar <= S
43   feature_stronglyMeasurable : forall i j,
44     StronglyMeasurable[filtration i] (fun trajectory =>
45       canonicalHistoryTrajectoryFeature actionFeature i trajectory j)
46   noise_stronglyAdapted :
47     StronglyAdapted filtration (fun t trajectory =>
48       match t with
49       | 0 => 0
50       | i + 1 => noise i trajectory)
51   projectionBound_nonneg : forall theta i,
52     0 <= projectionBound theta i
53   projection_le : forall theta i trajectory,
54     |dotProduct (WithLp.ofLp theta)
55         (canonicalHistoryTrajectoryFeature actionFeature i trajectory)| <=
56       projectionBound theta i
57   noise_hasCondSubgaussianMGF : forall i, i < horizon ->
58     ProbabilityTheory.HasCondSubgaussianMGF
59       (filtration i) (filtration.le i) (noise i)
60       (constantSquaredVarianceProxy R i)
61       (Thompson.canonicalHistoryTrajectoryMeasure algorithm environment)
62   response_eq : forall trajectory i, i < horizon ->
63     canonicalHistoryTrajectoryResponse i trajectory =
64       dotProduct thetaStar
65         (canonicalHistoryTrajectoryFeature actionFeature i trajectory) +
66       noise i trajectory

111 /--
112 Strict violation of the compiled successor-window OFUL good-event gap bound.
113 The range index `n` is charged to the actual action at time `n + 1`, so time
114 zero is not part of this event.
115 -/
116 def canonicalHistoryTrajectorySumRangeSuccGapViolationSet
117     {K : Nat} {Feature : Type u}
118     [Fintype Feature] [DecidableEq Feature]
119     (lambda : Real)
120     (thetaStar : Feature -> Real)
121     (actionFeature : Fin K -> Feature -> Real)
122     (R delta S : Real)
123     (horizon : Nat)
124     (comparator : Nat -> Fin K) :
125     Set (Nat -> Fin K × Real) :=
126   {trajectory |
127     (Finset.range horizon).sum (fun n =>
128         2 *
129           finiteHorizonScalarConfidenceRadius
130             (canonicalHistoryTrajectoryFeature actionFeature)
131             R (delta / ((horizon + 1 : Nat) : Real))
132             lambda S (n + 1) trajectory *
133           confidenceWidth
134             (finiteHorizonScalarGram lambda
135               (canonicalHistoryTrajectoryFeature actionFeature)
136               (n + 1) trajectory)
137             (actionFeature
138               (Thompson.canonicalHistoryTrajectoryAction
139                 trajectory (n + 1)))) <
140       (Finset.range horizon).sum (fun n =>
141         linearValue thetaStar (actionFeature (comparator (n + 1))) -
142           linearValue thetaStar
143             (actionFeature
144               (Thompson.canonicalHistoryTrajectoryAction
145                 trajectory (n + 1))))}

155     {K : Nat} {Feature : Type u}

157     (hK : 0 < K)
158     (lambda : Real) (hlambda : 0 < lambda)

169         hK lambda actionFeature R

171       environment thetaStar actionFeature R S horizon) :

181     finiteHistoryScalarRidgeOptimisticAlgorithm
182       hK lambda actionFeature R

185     Thompson.canonicalHistoryTrajectoryMeasure algorithm environment
