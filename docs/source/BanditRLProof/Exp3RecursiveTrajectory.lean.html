5 # Recursive measurable EXP3 trajectories
6 
7 This module turns a measurable finite-history score into the exploration-mixed
8 exponential policy used by EXP3.  It proves normalization, coordinate
9 measurability, and a uniform exploration floor, packages the policy as the
10 project's stochastic finite-history algorithm, and invokes the Mathlib-backed
11 Ionescu--Tulcea trajectory kernel.  The resulting theorem identifies every
12 successor action's conditional law with the explicit finite-action policy.
13 
14 The score is an input contract here.  Constructing it recursively from sampled
15 importance-weighted losses is the next layer of the EXP3 route.
16 -/

25 /-- Exponential weight generated by a finite-history cumulative score. -/
26 noncomputable def historyWeight {Action History : Type*}
27     (eta : Real) (score : History -> Action -> Real)
28     (history : History) (action : Action) : Real :=
29   Real.exp (-eta * score history action)
30 
31 /-- Total exponential weight on the explicit finite support. -/
32 noncomputable def historyTotalWeight {Action History : Type*}
33     (arms : Finset Action) (eta : Real)
34     (score : History -> Action -> Real) (history : History) : Real :=
35   arms.sum (historyWeight eta score history)
36 
37 /-- Normalized exponential history policy before explicit exploration mixing. -/
38 noncomputable def normalizedHistoryDistribution {Action History : Type*}
39     (arms : Finset Action) (eta : Real)
40     (score : History -> Action -> Real) (history : History)
41     (action : Action) : Real :=
42   historyWeight eta score history action /
43     historyTotalWeight arms eta score history
44 
45 /-- EXP3 policy: normalized exponential weights mixed with uniform exploration. -/
46 noncomputable def exploredHistoryDistribution {Action History : Type*}
47     (arms : Finset Action) (eta gamma : Real)
48     (score : History -> Action -> Real) (history : History)
49     (action : Action) : Real :=
50   (1 - gamma) * normalizedHistoryDistribution arms eta score history action +
51     gamma / (arms.card : Real)

53 theorem historyTotalWeight_pos
54     {Action History : Type*}
55     (arms : Finset Action) (harms : arms.Nonempty)
56     (eta : Real) (score : History -> Action -> Real) (history : History) :
57     0 < historyTotalWeight arms eta score history := by
58   unfold historyTotalWeight
59   refine Finset.sum_pos' (fun action _ => (Real.exp_pos _).le) ?_
60   rcases harms with ⟨action, haction⟩
61   exact ⟨action, haction, Real.exp_pos _⟩

63 theorem normalizedHistoryDistribution_nonneg
64     {Action History : Type*}
65     (arms : Finset Action) (harms : arms.Nonempty)
66     (eta : Real) (score : History -> Action -> Real)
67     (history : History) (action : Action) :
68     0 <= normalizedHistoryDistribution arms eta score history action := by
69   exact div_nonneg (Real.exp_pos _).le

72 theorem sum_normalizedHistoryDistribution
73     {Action History : Type*}
74     (arms : Finset Action) (harms : arms.Nonempty)
75     (eta : Real) (score : History -> Action -> Real) (history : History) :
76     arms.sum (normalizedHistoryDistribution arms eta score history) = 1 := by
77   rw [show arms.sum (normalizedHistoryDistribution arms eta score history) =
78       arms.sum (historyWeight eta score history) /
79         historyTotalWeight arms eta score history by
80     simp only [normalizedHistoryDistribution]
81     rw [Finset.sum_div]]
82   rw [show arms.sum (historyWeight eta score history) =
83       historyTotalWeight arms eta score history by rfl]
84   exact div_self (ne_of_gt
85     (historyTotalWeight_pos arms harms eta score history))

87 theorem exploredHistoryDistribution_nonneg
88     {Action History : Type*}
89     (arms : Finset Action) (harms : arms.Nonempty)
90     (eta gamma : Real) (score : History -> Action -> Real)
91     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
92     (history : History) (action : Action) :
93     0 <= exploredHistoryDistribution arms eta gamma score history action := by
94   unfold exploredHistoryDistribution
95   exact add_nonneg
96     (mul_nonneg (sub_nonneg.mpr hgamma_le_one)
97       (normalizedHistoryDistribution_nonneg arms harms eta score history action))
98     (div_nonneg hgamma_nonneg (Nat.cast_nonneg _))

100 theorem sum_exploredHistoryDistribution
101     {Action History : Type*}
102     (arms : Finset Action) (harms : arms.Nonempty)
103     (eta gamma : Real) (score : History -> Action -> Real)
104     (history : History) :
105     arms.sum (exploredHistoryDistribution arms eta gamma score history) = 1 := by
106   have hcard_pos : 0 < (arms.card : Real) := by
107     exact_mod_cast harms.card_pos
108   unfold exploredHistoryDistribution
109   rw [Finset.sum_add_distrib, ← Finset.mul_sum,
110     sum_normalizedHistoryDistribution arms harms]
111   simp only [Finset.sum_const, nsmul_eq_mul]
112   field_simp
113   ring

115 theorem exploredHistoryDistribution_floor
116     {Action History : Type*}
117     (arms : Finset Action) (harms : arms.Nonempty)
118     (eta gamma : Real) (score : History -> Action -> Real)
119     (hgamma_le_one : gamma <= 1)
120     (history : History) (action : Action) :
121     gamma / (arms.card : Real) <=
122       exploredHistoryDistribution arms eta gamma score history action := by
123   unfold exploredHistoryDistribution
124   exact le_add_of_nonneg_left
125     (mul_nonneg (sub_nonneg.mpr hgamma_le_one)

191 /-- Coordinate measurability for a score indexed by inclusive pair histories. -/
192 structure MeasurableFiniteHistoryScore
193     {Action : Type u} {Loss : Type v}
194     [MeasurableSpace Action] [MeasurableSpace Loss]
195     (arms : Finset Action)
196     (score : (n : Nat) -> History.FinitePairHistory Action Loss n ->
197       Action -> Real) : Prop where
198   measurable_score : forall n action, action ∈ arms ->
199     Measurable (fun history => score n history action)

201 /-- The un-explored normalized exponential-weights policy supplies measurable
202 probability vectors on every finite-history level. -/
203 def normalizedHistoryDistributionSource
204     {Action : Type u} {Loss : Type v}
205     [MeasurableSpace Action] [MeasurableSpace Loss]
206     (arms : Finset Action) (harms : arms.Nonempty)
207     (eta : Real)
208     (score : (n : Nat) -> History.FinitePairHistory Action Loss n ->
209       Action -> Real)
210     (hscore : MeasurableFiniteHistoryScore arms score)
211     (n : Nat) :
212     MeasurableFiniteActionDistribution arms
213       (normalizedHistoryDistribution arms eta (score n)) where
214   distribution := fun history =>
215     { nonneg := fun action _haction =>
216         normalizedHistoryDistribution_nonneg arms harms eta (score n)
217           history action
218       sum_eq_one :=
219         sum_normalizedHistoryDistribution arms harms eta (score n) history }
220   measurable_prob := fun action haction =>
221     measurable_normalizedHistoryDistribution arms eta (score n)
222       (fun selected hselected =>
223         hscore.measurable_score n selected hselected) action haction

263 noncomputable def exploredHistoryAlgorithm

265     [MeasurableSpace Action] [MeasurableSingletonClass Action]
266     [MeasurableSpace Loss]

270       Action -> Real)
271     (hscore : MeasurableFiniteHistoryScore arms score)
272     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1) :
273     Thompson.HistoryAlgorithm Action Loss where
274   policy n := finiteActionKernel arms
275     (exploredHistoryDistribution arms eta gamma (score n))

284   initialAction_isProbability := finiteActionMeasure_isProbabilityMeasure arms

345 /--
346 Every successor action of the recursive trajectory has the explicit
347 exploration-mixed EXP3 policy as its conditional law given the finite history.
348 -/
349 theorem exploredTrajectoryMeasure_condDistrib_action
350     {Env : Type w} {Action : Type u} {Loss : Type v}
351     [MeasurableSpace Env] [StandardBorelSpace Env]
352     [MeasurableSpace Action] [MeasurableSingletonClass Action]
353     [StandardBorelSpace Action] [Nonempty Action]
354     [MeasurableSpace Loss] [StandardBorelSpace Loss] [Nonempty Loss]
355     (prior : Measure Env) [IsFiniteMeasure prior]
356     (arms : Finset Action) (harms : arms.Nonempty)
357     (eta gamma : Real)
358     (score : (n : Nat) -> History.FinitePairHistory Action Loss n ->
359       Action -> Real)
360     (hscore : MeasurableFiniteHistoryScore arms score)
361     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
362     (environment : Thompson.MeasurableHistoryEnvironment Env Action Loss)
363     (n : Nat) :
364     condDistrib
365         (fun sample : Env × ((k : Nat) -> Action × Loss) =>
366           (sample.2 (n + 1)).1)
367         (fun sample => Preorder.frestrictLe n sample.2)
368         (prior ⊗ₘ exploredTrajectoryKernel arms harms eta gamma score hscore
369           hgamma_nonneg hgamma_le_one environment) =ᵐ[
370       (prior ⊗ₘ exploredTrajectoryKernel arms harms eta gamma score hscore
371         hgamma_nonneg hgamma_le_one environment).map
372           (fun sample => Preorder.frestrictLe n sample.2)]
373       finiteActionKernel arms
374         (exploredHistoryDistribution arms eta gamma (score n))
375         (exploredHistoryDistributionSource arms harms eta gamma score hscore
376           hgamma_nonneg hgamma_le_one n) := by
377   simpa [exploredTrajectoryKernel, exploredHistoryAlgorithm_policy] using
378     (Thompson.canonicalMeasurableEnvironmentTrajectoryMeasure_condDistrib_action
379       prior
380       (exploredHistoryAlgorithm arms harms eta gamma score hscore
381         hgamma_nonneg hgamma_le_one)
382       environment n)
