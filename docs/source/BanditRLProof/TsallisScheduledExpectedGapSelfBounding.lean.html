4 # Expected-gap self-bounding for scheduled half-Tsallis FTRL
5 
6 This module replaces the samplewise fixed-gap premise by the stochastic
7 first-moment law actually needed by self-bounding.  For every time and
8 suboptimal arm, the probability-weighted predictable loss difference has
9 expectation `gap action * E[p_t(action)]`.
10 -/

19 /-- The per-time, per-suboptimal-arm first-moment law needed by stochastic
20 self-bounding.  It is deliberately finer than the final summed regret law so
21 that later conditional-expectation producers can discharge it one coordinate
22 at a time. -/
23 def HasScheduledExpectedGapLaw
24     {Env : Type u} {Action : Type v}
25     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
26     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
27     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
28     (loss : Exp3.PredictableLossVector Env Action)
29     (best : Action) (gap : Action -> Real) (horizon : Nat) : Prop :=
30   ∀ t, t <= horizon -> ∀ action, action ∈ arms.erase best ->
31     integral mu (fun sample =>
32         sampledScheduledHalfTsallisProbabilityAtTime
33             arms harms eta t sample action *
34           (Exp3.predictableLossAt loss t sample action -
35             Exp3.predictableLossAt loss t sample best)) =
36       gap action * sampledScheduledHalfTsallisExpectedProbabilityAt
37         mu arms harms eta t action
38 
39 /-- One probability-weighted predictable loss difference is integrable. -/
40 theorem integrable_sampledScheduledHalfTsallisProbability_mul_predictableLossDiffAt
41     {Env : Type u} {Action : Type v}
42     [MeasurableSpace Env]
43     [MeasurableSpace Action] [MeasurableSingletonClass Action]
44     [DecidableEq Action]
45     (mu : Measure (Env × ((k : Nat) -> Action × Real))) [IsFiniteMeasure mu]
46     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
47     (loss : Exp3.PredictableLossVector Env Action)
48     (best action : Action) (haction : action ∈ arms) (t : Nat) :
49     Integrable (fun sample =>
50       sampledScheduledHalfTsallisProbabilityAtTime
51           arms harms eta t sample action *
52         (Exp3.predictableLossAt loss t sample action -
53           Exp3.predictableLossAt loss t sample best)) mu := by
54   have hp := integrable_sampledScheduledHalfTsallisProbabilityAtTime
55     mu arms harms eta t action haction
56   have hdiffMeas :=
57     (Exp3.measurable_predictableLossAt loss t action).sub
58       (Exp3.measurable_predictableLossAt loss t best)
59   refine hp.mul_bdd (c := 1) hdiffMeas.aestronglyMeasurable ?_
60   exact Filter.Eventually.of_forall fun sample => by
61     have hmem (candidate : Action) :
62         Exp3.predictableLossAt loss t sample candidate ∈ Set.Icc (0 : Real) 1 := by
63       cases t with
64       | zero => exact loss.initial_mem_unitInterval sample.1 candidate
65       | succ n =>
66           exact loss.successor_mem_unitInterval n sample.1
67             (Preorder.frestrictLe n sample.2) candidate
68     have ha := hmem action
69     have hb := hmem best
70     rcases ha with ⟨ha0, ha1⟩
71     rcases hb with ⟨hb0, hb1⟩
72     rw [Real.norm_eq_abs]
73     exact abs_le.2 ⟨by linarith, by linarith⟩

75 /-- Pathwise scheduled regret against a best-arm point mass is the finite sum
76 of probability-weighted predictable loss differences over suboptimal arms. -/
77 theorem sampledScheduledHalfTsallisPredictableEnvironmentRegret_pointMass_eq_weightedLossGapMass
78     {Env : Type u} {Action : Type v}
79     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
80     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
81     (loss : Exp3.PredictableLossVector Env Action)
82     {best : Action} (hbest : best ∈ arms) (horizon : Nat)
83     (sample : Env × ((k : Nat) -> Action × Real)) :
84     sampledScheduledHalfTsallisPredictableEnvironmentRegret
85         arms harms eta loss (pointMass best) horizon sample =
86       (Finset.range (horizon + 1)).sum (fun t =>
87         (arms.erase best).sum (fun action =>
88           sampledScheduledHalfTsallisProbabilityAtTime
89               arms harms eta t sample action *
90             (Exp3.predictableLossAt loss t sample action -
91               Exp3.predictableLossAt loss t sample best))) := by
92   unfold sampledScheduledHalfTsallisPredictableEnvironmentRegret
93   apply Finset.sum_congr rfl
94   intro t _ht
95   rw [linearLoss_sub_pointMass_eq_gapMass
96     arms hbest
97     (sampledScheduledHalfTsallisProbabilityAtTime arms harms eta t sample)
98     (Exp3.predictableLossAt loss t sample)
99     (fun action => Exp3.predictableLossAt loss t sample action -
100       Exp3.predictableLossAt loss t sample best)
101     (finiteSimplex_sampledScheduledHalfTsallisProbabilityAtTime
102       arms harms eta t sample)
103     (fun _action _haction => rfl)]
104   rw [← Finset.sum_erase_add _ _ hbest]
105   simp
