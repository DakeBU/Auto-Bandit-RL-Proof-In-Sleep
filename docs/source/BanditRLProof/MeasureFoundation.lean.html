4 /-!
5 # Minimal measurable action-event foundations
6 
7 This module starts the probability-facing layer with measurable events only.
8 It deliberately avoids measure, integration, probability, filtration, and
9 concentration imports.
10 -/
11 
12 universe u v w
13 
14 namespace BanditRLProof
15 
16 section ActionEvents
17 
18 /--
19 If every time-indexed action random variable is measurable, then the event that
20 the action at a fixed time equals a fixed arm is measurable.
21 
22 This is the `MEAS-FIN-ACTION` canary.  The statement is more general than finite
23 actions: it only needs singleton measurability of the action space.
24 -/
25 theorem measurableSet_actionTrace_eval_eq
26     {Omega : Type u} {Action : Type v}
27     [MeasurableSpace Omega] [MeasurableSpace Action]
28     [MeasurableSingletonClass Action]
29     (action : Omega -> ActionTrace Action)
30     (hmeas : forall t : Nat, Measurable (fun omega : Omega => action omega t))
31     (a : Action) (t : Nat) :
32     MeasurableSet {omega : Omega | action omega t = a} := by
33   simpa [Set.preimage] using
34     ((hmeas t) (MeasurableSet.singleton a))
35 
36 /--
37 The indicator of a measurable action-equality event with a constant value is
38 measurable.
39 
40 This is the `MEAS-PULL-INDICATOR` bridge.  It remains scalar-agnostic so later
41 expectation work can choose the codomain deliberately.
42 -/
43 theorem measurable_actionTrace_eval_eq_indicator_const
44     {Omega : Type u} {Action : Type v} {Beta : Type w}
45     [MeasurableSpace Omega] [MeasurableSpace Action]
46     [MeasurableSingletonClass Action]
47     [MeasurableSpace Beta] [Zero Beta]
48     (action : Omega -> ActionTrace Action)
49     (hmeas : forall t : Nat, Measurable (fun omega : Omega => action omega t))
50     (a : Action) (t : Nat) (c : Beta) :
51     Measurable
52       (({omega : Omega | action omega t = a} : Set Omega).indicator
53         (fun _ : Omega => c)) := by
54   exact Measurable.indicator measurable_const
55     (measurableSet_actionTrace_eval_eq action hmeas a t)
56 
57 /--
58 The selected-reward contribution for a fixed action event is measurable when
59 the action and reward traces are timewise measurable.
60 
61 This is the `MEAS-REWARD` bridge.  It deliberately stays at the
62 measurability layer and does not choose an expectation or scalar algebra route.
63 -/
64 theorem measurable_actionTrace_eval_eq_indicator_reward
65     {Omega : Type u} {Action : Type v} {Reward : Type w}
66     [MeasurableSpace Omega] [MeasurableSpace Action]
67     [MeasurableSingletonClass Action]
68     [MeasurableSpace Reward] [Zero Reward]
69     (action : Omega -> ActionTrace Action)
70     (reward : Omega -> RewardTrace Reward)
71     (haction : forall t : Nat, Measurable (fun omega : Omega => action omega t))
72     (hreward : forall t : Nat, Measurable (fun omega : Omega => reward omega t))
73     (a : Action) (t : Nat) :
74     Measurable
75       (({omega : Omega | action omega t = a} : Set Omega).indicator
76         (fun omega : Omega => reward omega t)) := by
77   exact Measurable.indicator (hreward t)
78     (measurableSet_actionTrace_eval_eq action haction a t)
