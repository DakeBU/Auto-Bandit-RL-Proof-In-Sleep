3 /-!
4 # Sampled importance-weighted EXP3 history scores
5 
6 This module closes the input-score boundary of `Exp3RecursiveTrajectory` for
7 real-valued observed losses.  The score at an inclusive finite history is the
8 previous score plus the importance-weighted loss of the newly observed pair.
9 The probability in that increment is exactly the exploration-mixed policy
10 computed from the preceding score and history prefix.
11 
12 The resulting theorem constructs the concrete recursive EXP3 trajectory and
13 identifies every successor action's conditional law.  Expected-regret assembly
14 and parameter optimization remain downstream.
15 -/

24 /-- Remove the newest coordinate from an inclusive successor pair history. -/
25 def previousPairHistory {Action : Type u} {n : Nat}
26     (history : History.FinitePairHistory Action Real (n + 1)) :
27     History.FinitePairHistory Action Real n :=
28   fun i => history
29     ⟨i.1, Finset.mem_Iic.mpr
30       ((Finset.mem_Iic.mp i.2).trans (Nat.le_succ n))⟩
31 
32 theorem measurable_previousPairHistory
33     {Action : Type u} [MeasurableSpace Action] {n : Nat} :
34     Measurable
35       (previousPairHistory (Action := Action) (n := n)) := by
36   refine measurable_pi_lambda _ ?_
37   intro i
38   let oldIndex : Finset.Iic (n + 1) :=
39     ⟨i.1, Finset.mem_Iic.mpr
40       ((Finset.mem_Iic.mp i.2).trans (Nat.le_succ n))⟩
41   exact measurable_pi_apply oldIndex

43 /--
44 Measurability of one importance-weighted coordinate when only the sampled
45 action and its scalar observed loss are available.
46 -/
47 theorem measurable_observedImportanceWeightedLoss
48     {History : Type*} {Action : Type u}
49     [MeasurableSpace History] [MeasurableSpace Action]
50     [MeasurableSingletonClass Action] [DecidableEq Action]
51     (prob : History -> Action -> Real)
52     (chosen : History -> Action) (observedLoss : History -> Real)
53     (action : Action)
54     (hprob : Measurable (fun history => prob history action))
55     (hchosen : Measurable chosen) (hloss : Measurable observedLoss) :
56     Measurable (fun history =>
57       importanceWeightedLoss (prob history)
58         (fun _ => observedLoss history) (chosen history) action) := by
59   unfold importanceWeightedLoss
60   refine Measurable.ite ?_ (hloss.div hprob) measurable_const
61   simpa only [Set.mem_setOf_eq] using
62     hchosen (measurableSet_singleton action)

64 /--
65 The cumulative sampled importance-weighted loss through an inclusive history.
66 
67 At time zero the estimator uses the initial action law.  At time `n + 1`, it
68 uses the exploration-mixed law generated from the score on the prefix through
69 `n`, exactly matching `HistoryAlgorithm.policy n`.
70 -/
71 noncomputable def sampledHistoryScore
72     {Action : Type u} [DecidableEq Action]
73     (arms : Finset Action) (eta gamma : Real) :
74     (n : Nat) -> History.FinitePairHistory Action Real n -> Action -> Real
75   | 0, history, action =>
76       importanceWeightedLoss
77         (initialExploredDistribution arms eta gamma)
78         (fun _ => (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).2)
79         (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 action
80   | n + 1, history, action =>
81       let previous := previousPairHistory history
82       sampledHistoryScore arms eta gamma n previous action +
83         importanceWeightedLoss
84           (exploredHistoryDistribution arms eta gamma
85             (sampledHistoryScore arms eta gamma n) previous)
86           (fun _ =>
87             (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).2)
88           (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).1 action

121 theorem measurable_sampledHistoryScore
122     {Action : Type u}
123     [MeasurableSpace Action] [MeasurableSingletonClass Action]
124     [DecidableEq Action]
125     (arms : Finset Action) (eta gamma : Real) :
126     forall n action, action ∈ arms ->
127       Measurable (fun history : History.FinitePairHistory Action Real n =>
128         sampledHistoryScore arms eta gamma n history action) := by
129   intro n
130   induction n with
131   | zero =>
132       intro action _haction
133       let zeroIndex : Finset.Iic 0 :=
134         ⟨0, Finset.mem_Iic.mpr le_rfl⟩
135       have hchosen : Measurable
136           (fun history : History.FinitePairHistory Action Real 0 =>
137             (history zeroIndex).1) :=
138         measurable_fst.comp (measurable_pi_apply zeroIndex)
139       have hloss : Measurable
140           (fun history : History.FinitePairHistory Action Real 0 =>
141             (history zeroIndex).2) :=
142         measurable_snd.comp (measurable_pi_apply zeroIndex)
143       simpa [sampledHistoryScore, zeroIndex] using
144         (measurable_observedImportanceWeightedLoss
145           (fun _history : History.FinitePairHistory Action Real 0 =>
146             initialExploredDistribution arms eta gamma)
147           (fun history => (history zeroIndex).1)
148           (fun history => (history zeroIndex).2)
149           action measurable_const hchosen hloss)
150   | succ n ih =>
151       intro action haction
152       let lastIndex : Finset.Iic (n + 1) :=
153         ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩
154       have hprevious : Measurable
155           (previousPairHistory (Action := Action) (n := n)) :=
156         measurable_previousPairHistory
157       have hscore : Measurable
158           (fun history : History.FinitePairHistory Action Real (n + 1) =>
159             sampledHistoryScore arms eta gamma n
160               (previousPairHistory history) action) :=
161         (ih action haction).comp hprevious
162       have hprob : Measurable
163           (fun history : History.FinitePairHistory Action Real (n + 1) =>
164             exploredHistoryDistribution arms eta gamma
165               (sampledHistoryScore arms eta gamma n)
166               (previousPairHistory history) action) := by
167         exact
168           (measurable_exploredHistoryDistribution arms eta gamma
169             (sampledHistoryScore arms eta gamma n)
170             (fun selected hselected => ih selected hselected)
171             action haction).comp hprevious
172       have hchosen : Measurable
173           (fun history : History.FinitePairHistory Action Real (n + 1) =>
174             (history lastIndex).1) :=
175         measurable_fst.comp (measurable_pi_apply lastIndex)
176       have hloss : Measurable
177           (fun history : History.FinitePairHistory Action Real (n + 1) =>
178             (history lastIndex).2) :=
179         measurable_snd.comp (measurable_pi_apply lastIndex)
180       have hincrement : Measurable
181           (fun history : History.FinitePairHistory Action Real (n + 1) =>
182             importanceWeightedLoss
183               (exploredHistoryDistribution arms eta gamma
184                 (sampledHistoryScore arms eta gamma n)
185                 (previousPairHistory history))
186               (fun _ => (history lastIndex).2)
187               (history lastIndex).1 action) :=
188         measurable_observedImportanceWeightedLoss
189           (fun history =>
190             exploredHistoryDistribution arms eta gamma
191               (sampledHistoryScore arms eta gamma n)
192               (previousPairHistory history))
193           (fun history => (history lastIndex).1)
194           (fun history => (history lastIndex).2)
195           action hprob hchosen hloss
196       simpa [sampledHistoryScore, lastIndex] using hscore.add hincrement

292 /--
293 Every successor action of the concrete sampled-loss EXP3 trajectory has the
294 exploration-mixed law generated from its recursively accumulated
295 importance-weighted score.
296 -/
297 theorem sampledImportanceWeightedTrajectoryMeasure_condDistrib_action
298     {Env : Type w} {Action : Type u}
299     [MeasurableSpace Env] [StandardBorelSpace Env]
300     [MeasurableSpace Action] [MeasurableSingletonClass Action]
301     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
302     (prior : Measure Env) [IsFiniteMeasure prior]
303     (arms : Finset Action) (harms : arms.Nonempty)
304     (eta gamma : Real)
305     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
306     (environment : Thompson.MeasurableHistoryEnvironment Env Action Real)
307     (n : Nat) :
308     condDistrib
309         (fun sample : Env × ((k : Nat) -> Action × Real) =>
310           (sample.2 (n + 1)).1)
311         (fun sample => Preorder.frestrictLe n sample.2)
312         (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
313           eta gamma hgamma_nonneg hgamma_le_one environment) =ᵐ[
314       (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
315         eta gamma hgamma_nonneg hgamma_le_one environment).map
316           (fun sample => Preorder.frestrictLe n sample.2)]
317       finiteActionKernel arms
318         (sampledHistoryDistribution arms eta gamma n)
319         (exploredHistoryDistributionSource arms harms eta gamma
320           (sampledHistoryScore arms eta gamma)
321           (measurableFiniteHistoryScore_sampledHistoryScore arms eta gamma)
322           hgamma_nonneg hgamma_le_one n) := by
323   simpa [sampledImportanceWeightedTrajectoryKernel,
324     sampledHistoryDistribution] using
325     (exploredTrajectoryMeasure_condDistrib_action prior arms harms eta gamma
326       (sampledHistoryScore arms eta gamma)
327       (measurableFiniteHistoryScore_sampledHistoryScore arms eta gamma)
328       hgamma_nonneg hgamma_le_one environment n)
