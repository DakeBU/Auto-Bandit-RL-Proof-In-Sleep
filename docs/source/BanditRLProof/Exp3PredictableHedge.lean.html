3 /-!
4 # Almost-sure Hedge control for predictable sampled EXP3
5 
6 This module discharges the pathwise nonnegative-feedback premise of
7 `Exp3SampledHedge` from the generated predictable `[0,1]` reward law.  It first
8 aggregates the time-zero and successor reward identifications into one
9 finite-horizon almost-sure event, then applies the concrete pathwise Hedge
10 bound on that event.
11 
12 The endpoint remains an almost-sure inequality.  Integrating it and comparing
13 the pure Hedge distribution with the exploration-mixed sampling distribution
14 are separate downstream steps.
15 -/

24 /-- Every observed scalar reward is nonnegative almost surely under the
25 generated predictable sampled-EXP3 trajectory law. -/
26 theorem sampledPredictableTrajectoryMeasure_reward_nonneg_ae
27     {Env : Type u} {Action : Type v}
28     [MeasurableSpace Env] [StandardBorelSpace Env]
29     [MeasurableSpace Action] [MeasurableSingletonClass Action]
30     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
31     (prior : Measure Env) [IsFiniteMeasure prior]
32     (arms : Finset Action) (harms : arms.Nonempty)
33     (eta gamma : Real)
34     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
35     (loss : PredictableLossVector Env Action) (t : Nat) :
36     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
37       eta gamma hgamma_nonneg hgamma_le_one loss.environment
38     ∀ᵐ sample ∂mu, 0 <= (sample.2 t).2 := by
39   dsimp only
40   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
41     eta gamma hgamma_nonneg hgamma_le_one loss.environment
42   cases t with
43   | zero =>
44       have hreward :
45           (fun sample : Env × ((k : Nat) -> Action × Real) =>
46             (sample.2 0).2) =ᵐ[mu]
47           (fun sample => loss.initial sample.1 (sample.2 0).1) := by
48         simpa [mu, sampledImportanceWeightedTrajectoryKernel] using
49           (canonicalPredictableTrajectoryMeasure_reward_zero_eq_initialLoss_ae
50             prior (sampledImportanceWeightedHistoryAlgorithm arms harms eta gamma
51               hgamma_nonneg hgamma_le_one) loss)
52       filter_upwards [hreward] with sample hsample
53       rw [hsample]
54       exact (loss.initial_mem_unitInterval sample.1 (sample.2 0).1).1
55   | succ n =>
56       have hreward :
57           (fun sample : Env × ((k : Nat) -> Action × Real) =>
58             (sample.2 (n + 1)).2) =ᵐ[mu]
59           (fun sample => loss.successor n sample.1
60             (Preorder.frestrictLe n sample.2) (sample.2 (n + 1)).1) := by
61         simpa [mu] using
62           (sampledPredictableTrajectoryMeasure_reward_eq_successorLoss_ae
63             prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss n)
64       filter_upwards [hreward] with sample hsample
65       rw [hsample]
66       exact (loss.successor_mem_unitInterval n sample.1
67         (Preorder.frestrictLe n sample.2) (sample.2 (n + 1)).1).1

69 /-- One common almost-sure event supplies reward nonnegativity at every time
70 strictly before a finite horizon. -/
71 theorem sampledPredictableTrajectoryMeasure_finiteHorizon_reward_nonneg_ae
72     {Env : Type u} {Action : Type v}
73     [MeasurableSpace Env] [StandardBorelSpace Env]
74     [MeasurableSpace Action] [MeasurableSingletonClass Action]
75     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
76     (prior : Measure Env) [IsFiniteMeasure prior]
77     (arms : Finset Action) (harms : arms.Nonempty)
78     (eta gamma : Real)
79     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
80     (loss : PredictableLossVector Env Action) (horizon : Nat) :
81     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
82       eta gamma hgamma_nonneg hgamma_le_one loss.environment
83     ∀ᵐ sample ∂mu, ∀ t, t < horizon -> 0 <= (sample.2 t).2 := by
84   dsimp only
85   rw [ae_all_iff]
86   intro t
87   have hnonneg := sampledPredictableTrajectoryMeasure_reward_nonneg_ae
88     prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss t
89   dsimp only at hnonneg
90   filter_upwards [hnonneg] with sample hsample
91   intro _ht
92   exact hsample

94 /-- The concrete finite-horizon Hedge inequality holds almost surely on the
95 generated predictable sampled-EXP3 trajectory. -/
96 theorem sampledPredictableTrajectoryMeasure_hedge_regret_le_ae
97     {Env : Type u} {Action : Type v}
98     [MeasurableSpace Env] [StandardBorelSpace Env]
99     [MeasurableSpace Action] [MeasurableSingletonClass Action]
100     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
101     (prior : Measure Env) [IsFiniteMeasure prior]
102     (arms : Finset Action) (harms : arms.Nonempty)
103     (eta gamma : Real) (heta : 0 < eta)
104     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
105     (loss : PredictableLossVector Env Action) (horizon : Nat)
106     (comparator : Action) (hcomparator : comparator ∈ arms) :
107     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
108       eta gamma hgamma_nonneg hgamma_le_one loss.environment
109     ∀ᵐ sample ∂mu,
110       (Finset.range horizon).sum (fun t =>
111           mixedLoss arms eta
112             (sampledTrajectoryObservedLoss arms eta gamma sample) t) -
113           cumulativeLoss (sampledTrajectoryObservedLoss arms eta gamma sample)
114             horizon comparator <=
115         Real.log arms.card / eta +
116           eta * (Finset.range horizon).sum (fun t =>
117             mixedSquaredLoss arms eta
118               (sampledTrajectoryObservedLoss arms eta gamma sample) t) := by
119   dsimp only
120   have hnonneg :=
121     sampledPredictableTrajectoryMeasure_finiteHorizon_reward_nonneg_ae
122       prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon
123   dsimp only at hnonneg
124   filter_upwards [hnonneg] with sample hsample
125   exact sampledTrajectory_hedge_regret_le arms harms eta gamma heta
126     hgamma_nonneg hgamma_le_one sample horizon hsample comparator hcomparator
