4 # Constrained quadratic optimization for self-bounding Tsallis analysis
5 
6 This module gives a rigorous finite-sum version of the one-round optimization
7 used after lambda interpolation.  In contrast to the informal paper lemma,
8 the quadratic coefficients are required to be strictly positive and the
9 simplex-derived square-root mass constraint is explicit.
10 -/

17 /-- A downward quadratic is bounded by its unconstrained vertex value. -/
18 theorem linear_sub_quadratic_le_sq_div_four
19     (b x c : Real) (hc : 0 < c) :
20     b * x - c * x ^ 2 ≤ b ^ 2 / (4 * c) := by
21   rw [le_div_iff₀ (by positivity : 0 < 4 * c)]
22   nlinarith [sq_nonneg (2 * c * x - b)]
23 
24 /-- Coordinatewise completion of squares gives the unconstrained finite-sum
25 branch of the paper's quadratic optimization. -/
26 theorem sum_linear_sub_quadratic_le_unconstrained
27     {Index : Type u} [DecidableEq Index]
28     (indices : Finset Index) (b : Real) (c x : Index → Real)
29     (hc : ∀ i ∈ indices, 0 < c i) :
30     indices.sum (fun i => b * x i - c i * x i ^ 2) ≤
31       b ^ 2 / 4 * indices.sum (fun i => 1 / c i) := by
32   calc
33     indices.sum (fun i => b * x i - c i * x i ^ 2) ≤
34         indices.sum (fun i => b ^ 2 / (4 * c i)) := by
35       apply Finset.sum_le_sum
36       intro i hi
37       exact linear_sub_quadratic_le_sq_div_four b (x i) (c i) (hc i hi)
38     _ = b ^ 2 / 4 * indices.sum (fun i => 1 / c i) := by
39       rw [Finset.mul_sum]
40       apply Finset.sum_congr rfl
41       intro i hi
42       field_simp [ne_of_gt (hc i hi)]

44 /-- If the unconstrained vertex lies beyond a finite `sum x ≤ M` constraint,
45 shift the common linear coefficient and apply completion of squares to obtain
46 the active-constraint branch. -/
47 theorem sum_linear_sub_quadratic_le_of_sum_le
48     {Index : Type u} [DecidableEq Index]
49     (indices : Finset Index) (b M : Real) (c x : Index → Real)
50     (hc : ∀ i ∈ indices, 0 < c i)
51     (hreciprocal : 0 < indices.sum (fun i => 1 / c i))
52     (hxsum : indices.sum x ≤ M)
53     (hthreshold :
54       2 * M ≤ b * indices.sum (fun i => 1 / c i)) :
55     indices.sum (fun i => b * x i - c i * x i ^ 2) ≤
56       b * M - M ^ 2 / indices.sum (fun i => 1 / c i) := by
57   let reciprocal := indices.sum (fun i => 1 / c i)
58   let shifted := 2 * M / reciprocal
59   let residual := b - shifted
60   have hresidual : 0 ≤ residual := by
61     dsimp [residual, shifted, reciprocal]
62     rw [sub_nonneg, div_le_iff₀ hreciprocal]
63     simpa [mul_comm] using hthreshold
64   have hvertex :
65       indices.sum (fun i => shifted * x i - c i * x i ^ 2) ≤
66         shifted ^ 2 / 4 * reciprocal := by
67     simpa [reciprocal] using
68       (sum_linear_sub_quadratic_le_unconstrained
69         indices shifted c x hc)
70   calc
71     indices.sum (fun i => b * x i - c i * x i ^ 2) =
72         residual * indices.sum x +
73           indices.sum (fun i => shifted * x i - c i * x i ^ 2) := by
74       rw [Finset.mul_sum, ← Finset.sum_add_distrib]
75       apply Finset.sum_congr rfl
76       intro i hi
77       dsimp [residual]
78       ring
79     _ ≤ residual * M + shifted ^ 2 / 4 * reciprocal :=
80       add_le_add (mul_le_mul_of_nonneg_left hxsum hresidual) hvertex
81     _ = b * M - M ^ 2 / indices.sum (fun i => 1 / c i) := by
82       dsimp [residual, shifted, reciprocal]
83       field_simp [ne_of_gt hreciprocal]
84       ring

86 /-- Positive coefficients on a nonempty finite set have a positive reciprocal
87 sum. -/
88 theorem sum_inv_pos_of_nonempty
89     {Index : Type u} [DecidableEq Index]
90     (indices : Finset Index) (hindices : indices.Nonempty)
91     (c : Index → Real) (hc : ∀ i ∈ indices, 0 < c i) :
92     0 < indices.sum (fun i => 1 / c i) := by
93   rcases hindices with ⟨i, hi⟩
94   exact Finset.sum_pos'
95     (fun j hj => (one_div_pos.mpr (hc j hj)).le)
96     ⟨i, hi, one_div_pos.mpr (hc i hi)⟩

98 /-- A finite simplex point has suboptimal square-root mass at most the square
99 root of the number of suboptimal coordinates. -/
100 theorem sum_erase_sqrt_le_sqrt_card
101     {Action : Type u} [DecidableEq Action]
102     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
103     (probability : Action → Real)
104     (hprobability : FTRL.finiteSimplex arms probability) :
105     (arms.erase best).sum (fun action => Real.sqrt (probability action)) ≤
106       Real.sqrt ((arms.erase best).card : Real) := by
107   let suboptimal := arms.erase best
108   have hprobNonneg : ∀ action,
109       0 ≤ if action ∈ suboptimal then probability action else 0 := by
110     intro action
111     by_cases haction : action ∈ suboptimal
112     · rw [if_pos haction]
113       exact hprobability.1 action (Finset.mem_of_mem_erase haction)
114     · simp [haction]
115   have hsqrt := Real.sum_sqrt_mul_sqrt_le suboptimal
116     hprobNonneg (fun _ => (zero_le_one : (0 : Real) ≤ 1))
117   have hsumProbability : suboptimal.sum probability ≤ 1 := by
118     have hsplit := Finset.add_sum_erase arms probability hbest
119     rw [hprobability.2] at hsplit
120     have hbestNonneg := hprobability.1 best hbest
121     dsimp [suboptimal]
122     linarith
123   have hsumIf :
124       suboptimal.sum (fun action =>
125         if action ∈ suboptimal then probability action else 0) =
126         suboptimal.sum probability := by
127     apply Finset.sum_congr rfl
128     intro action haction
129     simp [haction]
130   calc
131     suboptimal.sum (fun action => Real.sqrt (probability action)) =
132         suboptimal.sum (fun action =>
133           Real.sqrt (if action ∈ suboptimal then probability action else 0) *
134             Real.sqrt (1 : Real)) := by
135       apply Finset.sum_congr rfl
136       intro action haction
137       simp [haction]
138     _ ≤ Real.sqrt
139           (suboptimal.sum (fun action =>
140             if action ∈ suboptimal then probability action else 0)) *
141         Real.sqrt (suboptimal.sum (fun _ => (1 : Real))) :=
142       hsqrt
143     _ = Real.sqrt (suboptimal.sum probability) *
144         Real.sqrt (suboptimal.card : Real) := by
145       rw [hsumIf]
146       simp
147     _ ≤ Real.sqrt 1 * Real.sqrt (suboptimal.card : Real) := by
148       gcongr
149     _ = Real.sqrt (suboptimal.card : Real) := by simp

151 /-- The unconstrained one-round branch after substituting
152 `x action = sqrt (probability action)` and
153 `c action = lambda * gap action`. -/
154 theorem sum_erase_sqrt_probability_sub_gap_le_unconstrained
155     {Action : Type u} [DecidableEq Action]
156     (arms : Finset Action) {best : Action}
157     (probability gap : Action → Real)
158     (hprobability : FTRL.finiteSimplex arms probability)
159     (hgap : ∀ action ∈ arms.erase best, 0 < gap action)
160     (b lambda : Real) (hlambda : 0 < lambda) :
161     (arms.erase best).sum (fun action =>
162         b * Real.sqrt (probability action) -
163           lambda * gap action * probability action) ≤
164       b ^ 2 / 4 * (arms.erase best).sum (fun action =>
165         1 / (lambda * gap action)) := by
166   have hc : ∀ action ∈ arms.erase best,
167       0 < lambda * gap action := by
168     intro action haction
169     exact mul_pos hlambda (hgap action haction)
170   calc
171     (arms.erase best).sum (fun action =>
172         b * Real.sqrt (probability action) -
173           lambda * gap action * probability action) =
174         (arms.erase best).sum (fun action =>
175           b * Real.sqrt (probability action) -
176             (lambda * gap action) *
177               Real.sqrt (probability action) ^ 2) := by
178       apply Finset.sum_congr rfl
179       intro action haction
180       rw [Real.sq_sqrt
181         (hprobability.1 action (Finset.mem_of_mem_erase haction))]
182     _ ≤ b ^ 2 / 4 * (arms.erase best).sum (fun action =>
183         1 / (lambda * gap action)) :=
184       sum_linear_sub_quadratic_le_unconstrained
185         (arms.erase best) b (fun action => lambda * gap action)
186           (fun action => Real.sqrt (probability action)) hc

188 /-- On the active-constraint branch, finite-simplex square-root mass sharpens
189 the unconstrained coordinatewise bound by the common mass constraint. -/
190 theorem sum_erase_sqrt_probability_sub_gap_le_of_threshold
191     {Action : Type u} [DecidableEq Action]
192     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
193     (probability gap : Action → Real)
194     (hprobability : FTRL.finiteSimplex arms probability)
195     (hsuboptimal : (arms.erase best).Nonempty)
196     (hgap : ∀ action ∈ arms.erase best, 0 < gap action)
197     (b lambda : Real) (hlambda : 0 < lambda)
198     (hthreshold :
199       2 * Real.sqrt ((arms.erase best).card : Real) ≤
200         b * (arms.erase best).sum (fun action =>
201           1 / (lambda * gap action))) :
202     (arms.erase best).sum (fun action =>
203         b * Real.sqrt (probability action) -
204           lambda * gap action * probability action) ≤
205       b * Real.sqrt ((arms.erase best).card : Real) -
206         (arms.erase best).card /
207           (arms.erase best).sum (fun action =>
208             1 / (lambda * gap action)) := by
209   have hc : ∀ action ∈ arms.erase best,
210       0 < lambda * gap action := by
211     intro action haction
212     exact mul_pos hlambda (hgap action haction)
213   have hreciprocal :
214       0 < (arms.erase best).sum (fun action =>
215         1 / (lambda * gap action)) :=
216     sum_inv_pos_of_nonempty (arms.erase best) hsuboptimal
217       (fun action => lambda * gap action) hc
218   have hbound := sum_linear_sub_quadratic_le_of_sum_le
219     (arms.erase best) b (Real.sqrt ((arms.erase best).card : Real))
220       (fun action => lambda * gap action)
221       (fun action => Real.sqrt (probability action))
222       hc hreciprocal
223       (sum_erase_sqrt_le_sqrt_card arms hbest probability hprobability)
224       hthreshold
225   calc
226     (arms.erase best).sum (fun action =>
227         b * Real.sqrt (probability action) -
228           lambda * gap action * probability action) =
229         (arms.erase best).sum (fun action =>
230           b * Real.sqrt (probability action) -
231             (lambda * gap action) *
232               Real.sqrt (probability action) ^ 2) := by
233       apply Finset.sum_congr rfl
234       intro action haction
235       rw [Real.sq_sqrt
236         (hprobability.1 action (Finset.mem_of_mem_erase haction))]
237     _ ≤ b * Real.sqrt ((arms.erase best).card : Real) -
238         Real.sqrt ((arms.erase best).card : Real) ^ 2 /
239           (arms.erase best).sum (fun action =>
240             1 / (lambda * gap action)) := hbound
241     _ = b * Real.sqrt ((arms.erase best).card : Real) -
242         (arms.erase best).card /
243           (arms.erase best).sum (fun action =>
244             1 / (lambda * gap action)) := by
245       rw [Real.sq_sqrt (by positivity :
246         (0 : Real) ≤ ((arms.erase best).card : Real))]

248 /-- Generated expected-probability specialization of the unconstrained
249 one-round quadratic branch. All measure and Jensen obligations are discharged
250 by the existing finite-simplex expectation theorem. -/
251 theorem sampledScheduledHalfTsallisExpectedProbability_sum_le_unconstrained
252     {Env : Type u} {Action : Type*}
253     [MeasurableSpace Env]
254     [MeasurableSpace Action] [MeasurableSingletonClass Action]
255     [DecidableEq Action]
256     (mu : MeasureTheory.Measure (Env × ((k : Nat) → Action × Real)))
257     [MeasureTheory.IsProbabilityMeasure mu]
258     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat → Real)
259     (t : Nat) {best : Action} (gap : Action → Real)
260     (hgap : ∀ action ∈ arms.erase best, 0 < gap action)
261     (b lambda : Real) (hlambda : 0 < lambda) :
262     (arms.erase best).sum (fun action =>
263         b * Real.sqrt (sampledScheduledHalfTsallisExpectedProbabilityAt
264           mu arms harms eta t action) -
265         lambda * gap action *

267             mu arms harms eta t action) ≤
268       b ^ 2 / 4 * (arms.erase best).sum (fun action =>
