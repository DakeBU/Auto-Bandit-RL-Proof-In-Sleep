4 /-!
5 # Random estimator-square EXP3 high-probability regret
6 
7 The pathwise EXP3 assembly bounds the mixed importance-weighted estimator-square
8 sum by `|arms| * T / gamma`.  Its expectation is at most `|arms| * T`.
9 This module turns that expectation bound into a Markov tail and includes the
10 square event beside the two existing Bernstein confidence events.  The resulting
11 regret theorem removes the reciprocal exploration factor from the Hedge-square
12 budget, at the honest cost of a `1 / deltaSquare` failure allocation.
13 -/

21 /-- Finite-horizon sum of the probability-mixed squared importance-weighted
22 loss estimates observed on a sampled trajectory. -/
23 noncomputable def sampledObservedMixedSquaredSum
24     {Env : Type u} {Action : Type v} [DecidableEq Action]
25     (arms : Finset Action) (eta gamma : Real) (horizon : Nat)
26     (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
27   (Finset.range horizon).sum (fun t =>

35     (hgamma_le_one : gamma <= 1) (t : Nat)
36     (sample : Env × ((k : Nat) -> Action × Real)) :
37     0 <= observedMixedSquaredImportanceWeightedLossAt
38       arms eta gamma t sample := by
39   unfold observedMixedSquaredImportanceWeightedLossAt
40   unfold mixedSquaredImportanceWeightedLoss
41   exact Finset.sum_nonneg fun action _haction =>
42     mul_nonneg
43       (sampledTrajectoryProbabilityAt_nonneg arms harms eta gamma
44         hgamma_nonneg hgamma_le_one t sample action)

105 /-- Markov tail for the random finite-horizon estimator-square sum.  Its
106 threshold is `|arms| * T / deltaSquare`, with no reciprocal exploration-rate
107 factor. -/
108 theorem sampledPredictableObservedMixedSquared_sum_tail_markov
109     {Env : Type u} {Action : Type v}
110     [MeasurableSpace Env] [StandardBorelSpace Env]
111     [MeasurableSpace Action] [MeasurableSingletonClass Action]
112     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
113     (prior : Measure Env) [IsProbabilityMeasure prior]
114     (arms : Finset Action) (harms : arms.Nonempty)
115     (eta gamma : Real) (hgamma_pos : 0 < gamma)
116     (hgamma_le_one : gamma <= 1)
117     (loss : PredictableLossVector Env Action)
118     (horizon : Nat) (hhorizon : 0 < horizon)
119     (deltaSquare : Real) (hdeltaSquare : 0 < deltaSquare) :
120     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
121       eta gamma hgamma_pos.le hgamma_le_one loss.environment
122     mu {sample |
123         (arms.card : Real) * (horizon : Real) / deltaSquare <=
124           sampledObservedMixedSquaredSum arms eta gamma horizon sample} <=
125       ENNReal.ofReal deltaSquare := by
126   dsimp only
127   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
128     eta gamma hgamma_pos.le hgamma_le_one loss.environment
129   let secondMoment := sampledObservedMixedSquaredSum
130     (Env := Env) arms eta gamma horizon
131   let budget := (arms.card : Real) * (horizon : Real)
132   let scale := deltaSquare / budget
133   have hcard_pos : 0 < (arms.card : Real) := by
134     exact_mod_cast harms.card_pos
135   have hhorizon_real : 0 < (horizon : Real) := by exact_mod_cast hhorizon
136   have hbudget_pos : 0 < budget := by
137     dsimp [budget]
138     positivity
139   have hscale_pos : 0 < scale := by
140     dsimp [scale]
141     positivity
142   have hsecondIntegrable : Integrable secondMoment mu := by
143     have h := integrable_sampledPredictableObservedMixedSquaredSum
144       prior arms harms eta gamma hgamma_pos hgamma_le_one loss horizon
145     simpa [mu, secondMoment] using h
146   have hsecondNonneg : ∀ sample, 0 <= secondMoment sample := by
147     intro sample
148     exact sampledObservedMixedSquaredSum_nonneg arms harms eta gamma
149       hgamma_pos.le hgamma_le_one horizon sample
150   have hmean : integral mu secondMoment <= budget := by
151     have h := sampledPredictableObserved_finiteHorizon_secondMoment_integral_le_card_mul
152       prior arms harms eta gamma hgamma_pos hgamma_le_one loss horizon
153     simpa [mu, secondMoment, sampledObservedMixedSquaredSum, budget] using h
154   have hscaledIntegrable :
155       Integrable (fun sample => scale * secondMoment sample) mu :=
156     hsecondIntegrable.const_mul scale
157   have hscaledNonneg : ∀ᵐ sample ∂mu, 0 <= scale * secondMoment sample :=
158     Filter.Eventually.of_forall fun sample =>
159       mul_nonneg hscale_pos.le (hsecondNonneg sample)
160   have hnormalize : scale * (budget / deltaSquare) = 1 := by
161     dsimp [scale]
162     field_simp [hbudget_pos.ne', hdeltaSquare.ne']
163   have hscaleBudget : scale * budget = deltaSquare := by
164     dsimp [scale]
165     field_simp [hbudget_pos.ne']
166   have hmarkov :
167       mu {sample | budget / deltaSquare <= secondMoment sample} <=
168         ENNReal.ofReal (integral mu (fun sample => scale * secondMoment sample)) := by
169     apply hscaledIntegrable.measure_le_integral hscaledNonneg
170     intro sample hsample
171     rw [← hnormalize]
172     exact mul_le_mul_of_nonneg_left hsample hscale_pos.le
173   have hintegral :
174       integral mu (fun sample => scale * secondMoment sample) <= deltaSquare := by
175     rw [MeasureTheory.integral_const_mul]
176     calc
177       scale * integral mu secondMoment <= scale * budget :=
178         mul_le_mul_of_nonneg_left hmean hscale_pos.le
179       _ = deltaSquare := hscaleBudget
180   exact hmarkov.trans (ENNReal.ofReal_le_ofReal hintegral)

182 /-- Predictable regret budget with a caller-visible random-square failure
183 allocation and the two variance-sensitive confidence radii. -/
184 noncomputable def sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
185     {Action : Type v} (arms : Finset Action) (eta gamma : Real)
186     (horizon : Nat) (deltaSquare deltaConfidence : Real) : Real :=
187   Real.log arms.card / eta +
188     (eta * (1 / (1 - gamma))) *
189       ((arms.card : Real) * (horizon : Real) / deltaSquare) +
190     gamma * (horizon : Real) +
191     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
192       arms gamma horizon deltaConfidence +
193     sampledComparatorEstimatorBernsteinConfidenceRadius
194       arms gamma horizon deltaConfidence

196 /-- Generated predictable EXP3 regret with a Markov estimator-square event and
197 the two compiled Bernstein confidence events. -/
198 theorem sampledPredictable_randomSquareBernsteinHighProbabilityRegret_tail
199     {Env : Type u} {Action : Type v}
200     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
201     [MeasurableSpace Action] [MeasurableSingletonClass Action]
202     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
203     (prior : Measure Env) [IsProbabilityMeasure prior]
204     (arms : Finset Action) (harms : arms.Nonempty)
205     (eta gamma : Real) (heta : 0 < eta)
206     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
207     (loss : PredictableLossVector Env Action)
208     (comparator : Action) (hcomparator : comparator ∈ arms)
209     (horizon : Nat) (hhorizon : 0 < horizon)
210     (deltaSquare deltaConfidence : Real)
211     (hdeltaSquare : 0 < deltaSquare)
212     (hdeltaConfidence : 0 < deltaConfidence) :
213     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
214       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
215     mu {sample |
216         sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
217             arms eta gamma horizon deltaSquare deltaConfidence <=
218           (Finset.range horizon).sum (fun t =>
219               sampledTrajectoryExploredPredictableLossAt
220                 arms eta gamma loss t sample) -
221             (Finset.range horizon).sum (fun t =>
222               predictableLossAt loss t sample comparator)} <=
223       (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
224         ENNReal.ofReal deltaConfidence := by
225   dsimp only
226   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
227     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
228   let purePredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
229     (Finset.range horizon).sum (fun t =>
230       sampledTrajectoryPurePredictableLossAt arms eta gamma loss t sample)
231   let pureObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
232     (Finset.range horizon).sum (fun t =>
233       sampledTrajectoryPureObservedLossAt arms eta gamma t sample)
234   let comparatorObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
235     (Finset.range horizon).sum (fun t =>
236       observedImportanceWeightedLossAt arms eta gamma t sample comparator)
237   let comparatorTrue := fun sample : Env × ((k : Nat) -> Action × Real) =>
238     (Finset.range horizon).sum (fun t =>
239       predictableLossAt loss t sample comparator)
240   let exploredPredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
241     (Finset.range horizon).sum (fun t =>
242       sampledTrajectoryExploredPredictableLossAt arms eta gamma loss t sample)
243   let secondMoment := sampledObservedMixedSquaredSum
244     (Env := Env) arms eta gamma horizon
245   let secondMomentBudget :=
246     (arms.card : Real) * (horizon : Real) / deltaSquare
247   let pureRadius :=
248     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
249       arms gamma horizon deltaConfidence
250   let comparatorRadius := sampledComparatorEstimatorBernsteinConfidenceRadius
251     arms gamma horizon deltaConfidence
252   let secondMomentBad : Set (Env × ((k : Nat) -> Action × Real)) :=
253     {sample | secondMomentBudget <= secondMoment sample}
254   let pureBad : Set (Env × ((k : Nat) -> Action × Real)) :=
255     {sample | pureRadius <= purePredictable sample - pureObserved sample}
256   let comparatorBad : Set (Env × ((k : Nat) -> Action × Real)) :=
257     {sample | comparatorRadius <= comparatorObserved sample - comparatorTrue sample}
258   have hsecondTail : mu secondMomentBad <= ENNReal.ofReal deltaSquare := by
259     have h := sampledPredictableObservedMixedSquared_sum_tail_markov
260       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss horizon
261         hhorizon deltaSquare hdeltaSquare
262     dsimp only at h
263     simpa [mu, secondMomentBad, secondMomentBudget, secondMoment] using h
264   have hpureTail : mu pureBad <= ENNReal.ofReal deltaConfidence := by
265     have h := sampledPurePredictableMinusObserved_sum_tail_bernstein_delta
266       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss
267         horizon deltaConfidence hdeltaConfidence
268     dsimp only at h
269     simpa [mu, pureBad, pureRadius, purePredictable, pureObserved] using h
270   have hcomparatorTail : mu comparatorBad <= ENNReal.ofReal deltaConfidence := by
271     have h := sampledObservedComparatorEstimatorDeviation_sum_tail_bernstein_delta
272       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss comparator
273         hcomparator horizon deltaConfidence hdeltaConfidence
274     dsimp only at h
275     simpa [mu, comparatorBad, comparatorRadius, comparatorObserved,
276       comparatorTrue,
277       sampledTrajectoryObservedComparatorEstimatorDeviationAt] using h
278   have hhedge :=
279     sampledPredictableTrajectoryMeasure_hedge_exploredSecondMoment_le_ae
280       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss horizon
281         comparator hcomparator
282   dsimp only at hhedge
283   have hsubset : ∀ᵐ sample ∂mu,
284       sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
285           arms eta gamma horizon deltaSquare deltaConfidence <=
286         exploredPredictable sample - comparatorTrue sample ->
287       sample ∈ (secondMomentBad ∪ pureBad) ∪ comparatorBad := by
288     filter_upwards [hhedge] with sample hsampleHedge
289     intro hregret
290     by_cases hsecond : sample ∈ secondMomentBad
291     · exact Or.inl (Or.inl hsecond)
292     by_cases hpure : sample ∈ pureBad
293     · exact Or.inl (Or.inr hpure)
294     by_cases hcomparator : sample ∈ comparatorBad
295     · exact Or.inr hcomparator
296     exfalso
297     have hsecondGood : secondMoment sample < secondMomentBudget := by
298       change ¬ secondMomentBudget <= secondMoment sample at hsecond
299       exact lt_of_not_ge hsecond
300     have hpureGood :
301         purePredictable sample - pureObserved sample < pureRadius := by
302       change ¬ pureRadius <= purePredictable sample - pureObserved sample at hpure
303       exact lt_of_not_ge hpure
304     have hcomparatorGood :
305         comparatorObserved sample - comparatorTrue sample <
306           comparatorRadius := by
307       change ¬ comparatorRadius <=
308         comparatorObserved sample - comparatorTrue sample at hcomparator
309       exact lt_of_not_ge hcomparator
310     have hexploration :
311         exploredPredictable sample <=
312           purePredictable sample + gamma * (horizon : Real) := by
313       have h := (sampledTrajectory_finiteHorizon_explorationBias_secondMoment
314         arms harms eta gamma hgamma_pos.le hgamma_lt_one loss horizon sample).1
315       simpa [exploredPredictable, purePredictable,
316         sampledTrajectoryExploredPredictableLossAt,
317         sampledTrajectoryPurePredictableLossAt,
318         sampledTrajectoryPureProbabilityAt] using h
319     have hcoef_pos : 0 < eta * (1 / (1 - gamma)) :=
320       mul_pos heta (one_div_pos.mpr (sub_pos.mpr hgamma_lt_one))
321     have hweightedSecond :
322         (eta * (1 / (1 - gamma))) * secondMoment sample <
323           (eta * (1 / (1 - gamma))) * secondMomentBudget :=
324       mul_lt_mul_of_pos_left hsecondGood hcoef_pos
325     have hhedge' :
326         pureObserved sample - comparatorObserved sample <=
327           Real.log arms.card / eta +
328             (eta * (1 / (1 - gamma))) * secondMoment sample := by
329       simpa [pureObserved, comparatorObserved, secondMoment,
330         sampledObservedMixedSquaredSum] using hsampleHedge
331     have hfinal : exploredPredictable sample - comparatorTrue sample <
332         sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
333           arms eta gamma horizon deltaSquare deltaConfidence := by
334       dsimp [sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget,
335         secondMomentBudget, pureRadius, comparatorRadius]
336       linarith
337     exact (not_lt_of_ge hregret) hfinal
338   calc
339     mu {sample |
340         sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
341             arms eta gamma horizon deltaSquare deltaConfidence <=
342           (Finset.range horizon).sum (fun t =>
343               sampledTrajectoryExploredPredictableLossAt
344                 arms eta gamma loss t sample) -
345             (Finset.range horizon).sum (fun t =>
346               predictableLossAt loss t sample comparator)} <=
347         mu ((secondMomentBad ∪ pureBad) ∪ comparatorBad) := by
348       apply measure_mono_ae
349       simpa [exploredPredictable, comparatorTrue, mu] using hsubset
350     _ <= (mu secondMomentBad + mu pureBad) + mu comparatorBad := by
351       exact (measure_union_le _ _).trans
352         (add_le_add (measure_union_le _ _) le_rfl)
353     _ <= (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
354         ENNReal.ofReal deltaConfidence :=
355       add_le_add (add_le_add hsecondTail hpureTail) hcomparatorTail

357 /-- Total-failure form: the estimator-square event and both Bernstein
358 confidence events each receive `delta / 3`. -/
359 theorem sampledPredictable_randomSquareBernsteinHighProbabilityRegret_tail_total_delta
360     {Env : Type u} {Action : Type v}
361     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
362     [MeasurableSpace Action] [MeasurableSingletonClass Action]
363     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
364     (prior : Measure Env) [IsProbabilityMeasure prior]
365     (arms : Finset Action) (harms : arms.Nonempty)
366     (eta gamma : Real) (heta : 0 < eta)
367     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
368     (loss : PredictableLossVector Env Action)
369     (comparator : Action) (hcomparator : comparator ∈ arms)
370     (horizon : Nat) (hhorizon : 0 < horizon)
371     (delta : Real) (hdelta : 0 < delta) :
372     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
373       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
374     mu {sample |
375         sampledPredictableRandomSquareBernsteinHighProbabilityRegretBudget
376             arms eta gamma horizon (delta / 3) (delta / 3) <=
377           (Finset.range horizon).sum (fun t =>
378               sampledTrajectoryExploredPredictableLossAt
379                 arms eta gamma loss t sample) -
380             (Finset.range horizon).sum (fun t =>
381               predictableLossAt loss t sample comparator)} <=
382       ENNReal.ofReal delta := by
383   have hthird_pos : 0 < delta / 3 := div_pos hdelta (by norm_num)
384   have h := sampledPredictable_randomSquareBernsteinHighProbabilityRegret_tail
385     prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
386       hcomparator horizon hhorizon (delta / 3) (delta / 3)
387         hthird_pos hthird_pos
388   dsimp only at h ⊢
389   have hthird_nonneg : 0 <= delta / 3 := hthird_pos.le
390   have hprob :
391       (ENNReal.ofReal (delta / 3) + ENNReal.ofReal (delta / 3)) +
392           ENNReal.ofReal (delta / 3) = ENNReal.ofReal delta := by
393     rw [← ENNReal.ofReal_add hthird_nonneg hthird_nonneg]
394     rw [← ENNReal.ofReal_add (add_nonneg hthird_nonneg hthird_nonneg)
395       hthird_nonneg]
396     congr 1
397     ring
398   exact h.trans_eq hprob
