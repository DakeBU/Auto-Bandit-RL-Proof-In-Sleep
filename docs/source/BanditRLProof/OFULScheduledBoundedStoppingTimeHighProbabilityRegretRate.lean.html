5 # Bounded stopping-time high-probability OFUL pseudo-regret rate
6 
7 This module evaluates the explicit one-policy all-horizon OFUL rate at a
8 bounded stopping time. The probability argument is pathwise event domination,
9 not optional stopping and not a new union bound.
10 -/

20 /--
21 The canonical trajectory filtration through the current round. At level `T`
22 it contains exactly the coordinates `0, ..., T`.
23 -/
24 def canonicalHistoryTrajectoryAllRoundFiltration
25     {K : Nat} :
26     Filtration Nat
27       (inferInstance : MeasurableSpace (Nat -> Fin K × Real)) :=
28   Filtration.piLE (X := fun _ : Nat => Fin K × Real)
29 
30 @[simp]
31 theorem canonicalHistoryTrajectoryAllRoundFiltration_apply
32     {K : Nat} (horizon : Nat) :
33     (canonicalHistoryTrajectoryAllRoundFiltration (K := K) horizon :
34       MeasurableSpace (Nat -> Fin K × Real)) =
35       Filtration.piLE (X := fun _ : Nat => Fin K × Real) horizon :=
36   rfl

38 /-- Every coordinate `t <= T` is measurable at all-round level `T`. -/
39 theorem measurable_canonicalHistoryTrajectory_coordinate_allRound
40     {K : Nat} {t horizon : Nat} (ht : t <= horizon) :
41     @Measurable (Nat -> Fin K × Real) (Fin K × Real)
42       (canonicalHistoryTrajectoryAllRoundFiltration (K := K) horizon)
43       inferInstance
44       (fun trajectory => trajectory t) := by
45   have hprefix :
46       @Measurable (Nat -> Fin K × Real)
47         ((i : Finset.Iic horizon) -> Fin K × Real)
48         (canonicalHistoryTrajectoryAllRoundFiltration (K := K) horizon)
49         inferInstance
50         (Preorder.frestrictLe horizon) := by
51     rw [canonicalHistoryTrajectoryAllRoundFiltration_apply,
52       Filtration.piLE_eq_comap_frestrictLe]
53     exact Measurable.of_comap_le le_rfl
54   let index : Finset.Iic horizon := ⟨t, Finset.mem_Iic.mpr ht⟩
55   have heval :
56       Measurable
57         (fun history : (i : Finset.Iic horizon) -> Fin K × Real =>
58           history index) :=
59     measurable_pi_apply index
60   simpa [index, Preorder.frestrictLe] using heval.comp hprefix

62 /-- A trajectory time below a finite deterministic horizon cannot be `top`. -/
63 theorem boundedTrajectoryTime_ne_top
64     {K : Nat}
65     (tau : (Nat -> Fin K × Real) -> WithTop Nat)
66     (maxHorizon : Nat)
67     (htau_le : forall trajectory,
68       tau trajectory <= (maxHorizon : WithTop Nat)) :
69     forall trajectory, tau trajectory ≠ (⊤ : WithTop Nat) := by
70   intro trajectory
71   exact
72     ne_top_of_le_ne_top WithTop.coe_ne_top (htau_le trajectory)

74 /-- Under a finite bound, `untopA` recovers the actual stopping-time value. -/
75 theorem coe_untopA_boundedTrajectoryTime
76     {K : Nat}
77     (tau : (Nat -> Fin K × Real) -> WithTop Nat)
78     (maxHorizon : Nat)
79     (htau_le : forall trajectory,
80       tau trajectory <= (maxHorizon : WithTop Nat))
81     (trajectory : Nat -> Fin K × Real) :
82     ((tau trajectory).untopA : WithTop Nat) = tau trajectory := by
83   have hne : tau trajectory ≠ (⊤ : WithTop Nat) :=
84     boundedTrajectoryTime_ne_top tau maxHorizon htau_le trajectory
85   rw [WithTop.untopA_eq_untop hne]
86   exact WithTop.coe_untop _ hne

88 /--
89 Complete fixed-best pseudo-regret through horizon `T` is measurable using only
90 trajectory coordinates `0, ..., T`.
91 -/
92 theorem measurable_canonicalStandardHighProbabilityPseudoRegret_allRound
93     {K : Nat} {Feature : Type u}
94     [Fintype Feature]
95     (thetaStar : Feature -> Real)
96     (actionFeature : Fin K -> Feature -> Real)
97     (best : Fin K) (horizon : Nat) :
98     @Measurable (Nat -> Fin K × Real) Real
99       (canonicalHistoryTrajectoryAllRoundFiltration (K := K) horizon)
100       inferInstance
101       (canonicalStandardHighProbabilityPseudoRegret
102         thetaStar actionFeature best horizon) := by
103   have hprefix :
104       @Measurable (Nat -> Fin K × Real)
105         ((i : Finset.Iic horizon) -> Fin K × Real)
106         (canonicalHistoryTrajectoryAllRoundFiltration (K := K) horizon)
107         inferInstance
108         (Preorder.frestrictLe horizon) := by
109     rw [canonicalHistoryTrajectoryAllRoundFiltration_apply,
110       Filtration.piLE_eq_comap_frestrictLe]
111     exact Measurable.of_comap_le le_rfl
112   have hfinite :
113       Measurable
114         (fun history : (i : Finset.Iic horizon) -> Fin K × Real =>
115           (Finset.range (horizon + 1)).sum (fun t =>
116             linearValue thetaStar (actionFeature best) -
117               linearValue thetaStar
118                 (actionFeature
119                   (if ht : t <= horizon then
120                     (history ⟨t, Finset.mem_Iic.mpr ht⟩).1
121                   else best)))) := by
122     refine Finset.measurable_sum (Finset.range (horizon + 1)) fun t ht => ?_
123     have ht_le : t <= horizon :=
124       Nat.le_of_lt_succ (Finset.mem_range.mp ht)
125     let index : Finset.Iic horizon :=
126       ⟨t, Finset.mem_Iic.mpr ht_le⟩
127     simp only [dif_pos ht_le]
128     exact measurable_const.sub
129       ((measurable_of_countable
130         (fun action : Fin K =>
131           linearValue thetaStar (actionFeature action))).comp
132         (measurable_fst.comp (measurable_pi_apply index)))
133   have hcomp := hfinite.comp hprefix
134   unfold canonicalStandardHighProbabilityPseudoRegret
135     canonicalHistoryTrajectorySumRangeAllGap
136   convert hcomp using 1
137   funext trajectory
138   apply Finset.sum_congr rfl
139   intro t ht
140   have ht_le : t <= horizon :=
141     Nat.le_of_lt_succ (Finset.mem_range.mp ht)
142   simp only [dif_pos ht_le, Thompson.canonicalHistoryTrajectoryAction,
143     Preorder.frestrictLe]
144   rfl
