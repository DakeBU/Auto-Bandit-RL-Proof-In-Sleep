4 # Finite-arm independent reward laws with drifting means
5 
6 Each round may use a different probability law for every arm.  The roundwise
7 arm means may drift from a fixed finite-bandit model, with a deterministic
8 coordinatewise deviation envelope.  The resulting regret theorem charges the
9 induced two-arm gap deviation explicitly.
10 -/

18 noncomputable def finiteArmIndependentRewardMean {K : Nat}
19     (armLaw : Nat -> Fin K -> Measure Rat) (t : Nat) (arm : Fin K) : Real :=

22 /-- Under the unit-support contract, the roundwise product-law loss gap is the

24 theorem independentLossStateTimeVaryingMeanGap_finiteArmIndependentRewardVectorLoss_eq_mean_sub

28       ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1)

31         (finiteArmIndependentRewardVectorLaw armLaw)
32         (fun _ => finiteArmIIDRewardVectorLoss)
33         t best arm =
34       finiteArmIndependentRewardMean armLaw t best -
35         finiteArmIndependentRewardMean armLaw t arm := by
36   letI (i : Fin K) : IsProbabilityMeasure (armLaw t i) := hprob t i
37   have hbestIntegrable : Integrable
38       (fun state : Fin K -> Rat => clippedUnitReward (state best))
39       (finiteArmIIDRewardVectorLaw (armLaw t)) := by
40     exact MeasureTheory.integrable_comp_eval
41       (integrable_clippedUnitReward (armLaw t best))
42   have harmIntegrable : Integrable
43       (fun state : Fin K -> Rat => clippedUnitReward (state arm))

45     exact MeasureTheory.integrable_comp_eval
46       (integrable_clippedUnitReward (armLaw t arm))

51     rw [finiteArmIIDRewardVectorLaw,
52       MeasureTheory.integral_comp_eval
53         (integrable_clippedUnitReward (armLaw t action)).aestronglyMeasurable]
54     apply integral_congr_ae
55     filter_upwards [hbound t action] with reward hreward
56     exact clippedUnitReward_eq_of_mem_Icc reward hreward
57   rw [independentLossStateTimeVaryingMeanGap,
58     finiteArmIndependentRewardVectorLaw]
59   calc
60     integral (finiteArmIIDRewardVectorLaw (armLaw t))
61         (fun state => finiteArmIIDRewardVectorLoss state arm -
62           finiteArmIIDRewardVectorLoss state best) =
63         integral (finiteArmIIDRewardVectorLaw (armLaw t))
64           (fun state => clippedUnitReward (state best) -
65             clippedUnitReward (state arm)) := by
66       congr 1
67       funext state
68       simp only [finiteArmIIDRewardVectorLoss]
69       ring
70     _ = integral (finiteArmIIDRewardVectorLaw (armLaw t))
71           (fun state => clippedUnitReward (state best)) -
72         integral (finiteArmIIDRewardVectorLaw (armLaw t))
73           (fun state => clippedUnitReward (state arm)) :=
74       integral_sub hbestIntegrable harmIntegrable
75     _ = finiteArmIndependentRewardMean armLaw t best -
76         finiteArmIndependentRewardMean armLaw t arm := by
77       rw [hcoordinate best, hcoordinate arm]

103         ((model.mean model.bestArm : Rat) : Real) -
104           ((model.mean arm : Rat) : Real) := by

106   rw [hgap]
107   calc
108     |(finiteArmIndependentRewardMean armLaw t model.bestArm -
109           finiteArmIndependentRewardMean armLaw t arm) -
110         (((model.mean model.bestArm : Rat) : Real) -
111           ((model.mean arm : Rat) : Real))| =
112         |(finiteArmIndependentRewardMean armLaw t model.bestArm -
113             ((model.mean model.bestArm : Rat) : Real)) -
114           (finiteArmIndependentRewardMean armLaw t arm -
115             ((model.mean arm : Rat) : Real))| := by ring_nf
116     _ <= |finiteArmIndependentRewardMean armLaw t model.bestArm -
117           ((model.mean model.bestArm : Rat) : Real)| +
118         |finiteArmIndependentRewardMean armLaw t arm -
119           ((model.mean arm : Rat) : Real)| := abs_sub _ _
120     _ <= meanDeviation t model.bestArm + meanDeviation t arm :=
