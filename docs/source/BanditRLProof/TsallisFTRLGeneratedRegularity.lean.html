19 /-- Every supported coordinate of a finite simplex point is at most one. -/
20 theorem finiteSimplex_apply_le_one
21     {Action : Type u} {arms : Finset Action} {p : Action -> Real}
22     (hp : FTRL.finiteSimplex arms p) {action : Action}
23     (haction : action ∈ arms) :
24     p action <= 1 := by
25   calc
26     p action <= arms.sum p := Finset.single_le_sum hp.1 haction
27     _ = 1 := hp.2

29 /-- Sampling mass cancels the inverse probability in one realized stability
30 score, leaving a bound by the current and updated selected coordinates. -/
31 theorem prob_mul_abs_importanceWeightedStabilityScore_le
32     {History : Type u} {Action : Type v} [DecidableEq Action]
33     (arms : Finset Action) (prob loss : History -> Action -> Real)
34     (next : History -> Action -> Action -> Real)
35     (history : History) (chosen : Action) (hchosen : chosen ∈ arms)
36     (hnextSimplex : FTRL.finiteSimplex arms (next history chosen))
37     (hprobPos : 0 < prob history chosen)
38     (hloss : 0 <= loss history chosen ∧ loss history chosen <= 1) :
39     prob history chosen *
40         |importanceWeightedStabilityScore arms prob loss next (history, chosen)| <=
41       prob history chosen + next history chosen chosen := by
42   have hprobNonneg : 0 <= prob history chosen := hprobPos.le
43   have hnextNonneg : 0 <= next history chosen chosen :=
44     hnextSimplex.1 chosen hchosen
45   have hprobNe : prob history chosen ≠ 0 := ne_of_gt hprobPos
46   have hcurrent :
47       FTRL.linearLoss arms (prob history)
48           (Exp3.importanceWeightedLoss (prob history) (loss history) chosen) =
49         loss history chosen := by
50     simpa [FTRL.linearLoss, Exp3.mixedImportanceWeightedLoss] using
51       (Exp3.mixedImportanceWeightedLoss_eq_selectedLoss
52         arms (prob history) (loss history) chosen hchosen hprobNe)
53   have hupdated :
54       FTRL.linearLoss arms (next history chosen)
55           (Exp3.importanceWeightedLoss (prob history) (loss history) chosen) =
56         next history chosen chosen *
57           (loss history chosen / prob history chosen) := by
58     unfold FTRL.linearLoss
59     rw [Finset.sum_eq_single chosen]
60     · simp [Exp3.importanceWeightedLoss]
61     · intro action haction hne
62       simp [Exp3.importanceWeightedLoss, Ne.symm hne]
63     · exact fun hnotmem => (hnotmem hchosen).elim
64   have hweighted :
65       prob history chosen *
66           (loss history chosen -
67             next history chosen chosen *
68               (loss history chosen / prob history chosen)) =
69         prob history chosen * loss history chosen -
70           next history chosen chosen * loss history chosen := by
71     field_simp
72   rw [importanceWeightedStabilityScore, hcurrent, hupdated]
73   calc
74     prob history chosen *
75         |loss history chosen -
76           next history chosen chosen *
77             (loss history chosen / prob history chosen)| =
78       |prob history chosen *
79         (loss history chosen -
80           next history chosen chosen *
81             (loss history chosen / prob history chosen))| := by
82         rw [abs_mul, abs_of_nonneg hprobNonneg]
83     _ = |prob history chosen * loss history chosen -
84           next history chosen chosen * loss history chosen| := by
85         rw [hweighted]
86     _ <= |prob history chosen * loss history chosen| +
87           |next history chosen chosen * loss history chosen| := by
88         simpa using (_root_.abs_sub_le
89           (prob history chosen * loss history chosen) 0
90           (next history chosen chosen * loss history chosen))
91     _ = prob history chosen * loss history chosen +
92           next history chosen chosen * loss history chosen := by
93         rw [abs_of_nonneg (mul_nonneg hprobNonneg hloss.1),
94           abs_of_nonneg (mul_nonneg hnextNonneg hloss.1)]
95     _ <= prob history chosen + next history chosen chosen := by
96         nlinarith

108 /-- A measurable one-round stability score is automatically integrable under
109 its finite sampling kernel.  No uniform lower probability floor is needed. -/
110 theorem integrable_importanceWeightedStabilityScore_finiteActionKernel
111     {History : Type u} {Action : Type v}
112     [MeasurableSpace History]
113     [MeasurableSpace Action] [MeasurableSingletonClass Action]
114     [DecidableEq Action]
115     (historyMu : Measure History) [IsFiniteMeasure historyMu]
116     (arms : Finset Action) (prob loss : History -> Action -> Real)
117     (next : History -> Action -> Action -> Real)
118     (source : Exp3.MeasurableFiniteActionDistribution arms prob)
119     (hprobPos : forall history action, action ∈ arms ->
120       0 < prob history action)
121     (hnextSimplex : forall history chosen, chosen ∈ arms ->
122       FTRL.finiteSimplex arms (next history chosen))
123     (hloss : forall history action, action ∈ arms ->
124       0 <= loss history action ∧ loss history action <= 1)
125     (hscore : Measurable
126       (importanceWeightedStabilityScore arms prob loss next)) :
127     Integrable (importanceWeightedStabilityScore arms prob loss next)
128       (historyMu ⊗ₘ Exp3.finiteActionKernel arms prob source) := by
129   let stability := importanceWeightedStabilityScore arms prob loss next
130   have hconditional (history : History) : Integrable
131       (fun chosen => stability (history, chosen))
132       (Exp3.finiteActionKernel arms prob source history) := by
133     rw [Exp3.finiteActionKernel_apply]
134     exact integrable_finiteActionMeasure arms (prob history)
135       (fun chosen => stability (history, chosen))
136   have hsumMeasurable : Measurable (fun history => arms.sum (fun chosen =>
137       prob history chosen * |stability (history, chosen)|)) := by
138     refine Finset.measurable_sum arms fun chosen hchosen => ?_
139     exact (source.measurable_prob chosen hchosen).mul
140       ((hscore.comp (measurable_id.prodMk measurable_const)).abs)
141   have hsumBound (history : History) :
142       arms.sum (fun chosen =>
143           prob history chosen * |stability (history, chosen)|) <=
144         1 + (arms.card : Real) := by
145     calc
146       arms.sum (fun chosen =>
147           prob history chosen * |stability (history, chosen)|) <=
148         arms.sum (fun chosen =>
149           prob history chosen + next history chosen chosen) := by
150             exact Finset.sum_le_sum fun chosen hchosen =>
151               prob_mul_abs_importanceWeightedStabilityScore_le
152                 arms prob loss next history chosen hchosen
153                 (hnextSimplex history chosen hchosen)
154                 (hprobPos history chosen hchosen)
155                 (hloss history chosen hchosen)
156       _ = arms.sum (prob history) +
157           arms.sum (fun chosen => next history chosen chosen) := by
158             rw [Finset.sum_add_distrib]
159       _ <= 1 + arms.sum (fun _chosen => (1 : Real)) := by
160             rw [(source.distribution history).sum_eq_one]
161             have hnextSum := Finset.sum_le_sum fun chosen hchosen =>
162               finiteSimplex_apply_le_one
163                 (hnextSimplex history chosen hchosen) hchosen
164             linarith
165       _ = 1 + (arms.card : Real) := by simp
166   have hsumIntegrable : Integrable (fun history => arms.sum (fun chosen =>
167       prob history chosen * |stability (history, chosen)|)) historyMu := by
168     refine Integrable.of_bound hsumMeasurable.aestronglyMeasurable
169       (1 + (arms.card : Real)) ?_
170     exact Filter.Eventually.of_forall fun history => by
171       rw [Real.norm_eq_abs, abs_of_nonneg]
172       · exact hsumBound history
173       · exact Finset.sum_nonneg fun chosen hchosen =>
174           mul_nonneg ((source.distribution history).nonneg chosen hchosen)
175             (abs_nonneg _)
176   apply (Measure.integrable_compProd_iff hscore.aestronglyMeasurable).2
177   constructor
178   · exact Filter.Eventually.of_forall hconditional
179   · have hintegralEq :
180         (fun history => integral
181           (Exp3.finiteActionKernel arms prob source history)
182           (fun chosen => ‖stability (history, chosen)‖)) =
183         (fun history => arms.sum (fun chosen =>
184           prob history chosen * |stability (history, chosen)|)) := by
185       funext history
186       rw [Exp3.finiteActionKernel_apply,
187         Exp3.integral_finiteActionMeasure_eq_sum arms (prob history)
188           (source.distribution history)]
189       simp only [Real.norm_eq_abs]
190     rw [hintegralEq]
191     exact hsumIntegrable

193 /-- The half-power stability budget is uniformly bounded and hence integrable
194 under every finite history measure. -/
195 theorem integrable_halfPowerStabilityBound_of_finiteSimplex
196     {History : Type u} {Action : Type v} [MeasurableSpace History]
197     (historyMu : Measure History) [IsFiniteMeasure historyMu]
198     (arms : Finset Action) (eta : Real)
199     (prob : History -> Action -> Real)
200     (source : Exp3.MeasurableFiniteActionDistribution arms prob) :
201     Integrable (halfPowerStabilityBound arms eta prob) historyMu := by
202   have hmeasurable : Measurable (halfPowerStabilityBound arms eta prob) := by
203     unfold halfPowerStabilityBound powerSum
204     exact measurable_const.mul
205       (Finset.measurable_sum arms fun action haction =>
206         (Real.continuous_rpow_const (by norm_num :
207           (0 : Real) <= (1 / 2 : Real))).measurable.comp
208             (source.measurable_prob action haction))
209   refine Integrable.of_bound hmeasurable.aestronglyMeasurable
210     (2 * |eta| * (arms.card : Real)) ?_
211   exact Filter.Eventually.of_forall fun history => by
212     have hpowerNonneg : 0 <= powerSum arms (1 / 2 : Real) (prob history) :=
213       powerSum_nonneg_of_finiteSimplex arms (1 / 2 : Real) (prob history)
214         ⟨(source.distribution history).nonneg,
215           (source.distribution history).sum_eq_one⟩
216     have hpowerLe :
217         powerSum arms (1 / 2 : Real) (prob history) <=
218           (arms.card : Real) := by
219       unfold powerSum
220       calc
221         arms.sum (fun action => (prob history action) ^ (1 / 2 : Real)) <=
222             arms.sum (fun _action => (1 : Real)) := by
223           exact Finset.sum_le_sum fun action haction =>
224             Real.rpow_le_one
225               ((source.distribution history).nonneg action haction)
226               (finiteSimplex_apply_le_one
227                 ⟨(source.distribution history).nonneg,
228                   (source.distribution history).sum_eq_one⟩ haction)
229               (by norm_num)
230         _ = (arms.card : Real) := by simp
231     rw [halfPowerStabilityBound, Real.norm_eq_abs, abs_mul, abs_mul,
232       abs_of_nonneg (by norm_num : (0 : Real) <= 2),
233       abs_of_nonneg hpowerNonneg]
234     nlinarith [abs_nonneg eta]
