7 # Self-bounding interfaces for the half-Tsallis route
8 
9 This module records the paper-facing `(Delta, C, T)` self-bounding condition,
10 identifies it on the generated predictable trajectory under an exact gap law,
11 and proves the finite completion-of-squares conversion used after a refined
12 suboptimal-arm stability bound is available.
13 
14 The currently compiled half-power stability endpoint contains the optimal-arm
15 term.  The final counterexample below records why that endpoint cannot be fed
16 directly into the gap conversion.
17 -/

26 /-- Unit mass at one action, used as the fixed optimal-arm comparator. -/
27 def pointMass {Action : Type u} [DecidableEq Action]
28     (best : Action) : Action -> Real :=
29   fun action => if action = best then 1 else 0

31 theorem finiteSimplex_pointMass
32     {Action : Type u} [DecidableEq Action]
33     (arms : Finset Action) {best : Action} (hbest : best ∈ arms) :
34     FTRL.finiteSimplex arms (pointMass best) := by
35   constructor
36   · intro action _haction
37     by_cases h : action = best <;> simp [pointMass, h]
38   · simp [pointMass, hbest]

40 theorem linearLoss_pointMass
41     {Action : Type u} [DecidableEq Action]
42     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
43     (loss : Action -> Real) :
44     FTRL.linearLoss arms (pointMass best) loss = loss best := by
45   unfold FTRL.linearLoss pointMass
46   simp [hbest]

48 /-- One-round regret against a point mass is the probability-weighted gap. -/
49 theorem linearLoss_sub_pointMass_eq_gapMass
50     {Action : Type u} [DecidableEq Action]
51     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
52     (prob loss gap : Action -> Real)
53     (hprob : FTRL.finiteSimplex arms prob)
54     (hgap : ∀ action ∈ arms, loss action - loss best = gap action) :
55     FTRL.linearLoss arms prob loss -
56         FTRL.linearLoss arms (pointMass best) loss =
57       arms.sum (fun action => prob action * gap action) := by
58   rw [linearLoss_pointMass arms hbest loss]
59   have hconstant :
60       arms.sum (fun action => prob action * loss best) = loss best := by
61     rw [← Finset.sum_mul, hprob.2, one_mul]
62   calc
63     FTRL.linearLoss arms prob loss - loss best =
64         arms.sum (fun action => prob action * loss action) -
65           arms.sum (fun action => prob action * loss best) := by
66       rw [hconstant]
67       rfl
68     _ = arms.sum (fun action =>
69         prob action * loss action - prob action * loss best) := by
70       rw [Finset.sum_sub_distrib]
71     _ = arms.sum (fun action => prob action * gap action) := by
72       apply Finset.sum_congr rfl
73       intro action haction
74       rw [← hgap action haction]
75       ring

77 /-- Gap mass accumulated by the generated actual-time sampling laws. -/
78 noncomputable def sampledHalfTsallisPredictableGapMass
79     {Env : Type u} {Action : Type v} [DecidableEq Action]
80     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
81     (gap : Action -> Real) (horizon : Nat)
82     (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
83   (Finset.range (horizon + 1)).sum (fun t =>
84     arms.sum (fun action =>
85       sampledHalfTsallisProbabilityAtTime
86         arms harms eta t sample action * gap action))

88 /-- Paper-facing scalar form of a `(Delta, C, T)` self-bounding constraint. -/
89 def HasSelfBoundingRegret
90     (expectedGapMass regret corruption : Real) : Prop :=
91   expectedGapMass - corruption ≤ regret

93 /-- A predictable fixed-gap law identifies environment regret pathwise. -/
94 theorem sampledHalfTsallisPredictableEnvironmentRegret_pointMass_eq_gapMass
95     {Env : Type u} {Action : Type v}
96     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
97     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
98     (loss : Exp3.PredictableLossVector Env Action)
99     {best : Action} (hbest : best ∈ arms) (gap : Action -> Real)
100     (horizon : Nat)
101     (hgap : ∀ t sample action, action ∈ arms ->
102       Exp3.predictableLossAt loss t sample action -
103         Exp3.predictableLossAt loss t sample best = gap action)
104     (sample : Env × ((k : Nat) -> Action × Real)) :
105     sampledHalfTsallisPredictableEnvironmentRegret
106         arms harms eta loss (pointMass best) horizon sample =
107       sampledHalfTsallisPredictableGapMass
108         arms harms eta gap horizon sample := by
109   unfold sampledHalfTsallisPredictableEnvironmentRegret
110     sampledHalfTsallisPredictableGapMass
111   apply Finset.sum_congr rfl
112   intro t _ht
113   exact linearLoss_sub_pointMass_eq_gapMass
114     arms hbest
115     (sampledHalfTsallisProbabilityAtTime arms harms eta t sample)
116     (Exp3.predictableLossAt loss t sample) gap
117     (finiteSimplex_sampledHalfTsallisProbabilityAtTime
118       arms harms eta t sample)
119     (hgap t sample)

121 /-- The fixed-gap predictable environment satisfies the integrated
122 self-bounding condition, with any nonnegative corruption allowance. -/
123 theorem integral_sampledHalfTsallisPredictableEnvironmentRegret_hasSelfBounding
124     {Env : Type u} {Action : Type v}
125     [MeasurableSpace Env] [StandardBorelSpace Env]
126     [MeasurableSpace Action] [MeasurableSingletonClass Action]
127     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
128     (prior : Measure Env) [IsProbabilityMeasure prior]
129     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
130     (loss : Exp3.PredictableLossVector Env Action)
131     {best : Action} (hbest : best ∈ arms) (gap : Action -> Real)
132     (horizon : Nat)
133     (hgap : ∀ t sample action, action ∈ arms ->
134       Exp3.predictableLossAt loss t sample action -
135         Exp3.predictableLossAt loss t sample best = gap action)
136     (corruption : Real) (hcorruption : 0 ≤ corruption) :
137     let selector := canonicalHalfTsallisFiniteHistorySelectorMeasurability
138       arms harms eta
139     let mu := prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
140       arms harms eta selector loss.environment
141     HasSelfBoundingRegret
142       (integral mu (sampledHalfTsallisPredictableGapMass
143         arms harms eta gap horizon))
144       (integral mu (sampledHalfTsallisPredictableEnvironmentRegret
145         arms harms eta loss (pointMass best) horizon))
146       corruption := by
147   dsimp only
148   unfold HasSelfBoundingRegret
149   have heq :
150       integral
151           (prior ⊗ₘ sampledHalfTsallisTrajectoryKernel arms harms eta
152             (canonicalHalfTsallisFiniteHistorySelectorMeasurability
153               arms harms eta) loss.environment)
154           (sampledHalfTsallisPredictableGapMass
155             arms harms eta gap horizon) =
156         integral
157           (prior ⊗ₘ sampledHalfTsallisTrajectoryKernel arms harms eta
158             (canonicalHalfTsallisFiniteHistorySelectorMeasurability
159               arms harms eta) loss.environment)
160           (sampledHalfTsallisPredictableEnvironmentRegret
161             arms harms eta loss (pointMass best) horizon) := by
162     apply integral_congr_ae
163     exact Filter.Eventually.of_forall (fun sample =>
164       (sampledHalfTsallisPredictableEnvironmentRegret_pointMass_eq_gapMass
165         arms harms eta loss hbest gap horizon hgap sample).symm)
166   linarith

168 /-- Actual generated environment-regret upper bound specialized to the
169 optimal-arm point-mass comparator. -/
170 theorem integral_sampledHalfTsallisPredictableEnvironmentRegret_pointMass_le
171     {Env : Type u} {Action : Type v}
172     [MeasurableSpace Env] [StandardBorelSpace Env]
173     [MeasurableSpace Action] [MeasurableSingletonClass Action]
174     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
175     (prior : Measure Env) [IsProbabilityMeasure prior]
176     (arms : Finset Action) (harms : arms.Nonempty)
177     (eta : Real) (heta : 0 < eta)
178     (loss : Exp3.PredictableLossVector Env Action)
179     {best : Action} (hbest : best ∈ arms) (horizon : Nat) :
180     let selector := canonicalHalfTsallisFiniteHistorySelectorMeasurability
181       arms harms eta
182     let mu := prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
183       arms harms eta selector loss.environment
184     integral mu (sampledHalfTsallisPredictableEnvironmentRegret
185         arms harms eta loss (pointMass best) horizon) <=
186       2 * eta * powerSum arms (1 / 2 : Real)
187           (initialHalfTsallisDistribution arms harms eta) +
188         integral mu (fun sample => (Finset.range horizon).sum (fun n =>
189           sampledHalfTsallisHalfPowerBoundAt (Env := Env)
190             arms harms eta n (sampledHalfTsallisHistoryAt n sample))) +
191         ((powerSum arms (1 / 2 : Real)
192               (initialHalfTsallisDistribution arms harms eta) -
193             powerSum arms (1 / 2 : Real) (pointMass best)) /
194           (1 - (1 / 2 : Real))) / eta := by
195   exact integral_sampledHalfTsallisPredictableEnvironmentRegret_le
196     prior arms harms eta heta loss (pointMass best)
197       (finiteSimplex_pointMass arms hbest) horizon

199 /-- Coordinate completion of squares used by the self-bounding conversion. -/
200 theorem two_mul_coeff_mul_sqrt_sub_gap_mul_le_sq_div_gap
201     (coeff probability gap : Real) (hprobability : 0 ≤ probability)
202     (hgap : 0 < gap) :
203     2 * coeff * Real.sqrt probability - gap * probability ≤
204       coeff ^ 2 / gap := by
205   apply (le_div_iff₀ hgap).2
206   have hsqrtSq := Real.sq_sqrt hprobability
207   have hsq := sq_nonneg (coeff - gap * Real.sqrt probability)
208   calc
209     (2 * coeff * Real.sqrt probability - gap * probability) * gap =
210         2 * coeff * Real.sqrt probability * gap - gap ^ 2 * probability := by
211       ring
212     _ = 2 * coeff * Real.sqrt probability * gap -
213         gap ^ 2 * (Real.sqrt probability) ^ 2 := by
214       rw [hsqrtSq]
215     _ = coeff ^ 2 - (coeff - gap * Real.sqrt probability) ^ 2 := by
216       ring
217     _ <= coeff ^ 2 := sub_le_self _ hsq

219 /-- Finite `(Delta, C, T)` self-bounding conversion after a refined upper
220 bound has removed every zero-gap coordinate. -/
221 theorem regret_le_two_mul_base_add_sum_sq_div_gap_add_corruption
222     {Index : Type u} (indices : Finset Index)
223     (probability coeff gap : Index -> Real)
224     (regret base corruption : Real)
225     (hprobability : ∀ index ∈ indices, 0 ≤ probability index)
226     (hgap : ∀ index ∈ indices, 0 < gap index)
227     (hselfBounding :
228       indices.sum (fun index => gap index * probability index) -
229           corruption ≤ regret)
230     (hupper :
231       regret ≤ base + indices.sum (fun index =>
232         coeff index * Real.sqrt (probability index))) :
233     regret ≤
234       2 * base + indices.sum (fun index => coeff index ^ 2 / gap index) +
235         corruption := by
236   have hcombine :
237       regret ≤
238         2 * (base + indices.sum (fun index =>
239           coeff index * Real.sqrt (probability index))) -
240         indices.sum (fun index => gap index * probability index) +
241         corruption := by
242     linarith
243   have hcoordinates :
244       indices.sum (fun index =>
245           2 * coeff index * Real.sqrt (probability index) -
246             gap index * probability index) ≤
247         indices.sum (fun index => coeff index ^ 2 / gap index) := by
248     apply Finset.sum_le_sum
249     intro index hindex
250     exact two_mul_coeff_mul_sqrt_sub_gap_mul_le_sq_div_gap
251       (coeff index) (probability index) (gap index)
252       (hprobability index hindex) (hgap index hindex)
253   have hrewrite :
254       2 * (base + indices.sum (fun index =>
255           coeff index * Real.sqrt (probability index))) -
256           indices.sum (fun index => gap index * probability index) +
257           corruption =
258         2 * base + indices.sum (fun index =>
259           2 * coeff index * Real.sqrt (probability index) -
260             gap index * probability index) + corruption := by
261     rw [Finset.sum_sub_distrib]
262     simp_rw [mul_assoc]
263     rw [← Finset.mul_sum]
264     ring
265   rw [hrewrite] at hcombine
266   linarith

269     {Action : Type u} [DecidableEq Action]
270     (arms : Finset Action) {best : Action} (hbest : best ∈ arms) :

275 theorem sum_erase_sqrt_pointMass
276     {Action : Type u} [DecidableEq Action]
277     (arms : Finset Action) (best : Action) :
278     (arms.erase best).sum (fun action =>
279       Real.sqrt (pointMass best action)) = 0 := by
280   apply Finset.sum_eq_zero
281   intro action haction
282   have hne : action ≠ best := (Finset.mem_erase.mp haction).1
283   simp [pointMass, hne]

285 /-- The existing total half-power budget cannot uniformly be replaced by a
286 suboptimal-arm square-root budget.  A refined `(1-p)` factor is genuinely
287 needed before the self-bounding conversion can consume the trajectory bound. -/
288 theorem not_forall_powerSum_half_le_sum_erase_sqrt
289     {Action : Type u} [DecidableEq Action]
290     (arms : Finset Action) {best : Action} (hbest : best ∈ arms) :
291     ¬ ∀ p : Action -> Real, FTRL.finiteSimplex arms p ->
292       powerSum arms (1 / 2 : Real) p ≤
293         (arms.erase best).sum (fun action => Real.sqrt (p action)) := by
294   intro h
295   have hbad := h (pointMass best) (finiteSimplex_pointMass arms hbest)
296   rw [powerSum_pointMass_half arms hbest,
297     sum_erase_sqrt_pointMass arms best] at hbad
298   norm_num at hbad
