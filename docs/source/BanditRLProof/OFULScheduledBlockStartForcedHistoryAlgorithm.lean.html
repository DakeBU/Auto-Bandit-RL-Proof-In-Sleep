3 /-!
4 # Concrete block-start forced scheduling for telescoping OFUL
5 
6 This module packages measurable finite-history selectors as deterministic
7 history algorithms and proves their canonical successor-action graph. It then
8 defines a modified telescoping OFUL selector that forces one prescribed action
9 after every block-start history while retaining the ordinary optimistic
10 selector at all other history indices.
11 
12 The final theorem constructs the aligned-window positive-action-cost contract
13 under the modified algorithm's own canonical trajectory measure. No ordinary
14 OFUL regret theorem is transported to that changed measure here; forced-round
15 regret accounting remains a separate route.
16 -/

55 /--
56 The canonical successor action of a deterministic history algorithm lies on
57 the graph of its measurable selector almost surely.
58 -/
59 theorem
60     canonicalHistoryTrajectory_action_succ_ae_eq_deterministicHistorySelector
61     {Action : Type u} {Reward : Type v}
62     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
63     [MeasurableEq Action]
64     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
65     (initialAction : Action)
66     (selector :
67       (n : Nat) -> History.FinitePairHistory Action Reward n -> Action)
68     (hselector : forall n, Measurable (selector n))
69     (environment : HistoryEnvironment Action Reward)
70     (n : Nat) :
71     ∀ᵐ trajectory ∂
72         canonicalHistoryTrajectoryMeasure
73           (deterministicHistoryAlgorithm initialAction selector hselector)
74           environment,
75       canonicalHistoryTrajectoryAction trajectory (n + 1) =
76         selector n
77           (History.finitePairHistoryOfTrace
78             (canonicalHistoryTrajectoryAction trajectory)
79             (canonicalHistoryTrajectoryReward trajectory) n) := by
80   let algorithm :=
81     deterministicHistoryAlgorithm initialAction selector hselector
82   let mu := canonicalHistoryTrajectoryMeasure algorithm environment
83   let history := fun trajectory : (k : Nat) -> Action × Reward =>
84     History.finitePairHistoryOfTrace
85       (canonicalHistoryTrajectoryAction trajectory)
86       (canonicalHistoryTrajectoryReward trajectory) n
87   let nextAction := fun trajectory : (k : Nat) -> Action × Reward =>
88     canonicalHistoryTrajectoryAction trajectory (n + 1)
89   have hhistory : Measurable history :=
90     History.measurable_finitePairHistoryOfTrace
91       canonicalHistoryTrajectoryAction
92       canonicalHistoryTrajectoryReward
93       measurable_canonicalHistoryTrajectoryAction_apply
94       measurable_canonicalHistoryTrajectoryReward_apply n
95   have hnextAction : Measurable nextAction :=
96     measurable_canonicalHistoryTrajectoryAction_apply (n + 1)
97   have hpolicy :=
98     policy_condDistrib_of_historyAlgorithmEnvironmentSequence
99       mu
100       canonicalHistoryTrajectoryAction
101       canonicalHistoryTrajectoryReward
102       algorithm environment
103       (canonicalHistoryAlgorithmEnvironmentSequence algorithm environment)
104       n
105   have hjoint :
106       mu.map (fun trajectory => (history trajectory, nextAction trajectory)) =
107         mu.map history ⊗ₘ algorithm.policy n :=
108     (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
109       history hnextAction.aemeasurable (algorithm.policy n)).mp <| by
110         simpa [history, nextAction] using hpolicy
111   have hp : MeasurableSet
112       {pair : History.FinitePairHistory Action Reward n × Action |
113         pair.2 = selector n pair.1} :=
114     measurableSet_eq_fun measurable_snd
115       ((hselector n).comp measurable_fst)
116   have hgraph :
117       ∀ᵐ pair ∂ mu.map history ⊗ₘ algorithm.policy n,
118         pair.2 = selector n pair.1 := by
119     apply Measure.ae_compProd_of_ae_ae hp
120     filter_upwards [] with historyValue
121     change ∀ᵐ b ∂
122       (deterministicHistoryAlgorithm
123         initialAction selector hselector).policy n historyValue,
124       b = selector n historyValue
125     rw [deterministicHistoryAlgorithm_policy_apply]
126     simp
127   rw [← hjoint] at hgraph
128   exact
129     (ae_map_iff
130       ((hhistory.prod hnextAction).aemeasurable) hp).mp hgraph

136 /--
137 Use the forced block action at divisible history indices and the ordinary
138 telescoping OFUL selector at all other indices.

140 noncomputable def finiteHistoryBlockStartForcedTelescopingScalarRidgeAction
141     {K : Nat} {Feature : Type u}
142     [Fintype Feature] [DecidableEq Feature]
143     (hK : 0 < K)
144     (lambda : Real)
145     (actionFeature : Fin K -> Feature -> Real)
146     (R delta S : Real)
147     (forcedAction : Nat -> Fin K)
148     (window n : Nat)
149     (history : History.FinitePairHistory (Fin K) Real n) :
150     Fin K :=
151   if n % window = 0 then
152     forcedAction (n / window)
