19     (action : Omega -> ActionTrace Action)
20     (reward : Omega -> RewardTrace Reward)
21     (a : Action) (n : Nat) (omega : Omega) :
22     sumRewards (action omega) (reward omega) a n =
23       (Finset.range n).sum
24         (fun t : Nat =>
25           (({omega' : Omega | action omega' t = a} : Set Omega).indicator
26             (fun omega' : Omega => reward omega' t)) omega) := by

28     sumRewards (action omega) (reward omega) a n
29         =
30       ((Finset.range n).filter
31         (fun t : Nat => action omega t = a)).sum

37               (a := a)

40       (Finset.range n).sum

42           (({omega' : Omega | action omega' t = a} : Set Omega).indicator

60 theorem measurable_sumRewards
61     {Omega : Type u} {Action : Type v} {Reward : Type v}
62     [MeasurableSpace Omega] [MeasurableSpace Action]
63     [MeasurableSingletonClass Action]
64     [MeasurableSpace Reward] [AddCommMonoid Reward] [MeasurableAdd₂ Reward]
65     [DecidableEq Action]
66     (action : Omega -> ActionTrace Action)
67     (reward : Omega -> RewardTrace Reward)
68     (haction : forall t : Nat, Measurable (fun omega : Omega => action omega t))
69     (hreward : forall t : Nat, Measurable (fun omega : Omega => reward omega t))
70     (a : Action) (n : Nat) :
71     Measurable
72       (fun omega : Omega => sumRewards (action omega) (reward omega) a n) := by
73   have hsum :
74       Measurable
75         (fun omega : Omega =>
76           (Finset.range n).sum
77             (fun t : Nat =>
78               (({omega' : Omega | action omega' t = a} : Set Omega).indicator
79                 (fun omega' : Omega => reward omega' t)) omega)) := by
80     exact measurable_finset_sum_indicator_reward
81       action reward haction hreward a (Finset.range n)
82   have hfun :
83       (fun omega : Omega => sumRewards (action omega) (reward omega) a n)
84         =
85       (fun omega : Omega =>
86         (Finset.range n).sum
87           (fun t : Nat =>
88             (({omega' : Omega | action omega' t = a} : Set Omega).indicator
89               (fun omega' : Omega => reward omega' t)) omega)) := by
90     funext omega
91     exact sumRewards_eq_finset_range_indicator_reward action reward a n omega
92   rw [hfun]
93   exact hsum
