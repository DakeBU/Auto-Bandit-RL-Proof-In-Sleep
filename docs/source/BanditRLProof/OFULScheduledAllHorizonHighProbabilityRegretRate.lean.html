29           ((horizon + 2 : Nat) : Real)) / delta)

121         (Feature := Feature) R delta lambda S horizon L2 =
122       telescopingHighProbabilityPseudoRegretBound
123         (Feature := Feature) R delta lambda S horizon L2 := by
124   have hden_pos :
125       0 < (((horizon + 2 : Nat) : Real)) := by
126     positivity
