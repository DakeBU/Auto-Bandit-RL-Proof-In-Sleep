3 /-!
4 # Learning-rate tuning for exponential-square realized EXP3 regret
5 
6 The exponential mixed-square route replaces the Markov square threshold by
7 `K * T + sampledMixedSquaredConfidenceRadius`. This module chooses the exact
8 learning rate
9 
10 `eta = sqrt (log K / (K * T + squareRadius))`
11 
12 at the public four-event allocation. It balances entropy against the complete
13 exponential-square stability scale. The exploration, two Bernstein, and
14 realized-deviation contributions remain explicit for a later gamma schedule.
15 -/

23 /-- Positive scale appearing in the exponential-square Hedge term when the
24 square event receives `delta / 4`. -/
25 noncomputable def exponentialSquareHighProbabilityScale
26     {Action : Type v} (arms : Finset Action) (gamma : Real)
27     (horizon : Nat) (delta : Real) : Real :=
28   (arms.card : Real) * (horizon : Real) +
29     sampledMixedSquaredConfidenceRadius arms gamma horizon (delta / 4)
30 
31 theorem exponentialSquareHighProbabilityScale_pos
32     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
33     (gamma : Real) (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
34     0 < exponentialSquareHighProbabilityScale arms gamma horizon delta := by
35   have hK : 0 < (arms.card : Real) := by
36     positivity
37   have hT : 0 < (horizon : Real) := by
38     exact_mod_cast hhorizon
39   have hbase : 0 < (arms.card : Real) * (horizon : Real) := mul_pos hK hT
40   have hradius :
41       0 <= sampledMixedSquaredConfidenceRadius
42         arms gamma horizon (delta / 4) := by
43     exact Real.sqrt_nonneg _
44   dsimp [exponentialSquareHighProbabilityScale]
45   linarith

47 /-- Learning rate balancing entropy against the complete exponential-square
48 stability scale at the public `delta / 4` square allocation. -/
49 noncomputable def exponentialSquareHighProbabilityLearningRate
50     {Action : Type v} (arms : Finset Action) (gamma : Real)
51     (horizon : Nat) (delta : Real) : Real :=
52   Real.sqrt
53     (Real.log (arms.card : Real) /
54       exponentialSquareHighProbabilityScale arms gamma horizon delta)
55 
56 theorem exponentialSquareHighProbabilityLearningRate_pos
57     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
58     (gamma : Real) (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
59     0 < exponentialSquareHighProbabilityLearningRate
60       arms gamma horizon delta := by
61   have hK_one : 1 < (arms.card : Real) := by
62     exact_mod_cast hcard_two
63   have hscale := exponentialSquareHighProbabilityScale_pos
64     arms hcard_two gamma horizon hhorizon delta
65   apply Real.sqrt_pos.2
66   exact div_pos (Real.log_pos hK_one) hscale
67 
68 theorem exponentialSquareHighProbabilityLearningRate_sq_mul_scale
69     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
70     (gamma : Real) (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
71     exponentialSquareHighProbabilityLearningRate arms gamma horizon delta ^ 2 *
72         exponentialSquareHighProbabilityScale arms gamma horizon delta =
73       Real.log (arms.card : Real) := by
74   have hK_one : 1 < (arms.card : Real) := by
75     exact_mod_cast hcard_two
76   have hscale := exponentialSquareHighProbabilityScale_pos
77     arms hcard_two gamma horizon hhorizon delta
78   unfold exponentialSquareHighProbabilityLearningRate
79   rw [Real.sq_sqrt
80     (div_nonneg (Real.log_pos hK_one).le hscale.le)]
81   field_simp [ne_of_gt hscale]
82 
83 /-- With `gamma <= 1/2`, entropy and the stability-amplified exponential-square
84 scale cost at most three copies of their balanced square-root scale. -/
85 theorem exponentialSquareHighProbabilityHedgeBudget_le_three_mul_sqrt
86     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
87     (gamma : Real) (hgamma_pos : 0 < gamma)
88     (hgamma_le_half : gamma <= 1 / 2)
89     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
90     Real.log (arms.card : Real) /
91           exponentialSquareHighProbabilityLearningRate
92             arms gamma horizon delta +
93         (exponentialSquareHighProbabilityLearningRate
94             arms gamma horizon delta * (1 / (1 - gamma))) *
95           exponentialSquareHighProbabilityScale arms gamma horizon delta <=
96       3 * Real.sqrt
97         (Real.log (arms.card : Real) *
98           exponentialSquareHighProbabilityScale arms gamma horizon delta) := by
99   let eta := exponentialSquareHighProbabilityLearningRate
100     arms gamma horizon delta
101   let scale := exponentialSquareHighProbabilityScale
102     arms gamma horizon delta
103   let entropy := Real.log (arms.card : Real) / eta
104   have hK_one : 1 < (arms.card : Real) := by
105     exact_mod_cast hcard_two
106   have heta : 0 < eta := by
107     dsimp [eta]
108     exact exponentialSquareHighProbabilityLearningRate_pos
109       arms hcard_two gamma horizon hhorizon delta
110   have hscale : 0 < scale := by
111     dsimp [scale]
112     exact exponentialSquareHighProbabilityScale_pos
113       arms hcard_two gamma horizon hhorizon delta
114   have hbalance : eta ^ 2 * scale = Real.log (arms.card : Real) := by
115     dsimp [eta, scale]
116     exact exponentialSquareHighProbabilityLearningRate_sq_mul_scale
117       arms hcard_two gamma horizon hhorizon delta
118   have hentropy_nonneg : 0 <= entropy := by
119     exact div_nonneg (Real.log_pos hK_one).le heta.le
120   have hentropy_eq_unscaled : entropy = eta * scale := by
121     dsimp [entropy]
122     field_simp [ne_of_gt heta]
123     nlinarith [hbalance]
124   have hden : 0 < 1 - gamma := by linarith
125   have hfactor : 1 <= 2 * (1 - gamma) := by linarith
126   have hstable :
127       (eta * (1 / (1 - gamma))) * scale <= 2 * entropy := by
128     rw [show (eta * (1 / (1 - gamma))) * scale =
129       (eta * scale) / (1 - gamma) by ring]
130     rw [← hentropy_eq_unscaled, div_le_iff₀ hden]
131     have hmul := mul_le_mul_of_nonneg_left hfactor hentropy_nonneg
132     nlinarith
133   have hbalanced_sq :
134       entropy ^ 2 = Real.log (arms.card : Real) * scale := by
135     rw [hentropy_eq_unscaled]
136     calc
137       (eta * scale) ^ 2 = (eta ^ 2 * scale) * scale := by ring
138       _ = Real.log (arms.card : Real) * scale := by rw [hbalance]
139   have hradicand_nonneg :
140       0 <= Real.log (arms.card : Real) * scale :=
141     mul_nonneg (Real.log_pos hK_one).le hscale.le
142   have hentropy_eq_sqrt :
143       entropy = Real.sqrt
144         (Real.log (arms.card : Real) * scale) := by
145     have hsqrt_nonneg := Real.sqrt_nonneg
146       (Real.log (arms.card : Real) * scale)
147     have hsqrt_sq := Real.sq_sqrt hradicand_nonneg
148     nlinarith [hbalanced_sq]
149   change entropy + (eta * (1 / (1 - gamma))) * scale <=
150     3 * Real.sqrt (Real.log (arms.card : Real) * scale)
151   rw [← hentropy_eq_sqrt]
152   linarith

154 /-- Explicit threshold after tuning all learning-rate-dependent terms. The
155 exploration and three confidence contributions remain visible. -/
156 noncomputable def exponentialSquareBernsteinRealizedTunedThreshold
157     {Action : Type v} (arms : Finset Action) (gamma : Real)
158     (horizon : Nat) (delta : Real) : Real :=
159   3 * Real.sqrt
160       (Real.log (arms.card : Real) *
161         exponentialSquareHighProbabilityScale arms gamma horizon delta) +
162     gamma * (horizon : Real) +
163     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
164       arms gamma horizon (delta / 4) +
165     sampledComparatorEstimatorBernsteinConfidenceRadius
166       arms gamma horizon (delta / 4) +
167     sampledPredictableRealizedDeviationConfidenceRadius horizon (delta / 4)
168 
169 /-- The complete exponential-square four-event realized budget is bounded by
170 the learning-rate-tuned threshold. -/
171 theorem sampledPredictableExponentialSquareBernsteinRealizedHighProbabilityRegretBudget_le_tunedThreshold
172     {Action : Type v} [DecidableEq Action]
173     (arms : Finset Action) (hcard_two : 2 <= arms.card)
174     (gamma : Real) (hgamma_pos : 0 < gamma)
175     (hgamma_le_half : gamma <= 1 / 2)
176     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
177     sampledPredictableExponentialSquareBernsteinRealizedHighProbabilityRegretBudget
178         arms
179         (exponentialSquareHighProbabilityLearningRate
180           arms gamma horizon delta)
181         gamma horizon (delta / 4) (delta / 4) (delta / 4) <=
182       exponentialSquareBernsteinRealizedTunedThreshold
183         arms gamma horizon delta := by
184   have hhedge :=
185     exponentialSquareHighProbabilityHedgeBudget_le_three_mul_sqrt
186       arms hcard_two gamma hgamma_pos hgamma_le_half horizon hhorizon delta
187   dsimp [exponentialSquareHighProbabilityScale] at hhedge
188   dsimp [
189     sampledPredictableExponentialSquareBernsteinRealizedHighProbabilityRegretBudget,
190     sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget,
191     exponentialSquareBernsteinRealizedTunedThreshold,
192     exponentialSquareHighProbabilityScale]
193   linarith

195 /-- Generated realized-regret tail with the exponential-square-balanced
196 learning rate. This removes the old Markov tuning and leaves only the gamma
197 schedule and confidence-radius simplification for downstream consumers. -/
198 theorem sampledPredictable_tunedExponentialSquareBernsteinRealizedRegret_tail
199     {Env : Type u} {Action : Type v}
200     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
201     [MeasurableSpace Action] [MeasurableSingletonClass Action]
202     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
203     (prior : Measure Env) [IsProbabilityMeasure prior]
204     (arms : Finset Action) (harms : arms.Nonempty)
205     (hcard_two : 2 <= arms.card)
206     (gamma : Real) (hgamma_pos : 0 < gamma)
207     (hgamma_le_half : gamma <= 1 / 2)
208     (loss : PredictableLossVector Env Action)
209     (comparator : Action) (hcomparator : comparator ∈ arms)
210     (horizon : Nat) (hhorizon : 0 < horizon)
211     (delta : Real) (hdelta : 0 < delta) :
212     let eta := exponentialSquareHighProbabilityLearningRate
213       arms gamma horizon delta
214     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
215       eta gamma hgamma_pos.le (by linarith : gamma <= 1) loss.environment
216     mu {sample |
217         exponentialSquareBernsteinRealizedTunedThreshold
218             arms gamma horizon delta <=
219           (Finset.range horizon).sum (fun t =>
220               sampledTrajectoryRealizedLossAt t sample) -
221             (Finset.range horizon).sum (fun t =>
222               predictableLossAt loss t sample comparator)} <=
223       ENNReal.ofReal delta := by
224   dsimp only
225   have heta :
226       0 < exponentialSquareHighProbabilityLearningRate
227         arms gamma horizon delta :=
228     exponentialSquareHighProbabilityLearningRate_pos
229       arms hcard_two gamma horizon hhorizon delta
230   have hgamma_lt_one : gamma < 1 := by linarith
231   have hbudget :=
232     sampledPredictableExponentialSquareBernsteinRealizedHighProbabilityRegretBudget_le_tunedThreshold
233       arms hcard_two gamma hgamma_pos hgamma_le_half horizon hhorizon delta
234   have htail :=
235     sampledPredictable_exponentialSquareBernsteinRealizedHighProbabilityRegret_tail_total_delta
236       prior arms harms
237         (exponentialSquareHighProbabilityLearningRate
238           arms gamma horizon delta)
239         gamma heta hgamma_pos hgamma_lt_one loss comparator hcomparator
240         horizon hhorizon delta hdelta
241   dsimp only at htail ⊢
242   calc
243     (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
244         (exponentialSquareHighProbabilityLearningRate
245           arms gamma horizon delta)
246         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
247         {sample |
248           exponentialSquareBernsteinRealizedTunedThreshold
249               arms gamma horizon delta <=
250             (Finset.range horizon).sum (fun t =>
251                 sampledTrajectoryRealizedLossAt t sample) -
252               (Finset.range horizon).sum (fun t =>
253                 predictableLossAt loss t sample comparator)} <=
254       (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
255         (exponentialSquareHighProbabilityLearningRate
256           arms gamma horizon delta)
257         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
258         {sample |
259           sampledPredictableExponentialSquareBernsteinRealizedHighProbabilityRegretBudget
260               arms
261                 (exponentialSquareHighProbabilityLearningRate
262                   arms gamma horizon delta)
263               gamma horizon (delta / 4) (delta / 4) (delta / 4) <=
264             (Finset.range horizon).sum (fun t =>
265                 sampledTrajectoryRealizedLossAt t sample) -
266               (Finset.range horizon).sum (fun t =>
267                 predictableLossAt loss t sample comparator)} := by
268         apply measure_mono
269         intro sample hsample
270         exact hbudget.trans hsample
271     _ <= ENNReal.ofReal delta := htail
