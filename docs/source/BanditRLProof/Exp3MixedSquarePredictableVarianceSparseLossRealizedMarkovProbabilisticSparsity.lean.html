3 /-!
4 # Probabilistic sparse-loss control of realized predictable-variance EXP3 regret
5 
6 This module allows the per-round support-cardinality contract to fail on an
7 explicit generated-trajectory event. On paths outside that event, sparsity
8 still supplies the `sparsity * horizon` loss-mass budget used by the observed
9 mixed-square and Hedge terms. The Markov variance closure instead uses the
10 global `arms.card * horizon` loss-mass envelope, so the exceptional event costs
11 its actual generated-measure probability without assuming it is null.
12 
13 The resulting theorem is a caller-parameterized `delta + epsilon` route. It
14 does not retain the sharper sparse variance mean on exceptional paths and does
15 not claim the tuned all-horizon `14 * gamma * horizon` threshold.
16 -/

24 /-- Generated trajectories on which the requested per-round support cap fails
25 at some time before the horizon. -/
26 noncomputable def sampledPredictableSparsityFailure
27     {Env : Type u} {Action : Type v}
28     [MeasurableSpace Env] [MeasurableSpace Action]
29     (arms : Finset Action) (loss : PredictableLossVector Env Action)
30     (horizon sparsity : Nat) :
31     Set (Env × ((k : Nat) → Action × Real)) :=
32   {sample | ∃ t, t < horizon ∧
33     sparsity < (sampledPredictableLossSupport arms loss t sample).card}

35 /-- Every trajectory either obeys the sparse armwise loss-mass budget or lies
36 in the explicit sparsity-failure event. -/
37 theorem sampledPredictableLossMassSum_le_or_mem_sparsityFailure
38     {Env : Type u} {Action : Type v}
39     [MeasurableSpace Env] [MeasurableSpace Action]
40     (arms : Finset Action) (loss : PredictableLossVector Env Action)
41     (horizon sparsity : Nat)
42     (sample : Env × ((k : Nat) → Action × Real)) :
43     sampledPredictableLossMassSum arms loss horizon sample ≤
44         (sparsity : Real) * (horizon : Real) ∨
45       sample ∈ sampledPredictableSparsityFailure arms loss horizon sparsity := by
46   classical
47   by_cases hsparse : ∀ t, t < horizon →
48       (sampledPredictableLossSupport arms loss t sample).card ≤ sparsity
49   · exact Or.inl
50       (sampledPredictableLossMassSum_le_sparsity_mul_horizon_of_sample
51         arms loss horizon sparsity sample hsparse)
52   · right
53     simp only [sampledPredictableSparsityFailure, Set.mem_setOf_eq]
54     push Not at hsparse
55     exact hsparse

57 /-- The number of nonzero active coordinates is always at most the number of
58 active arms, giving a global pathwise loss-mass envelope. -/
59 theorem sampledPredictableLossMassSum_le_card_mul_horizon
60     {Env : Type u} {Action : Type v}
61     [MeasurableSpace Env] [MeasurableSpace Action]
62     (arms : Finset Action) (loss : PredictableLossVector Env Action)
63     (horizon : Nat) (sample : Env × ((k : Nat) → Action × Real)) :
64     sampledPredictableLossMassSum arms loss horizon sample ≤
65       (arms.card : Real) * (horizon : Real) := by
66   classical
67   apply sampledPredictableLossMassSum_le_sparsity_mul_horizon_of_sample
68   intro t _ht
69   exact Finset.card_le_card (Finset.filter_subset _ _)

71 /-- Global predictable-variance mean envelope used when sparsity may fail with
72 positive probability. -/
73 noncomputable def sampledPredictableGlobalVarianceMeanBudget
74     {Action : Type v} [DecidableEq Action]
75     (arms : Finset Action) (gamma : Real) (horizon : Nat) : Real :=
76   (1 / (gamma / (arms.card : Real))) *
77     ((arms.card : Real) * (horizon : Real))
