3 /-!
4 # Expected scheduled half-Tsallis stability at arbitrary positive rates
5 
6 This module supplies the coarse branch omitted by the refined scheduled
7 stability theorem.  Ordinary importance weighting has one-round expected
8 conjugate-potential stability at most one for every positive local rate.  The
9 final generated-trajectory theorem combines that fallback with the refined
10 bound available when the local ABRL rate is at most `1 / 2`.
11 -/

20 /-- Comparing the old objective at its minimizer with the updated minimizer
21 reduces conjugate-potential stability to a difference of linear losses. -/
22 theorem halfTsallisPotentialStability_le_linearLoss_sub_of_minimizers
23     {Action : Type u}
24     (arms : Finset Action) (eta : Real)
25     (score probability estimate next : Action -> Real)
26     (heta : 0 < eta)
27     (hprobabilityMin : FTRL.IsRegularizedMinimizer
28       (FTRL.finiteSimplex arms) arms eta
29       (negEntropyRegularizer arms (1 / 2 : Real)) score probability)
30     (hnextMin : FTRL.IsRegularizedMinimizer
31       (FTRL.finiteSimplex arms) arms eta
32       (negEntropyRegularizer arms (1 / 2 : Real))
33       (fun action => score action + estimate action) next) :
34     halfTsallisPotentialStability
35         arms eta score probability estimate next <=
36       FTRL.linearLoss arms probability estimate -
37         FTRL.linearLoss arms next estimate := by
38   have hmin := hprobabilityMin.2 next hnextMin.1
39   have hobjective :
40       FTRL.regularizedObjective arms eta
41           (negEntropyRegularizer arms (1 / 2 : Real))
42           (fun action => score action + estimate action) next =
43         FTRL.regularizedObjective arms eta
44             (negEntropyRegularizer arms (1 / 2 : Real)) score next +
45           eta * FTRL.linearLoss arms next estimate := by
46     unfold FTRL.regularizedObjective
47     rw [FTRL.linearLoss_add_right]
48     ring
49   have hdiv :
50       (FTRL.regularizedObjective arms eta
51           (negEntropyRegularizer arms (1 / 2 : Real)) score probability -
52         FTRL.regularizedObjective arms eta
53           (negEntropyRegularizer arms (1 / 2 : Real)) score next) / eta <= 0 :=
54     div_nonpos_of_nonpos_of_nonneg (sub_nonpos.mpr hmin) heta.le
55   unfold halfTsallisPotentialStability halfTsallisPotentialValue
56   rw [hobjective]
57   calc
58     _ =
59       FTRL.linearLoss arms probability estimate -
60           FTRL.linearLoss arms next estimate +
61         (FTRL.regularizedObjective arms eta
62             (negEntropyRegularizer arms (1 / 2 : Real)) score probability -
63           FTRL.regularizedObjective arms eta
64             (negEntropyRegularizer arms (1 / 2 : Real)) score next) / eta := by
65       field_simp [ne_of_gt heta]
66       ring
67     _ <= FTRL.linearLoss arms probability estimate -
68         FTRL.linearLoss arms next estimate := by linarith
