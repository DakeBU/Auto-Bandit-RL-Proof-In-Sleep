5 # Gaussian mixture identities for OFUL
6 
7 This module develops the exact Gaussian quadratic-exponential integrals used
8 after the fixed-direction exponential-supermartingale bound. The scalar
9 identity is the normalization step needed before the finite-dimensional
10 spectral/product assembly.
11 -/

20 /--
21 The exact scalar quadratic-exponential integral under a standard Gaussian.
22 
23 This is the one-coordinate completed-square identity used by the Gaussian
24 method of mixtures. The nonnegative quadratic coefficient is the scalar
25 eigenvalue contract that will arise from a positive-semidefinite Gram matrix.
26 -/
27 theorem integral_exp_linear_sub_quadratic_gaussianReal_zero_one
28     (s q : Real) (hq : 0 <= q) :
29     integral
30         (ProbabilityTheory.gaussianReal 0 1)
31         (fun x : Real => Real.exp (s * x - q * x ^ 2 / 2)) =
32       (Real.sqrt (1 + q))⁻¹ *
33         Real.exp (s ^ 2 / (2 * (1 + q))) := by
34   let a : Real := 1 + q
35   have ha : 0 < a := by
36     dsimp [a]
37     linarith
38   let v : NNReal := ⟨a⁻¹, inv_nonneg.mpr ha.le⟩
39   have hv : v ≠ 0 := by
40     intro hv0
41     have hcoe : (v : Real) = 0 := by rw [hv0]; rfl
42     dsimp [v] at hcoe
43     exact (inv_ne_zero (ne_of_gt ha)) hcoe
44   let m : Real := s / a
45   rw [ProbabilityTheory.integral_gaussianReal_eq_integral_smul
46     (by norm_num : (1 : NNReal) ≠ 0)]
47   simp only [smul_eq_mul]
48   have hpoint : forall x : Real,
49       ProbabilityTheory.gaussianPDFReal 0 1 x *
50           Real.exp (s * x - q * x ^ 2 / 2) =
51         (Real.sqrt a)⁻¹ * Real.exp (s ^ 2 / (2 * a)) *
52           ProbabilityTheory.gaussianPDFReal m v x := by
53     intro x
54     rw [ProbabilityTheory.gaussianPDFReal_def,
55       ProbabilityTheory.gaussianPDFReal_def]
56     dsimp [m, v]
57     have ha0 : a ≠ 0 := ne_of_gt ha
58     have hsqrt_a : Real.sqrt a ≠ 0 := ne_of_gt (Real.sqrt_pos.2 ha)
59     have hsqrt_two_pi : Real.sqrt (2 * Real.pi) ≠ 0 := by
60       positivity
61     have hsqrt_div :
62         Real.sqrt (2 * Real.pi * a⁻¹) =
63           Real.sqrt (2 * Real.pi) / Real.sqrt a := by
64       rw [show 2 * Real.pi * a⁻¹ = (2 * Real.pi) / a by
65         field_simp]
66       exact Real.sqrt_div (by positivity : 0 <= 2 * Real.pi) a
67     simp only [mul_one, sub_zero, hsqrt_div]
68     have hconst :
69         (Real.sqrt (2 * Real.pi))⁻¹ =
70           (Real.sqrt a)⁻¹ *
71             (Real.sqrt (2 * Real.pi) / Real.sqrt a)⁻¹ := by
72       field_simp
73     have hexponent :
74         -x ^ 2 / 2 + (s * x - q * x ^ 2 / 2) =
75           s ^ 2 / (2 * a) -
76             (x - s / a) ^ 2 / (2 * a⁻¹) := by
77       have haq : 1 + q ≠ 0 := by linarith
78       dsimp [a]
79       field_simp [haq]
80       ring
81     calc
82       (Real.sqrt (2 * Real.pi))⁻¹ * Real.exp (-x ^ 2 / 2) *
83             Real.exp (s * x - q * x ^ 2 / 2) =
84           (Real.sqrt a)⁻¹ *
85             (Real.sqrt (2 * Real.pi) / Real.sqrt a)⁻¹ *
86               (Real.exp (-x ^ 2 / 2) *
87                 Real.exp (s * x - q * x ^ 2 / 2)) := by
88         rw [hconst]
89         ring
90       _ = (Real.sqrt a)⁻¹ *
91             (Real.sqrt (2 * Real.pi) / Real.sqrt a)⁻¹ *
92               Real.exp (-x ^ 2 / 2 + (s * x - q * x ^ 2 / 2)) := by
93         rw [Real.exp_add]
94       _ = (Real.sqrt a)⁻¹ *
95             (Real.sqrt (2 * Real.pi) / Real.sqrt a)⁻¹ *
96               Real.exp
97                 (s ^ 2 / (2 * a) -
98                   (x - s / a) ^ 2 / (2 * a⁻¹)) := by
99         rw [hexponent]
100       _ = (Real.sqrt a)⁻¹ * Real.exp (s ^ 2 / (2 * a)) *
101             ((Real.sqrt (2 * Real.pi) / Real.sqrt a)⁻¹ *
102               Real.exp (-(x - s / a) ^ 2 / (2 * a⁻¹))) := by
103         rw [show
104           s ^ 2 / (2 * a) - (x - s / a) ^ 2 / (2 * a⁻¹) =
105             s ^ 2 / (2 * a) +
106               (-(x - s / a) ^ 2 / (2 * a⁻¹)) by ring,
107           Real.exp_add]
108         ring
109   simp_rw [hpoint]
110   rw [MeasureTheory.integral_const_mul,
111     ProbabilityTheory.integral_gaussianPDFReal_eq_one m hv, mul_one]

113 /--
114 Finite independent-coordinate version of the scalar completed-square identity.
115 
116 The right-hand side is deliberately left as a finite product. The following
117 matrix-facing lemmas identify that product with the square-root determinant
118 factor and the exponential term with an inverse-diagonal quadratic form.
119 -/
120 theorem integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal
121     {Feature : Type*} [Fintype Feature]
122     (score quadratic : Feature -> Real)
123     (hquadratic : forall i, 0 <= quadratic i) :
124     integral
125         (Measure.pi (fun _ : Feature =>
126           ProbabilityTheory.gaussianReal 0 1))
127         (fun z : Feature -> Real =>
128           Real.exp
129             (Finset.univ.sum (fun i =>
130               score i * z i - quadratic i * (z i) ^ 2 / 2))) =
131       Finset.univ.prod (fun i =>
132         (Real.sqrt (1 + quadratic i))⁻¹ *
133           Real.exp (score i ^ 2 / (2 * (1 + quadratic i)))) := by
134   simp_rw [Real.exp_sum]
135   have hprod :
136       integral
137           (Measure.pi (fun _ : Feature =>
138             ProbabilityTheory.gaussianReal 0 1))
139           (fun z : Feature -> Real =>
140             Finset.univ.prod (fun i =>
141               Real.exp
142                 (score i * z i - quadratic i * (z i) ^ 2 / 2))) =
143         Finset.univ.prod (fun i =>
144           integral
145             (ProbabilityTheory.gaussianReal 0 1)
146             (fun x : Real =>
147               Real.exp
148                 (score i * x - quadratic i * x ^ 2 / 2))) := by
149     exact MeasureTheory.integral_fintype_prod_eq_prod
150       (fun i (x : Real) =>
151         Real.exp (score i * x - quadratic i * x ^ 2 / 2))
152   rw [hprod]
153   congr with i
154   exact integral_exp_linear_sub_quadratic_gaussianReal_zero_one
155     (score i) (quadratic i) (hquadratic i)

157 /--
158 Finite product identity with the normalization and exponential factors
159 collected separately.
160 -/
161 theorem integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal_eq
162     {Feature : Type*} [Fintype Feature]
163     (score quadratic : Feature -> Real)
164     (hquadratic : forall i, 0 <= quadratic i) :
165     integral
166         (Measure.pi (fun _ : Feature =>
167           ProbabilityTheory.gaussianReal 0 1))
168         (fun z : Feature -> Real =>
169           Real.exp
170             (Finset.univ.sum (fun i =>
171               score i * z i - quadratic i * (z i) ^ 2 / 2))) =
172       (Finset.univ.prod (fun i =>
173         Real.sqrt (1 + quadratic i)))⁻¹ *
174         Real.exp
175           (Finset.univ.sum (fun i =>
176             score i ^ 2 / (2 * (1 + quadratic i)))) := by
177   rw [integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal
178     score quadratic hquadratic]
179   rw [Finset.prod_mul_distrib, Finset.prod_inv_distrib, ← Real.exp_sum]

181 /--
182 Diagonal matrix form of the finite Gaussian quadratic-exponential identity.
183 
184 This is the diagonal-coordinate determinant-ratio and inverse-quadratic
185 expression intended for later transport through a PSD matrix eigenbasis.
186 -/
187 theorem integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal_det
188     {Feature : Type*} [Fintype Feature] [DecidableEq Feature]
189     (score quadratic : Feature -> Real)
190     (hquadratic : forall i, 0 <= quadratic i) :
191     integral
192         (Measure.pi (fun _ : Feature =>
193           ProbabilityTheory.gaussianReal 0 1))
194         (fun z : Feature -> Real =>
195           Real.exp
196             (Finset.univ.sum (fun i =>
197               score i * z i - quadratic i * (z i) ^ 2 / 2))) =
198       (Real.sqrt
199         (Matrix.det
200           (Matrix.diagonal (fun i : Feature =>
201             1 + quadratic i))))⁻¹ *
202         Real.exp
203           (score ⬝ᵥ
204             ((Matrix.diagonal (fun i : Feature =>
205               1 + quadratic i))⁻¹).mulVec score / 2) := by
206   rw [integral_exp_sum_linear_sub_diagonal_quadratic_pi_gaussianReal_eq
207     score quadratic hquadratic]
208   have hdiag_nonneg : forall i : Feature, 0 <= 1 + quadratic i := by
209     intro i
210     linarith [hquadratic i]
211   have hdiag_ne : forall i : Feature, 1 + quadratic i ≠ 0 := by
212     intro i
213     linarith [hquadratic i]
214   have hdet :
215       Real.sqrt
216           (Matrix.det
217             (Matrix.diagonal (fun i : Feature =>
218               1 + quadratic i))) =
219         Finset.univ.prod (fun i =>
220           Real.sqrt (1 + quadratic i)) := by
221     rw [Matrix.det_diagonal]
222     exact Real.sqrt_prod Finset.univ (fun i _ => hdiag_nonneg i)
223   have hdiag_unit :
224       IsUnit (fun i : Feature => 1 + quadratic i) := by
225     rw [Pi.isUnit_iff]
226     intro i
227     exact isUnit_iff_ne_zero.mpr (hdiag_ne i)
228   have hdiag_inverse :
229       Ring.inverse (fun i : Feature => 1 + quadratic i) =
230         fun i => (1 + quadratic i)⁻¹ := by
231     funext i
232     apply eq_inv_of_mul_eq_one_right
233     simpa only [Pi.mul_apply, Pi.one_apply] using
234       congrFun
235         (Ring.mul_inverse_cancel
236           (fun i : Feature => 1 + quadratic i) hdiag_unit) i
237   have hquad :
238       score ⬝ᵥ
239           ((Matrix.diagonal (fun i : Feature =>
240             1 + quadratic i))⁻¹).mulVec score =
241         Finset.univ.sum (fun i =>
242           score i ^ 2 / (1 + quadratic i)) := by
243     rw [Matrix.inv_diagonal, hdiag_inverse]
244     simp only [dotProduct]
245     apply Finset.sum_congr rfl
246     intro i hi
247     rw [Matrix.mulVec_diagonal]
248     ring
249   have hsum :
250       Finset.univ.sum (fun i =>
251           score i ^ 2 / (2 * (1 + quadratic i))) =
252         Finset.univ.sum (fun i =>
253           score i ^ 2 / (1 + quadratic i)) / 2 := by
254     rw [Finset.sum_div]
255     apply Finset.sum_congr rfl
256     intro i hi
257     field_simp [hdiag_ne i]
258   rw [hdet, hquad, hsum]
