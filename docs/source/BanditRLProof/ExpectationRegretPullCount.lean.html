7 # Bochner expected-regret pull-count decomposition
8 
9 This module lifts the deterministic `REGRET-PULLCOUNT` equality to a
10 Real-valued Bochner expectation statement.  It stays at the bookkeeping layer:
11 the only probabilistic regularity assumption is integrability of each finite
12 horizon pull-count random variable after casting to `Real`.
13 -/

21 private theorem real_pseudoRegret_eq_univ_sum_gap_mul_natCast_pullCount
22     {K : Nat}
23     (model : FiniteBanditModel K)
24     (action : ActionTrace (Fin K))
25     (n : Nat) :
26     ((pseudoRegret model action n : Rat) : Real) =
27       (Finset.univ : Finset (Fin K)).sum
28         (fun a : Fin K =>
29           ((model.gap a : Rat) : Real) *
30             ((pullCount action a n : Nat) : Real)) := by

33 /--
34 The Real cast of a finite-horizon pull count is integrable under any finite
35 measure when the action trace is measurable one time coordinate at a time.
36 
37 This is the generic regularity adapter behind Real expected-regret wrappers:
38 measurability comes from `measurable_natCast_pullCount`, while
39 `pullCount_le_time` supplies the deterministic integrable bound.
40 -/
41 theorem integrable_real_pullCount_of_measurable_action
42     {Omega : Type u} {Action : Type*}
43     [MeasurableSpace Omega] [MeasurableSpace Action]
44     [MeasurableSingletonClass Action] [DecidableEq Action]
45     (mu : Measure Omega) [MeasureTheory.IsFiniteMeasure mu]
46     (action : Omega -> ActionTrace Action)
47     (haction : forall t : Nat,
48       Measurable (fun omega : Omega => action omega t))
49     (arm : Action) (n : Nat) :
50     Integrable
51       (fun omega : Omega =>
52         ((pullCount (action omega) arm n : Nat) : Real)) mu := by
53   have hmeas :
54       Measurable
55         (fun omega : Omega =>
56           ((pullCount (action omega) arm n : Nat) : Real)) :=
57     measurable_natCast_pullCount action haction arm n
58   refine Integrable.of_bound hmeas.aestronglyMeasurable (n : Real) ?_
59   filter_upwards [] with omega
60   rw [Real.norm_eq_abs, abs_of_nonneg (Nat.cast_nonneg _)]
61   exact_mod_cast pullCount_le_time (action omega) arm n
62 
63 /--
64 If every finite-horizon pull count is integrable after casting to `Real`, then
65 the corresponding Real-valued pseudo-regret is integrable.
66 
67 This is the regularity adapter used by the Bochner expected-regret
68 decomposition.  It does not prove measurability or integrability from a policy
69 model; callers provide the pull-count integrability witnesses.
70 -/
71 theorem integrable_real_pseudoRegret_of_integrable_pullCount
72     {Omega : Type u} {K : Nat}
73     [MeasurableSpace Omega]
74     (mu : Measure Omega)
75     (model : FiniteBanditModel K)
76     (action : Omega -> ActionTrace (Fin K))
77     (n : Nat)
78     (hcount : forall a : Fin K,
79       Integrable
80         (fun omega : Omega =>
81           ((pullCount (action omega) a n : Nat) : Real)) mu) :
82     Integrable
83       (fun omega : Omega =>
84         ((pseudoRegret model (action omega) n : Rat) : Real)) mu := by
85   let term : Fin K -> Omega -> Real :=
86     fun a omega =>
87       ((model.gap a : Rat) : Real) *
88         ((pullCount (action omega) a n : Nat) : Real)
89   have hterm : forall a : Fin K, Integrable (term a) mu := by
90     intro a
91     exact (hcount a).const_mul ((model.gap a : Rat) : Real)
92   have hsum :
93       Integrable
94         (fun omega : Omega =>
95           (Finset.univ : Finset (Fin K)).sum
96             (fun a : Fin K => term a omega)) mu := by
97     exact
98       IntegrabilitySums.integrable_univ_sum
99         (mu := mu)
100         (f := term)
101         (hf := hterm)
102   have hfun :
103       (fun omega : Omega =>
104           ((pseudoRegret model (action omega) n : Rat) : Real))
105         =
106       (fun omega : Omega =>
107         (Finset.univ : Finset (Fin K)).sum
108           (fun a : Fin K => term a omega)) := by
109     funext omega
110     exact
111       real_pseudoRegret_eq_univ_sum_gap_mul_natCast_pullCount
112         (model := model)
113         (action := action omega)
114         (n := n)
115   rwa [hfun]

117 /--
118 The Real-valued Bochner expectation of pseudo-regret is the finite sum of each
119 arm gap multiplied by the Bochner expectation of that arm's pull count.
120 
121 This is the local `EXP-REGRET-PULLCOUNT` leaf.  It consumes the deterministic
122 `REGRET-PULLCOUNT` bridge and the Mathlib-backed `EXP-FINITE-SUM` wrapper.
123 -/
124 theorem integral_real_pseudoRegret_eq_sum_gap_mul_integral_pullCount
125     {Omega : Type u} {K : Nat}
126     [MeasurableSpace Omega]
127     (mu : Measure Omega)
128     (model : FiniteBanditModel K)
129     (action : Omega -> ActionTrace (Fin K))
130     (n : Nat)
131     (hcount : forall a : Fin K,
132       Integrable
133         (fun omega : Omega =>
134           ((pullCount (action omega) a n : Nat) : Real)) mu) :
135     MeasureTheory.integral mu
136       (fun omega : Omega =>
137         ((pseudoRegret model (action omega) n : Rat) : Real))
138       =
139     (Finset.univ : Finset (Fin K)).sum
140       (fun a : Fin K =>
141         ((model.gap a : Rat) : Real) *
142           MeasureTheory.integral mu
143             (fun omega : Omega =>
144               ((pullCount (action omega) a n : Nat) : Real))) := by
145   let term : Fin K -> Omega -> Real :=
146     fun a omega =>
147       ((model.gap a : Rat) : Real) *
148         ((pullCount (action omega) a n : Nat) : Real)
149   have hterm : forall a : Fin K, Integrable (term a) mu := by
150     intro a
151     exact (hcount a).const_mul ((model.gap a : Rat) : Real)
152   have hfun :
153       (fun omega : Omega =>
154           ((pseudoRegret model (action omega) n : Rat) : Real))
155         =
156       (fun omega : Omega =>
157         (Finset.univ : Finset (Fin K)).sum
158           (fun a : Fin K => term a omega)) := by
159     funext omega
160     exact
161       real_pseudoRegret_eq_univ_sum_gap_mul_natCast_pullCount
162         (model := model)
163         (action := action omega)
164         (n := n)
165   calc
166     MeasureTheory.integral mu
167       (fun omega : Omega =>
168         ((pseudoRegret model (action omega) n : Rat) : Real))
169         =
170       MeasureTheory.integral mu
171         (fun omega : Omega =>
172           (Finset.univ : Finset (Fin K)).sum
173             (fun a : Fin K => term a omega)) := by
174           rw [hfun]
175     _ =
176       (Finset.univ : Finset (Fin K)).sum
177         (fun a : Fin K =>
178           MeasureTheory.integral mu (term a)) := by
179           exact
180             ExpectationBochnerSums.integral_univ_sum
181               (mu := mu)
182               (f := term)
183               (hf := hterm)
184     _ =
185       (Finset.univ : Finset (Fin K)).sum
186         (fun a : Fin K =>
187           ((model.gap a : Rat) : Real) *
188             MeasureTheory.integral mu
189               (fun omega : Omega =>
190                 ((pullCount (action omega) a n : Nat) : Real))) := by
191           apply Finset.sum_congr rfl
192           intro a _ha
193           simp [term, MeasureTheory.integral_const_mul]
