3 /-!
4 # Uniform variance proxies over finite context-action families
5 
6 This module computes one common `NNReal` proxy for a finite context space and
7 finite arm set. It is independent of any particular bandit algorithm.
8 -/
9 
10 namespace BanditRLProof
11 namespace Concentration
12 
13 /-- The largest variance proxy over a finite context-action family. -/
14 noncomputable def finiteContextArmVarianceProxy
15     {Context : Type} [Fintype Context] {K : Nat}
16     (varianceProxy : Context -> Fin K -> NNReal) : NNReal :=
17   Finset.univ.sup fun context => Finset.univ.sup (varianceProxy context)
18 
19 /-- Every context-action proxy is bounded by the finite-family maximum. -/
20 theorem varianceProxy_le_finiteContextArmVarianceProxy
21     {Context : Type} [Fintype Context] {K : Nat}
22     (varianceProxy : Context -> Fin K -> NNReal)
23     (context : Context) (arm : Fin K) :
24     varianceProxy context arm <=
25       finiteContextArmVarianceProxy varianceProxy := by
26   apply le_trans
27     (Finset.le_sup
28       (s := (Finset.univ : Finset (Fin K)))
29       (f := varianceProxy context)
30       (Finset.mem_univ arm))
31   exact Finset.le_sup
32     (s := (Finset.univ : Finset Context))
33     (f := fun context => Finset.univ.sup (varianceProxy context))
34     (Finset.mem_univ context)
35 
36 /-- A positive member makes the finite context-action maximum positive. -/
37 theorem finiteContextArmVarianceProxy_pos_of_exists
38     {Context : Type} [Fintype Context] {K : Nat}
39     (varianceProxy : Context -> Fin K -> NNReal)
40     (hpos : exists context arm,
41       0 < ((varianceProxy context arm : NNReal) : Real)) :
42     0 < ((finiteContextArmVarianceProxy varianceProxy : NNReal) : Real) := by
43   rcases hpos with ⟨context, arm, hproxy⟩
44   have hle := varianceProxy_le_finiteContextArmVarianceProxy
45     varianceProxy context arm
46   exact hproxy.trans_le (by exact_mod_cast hle)
47 
48 /--
49 The finite context-action maximum padded by one. This gives algorithms that
50 require a strictly positive tuning parameter a uniform proxy even when every
51 genuine variance proxy is zero.
52 -/
53 noncomputable def finiteContextArmPositiveVarianceProxy
54     {Context : Type} [Fintype Context] {K : Nat}
55     (varianceProxy : Context -> Fin K -> NNReal) : NNReal :=
56   max 1 (finiteContextArmVarianceProxy varianceProxy)
57 
58 /-- Every genuine proxy is bounded by the positive padded proxy. -/
59 theorem varianceProxy_le_finiteContextArmPositiveVarianceProxy
60     {Context : Type} [Fintype Context] {K : Nat}
61     (varianceProxy : Context -> Fin K -> NNReal)
62     (context : Context) (arm : Fin K) :
63     varianceProxy context arm <=
64       finiteContextArmPositiveVarianceProxy varianceProxy := by
65   exact
66     (varianceProxy_le_finiteContextArmVarianceProxy
67       varianceProxy context arm).trans (le_max_right _ _)
68 
69 /-- The padded finite context-action proxy is always strictly positive. -/
70 theorem finiteContextArmPositiveVarianceProxy_pos
71     {Context : Type} [Fintype Context] {K : Nat}
72     (varianceProxy : Context -> Fin K -> NNReal) :
73     0 < ((finiteContextArmPositiveVarianceProxy varianceProxy : NNReal) : Real) := by
74   have hone :
75       (1 : NNReal) <= finiteContextArmPositiveVarianceProxy varianceProxy :=
76     le_max_left _ _
77   exact zero_lt_one.trans_le (by exact_mod_cast hone)
78
