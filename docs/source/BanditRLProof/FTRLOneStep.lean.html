7 /-!
8 # FTRL one-step finite-sum wrapper
9 
10 This module records the deterministic optimization primitive used by FTRL/OMD
11 routes.  It consumes an explicit minimizer certificate for the regularized
12 finite-action objective and returns the one-step linear-loss inequality against
13 any feasible comparator.
14 
15 It does not prove convexity, minimizer existence, a Tsallis regularizer, a
16 stability/penalty decomposition, or a regret theorem.
17 -/
18 
19 namespace BanditRLProof
20 namespace FTRL
21 
22 universe u
23 
24 /-- Finite-action linear loss of a weight vector against a loss vector. -/
25 noncomputable def linearLoss {Action : Type u}
26     (arms : Finset Action) (p loss : Action -> Real) : Real :=
27   arms.sum (fun a => p a * loss a)
28 
29 /-- The finite probability-simplex predicate over an explicit action set. -/
30 def finiteSimplex {Action : Type u}
31     (arms : Finset Action) (p : Action -> Real) : Prop :=
32   (forall a, a ∈ arms -> 0 <= p a) ∧ arms.sum p = 1
33 
34 /--
35 Regularized one-round FTRL objective `eta * <p, loss> + R p`.
36 
37 The learning-rate positivity contract is kept on the theorem, not the
38 definition, so future leaves can reuse the objective algebraically.
39 -/
40 noncomputable def regularizedObjective {Action : Type u}
41     (arms : Finset Action) (eta : Real)
42     (regularizer : (Action -> Real) -> Real)
43     (loss : Action -> Real) (p : Action -> Real) : Real :=
44   eta * linearLoss arms p loss + regularizer p
45 
46 /-- A point `p` minimizes the regularized objective over a feasible predicate. -/
47 def IsRegularizedMinimizer {Action : Type u}
48     (feasible : (Action -> Real) -> Prop)
49     (arms : Finset Action) (eta : Real)
50     (regularizer : (Action -> Real) -> Real)
51     (loss : Action -> Real) (p : Action -> Real) : Prop :=
52   feasible p ∧
53     forall q, feasible q ->
54       regularizedObjective arms eta regularizer loss p <=
55         regularizedObjective arms eta regularizer loss q
56 
57 /--
58 FTRL one-step inequality from an explicit regularized-objective minimizer.
59 
60 The conclusion is the deterministic algebraic form used before later leaves
61 choose a concrete regularizer or prove a stability/penalty sum.
62 -/
63 theorem linearLoss_sub_le_regularizer_sub_div_of_isRegularizedMinimizer
64     {Action : Type u}
65     (feasible : (Action -> Real) -> Prop)
66     (arms : Finset Action) (eta : Real)
67     (regularizer : (Action -> Real) -> Real)
68     (loss : Action -> Real) (p q : Action -> Real)
69     (heta : 0 < eta)
70     (hp : IsRegularizedMinimizer feasible arms eta regularizer loss p)
71     (hq : feasible q) :
72     linearLoss arms p loss - linearLoss arms q loss <=
73       (regularizer q - regularizer p) / eta := by
74   set Lp := linearLoss arms p loss
75   set Lq := linearLoss arms q loss
76   set Rp := regularizer p
77   set Rq := regularizer q
78   have hmin :
79       eta * Lp + Rp <= eta * Lq + Rq := by
80     simpa [IsRegularizedMinimizer, regularizedObjective, Lp, Lq, Rp, Rq]
81       using hp.2 q hq
82   have hscaled : eta * Lp - eta * Lq <= Rq - Rp := by
83     linarith
84   have hmul : (Lp - Lq) * eta <= Rq - Rp := by
85     calc
86       (Lp - Lq) * eta = eta * Lp - eta * Lq := by
87         ring
88       _ <= Rq - Rp := hscaled
89   have hdiv : Lp - Lq <= (Rq - Rp) / eta :=
90     (le_div_iff₀ heta).2 hmul
91   simpa [Lp, Lq, Rp, Rq] using hdiv
92 
93 /-- FTRL one-step inequality specialized to the finite simplex. -/
94 theorem linearLoss_sub_le_regularizer_sub_div_of_simplex_minimizer
95     {Action : Type u}
96     (arms : Finset Action) (eta : Real)
97     (regularizer : (Action -> Real) -> Real)
98     (loss : Action -> Real) (p q : Action -> Real)
99     (heta : 0 < eta)
100     (hp : IsRegularizedMinimizer (finiteSimplex arms) arms eta
101       regularizer loss p)
102     (hq : finiteSimplex arms q) :
103     linearLoss arms p loss - linearLoss arms q loss <=
104       (regularizer q - regularizer p) / eta := by
105   exact
106     linearLoss_sub_le_regularizer_sub_div_of_isRegularizedMinimizer
107       (finiteSimplex arms) arms eta regularizer loss p q heta hp hq
