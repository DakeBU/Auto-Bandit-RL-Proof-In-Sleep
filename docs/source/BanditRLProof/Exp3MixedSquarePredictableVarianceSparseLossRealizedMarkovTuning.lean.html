3 /-!
4 # Learning-rate tuning for sparse-loss predictable-variance EXP3 regret
5 
6 This module chooses the learning rate for the compiled sparse-loss realized
7 Markov route. With `L = sparsity * horizon`, the exact Hedge stability scale is
8 
9 `L + sampledMixedSquaredPredictableVarianceRadius arms gamma v (delta / 5)`,
10 
11 where `v = ((1 / (gamma / K)) * L) / (delta / 5)` is the Markov variance
12 threshold used by the five-event theorem. The learning rate
13 
14 `eta = sqrt (log K / scale)`
15 
16 balances entropy against this full scale. Under `gamma <= 1 / 2`, the two
17 learning-rate-dependent terms cost at most
18 `3 * sqrt (log K * scale)`.
19 
20 The theorem remains a pathwise armwise sparse-loss result. It does not tune
21 `gamma`, replace Markov overflow, or claim a best-arm first-order rate.
22 -/

30 /-- Markov threshold for cumulative predictable mixed-square variance under
31 the sparse armwise loss-mass budget. -/
32 noncomputable def sparseLossPredictableVarianceBudget
33     {Action : Type v} (arms : Finset Action) (gamma : Real)
34     (horizon sparsity : Nat) (delta : Real) : Real :=
35   ((1 / (gamma / (arms.card : Real))) *
36       ((sparsity : Real) * (horizon : Real))) / (delta / 5)
37 
38 theorem sparseLossPredictableVarianceBudget_pos
39     {Action : Type v} (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
40     (gamma : Real) (hgamma_pos : 0 < gamma)
41     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
42     (hsparsity : 0 < sparsity) (delta : Real) (hdelta : 0 < delta) :
43     0 < sparseLossPredictableVarianceBudget
44       arms gamma horizon sparsity delta := by
45   have hcard_pos : 0 < (arms.card : Real) := by positivity
46   have hfloor_pos : 0 < gamma / (arms.card : Real) :=
47     div_pos hgamma_pos hcard_pos
48   have hloss_pos :
49       0 < (sparsity : Real) * (horizon : Real) :=
50     mul_pos (Nat.cast_pos.2 hsparsity) (Nat.cast_pos.2 hhorizon)
51   unfold sparseLossPredictableVarianceBudget
52   exact div_pos (mul_pos (one_div_pos.2 hfloor_pos) hloss_pos)
53     (div_pos hdelta (by norm_num))

55 /-- Complete sparse-loss Hedge scale at the public five-event allocation. -/
56 noncomputable def sparseLossPredictableVarianceHighProbabilityScale
57     {Action : Type v} [DecidableEq Action]
58     (arms : Finset Action) (gamma : Real)
59     (horizon sparsity : Nat) (delta : Real) : Real :=
60   (sparsity : Real) * (horizon : Real) +
61     sampledMixedSquaredPredictableVarianceRadius arms gamma
62       (sparseLossPredictableVarianceBudget
63         arms gamma horizon sparsity delta)
64       (delta / 5)

66 theorem sparseLossPredictableVarianceHighProbabilityScale_pos
67     {Action : Type v} [DecidableEq Action]
68     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
69     (gamma : Real) (hgamma_pos : 0 < gamma)
70     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
71     (hsparsity : 0 < sparsity) (delta : Real) (hdelta : 0 < delta) :
72     0 < sparseLossPredictableVarianceHighProbabilityScale
73       arms gamma horizon sparsity delta := by
74   have hcard_pos : 0 < (arms.card : Real) := by positivity
75   have hfloor_pos : 0 < gamma / (arms.card : Real) :=
76     div_pos hgamma_pos hcard_pos
77   have hvariance_pos :=
78     sparseLossPredictableVarianceBudget_pos
79       arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity
80         delta hdelta
81   have hbudget_nonneg : 0 ≤ max (Real.log (1 / (delta / 5))) 0 :=
82     le_max_right _ _
83   have hradius_nonneg :
84       0 ≤ sampledMixedSquaredPredictableVarianceRadius arms gamma
85         (sparseLossPredictableVarianceBudget
86           arms gamma horizon sparsity delta)
87         (delta / 5) := by
88     dsimp [sampledMixedSquaredPredictableVarianceRadius]
89     exact add_nonneg
90       (mul_nonneg (by norm_num) (Real.sqrt_nonneg _))
91       (div_nonneg hbudget_nonneg hfloor_pos.le)
92   have hloss_pos :
93       0 < (sparsity : Real) * (horizon : Real) :=
94     mul_pos (Nat.cast_pos.2 hsparsity) (Nat.cast_pos.2 hhorizon)
95   dsimp [sparseLossPredictableVarianceHighProbabilityScale]
96   linarith

148 /-- With `gamma <= 1/2`, entropy and sparse-loss stability cost at most three
149 copies of their balanced square-root scale. -/
150 theorem sparseLossPredictableVarianceHighProbabilityHedgeBudget_le_three_mul_sqrt
151     {Action : Type v} [DecidableEq Action]
152     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
153     (gamma : Real) (hgamma_pos : 0 < gamma)
154     (hgamma_le_half : gamma ≤ 1 / 2)
155     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
156     (hsparsity : 0 < sparsity) (delta : Real) (hdelta : 0 < delta) :
157     Real.log (arms.card : Real) /
158           sparseLossPredictableVarianceHighProbabilityLearningRate
159             arms gamma horizon sparsity delta +
160         (sparseLossPredictableVarianceHighProbabilityLearningRate
161             arms gamma horizon sparsity delta * (1 / (1 - gamma))) *
162           sparseLossPredictableVarianceHighProbabilityScale
163             arms gamma horizon sparsity delta ≤
164       3 * Real.sqrt
165         (Real.log (arms.card : Real) *
166           sparseLossPredictableVarianceHighProbabilityScale
167             arms gamma horizon sparsity delta) := by
168   let eta := sparseLossPredictableVarianceHighProbabilityLearningRate
169     arms gamma horizon sparsity delta
170   let scale := sparseLossPredictableVarianceHighProbabilityScale
171     arms gamma horizon sparsity delta
172   let entropy := Real.log (arms.card : Real) / eta
173   have hK_one : 1 < (arms.card : Real) := by
174     exact_mod_cast hcard_two
175   have heta : 0 < eta := by
176     dsimp [eta]
177     exact sparseLossPredictableVarianceHighProbabilityLearningRate_pos
178       arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity
179         delta hdelta
180   have hscale : 0 < scale := by
181     dsimp [scale]
182     exact sparseLossPredictableVarianceHighProbabilityScale_pos
183       arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity
184         delta hdelta
185   have hbalance : eta ^ 2 * scale = Real.log (arms.card : Real) := by
186     dsimp [eta, scale]
187     exact
188       sparseLossPredictableVarianceHighProbabilityLearningRate_sq_mul_scale
189         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity
190           delta hdelta
191   have hentropy_nonneg : 0 ≤ entropy :=
192     div_nonneg (Real.log_pos hK_one).le heta.le
193   have hentropy_eq_unscaled : entropy = eta * scale := by
194     dsimp [entropy]
195     field_simp [ne_of_gt heta]
196     nlinarith [hbalance]
197   have hden : 0 < 1 - gamma := by linarith
198   have hfactor : 1 ≤ 2 * (1 - gamma) := by linarith
199   have hstable :
200       (eta * (1 / (1 - gamma))) * scale ≤ 2 * entropy := by
201     rw [show (eta * (1 / (1 - gamma))) * scale =
202       (eta * scale) / (1 - gamma) by ring]
203     rw [← hentropy_eq_unscaled, div_le_iff₀ hden]
204     have hmul := mul_le_mul_of_nonneg_left hfactor hentropy_nonneg
205     nlinarith
206   have hbalanced_sq :
207       entropy ^ 2 = Real.log (arms.card : Real) * scale := by
208     rw [hentropy_eq_unscaled]
209     calc
210       (eta * scale) ^ 2 = (eta ^ 2 * scale) * scale := by ring
211       _ = Real.log (arms.card : Real) * scale := by rw [hbalance]
212   have hradicand_nonneg :
213       0 ≤ Real.log (arms.card : Real) * scale :=
214     mul_nonneg (Real.log_pos hK_one).le hscale.le
215   have hentropy_eq_sqrt :
216       entropy = Real.sqrt
217         (Real.log (arms.card : Real) * scale) := by
218     have hsqrt_nonneg :=
219       Real.sqrt_nonneg (Real.log (arms.card : Real) * scale)
220     have hsqrt_sq := Real.sq_sqrt hradicand_nonneg
221     nlinarith [hbalanced_sq]
222   change entropy + (eta * (1 / (1 - gamma))) * scale ≤
223     3 * Real.sqrt (Real.log (arms.card : Real) * scale)
224   rw [← hentropy_eq_sqrt]
225   linarith

228 term. Gamma and the three non-square confidence contributions remain visible. -/

230     {Action : Type v} [DecidableEq Action]
231     (arms : Finset Action) (gamma : Real)
232     (horizon sparsity : Nat) (delta : Real) : Real :=
233   3 * Real.sqrt
