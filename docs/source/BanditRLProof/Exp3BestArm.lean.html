3 /-!
4 # Shared finite best-arm transport for EXP3
5 
6 This module packages the comparator-independent order step used by finite-arm
7 best-in-hindsight regret wrappers. It contains no probability bound: downstream
8 routes identify the best-arm event with a finite union of fixed-comparator
9 events and supply their own confidence schedules.
10 -/
11 
12 namespace BanditRLProof.Exp3
13 
14 universe u v
15 
16 /-- Cumulative predictable loss of the best supported arm in hindsight. -/
17 noncomputable def sampledPredictableBestArmCumulativeLoss
18     {Env : Type u} {Action : Type v}
19     [MeasurableSpace Env] [MeasurableSpace Action]
20     (arms : Finset Action) (harms : arms.Nonempty)
21     (loss : PredictableLossVector Env Action) (horizon : Nat)
22     (sample : Env × ((k : Nat) → Action × Real)) : Real :=
23   arms.inf' harms fun comparator =>
24     (Finset.range horizon).sum fun t =>
25       predictableLossAt loss t sample comparator
26 
27 /-- The best-arm regret event is exactly the finite existential union of the
28 fixed-comparator regret events. -/
29 theorem threshold_le_sampledPredictableRealizedLoss_sub_bestArmCumulativeLoss_iff
30     {Env : Type u} {Action : Type v}
31     [MeasurableSpace Env] [MeasurableSpace Action]
32     (arms : Finset Action) (harms : arms.Nonempty)
33     (loss : PredictableLossVector Env Action) (horizon : Nat)
34     (sample : Env × ((k : Nat) → Action × Real)) (threshold : Real) :
35     threshold <=
36         (Finset.range horizon).sum (fun t =>
37             sampledTrajectoryRealizedLossAt t sample) -
38           sampledPredictableBestArmCumulativeLoss
39             arms harms loss horizon sample ↔
40       ∃ comparator ∈ arms,
41         threshold <=
42           (Finset.range horizon).sum (fun t =>
43               sampledTrajectoryRealizedLossAt t sample) -
44             (Finset.range horizon).sum (fun t =>
45               predictableLossAt loss t sample comparator) := by
46   let realizedLoss :=
47     (Finset.range horizon).sum fun t =>
48       sampledTrajectoryRealizedLossAt t sample
49   let comparatorLoss := fun comparator =>
50     (Finset.range horizon).sum fun t =>
51       predictableLossAt loss t sample comparator
52   change threshold <= realizedLoss - arms.inf' harms comparatorLoss ↔
53     ∃ comparator ∈ arms, threshold <= realizedLoss - comparatorLoss comparator
54   constructor
55   · intro hbest
56     have hinf :
57         arms.inf' harms comparatorLoss <= realizedLoss - threshold := by
58       linarith
59     rcases (Finset.inf'_le_iff harms).1 hinf with
60       ⟨comparator, hcomparator, hcomparatorLoss⟩
61     exact ⟨comparator, hcomparator, by linarith⟩
62   · rintro ⟨comparator, hcomparator, hregret⟩
63     have hinf :
64         arms.inf' harms comparatorLoss <= comparatorLoss comparator :=
65       Finset.inf'_le _ hcomparator
66     linarith
67 
68 end BanditRLProof.Exp3
