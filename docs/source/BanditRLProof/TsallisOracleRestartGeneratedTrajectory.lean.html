18 /-- Start time of the current epoch, with either continuation or a fresh
19 restart at every successor time. -/
20 structure OracleRestartSchedule where
21   start : Nat -> Nat
22   start_zero : start 0 = 0
23   start_le : forall t, start t <= t
24   start_succ : forall n, start (n + 1) = start n ∨ start (n + 1) = n + 1
25 
26 /-- One epoch containing the whole trajectory. -/
27 def oracleNeverRestartSchedule : OracleRestartSchedule where
28   start := fun _ => 0
29   start_zero := rfl
30   start_le := fun t => Nat.zero_le t
31   start_succ := fun _ => Or.inl rfl
32 
33 /-- A fresh one-round epoch at every actual time. -/
34 def oracleRestartEveryRoundSchedule : OracleRestartSchedule where
35   start := id
36   start_zero := rfl
37   start_le := fun _ => le_rfl
38   start_succ := fun _ => Or.inr rfl

40 /-- Away from a restart boundary, the current epoch starts no later than the
41 last observed round. -/
42 theorem OracleRestartSchedule.start_succ_le_of_ne
43     (schedule : OracleRestartSchedule) (n : Nat)
44     (hboundary : schedule.start (n + 1) ≠ n + 1) :
45     schedule.start (n + 1) <= n := by
46   have hle := schedule.start_le (n + 1)
47   omega

49 /-- Reindex the inclusive global history segment `start..n` as a local history
50 through `n-start`. -/
51 def oracleRestartLocalPairHistory
52     {Action Reward : Type*} (start n : Nat) (hstart : start <= n)
53     (history : History.FinitePairHistory Action Reward n) :
54     History.FinitePairHistory Action Reward (n - start) :=
55   fun i =>
56     history ⟨start + i.1, Finset.mem_Iic.mpr (by
57       have hi := Finset.mem_Iic.mp i.2
58       omega)⟩

64     oracleRestartLocalPairHistory 0 n (Nat.zero_le n) history = history := by

73     Measurable

76   refine measurable_pi_lambda _ ?_

84 /-- Restarted successor action distribution. Boundary times use a fresh
85 initial law; continuation times use only the current epoch's local suffix. -/
86 noncomputable def sampledOracleRestartHalfTsallisHistoryDistribution
87     {Action : Type u} [DecidableEq Action]
88     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
89     (schedule : OracleRestartSchedule) (n : Nat) :
90     History.FinitePairHistory Action Real n -> Action -> Real :=
91   fun history =>
92     if hboundary : schedule.start (n + 1) = n + 1 then
93       initialHalfTsallisDistribution arms harms (eta 0)
94     else
95       sampledScheduledHalfTsallisHistoryDistribution
96         arms harms eta (n - schedule.start (n + 1))
97         (oracleRestartLocalPairHistory
98           (schedule.start (n + 1)) n
99           (schedule.start_succ_le_of_ne n hboundary) history)

189 /-- Measurable finite-action source for every restarted successor policy. -/
190 noncomputable def sampledOracleRestartHalfTsallisHistoryDistributionSource
191     {Action : Type u}
192     [MeasurableSpace Action] [MeasurableSingletonClass Action]
193     [DecidableEq Action]
194     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
195     (schedule : OracleRestartSchedule) (n : Nat) :
196     Exp3.MeasurableFiniteActionDistribution arms
197       (sampledOracleRestartHalfTsallisHistoryDistribution
198         arms harms eta schedule n) := by
199   classical
200   let selector :=
201     canonicalHalfTsallisScheduleFiniteHistorySelectorMeasurability
202       arms harms eta
203   by_cases hboundary : schedule.start (n + 1) = n + 1
204   · refine
205       { distribution := fun _history => ?_
206         measurable_prob := fun _action _haction => ?_ }
207     · simpa [sampledOracleRestartHalfTsallisHistoryDistribution, hboundary] using
208         finiteActionDistribution_initialHalfTsallisDistribution
209           arms harms (eta 0)
210     · simp [sampledOracleRestartHalfTsallisHistoryDistribution, hboundary]
211   · let hstart := schedule.start_succ_le_of_ne n hboundary
212     let localSource :=
213       sampledScheduledHalfTsallisHistoryDistributionSource
214         arms harms eta selector (n - schedule.start (n + 1))
215     refine
216       { distribution := fun history => ?_
217         measurable_prob := fun action haction => ?_ }
218     · simpa [sampledOracleRestartHalfTsallisHistoryDistribution, hboundary,
219         hstart, localSource] using
220         localSource.distribution
221           (oracleRestartLocalPairHistory
222             (schedule.start (n + 1)) n hstart history)
223     · simpa [sampledOracleRestartHalfTsallisHistoryDistribution, hboundary,
224         hstart, localSource] using
225         (localSource.measurable_prob action haction).comp
226           (measurable_oracleRestartLocalPairHistory
227             (Action := Action) (Reward := Real)
228             (schedule.start (n + 1)) n hstart)
