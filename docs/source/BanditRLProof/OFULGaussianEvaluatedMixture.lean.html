19 /--
20 The positive-semidefinite Gaussian quadratic exponential is integrable.
21 Fernique supplies a square-exponential envelope for the linear term.
22 -/
23 theorem integrable_exp_inner_sub_quadratic_multivariateGaussian_zero_inv
24     [Fintype Feature] [DecidableEq Feature]
25     (V0 G : Matrix Feature Feature Real)
26     (hG : G.PosSemidef) (score : EuclideanSpace Real Feature) :
27     Integrable
28       (fun z : EuclideanSpace Real Feature =>
29         Real.exp
30           (⟪score, z⟫_ℝ -
31             ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2))
32       (ProbabilityTheory.multivariateGaussian 0 V0⁻¹) := by
33   let nu :=
34     ProbabilityTheory.multivariateGaussian
35       (0 : EuclideanSpace Real Feature) V0⁻¹
36   obtain ⟨C, hC, hFernique⟩ :=
37     IsGaussian.exists_integrable_exp_sq nu
38   have hdom :
39       Integrable
40         (fun z : EuclideanSpace Real Feature =>
41           Real.exp (‖score‖ ^ 2 / (4 * C)) *
42             Real.exp (C * ‖z‖ ^ 2))
43         nu := by
44     exact hFernique.const_mul (Real.exp (‖score‖ ^ 2 / (4 * C)))
45   refine Integrable.mono' hdom (by fun_prop) ?_
46   exact Filter.Eventually.of_forall (fun z => by
47     have hquad :
48         0 <=
49           ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ := by
50       rw [Matrix.inner_toEuclideanCLM]
51       have hnonneg := hG.dotProduct_mulVec_nonneg (WithLp.ofLp z)
52       have hstar : star (WithLp.ofLp z) = WithLp.ofLp z := by
53         funext i
54         simp
55       simpa [hstar] using hnonneg
56     have hinner :
57         ⟪score, z⟫_ℝ <= ‖score‖ * ‖z‖ :=
58       real_inner_le_norm score z
59     have hyoung :
60         ‖score‖ * ‖z‖ <=
61           C * ‖z‖ ^ 2 + ‖score‖ ^ 2 / (4 * C) := by
62       have h :=
63         two_mul_le_add_mul_sq
64           (a := ‖z‖) (b := ‖score‖) (ε := 2 * C) (by positivity)
65       calc
66         ‖score‖ * ‖z‖ =
67             (2 * ‖z‖ * ‖score‖) / 2 := by ring
68         _ <=
69             ((2 * C) * ‖z‖ ^ 2 +
70               (2 * C)⁻¹ * ‖score‖ ^ 2) / 2 := by
71               exact div_le_div_of_nonneg_right h (by norm_num)
72         _ =
73             C * ‖z‖ ^ 2 + ‖score‖ ^ 2 / (4 * C) := by
74               field_simp [ne_of_gt hC]
75               ring
76     have hexponent :
77         ⟪score, z⟫_ℝ -
78             ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2 <=
79           ‖score‖ ^ 2 / (4 * C) + C * ‖z‖ ^ 2 := by
80       calc
81         ⟪score, z⟫_ℝ -
82               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2
83             <= ⟪score, z⟫_ℝ := by linarith
84         _ <= ‖score‖ * ‖z‖ := hinner
85         _ <= C * ‖z‖ ^ 2 + ‖score‖ ^ 2 / (4 * C) := hyoung
86         _ = ‖score‖ ^ 2 / (4 * C) + C * ‖z‖ ^ 2 := by ring
87     rw [Real.norm_eq_abs, abs_of_pos (Real.exp_pos _)]
88     calc
89       Real.exp
90           (⟪score, z⟫_ℝ -
91             ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2) <=
92           Real.exp
93             (‖score‖ ^ 2 / (4 * C) + C * ‖z‖ ^ 2) :=
94         Real.exp_le_exp.mpr hexponent
95       _ =
96           Real.exp (‖score‖ ^ 2 / (4 * C)) *
97             Real.exp (C * ‖z‖ ^ 2) := Real.exp_add _ _)

99 /--
100 The `ENNReal` Gaussian direction integral equals the completed-square
101 determinant-ratio expression.
102 -/
103 theorem lintegral_gaussianQuadraticExponentialENNReal_multivariateGaussian_zero_inv
104     [Fintype Feature] [DecidableEq Feature]
105     (V0 G : Matrix Feature Feature Real) (hV0 : V0.PosDef)
106     (hG : G.PosSemidef) (score : EuclideanSpace Real Feature) :
107     ∫⁻ z : EuclideanSpace Real Feature,
108         ENNReal.ofReal
109           (Real.exp
110             (⟪score, z⟫_ℝ -
111               ⟪z, Matrix.toEuclideanCLM (𝕜 := Real) G z⟫_ℝ / 2))
112       ∂ProbabilityTheory.multivariateGaussian 0 V0⁻¹ =
113       ENNReal.ofReal
114         (Real.sqrt (Matrix.det V0 / Matrix.det (V0 + G)) *
115           Real.exp
116             (score ⬝ᵥ (V0 + G)⁻¹.mulVec score / 2)) := by
117   rw [← ofReal_integral_eq_lintegral_ofReal
118     (integrable_exp_inner_sub_quadratic_multivariateGaussian_zero_inv
119       V0 G hG score)
120     (Filter.Eventually.of_forall (fun _ => (Real.exp_pos _).le))]
121   rw [integral_exp_inner_sub_quadratic_multivariateGaussian_zero_inv
122     V0 G hV0 hG score]

124 /--
125 The evaluated determinant-ratio inverse-Gram exponential for a finite-horizon
126 score and variance Gram.
127 -/
128 noncomputable def finiteHorizonDetRatioInvGramExponential
129     [Fintype Feature] [DecidableEq Feature]
130     (V0 : Matrix Feature Feature Real)
131     (feature : Nat -> Omega -> Feature -> Real)
132     (noise : Nat -> Omega -> Real)
133     (varianceProxy : Nat -> NNReal)
134     (n : Nat) (omega : Omega) : ENNReal :=
135   ENNReal.ofReal
136     (Real.sqrt
137         (Matrix.det V0 /
138           Matrix.det
139             (V0 +
140               finiteHorizonVarianceGram feature varianceProxy n omega)) *
141       Real.exp
142         ((finiteHorizonNoiseScore feature noise n omega) ⬝ᵥ
143             (V0 +
144               finiteHorizonVarianceGram
145                 feature varianceProxy n omega)⁻¹.mulVec
146               (finiteHorizonNoiseScore feature noise n omega) / 2))

178 The evaluated finite-horizon mixture is measurable in the sample.
179 -/
180 theorem measurable_finiteHorizonDetRatioInvGramExponential
181     [MeasurableSpace Omega]
182     [Fintype Feature] [DecidableEq Feature]
183     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
184     (feature : Nat -> Omega -> Feature -> Real)
185     (noise : Nat -> Omega -> Real)
186     (varianceProxy : Nat -> NNReal)
187     (n : Nat)
188     (hfeature : forall i j,
189       Measurable (fun omega => feature i omega j))
190     (hnoise : forall i, Measurable (noise i)) :
191     Measurable
192       (finiteHorizonDetRatioInvGramExponential
193         V0 feature noise varianceProxy n) := by
194   have hscore :
195       Measurable (finiteHorizonNoiseScore feature noise n) :=
196     measurable_finiteHorizonNoiseScore
197       feature noise n hfeature hnoise
198   have hgram : forall j k,
199       Measurable
200         (fun omega =>
201           finiteHorizonVarianceGram feature varianceProxy n omega j k) :=
202     measurable_finiteHorizonVarianceGram_apply
203       feature varianceProxy n hfeature
204   have hjoint :
205       Measurable
206         (gaussianQuadraticExponentialENNReal
207           (finiteHorizonNoiseScore feature noise n)
208           (finiteHorizonVarianceGram feature varianceProxy n)) :=
209     measurable_gaussianQuadraticExponentialENNReal
210       _ _ hscore hgram
211   rw [show
212       finiteHorizonDetRatioInvGramExponential
213           V0 feature noise varianceProxy n =
214         fun omega => ∫⁻ theta : EuclideanSpace Real Feature,
215           gaussianQuadraticExponentialENNReal
216             (finiteHorizonNoiseScore feature noise n)
217             (finiteHorizonVarianceGram feature varianceProxy n)
218             (omega, theta)
219           ∂ProbabilityTheory.multivariateGaussian 0 V0⁻¹ by
220     funext omega
221     exact
222       (lintegral_gaussianQuadraticExponentialENNReal_finiteHorizon_multivariateGaussian_zero_inv
223         V0 hV0 feature noise varianceProxy n omega).symm]
224   exact hjoint.lintegral_prod_right'

226 /--
227 The evaluated finite-horizon Gaussian mixture has `ENNReal` expectation at
228 most one.
229 -/
230 theorem lintegral_finiteHorizon_detRatio_invGramExponential_le_one
231     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
232     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
233     (mu : Measure Omega) [IsProbabilityMeasure mu]
234     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
235     (F : Filtration Nat mOmega)
236     (feature : Nat -> Omega -> Feature -> Real)
237     (noise : Nat -> Omega -> Real)
238     (varianceProxy : Nat -> NNReal)
239     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
240     (hfeature : forall i j,
241       StronglyMeasurable[F i] (fun omega => feature i omega j))
242     (hnoise : StronglyAdapted F (fun t omega =>
243       match t with
244       | 0 => 0
245       | i + 1 => noise i omega))
246     (hprojectionBound_nonneg : forall theta i,
247       0 <= projectionBound theta i)
248     (hprojectionBound : forall theta i omega,
249       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
250         projectionBound theta i)
251     (n : Nat)
252     (hsubGaussian : forall i, i < n ->
253       HasCondSubgaussianMGF
254         (F i) (F.le i) (noise i) (varianceProxy i) mu) :
255     ∫⁻ omega,
256         finiteHorizonDetRatioInvGramExponential
257           V0 feature noise varianceProxy n omega
258       ∂mu <= 1 := by
259   have hfeatureMeas : forall i j,
260       Measurable (fun omega => feature i omega j) := by
261     intro i j
262     exact (hfeature i j).measurable.mono (F.le i) le_rfl
263   have hnoiseMeas : forall i, Measurable (noise i) := by
264     intro i
265     have hi : StronglyMeasurable[F (i + 1)] (noise i) := by
266       simpa using hnoise (i + 1)
267     exact hi.measurable.mono (F.le (i + 1)) le_rfl
268   have hscore :
269       Measurable (finiteHorizonNoiseScore feature noise n) :=
270     measurable_finiteHorizonNoiseScore
271       feature noise n hfeatureMeas hnoiseMeas
272   have hgram : forall j k,
273       Measurable
274         (fun omega =>
275           finiteHorizonVarianceGram feature varianceProxy n omega j k) :=
276     measurable_finiteHorizonVarianceGram_apply
277       feature varianceProxy n hfeatureMeas
278   calc
279     ∫⁻ omega,
280         finiteHorizonDetRatioInvGramExponential
281           V0 feature noise varianceProxy n omega
282       ∂mu =
283         ∫⁻ omega, ∫⁻ theta,
284           gaussianQuadraticExponentialENNReal
285             (finiteHorizonNoiseScore feature noise n)
286             (finiteHorizonVarianceGram feature varianceProxy n)
287             (omega, theta)
288           ∂ProbabilityTheory.multivariateGaussian 0 V0⁻¹ ∂mu := by
289             apply lintegral_congr
290             intro omega
291             exact
292               (lintegral_gaussianQuadraticExponentialENNReal_finiteHorizon_multivariateGaussian_zero_inv
293                 V0 hV0 feature noise varianceProxy n omega).symm
294     _ =
295         ∫⁻ p,
296           gaussianQuadraticExponentialENNReal
297             (finiteHorizonNoiseScore feature noise n)
298             (finiteHorizonVarianceGram feature varianceProxy n) p
299           ∂mu.prod
300             (ProbabilityTheory.multivariateGaussian 0 V0⁻¹) := by
301               symm
302               exact
303                 lintegral_gaussianQuadraticExponentialENNReal_prod_multivariateGaussian_zero_inv
304                   mu V0
305                   (finiteHorizonNoiseScore feature noise n)
306                   (finiteHorizonVarianceGram feature varianceProxy n)
307                   hscore hgram
308     _ <= 1 :=
309       lintegral_gaussianQuadraticExponentialENNReal_finiteHorizon_prod_multivariateGaussian_zero_inv_le_one
310         mu V0 F feature noise varianceProxy projectionBound hfeature hnoise
311         hprojectionBound_nonneg hprojectionBound n hsubGaussian
