5 /-! # Telescoping exponential estimate for dyadic peeling
6 
7 A convex-exponential comparison gives a dyadic sum estimate strong enough
8 to imply the constant 15 used in source Lemma 9.3.
9 -/

11 namespace BanditRLProof.Concentration

13 open Real Finset
14 open scoped ENNReal NNReal

16 /-- A telescoping majorant obtained from `x/3 ≤ sinh(x/3)`. -/
17 theorem mul_exp_neg_le_exp_difference (x : ℝ) (hx : 0 ≤ x) :
18     x * exp (-x) ≤ (3 / 2 : ℝ) * (exp (-(2 * x / 3)) - exp (-(4 * x / 3))) := by
19   have hs : x / 3 ≤ sinh (x / 3) := self_le_sinh_iff.mpr (by positivity)
20   rw [sinh_eq] at hs
21   have h := mul_le_mul_of_nonneg_right hs (exp_pos (-x)).le
22   have h1 : exp (x / 3) * exp (-x) = exp (-(2 * x / 3)) := by
23     rw [← exp_add]
24     congr 1
25     ring
26   have h2 : exp (-(x / 3)) * exp (-x) = exp (-(4 * x / 3)) := by
27     rw [← exp_add]
28     congr 1
29     ring
30   have heq : (exp (x / 3) - exp (-(x / 3))) / 2 * exp (-x) =
31       (exp (-(2 * x / 3)) - exp (-(4 * x / 3))) / 2 := by
32     rw [div_mul_eq_mul_div, sub_mul, h1, h2]
33   rw [heq] at h
34   nlinarith

36 /-- Finite dyadic exponential sum, with the remaining terminal mass retained. -/
37 theorem sum_dyadic_mul_exp_neg_le (a : ℝ) (ha : 0 < a) (N : ℕ) :
38     ∑ j ∈ range N, (2 : ℝ) ^ j * exp (-(a * 2 ^ j)) ≤
39       3 / (2 * a) * (exp (-(2 * a / 3)) - exp (-(2 * (a * 2 ^ N) / 3))) := by
40   have hterm (j : ℕ) : (2 : ℝ) ^ j * exp (-(a * 2 ^ j)) ≤
41       3 / (2 * a) *
42         (exp (-(2 * (a * 2 ^ j) / 3)) - exp (-(2 * (a * 2 ^ (j+1)) / 3))) := by
43     have h := mul_exp_neg_le_exp_difference (a * 2 ^ j) (by positivity)
44     have heq : 4 * (a * (2 : ℝ) ^ j) / 3 = 2 * (a * 2 ^ (j+1)) / 3 := by
45       rw [pow_succ]
46       ring
47     rw [heq] at h
48     have h' := (div_le_div_iff_of_pos_right ha).mpr h
49     convert h' using 1 <;> field_simp <;> ring
50   induction N with
51   | zero => simp
52   | succ N ih =>
53     rw [sum_range_succ]
54     have h := hterm N
55     linarith
56 
57 /-- Uniform finite-prefix bound; no logarithm or integral comparison loss. -/
58 theorem sum_dyadic_mul_exp_neg_le_three_div_two (a : ℝ) (ha : 0 < a) (N : ℕ) :
59     ∑ j ∈ range N, (2 : ℝ) ^ j * exp (-(a * 2 ^ j)) ≤ 3 / (2 * a) := by
60   have h := sum_dyadic_mul_exp_neg_le a ha N
61   have he : exp (-(2 * a / 3)) ≤ 1 := exp_le_one_iff.mpr (by linarith)
62   have hn := (exp_pos (-(2 * (a * 2 ^ N) / 3))).le
63   have hp : 0 ≤ 3 / (2 * a) := by positivity
64   nlinarith

66 /-- Countable dyadic sum in the probability-friendly extended nonnegative reals. -/
67 theorem tsum_dyadic_mul_exp_neg_le (a : ℝ) (ha : 0 < a) :
68     (∑' j : ℕ, ENNReal.ofReal ((2 : ℝ) ^ j * exp (-(a * 2 ^ j)))) ≤
69       ENNReal.ofReal (3 / (2 * a)) := by
70   apply ENNReal.tsum_le_of_sum_range_le
71   intro N
72   rw [← ENNReal.ofReal_sum_of_nonneg (fun j _ => by positivity)]
73   exact ENNReal.ofReal_le_ofReal (sum_dyadic_mul_exp_neg_le_three_div_two a ha N)

75 /-- The geometric series in source Lemma 9.3 is at most 12 delta/gap^2. -/
76 theorem sum_moss_peeling_exponential_le_twelve (δ gap : ℝ)
77     (hδ : 0 ≤ δ) (hgap : 0 < gap) (N : ℕ) :
78     ∑ j ∈ range N, δ * (2 : ℝ) ^ (j+1) * exp (-(gap ^ 2 / 4 * 2 ^ j)) ≤
79       12 * δ / gap ^ 2 := by
80   have ha : 0 < gap ^ 2 / 4 := by positivity
81   have h := mul_le_mul_of_nonneg_left
82     (sum_dyadic_mul_exp_neg_le_three_div_two (gap ^ 2 / 4) ha N)
83     (show 0 ≤ 2 * δ by positivity)
84   have heq : (∑ j ∈ range N, δ * (2 : ℝ) ^ (j+1) * exp (-(gap ^ 2 / 4 * 2 ^ j))) =
85       2 * δ * ∑ j ∈ range N, (2 : ℝ) ^ j * exp (-(gap ^ 2 / 4 * 2 ^ j)) := by
86     rw [mul_sum]
87     apply sum_congr rfl
88     intro j _
89     rw [pow_succ]
90     ring
91   rw [heq]
92   convert h using 1 <;> field_simp <;> ring

94 /-- Source-constant countable series bound, with no weakened constant. -/
95 theorem tsum_moss_peeling_exponential_le_fifteen (δ gap : ℝ)
96     (hδ : 0 ≤ δ) (hgap : 0 < gap) :
97     (∑' j : ℕ, ENNReal.ofReal
98       (δ * (2 : ℝ) ^ (j+1) * exp (-(gap ^ 2 / 4 * 2 ^ j)))) ≤
99       ENNReal.ofReal (15 * δ / gap ^ 2) := by
100   apply ENNReal.tsum_le_of_sum_range_le
101   intro N
102   rw [← ENNReal.ofReal_sum_of_nonneg (fun j _ => by positivity)]
103   apply ENNReal.ofReal_le_ofReal
104   refine (sum_moss_peeling_exponential_le_twelve δ gap hδ hgap N).trans ?_
105   apply div_le_div_of_nonneg_right _ (sq_nonneg gap)
106   nlinarith
