18 /-- Deterministic selected-loss predictable-variance budget at prefix `n+1`. -/
19 noncomputable def sampledRealizedPredictableVarianceLinearBudget
20     (n : Nat) : Real :=
21   ((n + 1 : Nat) : Real)

23 theorem sampledRealizedPredictableVarianceLinearBudget_pos (n : Nat) :
24     0 < sampledRealizedPredictableVarianceLinearBudget n := by

32   sampledRealizedPredictableVarianceGeometricRadius

35 /-- Pure realized-deviation failure event over every positive prefix of one
36 generated EXP3 trajectory. -/
37 noncomputable def sampledRealizedDeviationGeometricAllTimeFailureSet
38     {Env : Type u} {Action : Type v}
39     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
40     (arms : Finset Action) (eta gamma : Real)
41     (loss : PredictableLossVector Env Action) (delta : Real) :
42     Set (Env × ((k : Nat) → Action × Real)) :=
43   ⋃ n, {sample |
44     sampledRealizedDeviationGeometricAllTimeRadius delta n ≤
45       (Finset.range (n + 1)).sum (fun i =>
46         sampledTrajectoryRealizedDeviationAt arms eta gamma loss i sample)}

48 theorem mem_sampledRealizedDeviationGeometricAllTimeFailureSet_iff
49     {Env : Type u} {Action : Type v}
50     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
51     (arms : Finset Action) (eta gamma : Real)
52     (loss : PredictableLossVector Env Action) (delta : Real)
53     (sample : Env × ((k : Nat) → Action × Real)) :
54     sample ∈ sampledRealizedDeviationGeometricAllTimeFailureSet
55         arms eta gamma loss delta ↔
56       ∃ n,
57         sampledRealizedDeviationGeometricAllTimeRadius delta n ≤
58           (Finset.range (n + 1)).sum (fun i =>
59             sampledTrajectoryRealizedDeviationAt
60               arms eta gamma loss i sample) := by
61   simp [sampledRealizedDeviationGeometricAllTimeFailureSet]

63 /-- With the deterministic budget `n+1`, the prior joint failure event is
64 exactly the pure realized-deviation crossing event. -/
65 theorem sampledPredictableRealizedDeviationAllTimeFailureSet_linearBudget_eq
66     {Env : Type u} {Action : Type v}
67     [MeasurableSpace Env]
68     [MeasurableSpace Action] [MeasurableSingletonClass Action]
69     [DecidableEq Action]
70     (arms : Finset Action) (harms : arms.Nonempty)
71     (eta gamma : Real) (hgamma_nonneg : 0 ≤ gamma)
72     (hgamma_le_one : gamma ≤ 1)
73     (loss : PredictableLossVector Env Action) (delta : Real) :
74     sampledPredictableRealizedDeviationAllTimeFailureSet
75         arms eta gamma loss sampledRealizedPredictableVarianceLinearBudget
76           delta =
77       sampledRealizedDeviationGeometricAllTimeFailureSet
78         arms eta gamma loss delta := by
79   ext sample
80   rw [mem_sampledPredictableRealizedDeviationAllTimeFailureSet_iff,
81     mem_sampledRealizedDeviationGeometricAllTimeFailureSet_iff]
82   constructor
83   · rintro ⟨n, hdeviation, _hvariance⟩
84     exact ⟨n, by
85       simpa [sampledRealizedDeviationGeometricAllTimeRadius] using hdeviation⟩
86   · rintro ⟨n, hdeviation⟩
87     refine ⟨n, ?_, ?_⟩
88     · simpa [sampledRealizedDeviationGeometricAllTimeRadius] using hdeviation
89     · simpa [sampledRealizedPredictableVarianceLinearBudget] using
90         (sampledPredictableRealizedVariance_sum_le_horizon
91           arms harms eta gamma hgamma_nonneg hgamma_le_one loss (n + 1)
92             sample)

94 /-- On one fixed generated EXP3 trajectory law, the realized selected-loss
95 deviation stays below its geometric-share quadratic radius at every positive
96 prefix outside a set of mass at most the outer confidence budget. -/
97 theorem measure_sampledRealizedDeviationGeometricAllTimeFailureSet_le
98     {Env : Type u} {Action : Type v}
99     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
100     [MeasurableSpace Action] [MeasurableSingletonClass Action]
101     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
102     (prior : Measure Env) [IsProbabilityMeasure prior]
103     (arms : Finset Action) (harms : arms.Nonempty)
104     (eta gamma : Real) (hgamma_pos : 0 < gamma)
105     (hgamma_le_one : gamma ≤ 1)
106     (loss : PredictableLossVector Env Action)
107     (delta : Real) (hdelta : 0 < delta) :
108     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
109       eta gamma hgamma_pos.le hgamma_le_one loss.environment
110     mu (sampledRealizedDeviationGeometricAllTimeFailureSet
111         arms eta gamma loss delta) ≤ ENNReal.ofReal delta := by
112   dsimp only
113   have htail :=
114     measure_sampledPredictableRealizedDeviationAllTimeFailureSet_le
115       prior arms harms eta gamma hgamma_pos hgamma_le_one loss
116         sampledRealizedPredictableVarianceLinearBudget
117         sampledRealizedPredictableVarianceLinearBudget_pos delta hdelta
