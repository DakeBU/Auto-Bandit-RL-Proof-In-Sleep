3 /-!
4 # Learning-rate tuning for Bernstein-square realized EXP3 regret
5 
6 The variance-sensitive mixed-square route uses
7 
8 `K * T + sampledMixedSquaredBernsteinConfidenceRadius`.
9 
10 This module chooses the exact learning rate
11 
12 `eta = sqrt (log K / (K * T + squareRadius))`
13 
14 at the public four-event allocation. It balances entropy against the complete
15 Bernstein-square stability scale. Exploration, the two action-confidence
16 radii, and realized deviation remain explicit for a later gamma schedule.
17 -/

25 /-- Positive scale appearing in the Bernstein-square Hedge term when the
26 square event receives `delta / 4`. -/
27 noncomputable def bernsteinSquareHighProbabilityScale
28     {Action : Type v} (arms : Finset Action) (gamma : Real)
29     (horizon : Nat) (delta : Real) : Real :=
30   (arms.card : Real) * (horizon : Real) +
31     sampledMixedSquaredBernsteinConfidenceRadius arms gamma horizon (delta / 4)
32 
33 theorem bernsteinSquareHighProbabilityScale_pos
34     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
35     (gamma : Real) (hgamma_pos : 0 < gamma)
36     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
37     0 < bernsteinSquareHighProbabilityScale arms gamma horizon delta := by
38   have hK : 0 < (arms.card : Real) := by
39     positivity
40   have hT : 0 < (horizon : Real) := by
41     exact_mod_cast hhorizon
42   have hepsilon : 0 < gamma / (arms.card : Real) := div_pos hgamma_pos hK
43   have hvariance :
44       0 < sampledMixedSquaredBernsteinVarianceCoefficient arms gamma := by
45     dsimp [sampledMixedSquaredBernsteinVarianceCoefficient]
46     exact div_pos hK hepsilon
47   have hbudget : 0 <= max (Real.log (1 / (delta / 4))) 0 := le_max_right _ _
48   have hradius :
49       0 <= sampledMixedSquaredBernsteinConfidenceRadius
50         arms gamma horizon (delta / 4) := by
51     dsimp [sampledMixedSquaredBernsteinConfidenceRadius]
52     exact add_nonneg
53       (mul_nonneg (by norm_num) (Real.sqrt_nonneg _))
54       (div_nonneg hbudget hepsilon.le)
55   have hbase : 0 < (arms.card : Real) * (horizon : Real) := mul_pos hK hT
56   dsimp [bernsteinSquareHighProbabilityScale]
57   linarith

59 /-- Learning rate balancing entropy against the complete Bernstein-square
60 stability scale at the public `delta / 4` square allocation. -/
61 noncomputable def bernsteinSquareHighProbabilityLearningRate
62     {Action : Type v} (arms : Finset Action) (gamma : Real)
63     (horizon : Nat) (delta : Real) : Real :=
64   Real.sqrt
65     (Real.log (arms.card : Real) /
66       bernsteinSquareHighProbabilityScale arms gamma horizon delta)
67 
68 theorem bernsteinSquareHighProbabilityLearningRate_pos
69     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
70     (gamma : Real) (hgamma_pos : 0 < gamma)
71     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
72     0 < bernsteinSquareHighProbabilityLearningRate
73       arms gamma horizon delta := by
74   have hK_one : 1 < (arms.card : Real) := by
75     exact_mod_cast hcard_two
76   have hscale := bernsteinSquareHighProbabilityScale_pos
77     arms hcard_two gamma hgamma_pos horizon hhorizon delta
78   apply Real.sqrt_pos.2
79   exact div_pos (Real.log_pos hK_one) hscale
80 
81 theorem bernsteinSquareHighProbabilityLearningRate_sq_mul_scale
82     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
83     (gamma : Real) (hgamma_pos : 0 < gamma)
84     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
85     bernsteinSquareHighProbabilityLearningRate arms gamma horizon delta ^ 2 *
86         bernsteinSquareHighProbabilityScale arms gamma horizon delta =
87       Real.log (arms.card : Real) := by
88   have hK_one : 1 < (arms.card : Real) := by
89     exact_mod_cast hcard_two
90   have hscale := bernsteinSquareHighProbabilityScale_pos
91     arms hcard_two gamma hgamma_pos horizon hhorizon delta
92   unfold bernsteinSquareHighProbabilityLearningRate
93   rw [Real.sq_sqrt
94     (div_nonneg (Real.log_pos hK_one).le hscale.le)]
95   field_simp [ne_of_gt hscale]

97 /-- With `gamma <= 1/2`, entropy and the stability-amplified Bernstein-square
98 scale cost at most three copies of their balanced square-root scale. -/
99 theorem bernsteinSquareHighProbabilityHedgeBudget_le_three_mul_sqrt
100     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
101     (gamma : Real) (hgamma_pos : 0 < gamma)
102     (hgamma_le_half : gamma <= 1 / 2)
103     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
104     Real.log (arms.card : Real) /
105           bernsteinSquareHighProbabilityLearningRate
106             arms gamma horizon delta +
107         (bernsteinSquareHighProbabilityLearningRate
108             arms gamma horizon delta * (1 / (1 - gamma))) *
109           bernsteinSquareHighProbabilityScale arms gamma horizon delta <=
110       3 * Real.sqrt
111         (Real.log (arms.card : Real) *
112           bernsteinSquareHighProbabilityScale arms gamma horizon delta) := by
113   let eta := bernsteinSquareHighProbabilityLearningRate
114     arms gamma horizon delta
115   let scale := bernsteinSquareHighProbabilityScale
116     arms gamma horizon delta
117   let entropy := Real.log (arms.card : Real) / eta
118   have hK_one : 1 < (arms.card : Real) := by
119     exact_mod_cast hcard_two
120   have heta : 0 < eta := by
121     dsimp [eta]
122     exact bernsteinSquareHighProbabilityLearningRate_pos
123       arms hcard_two gamma hgamma_pos horizon hhorizon delta
124   have hscale : 0 < scale := by
125     dsimp [scale]
126     exact bernsteinSquareHighProbabilityScale_pos
127       arms hcard_two gamma hgamma_pos horizon hhorizon delta
128   have hbalance : eta ^ 2 * scale = Real.log (arms.card : Real) := by
129     dsimp [eta, scale]
130     exact bernsteinSquareHighProbabilityLearningRate_sq_mul_scale
131       arms hcard_two gamma hgamma_pos horizon hhorizon delta
132   have hentropy_nonneg : 0 <= entropy := by
133     exact div_nonneg (Real.log_pos hK_one).le heta.le
134   have hentropy_eq_unscaled : entropy = eta * scale := by
135     dsimp [entropy]
136     field_simp [ne_of_gt heta]
137     nlinarith [hbalance]
138   have hden : 0 < 1 - gamma := by linarith
139   have hfactor : 1 <= 2 * (1 - gamma) := by linarith
140   have hstable :
141       (eta * (1 / (1 - gamma))) * scale <= 2 * entropy := by
142     rw [show (eta * (1 / (1 - gamma))) * scale =
143       (eta * scale) / (1 - gamma) by ring]
144     rw [← hentropy_eq_unscaled, div_le_iff₀ hden]
145     have hmul := mul_le_mul_of_nonneg_left hfactor hentropy_nonneg
146     nlinarith
147   have hbalanced_sq :
148       entropy ^ 2 = Real.log (arms.card : Real) * scale := by
149     rw [hentropy_eq_unscaled]
150     calc
151       (eta * scale) ^ 2 = (eta ^ 2 * scale) * scale := by ring
152       _ = Real.log (arms.card : Real) * scale := by rw [hbalance]
153   have hradicand_nonneg :
154       0 <= Real.log (arms.card : Real) * scale :=
155     mul_nonneg (Real.log_pos hK_one).le hscale.le
156   have hentropy_eq_sqrt :
157       entropy = Real.sqrt
158         (Real.log (arms.card : Real) * scale) := by
159     have hsqrt_nonneg := Real.sqrt_nonneg
160       (Real.log (arms.card : Real) * scale)
161     have hsqrt_sq := Real.sq_sqrt hradicand_nonneg
162     nlinarith [hbalanced_sq]
163   change entropy + (eta * (1 / (1 - gamma))) * scale <=
164     3 * Real.sqrt (Real.log (arms.card : Real) * scale)
165   rw [← hentropy_eq_sqrt]
166   linarith

168 /-- Explicit threshold after tuning all learning-rate-dependent terms. The
169 exploration and three confidence contributions remain visible. -/
170 noncomputable def bernsteinSquareRealizedTunedThreshold
171     {Action : Type v} (arms : Finset Action) (gamma : Real)
172     (horizon : Nat) (delta : Real) : Real :=
173   3 * Real.sqrt
174       (Real.log (arms.card : Real) *
175         bernsteinSquareHighProbabilityScale arms gamma horizon delta) +
176     gamma * (horizon : Real) +
177     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
178       arms gamma horizon (delta / 4) +
179     sampledComparatorEstimatorBernsteinConfidenceRadius
180       arms gamma horizon (delta / 4) +
181     sampledPredictableRealizedDeviationConfidenceRadius horizon (delta / 4)

183 /-- The complete Bernstein-square four-event realized budget is bounded by the
184 learning-rate-tuned threshold. -/
185 theorem sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget_le_tunedThreshold
186     {Action : Type v} [DecidableEq Action]
187     (arms : Finset Action) (hcard_two : 2 <= arms.card)
188     (gamma : Real) (hgamma_pos : 0 < gamma)
189     (hgamma_le_half : gamma <= 1 / 2)
190     (horizon : Nat) (hhorizon : 0 < horizon) (delta : Real) :
191     sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget
192         arms
193         (bernsteinSquareHighProbabilityLearningRate
194           arms gamma horizon delta)
195         gamma horizon (delta / 4) (delta / 4) (delta / 4) <=
196       bernsteinSquareRealizedTunedThreshold arms gamma horizon delta := by
197   have hhedge :=
198     bernsteinSquareHighProbabilityHedgeBudget_le_three_mul_sqrt
199       arms hcard_two gamma hgamma_pos hgamma_le_half horizon hhorizon delta
200   dsimp [bernsteinSquareHighProbabilityScale] at hhedge
201   dsimp [
202     sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget,
203     sampledPredictableBernsteinSquareHighProbabilityRegretBudget,
204     bernsteinSquareRealizedTunedThreshold,
205     bernsteinSquareHighProbabilityScale]
206   linarith

208 /-- Generated realized-regret tail with the Bernstein-square-balanced learning
209 rate. Gamma scheduling and confidence-radius simplification remain for
210 downstream consumers. -/
211 theorem sampledPredictable_tunedBernsteinSquareRealizedRegret_tail
212     {Env : Type u} {Action : Type v}
213     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
214     [MeasurableSpace Action] [MeasurableSingletonClass Action]
215     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
216     (prior : Measure Env) [IsProbabilityMeasure prior]
217     (arms : Finset Action) (harms : arms.Nonempty)
218     (hcard_two : 2 <= arms.card)
219     (gamma : Real) (hgamma_pos : 0 < gamma)
220     (hgamma_le_half : gamma <= 1 / 2)
221     (loss : PredictableLossVector Env Action)
222     (comparator : Action) (hcomparator : comparator ∈ arms)
223     (horizon : Nat) (hhorizon : 0 < horizon)
224     (delta : Real) (hdelta : 0 < delta) :
225     let eta := bernsteinSquareHighProbabilityLearningRate
226       arms gamma horizon delta
227     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
228       eta gamma hgamma_pos.le (by linarith : gamma <= 1) loss.environment
229     mu {sample |
230         bernsteinSquareRealizedTunedThreshold arms gamma horizon delta <=
231           (Finset.range horizon).sum (fun t =>
232               sampledTrajectoryRealizedLossAt t sample) -
233             (Finset.range horizon).sum (fun t =>
234               predictableLossAt loss t sample comparator)} <=
235       ENNReal.ofReal delta := by
236   dsimp only
237   have heta :
238       0 < bernsteinSquareHighProbabilityLearningRate
239         arms gamma horizon delta :=
240     bernsteinSquareHighProbabilityLearningRate_pos
241       arms hcard_two gamma hgamma_pos horizon hhorizon delta
242   have hgamma_lt_one : gamma < 1 := by linarith
243   have hbudget :=
244     sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget_le_tunedThreshold
245       arms hcard_two gamma hgamma_pos hgamma_le_half horizon hhorizon delta
246   have htail :=
247     sampledPredictable_bernsteinSquareRealizedHighProbabilityRegret_tail_total_delta
248       prior arms harms
249         (bernsteinSquareHighProbabilityLearningRate arms gamma horizon delta)
250         gamma heta hgamma_pos hgamma_lt_one loss comparator hcomparator
251         horizon hhorizon delta hdelta
252   dsimp only at htail ⊢
253   calc
254     (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
255         (bernsteinSquareHighProbabilityLearningRate arms gamma horizon delta)
256         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
257         {sample |
258           bernsteinSquareRealizedTunedThreshold arms gamma horizon delta <=
259             (Finset.range horizon).sum (fun t =>
260                 sampledTrajectoryRealizedLossAt t sample) -
261               (Finset.range horizon).sum (fun t =>
262                 predictableLossAt loss t sample comparator)} <=
263       (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
264         (bernsteinSquareHighProbabilityLearningRate arms gamma horizon delta)
265         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
266         {sample |
267           sampledPredictableBernsteinSquareRealizedHighProbabilityRegretBudget
268               arms
269                 (bernsteinSquareHighProbabilityLearningRate
270                   arms gamma horizon delta)
271               gamma horizon (delta / 4) (delta / 4) (delta / 4) <=
272             (Finset.range horizon).sum (fun t =>
273                 sampledTrajectoryRealizedLossAt t sample) -
274               (Finset.range horizon).sum (fun t =>
275                 predictableLossAt loss t sample comparator)} := by
276         apply measure_mono
277         intro sample hsample
278         exact hbudget.trans hsample
279     _ <= ENNReal.ofReal delta := htail
