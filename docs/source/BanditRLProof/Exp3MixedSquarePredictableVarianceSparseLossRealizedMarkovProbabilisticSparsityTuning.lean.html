3 /-!
4 # Learning-rate tuning under probabilistic sparse losses
5 
6 This module tunes `eta` for the generated realized predictable-variance EXP3
7 route whose support-sparsity contract may fail with positive probability.
8 The exact learning-rate scale is
9 
10 `S * T + sampledMixedSquaredPredictableVarianceRadius arms gamma v (delta / 5)`,
11 
12 where `v` is based on the unconditional `K * T` loss-mass envelope rather than
13 the sparse `S * T` envelope. The final theorem preserves the exact generated
14 sparsity-failure residual, and its practical consumer proves a
15 `delta + epsilon` tail.
16 
17 Gamma remains caller-selected. This module does not transfer the pathwise
18 sparse `14 * gamma * T` threshold to the probabilistic-sparsity setting.
19 -/

27 /-- Markov threshold for cumulative predictable mixed-square variance when
28 positive-probability sparsity failures require the global `K * T` envelope. -/
29 noncomputable def probabilisticSparseLossPredictableVarianceBudget
30     {Action : Type v} [DecidableEq Action]
31     (arms : Finset Action) (gamma : Real) (horizon : Nat)
32     (delta : Real) : Real :=
33   sampledPredictableGlobalVarianceMeanBudget arms gamma horizon / (delta / 5)
34 
35 theorem probabilisticSparseLossPredictableVarianceBudget_pos
36     {Action : Type v} [DecidableEq Action]
37     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
38     (gamma : Real) (hgamma_pos : 0 < gamma)
39     (horizon : Nat) (hhorizon : 0 < horizon)
40     (delta : Real) (hdelta : 0 < delta) :
41     0 < probabilisticSparseLossPredictableVarianceBudget
42       arms gamma horizon delta := by
43   have hcard_pos : 0 < (arms.card : Real) := by positivity
44   have hfloor_pos : 0 < gamma / (arms.card : Real) :=
45     div_pos hgamma_pos hcard_pos
46   have hglobal_pos :
47       0 < sampledPredictableGlobalVarianceMeanBudget
48         arms gamma horizon := by
49     unfold sampledPredictableGlobalVarianceMeanBudget
50     exact mul_pos (one_div_pos.2 hfloor_pos)
51       (mul_pos hcard_pos (Nat.cast_pos.2 hhorizon))
52   unfold probabilisticSparseLossPredictableVarianceBudget
53   exact div_pos hglobal_pos (div_pos hdelta (by norm_num))

55 /-- Complete Hedge scale with sparse observed-square mean and global Markov
56 variance control. -/
57 noncomputable def probabilisticSparseLossPredictableVarianceHighProbabilityScale
58     {Action : Type v} [DecidableEq Action]
59     (arms : Finset Action) (gamma : Real)
60     (horizon sparsity : Nat) (delta : Real) : Real :=
61   (sparsity : Real) * (horizon : Real) +
62     sampledMixedSquaredPredictableVarianceRadius arms gamma
63       (probabilisticSparseLossPredictableVarianceBudget
64         arms gamma horizon delta)
65       (delta / 5)

67 theorem probabilisticSparseLossPredictableVarianceHighProbabilityScale_pos
68     {Action : Type v} [DecidableEq Action]
69     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
70     (gamma : Real) (hgamma_pos : 0 < gamma)
71     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
72     (hsparsity : 0 < sparsity) (delta : Real) :
73     0 < probabilisticSparseLossPredictableVarianceHighProbabilityScale
74       arms gamma horizon sparsity delta := by
75   have hcard_pos : 0 < (arms.card : Real) := by positivity
76   have hfloor_pos : 0 < gamma / (arms.card : Real) :=
77     div_pos hgamma_pos hcard_pos
78   have hbudget_nonneg : 0 ≤ max (Real.log (1 / (delta / 5))) 0 :=
79     le_max_right _ _
80   have hradius_nonneg :
81       0 ≤ sampledMixedSquaredPredictableVarianceRadius arms gamma
82         (probabilisticSparseLossPredictableVarianceBudget
83           arms gamma horizon delta)
84         (delta / 5) := by
85     dsimp [sampledMixedSquaredPredictableVarianceRadius]
86     exact add_nonneg
87       (mul_nonneg (by norm_num) (Real.sqrt_nonneg _))
88       (div_nonneg hbudget_nonneg hfloor_pos.le)
89   have hloss_pos :
90       0 < (sparsity : Real) * (horizon : Real) :=
91     mul_pos (Nat.cast_pos.2 hsparsity) (Nat.cast_pos.2 hhorizon)
92   dsimp [probabilisticSparseLossPredictableVarianceHighProbabilityScale]
93   linarith

142 /-- Under `gamma ≤ 1/2`, entropy and stability cost at most three copies of
143 the balanced probabilistic-sparsity scale. -/
144 theorem probabilisticSparseLossPredictableVarianceHighProbabilityHedgeBudget_le_three_mul_sqrt
145     {Action : Type v} [DecidableEq Action]
146     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
147     (gamma : Real) (hgamma_pos : 0 < gamma)
148     (hgamma_le_half : gamma ≤ 1 / 2)
149     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
150     (hsparsity : 0 < sparsity) (delta : Real) :
151     Real.log (arms.card : Real) /
152           probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate
153             arms gamma horizon sparsity delta +
154         (probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate
155             arms gamma horizon sparsity delta * (1 / (1 - gamma))) *
156           probabilisticSparseLossPredictableVarianceHighProbabilityScale
157             arms gamma horizon sparsity delta ≤
158       3 * Real.sqrt
159         (Real.log (arms.card : Real) *
160           probabilisticSparseLossPredictableVarianceHighProbabilityScale
161             arms gamma horizon sparsity delta) := by
162   let eta :=
163     probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate
164       arms gamma horizon sparsity delta
165   let scale :=
166     probabilisticSparseLossPredictableVarianceHighProbabilityScale
167       arms gamma horizon sparsity delta
168   let entropy := Real.log (arms.card : Real) / eta
169   have hK_one : 1 < (arms.card : Real) := by
170     exact_mod_cast hcard_two
171   have heta : 0 < eta := by
172     dsimp [eta]
173     exact
174       probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate_pos
175         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
176   have hscale : 0 < scale := by
177     dsimp [scale]
178     exact
179       probabilisticSparseLossPredictableVarianceHighProbabilityScale_pos
180         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
181   have hbalance : eta ^ 2 * scale = Real.log (arms.card : Real) := by
182     dsimp [eta, scale]
183     exact
184       probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate_sq_mul_scale
185         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
186   have hentropy_nonneg : 0 ≤ entropy :=
187     div_nonneg (Real.log_pos hK_one).le heta.le
188   have hentropy_eq_unscaled : entropy = eta * scale := by
189     dsimp [entropy]
190     field_simp [ne_of_gt heta]
191     nlinarith [hbalance]
192   have hden : 0 < 1 - gamma := by linarith
193   have hfactor : 1 ≤ 2 * (1 - gamma) := by linarith
194   have hstable :
195       (eta * (1 / (1 - gamma))) * scale ≤ 2 * entropy := by
196     rw [show (eta * (1 / (1 - gamma))) * scale =
197       (eta * scale) / (1 - gamma) by ring]
198     rw [← hentropy_eq_unscaled, div_le_iff₀ hden]
199     have hmul := mul_le_mul_of_nonneg_left hfactor hentropy_nonneg
200     nlinarith
201   have hbalanced_sq :
202       entropy ^ 2 = Real.log (arms.card : Real) * scale := by
203     rw [hentropy_eq_unscaled]
204     calc
205       (eta * scale) ^ 2 = (eta ^ 2 * scale) * scale := by ring
206       _ = Real.log (arms.card : Real) * scale := by rw [hbalance]
207   have hradicand_nonneg :
208       0 ≤ Real.log (arms.card : Real) * scale :=
209     mul_nonneg (Real.log_pos hK_one).le hscale.le
210   have hentropy_eq_sqrt :
211       entropy = Real.sqrt
212         (Real.log (arms.card : Real) * scale) := by
213     have hsqrt_nonneg :=
214       Real.sqrt_nonneg (Real.log (arms.card : Real) * scale)
215     have hsqrt_sq := Real.sq_sqrt hradicand_nonneg
216     nlinarith [hbalanced_sq]
217   change entropy + (eta * (1 / (1 - gamma))) * scale ≤
218     3 * Real.sqrt (Real.log (arms.card : Real) * scale)
219   rw [← hentropy_eq_sqrt]
220   linarith

222 /-- Tuned regret threshold retaining the global Markov variance radius and the
223 three non-Hedge confidence terms. -/
224 noncomputable def probabilisticSparseLossPredictableVarianceRealizedMarkovTunedThreshold
225     {Action : Type v} [DecidableEq Action]
226     (arms : Finset Action) (gamma : Real)
227     (horizon sparsity : Nat) (delta : Real) : Real :=
228   3 * Real.sqrt
229       (Real.log (arms.card : Real) *
230         probabilisticSparseLossPredictableVarianceHighProbabilityScale
231           arms gamma horizon sparsity delta) +
232     gamma * (horizon : Real) +
233     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
234       arms gamma horizon (delta / 5) +
235     sampledComparatorEstimatorBernsteinConfidenceRadius
236       arms gamma horizon (delta / 5) +
237     sampledPredictableRealizedDeviationConfidenceRadius horizon (delta / 5)

355 /-- Practical eta-tuned `delta + epsilon` theorem under an exact generated-
356 measure bound on the sparsity-failure event. -/
357 theorem sampledPredictable_tunedProbabilisticSparseLossPredictableVarianceRealizedMarkovRegret_tail_of_sparsityFailure_le
358     {Env : Type u} {Action : Type v}
359     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
360     [MeasurableSpace Action] [MeasurableSingletonClass Action]
361     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
362     (prior : Measure Env) [IsProbabilityMeasure prior]
363     (arms : Finset Action) (harms : arms.Nonempty)
364     (hcard_two : 2 ≤ arms.card)
365     (gamma : Real) (hgamma_pos : 0 < gamma)
366     (hgamma_le_half : gamma ≤ 1 / 2)
367     (loss : PredictableLossVector Env Action)
368     (comparator : Action) (hcomparator : comparator ∈ arms)
369     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
370     (hsparsity : 0 < sparsity)
371     (delta epsilon : Real) (hdelta : 0 < delta) :
372     let eta :=
373       probabilisticSparseLossPredictableVarianceHighProbabilityLearningRate
374         arms gamma horizon sparsity delta
375     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
376       eta gamma hgamma_pos.le (by linarith : gamma ≤ 1) loss.environment
377     mu (sampledPredictableSparsityFailure arms loss horizon sparsity) ≤
378         ENNReal.ofReal epsilon →
379       mu {sample |
380           probabilisticSparseLossPredictableVarianceRealizedMarkovTunedThreshold
381               arms gamma horizon sparsity delta ≤
382             (Finset.range horizon).sum (fun t =>
383                 sampledTrajectoryRealizedLossAt t sample) -
384               (Finset.range horizon).sum (fun t =>
385                 predictableLossAt loss t sample comparator)} ≤
