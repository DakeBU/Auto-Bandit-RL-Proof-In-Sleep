3 /-!
4 # Refined expected penalty for scheduled half-Tsallis FTRL
5 
6 This module keeps the reciprocal-rate increments from the deterministic
7 time-varying penalty theorem instead of collapsing them into a terminal
8 potential mass.  The mass above the point-mass baseline is eliminated in
9 favor of suboptimal-arm terms, then transported through expectation by the
10 compiled square-root Jensen bridge.
11 -/

20 /-- The concavity tangent at one bounds the excess of `sqrt x` over `x`. -/
21 theorem sqrt_sub_self_le_half_one_sub (x : Real) (hx : 0 <= x) :
22     Real.sqrt x - x <= (1 - x) / 2 := by
23   have hsquare : 0 <= (Real.sqrt x - 1) ^ 2 := sq_nonneg _
24   have hsqrtSquare : (Real.sqrt x) ^ 2 = x := Real.sq_sqrt hx
25   nlinarith

27 /-- Above the point-mass baseline, half-Tsallis potential mass is controlled
28 by the paper's refined suboptimal-arm mass. -/
29 theorem halfTsallisPotentialMass_sub_one_le_two_mul_sum_erase_refined
30     {Action : Type u} [DecidableEq Action]
31     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
32     (probability : Action -> Real)
33     (hprobability : FTRL.finiteSimplex arms probability) :
34     halfTsallisPotentialMass arms probability - 1 <=
35       2 * (arms.erase best).sum (fun action =>
36         Real.sqrt (probability action) - probability action / 2) := by
37   rw [halfTsallisPotentialMass_eq_two_mul_powerSum_sub_one]
38   have hsplitSqrt := Finset.add_sum_erase arms
39     (fun action => Real.sqrt (probability action)) hbest
40   have hsplitProbability := Finset.add_sum_erase arms probability hbest
41   rw [hprobability.2] at hsplitProbability
42   have hbestBound := sqrt_sub_self_le_half_one_sub
43     (probability best) (hprobability.1 best hbest)
44   have heraseRewrite :
45       (arms.erase best).sum (fun action =>
46           Real.sqrt (probability action) - probability action / 2) =
47         (arms.erase best).sum (fun action => Real.sqrt (probability action)) -
48           (arms.erase best).sum probability / 2 := by
49     rw [Finset.sum_sub_distrib, Finset.sum_div]
