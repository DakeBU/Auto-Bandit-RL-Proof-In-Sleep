4 /-!
5 # Explicit exploration tuning for sparse pathwise variance
6 
7 This module closes the exploration-rate route for the four-event generated
8 realized-regret theorem under probabilistic sparsity. On the good event, the
9 predictable-variance budget is
10 
11 `(1 / (gamma / K)) * S * T`.
12 
13 Consequently the mixed fifth-root scale is driven by
14 
15 `K * S * (log K)^2 * log (4 / delta) / T^3`,
16 
17 with neither the extra factor `K` nor the polynomial `1 / delta` from the old
18 global-envelope Markov route. The final generated theorem retains the exact
19 sparsity-failure residual, and its practical endpoint has failure budget
20 `delta + epsilon` under the same internally eta/gamma-tuned measure.
21 
22 This is still an armwise aggregate sparse-loss theorem with bounded realized
23 deviation. It is not an all-horizon, best-arm first-order, Freedman, anytime,
24 or ideal EXP3.P result.
25 -/

46 /-- The sparse base, pathwise fifth-power, and cubic confidence contracts
47 bound the log-weighted predictable-variance radius by
48 `3 * gamma^2 * T^2`. -/

59         gamma ^ 2 * (horizon : Real))

101     rw [hlog_fourth]

113     mul_nonneg hvariance_pos.le hB.le
114   have hroot_nonneg :
115       0 <= Real.log K * Real.sqrt (variance * B) :=
116     mul_nonneg hlogK (Real.sqrt_nonneg _)
117   have hroot_right_nonneg : 0 <= gamma ^ 2 * T ^ 2 := by positivity

123         gamma ^ 2 * T ^ 2 := by
124     apply le_of_sq_le_sq
125     · rw [mul_pow, Real.sq_sqrt hradicand, hvariance]
126       field_simp [ne_of_gt hgamma_pos]
127       nlinarith [hmixed']
128     · exact hroot_right_nonneg
129   have hgamma_sq_le_one : gamma ^ 2 <= 1 := by
130     nlinarith [hgamma_pos, hgamma_le_half]
131   have hlog_le_Slog : Real.log K <= S * Real.log K := by
132     have hmul := mul_nonneg (sub_nonneg.mpr hS_one) hlogK
133     nlinarith
134   have hgamma_sq_T_le_T : gamma ^ 2 * T <= T := by
135     have hmul := mul_nonneg (sub_nonneg.mpr hgamma_sq_le_one) hT.le
136     nlinarith
137   have hlog_le_T : Real.log K <= T := by
138     have hbase' : S * Real.log K <= gamma ^ 2 * T := by
139       simpa [K, S, T] using hbase
140     linarith
141   have hconfidence' : K * B <= gamma ^ 3 * T := by
142     simpa [K, T, B] using hconfidence
143   have hconfidence_nonneg : 0 <= gamma ^ 3 * T := by positivity
144   have hprod_one := mul_le_mul_of_nonneg_right hconfidence' hlogK
145   have hprod_two :=

147   have hprod : (K * B) * Real.log K <= gamma ^ 3 * T ^ 2 := by
148     calc

150       _ <= (gamma ^ 3 * T) * T := hprod_two
