21 /-- A pointwise prescribed-arm gap ceiling bounds the forced charge by its cardinality. -/
22 theorem powerOfTwoForcedActionSuccessorPseudoRegret_le_card_mul
23     {K : Nat} {Feature : Type u}
24     [Fintype Feature]
25     (thetaStar : Feature -> Real)
26     (actionFeature : Fin K -> Feature -> Real)
27     (best : Fin K)
28     (forcedAction : Nat -> Fin K)
29     (horizon : Nat)
30     (forcedGapBound : Real)
31     (hforcedGap : forall exponent,
32       linearValue thetaStar (actionFeature best) -
33           linearValue thetaStar (actionFeature (forcedAction exponent)) <=
34         forcedGapBound) :
35     powerOfTwoForcedActionSuccessorPseudoRegret
36         thetaStar actionFeature best forcedAction horizon <=
37       ((powerOfTwoForcedIndexSet horizon).card : Real) *
38         forcedGapBound := by
39   have hsum :=
40     Finset.sum_le_card_nsmul
41       (powerOfTwoForcedIndexSet horizon)
42       (fun n =>
43         linearValue thetaStar (actionFeature best) -
44           linearValue thetaStar
45             (actionFeature (forcedAction (Nat.log2 (n + 1)))))
46       forcedGapBound
47       (fun n _hn => hforcedGap (Nat.log2 (n + 1)))
48   simpa only [
49     powerOfTwoForcedActionSuccessorPseudoRegret,
50     nsmul_eq_mul,
51     Nat.cast_ofNat] using hsum

54 theorem powerOfTwoForcedActionSuccessorPseudoRegret_le_log2_add_one_mul
55     {K : Nat} {Feature : Type u}
56     [Fintype Feature]
57     (thetaStar : Feature -> Real)
58     (actionFeature : Fin K -> Feature -> Real)
59     (best : Fin K)
60     (forcedAction : Nat -> Fin K)
61     (horizon : Nat)
62     (forcedGapBound : Real)
63     (hforcedGapBound : 0 <= forcedGapBound)
64     (hforcedGap : forall exponent,
65       linearValue thetaStar (actionFeature best) -
66           linearValue thetaStar (actionFeature (forcedAction exponent)) <=
67         forcedGapBound) :
68     powerOfTwoForcedActionSuccessorPseudoRegret
69         thetaStar actionFeature best forcedAction horizon <=
70       ((Nat.log2 horizon + 1 : Nat) : Real) * forcedGapBound := by
