5 /-!
6 # All-time realized EXP3 predictable-variance tail
7 
8 This module gives every positive prefix of one fixed generated EXP3 process a
9 geometric confidence share. The resulting countable union is controlled by
10 one outer confidence budget. This is countable outer-measure subadditivity,
11 not a Ville/Doob, mixture, optional-stopping, self-normalized, or general
12 Freedman theorem.
13 -/

24 /-- The selected-loss predictable-variance radius at prefix `n+1` under the
25 geometric confidence schedule. -/
26 noncomputable def sampledRealizedPredictableVarianceGeometricRadius
27     (varianceBudget : Nat -> Real) (delta : Real) (n : Nat) : Real :=
28   sampledRealizedPredictableVarianceRadius (varianceBudget n)
29     (Concentration.geometricConfidenceShare delta n)
30 
31 /-- Countable failure event over every positive prefix of one generated EXP3
32 trajectory. -/
33 noncomputable def sampledPredictableRealizedDeviationAllTimeFailureSet
34     {Env : Type u} {Action : Type v}
35     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
36     (arms : Finset Action) (eta gamma : Real)
37     (loss : PredictableLossVector Env Action)
38     (varianceBudget : Nat -> Real) (delta : Real) :
39     Set (Env × ((k : Nat) -> Action × Real)) :=
40   ⋃ n, {sample |
41       sampledRealizedPredictableVarianceGeometricRadius
42             varianceBudget delta n <=
43           (Finset.range (n + 1)).sum (fun i =>
44             sampledTrajectoryRealizedDeviationAt
45               arms eta gamma loss i sample) ∧
46         (Finset.range (n + 1)).sum (fun i =>
47           sampledTrajectoryPredictableRealizedVarianceAt
48             arms eta gamma loss i sample) <= varianceBudget n}
49 
50 /-- Membership is failure at at least one positive prefix. -/
51 theorem mem_sampledPredictableRealizedDeviationAllTimeFailureSet_iff
52     {Env : Type u} {Action : Type v}
53     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
54     (arms : Finset Action) (eta gamma : Real)
55     (loss : PredictableLossVector Env Action)
56     (varianceBudget : Nat -> Real) (delta : Real)
57     (sample : Env × ((k : Nat) -> Action × Real)) :
58     sample ∈ sampledPredictableRealizedDeviationAllTimeFailureSet
59         arms eta gamma loss varianceBudget delta ↔
60       ∃ n,
61         sampledRealizedPredictableVarianceGeometricRadius
62               varianceBudget delta n <=
63             (Finset.range (n + 1)).sum (fun i =>
64               sampledTrajectoryRealizedDeviationAt
65                 arms eta gamma loss i sample) ∧
66           (Finset.range (n + 1)).sum (fun i =>
67             sampledTrajectoryPredictableRealizedVarianceAt
68               arms eta gamma loss i sample) <= varianceBudget n := by
69   simp [sampledPredictableRealizedDeviationAllTimeFailureSet]

71 /-- On one generated EXP3 trajectory law, the joint
72 deviation/predictable-variance failures over all positive prefixes have total
73 mass at most the outer confidence budget. -/
74 theorem measure_sampledPredictableRealizedDeviationAllTimeFailureSet_le
75     {Env : Type u} {Action : Type v}
76     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
77     [MeasurableSpace Action] [MeasurableSingletonClass Action]
78     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
79     (prior : Measure Env) [IsProbabilityMeasure prior]
80     (arms : Finset Action) (harms : arms.Nonempty)
81     (eta gamma : Real) (hgamma_pos : 0 < gamma)
82     (hgamma_le_one : gamma <= 1)
83     (loss : PredictableLossVector Env Action)
84     (varianceBudget : Nat -> Real)
85     (hvarianceBudget : forall n, 0 < varianceBudget n)
86     (delta : Real) (hdelta : 0 < delta) :
87     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
88       eta gamma hgamma_pos.le hgamma_le_one loss.environment
89     mu (sampledPredictableRealizedDeviationAllTimeFailureSet
90         arms eta gamma loss varianceBudget delta) <=
91       ENNReal.ofReal delta := by
92   dsimp only
93   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
94     eta gamma hgamma_pos.le hgamma_le_one loss.environment
95   let deviation := fun n sample =>
96     (Finset.range (n + 1)).sum (fun i =>
97       sampledTrajectoryRealizedDeviationAt arms eta gamma loss i sample)
98   let predictableVariance := fun n sample =>
99     (Finset.range (n + 1)).sum (fun i =>
100       sampledTrajectoryPredictableRealizedVarianceAt
101         arms eta gamma loss i sample)
102   let varianceScale : Nat -> Real := fun _ => 1
103   let tiltCap : Nat -> Real := fun _ => 1
104   let deltaAt := Concentration.geometricConfidenceShare delta
105   have htail :=
106     Concentration.measure_iUnion_scheduled_deviation_ge_inter_variance_le_delta_of_fixedTilt_quadratic_tail
107       mu deviation predictableVariance varianceScale varianceBudget tiltCap
108         deltaAt delta
109         (fun _ => by simp [varianceScale])
110         hvarianceBudget
111         (fun _ => by simp [tiltCap])
112         (fun n => by
113           exact Concentration.geometricConfidenceShare_pos hdelta n)
114         (by
115           intro n tilt htilt_nonneg htilt_le
116           simpa [mu, deviation, predictableVariance, varianceScale, tiltCap,
117             deltaAt] using
118             (sampledPredictableRealizedDeviation_sum_tail_predictableVariance_fixedTilt
119               prior arms harms eta gamma hgamma_pos hgamma_le_one loss (n + 1)
120                 tilt htilt_nonneg htilt_le
121                 (Concentration.quadraticFixedMGFScheduledRadius
122                   varianceScale varianceBudget tiltCap deltaAt n)
123                 (varianceBudget n)))
124         (by
125           simpa [deltaAt] using
126             (Concentration.tsum_ofReal_geometricConfidenceShare hdelta.le).le)
127   simpa [mu, deviation, predictableVariance, varianceScale, tiltCap, deltaAt,
128     sampledPredictableRealizedDeviationAllTimeFailureSet,
129     sampledRealizedPredictableVarianceGeometricRadius,
130     sampledRealizedPredictableVarianceRadius,
131     Concentration.quadraticFixedMGFScheduledRadius,
132     Concentration.quadraticFixedMGFRadius] using htail
