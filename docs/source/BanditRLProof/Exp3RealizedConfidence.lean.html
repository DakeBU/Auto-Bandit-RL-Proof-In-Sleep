4 # Delta-shaped confidence for realized EXP3 deviation
5 
6 This module converts the finite-horizon ENNReal Azuma bound into an explicit
7 square-root confidence radius.  It closes the one-sided confidence theorem for
8 realized loss minus the exploration-mixed predictable conditional mean; it
9 does not identify the estimator-valued Hedge comparator with true comparator
10 loss.
11 -/

19 noncomputable def sampledPredictableRealizedDeviationConfidenceRadius
20     (horizon : Nat) (delta : Real) : Real :=
21   Real.sqrt
22     (2 * ((((horizon : NNReal) *
23       Concentration.intervalVarianceProxy 0 1 : NNReal)) : Real) *
24         Real.log (1 / delta))

45 theorem sampledPredictableRealizedDeviation_sum_tail_exp_neg_budget
46     {Env : Type u} {Action : Type v}
47     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
48     [MeasurableSpace Action] [MeasurableSingletonClass Action]
49     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
50     (prior : Measure Env) [IsProbabilityMeasure prior]
51     (arms : Finset Action) (harms : arms.Nonempty)
52     (eta gamma : Real)
53     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
54     (loss : PredictableLossVector Env Action) (horizon : Nat)
55     (hhorizon : 0 < horizon) (budget : Real) :
56     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
57       eta gamma hgamma_nonneg hgamma_le_one loss.environment
58     mu {sample | Real.sqrt
59           (2 * ((((horizon : NNReal) *
60             Concentration.intervalVarianceProxy 0 1 : NNReal)) : Real) * budget) <=
61         (Finset.range horizon).sum (fun i =>
62           sampledTrajectoryRealizedDeviationAt arms eta gamma loss i sample)} <=
63       ENNReal.ofReal (Real.exp (-budget)) := by
64   dsimp only
65   let variance : Real := ((((horizon : NNReal) *
66     Concentration.intervalVarianceProxy 0 1 : NNReal)) : Real)
67   let radius : Real := Real.sqrt (2 * variance * budget)
68   have hvariance_pos : 0 < variance := by
69     change 0 < (horizon : Real) *
70       ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real)
71     exact mul_pos (by exact_mod_cast hhorizon)
72       intervalVarianceProxy_zero_one_pos
73   have htail := sampledPredictableRealizedDeviation_sum_tail_ennreal
74     prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon
75       (eps := radius) (Real.sqrt_nonneg _)
76   have hradius_sq : 2 * variance * budget <= radius ^ 2 := by
77     dsimp [radius]
78     rw [Real.sq_sqrt']
79     exact le_max_left _ _
80   have hden_pos : 0 < 2 * variance := mul_pos (by norm_num) hvariance_pos
81   have hbudget_le : budget <= radius ^ 2 / (2 * variance) := by
82     rw [le_div_iff₀ hden_pos]
83     simpa [mul_assoc, mul_comm, mul_left_comm] using hradius_sq
84   have hexp_le :
85       Real.exp (-radius ^ 2 / (2 * variance)) <= Real.exp (-budget) := by
86     apply Real.exp_le_exp.mpr
87     simpa [neg_div] using neg_le_neg hbudget_le
88   exact htail.trans (ENNReal.ofReal_le_ofReal hexp_le)

90 theorem sampledPredictableRealizedDeviation_sum_tail_delta
91     {Env : Type u} {Action : Type v}
92     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
93     [MeasurableSpace Action] [MeasurableSingletonClass Action]
94     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
95     (prior : Measure Env) [IsProbabilityMeasure prior]
96     (arms : Finset Action) (harms : arms.Nonempty)
97     (eta gamma : Real)
98     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
99     (loss : PredictableLossVector Env Action) (horizon : Nat)
100     (hhorizon : 0 < horizon) (delta : Real) (hdelta : 0 < delta) :
101     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
102       eta gamma hgamma_nonneg hgamma_le_one loss.environment
103     mu {sample |
104         sampledPredictableRealizedDeviationConfidenceRadius horizon delta <=
105           (Finset.range horizon).sum (fun i =>
106             sampledTrajectoryRealizedDeviationAt arms eta gamma loss i sample)} <=
107       ENNReal.ofReal delta := by
108   dsimp only
109   have htail :=
110     sampledPredictableRealizedDeviation_sum_tail_exp_neg_budget
111       prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss horizon
112         hhorizon (Real.log (1 / delta))
113   have hscale : 0 < 1 / delta := one_div_pos.mpr hdelta
114   have hexp : Real.exp (-(Real.log (1 / delta))) = delta := by
115     rw [Real.exp_neg, Real.exp_log hscale]
116     field_simp
117   rw [hexp] at htail
118   simpa only [sampledPredictableRealizedDeviationConfidenceRadius] using htail
