9 # Deterministic regret/count bounds
10 
11 This module contains algorithm-neutral deterministic scaffolds that convert
12 per-arm pull-count upper bounds into pseudo-regret upper bounds.  It stays
13 below probability, expectation, filtrations, concentration, and algorithm
14 final theorem work.
15 -/

19 /--
20 If every arm's pull count is bounded by `B`, then pseudo-regret is bounded by
21 the corresponding gap-weighted count budget.
22 
23 This is the `REGRET-COUNT-BOUND` deterministic scaffold.  It consumes only the
24 compiled regret decomposition and model-derived gap nonnegativity.
25 -/
26 theorem pseudoRegret_le_finset_sum_gap_mul_count_bound
27     {K : Nat}
28     (model : FiniteBanditModel K)
29     (action : ActionTrace (Fin K))
30     (n : Nat)
31     (B : Fin K -> Rat)
32     (hB : forall a : Fin K,
33       ((pullCount action a n : Nat) : Rat) <= B a) :
34     pseudoRegret model action n <=
35       (Finset.univ : Finset (Fin K)).sum
36         (fun a : Fin K => model.gap a * B a) := by
37   rw [pseudoRegret_eq_finset_sum_gap_mul_pullCount]
38   apply Finset.sum_le_sum
39   intro a _ha
40   exact
41     mul_le_mul_of_nonneg_left
42       (hB a)
43       (FiniteBanditModel.gap_nonneg model a)

49 This is the `REGRET-NAT-COUNT-BOUND` adapter.  It is algorithm-neutral and

52 theorem pseudoRegret_le_finset_sum_gap_mul_nat_count_bound
53     {K : Nat}
54     (model : FiniteBanditModel K)
55     (action : ActionTrace (Fin K))
56     (n : Nat)
57     (B : Fin K -> Nat)
58     (hB : forall a : Fin K,
59       pullCount action a n <= B a) :
60     pseudoRegret model action n <=
61       (Finset.univ : Finset (Fin K)).sum
62         (fun a : Fin K => model.gap a * (((B a : Nat) : Rat))) := by
63   exact
64     pseudoRegret_le_finset_sum_gap_mul_count_bound
65       (model := model)

67       (n := n)
68       (B := fun a : Fin K => (((B a : Nat) : Rat)))
69       (hB := by
70         intro a
71         exact (@Nat.cast_le Rat _ _ _ _ _).mpr (hB a))

74 A uniform Nat-valued pull-count bound implies pseudo-regret is bounded by the
75 sum of model gaps times that uniform count budget.
76 
77 This is the `REGRET-UNIFORM-NAT-COUNT-BOUND` adapter.  It is still
78 algorithm-neutral and does not prove any ETC/UCB-specific count fact.

80 theorem pseudoRegret_le_sum_gap_mul_uniform_nat_count_bound
81     {K : Nat}

83     (action : ActionTrace (Fin K))
84     (n B : Nat)
85     (hB : forall a : Fin K,
86       pullCount action a n <= B) :
87     pseudoRegret model action n <=
88       ((Finset.univ : Finset (Fin K)).sum
89         (fun a : Fin K => model.gap a)) * (((B : Nat) : Rat)) := by
90   calc
91     pseudoRegret model action n
92         <=
93       (Finset.univ : Finset (Fin K)).sum
94         (fun a : Fin K =>
95           model.gap a * (((B : Nat) : Rat))) := by
96           exact
97             pseudoRegret_le_finset_sum_gap_mul_nat_count_bound
98               (model := model)
99               (action := action)
100               (n := n)
101               (B := fun _a : Fin K => B)
102               (hB := hB)
103     _ =
104       ((Finset.univ : Finset (Fin K)).sum
105         (fun a : Fin K => model.gap a)) * (((B : Nat) : Rat)) := by
106           exact

108               (s := (Finset.univ : Finset (Fin K)))
