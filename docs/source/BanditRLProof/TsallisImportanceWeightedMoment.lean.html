8 # Tsallis importance-weighted power moments
9 
10 This module isolates the finite-sum power-moment calculation used by the
11 Tsallis-INF stability analysis.  The sampled importance-weighted loss is
12 weighted by the inverse Hessian scale `p^(2 - alpha)`.  Taking the finite sum
13 weighted by the sampling masses gives exactly the power-weighted second moment
14 `sum_i loss_i^2 * p_i^(1 - alpha)`.
15 
16 The result is deterministic finite-sum algebra.  It does not prove the
17 preceding Hessian/conjugate-potential stability inequality, identify a
18 conditional action law, or establish a regret theorem.
19 -/
20 
21 namespace BanditRLProof
22 namespace Tsallis
23 
24 universe u
25 
26 /-- The inverse-Hessian-weighted square of one sampled loss estimate. -/
27 noncomputable def powerWeightedSquaredImportanceWeightedLoss
28     {Action : Type u}
29     (arms : Finset Action) (alpha : Real)
30     (prob loss : Action -> Real) (chosen : Action) : Real :=
31   arms.sum (fun action =>
32     (prob action) ^ (2 - alpha) *
33       (Exp3.importanceWeightedLoss prob loss chosen action) ^ 2)
34 
35 /--
36 Pathwise power-moment identity: only the sampled coordinate remains.
37 -/
38 theorem powerWeightedSquaredImportanceWeightedLoss_eq_selected
39     {Action : Type u} [DecidableEq Action]
40     (arms : Finset Action) (alpha : Real)
41     (prob loss : Action -> Real) (chosen : Action)
42     (hchosen : chosen ∈ arms) (hprob : 0 < prob chosen) :
43     powerWeightedSquaredImportanceWeightedLoss
44         arms alpha prob loss chosen =
45       (loss chosen) ^ 2 * (prob chosen) ^ (-alpha) := by
46   unfold powerWeightedSquaredImportanceWeightedLoss
47   rw [Finset.sum_eq_single chosen]
48   · simp only [Exp3.importanceWeightedLoss]
49     have hpow :
50         (prob chosen) ^ (2 - alpha) =
51           (prob chosen) ^ (2 : Real) * (prob chosen) ^ (-alpha) := by
52       rw [← Real.rpow_add hprob]
53       congr 1
54     rw [hpow, Real.rpow_two]
55     simp only [if_true]
56     field_simp [ne_of_gt hprob]
57   · intro action _haction hne
58     simp [Exp3.importanceWeightedLoss, Ne.symm hne]
59   · exact fun hnotmem => (hnotmem hchosen).elim
60 
61 /--
62 The sampling-mass-weighted finite sum equals the power-weighted loss square.
63 -/
64 theorem sum_prob_mul_powerWeightedSquaredImportanceWeightedLoss_eq
65     {Action : Type u} [DecidableEq Action]
66     (arms : Finset Action) (alpha : Real)
67     (prob loss : Action -> Real)
68     (hprob : forall action, action ∈ arms -> 0 < prob action) :
69     arms.sum (fun chosen =>
70         prob chosen *
71           powerWeightedSquaredImportanceWeightedLoss
72             arms alpha prob loss chosen) =
73       arms.sum (fun action =>
74         (loss action) ^ 2 * (prob action) ^ (1 - alpha)) := by
75   apply Finset.sum_congr rfl
76   intro chosen hchosen
77   rw [powerWeightedSquaredImportanceWeightedLoss_eq_selected
78     arms alpha prob loss chosen hchosen (hprob chosen hchosen)]
79   rw [show (1 - alpha : Real) = 1 + (-alpha) by ring,
80     Real.rpow_add (hprob chosen hchosen), Real.rpow_one]
81   ring
82 
83 /--
84 For losses in `[0,1]`, the Tsallis importance-weighted power moment is bounded
85 by the finite power sum with exponent `1 - alpha`.
86 -/
87 theorem sum_prob_mul_powerWeightedSquaredImportanceWeightedLoss_le_powerSum
88     {Action : Type u} [DecidableEq Action]
89     (arms : Finset Action) (alpha : Real)
90     (prob loss : Action -> Real)
91     (hprob : forall action, action ∈ arms -> 0 < prob action)
92     (hloss : forall action, action ∈ arms ->
93       0 <= loss action ∧ loss action <= 1) :
94     arms.sum (fun chosen =>
95         prob chosen *
96           powerWeightedSquaredImportanceWeightedLoss
97             arms alpha prob loss chosen) <=
98       powerSum arms (1 - alpha) prob := by
99   rw [sum_prob_mul_powerWeightedSquaredImportanceWeightedLoss_eq
100     arms alpha prob loss hprob]
101   unfold powerSum
102   apply Finset.sum_le_sum
103   intro action haction
104   have hl := hloss action haction
105   have hpow : 0 <= (prob action) ^ (1 - alpha) :=
106     Real.rpow_nonneg (le_of_lt (hprob action haction)) (1 - alpha)
107   nlinarith [sq_nonneg (loss action), mul_self_le_mul_self hl.1 hl.2]
108 
109 end Tsallis
