17 /-- Shift a generated trajectory so that `start` becomes local time zero. -/
18 def oracleRestartShiftedTrajectory
19     {Env : Type u} {Action : Type v}
20     (start : Nat) (sample : Env × ((k : Nat) -> Action × Real)) :
21     Env × ((k : Nat) -> Action × Real) :=
22   (sample.1, fun localTime => sample.2 (start + localTime))

40 /-- Restart epoch starts are monotone in actual time. -/
41 theorem OracleRestartSchedule.monotone_start
42     (schedule : OracleRestartSchedule) :
43     Monotone schedule.start := by
44   apply monotone_nat_of_le_succ
45   intro n
46   rcases schedule.start_succ n with hcontinue | hrestart
47   · exact hcontinue.ge
48   · rw [hrestart]
49     exact Nat.le_succ_of_le (schedule.start_le n)

51 /-- Every epoch identifier visited by the schedule is a fixed point of
52 `schedule.start`. -/
53 @[simp]
54 theorem OracleRestartSchedule.start_start
55     (schedule : OracleRestartSchedule) (t : Nat) :
56     schedule.start (schedule.start t) = schedule.start t := by
57   induction t with
58   | zero => simp [schedule.start_zero]
59   | succ n ih =>
60       rcases schedule.start_succ n with hcontinue | hrestart
61       · simpa [hcontinue] using ih
62       · simp [hrestart]

64 /-- A restart schedule cannot leave an epoch and later return to it. -/
65 theorem OracleRestartSchedule.start_eq_of_between
66     (schedule : OracleRestartSchedule) {epoch localTime t : Nat}
67     (ht : schedule.start t = epoch)
68     (hepoch : epoch <= localTime) (hlocalTime : localTime <= t) :
69     schedule.start localTime = epoch := by
70   have hlow := schedule.monotone_start
71     (show schedule.start t <= localTime by simpa [ht] using hepoch)
72   have hhigh := schedule.monotone_start hlocalTime
73   rw [schedule.start_start, ht] at hlow
74   rw [ht] at hhigh
75   omega

77 /-- At every actual time, the restarted probability is exactly the scheduled
78 probability at local time `t - schedule.start t` on the shifted trajectory. -/
79 theorem sampledOracleRestartHalfTsallisProbabilityAtTime_eq_scheduled_shift
80     {Env : Type u} {Action : Type v} [DecidableEq Action]
81     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
82     (schedule : OracleRestartSchedule) (t : Nat)
83     (sample : Env × ((k : Nat) -> Action × Real)) :
84     sampledOracleRestartHalfTsallisProbabilityAtTime
85         arms harms eta schedule t sample =
86       sampledScheduledHalfTsallisProbabilityAtTime
87         arms harms eta (t - schedule.start t)
88         (oracleRestartShiftedTrajectory (schedule.start t) sample) := by
89   cases t with
90   | zero =>
91       simp [sampledOracleRestartHalfTsallisProbabilityAtTime,
92         sampledScheduledHalfTsallisProbabilityAtTime,
93         schedule.start_zero, oracleRestartShiftedTrajectory]
94   | succ n =>
95       by_cases hboundary : schedule.start (n + 1) = n + 1
96       · simp [sampledOracleRestartHalfTsallisProbabilityAtTime,
97           sampledScheduledHalfTsallisProbabilityAtTime, hboundary,
98           sampledOracleRestartHalfTsallisHistoryDistribution_of_boundary,
99           oracleRestartShiftedTrajectory]
100       · have hstart : schedule.start (n + 1) <= n :=
101           schedule.start_succ_le_of_ne n hboundary
102         have hlocalTime :
103             n + 1 - schedule.start (n + 1) =
104               (n - schedule.start (n + 1)) + 1 := by
105           omega
106         rw [sampledOracleRestartHalfTsallisProbabilityAtTime,
107           sampledOracleRestartHalfTsallisHistoryDistribution_of_continuation
108             arms harms eta schedule n hboundary
109               (Preorder.frestrictLe n sample.2),
110           hlocalTime]
111         change
112           sampledScheduledHalfTsallisHistoryDistribution
113               arms harms eta (n - schedule.start (n + 1))
114               (oracleRestartLocalPairHistory
115                 (schedule.start (n + 1)) n hstart
116                 (Preorder.frestrictLe n sample.2)) =
117             sampledScheduledHalfTsallisHistoryDistribution
118               arms harms eta (n - schedule.start (n + 1))
119               (Preorder.frestrictLe (n - schedule.start (n + 1))
120                 (oracleRestartShiftedTrajectory
121                   (schedule.start (n + 1)) sample).2)
122         congr 1

264 /-- Every visited restart epoch fiber is a nonempty contiguous range starting
265 at its epoch identifier. -/
266 theorem exists_oracleRestartEpochRounds_eq_image_range
267     (schedule : OracleRestartSchedule) (horizon epoch : Nat)
268     (hepoch : epoch ∈ oracleRestartScheduleEpochs schedule horizon) :
269     ∃ localHorizon,
270       oracleRestartEpochRounds schedule.start horizon epoch =
271         (Finset.range (localHorizon + 1)).image
272           (fun localTime => epoch + localTime) := by
273   rw [oracleRestartScheduleEpochs] at hepoch
274   rcases Finset.mem_image.mp hepoch with
275     ⟨witness, hwitnessRange, hwitnessStart⟩
276   let rounds := oracleRestartEpochRounds schedule.start horizon epoch
277   have hwitnessRounds : witness ∈ rounds := by
278     simp [rounds, oracleRestartEpochRounds, hwitnessRange, hwitnessStart]
279   have hroundsNonempty : rounds.Nonempty :=
280     ⟨witness, hwitnessRounds⟩
281   let last := rounds.max' hroundsNonempty
282   have hlastRounds : last ∈ rounds :=
283     Finset.max'_mem rounds hroundsNonempty
284   have hlastInfo :
285       last ∈ Finset.range (horizon + 1) ∧
286         schedule.start last = epoch := by
287     simpa [rounds, oracleRestartEpochRounds] using hlastRounds
288   have hepochLeLast : epoch <= last := by
289     have hstartLe := schedule.start_le last
290     omega
291   refine ⟨last - epoch, ?_⟩
292   change rounds =
293     (Finset.range (last - epoch + 1)).image
294       (fun localTime => epoch + localTime)
295   ext actualTime
296   constructor
297   · intro hactualTime
298     have hactualInfo :
299         actualTime ∈ Finset.range (horizon + 1) ∧
300           schedule.start actualTime = epoch := by
301       simpa [rounds, oracleRestartEpochRounds] using hactualTime
302     have hepochLeActualTime : epoch <= actualTime := by
303       have hstartLe := schedule.start_le actualTime
304       omega
305     have hactualTimeLeLast : actualTime <= last :=
306       rounds.le_max' actualTime hactualTime
307     refine Finset.mem_image.mpr
308       ⟨actualTime - epoch, Finset.mem_range.mpr ?_, ?_⟩
309     · omega
310     · omega
311   · intro hactualTime
312     rcases Finset.mem_image.mp hactualTime with
313       ⟨localTime, hlocalTimeRange, hlocalTimeEq⟩
314     have hlocalTimeLt : localTime < last - epoch + 1 :=
315       Finset.mem_range.mp hlocalTimeRange
316     have hactualTimeLeLast : epoch + localTime <= last := by
317       omega
318     have hactualTimeStart :
319         schedule.start (epoch + localTime) = epoch :=
320       schedule.start_eq_of_between hlastInfo.2
321         (Nat.le_add_right epoch localTime) hactualTimeLeLast
322     have hlastLeHorizon : last <= horizon := by
323       exact Nat.le_of_lt_succ (Finset.mem_range.mp hlastInfo.1)
324     have hactualTimeRange :
325         epoch + localTime ∈ Finset.range (horizon + 1) := by
326       exact Finset.mem_range.mpr (by omega)
327     have hactualTimeRounds : epoch + localTime ∈ rounds := by
328       simp [rounds, oracleRestartEpochRounds, hactualTimeRange,
329         hactualTimeStart]
330     rw [← hlocalTimeEq]
331     exact hactualTimeRounds
