30 /-- An exact predictable fixed-gap law identifies scheduled environment
31 regret against the best-arm point mass with suboptimal-arm gap mass
32 pathwise. -/
33 theorem sampledScheduledHalfTsallisPredictableEnvironmentRegret_pointMass_eq_suboptimalGapMass
34     {Env : Type u} {Action : Type v}
35     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
36     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
37     (loss : Exp3.PredictableLossVector Env Action)
38     {best : Action} (hbest : best ∈ arms) (gap : Action -> Real)
39     (horizon : Nat)
40     (hgapLaw : ∀ t sample action, action ∈ arms ->
41       Exp3.predictableLossAt loss t sample action -
42         Exp3.predictableLossAt loss t sample best = gap action)
43     (sample : Env × ((k : Nat) -> Action × Real)) :
44     sampledScheduledHalfTsallisPredictableEnvironmentRegret
45         arms harms eta loss (pointMass best) horizon sample =
46       sampledScheduledHalfTsallisPredictableSuboptimalGapMass
47         arms harms eta best gap horizon sample := by
48   unfold sampledScheduledHalfTsallisPredictableEnvironmentRegret
49     sampledScheduledHalfTsallisPredictableSuboptimalGapMass
50   apply Finset.sum_congr rfl
51   intro t _ht
52   rw [linearLoss_sub_pointMass_eq_gapMass
53     arms hbest
54     (sampledScheduledHalfTsallisProbabilityAtTime arms harms eta t sample)
55     (Exp3.predictableLossAt loss t sample) gap
56     (finiteSimplex_sampledScheduledHalfTsallisProbabilityAtTime
57       arms harms eta t sample)
58     (hgapLaw t sample)]
59   rw [← Finset.sum_erase_add _ _ hbest]
60   have hgapBest : gap best = 0 := by
61     have h := hgapLaw t sample best hbest
62     linarith
63   simp [hgapBest]

65 /-- Integrating scheduled pathwise suboptimal gap mass gives the deterministic
66 time-by-arm sum of gaps times expected scheduled action probabilities. -/
67 theorem integral_sampledScheduledHalfTsallisPredictableSuboptimalGapMass_eq
68     {Env : Type u} {Action : Type v}
69     [MeasurableSpace Env]
70     [MeasurableSpace Action] [MeasurableSingletonClass Action]
71     [DecidableEq Action]
72     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
73     [IsProbabilityMeasure mu]
74     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
75     (best : Action) (gap : Action -> Real) (horizon : Nat) :
76     integral mu (sampledScheduledHalfTsallisPredictableSuboptimalGapMass
77         arms harms eta best gap horizon) =
78       (Finset.range (horizon + 1)).sum (fun t =>
79         (arms.erase best).sum (fun action =>
80           gap action * sampledScheduledHalfTsallisExpectedProbabilityAt
81             mu arms harms eta t action)) := by
82   unfold sampledScheduledHalfTsallisPredictableSuboptimalGapMass
83   rw [ExpectationBochnerSums.integral_finset_sum mu
84     (Finset.range (horizon + 1))]
85   · apply Finset.sum_congr rfl
86     intro t _ht
87     rw [ExpectationBochnerSums.integral_finset_sum mu (arms.erase best)]
88     · apply Finset.sum_congr rfl
89       intro action haction
90       rw [integral_mul_const]
91       unfold sampledScheduledHalfTsallisExpectedProbabilityAt
92       ring
93     · intro action haction
94       exact
95         (integrable_sampledScheduledHalfTsallisProbabilityAtTime
96           mu arms harms eta t action (Finset.mem_of_mem_erase haction)).mul_const
97           (gap action)
98   · intro t _ht
99     exact IntegrabilitySums.integrable_finset_sum mu (arms.erase best)
100       (fun action sample =>
101         sampledScheduledHalfTsallisProbabilityAtTime
102           arms harms eta t sample action * gap action)
103       (fun action haction =>
104         (integrable_sampledScheduledHalfTsallisProbabilityAtTime
105           mu arms harms eta t action (Finset.mem_of_mem_erase haction)).mul_const
106           (gap action))

192 /-- The generated scheduled regret bound with an explicit self-bounding
193 premise becomes automatic under an exact predictable fixed-gap law. -/
194 theorem integral_sampledScheduledHalfTsallisPredictableEnvironmentRegret_pointMass_le_of_fixedGap
195     {Env : Type u} {Action : Type v}
196     [MeasurableSpace Env] [StandardBorelSpace Env]
197     [MeasurableSpace Action] [MeasurableSingletonClass Action]
198     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
199     (prior : Measure Env) [IsProbabilityMeasure prior]
200     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
201     (loss : Exp3.PredictableLossVector Env Action)
202     {best : Action} (hbest : best ∈ arms) (horizon : Nat)
203     (heta : forall t, t <= horizon -> 0 < eta t)
204     (heta_le : forall t, t <= horizon -> eta t <= 1 / 2)
205     (hetaMono : forall t, t < horizon -> eta (t + 1) <= eta t)
206     (gap : Action -> Real)
207     (hgapPos : forall action, action ∈ arms.erase best -> 0 < gap action)
208     (hgapLaw : ∀ t sample action, action ∈ arms ->
209       Exp3.predictableLossAt loss t sample action -
210         Exp3.predictableLossAt loss t sample best = gap action)
211     (corruption : Real) (hcorruption : 0 <= corruption) :
212     let selector :=
213       canonicalHalfTsallisScheduleGeneratedSelectorMeasurability
214         arms harms eta loss
215     let mu := prior ⊗ₘ sampledScheduledHalfTsallisTrajectoryKernel
216       arms harms eta selector.finiteHistory loss.environment
217     integral mu (sampledScheduledHalfTsallisPredictableEnvironmentRegret
218         arms harms eta loss (pointMass best) horizon) <=
219       2 * ((Finset.range (horizon + 1)).sum (fun t =>
220           2 * (eta t) ^ 2) +
221         (halfTsallisPotentialMass arms
222             (initialHalfTsallisDistribution arms harms (eta 0)) / eta horizon -
223           1 / eta horizon)) +
224         ((Finset.range (horizon + 1)).product (arms.erase best)).sum
225           (fun index => (2 * eta index.1) ^ 2 / gap index.2) +
226         corruption := by
227   dsimp only
228   apply
229     integral_sampledScheduledHalfTsallisPredictableEnvironmentRegret_pointMass_le_of_selfBounding
230       prior arms harms eta loss hbest horizon heta heta_le hetaMono gap
231         hgapPos corruption
232   exact
233     integral_sampledScheduledHalfTsallisPredictableEnvironmentRegret_hasSelfBounding_of_fixedGap
234       prior arms harms eta loss hbest gap horizon hgapLaw corruption hcorruption
