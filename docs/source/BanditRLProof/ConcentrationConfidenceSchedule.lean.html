3 /-!
4 # Summable confidence schedules
5 
6 This module records reusable deterministic schedules for countable confidence
7 budgets. It contains no stochastic-process or algorithm assumptions.
8 -/
9 
10 namespace BanditRLProof.Concentration
11 
12 open scoped ENNReal
13 
14 /-- The geometric confidence schedule `delta / 2 / 2^n`. -/
15 noncomputable def geometricConfidenceShare
16     (delta : Real) (n : Nat) : Real :=
17   delta / 2 / 2 ^ n
18 
19 /-- Every geometric confidence share is positive when its outer budget is. -/
20 theorem geometricConfidenceShare_pos
21     {delta : Real} (hdelta : 0 < delta) (n : Nat) :
22     0 < geometricConfidenceShare delta n := by
23   unfold geometricConfidenceShare
24   positivity
25 
26 /-- The ENNReal masses of the geometric confidence shares sum exactly to the
27 outer budget. -/
28 theorem tsum_ofReal_geometricConfidenceShare
29     {delta : Real} (hdelta : 0 <= delta) :
30     ∑' n, ENNReal.ofReal (geometricConfidenceShare delta n) =
31       ENNReal.ofReal delta := by
32   have hreal : HasSum (geometricConfidenceShare delta) delta := by
33     simpa [geometricConfidenceShare] using hasSum_geometric_two' delta
34   have hnonneg : forall n, 0 <= geometricConfidenceShare delta n := by
35     intro n
36     unfold geometricConfidenceShare
37     positivity
38   have hnnreal :
39       HasSum
40           (fun n => (geometricConfidenceShare delta n).toNNReal)
41           delta.toNNReal :=
42     hreal.toNNReal hnonneg
43   have hcoe :
44       HasSum
45           (fun n => ((geometricConfidenceShare delta n).toNNReal : ENNReal))
46           (delta.toNNReal : ENNReal) :=
47     ENNReal.hasSum_coe.mpr hnnreal
48   have hofReal :
49       HasSum
50           (fun n => ENNReal.ofReal (geometricConfidenceShare delta n))
51           (ENNReal.ofReal delta) := by
52     simpa only [ENNReal.ofReal_eq_coe_nnreal] using hcoe
53   exact hofReal.tsum_eq
54 
55 /-- Positive telescoping weight `1 / ((n+1)(n+2))`.  Unlike the geometric
56 weight, its reciprocal grows only polynomially in time. -/
57 noncomputable def telescopingConfidenceWeight (n : Nat) : Real :=
58   1 / (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real))
59 
60 /-- The telescoping weight is a difference of consecutive reciprocals. -/
61 theorem telescopingConfidenceWeight_eq_sub (n : Nat) :
62     telescopingConfidenceWeight n =
63       1 / (((n + 1 : Nat) : Real)) -
64         1 / (((n + 2 : Nat) : Real)) := by
65   unfold telescopingConfidenceWeight
66   field_simp
67   norm_num [Nat.cast_add]
68 
69 /-- Exact finite partial sum of the telescoping weights. -/
70 theorem sum_range_telescopingConfidenceWeight (n : Nat) :
71     (Finset.range n).sum telescopingConfidenceWeight =
72       1 - 1 / (((n + 1 : Nat) : Real)) := by
73   rw [Finset.sum_congr rfl
74     (fun i _hi => telescopingConfidenceWeight_eq_sub i)]
75   rw [Finset.sum_range_sub']
76   norm_num
77 
78 /-- Every telescoping confidence weight is nonnegative. -/
79 theorem telescopingConfidenceWeight_nonneg (n : Nat) :
80     0 <= telescopingConfidenceWeight n := by
81   unfold telescopingConfidenceWeight
82   positivity
83 
84 /-- The telescoping weights sum exactly to one. -/
85 theorem hasSum_telescopingConfidenceWeight :
86     HasSum telescopingConfidenceWeight 1 := by
87   rw [hasSum_iff_tendsto_nat_of_nonneg telescopingConfidenceWeight_nonneg]
88   simp_rw [sum_range_telescopingConfidenceWeight]
89   have hcast :
90       Filter.Tendsto
91         (fun n : Nat => (((n + 1 : Nat) : Real)))
92         Filter.atTop Filter.atTop := by
93     exact tendsto_natCast_atTop_atTop.comp
94       (Filter.tendsto_add_atTop_nat 1)
95   have hinv :
96       Filter.Tendsto
97         (fun n : Nat => (1 / (((n + 1 : Nat) : Real))))
98         Filter.atTop (nhds 0) := by
99     simpa only [one_div] using tendsto_inv_atTop_zero.comp hcast
100   simpa using tendsto_const_nhds.sub hinv
101 
102 /-- Time-`n` confidence share `delta / ((n+1)(n+2))`. -/
103 noncomputable def telescopingConfidenceShare
104     (delta : Real) (n : Nat) : Real :=
105   delta * telescopingConfidenceWeight n
106 
107 /-- Display the telescoping share as the intended quotient. -/
108 theorem telescopingConfidenceShare_eq_div (delta : Real) (n : Nat) :
109     telescopingConfidenceShare delta n =
110       delta /
111         (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) := by
112   simp [telescopingConfidenceShare, telescopingConfidenceWeight,
113     div_eq_mul_inv]
114 
115 /-- Every telescoping confidence share is positive when its outer budget is. -/
116 theorem telescopingConfidenceShare_pos
117     {delta : Real} (hdelta : 0 < delta) (n : Nat) :
118     0 < telescopingConfidenceShare delta n := by
119   unfold telescopingConfidenceShare telescopingConfidenceWeight
120   positivity
121 
122 /-- The ENNReal masses of the telescoping confidence shares sum exactly to
123 the outer budget. -/
124 theorem tsum_ofReal_telescopingConfidenceShare
125     {delta : Real} (hdelta : 0 <= delta) :
126     ∑' n, ENNReal.ofReal (telescopingConfidenceShare delta n) =
127       ENNReal.ofReal delta := by
128   have hreal : HasSum (telescopingConfidenceShare delta) delta := by
129     simpa [telescopingConfidenceShare] using
130       hasSum_telescopingConfidenceWeight.mul_left delta
131   have hnonneg : forall n, 0 <= telescopingConfidenceShare delta n := by
132     intro n
133     exact mul_nonneg hdelta (telescopingConfidenceWeight_nonneg n)
134   have hnnreal :
135       HasSum
136           (fun n => (telescopingConfidenceShare delta n).toNNReal)
137           delta.toNNReal :=
138     hreal.toNNReal hnonneg
139   have hcoe :
140       HasSum
141           (fun n => ((telescopingConfidenceShare delta n).toNNReal : ENNReal))
142           (delta.toNNReal : ENNReal) :=
143     ENNReal.hasSum_coe.mpr hnnreal
144   have hofReal :
145       HasSum
146           (fun n => ENNReal.ofReal (telescopingConfidenceShare delta n))
147           (ENNReal.ofReal delta) := by
148     simpa only [ENNReal.ofReal_eq_coe_nnreal] using hcoe
149   exact hofReal.tsum_eq

151 end BanditRLProof.Concentration
