5 # Finite-index geometric all-time confidence
6 
7 This module composes two already-compiled outer-measure adapters: a finite
8 equal-share union at each time and a countable geometric schedule over time.
9 It is deliberately only a confidence-budget composition theorem. It does not
10 produce the per-index tails and is not a Ville/Doob, mixture, optional-stopping,
11 self-normalized, or general Freedman inequality.
12 -/

21 /-- A countable family of finite-index bad events is controlled by the sum of
22 its per-time budgets when every index receives an equal share. No event
23 measurability or probability-measure assumption is required. -/
24 theorem measure_iUnion_iUnion_fintype_le_tsum_of_uniform
25     {Omega : Type u} {Idx : Type v}
26     [MeasurableSpace Omega] [Fintype Idx] [Nonempty Idx]
27     (mu : Measure Omega)
28     (bad : Nat -> Idx -> Set Omega)
29     (deltaAt : Nat -> Real)
30     (hbad : forall n i,
31       mu (bad n i) <=
32         ENNReal.ofReal (deltaAt n / (Fintype.card Idx : Real))) :
33     mu (⋃ n, ⋃ i, bad n i) <=
34       ∑' n, ENNReal.ofReal (deltaAt n) := by
35   classical
36   calc
37     mu (⋃ n, ⋃ i, bad n i) <= ∑' n, mu (⋃ i, bad n i) :=
38       MeasureTheory.measure_iUnion_le _
39     _ <= ∑' n, ENNReal.ofReal (deltaAt n) := by
40       apply ENNReal.tsum_le_tsum
41       intro n
42       simpa using
43         (ProbabilityUnionBound.measure_biUnion_finset_le_of_uniform
44           mu (Finset.univ : Finset Idx) Finset.univ_nonempty
45             (deltaAt n) (bad n) (fun i _hi => hbad n i))

47 /-- Equal sharing across a nonempty finite index type at every time, followed
48 by the geometric schedule over time, gives one all-time outer confidence
49 budget. No event measurability or probability-measure assumption is required. -/
50 theorem measure_iUnion_iUnion_fintype_le_delta_of_geometricConfidenceShare
51     {Omega : Type u} {Idx : Type v}
52     [MeasurableSpace Omega] [Fintype Idx] [Nonempty Idx]
53     (mu : Measure Omega)
54     (bad : Nat -> Idx -> Set Omega)
55     (delta : Real) (hdelta : 0 <= delta)
56     (hbad : forall n i,
57       mu (bad n i) <= ENNReal.ofReal
58         (geometricConfidenceShare delta n / (Fintype.card Idx : Real))) :
59     mu (⋃ n, ⋃ i, bad n i) <= ENNReal.ofReal delta := by
60   calc
61     mu (⋃ n, ⋃ i, bad n i) <=
62         ∑' n, ENNReal.ofReal (geometricConfidenceShare delta n) :=
63       measure_iUnion_iUnion_fintype_le_tsum_of_uniform
64         mu bad (geometricConfidenceShare delta) hbad
65     _ = ENNReal.ofReal delta :=
66       tsum_ofReal_geometricConfidenceShare hdelta
