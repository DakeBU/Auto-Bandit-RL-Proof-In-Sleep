18 theorem regularizedPrefixFeatureGram_inv_quadratic_le_one
19     {Feature : Type u} [Fintype Feature] [DecidableEq Feature]
20     (lambda : Real) (hlambda : 0 < lambda)
21     (feature : Nat -> Feature -> Real) (T : Nat)
22     (x : Feature -> Real)
23     (hx : dotProduct x x <= lambda) :
24     dotProduct x
25         ((regularizedPrefixFeatureGram lambda feature T)⁻¹.mulVec x) <=
26       1 := by
27   let V := regularizedPrefixFeatureGram lambda feature T
28   let z := V⁻¹.mulVec x
29   have hV : V.PosDef :=
30     regularizedPrefixFeatureGram_posDef lambda hlambda feature T
31   have hcancel : V.mulVec z = x :=
32     posDef_mulVec_nonsingInv_mulVec V hV x
33   have hq_eq :
34       dotProduct x z = quadraticForm V z := by
35     rw [← hcancel, dotProduct_comm,
36       dotProduct_mulVec_eq_quadraticForm]
37   have hzz_nonneg : 0 <= dotProduct z z := by
38     rw [dotProduct]
39     exact Finset.sum_nonneg (fun _ _ => mul_self_nonneg _)
40   have hq_nonneg : 0 <= dotProduct x z := by
41     simpa [V, z] using
42       regularizedPrefixFeatureGram_inv_quadratic_nonneg
43         lambda hlambda feature T x
44   have hregularization :
45       lambda * dotProduct z z <= dotProduct x z := by
46     rw [hq_eq]
47     dsimp only [V]
48     rw [regularizedPrefixFeatureGram_quadraticForm_eq_sum_sq]
49     have hsum_nonneg :
50         0 <=
51           (Finset.range T).sum
52             (fun t =>
53               ((Finset.univ : Finset Feature).sum
54                 (fun i => feature t i * z i)) ^ 2) :=
55       Finset.sum_nonneg (fun _ _ => sq_nonneg _)
56     simpa [dotProduct, pow_two] using
57       (le_add_of_nonneg_right hsum_nonneg :
58         lambda * (Finset.univ : Finset Feature).sum (fun i => z i ^ 2) <=
59           lambda * (Finset.univ : Finset Feature).sum (fun i => z i ^ 2) +
60             (Finset.range T).sum
61               (fun t =>
62                 ((Finset.univ : Finset Feature).sum
63                   (fun i => feature t i * z i)) ^ 2))
64   have hcauchy :
65       dotProduct x z ^ 2 <= dotProduct x x * dotProduct z z := by
66     simpa [dotProduct, pow_two] using
67       (Finset.sum_mul_sq_le_sq_mul_sq
68         (Finset.univ : Finset Feature) x z)
69   have hsq_le : dotProduct x z ^ 2 <= dotProduct x z := by
70     calc
71       dotProduct x z ^ 2 <=
72           dotProduct x x * dotProduct z z := hcauchy
73       _ <= lambda * dotProduct z z :=
74         mul_le_mul_of_nonneg_right hx hzz_nonneg
75       _ <= dotProduct x z := hregularization
76   have hq_le_one : dotProduct x z <= 1 := by
77     nlinarith
78   simpa [V, z] using hq_le_one
