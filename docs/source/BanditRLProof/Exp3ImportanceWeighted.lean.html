7 # Finite-action importance-weighted loss identities
8 
9 This module supplies the finite-distribution algebra immediately above the
10 deterministic Hedge theorem and below the probabilistic EXP3 process.  It proves
11 the estimator's armwise weighted-sum cancellation and exact mixed-square identity on
12 an explicit finite action set.
13 
14 These statements are deterministic finite sums.  Calling them conditional
15 expectation facts requires a later law-transport theorem identifying the
16 conditional action law with the supplied probabilities.
17 -/
18 
19 namespace BanditRLProof
20 namespace Exp3

22 universe u

24 /-- Loss estimate that reveals only the loss of the sampled action. -/
25 noncomputable def importanceWeightedLoss {Action : Type u}
26     (prob loss : Action -> Real) (chosen action : Action) : Real := by
27   classical
28   exact if chosen = action then loss action / prob action else 0
29 
30 /-- The loss mixed by `prob` after replacing losses by one sampled estimate. -/
31 noncomputable def mixedImportanceWeightedLoss {Action : Type u}
32     (arms : Finset Action) (prob loss : Action -> Real) (chosen : Action) : Real :=
33   arms.sum (fun action =>
34     prob action * importanceWeightedLoss prob loss chosen action)
35 
36 /-- An importance-weighted estimate mixed by weights that may differ from the
37 sampling probabilities used in the estimator denominator. -/
38 noncomputable def weightedImportanceWeightedLoss {Action : Type u}
39     (arms : Finset Action) (prob weight loss : Action -> Real)
40     (chosen : Action) : Real :=
41   arms.sum (fun action =>
42     weight action * importanceWeightedLoss prob loss chosen action)
43 
44 /-- The mixed square of one sampled importance-weighted loss vector. -/
45 noncomputable def mixedSquaredImportanceWeightedLoss {Action : Type u}
46     (arms : Finset Action) (prob loss : Action -> Real) (chosen : Action) : Real :=
47   arms.sum (fun action =>
48     prob action * (importanceWeightedLoss prob loss chosen action) ^ 2)

50 theorem importanceWeightedLoss_nonneg {Action : Type u} [DecidableEq Action]
51     {prob loss : Action -> Real} {chosen action : Action}
52     (hprob : 0 <= prob action) (hloss : 0 <= loss action) :
53     0 <= importanceWeightedLoss prob loss chosen action := by
54   unfold importanceWeightedLoss
55   split_ifs
56   · exact div_nonneg hloss hprob
57   · exact le_refl 0
58 
59 /-- The probability-weighted finite sum of one coordinate recovers its loss. -/
60 theorem sum_prob_mul_importanceWeightedLoss_eq_loss {Action : Type u}
61     [DecidableEq Action]
62     (arms : Finset Action) (prob loss : Action -> Real)
63     (action : Action) (haction : action ∈ arms)
64     (hprob : prob action ≠ 0) :
65     arms.sum (fun chosen =>
66         prob chosen * importanceWeightedLoss prob loss chosen action) =
67       loss action := by
68   rw [Finset.sum_eq_single action]
69   · simp [importanceWeightedLoss]
70     rw [mul_comm]
71     exact div_mul_cancel₀ _ hprob
72   · intro chosen hchosen hne
73     simp [importanceWeightedLoss, hne]
74   · exact fun hnotmem => (hnotmem haction).elim
75 
76 /-- Pathwise cancellation: the mixed estimate is the sampled loss. -/
77 theorem mixedImportanceWeightedLoss_eq_selectedLoss {Action : Type u}
78     [DecidableEq Action]
79     (arms : Finset Action) (prob loss : Action -> Real)
80     (chosen : Action) (hchosen : chosen ∈ arms)
81     (hprob : prob chosen ≠ 0) :
82     mixedImportanceWeightedLoss arms prob loss chosen = loss chosen := by
83   unfold mixedImportanceWeightedLoss
84   rw [Finset.sum_eq_single chosen]
85   · simp [importanceWeightedLoss]
86     rw [mul_comm]
87     exact div_mul_cancel₀ _ hprob
88   · intro action haction hne
89     simp [importanceWeightedLoss, Ne.symm hne]
90   · exact fun hnotmem => (hnotmem hchosen).elim
91 
92 /-- Averaging the pathwise mixed estimate recovers the true mixed loss. -/
93 theorem sum_prob_mul_mixedImportanceWeightedLoss_eq_mixedLoss
94     {Action : Type u} [DecidableEq Action]
95     (arms : Finset Action) (prob loss : Action -> Real)
96     (hprob : forall action, action ∈ arms -> prob action ≠ 0) :
97     arms.sum (fun chosen =>
98         prob chosen * mixedImportanceWeightedLoss arms prob loss chosen) =
99       arms.sum (fun action => prob action * loss action) := by
100   apply Finset.sum_congr rfl
101   intro chosen hchosen
102   rw [mixedImportanceWeightedLoss_eq_selectedLoss arms prob loss chosen hchosen
103     (hprob chosen hchosen)]
104 
105 /-- Averaging an estimator mixed by arbitrary predictable weights recovers
106 the same weighted true loss. -/
107 theorem sum_prob_mul_weightedImportanceWeightedLoss_eq_weightedLoss
108     {Action : Type u} [DecidableEq Action]
109     (arms : Finset Action) (prob weight loss : Action -> Real)
110     (hprob : forall action, action ∈ arms -> prob action ≠ 0) :
111     arms.sum (fun chosen =>
112         prob chosen *
113           weightedImportanceWeightedLoss arms prob weight loss chosen) =
114       arms.sum (fun action => weight action * loss action) := by
115   unfold weightedImportanceWeightedLoss
116   simp_rw [Finset.mul_sum]
117   rw [Finset.sum_comm]
118   apply Finset.sum_congr rfl
119   intro action haction
120   calc
121     arms.sum (fun chosen =>
122         prob chosen *
123           (weight action * importanceWeightedLoss prob loss chosen action)) =
124         weight action * arms.sum (fun chosen =>
125           prob chosen * importanceWeightedLoss prob loss chosen action) := by
126       rw [Finset.mul_sum]
127       apply Finset.sum_congr rfl
128       intro chosen _hchosen
129       ring
130     _ = weight action * loss action := by
131       rw [sum_prob_mul_importanceWeightedLoss_eq_loss
132         arms prob loss action haction (hprob action haction)]

134 /-- Exact pathwise mixed-square formula for a sampled action. -/
135 theorem mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
136     {Action : Type u} [DecidableEq Action]
137     (arms : Finset Action) (prob loss : Action -> Real)
138     (chosen : Action) (hchosen : chosen ∈ arms)
139     (hprob : prob chosen ≠ 0) :
140     mixedSquaredImportanceWeightedLoss arms prob loss chosen =
141       (loss chosen) ^ 2 / prob chosen := by
142   unfold mixedSquaredImportanceWeightedLoss
143   rw [Finset.sum_eq_single chosen]
144   · simp [importanceWeightedLoss]
145     field_simp
146   · intro action haction hne
147     simp [importanceWeightedLoss, Ne.symm hne]
148   · exact fun hnotmem => (hnotmem hchosen).elim
149 
150 /-- Exact probability-weighted finite sum of the mixed estimator's square. -/
151 theorem sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
152     {Action : Type u} [DecidableEq Action]
153     (arms : Finset Action) (prob loss : Action -> Real)
154     (hprob : forall action, action ∈ arms -> prob action ≠ 0) :
155     arms.sum (fun chosen =>
156         prob chosen *
157           mixedSquaredImportanceWeightedLoss arms prob loss chosen) =
158       arms.sum (fun action => (loss action) ^ 2) := by
159   apply Finset.sum_congr rfl
160   intro chosen hchosen
161   rw [mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
162     arms prob loss chosen hchosen (hprob chosen hchosen)]
163   exact mul_div_cancel₀ _ (hprob chosen hchosen)
164 
165 /-- For losses in `[0,1]`, the probability-weighted mixed square is at most the arm count. -/
166 theorem sum_prob_mul_mixedSquaredImportanceWeightedLoss_le_card
167     {Action : Type u} [DecidableEq Action]
168     (arms : Finset Action) (prob loss : Action -> Real)
169     (hprob : forall action, action ∈ arms -> prob action ≠ 0)
170     (hloss : forall action, action ∈ arms ->
171       0 <= loss action ∧ loss action <= 1) :
172     arms.sum (fun chosen =>
173         prob chosen *
174           mixedSquaredImportanceWeightedLoss arms prob loss chosen) <=
175       arms.card := by
176   rw [sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
177     arms prob loss hprob]
178   calc
179     arms.sum (fun action => (loss action) ^ 2) <=
180         arms.sum (fun _action => (1 : Real)) := by
181       apply Finset.sum_le_sum
182       intro action haction
183       nlinarith [(hloss action haction).1, (hloss action haction).2]
184     _ = arms.card := by simp
