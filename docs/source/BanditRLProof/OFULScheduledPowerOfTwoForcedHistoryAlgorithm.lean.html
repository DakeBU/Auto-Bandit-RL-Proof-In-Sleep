5 # Power-of-two forced scheduling for telescoping OFUL
6 
7 This module packages one horizon-independent deterministic history algorithm.
8 At history indices one below a power of two it selects the arm indexed by that
9 power's exponent; at every other index it uses the ordinary telescoping OFUL
10 selector.
11 -/

22 /--
23 Use the exponent-indexed forced action at power-of-two successor indices and
24 the ordinary telescoping OFUL selector at all other history indices.
25 -/
26 noncomputable def finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAction
27     {K : Nat} {Feature : Type u}
28     [Fintype Feature] [DecidableEq Feature]
29     (hK : 0 < K)
30     (lambda : Real)
31     (actionFeature : Fin K -> Feature -> Real)
32     (R delta S : Real)
33     (forcedAction : Nat -> Fin K)
34     (n : Nat)
35     (history : History.FinitePairHistory (Fin K) Real n) :
36     Fin K :=
37   if isPowerOfTwoForcedIndex n then
38     forcedAction (Nat.log2 (n + 1))
39   else
40     finiteHistoryTelescopingScalarRidgeOptimisticAction
41       hK lambda actionFeature R delta S n history

109 /--
110 At history index `2 ^ exponent - 1`, the selector uses the action prescribed
111 for that exponent.
112 -/
113 @[simp]
114 theorem finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAction_pow_sub_one
115     {K : Nat} {Feature : Type u}
116     [Fintype Feature] [DecidableEq Feature]
117     (hK : 0 < K)
118     (lambda : Real)
119     (actionFeature : Fin K -> Feature -> Real)
120     (R delta S : Real)
121     (forcedAction : Nat -> Fin K)
122     (exponent : Nat)
123     (history :
124       History.FinitePairHistory (Fin K) Real (2 ^ exponent - 1)) :
125     finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAction
126         hK lambda actionFeature R delta S forcedAction
127         (2 ^ exponent - 1) history =
128       forcedAction exponent := by
129   have hpow : 0 < 2 ^ exponent := pow_pos (by omega) exponent
130   have hsucc : 2 ^ exponent - 1 + 1 = 2 ^ exponent := by omega
131   have hforced : isPowerOfTwoForcedIndex (2 ^ exponent - 1) :=
132     isPowerOfTwoForcedIndex_iff.mpr ⟨exponent, hsucc⟩
133   unfold finiteHistoryPowerOfTwoForcedTelescopingScalarRidgeAction
134   rw [if_pos hforced, hsucc, Nat.log2_eq_log_two,
135     Nat.log_pow Nat.one_lt_two]
