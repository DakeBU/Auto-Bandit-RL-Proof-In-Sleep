9 /-- A concrete history-adaptive corruption source. At successor time `n+1`,
10 the suboptimal-arm boost is active exactly when the action observed at time `n`
11 equals `triggerArm`. The best arm is never shifted. -/
12 noncomputable def previousActionGatedSuboptimalRewardBoostSource
13     {K : Nat} (model : FiniteBanditModel K) (triggerArm : Fin K)
14     (boost : Nat -> Fin K -> Real)
15     (hboost : forall t arm, 0 <= boost t arm) :
16     FiniteArmIIDHistoryAdaptiveRewardShiftSource K where
17   initial arm := if arm = model.bestArm then 0 else boost 0 arm
18   successor n history arm :=
19     if arm = model.bestArm then 0
20     else if (history ⟨n, Finset.mem_Iic.mpr le_rfl⟩).1 = triggerArm then
21       boost (n + 1) arm
22     else 0
23   measurable_successor n := by
24     let previousIndex : Finset.Iic n := ⟨n, Finset.mem_Iic.mpr le_rfl⟩
25     have hpreviousPair : Measurable
26         (fun input : History.FinitePairHistory (Fin K) Real n × Fin K =>
27           input.1 previousIndex) :=
28       (measurable_pi_apply previousIndex).comp measurable_fst
29     have hpreviousAction : Measurable
30         (fun input : History.FinitePairHistory (Fin K) Real n × Fin K =>
31           (input.1 previousIndex).1) :=
32       measurable_fst.comp hpreviousPair
33     have hbest : MeasurableSet
34         {input : History.FinitePairHistory (Fin K) Real n × Fin K |
35           input.2 = model.bestArm} :=
36       measurableSet_eq_fun measurable_snd measurable_const
37     have htrigger : MeasurableSet
38         {input : History.FinitePairHistory (Fin K) Real n × Fin K |
39           (input.1 previousIndex).1 = triggerArm} :=
40       measurableSet_eq_fun hpreviousAction measurable_const
41     have hscheduled : Measurable
42         (fun input : History.FinitePairHistory (Fin K) Real n × Fin K =>
43           boost (n + 1) input.2) :=
44       (measurable_of_countable (boost (n + 1))).comp measurable_snd
45     exact Measurable.ite hbest measurable_const
46       (Measurable.ite htrigger hscheduled measurable_const)
47   envelope t arm := if arm = model.bestArm then 0 else boost t arm
48   envelope_nonneg t arm := by
49     split_ifs
50     · exact le_rfl
51     · exact hboost t arm
52   initial_abs_le arm := by
53     split_ifs <;> simp [abs_of_nonneg (hboost 0 arm)]
54   successor_abs_le n _ arm := by
55     split_ifs
56     · simp
57     · simp [abs_of_nonneg (hboost (n + 1) arm)]
58     · simpa using hboost (n + 1) arm
