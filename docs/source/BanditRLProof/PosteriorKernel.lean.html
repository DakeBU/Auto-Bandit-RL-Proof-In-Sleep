5 # Posterior kernel surface
6 
7 This module records the narrow `POSTERIOR-KERNEL` leaf: a posterior over
8 environments, indexed by the observed history, is represented as a Mathlib
9 Markov kernel from histories to environments.  It deliberately does not prove a
10 Bayes formula, a regular-conditional-distribution existence theorem, Thompson
11 probability matching, or Bayesian regret.
12 -/
13

22 /--
23 A posterior distribution over environments indexed by observed histories.
24 
25 The underlying object is Mathlib's `ProbabilityTheory.Kernel`; the local
26 wrapper gives Thompson-sampling and Bayesian-regret leaves a stable project
27 name for the regularity contract.
28 -/
29 structure MarkovPosteriorKernel
30     (History : Type u) (Env : Type v)
31     [MeasurableSpace History] [MeasurableSpace Env] where
32   kernel : ProbabilityTheory.Kernel History Env
33   isMarkovKernel : ProbabilityTheory.IsMarkovKernel kernel
34 
35 variable {History : Type u} {Env : Type v}
36     [MeasurableSpace History] [MeasurableSpace Env]
37 
38 instance instIsMarkovKernel
39     (posterior : MarkovPosteriorKernel History Env) :
40     ProbabilityTheory.IsMarkovKernel posterior.kernel :=
41   posterior.isMarkovKernel

43 /-- Build the local posterior-kernel contract from an existing Mathlib kernel. -/
44 def ofKernel
45     (kernel : ProbabilityTheory.Kernel History Env)
46     (hkernel : ProbabilityTheory.IsMarkovKernel kernel) :
47     MarkovPosteriorKernel History Env where
48   kernel := kernel
49   isMarkovKernel := hkernel

51 /-- Build a posterior kernel from a measurable posterior-measure selector. -/
52 def ofMeasureSelector
53     (posterior : History -> Measure Env)
54     (hposterior : Measurable posterior)
55     (hprob : forall history, IsProbabilityMeasure (posterior history)) :
56     MarkovPosteriorKernel History Env where
57   kernel := ProbabilityTheory.Kernel.mk posterior hposterior
58   isMarkovKernel := ⟨hprob⟩
59 
60 @[simp]
61 theorem ofMeasureSelector_apply
62     (posterior : History -> Measure Env)
63     (hposterior : Measurable posterior)
64     (hprob : forall history, IsProbabilityMeasure (posterior history))
65     (history : History) :
66     (ofMeasureSelector posterior hposterior hprob).kernel history =
67       posterior history := rfl

69 /--
70 Build a posterior kernel on a countable/discrete history space.
71 
72 Finite histories in the bandit development are typically countable/discrete,
73 so Mathlib's `Kernel.ofFunOfCountable` can turn any probability-valued
74 posterior selector into a Markov kernel without a separate measurability proof.
75 -/
76 def ofCountableHistorySelector
77     [Countable History] [MeasurableSingletonClass History]
78     (posterior : History -> Measure Env)
79     (hprob : forall history, IsProbabilityMeasure (posterior history)) :
80     MarkovPosteriorKernel History Env where
81   kernel := ProbabilityTheory.Kernel.ofFunOfCountable posterior
82   isMarkovKernel := ⟨hprob⟩
83 
84 @[simp]
85 theorem ofCountableHistorySelector_apply
86     [Countable History] [MeasurableSingletonClass History]
87     (posterior : History -> Measure Env)
88     (hprob : forall history, IsProbabilityMeasure (posterior history))
89     (history : History) :
90     (ofCountableHistorySelector posterior hprob).kernel history =
91       posterior history := rfl

93 /-- The posterior kernel is measurable as a map from histories to measures. -/
94 theorem measurable_kernel
95     (posterior : MarkovPosteriorKernel History Env) :
96     Measurable posterior.kernel :=
97   posterior.kernel.measurable

99 /-- A measurable random history selects a measurable random posterior measure. -/
100 theorem measurable_apply_of_measurable_history
101     {Omega : Type u} [MeasurableSpace Omega]
102     (posterior : MarkovPosteriorKernel History Env)
103     (history : Omega -> History)
104     (hhistory : Measurable history) :
105     Measurable (fun omega : Omega => posterior.kernel (history omega)) :=
106   posterior.kernel.measurable.comp hhistory
107 
108 /--
109 For every measurable environment event, the posterior event probability is a
110 measurable scalar function of a measurable random history.
111 -/
112 theorem measurable_eventProbability_of_measurable_history
113     {Omega : Type u} [MeasurableSpace Omega]
114     (posterior : MarkovPosteriorKernel History Env)
115     (history : Omega -> History)
116     (hhistory : Measurable history)
117     {event : Set Env}
118     (hevent : MeasurableSet event) :
119     Measurable
120       (fun omega : Omega => posterior.kernel (history omega) event) :=
121   (ProbabilityTheory.Kernel.measurable_coe posterior.kernel hevent).comp
122     hhistory

124 /-- Every measure selected by a posterior kernel is a probability measure. -/
125 theorem isProbabilityMeasure_apply
126     (posterior : MarkovPosteriorKernel History Env)
127     (history : History) :
128     IsProbabilityMeasure (posterior.kernel history) := by
129   haveI : ProbabilityTheory.IsMarkovKernel posterior.kernel :=
130     posterior.isMarkovKernel
131   infer_instance

133 @[simp]
134 theorem apply_univ
135     (posterior : MarkovPosteriorKernel History Env)
136     (history : History) :
137     posterior.kernel history Set.univ = 1 := by
138   haveI : IsProbabilityMeasure (posterior.kernel history) :=
139     isProbabilityMeasure_apply posterior history

142 /-!
143 ## Canonical Bayesian posterior
144 
145 Mathlib's `ProbabilityTheory.posterior likelihood prior` is the posterior
146 kernel characterized by the swapped joint-law identity.  The declarations
147 below expose that object through the local posterior contract and identify it
148 with `condDistrib` on any source whose environment/history pair law is the
149 Bayesian composition product.
150 -/
151 
152 /-- The Mathlib posterior of a likelihood kernel under a prior measure. -/
153 noncomputable def canonicalPosterior
154     [StandardBorelSpace Env] [Nonempty Env]
155     (prior : Measure Env) [IsProbabilityMeasure prior]
156     (likelihood : ProbabilityTheory.Kernel Env History)
157     [ProbabilityTheory.IsMarkovKernel likelihood] :
158     MarkovPosteriorKernel History Env :=
159   ofKernel (ProbabilityTheory.posterior likelihood prior) inferInstance
160 
161 @[simp]
162 theorem canonicalPosterior_kernel
163     [StandardBorelSpace Env] [Nonempty Env]
164     (prior : Measure Env) [IsProbabilityMeasure prior]
165     (likelihood : ProbabilityTheory.Kernel Env History)
166     [ProbabilityTheory.IsMarkovKernel likelihood] :
167     (canonicalPosterior prior likelihood).kernel =
168       ProbabilityTheory.posterior likelihood prior := rfl
169 
170 /-- The canonical joint law generated by a prior followed by a likelihood. -/
171 noncomputable def canonicalJointMeasure
172     (prior : Measure Env) [IsProbabilityMeasure prior]
173     (likelihood : ProbabilityTheory.Kernel Env History)
174     [ProbabilityTheory.IsMarkovKernel likelihood] :
175     Measure (Env × History) :=
176   prior ⊗ₘ likelihood
177 
178 instance instCanonicalJointMeasureIsProbabilityMeasure
179     (prior : Measure Env) [IsProbabilityMeasure prior]
180     (likelihood : ProbabilityTheory.Kernel Env History)
181     [ProbabilityTheory.IsMarkovKernel likelihood] :
182     IsProbabilityMeasure (canonicalJointMeasure prior likelihood) := by
183   rw [canonicalJointMeasure]
184   infer_instance

186 /--
187 Identify the canonical posterior on any source with the prescribed Bayesian
188 environment/history pair law.
189 
190 The proof uses the defining composition-product identity for Mathlib's
191 `posterior`.  Thus the posterior conditional-law equality is produced from a
192 joint-law transport, rather than assumed as a separate Bayes-law field.
193 -/
194 theorem canonicalPosterior_kernel_ae_eq_condDistrib_of_pair_map_eq
195     {Omega : Type*} [MeasurableSpace Omega]
196     [StandardBorelSpace Env] [Nonempty Env]
197     (mu : Measure Omega) [IsFiniteMeasure mu]
198     (env : Omega -> Env) (history : Omega -> History)
199     (henv : Measurable env) (hhistory : Measurable history)
200     (prior : Measure Env) [IsProbabilityMeasure prior]
201     (likelihood : ProbabilityTheory.Kernel Env History)
202     [ProbabilityTheory.IsMarkovKernel likelihood]
203     (hpair :
204       mu.map (fun omega => (env omega, history omega)) =
205         canonicalJointMeasure prior likelihood) :
206     (canonicalPosterior prior likelihood).kernel =ᵐ[mu.map history]
207       ProbabilityTheory.condDistrib env history mu := by
208   have hpairMeas : Measurable (fun omega => (env omega, history omega)) :=
209     henv.prodMk hhistory
210   have hhistoryLaw :
211       mu.map history = likelihood ∘ₘ prior := by
212     calc
213       mu.map history =
214           (mu.map (fun omega => (env omega, history omega))).map Prod.snd := by
215             rw [Measure.map_map measurable_snd hpairMeas]
216             rfl
217       _ = (canonicalJointMeasure prior likelihood).map Prod.snd :=
218         congrArg (fun nu : Measure (Env × History) => nu.map Prod.snd) hpair
219       _ = likelihood ∘ₘ prior := by
220         change (prior ⊗ₘ likelihood).snd = likelihood ∘ₘ prior
221         rw [Measure.snd_compProd]
222   have hswappedJoint :
223       mu.map (fun omega => (history omega, env omega)) =
224         mu.map history ⊗ₘ (canonicalPosterior prior likelihood).kernel := by
225     calc
226       mu.map (fun omega => (history omega, env omega)) =
227           (mu.map (fun omega => (env omega, history omega))).map Prod.swap := by
228             rw [Measure.map_map measurable_swap hpairMeas]
229             rfl
230       _ = (canonicalJointMeasure prior likelihood).map Prod.swap :=
231         congrArg (fun nu : Measure (Env × History) => nu.map Prod.swap) hpair
232       _ = (likelihood ∘ₘ prior) ⊗ₘ
233           (canonicalPosterior prior likelihood).kernel := by
234         exact ProbabilityTheory.compProd_posterior_eq_map_swap.symm
235       _ = mu.map history ⊗ₘ
236           (canonicalPosterior prior likelihood).kernel := by
237         rw [hhistoryLaw]
238   exact ((ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
239     (μ := mu) history henv.aemeasurable
240     (canonicalPosterior prior likelihood).kernel).2 hswappedJoint).symm
241 
242 /--
243 On the canonical Bayesian product space itself, the Mathlib posterior is the
244 conditional distribution of the environment coordinate given the history
245 coordinate.
246 -/
247 theorem canonicalPosterior_kernel_ae_eq_condDistrib_fst_snd
248     [StandardBorelSpace Env] [Nonempty Env]
249     (prior : Measure Env) [IsProbabilityMeasure prior]
250     (likelihood : ProbabilityTheory.Kernel Env History)
251     [ProbabilityTheory.IsMarkovKernel likelihood] :
252     (canonicalPosterior prior likelihood).kernel =ᵐ[
253         (canonicalJointMeasure prior likelihood).map Prod.snd]
254       ProbabilityTheory.condDistrib Prod.fst Prod.snd
255         (canonicalJointMeasure prior likelihood) := by
256   apply canonicalPosterior_kernel_ae_eq_condDistrib_of_pair_map_eq
257     (canonicalJointMeasure prior likelihood) Prod.fst Prod.snd
258     measurable_fst measurable_snd prior likelihood
259   simp [canonicalJointMeasure]

261 /--
262 Minimal prior/likelihood/posterior package.
263 
264 This records the Bayesian objects that future Thompson-sampling leaves need to
265 name.  It does not assert that `posterior` satisfies Bayes' rule for `prior`
266 and `likelihood`.
267 -/
268 structure BayesianPosteriorSurface
269     (Env : Type u) (History : Type v)
270     [MeasurableSpace Env] [MeasurableSpace History] where
271   prior : Measure Env
272   likelihood : ProbabilityTheory.Kernel Env History
273   posterior : MarkovPosteriorKernel History Env
274   prior_isProbability : IsProbabilityMeasure prior
275   likelihood_isMarkovKernel : ProbabilityTheory.IsMarkovKernel likelihood
