    1 import Mathlib.Data.Nat.Log
2 import Mathlib.Data.Finset.Card
3 
4 namespace BanditRLProof
5 namespace OFUL
6 
7 /--
8 A horizon-independent forcing predicate for successor indices one below powers
9 of two. The `Nat.log2` equality makes the predicate decidable without a
10 classical search over exponents.
11 -/
12 def isPowerOfTwoForcedIndex (n : Nat) : Prop :=
13   2 ^ Nat.log2 (n + 1) = n + 1
14 
15 instance instDecidableIsPowerOfTwoForcedIndex (n : Nat) :
16     Decidable (isPowerOfTwoForcedIndex n) := by
17   unfold isPowerOfTwoForcedIndex
18   infer_instance
19 
20 /-- The computable predicate has the intended existential power-of-two semantics. -/
21 theorem isPowerOfTwoForcedIndex_iff {n : Nat} :
22     isPowerOfTwoForcedIndex n <-> exists k, n + 1 = 2 ^ k := by
23   constructor
24   · intro hn
25     exact ⟨Nat.log2 (n + 1), hn.symm⟩
26   · rintro ⟨k, hk⟩
27     unfold isPowerOfTwoForcedIndex
28     rw [hk, Nat.log2_eq_log_two, Nat.log_pow Nat.one_lt_two]
29 
30 /-- Power-of-two forced successor indices strictly below `horizon`. -/
31 def powerOfTwoForcedIndexSet (horizon : Nat) : Finset Nat :=
32   (Finset.range horizon).filter (fun n => isPowerOfTwoForcedIndex n)
33 
34 /-- Membership combines the prefix bound with the intended power-of-two equation. -/
35 theorem mem_powerOfTwoForcedIndexSet_iff {horizon n : Nat} :
36     n ∈ powerOfTwoForcedIndexSet horizon <->
37       n < horizon ∧ exists k, n + 1 = 2 ^ k := by
38   simp only [
39     powerOfTwoForcedIndexSet,
40     Finset.mem_filter,
41     Finset.mem_range,
42     isPowerOfTwoForcedIndex_iff]
43 
44 @[simp]
45 theorem powerOfTwoForcedIndexSet_zero :
46     powerOfTwoForcedIndexSet 0 = ∅ := by
47   ext n
48   simp [mem_powerOfTwoForcedIndexSet_iff]
49 
50 /-- Index zero is forced exactly in nonempty horizon prefixes. -/
51 theorem zero_mem_powerOfTwoForcedIndexSet_iff {horizon : Nat} :
52     0 ∈ powerOfTwoForcedIndexSet horizon <-> 0 < horizon := by
53   constructor
54   · intro h
55     exact (mem_powerOfTwoForcedIndexSet_iff.mp h).1
56   · intro h
57     exact mem_powerOfTwoForcedIndexSet_iff.mpr ⟨h, 0, by simp⟩
58 
59 /--
60 There are at most `Nat.log2 horizon + 1` power-of-two forced indices below a
61 horizon. Each member embeds into the image of the admissible exponent range.
62 -/
63 theorem card_powerOfTwoForcedIndexSet_le_log2_add_one (horizon : Nat) :
64     (powerOfTwoForcedIndexSet horizon).card <= Nat.log2 horizon + 1 := by
65   let exponentImage :=
66     (Finset.range (Nat.log2 horizon + 1)).image (fun k => 2 ^ k - 1)
67   have hsubset : powerOfTwoForcedIndexSet horizon ⊆ exponentImage := by
68     intro n hn
69     rcases mem_powerOfTwoForcedIndexSet_iff.mp hn with ⟨hn_lt, k, hk⟩
70     apply Finset.mem_image.mpr
71     refine ⟨k, Finset.mem_range.mpr ?_, ?_⟩
72     · apply Nat.lt_succ_iff.mpr
73       apply (Nat.le_log2 (Nat.ne_of_gt (Nat.zero_lt_of_lt hn_lt))).mpr
74       omega
75     · omega
76   calc
77     (powerOfTwoForcedIndexSet horizon).card <= exponentImage.card :=
78       Finset.card_le_card hsubset
79     _ <= (Finset.range (Nat.log2 horizon + 1)).card := Finset.card_image_le
80     _ = Nat.log2 horizon + 1 := Finset.card_range _
