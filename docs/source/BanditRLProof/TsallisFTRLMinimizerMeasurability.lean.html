6 # Measurability of the canonical half-Tsallis minimizer
7 
8 The project minimizer is chosen noncomputably on the ambient action space, but
9 the objective only sees the explicit finite arm set.  We restrict the selected
10 minimizer to that finite subtype and prove continuity in the restricted score
11 vector.  Compactness supplies cluster points and strict convexity identifies
12 every cluster point with the unique minimizer.
13 -/

22 /-- The canonical project minimizer, restricted to the explicit finite arm
23 subtype and parameterized by a score vector on that subtype. -/
24 noncomputable def restrictedHalfTsallisMinimizer
25     {Action : Type u} [DecidableEq Action]
26     (arms : Finset Action) (harms : arms.Nonempty)
27     (eta : Real) (score : ↥arms -> Real) : ↥arms -> Real :=
28   Finset.restrict arms
29     (halfTsallisMinimizer arms harms eta (extendFiniteWeights arms score))

31 theorem restrictedHalfTsallisMinimizer_mem_stdSimplex
32     {Action : Type u} [DecidableEq Action]
33     (arms : Finset Action) (harms : arms.Nonempty)
34     (eta : Real) (score : ↥arms -> Real) :
35     restrictedHalfTsallisMinimizer arms harms eta score ∈
36       stdSimplex Real ↥arms := by
37   exact restrict_mem_stdSimplex_of_finiteSimplex arms
38     (halfTsallisMinimizer_isRegularizedMinimizer arms harms eta
39       (extendFiniteWeights arms score)).1

41 theorem restrictedHalfTsallisMinimizer_isMinOn
42     {Action : Type u} [DecidableEq Action]
43     (arms : Finset Action) (harms : arms.Nonempty)
44     (eta : Real) (score : ↥arms -> Real) :
45     IsMinOn
46       (fun weights : ↥arms -> Real =>
47         FTRL.regularizedObjective arms.attach eta
48           (negEntropyRegularizer arms.attach (1 / 2 : Real)) score weights)
49       (stdSimplex Real ↥arms)
50       (restrictedHalfTsallisMinimizer arms harms eta score) := by
51   intro weights hweights
52   have hcandidate : FTRL.finiteSimplex arms
53       (extendFiniteWeights arms weights) :=
54     finiteSimplex_extendFiniteWeights arms hweights
55   have h := (halfTsallisMinimizer_isRegularizedMinimizer arms harms eta
56     (extendFiniteWeights arms score)).2
57       (extendFiniteWeights arms weights) hcandidate
58   rw [← regularizedObjective_half_restrict_eq arms eta
59       (extendFiniteWeights arms score)
60       (halfTsallisMinimizer arms harms eta (extendFiniteWeights arms score)),
61     regularizedObjective_half_extendFiniteWeights_eq arms eta
62       (extendFiniteWeights arms score) weights] at h
63   simpa [restrictedHalfTsallisMinimizer, extendFiniteWeights] using h

65 /-- The restricted half-Tsallis objective is jointly continuous in the finite
66 score vector and simplex vector. -/
67 theorem continuous_regularizedObjective_half_restricted_joint
68     {Action : Type u} [DecidableEq Action]
69     (arms : Finset Action) (eta : Real) :
70     Continuous (fun pair : (↥arms -> Real) × (↥arms -> Real) =>
71       FTRL.regularizedObjective arms.attach eta
72         (negEntropyRegularizer arms.attach (1 / 2 : Real)) pair.1 pair.2) := by
73   rw [Finset.attach_eq_univ]
74   simp_rw [regularizedObjective_half_eq, FTRL.linearLoss]
75   fun_prop

78 /-- The finite-coordinate canonical half-Tsallis minimizer is continuous in
79 the finite score vector. -/
80 theorem continuous_restrictedHalfTsallisMinimizer
81     {Action : Type u} [DecidableEq Action]
82     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real) :
83     Continuous (restrictedHalfTsallisMinimizer arms harms eta) := by
84   letI : Nonempty ↥arms := ⟨⟨harms.choose, harms.choose_spec⟩⟩
85   rw [continuous_iff_continuousAt]
86   intro score
87   let objective : (↥arms -> Real) -> (↥arms -> Real) -> Real :=
88     fun parameter weights =>
89       FTRL.regularizedObjective arms.attach eta
90         (negEntropyRegularizer arms.attach (1 / 2 : Real)) parameter weights
91   apply (isCompact_stdSimplex Real ↥arms).tendsto_nhds_of_unique_mapClusterPt
92   · exact Filter.Eventually.of_forall fun parameter =>
93       restrictedHalfTsallisMinimizer_mem_stdSimplex
94         arms harms eta parameter
95   · intro weights hweights hcluster
96     have hweightsMin : IsMinOn (objective score)
97         (stdSimplex Real ↥arms) weights := by
98       rw [mapClusterPt_iff_ultrafilter] at hcluster
99       obtain ⟨ultra, hultraScore, hultraWeights⟩ := hcluster
100       have hscore : Tendsto id (↑ultra) (nhds score) := by
101         simpa only [Filter.map_id] using hultraScore
102       have hpair : Tendsto
103           (fun parameter =>
104             (parameter,
105               restrictedHalfTsallisMinimizer arms harms eta parameter))
106           (↑ultra)
107           (nhds (score, weights)) :=
108         hscore.prodMk_nhds hultraWeights
109       intro candidate hcandidate
110       have hleft : Tendsto
111           (fun parameter => objective parameter
112             (restrictedHalfTsallisMinimizer arms harms eta parameter))
113           (↑ultra) (nhds (objective score weights)) := by
114         exact (continuous_regularizedObjective_half_restricted_joint
115           arms eta).continuousAt.tendsto.comp hpair
116       have hright : Tendsto
117           (fun parameter => objective parameter candidate)
118           (↑ultra) (nhds (objective score candidate)) := by
119         have hcontinuous : Continuous
120             (fun parameter => objective parameter candidate) := by
121           exact (continuous_regularizedObjective_half_restricted_joint
122             arms eta).comp (continuous_id.prodMk continuous_const)
123         exact hcontinuous.continuousAt.tendsto.comp hscore
124       exact le_of_tendsto_of_tendsto' hleft hright fun parameter =>
125         restrictedHalfTsallisMinimizer_isMinOn
126           arms harms eta parameter hcandidate
127     have hstrict : StrictConvexOn Real (stdSimplex Real ↥arms)
128         (objective score) := by
129       dsimp [objective]
130       rw [Finset.attach_eq_univ]
131       exact strictConvexOn_regularizedObjective_half_stdSimplex
132         (Action := ↥arms) eta score
133     exact hstrict.eq_of_isMinOn hweightsMin
134       (restrictedHalfTsallisMinimizer_isMinOn arms harms eta score)
135       hweights
136       (restrictedHalfTsallisMinimizer_mem_stdSimplex arms harms eta score)

138 /-- Changing score coordinates outside the explicit arm set does not change
139 the canonical half-Tsallis minimizer on supported coordinates. -/
140 theorem halfTsallisMinimizer_eq_on_arms_of_score_eq
141     {Action : Type u} [DecidableEq Action]
142     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
143     (score₁ score₂ : Action -> Real)
144     (hscore : forall action, action ∈ arms ->
145       score₁ action = score₂ action) :
146     forall action, action ∈ arms ->
147       halfTsallisMinimizer arms harms eta score₁ action =
148         halfTsallisMinimizer arms harms eta score₂ action := by
149   have hobjective : forall weights : Action -> Real,
150       FTRL.regularizedObjective arms eta
151           (negEntropyRegularizer arms (1 / 2 : Real)) score₁ weights =
152         FTRL.regularizedObjective arms eta
153           (negEntropyRegularizer arms (1 / 2 : Real)) score₂ weights := by
154     intro weights
155     unfold FTRL.regularizedObjective FTRL.linearLoss
156     congr 2
157     apply Finset.sum_congr rfl
158     intro action haction
159     rw [hscore action haction]
160   have hsecond : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
161       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score₁
162       (halfTsallisMinimizer arms harms eta score₂) := by
163     refine ⟨(halfTsallisMinimizer_isRegularizedMinimizer
164       arms harms eta score₂).1, ?_⟩
165     intro weights hweights
166     rw [hobjective, hobjective]
167     exact (halfTsallisMinimizer_isRegularizedMinimizer
168       arms harms eta score₂).2 weights hweights
169   exact isRegularizedMinimizer_half_eq_on_arms arms harms eta score₁
170     (halfTsallisMinimizer arms harms eta score₁)
171     (halfTsallisMinimizer arms harms eta score₂)
172     (halfTsallisMinimizer_isRegularizedMinimizer arms harms eta score₁)
173     hsecond

175 theorem restrictedHalfTsallisMinimizer_restrict_score_apply
176     {Action : Type u} [DecidableEq Action]
177     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
178     (score : Action -> Real) (action : ↥arms) :
179     restrictedHalfTsallisMinimizer arms harms eta
180         (Finset.restrict arms score) action =
181       halfTsallisMinimizer arms harms eta score action := by
182   unfold restrictedHalfTsallisMinimizer
183   exact halfTsallisMinimizer_eq_on_arms_of_score_eq
184     arms harms eta (extendFiniteWeights arms (Finset.restrict arms score)) score
185     (fun candidate hcandidate => by
186       simp [extendFiniteWeights, Finset.restrict, hcandidate])
187     action action.2

189 /-- Coordinatewise measurability of supported scores implies coordinatewise
190 measurability of the existing canonical project minimizer. -/
191 theorem measurable_halfTsallisMinimizer_comp
192     {Omega : Type*} {Action : Type u}
193     [MeasurableSpace Omega] [DecidableEq Action]
194     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
195     (score : Omega -> Action -> Real)
196     (hscore : forall action, action ∈ arms ->
197       Measurable (fun omega => score omega action))
198     (action : Action) (haction : action ∈ arms) :
199     Measurable (fun omega =>
200       halfTsallisMinimizer arms harms eta (score omega) action) := by
201   let restrictedScore : Omega -> ↥arms -> Real := fun omega candidate =>
202     score omega candidate
203   have hrestrictedScore : Measurable restrictedScore :=
204     measurable_pi_lambda _ fun candidate => hscore candidate candidate.2
205   have hrestrictedSelector : Measurable (fun omega =>
206       restrictedHalfTsallisMinimizer arms harms eta
207         (restrictedScore omega)) :=
208     (continuous_restrictedHalfTsallisMinimizer arms harms eta).measurable.comp
209       hrestrictedScore
210   have hcoordinate : Measurable (fun omega =>
211       restrictedHalfTsallisMinimizer arms harms eta
212         (restrictedScore omega) ⟨action, haction⟩) :=
213     (measurable_pi_apply
214       (X := fun _candidate : ↥arms => Real) ⟨action, haction⟩).comp
215         hrestrictedSelector
216   rw [← show (fun omega =>
217       restrictedHalfTsallisMinimizer arms harms eta
218         (restrictedScore omega) ⟨action, haction⟩) =
219       (fun omega =>
220         halfTsallisMinimizer arms harms eta (score omega) action) by
221     funext omega
222     exact restrictedHalfTsallisMinimizer_restrict_score_apply
223       arms harms eta (score omega) ⟨action, haction⟩]
224   exact hcoordinate
