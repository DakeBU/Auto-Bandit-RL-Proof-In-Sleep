6 # Refined half-Tsallis stability to suboptimal-arm budgets
7 
8 The paper's refined stability term contains
9 `sum_a sqrt (p a) * (1 - p a)`.  This module proves the finite-simplex
10 elimination of one optimal arm and connects that bound to the compiled
11 self-bounding completion-of-squares consumer.
12 
13 Producing the refined all-arm stability bound from the generated algorithm is
14 separate: it requires the conjugate-potential stability estimate and the
15 time-varying penalty route, rather than the existing fixed-learning-rate
16 `<p_t - p_(t+1), hatLoss_t>` upper bound.
17 -/

24 theorem probability_le_sqrt
25     (probability : Real) (hprobability : 0 <= probability)
26     (hprobability_le_one : probability <= 1) :
27     probability <= Real.sqrt probability := by
28   apply (Real.le_sqrt hprobability hprobability).2
29   nlinarith
30 
31 theorem sqrt_mul_one_sub_le_sqrt
32     (probability : Real) (hprobability : 0 <= probability) :
33     Real.sqrt probability * (1 - probability) <=
34       Real.sqrt probability := by
35   have hsqrtNonneg : 0 <= Real.sqrt probability := Real.sqrt_nonneg _
36   nlinarith [mul_nonneg hsqrtNonneg hprobability]
37 
38 theorem sqrt_mul_one_sub_le_one_sub
39     (probability : Real) (hprobability_le_one : probability <= 1) :
40     Real.sqrt probability * (1 - probability) <= 1 - probability := by
41   have hsqrtLeOne : Real.sqrt probability <= 1 :=
42     (Real.sqrt_le_one).2 hprobability_le_one
43   simpa only [one_mul] using
44     mul_le_mul_of_nonneg_right hsqrtLeOne
45       (sub_nonneg.mpr hprobability_le_one)

47 theorem one_sub_eq_sum_erase
48     {Action : Type u} [DecidableEq Action]
49     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
50     (probability : Action -> Real)
51     (hprobability : FTRL.finiteSimplex arms probability) :
52     1 - probability best =
53       (arms.erase best).sum probability := by
54   have hsplit := Finset.add_sum_erase arms probability hbest
55   rw [hprobability.2] at hsplit
56   linarith

58 /-- Eliminate one distinguished arm from the paper's refined half-power
59 stability budget. -/
60 theorem sum_sqrt_mul_one_sub_le_two_mul_sum_erase_sqrt
61     {Action : Type u} [DecidableEq Action]
62     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
63     (probability : Action -> Real)
64     (hprobability : FTRL.finiteSimplex arms probability) :
65     arms.sum (fun action =>
66         Real.sqrt (probability action) * (1 - probability action)) <=
67       2 * (arms.erase best).sum (fun action =>
68         Real.sqrt (probability action)) := by
69   have hbestUpper :
70       Real.sqrt (probability best) * (1 - probability best) <=
71         (arms.erase best).sum probability := by
72     rw [← one_sub_eq_sum_erase arms hbest probability hprobability]
73     exact sqrt_mul_one_sub_le_one_sub
74       (probability best) (finiteSimplex_apply_le_one hprobability hbest)
75   have heraseUpper :
76       (arms.erase best).sum (fun action =>
77           Real.sqrt (probability action) * (1 - probability action)) <=
78         (arms.erase best).sum (fun action => Real.sqrt (probability action)) := by
79     apply Finset.sum_le_sum
80     intro action haction
81     exact sqrt_mul_one_sub_le_sqrt
82       (probability action)
83       (hprobability.1 action (Finset.mem_of_mem_erase haction))
84   have hprobToSqrt :
85       (arms.erase best).sum probability <=
86         (arms.erase best).sum (fun action => Real.sqrt (probability action)) := by
87     apply Finset.sum_le_sum
88     intro action haction
89     have hairm : action ∈ arms := Finset.mem_of_mem_erase haction
90     exact probability_le_sqrt
91       (probability action) (hprobability.1 action hairm)
92       (finiteSimplex_apply_le_one hprobability hairm)
93   rw [← Finset.add_sum_erase arms
94     (fun action => Real.sqrt (probability action) *
95       (1 - probability action)) hbest]
96   linarith

98 /-- A finite time-by-suboptimal-arm consumer for the paper-shaped refined
99 stability budget.  The remaining algorithmic obligation is exactly `hupper`. -/
100 theorem regret_le_of_refinedHalfPowerSelfBounding
101     {Time : Type u} {Action : Type v}
102     [DecidableEq Time] [DecidableEq Action]
103     (times : Finset Time) (arms : Finset Action)
104     {best : Action} (hbest : best ∈ arms)
105     (probability : Time -> Action -> Real)
106     (coefficient : Time -> Real) (gap : Action -> Real)
107     (regret base corruption : Real)
108     (hprobability : ∀ time ∈ times,
109       FTRL.finiteSimplex arms (probability time))
110     (hcoefficient : ∀ time ∈ times, 0 <= coefficient time)
111     (hgap : ∀ action ∈ arms.erase best, 0 < gap action)
112     (hselfBounding :
113       times.sum (fun time => (arms.erase best).sum (fun action =>
114           gap action * probability time action)) - corruption <= regret)
115     (hupper :
116       regret <= base + times.sum (fun time =>
117         coefficient time * arms.sum (fun action =>
118           Real.sqrt (probability time action) *
119             (1 - probability time action)))) :
120     regret <=
121       2 * base +
122         (times.product (arms.erase best)).sum (fun index =>
123           (2 * coefficient index.1) ^ 2 / gap index.2) +
124         corruption := by
125   let indices := times.product (arms.erase best)
126   have hprobabilityIndex : ∀ index ∈ indices,
127       0 <= probability index.1 index.2 := by
128     intro index hindex
129     have htime : index.1 ∈ times := (Finset.mem_product.mp hindex).1
130     have hactionErase : index.2 ∈ arms.erase best :=
131       (Finset.mem_product.mp hindex).2
132     exact (hprobability index.1 htime).1 index.2
133       (Finset.mem_of_mem_erase hactionErase)
134   have hgapIndex : ∀ index ∈ indices, 0 < gap index.2 := by
135     intro index hindex
136     exact hgap index.2 (Finset.mem_product.mp hindex).2
137   have hselfBoundingIndex :
138       indices.sum (fun index =>
139           gap index.2 * probability index.1 index.2) - corruption <= regret := by
140     simpa [indices, Finset.sum_product] using hselfBounding
141   have hrefined :
142       times.sum (fun time =>
143           coefficient time * arms.sum (fun action =>
144             Real.sqrt (probability time action) *
145               (1 - probability time action))) <=
146         times.sum (fun time =>
147           2 * coefficient time * (arms.erase best).sum (fun action =>
148             Real.sqrt (probability time action))) := by
149     apply Finset.sum_le_sum
150     intro time htime
151     have htimeBound :=
152       sum_sqrt_mul_one_sub_le_two_mul_sum_erase_sqrt
153         arms hbest (probability time) (hprobability time htime)
154     simpa [mul_assoc, mul_comm, mul_left_comm] using
155       mul_le_mul_of_nonneg_left htimeBound (hcoefficient time htime)
156   have hupperIndex :
157       regret <= base + indices.sum (fun index =>
158         (2 * coefficient index.1) *
159           Real.sqrt (probability index.1 index.2)) := by
160     calc
161       regret <= base + times.sum (fun time =>
162           coefficient time * arms.sum (fun action =>
163             Real.sqrt (probability time action) *
164               (1 - probability time action))) := hupper
165       _ <= base + times.sum (fun time =>
166           2 * coefficient time * (arms.erase best).sum (fun action =>
167             Real.sqrt (probability time action))) :=
168         by simpa [add_comm] using add_le_add_left hrefined base
169       _ = base + indices.sum (fun index =>
170           (2 * coefficient index.1) *
171             Real.sqrt (probability index.1 index.2)) := by
172         simp [indices, Finset.sum_product, Finset.mul_sum, mul_assoc]
173   exact regret_le_two_mul_base_add_sum_sq_div_gap_add_corruption
174     indices
175     (fun index => probability index.1 index.2)
176     (fun index => 2 * coefficient index.1)
177     (fun index => gap index.2)
178     regret base corruption hprobabilityIndex hgapIndex
179       hselfBoundingIndex hupperIndex
