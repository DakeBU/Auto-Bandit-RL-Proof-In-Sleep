4 /-!
5 # Scheduled canonical all-horizon OFUL cumulative gap
6 
7 This module combines the one-policy all-time scheduled confidence event with a

22 theorem allTimeTelescopingDelta_antitone
23     {delta : Real} (hdelta : 0 <= delta)
24     {n T : Nat} (hnT : n <= T) :
25     allTimeTelescopingDelta delta T <=
26       allTimeTelescopingDelta delta n := by
27   unfold allTimeTelescopingDelta allTimeTelescopingWeight
28   apply mul_le_mul_of_nonneg_left _ hdelta
29   apply one_div_le_one_div_of_le (by positivity)
30   have hfirst :
31       (((n + 1 : Nat) : Real)) <= (((T + 1 : Nat) : Real)) := by
32     exact_mod_cast Nat.add_le_add_right hnT 1
33   have hsecond :
34       (((n + 2 : Nat) : Real)) <= (((T + 2 : Nat) : Real)) := by
35     exact_mod_cast Nat.add_le_add_right hnT 2
36   exact mul_le_mul hfirst hsecond (by positivity) (by positivity)
