4 # Finite-horizon OFUL least-squares confidence ellipsoid
5 
6 This module consumes the common-`R` vector self-normalized tail. It defines
7 the finite-horizon ridge estimator from scalar observations, proves its exact
8 error decomposition into the martingale score and regularization bias, and
9 transports the score tail to a parameter confidence ellipsoid.
10 -/

19 /-- Quadratic-form square root; it is a norm when the matrix is positive definite. -/
20 noncomputable def matrixNorm
21     [Fintype Feature]
22     (A : Matrix Feature Feature Real) (x : Feature -> Real) : Real :=
23   Real.sqrt (x ⬝ᵥ A.mulVec x)

25 /-- Triangle inequality for the norm induced by a positive-definite matrix. -/
26 theorem matrixNorm_add_le
27     [Fintype Feature] [DecidableEq Feature]
28     (A : Matrix Feature Feature Real) (hA : A.PosDef)
29     (x y : Feature -> Real) :
30     matrixNorm A (x + y) <= matrixNorm A x + matrixNorm A y := by
31   have hnorm (z : Feature -> Real) :
32       matrixNorm A z =
33         @norm (Feature -> Real) (A.toNormedAddCommGroup hA).toNorm z := by
34     change Real.sqrt (z ⬝ᵥ A.mulVec z) =
35       Real.sqrt ((A.mulVec z) ⬝ᵥ z)
36     rw [dotProduct_comm]
37   rw [hnorm, hnorm, hnorm]
38   exact
39     @norm_add_le (Feature -> Real)
40       (A.toNormedAddCommGroup hA).toSeminormedAddCommGroup.toSeminormedAddGroup x y
41 
42 /-- Triangle inequality for subtraction in a positive-definite matrix norm. -/
43 theorem matrixNorm_sub_le
44     [Fintype Feature] [DecidableEq Feature]
45     (A : Matrix Feature Feature Real) (hA : A.PosDef)
46     (x y : Feature -> Real) :
47     matrixNorm A (x - y) <= matrixNorm A x + matrixNorm A y := by
48   have hnorm (z : Feature -> Real) :
49       matrixNorm A z =
50         @norm (Feature -> Real) (A.toNormedAddCommGroup hA).toNorm z := by
51     change Real.sqrt (z ⬝ᵥ A.mulVec z) =
52       Real.sqrt ((A.mulVec z) ⬝ᵥ z)
53     rw [dotProduct_comm]
54   rw [hnorm, hnorm, hnorm]
55   exact
56     @norm_sub_le (Feature -> Real)
57       (A.toNormedAddCommGroup hA).toSeminormedAddCommGroup.toSeminormedAddGroup x y

59 /-- Squaring the positive-definite matrix norm recovers its quadratic form. -/
60 theorem matrixNorm_sq
61     [Fintype Feature] [DecidableEq Feature]
62     (A : Matrix Feature Feature Real) (hA : A.PosDef)
63     (x : Feature -> Real) :
64     matrixNorm A x ^ 2 = x ⬝ᵥ A.mulVec x := by
65   exact Real.sq_sqrt (hA.posSemidef.dotProduct_mulVec_nonneg x)

67 /-- The finite-horizon sufficient statistic `sum_{i<n} x_i y_i`. -/
68 noncomputable def finiteHorizonResponseVector
69     [Fintype Feature]
70     (feature : Nat -> Omega -> Feature -> Real)
71     (response : Nat -> Omega -> Real)
72     (n : Nat) (omega : Omega) : Feature -> Real :=
73   fun j =>
74     (Finset.range n).sum (fun i => feature i omega j * response i omega)
75 
76 /-- Ridge least-squares estimate with deterministic positive-definite base. -/
77 noncomputable def finiteHorizonRidgeEstimate
78     [Fintype Feature] [DecidableEq Feature]
79     (V0 : Matrix Feature Feature Real)
80     (feature : Nat -> Omega -> Feature -> Real)
81     (response : Nat -> Omega -> Real)
82     (n : Nat) (omega : Omega) : Feature -> Real :=
83   (V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
84     (finiteHorizonResponseVector feature response n omega)

86 /-- The squared self-normalized radius from the common-`R` Markov tail. -/
87 noncomputable def finiteHorizonConfidenceThreshold
88     [Fintype Feature] [DecidableEq Feature]
89     (V0 : Matrix Feature Feature Real)
90     (feature : Nat -> Omega -> Feature -> Real)
91     (R delta : Real) (n : Nat) (omega : Omega) : Real :=
92   2 * R ^ 2 *
93     Real.log
94       (Real.sqrt
95           (Matrix.det (V0 + finiteHorizonFeatureGram feature n omega) /
96             Matrix.det V0) /
97         delta)

99 /-- Confidence radius with an explicit deterministic regularization-bias cap. -/
100 noncomputable def finiteHorizonConfidenceRadius
101     [Fintype Feature] [DecidableEq Feature]
102     (V0 : Matrix Feature Feature Real)
103     (feature : Nat -> Omega -> Feature -> Real)
104     (R delta biasRadius : Real) (n : Nat) (omega : Omega) : Real :=
105   Real.sqrt
106       (finiteHorizonConfidenceThreshold V0 feature R delta n omega) +
107     biasRadius

109 /--
110 Adding a positive-semidefinite Gram to a positive-definite base cannot reduce
111 the determinant ratio below one.
112 -/
113 theorem one_le_det_add_posSemidef_div_det
114     [Fintype Feature] [DecidableEq Feature]
115     (V0 G : Matrix Feature Feature Real)
116     (hV0 : V0.PosDef) (hG : G.PosSemidef) :
117     1 <= Matrix.det (V0 + G) / Matrix.det V0 := by
118   let C := CFC.sqrt V0⁻¹
119   have hC : C.PosSemidef := by
120     exact (CFC.sqrt_nonneg V0⁻¹).posSemidef
121   have hCGC : (C * G * C).PosSemidef := by
122     have hcongr := hG.conjTranspose_mul_mul_same C
123     simpa only [hC.isHermitian.eq] using hcongr
124   rw [← det_one_add_sqrt_inv_congruence_eq_ratio V0 G hV0,
125     det_one_add_posSemidef_eq_prod_eigenvalues (C * G * C) hCGC]
126   exact Finset.one_le_prod (fun i hi => by
127     linarith [hCGC.eigenvalues_nonneg i])

129 /-- The squared confidence threshold is nonnegative for `0 < delta <= 1`. -/
130 theorem finiteHorizonConfidenceThreshold_nonneg
131     [Fintype Feature] [DecidableEq Feature]
132     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
133     (feature : Nat -> Omega -> Feature -> Real)
134     (R delta : Real) (n : Nat) (omega : Omega)
135     (hdelta : 0 < delta) (hdelta_one : delta <= 1) :
136     0 <= finiteHorizonConfidenceThreshold V0 feature R delta n omega := by
137   have hratio :
138       1 <= Matrix.det
139           (V0 + finiteHorizonFeatureGram feature n omega) /
140         Matrix.det V0 :=
141     one_le_det_add_posSemidef_div_det V0
142       (finiteHorizonFeatureGram feature n omega) hV0
143       (finiteHorizonFeatureGram_posSemidef feature n omega)
144   have hsqrt :
145       1 <= Real.sqrt
146         (Matrix.det (V0 + finiteHorizonFeatureGram feature n omega) /
147           Matrix.det V0) := Real.one_le_sqrt.mpr hratio
148   have harg :
149       1 <=
150         Real.sqrt
151             (Matrix.det (V0 + finiteHorizonFeatureGram feature n omega) /
152               Matrix.det V0) /
153           delta :=
154     (one_le_div hdelta).2 (hdelta_one.trans hsqrt)
155   exact mul_nonneg (mul_nonneg (by norm_num) (sq_nonneg R))
156     (Real.log_nonneg harg)

158 /--
159 Under the linear observation model, the response sufficient statistic is the
160 feature Gram applied to the true parameter plus the martingale-noise score.
161 -/
162 theorem finiteHorizonResponseVector_eq_featureGram_mulVec_add_noiseScore
163     [Fintype Feature]
164     (thetaStar : Feature -> Real)
165     (feature : Nat -> Omega -> Feature -> Real)
166     (response noise : Nat -> Omega -> Real)
167     (n : Nat) (omega : Omega)
168     (hresponse : forall i, i < n ->
169       response i omega =
170         dotProduct thetaStar (feature i omega) + noise i omega) :
171     finiteHorizonResponseVector feature response n omega =
172       (finiteHorizonFeatureGram feature n omega).mulVec thetaStar +
173         WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega) := by
174   funext j
175   have hmodel :
176       (Finset.range n).sum (fun i =>
177           feature i omega j * response i omega) =
178         (Finset.range n).sum (fun i =>
179           feature i omega j *
180             (dotProduct thetaStar (feature i omega) + noise i omega)) := by
181     apply Finset.sum_congr rfl
182     intro i hi
183     rw [hresponse i (Finset.mem_range.mp hi)]
184   rw [finiteHorizonResponseVector, hmodel]
185   simp only [mul_add, Finset.sum_add_distrib, Pi.add_apply]
186   congr 1
187   simp only [finiteHorizonFeatureGram, prefixFeatureGram, Matrix.mulVec, dotProduct]
188   simp_rw [Finset.mul_sum, Finset.sum_mul]
189   rw [Finset.sum_comm]
190   apply Finset.sum_congr rfl
191   intro i hi
192   apply Finset.sum_congr rfl
193   intro k hk
194   ring

196 /-- Applying the nonsingular inverse of a positive-definite matrix cancels it. -/
197 theorem posDef_nonsingInv_mulVec_mulVec
198     [Fintype Feature] [DecidableEq Feature]
199     (A : Matrix Feature Feature Real) (hA : A.PosDef)
200     (x : Feature -> Real) :
201     A⁻¹.mulVec (A.mulVec x) = x := by
202   rw [Matrix.mulVec_mulVec,
203     Matrix.nonsing_inv_mul A (A.isUnit_iff_isUnit_det.mp hA.isUnit),
204     Matrix.one_mulVec]
205 
206 /-- A positive-definite matrix also cancels its inverse on the left. -/
207 theorem posDef_mulVec_nonsingInv_mulVec
208     [Fintype Feature] [DecidableEq Feature]
209     (A : Matrix Feature Feature Real) (hA : A.PosDef)
210     (x : Feature -> Real) :
211     A.mulVec (A⁻¹.mulVec x) = x := by
212   rw [Matrix.mulVec_mulVec,
213     Matrix.mul_nonsing_inv A (A.isUnit_iff_isUnit_det.mp hA.isUnit),
214     Matrix.one_mulVec]

216 /--
217 The squared `A`-norm of `A⁻¹x` is the inverse quadratic form of `x`.
218 -/
219 theorem matrixNorm_nonsingInv_mulVec_sq
220     [Fintype Feature] [DecidableEq Feature]
221     (A : Matrix Feature Feature Real) (hA : A.PosDef)

223     matrixNorm A (A⁻¹.mulVec x) ^ 2 = x ⬝ᵥ A⁻¹.mulVec x := by
224   rw [matrixNorm_sq A hA,

228 /--
229 Exact ridge-estimation error decomposition into inverse-Gram noise score and
230 inverse-Gram regularization bias.
231 -/
232 theorem finiteHorizonRidgeEstimate_sub_eq_inverseScore_sub_inverseBias
233     [Fintype Feature] [DecidableEq Feature]
234     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
235     (thetaStar : Feature -> Real)
236     (feature : Nat -> Omega -> Feature -> Real)
237     (response noise : Nat -> Omega -> Real)
238     (n : Nat) (omega : Omega)
239     (hresponse : forall i, i < n ->
240       response i omega =
241         dotProduct thetaStar (feature i omega) + noise i omega) :
242     finiteHorizonRidgeEstimate V0 feature response n omega - thetaStar =
243       (V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
244           (WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
245         (V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
246           (V0.mulVec thetaStar) := by
247   let G := finiteHorizonFeatureGram feature n omega
248   let V := V0 + G
249   have hG : G.PosSemidef :=
250     finiteHorizonFeatureGram_posSemidef feature n omega
251   have hV : V.PosDef := hV0.add_posSemidef hG
252   have htheta : V⁻¹.mulVec (V.mulVec thetaStar) = thetaStar :=
253     posDef_nonsingInv_mulVec_mulVec V hV thetaStar
254   rw [finiteHorizonRidgeEstimate,
255     finiteHorizonResponseVector_eq_featureGram_mulVec_add_noiseScore
256       thetaStar feature response noise n omega hresponse]
257   change V⁻¹.mulVec
258       (G.mulVec thetaStar +
259         WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
260       thetaStar =
261     V⁻¹.mulVec
262         (WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
263       V⁻¹.mulVec (V0.mulVec thetaStar)
264   calc
265     _ = V⁻¹.mulVec
266           (G.mulVec thetaStar +
267             WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
268         V⁻¹.mulVec (V.mulVec thetaStar) :=
269       congrArg
270         (fun z =>
271           V⁻¹.mulVec
272               (G.mulVec thetaStar +
273                 WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
274             z)
275         htheta.symm
276     _ = V⁻¹.mulVec
277           ((G.mulVec thetaStar +
278               WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)) -
279             V.mulVec thetaStar) := by
280       rw [Matrix.mulVec_sub]
281     _ = V⁻¹.mulVec
282           (WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega) -
283             V0.mulVec thetaStar) := by
284       congr 1
285       rw [Matrix.add_mulVec]
286       abel
287     _ = _ := by
288       rw [Matrix.mulVec_sub]

291 Deterministic confidence decomposition: estimator error is bounded by the
292 self-normalized score plus the regularization-bias norm.
293 -/
294 theorem finiteHorizonRidgeEstimate_error_matrixNorm_le_score_add_bias
295     [Fintype Feature] [DecidableEq Feature]
296     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
297     (thetaStar : Feature -> Real)
298     (feature : Nat -> Omega -> Feature -> Real)
299     (response noise : Nat -> Omega -> Real)
300     (n : Nat) (omega : Omega)

302       response i omega =

307       matrixNorm

321 /--
322 Pointwise transport from confidence-ellipsoid failure to the compiled
323 self-normalized score bad event.
324 -/
325 theorem
326     selfNormalizedQuadratic_gt_of_ridgeEstimate_error_matrixNorm_gt_confidenceRadius
327     [Fintype Feature] [DecidableEq Feature]
328     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
329     (thetaStar : Feature -> Real)
330     (feature : Nat -> Omega -> Feature -> Real)
331     (response noise : Nat -> Omega -> Real)
332     (R delta biasRadius : Real)
333     (n : Nat) (omega : Omega)
334     (hdelta : 0 < delta) (hdelta_one : delta <= 1)
335     (hresponse : forall i, i < n ->
336       response i omega =
337         dotProduct thetaStar (feature i omega) + noise i omega)
338     (hbias :
339       matrixNorm
340           (V0 + finiteHorizonFeatureGram feature n omega)
341           ((V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
342             (V0.mulVec thetaStar)) <=
343         biasRadius)
344     (hbad :
345       matrixNorm
346           (V0 + finiteHorizonFeatureGram feature n omega)
347           (finiteHorizonRidgeEstimate V0 feature response n omega -
348             thetaStar) >
349         finiteHorizonConfidenceRadius
350           V0 feature R delta biasRadius n omega) :
351     (finiteHorizonNoiseScore feature noise n omega) ⬝ᵥ
352           (V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
353             (finiteHorizonNoiseScore feature noise n omega) >
354       finiteHorizonConfidenceThreshold V0 feature R delta n omega := by
355   let V := V0 + finiteHorizonFeatureGram feature n omega
356   let score : Feature -> Real :=
357     WithLp.ofLp (finiteHorizonNoiseScore feature noise n omega)
358   have hV : V.PosDef :=
359     hV0.add_posSemidef
360       (finiteHorizonFeatureGram_posSemidef feature n omega)
361   have herror_le :
362       matrixNorm V
363           (finiteHorizonRidgeEstimate V0 feature response n omega -
364             thetaStar) <=
365         matrixNorm V (V⁻¹.mulVec score) + biasRadius := by
366     have hbias_add :
367         matrixNorm V (V⁻¹.mulVec score) +
368             matrixNorm V (V⁻¹.mulVec (V0.mulVec thetaStar)) <=
369           matrixNorm V (V⁻¹.mulVec score) + biasRadius :=
370       add_le_add_right hbias _
371     exact
372       (finiteHorizonRidgeEstimate_error_matrixNorm_le_score_add_bias
373         V0 hV0 thetaStar feature response noise n omega hresponse).trans
374         hbias_add
375   have hsqrt_lt :
376       Real.sqrt
377           (finiteHorizonConfidenceThreshold V0 feature R delta n omega) <
378         matrixNorm V (V⁻¹.mulVec score) := by
379     rw [finiteHorizonConfidenceRadius] at hbad
380     linarith
381   have hthreshold_nonneg :
382       0 <= finiteHorizonConfidenceThreshold V0 feature R delta n omega :=
383     finiteHorizonConfidenceThreshold_nonneg
384       V0 hV0 feature R delta n omega hdelta hdelta_one
385   have hsquared :
386       finiteHorizonConfidenceThreshold V0 feature R delta n omega <
387         matrixNorm V (V⁻¹.mulVec score) ^ 2 := by

396 /--
397 Finite-horizon OFUL least-squares confidence ellipsoid. With probability at
398 least `1 - delta`, the ridge-estimation error lies in the current Gram norm
399 inside the self-normalized radius plus the supplied regularization-bias cap.
400 -/
401 theorem
402     measure_finiteHorizonRidgeEstimate_error_matrixNorm_gt_confidenceRadius_le
403     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
404     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
405     (mu : Measure Omega) [IsProbabilityMeasure mu]
406     (V0 : Matrix Feature Feature Real) (hV0 : V0.PosDef)
407     (thetaStar : Feature -> Real)
408     (F : Filtration Nat mOmega)
409     (feature : Nat -> Omega -> Feature -> Real)
410     (response noise : Nat -> Omega -> Real)
411     (R : Real) (hR : 0 < R)
412     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
413     (hfeature : forall i j,
414       StronglyMeasurable[F i] (fun omega => feature i omega j))
415     (hnoise : StronglyAdapted F (fun t omega =>
416       match t with
417       | 0 => 0
418       | i + 1 => noise i omega))
419     (hprojectionBound_nonneg : forall theta i,
420       0 <= projectionBound theta i)
421     (hprojectionBound : forall theta i omega,
422       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
423         projectionBound theta i)
424     (n : Nat)
425     (hsubGaussian : forall i, i < n ->
426       HasCondSubgaussianMGF
427         (F i) (F.le i) (noise i)
428         (constantSquaredVarianceProxy R i) mu)
429     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1)
430     (biasRadius : Real)
431     (hresponse : forall omega i, i < n ->
432       response i omega =
433         dotProduct thetaStar (feature i omega) + noise i omega)
434     (hbias : forall omega,
435       matrixNorm
436           (V0 + finiteHorizonFeatureGram feature n omega)
437           ((V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
438             (V0.mulVec thetaStar)) <=
439         biasRadius) :
440     mu {omega |
441         matrixNorm
442             (V0 + finiteHorizonFeatureGram feature n omega)
443             (finiteHorizonRidgeEstimate V0 feature response n omega -
444               thetaStar) >
445           finiteHorizonConfidenceRadius
446             V0 feature R delta biasRadius n omega} <=
447       ENNReal.ofReal delta := by
448   calc
449     _ <=
450         mu {omega |
451           (finiteHorizonNoiseScore feature noise n omega) ⬝ᵥ
452                 (V0 + finiteHorizonFeatureGram feature n omega)⁻¹.mulVec
453                   (finiteHorizonNoiseScore feature noise n omega) >
454             finiteHorizonConfidenceThreshold
455               V0 feature R delta n omega} := by
456       apply measure_mono
457       intro omega hbad
458       exact
459         selfNormalizedQuadratic_gt_of_ridgeEstimate_error_matrixNorm_gt_confidenceRadius
460           V0 hV0 thetaStar feature response noise R delta biasRadius n omega
461           hdelta hdelta_one (hresponse omega) (hbias omega) hbad
462     _ <= ENNReal.ofReal delta := by
463       simpa [finiteHorizonConfidenceThreshold] using
464         measure_finiteHorizon_selfNormalizedQuadratic_gt_two_mul_sq_mul_log_detRatio_div_le
465           mu V0 hV0 F feature noise R hR projectionBound hfeature hnoise
466           hprojectionBound_nonneg hprojectionBound n hsubGaussian delta hdelta
