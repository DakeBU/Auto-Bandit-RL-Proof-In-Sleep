10 /-!
11 # OFUL elliptical-potential linear algebra wrappers
12 
13 This module records the deterministic finite-dimensional linear-algebra
14 surface needed by OFUL/LinUCB routes.  It packages rank-one and finite-history
15 feature Gram matrices, positive-definiteness and inverse facts, determinant
16 updates, log-determinant telescoping, trace/eigenvalue determinant bounds, and
17 the standard logarithmic clipped elliptical-potential endpoint.
18 
19 It does not prove self-normalized concentration, confidence ellipsoids,
20 optimism, or an OFUL regret theorem.

28 /-- Rank-one Gram matrix generated by one feature vector. -/
29 def rankOneGram {Feature : Type u}
30     (x : Feature -> Real) : Matrix Feature Feature Real :=
31   fun i j => x i * x j

157 /-- Quadratic forms distribute over matrix addition. -/
158 theorem quadraticForm_add
159     {Feature : Type u} [Fintype Feature]
160     (A B : Matrix Feature Feature Real) (y : Feature -> Real) :
161     quadraticForm (A + B) y = quadraticForm A y + quadraticForm B y := by
162   unfold quadraticForm
163   calc
164     (Finset.univ : Finset Feature).sum
165         (fun i => (Finset.univ : Finset Feature).sum
166           (fun j => y i * (A i j + B i j) * y j)) =
167       (Finset.univ : Finset Feature).sum
168         (fun i => (Finset.univ : Finset Feature).sum
169           (fun j => y i * A i j * y j + y i * B i j * y j)) := by
170       apply Finset.sum_congr rfl
171       intro i _hi
172       apply Finset.sum_congr rfl
173       intro j _hj
174       ring
175     _ =
176       (Finset.univ : Finset Feature).sum
177         (fun i =>
178           (Finset.univ : Finset Feature).sum
179             (fun j => y i * A i j * y j) +
180           (Finset.univ : Finset Feature).sum
181             (fun j => y i * B i j * y j)) := by
182       apply Finset.sum_congr rfl
183       intro i _hi
184       rw [Finset.sum_add_distrib]
185     _ =
186       (Finset.univ : Finset Feature).sum
187         (fun i => (Finset.univ : Finset Feature).sum

189       (Finset.univ : Finset Feature).sum

423 /-- Finite nonnegative product is bounded by the corresponding arithmetic mean power. -/
424 theorem finset_prod_le_pow_sum_div_card_of_nonneg
425     {ι : Type v} (s : Finset ι) (hs : s.Nonempty)
426     (z : ι -> Real) (hz : forall i : ι, i ∈ s -> 0 <= z i) :
427     (s.prod z) <= ((s.sum z) / (s.card : Real)) ^ s.card := by
428   have hcard_ne : s.card ≠ 0 := hs.card_ne_zero
429   have hsumw : (s.sum (fun _ : ι => (1 : Real))) = (s.card : Real) := by
430     simp
431   have hwpos : 0 < s.sum (fun _ : ι => (1 : Real)) := by
432     rw [hsumw]
433     exact_mod_cast hs.card_pos
434   have hamgm := Real.geom_mean_le_arith_mean s
435     (fun _ : ι => (1 : Real)) z
436     (fun _ _ => by norm_num) hwpos hz
437   have hgeom :
438       (s.prod z) ^ ((s.card : Real)⁻¹) <=
439         (s.sum z) / (s.card : Real) := by
440     simpa [hsumw] using hamgm
441   have hprod_nonneg : 0 <= s.prod z :=
442     Finset.prod_nonneg (fun i hi => hz i hi)
443   have hpow := pow_le_pow_left₀
444     (Real.rpow_nonneg hprod_nonneg _) hgeom s.card
445   simpa [Real.rpow_inv_natCast_pow hprod_nonneg hcard_ne] using hpow
