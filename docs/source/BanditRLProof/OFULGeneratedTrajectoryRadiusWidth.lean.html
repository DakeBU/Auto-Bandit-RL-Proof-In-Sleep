20 /-- The standard log-determinant budget for a bounded scalar-ridge prefix. -/
21 noncomputable def standardScalarLogDetBudget
22     {Feature : Type u} [Fintype Feature]
23     (lambda : Real) (T : Nat) (L2 : Real) : Real :=
24   (Fintype.card Feature : Real) *
25     Real.log
26       (1 +
27         (T * L2) /
28           ((Fintype.card Feature : Real) * lambda))
29 
30 /--
31 A deterministic upper radius obtained by replacing the random determinant
32 ratio with `exp (standardScalarLogDetBudget ...)`.
33 -/
34 noncomputable def standardScalarConfidenceRadiusUpper
35     {Feature : Type u} [Fintype Feature]
36     (R delta lambda S : Real) (T : Nat) (L2 : Real) : Real :=
37   Real.sqrt
38       (2 * R ^ 2 *
39         Real.log
40           (Real.sqrt
41               (Real.exp
42                 (standardScalarLogDetBudget
43                   (Feature := Feature) lambda T L2)) /
44             delta)) +
45     Real.sqrt lambda * S

47 /-- The selected-width budget paired with the standard log-determinant term. -/
48 noncomputable def standardSelectedWidthBudget
49     {Feature : Type u} [Fintype Feature]
50     (lambda : Real) (T : Nat) (L2 : Real) : Real :=
51   Real.sqrt (T : Real) *
52     Real.sqrt
53       (2 *
54         standardScalarLogDetBudget
55           (Feature := Feature) lambda T L2)

57 /-- Deterministic radius-times-width budget used by the canonical gap theorem. -/
58 noncomputable def standardScalarRadiusWidthBound
59     {Feature : Type u} [Fintype Feature]
60     (R delta lambda S : Real) (T : Nat) (L2 : Real) : Real :=
61   2 *
62     standardScalarConfidenceRadiusUpper
63       (Feature := Feature) R delta lambda S T L2 *
64     standardSelectedWidthBudget

78 /-- The standard log-determinant budget is monotone in the horizon. -/
79 theorem standardScalarLogDetBudget_mono
80     {Feature : Type u}
81     [Fintype Feature] [Nonempty Feature]
82     (lambda : Real) (hlambda : 0 < lambda)
83     (L2 : Real) (hL2 : 0 <= L2)
84     {n T : Nat} (hnT : n <= T) :
85     standardScalarLogDetBudget (Feature := Feature) lambda n L2 <=
86       standardScalarLogDetBudget (Feature := Feature) lambda T L2 := by
87   have hcard_pos : 0 < (Fintype.card Feature : Real) := by
88     exact_mod_cast (Fintype.card_pos_iff.mpr inferInstance :
89       0 < Fintype.card Feature)
90   have hdenom_pos :
91       0 < (Fintype.card Feature : Real) * lambda :=
92     mul_pos hcard_pos hlambda
93   have hnT_real : (n : Real) <= (T : Real) := by
94     exact_mod_cast hnT
95   have hmul :
96       (n : Real) * L2 <= (T : Real) * L2 :=
97     mul_le_mul_of_nonneg_right hnT_real hL2
98   have hfrac :
99       (n * L2) / ((Fintype.card Feature : Real) * lambda) <=
100         (T * L2) / ((Fintype.card Feature : Real) * lambda) :=
101     (div_le_div_iff_of_pos_right hdenom_pos).2 hmul
102   have harg_nonneg :
103       0 <=
104         (n * L2) / ((Fintype.card Feature : Real) * lambda) :=
105     div_nonneg
106       (mul_nonneg (Nat.cast_nonneg _) hL2) hdenom_pos.le
107   have hlog :
108       Real.log
109           (1 + (n * L2) /
110             ((Fintype.card Feature : Real) * lambda)) <=
111         Real.log
112           (1 + (T * L2) /
113             ((Fintype.card Feature : Real) * lambda)) :=
114     Real.log_le_log (by linarith) (add_le_add_right hfrac 1)
115   exact mul_le_mul_of_nonneg_left hlog (Nat.cast_nonneg _)

138 /--
139 Every prefix scalar confidence radius is bounded by the standard deterministic
140 radius at a later horizon.
141 -/
142 theorem finiteHorizonScalarConfidenceRadius_le_standardUpper
143     {Omega Feature : Type*}
144     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
145     (lambda : Real) (hlambda : 0 < lambda)
146     (feature : Nat -> Omega -> Feature -> Real)
147     (R delta S : Real)
148     (omega : Omega) (n T : Nat) (hnT : n <= T)
149     (L2 : Real) (hL2 : 0 <= L2)
150     (hbound : forall t, t < T ->
151       dotProduct (feature t omega) (feature t omega) <= L2)
152     (hdelta : 0 < delta) :
153     finiteHorizonScalarConfidenceRadius
154         feature R delta lambda S n omega <=
155       standardScalarConfidenceRadiusUpper
156         (Feature := Feature) R delta lambda S T L2 := by
157   let history : Nat -> Feature -> Real := fun t => feature t omega
158   let Bn :=
159     standardScalarLogDetBudget (Feature := Feature) lambda n L2
160   let BT :=
161     standardScalarLogDetBudget (Feature := Feature) lambda T L2
162   have hbound_n : forall t, t < n ->
163       dotProduct (history t) (history t) <= L2 := by
164     intro t htn
165     exact hbound t (lt_of_lt_of_le htn hnT)
166   have hdet_n :
167       (regularizedPrefixFeatureGram lambda history n).det <=
168         lambda ^ Fintype.card Feature * Real.exp Bn := by
169     simpa only [Bn, standardScalarLogDetBudget] using
170       (standardLogDeterminantAndEllipticalPotential
171         lambda hlambda history n L2 hL2 hbound_n).1
172   have hbudget : Bn <= BT := by
173     exact standardScalarLogDetBudget_mono
174       (Feature := Feature) lambda hlambda L2 hL2 hnT
175   have hdet :
176       (regularizedPrefixFeatureGram lambda history n).det <=
177         lambda ^ Fintype.card Feature * Real.exp BT :=
178     hdet_n.trans
179       (mul_le_mul_of_nonneg_left
180         (Real.exp_le_exp.mpr hbudget)
181         (pow_nonneg hlambda.le _))
182   have hbase_pos : 0 < lambda ^ Fintype.card Feature :=
183     pow_pos hlambda _
184   have hratio :
185       (regularizedPrefixFeatureGram lambda history n).det /
186           (Matrix.scalar Feature lambda).det <=
187         Real.exp BT := by
188     rw [det_scalar_identity]
189     calc
190       (regularizedPrefixFeatureGram lambda history n).det /
191           lambda ^ Fintype.card Feature <=
192         (lambda ^ Fintype.card Feature * Real.exp BT) /
193           lambda ^ Fintype.card Feature :=
194         (div_le_div_iff_of_pos_right hbase_pos).2 hdet
195       _ = Real.exp BT := by
196         field_simp
197   have hdet_pos :
198       0 < (regularizedPrefixFeatureGram lambda history n).det :=
199     regularizedPrefixFeatureGram_det_pos lambda hlambda history n
200   have hratio_pos :
201       0 <
202         (regularizedPrefixFeatureGram lambda history n).det /
203           (Matrix.scalar Feature lambda).det := by
204     rw [det_scalar_identity]
205     exact div_pos hdet_pos hbase_pos
206   have hsqrt :
207       Real.sqrt
208           ((regularizedPrefixFeatureGram lambda history n).det /
209             (Matrix.scalar Feature lambda).det) <=
210         Real.sqrt (Real.exp BT) :=
211     Real.sqrt_le_sqrt hratio
212   have hlog :
213       Real.log
214           (Real.sqrt
215               ((regularizedPrefixFeatureGram lambda history n).det /
216                 (Matrix.scalar Feature lambda).det) /
217             delta) <=
218         Real.log (Real.sqrt (Real.exp BT) / delta) := by
219     apply Real.log_le_log
220     · exact div_pos (Real.sqrt_pos.2 hratio_pos) hdelta
221     · exact (div_le_div_iff_of_pos_right hdelta).2 hsqrt
222   have hthreshold :
223       2 * R ^ 2 *
224           Real.log
225             (Real.sqrt
226                 ((regularizedPrefixFeatureGram lambda history n).det /
227                   (Matrix.scalar Feature lambda).det) /
228               delta) <=
229         2 * R ^ 2 *
230           Real.log (Real.sqrt (Real.exp BT) / delta) :=
231     mul_le_mul_of_nonneg_left hlog
232       (mul_nonneg (by norm_num) (sq_nonneg R))
233   unfold finiteHorizonScalarConfidenceRadius finiteHorizonConfidenceRadius
234     finiteHorizonConfidenceThreshold standardScalarConfidenceRadiusUpper
235   change
236     Real.sqrt
237           (2 * R ^ 2 *
238             Real.log
239               (Real.sqrt
240                   ((regularizedPrefixFeatureGram lambda history n).det /
241                     (Matrix.scalar Feature lambda).det) /
242                 delta)) +
243         Real.sqrt lambda * S <=
244       Real.sqrt
245           (2 * R ^ 2 *
246             Real.log (Real.sqrt (Real.exp BT) / delta)) +
247         Real.sqrt lambda * S
248   exact add_le_add (Real.sqrt_le_sqrt hthreshold) le_rfl
