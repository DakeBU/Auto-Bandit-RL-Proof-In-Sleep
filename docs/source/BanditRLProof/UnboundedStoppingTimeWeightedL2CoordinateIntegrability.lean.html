5 # Square-summable weights at an unbounded stopping time
6 
7 This module replaces a moment assumption on an a.e.-finite stopping time by a
8 square-summable deterministic weight.  Uniform L2 control of the deterministic
9 coordinates and Cauchy--Schwarz over the measurable stopping fibers then give
10 an integrable weighted stopped value.  This is a countable-fiber argument, not
11 optional stopping.
12 -/

21 /-- A stopped value is measurable when the stopping index and every
22 deterministic coordinate are measurable.  The proof decomposes the dynamic
23 evaluation into countably many natural-number fibers. -/
24 theorem measurable_stoppedValue_of_measurable_coordinates
25     {Omega : Type u} [MeasurableSpace Omega]
26     (tau : Omega -> WithTop Nat) (htau : Measurable tau)
27     (process : Nat -> Omega -> Real)
28     (hprocess : forall n, Measurable (process n)) :
29     Measurable (stoppedValue process tau) := by
30   let index : Omega -> Nat := fun omega => (tau omega).untopA
31   have hindex : Measurable index := htau.untopA
32   intro s hs
33   have hpreimage :
34       stoppedValue process tau ⁻¹' s =
35         ⋃ n : Nat, {omega | index omega = n} ∩ process n ⁻¹' s := by
36     ext omega
37     simp only [Set.mem_preimage, Set.mem_iUnion, Set.mem_inter_iff,
38       Set.mem_setOf_eq]
39     constructor
40     · intro homega
41       refine ⟨index omega, rfl, ?_⟩
42       simpa only [index, stoppedValue] using homega
43     · rintro ⟨n, hindexEq, homega⟩
44       rw [← hindexEq] at homega
45       simpa only [index, stoppedValue] using homega
46   rw [hpreimage]
47   exact MeasurableSet.iUnion fun n =>
48     (measurableSet_eq_fun hindex measurable_const).inter
49       ((hprocess n) hs)

51 /-- Square-summable deterministic weights are summable against the square
52 roots of the real masses of measurable stopping fibers. -/
53 theorem
54     summable_abs_weight_mul_sqrt_stoppingFiberRealMeasure_and_tsum_le
55     {Omega : Type u} [MeasurableSpace Omega]
56     (mu : Measure Omega) [IsFiniteMeasure mu]
57     (tau : Omega -> WithTop Nat) (htau : Measurable tau)
58     (weight : Nat -> Real)
59     (hweightSq : Summable (fun n => weight n ^ 2)) :
60     Summable (fun n =>
61         |weight n| *
62           Real.sqrt (mu.real {omega | tau omega = (n : WithTop Nat)})) /\
63       (∑' n : Nat,
64           |weight n| *
65             Real.sqrt (mu.real {omega | tau omega = (n : WithTop Nat)})) <=
66         Real.sqrt (∑' n : Nat, weight n ^ 2) *
67           Real.sqrt (mu.real Set.univ) := by
68   let fiber : Nat -> Set Omega := fun n =>
69     {omega | tau omega = (n : WithTop Nat)}
70   have hfiberMeasurable (n : Nat) : MeasurableSet (fiber n) := by
71     exact measurableSet_eq_fun htau measurable_const
72   have hfiberDisjoint : Pairwise (fun i j => Disjoint (fiber i) (fiber j)) := by
73     intro i j hij
74     rw [Set.disjoint_left]
75     intro omega hi hj
76     have hijTop : (i : WithTop Nat) = (j : WithTop Nat) := by
77       exact hi.symm.trans hj
78     exact hij (WithTop.coe_injective hijTop)
79   have hfiberSummable : Summable (fun n => mu.real (fiber n)) :=
80     summable_measure_toReal hfiberMeasurable hfiberDisjoint
81   have hfiberTsumLe :
82       (∑' n : Nat, mu.real (fiber n)) <= mu.real Set.univ := by
83     calc
84       (∑' n : Nat, mu.real (fiber n)) =
85           (∑' n : Nat, mu (fiber n)).toReal := by
86         simp only [measureReal_def]
87         rw [ENNReal.tsum_toReal_eq (fun n => measure_ne_top mu (fiber n))]
88       _ = mu.real (Set.iUnion fiber) := by
89         rw [measureReal_def,
90           MeasureTheory.measure_iUnion hfiberDisjoint hfiberMeasurable]
91       _ <= mu.real Set.univ := measureReal_mono (Set.subset_univ _)
92   have hweightAbsSq : Summable (fun n => |weight n| ^ (2 : Real)) := by
93     simpa [Real.rpow_two, sq_abs] using hweightSq
94   have hfiberSqrtSq : Summable (fun n =>
95       (Real.sqrt (mu.real (fiber n))) ^ (2 : Real)) := by
96     simpa [Real.rpow_two, Real.sq_sqrt measureReal_nonneg] using
97       hfiberSummable
98   have hpq : (2 : Real).HolderConjugate 2 := by
99     rw [Real.holderConjugate_iff]
100     norm_num
101   have hholder :=
102     Real.summable_and_inner_le_Lp_mul_Lq_tsum_of_nonneg hpq
103       (fun n => abs_nonneg (weight n))
104       (fun n => Real.sqrt_nonneg _)
105       hweightAbsSq hfiberSqrtSq
106   refine ⟨?_, ?_⟩
107   · simpa only [fiber] using hholder.1
108   · have hraw :
109         (∑' n : Nat, |weight n| * Real.sqrt (mu.real (fiber n))) <=
110           Real.sqrt (∑' n : Nat, weight n ^ 2) *
111             Real.sqrt (∑' n : Nat, mu.real (fiber n)) := by
112       simpa only [← Real.sqrt_eq_rpow, Real.rpow_two, sq_abs,
113         Real.sq_sqrt measureReal_nonneg] using hholder.2
114     calc
115       (∑' n : Nat,
116           |weight n| *
117             Real.sqrt (mu.real {omega | tau omega = (n : WithTop Nat)})) =
118           (∑' n : Nat,
119             |weight n| * Real.sqrt (mu.real (fiber n))) := rfl
120       _ <= Real.sqrt (∑' n : Nat, weight n ^ 2) *
121           Real.sqrt (∑' n : Nat, mu.real (fiber n)) := hraw
122       _ <= Real.sqrt (∑' n : Nat, weight n ^ 2) *
123           Real.sqrt (mu.real Set.univ) := by
124         exact mul_le_mul_of_nonneg_left (Real.sqrt_le_sqrt hfiberTsumLe)
125           (Real.sqrt_nonneg _)

127 /-- Uniform deterministic-coordinate second moments and a square-summable
128 deterministic weight make the weighted stopped value integrable and bound its
129 absolute first moment. -/
130 theorem
131     integrable_and_integral_abs_stoppedValue_weight_mul_le
132     {Omega : Type u} [MeasurableSpace Omega]
133     (mu : Measure Omega) [IsFiniteMeasure mu]
134     (tau : Omega -> WithTop Nat) (htau : Measurable tau)
135     (hfinite : ∀ᵐ omega ∂mu, tau omega ≠ ⊤)
136     (process : Nat -> Omega -> Real)
137     (weight : Nat -> Real)
138     (hweightSq : Summable (fun n => weight n ^ 2))
139     (hstoppedMeasurable :
140       Measurable (stoppedValue (fun n omega => weight n * process n omega) tau))
141     (secondMomentEnvelope : Real)
142     (hprocessMemLp : forall n, MemLp (process n) 2 mu)
143     (hprocessSecondMoment : forall n,
144       integral mu (fun omega => process n omega ^ 2) <=
145         secondMomentEnvelope) :
146     Integrable
147         (stoppedValue (fun n omega => weight n * process n omega) tau) mu /\
148       integral mu (fun omega =>
149           |stoppedValue (fun n omega => weight n * process n omega) tau omega|) <=
150         Real.sqrt secondMomentEnvelope *
151           (Real.sqrt (∑' n : Nat, weight n ^ 2) *
152             Real.sqrt (mu.real Set.univ)) := by
153   let fiber : Nat -> Set Omega := fun n =>
154     {omega | tau omega = (n : WithTop Nat)}
155   let term : Nat -> Omega -> Real := fun n =>
156     (fiber n).indicator (fun omega => |weight n * process n omega|)
157   let majorant : Nat -> Real := fun n =>
158     Real.sqrt secondMomentEnvelope *
159       (|weight n| * Real.sqrt (mu.real (fiber n)))
160   have hfiberMeasurable (n : Nat) : MeasurableSet (fiber n) := by
161     exact measurableSet_eq_fun htau measurable_const
162   have hweightedFiber :=
163     summable_abs_weight_mul_sqrt_stoppingFiberRealMeasure_and_tsum_le
164       mu tau htau weight hweightSq
165   have hmajorantSummable : Summable majorant := by
166     exact hweightedFiber.1.mul_left (Real.sqrt secondMomentEnvelope)
167   have htermIntegrable (n : Nat) : Integrable (term n) mu := by
168     dsimp only [term]
169     exact (((hprocessMemLp n).const_mul (weight n)).integrable
170       (by norm_num)).abs.indicator (hfiberMeasurable n)
171   have htermNonneg (n : Nat) (omega : Omega) : 0 <= term n omega := by
172     by_cases homega : omega ∈ fiber n
173     · simp only [term, Set.indicator_of_mem homega]
174       exact abs_nonneg _
175     · simp [term, homega]
176   have htermIntegral (n : Nat) :
177       integral mu (term n) <= majorant n := by
178     have hholder :=
179       integral_indicator_le_sqrt_secondMoment_mul_sqrt_real_measure
180         mu (fun omega => |process n omega|) (fun _ => abs_nonneg _)
181           (hprocessMemLp n).abs (fiber n) (hfiberMeasurable n)
182     have hsqrtMoment :
183         Real.sqrt (integral mu (fun omega => |process n omega| ^ 2)) <=
184           Real.sqrt secondMomentEnvelope := by
185       apply Real.sqrt_le_sqrt
186       simpa only [sq_abs] using hprocessSecondMoment n
187     calc
188       integral mu (term n) =
189           |weight n| *
190             integral mu ((fiber n).indicator (fun omega => |process n omega|)) := by
191         have htermEq : term n = fun omega =>
192             |weight n| *
193               (fiber n).indicator (fun omega => |process n omega|) omega := by
194           funext omega
195           by_cases homega : omega ∈ fiber n <;>
196             simp [term, homega, abs_mul]
197         rw [htermEq, integral_const_mul]
198       _ <= |weight n| *
199           (Real.sqrt (integral mu (fun omega => |process n omega| ^ 2)) *
200             Real.sqrt (mu.real (fiber n))) :=
201         mul_le_mul_of_nonneg_left hholder (abs_nonneg _)
202       _ <= |weight n| *
203           (Real.sqrt secondMomentEnvelope *
204             Real.sqrt (mu.real (fiber n))) := by
205         gcongr
206       _ = majorant n := by
207         simp [majorant]
208         ring
209   have htermIntegralNormSummable :
210       Summable (fun n => integral mu (fun omega => ‖term n omega‖)) := by
211     apply hmajorantSummable.of_nonneg_of_le
212     · intro n
213       exact integral_nonneg fun omega => norm_nonneg (term n omega)
214     · intro n
215       rw [show integral mu (fun omega => ‖term n omega‖) =
216           integral mu (term n) by
217         apply integral_congr_ae
218         filter_upwards with omega
219         rw [Real.norm_eq_abs, abs_of_nonneg (htermNonneg n omega)]]
220       exact htermIntegral n
221   have htermIntegralSummable :
222       Summable (fun n => integral mu (term n)) := by
223     apply hmajorantSummable.of_nonneg_of_le
224     · intro n
225       exact integral_nonneg (htermNonneg n)
226     · exact htermIntegral
227   have hpoint : ∀ᵐ omega ∂mu,
228       |stoppedValue (fun n omega => weight n * process n omega) tau omega| =
229         ∑' n : Nat, term n omega := by
230     filter_upwards [hfinite] with omega htop
231     lift tau omega to Nat using htop with hit hhit
232     dsimp only [term, fiber]
233     rw [tsum_eq_single hit]
234     · rw [Set.indicator_of_mem]
235       · simp only [stoppedValue]
236         have huntopA : (tau omega).untopA = hit := by
237           rw [← hhit]
238           rfl
239         rw [huntopA]
240       · exact hhit.symm
241     · intro n hn
242       rw [Set.indicator_of_notMem]
243       intro hmem
244       have heq : hit = n := by
245         exact WithTop.coe_injective (hhit.trans hmem)
246       exact hn heq.symm
247   let ennTerm : Nat -> Omega -> ENNReal := fun n =>
248     (fiber n).indicator (fun omega => ENNReal.ofReal |weight n * process n omega|)
249   have hennTermMeasurable (n : Nat) : AEMeasurable (ennTerm n) mu := by
250     exact (((hprocessMemLp n).const_mul (weight n)).aestronglyMeasurable.norm
251       |>.aemeasurable.ennreal_ofReal.indicator (hfiberMeasurable n))
252   have hennTermIntegral (n : Nat) :
253       ∫⁻ omega, ennTerm n omega ∂mu <= ENNReal.ofReal (majorant n) := by
254     have htermIntegralNonneg : 0 <= integral mu (term n) :=
255       integral_nonneg (htermNonneg n)
256     have heq :
257         (∫⁻ omega, ennTerm n omega ∂mu) =
258           ENNReal.ofReal (integral mu (term n)) := by
259       rw [ofReal_integral_eq_lintegral_ofReal (htermIntegrable n)
260         (Filter.Eventually.of_forall (htermNonneg n))]
261       apply lintegral_congr
262       intro omega
263       by_cases homega : omega ∈ fiber n <;>
264         simp [ennTerm, term, homega]
265     rw [heq]
266     exact ENNReal.ofReal_le_ofReal (htermIntegral n)
267   have hennPoint : ∀ᵐ omega ∂mu,
268       ENNReal.ofReal
269           |stoppedValue (fun n omega => weight n * process n omega) tau omega| =
270         ∑' n : Nat, ennTerm n omega := by
271     filter_upwards [hfinite] with omega htop
272     lift tau omega to Nat using htop with hit hhit
273     dsimp only [ennTerm, fiber]
274     rw [tsum_eq_single hit]
275     · rw [Set.indicator_of_mem]
276       · simp only [stoppedValue]
277         have huntopA : (tau omega).untopA = hit := by
278           rw [← hhit]
279           rfl
280         rw [huntopA]
281       · exact hhit.symm
282     · intro n hn
283       rw [Set.indicator_of_notMem]
284       intro hmem
285       have heq : hit = n := by
286         exact WithTop.coe_injective (hhit.trans hmem)
287       exact hn heq.symm
288   have hennSumFinite :
289       (∑' n : Nat, ∫⁻ omega, ennTerm n omega ∂mu) < ⊤ := by
290     apply lt_of_le_of_lt (ENNReal.tsum_le_tsum hennTermIntegral)
291     exact lt_top_iff_ne_top.mpr hmajorantSummable.tsum_ofReal_ne_top
292   have hstoppedIntegrable :
293       Integrable
294         (stoppedValue (fun n omega => weight n * process n omega) tau) mu := by
295     refine ⟨hstoppedMeasurable.aestronglyMeasurable, ?_⟩
296     rw [hasFiniteIntegral_iff_norm]
297     calc
298       ∫⁻ omega,
299           ENNReal.ofReal
300             ‖stoppedValue (fun n omega => weight n * process n omega) tau omega‖
301           ∂mu =
302           ∫⁻ omega, ∑' n : Nat, ennTerm n omega ∂mu := by
303         apply lintegral_congr_ae
304         filter_upwards [hennPoint] with omega homega
305         simpa only [Real.norm_eq_abs] using homega
306       _ = ∑' n : Nat, ∫⁻ omega, ennTerm n omega ∂mu := by
307         rw [lintegral_tsum hennTermMeasurable]
308       _ < ⊤ := hennSumFinite
309   refine ⟨hstoppedIntegrable, ?_⟩
310   calc
311     integral mu (fun omega =>
312         |stoppedValue (fun n omega => weight n * process n omega) tau omega|) =
313         integral mu (fun omega => ∑' n : Nat, term n omega) :=
314       integral_congr_ae hpoint
315     _ = ∑' n : Nat, integral mu (term n) :=
316       (integral_tsum_of_summable_integral_norm htermIntegrable
317         htermIntegralNormSummable).symm
318     _ <= ∑' n : Nat, majorant n := by
319       exact htermIntegralSummable.tsum_le_tsum htermIntegral
320         hmajorantSummable
321     _ = Real.sqrt secondMomentEnvelope *
322         (∑' n : Nat,
323           |weight n| * Real.sqrt (mu.real (fiber n))) := by
324       rw [tsum_mul_left]
325     _ <= Real.sqrt secondMomentEnvelope *
326         (Real.sqrt (∑' n : Nat, weight n ^ 2) *
327           Real.sqrt (mu.real Set.univ)) :=
328       mul_le_mul_of_nonneg_left hweightedFiber.2 (Real.sqrt_nonneg _)
