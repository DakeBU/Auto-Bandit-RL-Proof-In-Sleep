18 /-- A nondegenerate interval has a strictly positive Hoeffding proxy. -/
19 theorem intervalVarianceProxy_pos_of_lt
20     {lo hi : Real} (hlohi : lo < hi) :
21     0 < ((intervalVarianceProxy lo hi : NNReal) : Real) := by
22   have hdiff : 0 < hi - lo := sub_pos.mpr hlohi
23   have hnorm : 0 < ‖hi - lo‖ := norm_pos_iff.mpr (ne_of_gt hdiff)
24   rw [show ((intervalVarianceProxy lo hi : NNReal) : Real) =
25       (‖hi - lo‖ / 2) ^ 2 by
26     simp [intervalVarianceProxy]]
27   positivity

33 /--
34 Exact pointwise means and centered sub-Gaussian MGF witnesses form a centered
35 law for an arbitrary context/action Markov reward kernel.
36 -/
37 noncomputable def centeredRewardKernelLaw_of_hasSubgaussianMGF
38     {Context Action : Type}
39     [MeasurableSpace Context] [MeasurableSpace Action]
40     (rewardKernel : MarkovRewardKernel (Context × Action) Rat)
41     (mean : Context -> Action -> Rat)
42     (varianceProxy : Context -> Action -> NNReal)
43     (hmean : forall context arm,
44       integral (selectedMeasure rewardKernel context arm)
45           (fun reward : Rat => ((reward : Rat) : Real)) =
46         ((mean context arm : Rat) : Real))
47     (hsubG : forall context arm,
48       HasSubgaussianMGF
49         (fun reward : Rat =>
50           (((reward - mean context arm : Rat) : Real)))
51         (varianceProxy context arm)
52         (selectedMeasure rewardKernel context arm)) :
53     CenteredRewardKernelLaw rewardKernel mean varianceProxy where
54   integrable := by
55     intro context arm
56     exact (hsubG context arm).integrable
57   integral_eq_zero := by
58     intro context arm
59     haveI : IsProbabilityMeasure
60         (selectedMeasure rewardKernel context arm) := by
61       change IsProbabilityMeasure (rewardKernel.kernel (context, arm))
62       exact isProbabilityMeasure_apply rewardKernel (context, arm)
63     have hcenter := (hsubG context arm).integrable
64     have hconst : Integrable
65         (fun _reward : Rat => ((mean context arm : Rat) : Real))
66         (selectedMeasure rewardKernel context arm) := integrable_const _
67     have hraw : Integrable
68         (fun reward : Rat => ((reward : Rat) : Real))
69         (selectedMeasure rewardKernel context arm) := by
70       convert hcenter.add hconst using 1
71       funext reward
72       simp
73     simp_rw [Rat.cast_sub]
74     rw [integral_sub hraw hconst, hmean context arm]
75     simp
76   hasSubgaussianMGF := hsubG

78 /--
79 Common almost-sure reward bounds and exact pointwise means form a centered law
80 for an arbitrary context/action Markov reward kernel.
81 -/
82 noncomputable def boundedCenteredRewardKernelLaw
83     {Context Action : Type}
84     [MeasurableSpace Context] [MeasurableSpace Action]
85     (rewardKernel : MarkovRewardKernel (Context × Action) Rat)
86     (mean : Context -> Action -> Rat)
87     (lo hi : Real)
88     (hmeas : forall context arm,
89       AEMeasurable (fun reward : Rat => ((reward : Rat) : Real))
90         (selectedMeasure rewardKernel context arm))
91     (hbound : forall context arm,
92       Filter.Eventually
93         (fun reward : Rat => Set.Icc lo hi ((reward : Rat) : Real))
94         (ae (selectedMeasure rewardKernel context arm)))
95     (hmean : forall context arm,
96       integral (selectedMeasure rewardKernel context arm)
97           (fun reward : Rat => ((reward : Rat) : Real)) =
98         ((mean context arm : Rat) : Real)) :
99     CenteredRewardKernelLaw rewardKernel mean
100       (fun _ _ => Concentration.intervalVarianceProxy lo hi) := by
101   apply centeredRewardKernelLaw_of_hasSubgaussianMGF
102     rewardKernel mean
103       (fun _ _ => Concentration.intervalVarianceProxy lo hi) hmean
104   intro context arm
105   haveI : IsProbabilityMeasure
106       (selectedMeasure rewardKernel context arm) := by
107     change IsProbabilityMeasure (rewardKernel.kernel (context, arm))
108     exact isProbabilityMeasure_apply rewardKernel (context, arm)
109   simpa [Rat.cast_sub] using
110     (Concentration.boundedCentered_hasSubgaussianMGF_of_mem_Icc_integral_eq
111       (mu := selectedMeasure rewardKernel context arm)
112       (X := fun reward : Rat => ((reward : Rat) : Real))
113       (lo := lo)
114       (hi := hi)
115       (mean := ((mean context arm : Rat) : Real))
116       (hmeas context arm) (hbound context arm) (hmean context arm))
