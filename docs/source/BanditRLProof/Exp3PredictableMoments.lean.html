21 Mixing fixed-environment trajectory laws preserves a history-dependent output
22 kernel when the conditioning variable retains the environment coordinate.

24 theorem trajectoryMixture_map_environment_history_output_eq_compProd
25     {Env : Type u} {Omega : Type v} {History : Type w} {Output : Type x}
26     [MeasurableSpace Env] [MeasurableSpace Omega]
27     [MeasurableSpace History] [MeasurableSpace Output]
28     (prior : Measure Env) [IsFiniteMeasure prior]
29     (trajectory : Kernel Env Omega) [IsMarkovKernel trajectory]
30     (history : Omega -> History) (hhistory : Measurable history)
31     (output : Omega -> Output) (houtput : Measurable output)
32     (outputKernel : Kernel (Env × History) Output)
33     [IsMarkovKernel outputKernel]
34     (hlaw : forall env,
35       (trajectory env).map (fun omega => (history omega, output omega)) =
36         (trajectory env).map history ⊗ₘ
37           outputKernel.comap (fun h => (env, h))
38             (measurable_const.prodMk measurable_id)) :
39     (prior ⊗ₘ trajectory).map
40         (fun sample : Env × Omega =>
41           ((sample.1, history sample.2), output sample.2)) =
42       (prior ⊗ₘ trajectory).map
43           (fun sample : Env × Omega => (sample.1, history sample.2)) ⊗ₘ
44         outputKernel := by
45   apply Measure.ext_prod
46   intro environmentHistoryEvent outputEvent hEnvironmentHistoryEvent hOutputEvent
47   let pairMap : Omega -> History × Output :=
48     fun omega => (history omega, output omega)
49   let globalPairMap : Env × Omega -> (Env × History) × Output :=
50     fun sample => ((sample.1, history sample.2), output sample.2)
51   let globalHistory : Env × Omega -> Env × History :=
52     fun sample => (sample.1, history sample.2)
53   have hpairMap : Measurable pairMap := hhistory.prodMk houtput
54   have hglobalPairMap : Measurable globalPairMap :=
55     (measurable_fst.prodMk (hhistory.comp measurable_snd)).prodMk
56       (houtput.comp measurable_snd)
57   have hglobalHistory : Measurable globalHistory :=
58     measurable_fst.prodMk (hhistory.comp measurable_snd)
59   rw [Measure.map_apply hglobalPairMap
60       (hEnvironmentHistoryEvent.prod hOutputEvent),
61     Measure.compProd_apply_prod hEnvironmentHistoryEvent hOutputEvent]
62   conv_rhs =>
63     change ∫⁻ a in environmentHistoryEvent,
64       outputKernel a outputEvent
65         ∂Measure.map globalHistory (prior ⊗ₘ trajectory)
66     rw [MeasureTheory.setLIntegral_map hEnvironmentHistoryEvent
67       (outputKernel.measurable_coe hOutputEvent) hglobalHistory]
68   have hhistorySection (env : Env) : MeasurableSet
69       {h : History | (env, h) ∈ environmentHistoryEvent} :=
70     hEnvironmentHistoryEvent.preimage
71       (measurable_const.prodMk measurable_id)
72   have hsection (env : Env) : MeasurableSet
73       {omega : Omega | (env, history omega) ∈ environmentHistoryEvent} :=
74     (hhistorySection env).preimage hhistory
75   let integrand : Env × Omega -> ENNReal :=
76     fun sample => outputKernel (sample.1, history sample.2) outputEvent
77   have hintegrand : Measurable integrand :=
78     (outputKernel.measurable_coe hOutputEvent).comp
79       (measurable_fst.prodMk (hhistory.comp measurable_snd))
80   have hglobalHistoryEvent :
81       MeasurableSet (globalHistory ⁻¹' environmentHistoryEvent) :=
82     hglobalHistory hEnvironmentHistoryEvent
83   conv_rhs =>
84     change ∫⁻ sample in globalHistory ⁻¹' environmentHistoryEvent,
85       integrand sample ∂(prior ⊗ₘ trajectory)
86     rw [← MeasureTheory.lintegral_indicator hglobalHistoryEvent,
87       Measure.lintegral_compProd
88         (hintegrand.indicator hglobalHistoryEvent)]
89   rw [Measure.compProd_apply]
90   · apply lintegral_congr
91     intro env
92     have hpointwise :
93         trajectory env
94             (pairMap ⁻¹' ({h : History | (env, h) ∈ environmentHistoryEvent} ×ˢ
95               outputEvent)) =
96           ∫⁻ h in {h : History | (env, h) ∈ environmentHistoryEvent},
97             outputKernel (env, h) outputEvent ∂(trajectory env).map history := by
98       rw [← Measure.map_apply hpairMap
99           ((hhistorySection env).prod hOutputEvent),
100         hlaw env,
101         Measure.compProd_apply_prod (hhistorySection env) hOutputEvent]
102       simp only [Kernel.coe_comap, Function.comp_apply]
103     have hpreimage :
104         (Prod.mk env) ⁻¹'
105             (globalPairMap ⁻¹'
106               (environmentHistoryEvent ×ˢ outputEvent)) =
107           pairMap ⁻¹' ({h : History | (env, h) ∈ environmentHistoryEvent} ×ˢ
108             outputEvent) := by
109       ext omega
110       simp [pairMap, globalPairMap]
111     have hmap :
112         (∫⁻ h in {h : History | (env, h) ∈ environmentHistoryEvent},
113             outputKernel (env, h) outputEvent
114             ∂(trajectory env).map history) =
115           ∫⁻ omega in {omega : Omega |
116               (env, history omega) ∈ environmentHistoryEvent},
117             outputKernel (env, history omega) outputEvent
118             ∂trajectory env := by
119       simpa only [Function.comp_apply] using
120         (MeasureTheory.setLIntegral_map (hhistorySection env)
121           ((outputKernel.measurable_coe hOutputEvent).comp
122             (measurable_const.prodMk measurable_id)) hhistory
123           (μ := trajectory env))
124     rw [hpreimage, hpointwise, hmap]
125     conv_rhs =>
126       change ∫⁻ omega,
127         (globalHistory ⁻¹' environmentHistoryEvent).indicator integrand
128           (env, omega) ∂trajectory env
129       rw [show
130           (fun omega =>
131             (globalHistory ⁻¹' environmentHistoryEvent).indicator integrand
132               (env, omega)) =
133             {omega : Omega |
134               (env, history omega) ∈ environmentHistoryEvent}.indicator
135                 (fun omega =>
136                   outputKernel (env, history omega) outputEvent) by
137           funext omega
138           by_cases hmem :
139               (env, history omega) ∈ environmentHistoryEvent <;>
140             simp [globalHistory, integrand, Set.indicator, hmem]]
141       rw [MeasureTheory.lintegral_indicator (hsection env)]
142   · exact hglobalPairMap
143       (hEnvironmentHistoryEvent.prod hOutputEvent)
