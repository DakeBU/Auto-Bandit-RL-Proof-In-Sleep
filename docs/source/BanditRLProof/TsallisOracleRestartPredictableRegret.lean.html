5 # Predictable regret for generated oracle-restart trajectories
6 
7 This module puts the generated restart probability into the pointwise regret
8 summand and aligns the deterministic epoch assembly with `schedule.start`.
9 Epoch-local regret certificates remain explicit downstream inputs.
10 -/

21 against a fixed comparator distribution through the inclusive horizon. -/
22 noncomputable def sampledOracleRestartHalfTsallisPredictableEnvironmentRegret
23     {Env : Type u} {Action : Type v}

31     FTRL.linearLoss arms

86 /-- Restarted moving-comparator regret is fixed point-mass regret plus the
87 cumulative advantage of the moving comparator over the fixed arm. -/
88 theorem sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret_eq_fixed_add
89     {Env : Type u} {Action : Type v}
90     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
91     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
92     (schedule : OracleRestartSchedule)
93     (loss : Exp3.PredictableLossVector Env Action)
94     {best : Action} (hbest : best ∈ arms)
95     (comparator : Nat -> Action) (horizon : Nat)
96     (sample : Env × ((k : Nat) -> Action × Real)) :
97     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret
98         arms harms eta schedule loss comparator horizon sample =
99       sampledOracleRestartHalfTsallisPredictableEnvironmentRegret
100           arms harms eta schedule loss (pointMass best) horizon sample +
101         (Finset.range (horizon + 1)).sum (fun t =>
102           Exp3.predictableLossAt loss t sample best -
103             Exp3.predictableLossAt loss t sample (comparator t)) := by
104   unfold
105     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret
106     sampledOracleRestartHalfTsallisPredictableEnvironmentRegret
107   simp_rw [linearLoss_pointMass arms hbest]
108   rw [← Finset.sum_add_distrib]
109   apply Finset.sum_congr rfl
110   intro t _ht
111   ring

113 /-- Epoch ids actually visited by the restart schedule through the inclusive
114 horizon. The epoch assignment is exactly `schedule.start`. -/
115 def oracleRestartScheduleEpochs
116     (schedule : OracleRestartSchedule) (horizon : Nat) : Finset Nat :=
117   (Finset.range (horizon + 1)).image schedule.start

173 /-- Schedule-aligned epoch certificates assemble into a global square-root
174 bound for the generated restart regret surface. -/
175 theorem sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret_le_scheduleSqrt
176     {Env : Type u} {Action : Type v}
177     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
178     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
179     (schedule : OracleRestartSchedule)
180     (loss : Exp3.PredictableLossVector Env Action)
181     (epochComparator : Nat -> Action) (horizon : Nat)
182     (coefficient : Real) (hcoefficient : 0 <= coefficient)
183     (sample : Env × ((k : Nat) -> Action × Real))
184     (hEpochRegret :
185       ∀ epoch ∈ oracleRestartScheduleEpochs schedule horizon,
186         sampledOracleRestartHalfTsallisPredictableScheduleEpochRegret
187             arms harms eta schedule loss epochComparator horizon epoch sample <=
188           coefficient *
189             Real.sqrt
190               ((oracleRestartEpochRounds
191                 schedule.start horizon epoch).card : Real)) :
192     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret
193         arms harms eta schedule loss
194         (fun t => epochComparator (schedule.start t)) horizon sample <=
195       coefficient *
196           Real.sqrt ((oracleRestartScheduleEpochs schedule horizon).card : Real) *
197         Real.sqrt (((horizon + 1 : Nat) : Real)) := by
198   rw [
199     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret_eq_sum_scheduleEpochRegret
200       arms harms eta schedule loss epochComparator horizon sample]
201   calc
202     _ <= (oracleRestartScheduleEpochs schedule horizon).sum (fun epoch =>
203         coefficient *
204           Real.sqrt
205             ((oracleRestartEpochRounds
206               schedule.start horizon epoch).card : Real)) := by
207       exact Finset.sum_le_sum (fun epoch hepoch =>
208         hEpochRegret epoch hepoch)
209     _ = coefficient *
210         (oracleRestartScheduleEpochs schedule horizon).sum (fun epoch =>
211           Real.sqrt
212             ((oracleRestartEpochRounds
213               schedule.start horizon epoch).card : Real)) := by
214       rw [Finset.mul_sum]
215     _ <= coefficient *
216         (Real.sqrt
217             ((oracleRestartScheduleEpochs schedule horizon).card : Real) *
218           Real.sqrt (((horizon + 1 : Nat) : Real))) := by
219       exact mul_le_mul_of_nonneg_left
220         (sum_sqrt_oracleRestartEpochRounds_card_le
221           (oracleRestartScheduleEpochs schedule horizon)
222           schedule.start horizon
223           (fun t ht =>
224             oracleRestartSchedule_start_mem_epochs schedule horizon t ht))
225         hcoefficient
226     _ = coefficient *
227           Real.sqrt
228             ((oracleRestartScheduleEpochs schedule horizon).card : Real) *
229         Real.sqrt (((horizon + 1 : Nat) : Real)) := by ring

234     {Env : Type u} {Action : Type v}

241       (oracleRestartScheduleEpochs schedule horizon).card <= switches + 1)
242     (coefficient : Real) (hcoefficient : 0 <= coefficient)
243     (sample : Env × ((k : Nat) -> Action × Real))
244     (hEpochRegret :
245       ∀ epoch ∈ oracleRestartScheduleEpochs schedule horizon,
246         sampledOracleRestartHalfTsallisPredictableScheduleEpochRegret
247             arms harms eta schedule loss epochComparator horizon epoch sample <=
248           coefficient *
249             Real.sqrt
250               ((oracleRestartEpochRounds
251                 schedule.start horizon epoch).card : Real)) :
252     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret
253         arms harms eta schedule loss
254         (fun t => epochComparator (schedule.start t)) horizon sample <=
255       coefficient * Real.sqrt (((switches + 1 : Nat) : Real)) *
256         Real.sqrt (((horizon + 1 : Nat) : Real)) := by
257   have hrestart :=
258     sampledOracleRestartHalfTsallisPredictableMovingComparatorEnvironmentRegret_le_scheduleSqrt
259       arms harms eta schedule loss epochComparator horizon
260       coefficient hcoefficient sample hEpochRegret
261   have hcardReal :
262       ((oracleRestartScheduleEpochs schedule horizon).card : Real) <=

264     exact_mod_cast hEpochCard
