10 /-- History indices whose successor action is prescribed by the block-start schedule. -/
11 def blockStartForcedIndexSet (window horizon : Nat) : Finset Nat :=
12   (Finset.range horizon).filter (fun n => n % window = 0)
13 
14 /--
15 Every forced index below `horizon` belongs to the image of the quotient-block
16 range under `block ↦ block * window`, so its cardinality is at most the number
17 of quotient blocks. With `window = 0`, the modulo condition reduces to `n = 0`,
18 and the same conservative bound remains valid.
19 -/
20 theorem card_blockStartForcedIndexSet_le_div_add_one
21     (window horizon : Nat) :
22     (blockStartForcedIndexSet window horizon).card <=
23       horizon / window + 1 := by
24   let quotientMultiples :=
25     (Finset.range (horizon / window + 1)).image (fun block => block * window)
26   have hsubset :
27       blockStartForcedIndexSet window horizon ⊆ quotientMultiples := by
28     intro n hn
29     have hn_filter := Finset.mem_filter.mp hn
30     have hn_lt : n < horizon := Finset.mem_range.mp hn_filter.1
31     have hn_mod : n % window = 0 := hn_filter.2
32     apply Finset.mem_image.mpr
33     refine ⟨n / window, Finset.mem_range.mpr ?_, ?_⟩
34     · exact Nat.lt_succ_of_le (Nat.div_le_div_right (Nat.le_of_lt hn_lt))
35     · exact Nat.div_mul_cancel (Nat.dvd_of_mod_eq_zero hn_mod)
36   calc
37     (blockStartForcedIndexSet window horizon).card <=
38         quotientMultiples.card :=
39       Finset.card_le_card hsubset
40     _ <= (Finset.range (horizon / window + 1)).card := by
41       exact Finset.card_image_le
42     _ = horizon / window + 1 := Finset.card_range _

44 /-- A pointwise forced-arm gap ceiling bounds the forced charge by its cardinality. -/
45 theorem blockStartForcedActionSuccessorPseudoRegret_le_card_mul
46     {K : Nat} {Feature : Type u}
47     [Fintype Feature]
48     (thetaStar : Feature -> Real)
49     (actionFeature : Fin K -> Feature -> Real)
50     (best : Fin K)
51     (forcedAction : Nat -> Fin K)
52     (window horizon : Nat)
53     (forcedGapBound : Real)
54     (hforcedGap : forall block,
55       linearValue thetaStar (actionFeature best) -
56           linearValue thetaStar (actionFeature (forcedAction block)) <=
57         forcedGapBound) :
58     blockStartForcedActionSuccessorPseudoRegret
59         thetaStar actionFeature best forcedAction window horizon <=
60       ((blockStartForcedIndexSet window horizon).card : Real) *
61         forcedGapBound := by
62   have hsum :=
63     Finset.sum_le_card_nsmul
64       (blockStartForcedIndexSet window horizon)
65       (fun n =>
66         linearValue thetaStar (actionFeature best) -
67           linearValue thetaStar (actionFeature (forcedAction (n / window))))
68       forcedGapBound
69       (fun n _hn => hforcedGap (n / window))
70   simpa only [
71     blockStartForcedActionSuccessorPseudoRegret,
72     blockStartForcedIndexSet,
73     nsmul_eq_mul,
74     Nat.cast_ofNat] using hsum

82     (best : Fin K)
83     (forcedAction : Nat -> Fin K)
84     (window horizon : Nat)

89           linearValue thetaStar (actionFeature (forcedAction block)) <=
90         forcedGapBound) :
91     blockStartForcedActionSuccessorPseudoRegret
92         thetaStar actionFeature best forcedAction window horizon <=
93       ((horizon / window + 1 : Nat) : Real) * forcedGapBound := by
