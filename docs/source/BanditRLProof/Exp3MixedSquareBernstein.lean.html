5 /-!
6 # Variance-sensitive mixed-square EXP3 concentration
7 
8 The interval-sub-Gaussian route treats the mixed estimator square as an
9 arbitrary variable in `[0, 1 / epsilon]`, producing a variance proxy quadratic
10 in the reciprocal exploration floor.  Here the exact finite sampling law gives
11 a centered second moment at most `K / epsilon`.  A fixed-tilt conditional MGF
12 argument then yields a generated finite-horizon Bernstein tail whose square-root
13 term is linear, rather than quadratic, in that reciprocal floor.
14 -/

39 /-- The centered mixed-square score has second moment at most `K / epsilon`
40 under a uniform probability floor. -/
41 theorem sum_prob_mul_sq_mixedSquaredEstimatorDeviation_le_card_div_floor
42     {Action : Type u} [DecidableEq Action]
43     (arms : Finset Action) (prob loss : Action -> Real)
44     (hdist : FiniteActionDistribution arms prob)
45     (epsilon : Real) (hepsilon : 0 < epsilon)
46     (hfloor : forall action, action ∈ arms -> epsilon <= prob action)
47     (hloss : forall action, action ∈ arms ->
48       loss action ∈ Set.Icc (0 : Real) 1) :
49     arms.sum (fun chosen =>
50         prob chosen *
51           (mixedSquaredImportanceWeightedLoss arms prob loss chosen -
52             arms.sum (fun action => (loss action) ^ 2)) ^ 2) <=
53       (arms.card : Real) / epsilon := by
54   let raw := fun chosen =>
55     mixedSquaredImportanceWeightedLoss arms prob loss chosen
56   let mean := arms.sum (fun action => (loss action) ^ 2)
57   have hprob_pos (action : Action) (haction : action ∈ arms) :
58       0 < prob action := hepsilon.trans_le (hfloor action haction)
59   have hfirst : arms.sum (fun chosen => prob chosen * raw chosen) = mean := by
60     simpa [raw, mean] using
61       sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
62         arms prob loss (fun action haction => (hprob_pos action haction).ne')
63   have hsecond : arms.sum (fun chosen => prob chosen * (raw chosen) ^ 2) =
64       arms.sum (fun action => (loss action) ^ 4 / prob action) := by
65     simpa [raw] using
66       sum_prob_mul_sq_mixedSquaredImportanceWeightedLoss_eq
67         arms prob loss (fun action haction => (hprob_pos action haction).ne')
68   have hcenter (chosen : Action) :
69       prob chosen * (raw chosen - mean) ^ 2 =
70         (prob chosen * (raw chosen) ^ 2 -
71           2 * mean * (prob chosen * raw chosen)) +
72           mean ^ 2 * prob chosen := by
73     ring
74   have hvariance :
75       arms.sum (fun chosen => prob chosen * (raw chosen - mean) ^ 2) =
76         arms.sum (fun action => (loss action) ^ 4 / prob action) - mean ^ 2 := by
77     simp_rw [hcenter]
78     rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
79       ← Finset.mul_sum, ← Finset.mul_sum, hsecond, hfirst, hdist.sum_eq_one]
80     ring
81   have hfourth (action : Action) (haction : action ∈ arms) :
82       (loss action) ^ 4 <= 1 := by
83     have hsq : (loss action) ^ 2 <= 1 := by
84       simpa using (sq_le_sq₀ (hloss action haction).1 zero_le_one).2
85         (hloss action haction).2
86     have hsq_nonneg : 0 <= (loss action) ^ 2 := sq_nonneg _
87     nlinarith [sq_nonneg ((loss action) ^ 2)]
88   have hterm (action : Action) (haction : action ∈ arms) :
89       (loss action) ^ 4 / prob action <= 1 / epsilon := by
90     calc
91       (loss action) ^ 4 / prob action <= 1 / prob action :=
92         div_le_div_of_nonneg_right (hfourth action haction)
93           (hprob_pos action haction).le
94       _ <= 1 / epsilon := one_div_le_one_div_of_le hepsilon
95         (hfloor action haction)
96   have hsecond_le :
97       arms.sum (fun action => (loss action) ^ 4 / prob action) <=
98         (arms.card : Real) / epsilon := by
99     calc
100       arms.sum (fun action => (loss action) ^ 4 / prob action) <=
101           arms.sum (fun _action => 1 / epsilon) :=
102         Finset.sum_le_sum fun action haction => hterm action haction
103       _ = (arms.card : Real) / epsilon := by simp [div_eq_mul_inv]
104   change arms.sum (fun chosen => prob chosen * (raw chosen - mean) ^ 2) <= _
105   rw [hvariance]
106   nlinarith [sq_nonneg mean]

108 /-- Fixed-tilt MGF budget for a centered mixed estimator square under a finite
109 sampling law. The quadratic coefficient is `K / epsilon`, while the admissible
110 tilt is controlled by the sharper range cap `epsilon`. -/

116     (epsilon : Real) (hepsilon : 0 < epsilon)

118     (hloss : forall action, action ∈ arms ->
119       loss action ∈ Set.Icc (0 : Real) 1)
120     (tilt : Real) (htilt_nonneg : 0 <= tilt) (htilt_le : tilt <= epsilon) :
121     Concentration.HasMGFUpperBoundAt
122       (fun chosen =>
123         mixedSquaredImportanceWeightedLoss arms prob loss chosen -
124           arms.sum (fun action => (loss action) ^ 2))
125       tilt (tilt ^ 2 * ((arms.card : Real) / epsilon))
126       (finiteActionMeasure arms prob) := by

128     mixedSquaredImportanceWeightedLoss arms prob loss chosen
129   let mean := arms.sum (fun action => (loss action) ^ 2)
130   let X := fun chosen => raw chosen - mean

132     finiteActionMeasure_isProbabilityMeasure arms prob hdist

134       0 < prob action := hepsilon.trans_le (hfloor action haction)
135   have hinv_nonneg : 0 <= 1 / epsilon := (one_div_pos.mpr hepsilon).le
136   have hmean_nonneg : 0 <= mean := by
137     exact Finset.sum_nonneg fun action _haction => sq_nonneg (loss action)
138   have hmean_le_card : mean <= (arms.card : Real) := by
139     dsimp only [mean]
140     calc
141       arms.sum (fun action => (loss action) ^ 2) <=
142           arms.sum (fun _action => (1 : Real)) := by

144         intro action haction
145         simpa using (sq_le_sq₀ (hloss action haction).1 zero_le_one).2
146           (hloss action haction).2
147       _ = (arms.card : Real) := by simp
148   have hcard_le_inv : (arms.card : Real) <= 1 / epsilon := by
149     rw [le_div_iff₀ hepsilon]
150     have hsum : arms.sum (fun _action => epsilon) <= arms.sum prob := by

153   have hX_bound (chosen : Action) (hchosen : chosen ∈ arms) :
154       |X chosen| <= 1 / epsilon := by
155     have hraw_nonneg : 0 <= raw chosen := by
156       unfold raw mixedSquaredImportanceWeightedLoss
157       exact Finset.sum_nonneg fun action haction =>
158         mul_nonneg (hdist.nonneg action haction) (sq_nonneg _)
159     have hraw_le : raw chosen <= 1 / epsilon := by
160       dsimp only [raw]
161       rw [mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
162         arms prob loss chosen hchosen (hprob_pos chosen hchosen).ne']
163       have hsq : (loss chosen) ^ 2 <= 1 := by
164         simpa using (sq_le_sq₀ (hloss chosen hchosen).1 zero_le_one).2
165           (hloss chosen hchosen).2
166       calc
167         (loss chosen) ^ 2 / prob chosen <= 1 / prob chosen :=
168           div_le_div_of_nonneg_right hsq (hprob_pos chosen hchosen).le
169         _ <= 1 / epsilon := one_div_le_one_div_of_le hepsilon
170           (hfloor chosen hchosen)
171     dsimp only [X]
172     rw [abs_le]
173     constructor
174     · nlinarith [hmean_le_card, hcard_le_inv]
175     · nlinarith
176   constructor
177   · intro s
178     rw [finiteActionMeasure]
179     refine integrable_finset_sum_measure.2 fun action haction => ?_
180     exact (integrable_dirac (by simp)).smul_measure ENNReal.ofReal_ne_top
181   · rw [ProbabilityTheory.mgf, integral_finiteActionMeasure_eq_sum arms prob hdist]
182     have hexp (chosen : Action) (hchosen : chosen ∈ arms) :
183         Real.exp (tilt * X chosen) <=
184           1 + tilt * X chosen + (tilt * X chosen) ^ 2 := by
185       apply Concentration.exp_le_one_add_self_add_sq_of_abs_le_one
186       rw [abs_mul, abs_of_nonneg htilt_nonneg]
187       calc
188         tilt * |X chosen| <= tilt * (1 / epsilon) :=
189           mul_le_mul_of_nonneg_left (hX_bound chosen hchosen) htilt_nonneg
190         _ <= epsilon * (1 / epsilon) :=
191           mul_le_mul_of_nonneg_right htilt_le hinv_nonneg
192         _ = 1 := by field_simp
193     calc
194       arms.sum (fun chosen => prob chosen * Real.exp (tilt * X chosen)) <=
195           arms.sum (fun chosen =>
196             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2)) := by
197         apply Finset.sum_le_sum
198         intro chosen hchosen
199         exact mul_le_mul_of_nonneg_left (hexp chosen hchosen)
200           (hdist.nonneg chosen hchosen)
201       _ = 1 + tilt * arms.sum (fun chosen => prob chosen * X chosen) +
202           tilt ^ 2 * arms.sum (fun chosen => prob chosen * (X chosen) ^ 2) := by
203         have hterm (chosen : Action) :
204             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2) =
205               (prob chosen + tilt * (prob chosen * X chosen)) +
206                 tilt ^ 2 * (prob chosen * (X chosen) ^ 2) := by ring
207         simp_rw [hterm, Finset.sum_add_distrib, ← Finset.mul_sum, hdist.sum_eq_one]
208       _ <= 1 + tilt ^ 2 * ((arms.card : Real) / epsilon) := by
209         have hmean : arms.sum (fun chosen => prob chosen * X chosen) = 0 := by
210           dsimp only [X]
211           simp_rw [mul_sub]
212           rw [Finset.sum_sub_distrib,
213             sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
214               arms prob loss
215                 (fun action haction => (hprob_pos action haction).ne'),
216             ← Finset.sum_mul, hdist.sum_eq_one]
217           simp [mean]
218         rw [hmean, mul_zero, add_zero]
219         have hsecond := mul_le_mul_of_nonneg_left
220           (by
221             simpa [X, mean] using
222               sum_prob_mul_sq_mixedSquaredEstimatorDeviation_le_card_div_floor
223                 arms prob loss hdist epsilon hepsilon hfloor hloss)
224           (sq_nonneg tilt)
225         simpa [X, add_comm] using add_le_add_left hsecond 1
226       _ <= Real.exp (tilt ^ 2 * ((arms.card : Real) / epsilon)) := by
227         simpa [add_comm] using
228           Real.add_one_le_exp (tilt ^ 2 * ((arms.card : Real) / epsilon))
