3 /-!
4 # Pathwise Hedge control for the concrete sampled EXP3 score
5 
6 This module identifies the recursively accumulated `sampledHistoryScore` with
7 the deterministic cumulative-loss surface used by `Exp3HedgeRegret`.  It also
8 identifies the corresponding pure exponential-weights distribution and the
9 exploration-mixed trajectory probability.  The final theorem specializes the
10 deterministic second-order Hedge bound to one concrete sampled trajectory.
11 
12 The result is pathwise.  Its scalar-feedback nonnegativity premise is intended
13 to be discharged almost surely by the predictable `[0,1]` reward law before
14 integrating the bound.
15 -/

22 /-- The complete importance-weighted loss vector observed at an actual time. -/
23 noncomputable def sampledTrajectoryObservedLoss
24     {Env : Type u} {Action : Type v} [DecidableEq Action]
25     (arms : Finset Action) (eta gamma : Real)
26     (sample : Env × ((k : Nat) -> Action × Real)) :
27     Nat -> Action -> Real :=
28   fun t action =>
29     observedImportanceWeightedLossAt arms eta gamma t sample action

40 /-- The inclusive sampled score through `n` is Hedge cumulative loss at `n+1`. -/
41 theorem sampledHistoryScore_frestrictLe_eq_cumulativeLoss
42     {Env : Type u} {Action : Type v} [DecidableEq Action]
43     (arms : Finset Action) (eta gamma : Real)
44     (sample : Env × ((k : Nat) -> Action × Real))
45     (n : Nat) (action : Action) :
46     sampledHistoryScore arms eta gamma n
47         (Preorder.frestrictLe n sample.2) action =
48       cumulativeLoss (sampledTrajectoryObservedLoss arms eta gamma sample)
49         (n + 1) action := by
50   induction n with
51   | zero =>
52       simp [sampledTrajectoryObservedLoss, cumulativeLoss,
53         observedImportanceWeightedLossAt, sampledTrajectoryProbabilityAt]
54   | succ n ih =>
55       rw [sampledHistoryScore_succ, previousPairHistory_frestrictLe, ih]
56       rw [show cumulativeLoss
57             (sampledTrajectoryObservedLoss arms eta gamma sample) (n + 1 + 1) action =
58           cumulativeLoss (sampledTrajectoryObservedLoss arms eta gamma sample)
59               (n + 1) action +
60             sampledTrajectoryObservedLoss arms eta gamma sample (n + 1) action by
61         exact cumulativeLoss_succ
62           (sampledTrajectoryObservedLoss arms eta gamma sample) (n + 1) action]
63       rfl

65 /-- At a successor time, the Hedge distribution is the normalized sampled score. -/
66 theorem distribution_sampledTrajectoryObservedLoss_succ
67     {Env : Type u} {Action : Type v} [DecidableEq Action]
68     (arms : Finset Action) (eta gamma : Real)
69     (sample : Env × ((k : Nat) -> Action × Real))
70     (n : Nat) (action : Action) :
71     distribution arms eta (sampledTrajectoryObservedLoss arms eta gamma sample)
72         (n + 1) action =
73       normalizedHistoryDistribution arms eta
74         (sampledHistoryScore arms eta gamma n)
75         (Preorder.frestrictLe n sample.2) action := by
76   unfold distribution weight totalWeight normalizedHistoryDistribution
77     historyWeight historyTotalWeight
78   rw [sampledHistoryScore_frestrictLe_eq_cumulativeLoss]
79   congr 1
80   apply Finset.sum_congr rfl
81   intro candidate _hcandidate
82   unfold weight historyWeight
83   rw [sampledHistoryScore_frestrictLe_eq_cumulativeLoss]

85 /-- The concrete sampling law is uniform exploration mixed with the Hedge law. -/
86 theorem sampledTrajectoryProbabilityAt_eq_mix_distribution
87     {Env : Type u} {Action : Type v} [DecidableEq Action]
88     (arms : Finset Action) (eta gamma : Real) (t : Nat)
89     (sample : Env × ((k : Nat) -> Action × Real)) (action : Action) :
90     sampledTrajectoryProbabilityAt arms eta gamma t sample action =
91       (1 - gamma) *
92           distribution arms eta
93             (sampledTrajectoryObservedLoss arms eta gamma sample) t action +
94         gamma / (arms.card : Real) := by
95   cases t with
96   | zero =>
97       rfl
98   | succ n =>
99       rw [sampledTrajectoryProbabilityAt]
100       unfold sampledHistoryDistribution exploredHistoryDistribution
101       rw [← distribution_sampledTrajectoryObservedLoss_succ]

103 theorem sampledTrajectoryProbabilityAt_nonneg
104     {Env : Type u} {Action : Type v} [DecidableEq Action]
105     (arms : Finset Action) (harms : arms.Nonempty)
106     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
107     (hgamma_le_one : gamma <= 1) (t : Nat)
108     (sample : Env × ((k : Nat) -> Action × Real)) (action : Action) :
109     0 <= sampledTrajectoryProbabilityAt arms eta gamma t sample action := by
110   cases t with
111   | zero =>
112       simpa [sampledTrajectoryProbabilityAt, initialExploredDistribution] using
113         (exploredHistoryDistribution_nonneg arms harms eta gamma

115   | succ n =>
116       simpa [sampledTrajectoryProbabilityAt, sampledHistoryDistribution] using
117         (exploredHistoryDistribution_nonneg arms harms eta gamma

134 /-- Concrete finite-horizon sampled-trajectory specialization of Hedge. -/
135 theorem sampledTrajectory_hedge_regret_le
136     {Env : Type u} {Action : Type v} [DecidableEq Action]
137     (arms : Finset Action) (harms : arms.Nonempty)
138     (eta gamma : Real) (heta : 0 < eta)
139     (hgamma_nonneg : 0 <= gamma) (hgamma_le_one : gamma <= 1)
140     (sample : Env × ((k : Nat) -> Action × Real)) (horizon : Nat)
141     (hreward_nonneg : forall t, t < horizon -> 0 <= (sample.2 t).2)
142     (comparator : Action) (hcomparator : comparator ∈ arms) :
143     (Finset.range horizon).sum (fun t =>
144         mixedLoss arms eta
145           (sampledTrajectoryObservedLoss arms eta gamma sample) t) -
146         cumulativeLoss (sampledTrajectoryObservedLoss arms eta gamma sample)
147           horizon comparator <=
148       Real.log arms.card / eta +
149         eta * (Finset.range horizon).sum (fun t =>
150           mixedSquaredLoss arms eta
151             (sampledTrajectoryObservedLoss arms eta gamma sample) t) := by
152   apply hedge_regret_le_log_card_div_add_eta_mul_mixedSquaredLoss_of_nonneg
153     arms harms eta heta
154       (sampledTrajectoryObservedLoss arms eta gamma sample) horizon
155   · intro t ht action _haction
156     exact sampledTrajectoryObservedLoss_nonneg arms harms eta gamma
157       hgamma_nonneg hgamma_le_one sample t action (hreward_nonneg t ht)
158   · exact hcomparator
