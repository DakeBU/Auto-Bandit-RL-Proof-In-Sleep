18 /--
19 The confidence logarithm for outer failure probability `delta` over the
20 complete finite window `0, ..., horizon`.
21 -/
22 noncomputable def standardHighProbabilityRegretLogBudget
23     {Feature : Type u} [Fintype Feature]
24     (lambda delta : Real) (horizon : Nat) (L2 : Real) : Real :=
25   standardScalarLogDetBudget
26       (Feature := Feature) lambda (horizon + 1) L2 +
27     2 * Real.log ((((horizon + 1 : Nat) : Real)) / delta)

29 /--
30 At algorithm parameter `delta / (T+1)`, the standard confidence radius has
31 the explicit form
32 `R * sqrt (B_T + 2 * log ((T+1)/delta)) + sqrt lambda * S`.
33 -/
34 theorem standardScalarConfidenceRadiusUpper_highProbabilityRegret
35     {Feature : Type u}
36     [Fintype Feature] [Nonempty Feature]
37     (R : Real) (hR : 0 < R)
38     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1)
39     (lambda : Real) (hlambda : 0 < lambda)
40     (S : Real)
41     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2) :
42     standardScalarConfidenceRadiusUpper
43         (Feature := Feature) R
44         (delta / (((horizon + 1 : Nat) : Real)))
45         lambda S (horizon + 1) L2 =
46       R * Real.sqrt
47           (standardHighProbabilityRegretLogBudget
48             (Feature := Feature) lambda delta horizon L2) +
49         Real.sqrt lambda * S := by
50   let n : Real := (((horizon + 1 : Nat) : Real))
51   let B : Real :=
52     standardScalarLogDetBudget
53       (Feature := Feature) lambda (horizon + 1) L2
54   have hn_pos : 0 < n := by
55     dsimp [n]
56     positivity
57   have hn_one : 1 <= n := by
58     dsimp [n]
59     exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
60   have hB : 0 <= B := by
61     exact standardScalarLogDetBudget_nonneg
62       (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
63   have hscale_pos : 0 < n / delta := div_pos hn_pos hdelta
64   have hscale_one : 1 <= n / delta := by
65     rw [le_div_iff₀ hdelta]
66     simpa using hdelta_one.trans hn_one
67   have hlogscale : 0 <= Real.log (n / delta) :=
68     Real.log_nonneg hscale_one
69   have hX : 0 <= B + 2 * Real.log (n / delta) := by positivity
70   have hquot :
71       Real.sqrt (Real.exp B) / (delta / n) =
72         Real.sqrt (Real.exp B) * (n / delta) := by
73     field_simp
74   have hlog :
75       Real.log (Real.sqrt (Real.exp B) / (delta / n)) =
76         B / 2 + Real.log (n / delta) := by
77     rw [hquot]
78     rw [Real.log_mul (Real.sqrt_pos.2 (Real.exp_pos B)).ne'
79       hscale_pos.ne']
80     rw [Real.log_sqrt (Real.exp_pos B).le, Real.log_exp]
81   unfold standardScalarConfidenceRadiusUpper
82     standardHighProbabilityRegretLogBudget
83   change
84     Real.sqrt
85           (2 * R ^ 2 *
86             Real.log (Real.sqrt (Real.exp B) / (delta / n))) +
87         Real.sqrt lambda * S =
88       R * Real.sqrt (B + 2 * Real.log (n / delta)) +
89         Real.sqrt lambda * S
90   rw [hlog]
91   congr 1
92   have hrad :
93       2 * R ^ 2 * (B / 2 + Real.log (n / delta)) =
94         R ^ 2 * (B + 2 * Real.log (n / delta)) := by ring
95   rw [hrad, Real.sqrt_mul (sq_nonneg R), Real.sqrt_sq hR.le]

98 noncomputable def standardHighProbabilityPseudoRegretBound
99     {Feature : Type u} [Fintype Feature]
100     (R delta lambda S : Real) (horizon : Nat) (L2 : Real) : Real :=
101   2 * S * Real.sqrt L2 +
102     2 *
103       (R * Real.sqrt
104           (standardHighProbabilityRegretLogBudget
105             (Feature := Feature) lambda delta horizon L2) +
106         Real.sqrt lambda * S) *
107       (Real.sqrt (((horizon + 1 : Nat) : Real)) *
108         Real.sqrt
109           (2 *

111               (Feature := Feature) lambda (horizon + 1) L2))
