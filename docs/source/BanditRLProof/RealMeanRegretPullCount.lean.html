6 /-!
7 # Real mean-regret pull-count decomposition
8 
9 This module provides the Real-valued finite-arm bookkeeping surface needed by
10 the exact LML ETC route.  It is parameterized by an arm-mean function, so a
11 later kernel bridge can instantiate `mean a` with the integral of the identity
12 under arm `a` without changing the deterministic or Bochner proofs here.
13 -/
14 
15 namespace BanditRLProof
16 
17 open MeasureTheory
18 
19 universe u
20 
21 /-- Gap from the supremum finite-arm mean, matching the scalar semantics of LML's bandit gap. -/
22 noncomputable def realMeanGap {K : Nat} (mean : Fin K -> Real) (a : Fin K) : Real :=
23   (⨆ b : Fin K, mean b) - mean a
24 
25 /-- Real pseudo-regret written directly from arm means over a finite horizon. -/
26 noncomputable def realMeanRegret {K : Nat}
27     (mean : Fin K -> Real) (action : ActionTrace (Fin K)) (n : Nat) : Real :=
28   (n : Real) * (⨆ a : Fin K, mean a) -
29     (Finset.range n).sum (fun t => mean (action t))
30 
31 /-- Real mean regret is the time-indexed finite sum of selected arm gaps. -/
32 theorem realMeanRegret_eq_finset_sum_gap
33     {K : Nat}
34     (mean : Fin K -> Real)
35     (action : ActionTrace (Fin K))
36     (n : Nat) :
37     realMeanRegret mean action n =
38       (Finset.range n).sum (fun t => realMeanGap mean (action t)) := by
39   simp [realMeanRegret, realMeanGap, Finset.sum_sub_distrib]
40 
41 /--
42 Real mean regret decomposes into each arm gap times its finite-horizon pull count.
43 
44 This is the deterministic half of `REAL-MEAN-REGRET-PULLCOUNT`.  The definition
45 uses the same supremum-minus-mean gap as the exact LML theorem card; no rational
46 model, kernel law, measurability, or concentration assumption appears here.
47 -/
48 theorem realMeanRegret_eq_sum_gap_mul_pullCount
49     {K : Nat}
50     (mean : Fin K -> Real)
51     (action : ActionTrace (Fin K))
52     (n : Nat) :
53     realMeanRegret mean action n =
54       (Finset.univ : Finset (Fin K)).sum
55         (fun a => realMeanGap mean a * (pullCount action a n : Real)) := by
56   rw [realMeanRegret_eq_finset_sum_gap]
57   rw [(Finset.sum_fiberwise'
58     (s := Finset.range n)
59     (g := action)
60     (f := fun a : Fin K => realMeanGap mean a)).symm]
61   apply Finset.sum_congr rfl
62   intro a _ha
63   rw [Finset.sum_const]
64   have hcount :=
65     pullCount_eq_finset_filter_card (action := action) (a := a) (t := n)
66   rw [hcount.symm]
67   exact nsmul_eq_mul' (realMeanGap mean a) (pullCount action a n)
68 
69 /-- Pull-count integrability implies integrability of Real mean regret. -/
70 theorem integrable_realMeanRegret_of_integrable_pullCount
71     {Omega : Type u} {K : Nat}
72     [MeasurableSpace Omega]
73     (mu : Measure Omega)
74     (mean : Fin K -> Real)
75     (action : Omega -> ActionTrace (Fin K))
76     (n : Nat)
77     (hcount : forall a : Fin K,
78       Integrable (fun omega => (pullCount (action omega) a n : Real)) mu) :
79     Integrable (fun omega => realMeanRegret mean (action omega) n) mu := by
80   let term : Fin K -> Omega -> Real :=
81     fun a omega => realMeanGap mean a * (pullCount (action omega) a n : Real)
82   have hterm : forall a : Fin K, Integrable (term a) mu := by
83     intro a
84     exact (hcount a).const_mul (realMeanGap mean a)
85   have hsum :
86       Integrable
87         (fun omega =>
88           (Finset.univ : Finset (Fin K)).sum (fun a => term a omega)) mu :=
89     IntegrabilitySums.integrable_univ_sum mu term hterm
90   have hfun :
91       (fun omega => realMeanRegret mean (action omega) n) =
92         (fun omega =>
93           (Finset.univ : Finset (Fin K)).sum (fun a => term a omega)) := by
94     funext omega
95     exact realMeanRegret_eq_sum_gap_mul_pullCount mean (action omega) n
96   rwa [hfun]
97 
98 /--
99 The Bochner expectation of Real mean regret is the gap-weighted sum of expected
100 pull counts.
101 
102 This is the expectation half of `REAL-MEAN-REGRET-PULLCOUNT`.  Callers supply
103 only pull-count integrability; probability-space, policy, reward-law, and
104 sub-Gaussian contracts remain outside this bookkeeping leaf.
105 -/
106 theorem integral_realMeanRegret_eq_sum_gap_mul_integral_pullCount
107     {Omega : Type u} {K : Nat}
108     [MeasurableSpace Omega]
109     (mu : Measure Omega)
110     (mean : Fin K -> Real)
111     (action : Omega -> ActionTrace (Fin K))
112     (n : Nat)
113     (hcount : forall a : Fin K,
114       Integrable (fun omega => (pullCount (action omega) a n : Real)) mu) :
115     integral mu (fun omega => realMeanRegret mean (action omega) n) =
116       (Finset.univ : Finset (Fin K)).sum
117         (fun a =>
118           realMeanGap mean a *
119             integral mu (fun omega => (pullCount (action omega) a n : Real))) := by
120   let term : Fin K -> Omega -> Real :=
121     fun a omega => realMeanGap mean a * (pullCount (action omega) a n : Real)
122   have hterm : forall a : Fin K, Integrable (term a) mu := by
123     intro a
124     exact (hcount a).const_mul (realMeanGap mean a)
125   have hfun :
126       (fun omega => realMeanRegret mean (action omega) n) =
127         (fun omega =>
128           (Finset.univ : Finset (Fin K)).sum (fun a => term a omega)) := by
129     funext omega
130     exact realMeanRegret_eq_sum_gap_mul_pullCount mean (action omega) n
131   calc
132     integral mu (fun omega => realMeanRegret mean (action omega) n) =
133         integral mu
134           (fun omega =>
135             (Finset.univ : Finset (Fin K)).sum (fun a => term a omega)) := by
136       rw [hfun]
137     _ = (Finset.univ : Finset (Fin K)).sum
138           (fun a => integral mu (term a)) :=
139       ExpectationBochnerSums.integral_univ_sum mu term hterm
140     _ = (Finset.univ : Finset (Fin K)).sum
141           (fun a =>
142             realMeanGap mean a *
143               integral mu (fun omega => (pullCount (action omega) a n : Real))) := by
144       apply Finset.sum_congr rfl
145       intro a _ha
146       simp [term, MeasureTheory.integral_const_mul]
