3 /-!
4 # Path-variation dynamic regret for independent nonidentical reward laws
5 
6 This module derives the mean-deviation envelope used by the compiled
7 drifting-mean dynamic-regret theorem from the actual reward-mean path. The
8 only model-alignment premise is equality of the actual and baseline means at
9 round zero.
10 -/
11

13 open MeasureTheory ProbabilityTheory Set Filter Finset

18 /-- Cumulative absolute variation of one arm's actual reward mean before
19 round `t`. -/
20 noncomputable def finiteArmIndependentCumulativeMeanPathVariation {K : Nat}
21     (armLaw : Nat -> Fin K -> Measure Rat) (t : Nat) (arm : Fin K) : Real :=
22   (Finset.range t).sum (fun s =>
23     |finiteArmIndependentRewardMean armLaw (s + 1) arm -
24       finiteArmIndependentRewardMean armLaw s arm|)
25 
26 @[simp]
27 theorem finiteArmIndependentCumulativeMeanPathVariation_zero {K : Nat}
28     (armLaw : Nat -> Fin K -> Measure Rat) (arm : Fin K) :
29     finiteArmIndependentCumulativeMeanPathVariation armLaw 0 arm = 0 := by
30   simp [finiteArmIndependentCumulativeMeanPathVariation]
31 
32 theorem abs_finiteArmIndependentRewardMean_sub_zero_le_cumulativeMeanPathVariation
33     {K : Nat} (armLaw : Nat -> Fin K -> Measure Rat)
34     (t : Nat) (arm : Fin K) :
35     |finiteArmIndependentRewardMean armLaw t arm -
36         finiteArmIndependentRewardMean armLaw 0 arm| <=
37       finiteArmIndependentCumulativeMeanPathVariation armLaw t arm := by
38   induction t with
39   | zero =>
40       simp [finiteArmIndependentCumulativeMeanPathVariation]
41   | succ t ih =>
42       rw [finiteArmIndependentCumulativeMeanPathVariation,
43         Finset.sum_range_succ]
44       calc
45         |finiteArmIndependentRewardMean armLaw (t + 1) arm -
46             finiteArmIndependentRewardMean armLaw 0 arm| =
47             |(finiteArmIndependentRewardMean armLaw (t + 1) arm -
48                 finiteArmIndependentRewardMean armLaw t arm) +
49               (finiteArmIndependentRewardMean armLaw t arm -
50                 finiteArmIndependentRewardMean armLaw 0 arm)| := by
51           congr 1
52           ring
53         _ <=
54             |finiteArmIndependentRewardMean armLaw (t + 1) arm -
55                 finiteArmIndependentRewardMean armLaw t arm| +
56               |finiteArmIndependentRewardMean armLaw t arm -
57                 finiteArmIndependentRewardMean armLaw 0 arm| :=
58           abs_add_le _ _
59         _ <=
60             |finiteArmIndependentRewardMean armLaw (t + 1) arm -
61                 finiteArmIndependentRewardMean armLaw t arm| +
62               finiteArmIndependentCumulativeMeanPathVariation armLaw t arm :=
63           add_le_add (le_refl _) ih
64         _ =
65             finiteArmIndependentCumulativeMeanPathVariation armLaw t arm +
66               |finiteArmIndependentRewardMean armLaw (t + 1) arm -
67                 finiteArmIndependentRewardMean armLaw t arm| := by
68           ring

70 /-- Initial mean matching turns cumulative actual-mean path variation into the
71 all-time model-deviation envelope required by the drifting-mean theorem. -/
72 theorem abs_finiteArmIndependentRewardMean_sub_model_le_cumulativeMeanPathVariation
73     {K : Nat} (model : FiniteBanditModel K)
74     (armLaw : Nat -> Fin K -> Measure Rat)
75     (hinitialMean : forall arm,
76       finiteArmIndependentRewardMean armLaw 0 arm =
77         ((model.mean arm : Rat) : Real))
78     (t : Nat) (arm : Fin K) :
79     |finiteArmIndependentRewardMean armLaw t arm -
80         ((model.mean arm : Rat) : Real)| <=
81       finiteArmIndependentCumulativeMeanPathVariation armLaw t arm := by
82   rw [show ((model.mean arm : Rat) : Real) =
83       finiteArmIndependentRewardMean armLaw 0 arm by
84     exact (hinitialMean arm).symm]
85   exact
86     abs_finiteArmIndependentRewardMean_sub_zero_le_cumulativeMeanPathVariation
87       armLaw t arm

89 /-- The compiled dynamic all-regimes bound specialized to the law-derived
90 cumulative population-mean path-variation envelope. -/
91 noncomputable def finiteArmIndependentPathVariationDynamicAllRegimeBound
92     {K : Nat} (model : FiniteBanditModel K)
93     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat) : Real :=
94   finiteArmIndependentDriftingMeanDynamicAllRegimeBound
95     model armLaw horizon
96       (finiteArmIndependentCumulativeMeanPathVariation armLaw)

107 /-- Generated expected predictable-environment dynamic regret with the
108 all-time deviation envelope derived from the reward laws' population-mean
109 path variation. No caller supplies a deviation envelope or the moving
110 comparator. This retains one cumulative prefix envelope at every included
111 time; it is not a horizon-compressed or minimax-sharp standard `V_T` bound. -/
112 theorem integral_sampledScheduledHalfTsallisFiniteArmIndependentPathVariationDynamicRegret_le_allRegimes
113     {K : Nat} (model : FiniteBanditModel K)
114     (armLaw : Nat -> Fin K -> Measure Rat)
115     (hprob : forall t arm, IsProbabilityMeasure (armLaw t arm))
116     (hbound : forall t arm, ∀ᵐ reward ∂armLaw t arm,
117       ((reward : Rat) : Real) ∈ Set.Icc (0 : Real) 1)
118     (hinitialMean : forall arm,
119       finiteArmIndependentRewardMean armLaw 0 arm =
120         ((model.mean arm : Rat) : Real))
121     (hgapPos : forall arm, arm ≠ model.bestArm ->
122       0 < ((model.gap arm : Rat) : Real))
123     (hgapLeOne : forall arm, arm ≠ model.bestArm ->
124       ((model.gap arm : Rat) : Real) <= 1)
125     (horizon : Nat) :
126     letI : Nonempty (Fin K) := ⟨model.bestArm⟩
127     let law := finiteArmIndependentRewardVectorLaw armLaw
128     let value := fun _ : Nat => finiteArmIIDRewardVectorLoss
129     let loss := iidTimeVaryingLossStatePredictableLossVector value
130       (fun _ => measurable_finiteArmIIDRewardVectorLoss)
131       (fun _ => finiteArmIIDRewardVectorLoss_nonneg)
132       (fun _ => finiteArmIIDRewardVectorLoss_le_one)
133     let selector :=
134       canonicalHalfTsallisScheduleGeneratedSelectorMeasurability
135         (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
136         sampledScheduledHalfTsallisSqrtSchedule loss
137     let prior := Measure.infinitePi law
138     let mu := prior.compProd
139       (sampledScheduledHalfTsallisTrajectoryKernel
140         (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
141         sampledScheduledHalfTsallisSqrtSchedule
142         selector.finiteHistory loss.environment)
143     integral mu
144         (sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret
145           (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
146           sampledScheduledHalfTsallisSqrtSchedule loss
147           (finiteArmIndependentBestArmAt model armLaw) horizon) <=
148       finiteArmIndependentPathVariationDynamicAllRegimeBound
149         model armLaw horizon := by
150   simpa only [finiteArmIndependentPathVariationDynamicAllRegimeBound] using
151     integral_sampledScheduledHalfTsallisFiniteArmIndependentDriftingMeanDynamicRegret_le_allRegimes
152       model armLaw hprob hbound
153       (finiteArmIndependentCumulativeMeanPathVariation armLaw)
154       (abs_finiteArmIndependentRewardMean_sub_model_le_cumulativeMeanPathVariation
155         model armLaw hinitialMean)
156       hgapPos hgapLeOne horizon
