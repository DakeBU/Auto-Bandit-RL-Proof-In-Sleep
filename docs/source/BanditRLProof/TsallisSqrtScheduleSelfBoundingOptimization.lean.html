4 /-!
5 # Square-root schedule self-bounding optimization
6 
7 This module specializes the refined generated quadratic split to the local
8 schedule `eta_t = 1 / (2 * sqrt (t + 1))`. The refined coefficient is replaced
9 by its compiled `5 / sqrt (t + 1)` envelope and the deterministic rate-square
10 base is identified with one half of the harmonic budget.

20 /-- Integral-comparison bound for the shifted inverse-square-root prefix. -/
21 theorem sum_range_one_div_sqrt_natSucc_le_two_sqrt (n : Nat) :
22     (Finset.range n).sum (fun t =>
23         1 / Real.sqrt (((t + 1 : Nat) : Real))) ≤
24       2 * Real.sqrt (n : Real) := by
25   induction n with
26   | zero => simp
27   | succ n ih =>
28       rw [Finset.sum_range_succ]
29       have hincrement :
30           1 / Real.sqrt (((n + 1 : Nat) : Real)) ≤
31             2 * (Real.sqrt ((n + 1 : Nat) : Real) -
32               Real.sqrt (n : Real)) := by
33         by_cases hn : n = 0
34         · subst n
35           norm_num
36         · have hsqrtSuccPos :
37               0 < Real.sqrt (((n + 1 : Nat) : Real)) := by positivity
38           have hsqrtSucc :
39               Real.sqrt (((n + 1 : Nat) : Real)) *
40                   Real.sqrt (((n + 1 : Nat) : Real)) =
41                 ((n + 1 : Nat) : Real) :=
42             Real.mul_self_sqrt (by positivity)
43           have hsqrt :
44               Real.sqrt (n : Real) * Real.sqrt (n : Real) =
45                 (n : Real) := Real.mul_self_sqrt (by positivity)
46           have hcast : (((n + 1 : Nat) : Real)) = (n : Real) + 1 := by
47             norm_num
48           field_simp
49           nlinarith [sq_nonneg
50             (Real.sqrt (((n + 1 : Nat) : Real)) - Real.sqrt (n : Real))]
51       norm_num [Nat.cast_add, Nat.cast_one] at *
52       linarith

54 /-- Harmonic tail bound with the exact logarithmic ratio needed by the
55 self-bounding threshold split. -/
56 theorem sum_Ico_one_div_natSucc_le_log_div
57     (m n : Nat) (hm : 0 < m) (hmn : m ≤ n) :
58     (Finset.Ico m n).sum (fun t =>
59         1 / (((t + 1 : Nat) : Real))) ≤
60       Real.log ((n : Real) / (m : Real)) := by
61   have hmReal : (0 : Real) < (m : Real) := by exact_mod_cast hm
62   have hmnReal : (m : Real) ≤ (n : Real) := by exact_mod_cast hmn
63   have hsum := AntitoneOn.sum_le_integral_Ico hmn
64     (inv_antitoneOn_Icc_right (b := (n : Real)) hmReal)
65   have hzero : (0 : Real) ∉ Set.uIcc (m : Real) (n : Real) := by
66     rw [Set.uIcc_of_le hmnReal]
67     intro hmem
68     exact (not_le_of_gt hmReal) hmem.1
69   calc
70     (Finset.Ico m n).sum (fun t =>
71         1 / (((t + 1 : Nat) : Real))) =
72         (Finset.Ico m n).sum (fun t =>
73           (((t + 1 : Nat) : Real))⁻¹) := by
74       apply Finset.sum_congr rfl
75       intro t ht
76       rw [one_div]
77     _ ≤ ∫ x in (m : Real)..(n : Real), x⁻¹ := by
78       simpa using hsum
79     _ = Real.log ((n : Real) / (m : Real)) := integral_inv hzero

81 /-- The active-prefix branch of the square-root schedule has a closed-form
82 inverse-square-root bound. -/
83 theorem sum_range_sqrtSchedule_activeBranch_le_closedForm
84     (cutoff : Nat) (amplitude sqrtCard card reciprocalGap : Real)
85     (hamplitude : 0 <= amplitude) (hsqrtCard : 0 <= sqrtCard) :
86     (Finset.range cutoff).sum (fun t =>
87         amplitude / Real.sqrt (((t + 1 : Nat) : Real)) * sqrtCard -
88           card / reciprocalGap) <=
89       2 * amplitude * sqrtCard * Real.sqrt (cutoff : Real) -
90         (cutoff : Real) * card / reciprocalGap := by
91   have hcoefficient : 0 <= amplitude * sqrtCard :=
92     mul_nonneg hamplitude hsqrtCard
93   have hsum := sum_range_one_div_sqrt_natSucc_le_two_sqrt cutoff
94   calc
95     (Finset.range cutoff).sum (fun t =>
96         amplitude / Real.sqrt (((t + 1 : Nat) : Real)) * sqrtCard -
97           card / reciprocalGap) =
98         amplitude * sqrtCard *
99             (Finset.range cutoff).sum (fun t =>
100               1 / Real.sqrt (((t + 1 : Nat) : Real))) -
101           (cutoff : Real) * card / reciprocalGap := by
102       rw [Finset.sum_sub_distrib, Finset.mul_sum]
103       congr 1
104       · apply Finset.sum_congr rfl
105         intro t ht
106         ring
107       · simp
108         ring
109     _ <= amplitude * sqrtCard * (2 * Real.sqrt (cutoff : Real)) -
110           (cutoff : Real) * card / reciprocalGap := by
111       exact sub_le_sub_right (mul_le_mul_of_nonneg_left hsum hcoefficient) _
112     _ = 2 * amplitude * sqrtCard * Real.sqrt (cutoff : Real) -
113           (cutoff : Real) * card / reciprocalGap := by ring
