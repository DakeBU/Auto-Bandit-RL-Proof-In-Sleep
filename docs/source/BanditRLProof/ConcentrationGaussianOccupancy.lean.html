166 theorem sum_le_occupancy_bound (p : ℕ → ℝ) (a ε : ℝ) (ha : 0 < a) (hε : 0 < ε)
167     (h1 : ∀ s, p s ≤ 1)
168     (htail : ∀ s : ℕ, 2*a/ε^2 < (s : ℝ) → p s ≤ occupancyTail a ε s) (n : ℕ) :
169     (∑ i ∈ Finset.range n, p (i+1)) ≤ 1+(2/ε^2)*(a+sqrt (Real.pi*a)+1) := by
170   let u := 2*a/ε^2
171   let m : ℕ := ⌈u⌉₊
172   have hu : 0 ≤ u := by dsimp [u]; positivity
173   have hm : u ≤ (m : ℝ) := Nat.le_ceil u
174   have hm1 : (m : ℝ) ≤ u+1 := (Nat.ceil_lt_add_one hu).le
175   have hbasic (N : ℕ) : (∑ i ∈ Finset.range N, p (i+1)) ≤ (N : ℝ) := by
176     calc
177       _ ≤ ∑ _i ∈ Finset.range N, (1 : ℝ) := Finset.sum_le_sum (fun i _ => h1 (i+1))
178       _ = _ := by simp
179   have hnon : 0 ≤ (2/ε^2)*(1+sqrt (Real.pi*a)) := by positivity
180   have hformula : 1+(2/ε^2)*(a+sqrt (Real.pi*a)+1) =
181       2*a/ε^2+1+(2/ε^2)*(1+sqrt (Real.pi*a)) := by ring
182   rw [hformula]
183   by_cases hnm : n ≤ m
184   · have hn := (hbasic n).trans ((Nat.cast_le.mpr hnm).trans hm1)
185     dsimp [u] at hn
186     nlinarith
187   · have hmn : m ≤ n := by omega
188     have hsplit : n = m+(n-m) := (Nat.add_sub_of_le hmn).symm
189     rw [hsplit, Finset.sum_range_add]
190     have ht : (∑ i ∈ Finset.range (n-m), p (m+i+1)) ≤
191         (2/ε^2)*(1+sqrt (Real.pi*a)) := by
192       calc
193         _ ≤ ∑ i ∈ Finset.range (n-m), occupancyTail a ε ((m : ℝ)+(i+1 : ℕ)) := by
194           apply Finset.sum_le_sum
195           intro i hi
196           have hlarge : 2*a/ε^2 < ((m+i+1 : ℕ) : ℝ) := by
197             push_cast
198             dsimp [u] at hm
199             linarith [Nat.cast_nonneg i (α := ℝ)]
200           simpa [Nat.cast_add, Nat.cast_one, add_assoc] using htail (m+i+1) hlarge
201         _ ≤ _ := sum_occupancyTail_shift_le a ε m ha hε hm (n-m)
