5 /-!
6 # Stationary corrupted finite-arm IID reward laws for scheduled half-Tsallis
7 
8 This module replaces the free nonnegative `corruption` parameter of the
9 abstract self-bounding endpoint by a quantity derived from a concrete process.
10 Each arm receives a fixed real reward shift, the shifted reward is projected
11 back to `[0, 1]`, and fresh finite reward vectors remain IID across rounds.
12 
13 The route is deliberately stationary and oblivious.  History-dependent or
14 time-varying corruptions require a later predictable-law transport.
15 -/

22 /-- Projection of a real value to the unit interval. -/
23 noncomputable def clippedUnitReal (value : Real) : Real :=
24   (Set.projIcc (0 : Real) 1 zero_le_one value : Real)
25 
26 theorem clippedUnitReal_mem_Icc (value : Real) :
27     clippedUnitReal value ∈ Set.Icc (0 : Real) 1 := by
28   exact (Set.projIcc (0 : Real) 1 zero_le_one value).property
29 
30 theorem clippedUnitReal_eq_of_mem_Icc (value : Real)
31     (hvalue : value ∈ Set.Icc (0 : Real) 1) :
32     clippedUnitReal value = value := by
33   simpa [clippedUnitReal] using
34     congrArg Subtype.val (Set.projIcc_of_mem zero_le_one hvalue)
35 
36 theorem abs_clippedUnitReal_add_sub_self_le (value shift : Real)
37     (hvalue : value ∈ Set.Icc (0 : Real) 1) :
38     |clippedUnitReal (value + shift) - value| <= |shift| := by
39   have hcontraction := Set.abs_projIcc_sub_projIcc
40     (a := (0 : Real)) (b := 1) zero_le_one
41     (c := value + shift) (d := value)
42   calc
43     |clippedUnitReal (value + shift) - value| =
44         |clippedUnitReal (value + shift) - clippedUnitReal value| := by
45       rw [clippedUnitReal_eq_of_mem_Icc value hvalue]
46     _ <= |(value + shift) - value| := by
47       simpa only [clippedUnitReal] using hcontraction
48     _ = |shift| := by ring_nf

50 /-- Reward after a stationary arm-dependent shift and unit-interval clipping. -/
51 noncomputable def finiteArmIIDStationaryCorruptedReward {K : Nat}
52     (rewardShift : Fin K -> Real) (state : Fin K -> Rat) (arm : Fin K) : Real :=
53   clippedUnitReal (clippedUnitReward (state arm) + rewardShift arm)

94 theorem abs_stationaryCorruptedLossDiff_sub_baseLossDiff_le {K : Nat}
95     (rewardShift : Fin K -> Real) (state : Fin K -> Rat)
96     (best arm : Fin K) :
97     |(finiteArmIIDStationaryCorruptedRewardVectorLoss rewardShift state arm -
98           finiteArmIIDStationaryCorruptedRewardVectorLoss rewardShift state best) -
99         (finiteArmIIDRewardVectorLoss state arm -
100           finiteArmIIDRewardVectorLoss state best)| <=
101       |rewardShift arm| + |rewardShift best| := by
102   have harm := abs_finiteArmIIDStationaryCorruptedReward_sub_base_le
103     rewardShift state arm
104   have hbest := abs_finiteArmIIDStationaryCorruptedReward_sub_base_le
105     rewardShift state best
106   calc
107     |(finiteArmIIDStationaryCorruptedRewardVectorLoss rewardShift state arm -
108           finiteArmIIDStationaryCorruptedRewardVectorLoss rewardShift state best) -
109         (finiteArmIIDRewardVectorLoss state arm -
110           finiteArmIIDRewardVectorLoss state best)| =
111         |(finiteArmIIDStationaryCorruptedReward rewardShift state best -
112             clippedUnitReward (state best)) -
113           (finiteArmIIDStationaryCorruptedReward rewardShift state arm -
114             clippedUnitReward (state arm))| := by
115       congr 1
116       simp only [finiteArmIIDStationaryCorruptedRewardVectorLoss,
117         finiteArmIIDRewardVectorLoss]
118       ring
119     _ <= |finiteArmIIDStationaryCorruptedReward rewardShift state best -
120             clippedUnitReward (state best)| +
121           |finiteArmIIDStationaryCorruptedReward rewardShift state arm -
122             clippedUnitReward (state arm)| := abs_sub _ _
123     _ <= |rewardShift best| + |rewardShift arm| := add_le_add hbest harm
124     _ = |rewardShift arm| + |rewardShift best| := add_comm _ _
