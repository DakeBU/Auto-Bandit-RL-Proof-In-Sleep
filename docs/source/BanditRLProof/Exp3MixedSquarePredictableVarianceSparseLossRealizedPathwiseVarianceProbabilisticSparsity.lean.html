3 /-!
4 # Pathwise sparse variance under probabilistic sparsity
5 
6 This module removes the global `arms.card * horizon` Markov envelope from the
7 positive-probability sparsity route. Outside the explicit sparsity-failure
8 event, the armwise loss mass is at most `sparsity * horizon`; the existing
9 pointwise mixed-square variance inequality therefore gives the deterministic
10 variance budget
11 
12 `(1 / (gamma / arms.card)) * (sparsity * horizon)`.
13 
14 The realized-regret proof uses four confidence events at `delta / 4`. The
15 sparsity-failure event is charged exactly once, and no measurability assumption
16 on that event is required.
17 -/

25 /-- Deterministic cumulative predictable-variance budget on trajectories that
26 obey the requested support-cardinality cap. -/
27 noncomputable def sampledPredictableSparsePathwiseVarianceBudget
28     {Action : Type v} [DecidableEq Action]
29     (arms : Finset Action) (gamma : Real) (horizon sparsity : Nat) : Real :=
30   (1 / (gamma / (arms.card : Real))) *
31     ((sparsity : Real) * (horizon : Real))
32 
33 /-- Every trajectory either obeys the sparse pathwise variance budget or lies
34 in the explicit sparsity-failure event. -/
35 theorem sampledPredictableMixedSquaredVarianceSum_le_sparsePathwiseVarianceBudget_or_mem_sparsityFailure
36     {Env : Type u} {Action : Type v}
37     [MeasurableSpace Env]
38     [MeasurableSpace Action] [MeasurableSingletonClass Action]
39     [DecidableEq Action]
40     (arms : Finset Action) (harms : arms.Nonempty)
41     (eta gamma : Real) (hgamma_pos : 0 < gamma)
42     (hgamma_le_one : gamma ≤ 1)
43     (loss : PredictableLossVector Env Action)
44     (horizon sparsity : Nat)
45     (sample : Env × ((k : Nat) → Action × Real)) :
46     sampledPredictableMixedSquaredVarianceSum
47         arms eta gamma loss horizon sample ≤
48         sampledPredictableSparsePathwiseVarianceBudget
49           arms gamma horizon sparsity ∨
50       sample ∈ sampledPredictableSparsityFailure
51         arms loss horizon sparsity := by
52   rcases sampledPredictableLossMassSum_le_or_mem_sparsityFailure
53       arms loss horizon sparsity sample with hmass | hbad
54   · left
55     have hfloor_pos : 0 < gamma / (arms.card : Real) :=
56       div_pos hgamma_pos (Nat.cast_pos.2 (Finset.card_pos.mpr harms))
57     exact
58       (sampledPredictableMixedSquaredVarianceSum_le_inv_floor_mul_lossMassSum
59         arms harms eta gamma hgamma_pos hgamma_le_one loss horizon sample).trans
60         (mul_le_mul_of_nonneg_left hmass (one_div_nonneg.2 hfloor_pos.le))
61   · exact Or.inr hbad

196 /-- Generated realized regret under probabilistic sparsity and deterministic
