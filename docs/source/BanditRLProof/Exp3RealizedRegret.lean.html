3 /-!
4 # Realized predictable EXP3 regret
5 
6 This module transports the compiled predictable `p_t`-mixed expected-regret
7 bound to the scalar loss actually observed on the generated trajectory.  The
8 only probabilistic step is the existing conditional action law: conditionally
9 on the pre-action history, the sampled action has finite distribution `p_t`.
10 -/

20 def sampledTrajectoryRealizedLossAt
21     {Env : Type u} {Action : Type v}
22     (t : Nat) (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
23   (sample.2 t).2

214 /-- At a successor time, conditioning on the retained environment/history
215 prefix converts the expected selected predictable coordinate to the `p_t`

223     (arms : Finset Action) (harms : arms.Nonempty)
224     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
225     (hgamma_le_one : gamma <= 1)
226     (loss : PredictableLossVector Env Action) (n : Nat) :
227     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
228       eta gamma hgamma_nonneg hgamma_le_one loss.environment

230       integral mu
231         (sampledTrajectoryExploredPredictableLossAt
232           arms eta gamma loss (n + 1)) := by

235     eta gamma hgamma_nonneg hgamma_le_one loss.environment

239     (sample.2 (n + 1)).1

243     loss.successor n input.1 input.2

249     (Env := Env) arms harms eta gamma hgamma_nonneg hgamma_le_one n

272     simpa [mu, history, action, policy, localSource] using
273       (sampledImportanceWeightedTrajectoryMeasure_condDistrib_action_given_environment
274         prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss n)
275   have hscore : Measurable score := by
276     exact (loss.measurable_successor n).comp
277       ((measurable_fst.comp measurable_fst).prodMk
278         ((measurable_snd.comp measurable_fst).prodMk measurable_snd))
279   have hintegrable : Integrable score (mu.map history ⊗ₘ policy) := by
280     refine Integrable.of_bound hscore.aestronglyMeasurable 1 ?_
281     exact Filter.Eventually.of_forall fun z => by
282       have hmem := loss.successor_mem_unitInterval n z.1.1 z.1.2 z.2
283       simpa [score, roundLoss, Real.norm_eq_abs, abs_of_nonneg hmem.1] using hmem.2
284   have hvector :=
285     integral_historyAction_eq_integral_sum_of_condDistrib_ae_eq_finiteActionMeasure
286       mu history hhistory action haction arms prob
287       (fun input => source.distribution input) policy hpolicy hcond score hscore
288       hintegrable

291       prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss (n + 1)

293   have hmix : Measurable (fun input =>
294       arms.sum (fun selected => prob input selected * roundLoss input selected)) := by
295     refine Finset.measurable_sum arms fun selected hselected => ?_
296     exact (source.measurable_prob selected hselected).mul
297       ((loss.measurable_successor n).comp
298         (measurable_fst.prodMk
299           (measurable_snd.prodMk
300             (measurable_const : Measurable
301               (fun _ : Env × History.FinitePairHistory Action Real n => selected)))))
302   have hmap := integral_map (μ := mu) hhistory.aemeasurable
303     hmix.aestronglyMeasurable
304   calc

306         integral mu (fun sample => score (history sample, action sample)) := by

314           arms eta gamma loss (n + 1)) := by

394 /-- Unoptimized expected regret for the scalar losses actually realized by the
395 sampled predictable EXP3 trajectory. -/
396 theorem sampledPredictable_realizedExpectedRegret_le
397     {Env : Type u} {Action : Type v}
398     [MeasurableSpace Env] [StandardBorelSpace Env]
399     [MeasurableSpace Action] [MeasurableSingletonClass Action]
400     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
401     (prior : Measure Env) [IsProbabilityMeasure prior]
402     (arms : Finset Action) (harms : arms.Nonempty)
403     (eta gamma : Real) (heta : 0 < eta)
404     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
405     (loss : PredictableLossVector Env Action) (horizon : Nat)
406     (comparator : Action) (hcomparator : comparator ∈ arms) :
407     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
408       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
409     integral mu (fun sample =>
410       (Finset.range horizon).sum (fun t =>
411           sampledTrajectoryRealizedLossAt t sample) -
412         (Finset.range horizon).sum (fun t =>
413           predictableLossAt loss t sample comparator)) <=
414       Real.log arms.card / eta +
415         (eta * (1 / (1 - gamma))) *
416           ((arms.card : Real) * (horizon : Real)) +
417         gamma * (horizon : Real) := by
418   dsimp only
419   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
420     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
421   have htransport :=
422     sampledPredictableRealized_finiteHorizon_integral_eq_explored
423       prior arms harms eta gamma hgamma_pos.le hgamma_lt_one.le loss horizon
424   have hbase := sampledPredictable_expectedRegret_le
425     prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss horizon
426       comparator hcomparator
427   dsimp only at htransport hbase
428   have hrealizedSum : Integrable (fun sample =>
429       (Finset.range horizon).sum (fun t =>
430         sampledTrajectoryRealizedLossAt t sample)) mu :=
431     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
432       (fun t sample => sampledTrajectoryRealizedLossAt t sample)
433       (fun t _ht => integrable_sampledTrajectoryRealizedLossAt
434         prior arms harms eta gamma hgamma_pos.le hgamma_lt_one.le loss t)
435   have hexploredSum : Integrable (fun sample =>
436       (Finset.range horizon).sum (fun t =>
437         sampledTrajectoryExploredPredictableLossAt
438           arms eta gamma loss t sample)) mu :=
439     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
440       (fun t sample => sampledTrajectoryExploredPredictableLossAt
441         arms eta gamma loss t sample)
442       (fun t _ht => integrable_sampledTrajectoryExploredPredictableLossAt
443         mu arms harms eta gamma hgamma_pos.le hgamma_lt_one.le loss t)
444   have hcomparatorSum : Integrable (fun sample =>
445       (Finset.range horizon).sum (fun t =>
446         predictableLossAt loss t sample comparator)) mu :=
447     IntegrabilitySums.integrable_finset_sum mu (Finset.range horizon)
448       (fun t sample => predictableLossAt loss t sample comparator)
449       (fun t _ht => integrable_predictableLossAt mu loss t comparator)
450   rw [integral_sub hexploredSum hcomparatorSum] at hbase
451   rw [integral_sub hrealizedSum hcomparatorSum]
452   rw [htransport]
453   exact hbase
