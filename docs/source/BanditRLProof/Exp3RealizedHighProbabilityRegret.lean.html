4 /-!
5 # Generated realized EXP3 high-probability regret
6 
7 This module composes the generated predictable EXP3 high-probability theorem
8 with the one-sided realized-minus-exploration-mixed deviation tail. The primary
9 endpoint controls the scalar loss stored in the generated trajectory, with the
10 requested total failure probability split equally across the pure-q,
11 comparator-estimator, and realized-deviation events.
12 
13 The theorem inherits the range-based importance-weighted confidence radii from
14 the predictable route. It is therefore a valid realized selected-loss theorem,
15 but it is not presented as the ideal EXP3.P/Freedman rate.
16 -/

26     (horizon : Nat) (delta : Real) : Real :=

31 /-- Raw three-event form of the generated realized EXP3 regret theorem. The
32 same `delta` is used for the pure-q, comparator-estimator, and realized
33 deviation tails, so the displayed failure probability is their three-term
34 sum. -/
35 theorem sampledPredictable_realizedHighProbabilityRegret_tail_delta
36     {Env : Type u} {Action : Type v}
37     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
38     [MeasurableSpace Action] [MeasurableSingletonClass Action]
39     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
40     (prior : Measure Env) [IsProbabilityMeasure prior]
41     (arms : Finset Action) (harms : arms.Nonempty)
42     (eta gamma : Real) (heta : 0 < eta)
43     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
44     (loss : PredictableLossVector Env Action)
45     (comparator : Action) (hcomparator : comparator ∈ arms)
46     (horizon : Nat) (hhorizon : 0 < horizon)
47     (delta : Real) (hdelta : 0 < delta) :
48     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
49       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
50     mu {sample |
51         sampledPredictableRealizedHighProbabilityRegretBudget
52             arms eta gamma horizon delta <=
53           (Finset.range horizon).sum (fun t =>
54               sampledTrajectoryRealizedLossAt t sample) -
55             (Finset.range horizon).sum (fun t =>
56               predictableLossAt loss t sample comparator)} <=
57       (ENNReal.ofReal delta + ENNReal.ofReal delta) + ENNReal.ofReal delta := by
58   dsimp only
59   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
60     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
61   let predictableRegret :=
62     fun sample : Env × ((k : Nat) -> Action × Real) =>
63       (Finset.range horizon).sum (fun t =>
64           sampledTrajectoryExploredPredictableLossAt
65             arms eta gamma loss t sample) -
66         (Finset.range horizon).sum (fun t =>
67           predictableLossAt loss t sample comparator)
68   let realizedDeviation :=
69     fun sample : Env × ((k : Nat) -> Action × Real) =>
70       (Finset.range horizon).sum (fun t =>
71         sampledTrajectoryRealizedDeviationAt arms eta gamma loss t sample)
72   let realizedRegret :=
73     fun sample : Env × ((k : Nat) -> Action × Real) =>
74       (Finset.range horizon).sum (fun t =>
75           sampledTrajectoryRealizedLossAt t sample) -
76         (Finset.range horizon).sum (fun t =>
77           predictableLossAt loss t sample comparator)
78   let predictableBudget := sampledPredictableHighProbabilityRegretBudget
79     arms eta gamma horizon delta
80   let realizedRadius :=
81     sampledPredictableRealizedDeviationConfidenceRadius horizon delta
82   let predictableBad : Set (Env × ((k : Nat) -> Action × Real)) :=
83     {sample | predictableBudget <= predictableRegret sample}
84   let realizedBad : Set (Env × ((k : Nat) -> Action × Real)) :=
85     {sample | realizedRadius <= realizedDeviation sample}
86   have hpredictableTail :
87       mu predictableBad <= ENNReal.ofReal delta + ENNReal.ofReal delta := by
88     have h := sampledPredictable_highProbabilityRegret_tail_delta
89       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
90         hcomparator horizon hhorizon delta hdelta
91     dsimp only at h
92     simpa [mu, predictableBad, predictableBudget, predictableRegret] using h
93   have hrealizedTail : mu realizedBad <= ENNReal.ofReal delta := by
94     have h := sampledPredictableRealizedDeviation_sum_tail_delta
95       prior arms harms eta gamma hgamma_pos.le hgamma_lt_one.le loss horizon
96         hhorizon delta hdelta
97     dsimp only at h
98     simpa [mu, realizedBad, realizedRadius, realizedDeviation] using h
99   have hdecomp : ∀ sample,
100       realizedRegret sample =
101         predictableRegret sample + realizedDeviation sample := by
102     intro sample
103     dsimp [realizedRegret, predictableRegret, realizedDeviation,
104       sampledTrajectoryRealizedDeviationAt]
105     rw [Finset.sum_sub_distrib]
106     ring
107   have hsubset :
108       {sample |
109         predictableBudget + realizedRadius <= realizedRegret sample} ⊆
110         predictableBad ∪ realizedBad := by
111     intro sample hregret
112     by_cases hpredictable : sample ∈ predictableBad
113     · exact Or.inl hpredictable
114     by_cases hrealized : sample ∈ realizedBad
115     · exact Or.inr hrealized
116     exfalso
117     have hpredictableGood :
118         predictableRegret sample < predictableBudget := by
119       change ¬ predictableBudget <= predictableRegret sample at hpredictable
120       exact lt_of_not_ge hpredictable
121     have hrealizedGood : realizedDeviation sample < realizedRadius := by
122       change ¬ realizedRadius <= realizedDeviation sample at hrealized
123       exact lt_of_not_ge hrealized
124     have hregret' :
125         predictableBudget + realizedRadius <= realizedRegret sample := hregret
126     rw [hdecomp sample] at hregret'
127     linarith
128   calc
129     mu {sample |
130         sampledPredictableRealizedHighProbabilityRegretBudget
131             arms eta gamma horizon delta <=
132           (Finset.range horizon).sum (fun t =>
133               sampledTrajectoryRealizedLossAt t sample) -
134             (Finset.range horizon).sum (fun t =>
135               predictableLossAt loss t sample comparator)} <=
136         mu (predictableBad ∪ realizedBad) := by
137       apply measure_mono
138       simpa [sampledPredictableRealizedHighProbabilityRegretBudget,
139         predictableBudget, realizedRadius, realizedRegret, mu] using hsubset
140     _ <= mu predictableBad + mu realizedBad := measure_union_le _ _
141     _ <= (ENNReal.ofReal delta + ENNReal.ofReal delta) +
142           ENNReal.ofReal delta :=
143       add_le_add hpredictableTail hrealizedTail

145 /-- Standard total-failure-probability form of generated realized EXP3 regret.
146 Each of the three underlying confidence events receives `delta / 3`. -/
147 theorem sampledPredictable_realizedHighProbabilityRegret_tail_total_delta
148     {Env : Type u} {Action : Type v}
149     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
150     [MeasurableSpace Action] [MeasurableSingletonClass Action]
151     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
152     (prior : Measure Env) [IsProbabilityMeasure prior]
153     (arms : Finset Action) (harms : arms.Nonempty)
154     (eta gamma : Real) (heta : 0 < eta)
155     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
156     (loss : PredictableLossVector Env Action)
157     (comparator : Action) (hcomparator : comparator ∈ arms)
158     (horizon : Nat) (hhorizon : 0 < horizon)
159     (delta : Real) (hdelta : 0 < delta) :
160     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
161       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
162     mu {sample |
163         sampledPredictableRealizedHighProbabilityRegretBudget
164             arms eta gamma horizon (delta / 3) <=
165           (Finset.range horizon).sum (fun t =>
166               sampledTrajectoryRealizedLossAt t sample) -
167             (Finset.range horizon).sum (fun t =>
168               predictableLossAt loss t sample comparator)} <=
169       ENNReal.ofReal delta := by
170   have hthird_pos : 0 < delta / 3 := div_pos hdelta (by norm_num)
171   have h := sampledPredictable_realizedHighProbabilityRegret_tail_delta
172     prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
173       hcomparator horizon hhorizon (delta / 3) hthird_pos
174   dsimp only at h ⊢
175   have hthird_nonneg : 0 <= delta / 3 := hthird_pos.le
176   have hprob :
177       (ENNReal.ofReal (delta / 3) + ENNReal.ofReal (delta / 3)) +
178           ENNReal.ofReal (delta / 3) = ENNReal.ofReal delta := by
179     rw [← ENNReal.ofReal_add hthird_nonneg hthird_nonneg]
180     rw [← ENNReal.ofReal_add (add_nonneg hthird_nonneg hthird_nonneg)
181       hthird_nonneg]
182     congr 1
183     ring
184   exact h.trans_eq hprob
