4 /-!
5 # Finite-bandit mean losses for scheduled half-Tsallis FTRL
6 
7 This module turns a bounded finite-bandit mean-reward model into the
8 history-independent predictable loss family `1 - mean`.  Its loss differences
9 are exactly the model gaps, so the compiled square-root-schedule fixed-gap
10 theorem applies without a caller-supplied predictable gap law.
11 
12 The generated feedback remains the deterministic mean-loss environment of
13 `Exp3.PredictableLossVector`; this module does not identify it with a stochastic
14 reward-kernel trajectory.
15 -/

25 /-- The stationary predictable loss family obtained from bounded arm means. -/
26 noncomputable def finiteBanditMeanLoss
27     {K : Nat} {Env : Type u} [MeasurableSpace Env]
28     (model : FiniteBanditModel K)
29     (hmean : forall arm,
30       ((model.mean arm : Rat) : Real) ∈ Set.Icc (0 : Real) 1) :
31     PredictableLossVector Env (Fin K) where
32   initial := fun _ arm => 1 - ((model.mean arm : Rat) : Real)
33   successor := fun _ _ _ arm => 1 - ((model.mean arm : Rat) : Real)
34   measurable_initial :=
35     measurable_const.sub
36       ((measurable_of_countable (fun arm : Fin K =>
37         ((model.mean arm : Rat) : Real))).comp measurable_snd)
38   measurable_successor := fun _ =>
39     measurable_const.sub
40       ((measurable_of_countable (fun arm : Fin K =>
41         ((model.mean arm : Rat) : Real))).comp
42           (measurable_snd.comp measurable_snd))
43   initial_nonneg := fun _ arm => by
44     exact sub_nonneg.mpr (hmean arm).2
45   initial_le_one := fun _ arm => by
46     linarith [(hmean arm).1]
47   successor_nonneg := fun _ _ _ arm => by
48     exact sub_nonneg.mpr (hmean arm).2
49   successor_le_one := fun _ _ _ arm => by
50     linarith [(hmean arm).1]

111 The square-root scheduled half-Tsallis algorithm has logarithmic fixed-gap
112 regret on the deterministic mean-loss environment of a bounded finite-bandit
113 model with strictly positive non-best gaps.
114 -/
115 theorem integral_sampledScheduledHalfTsallisFiniteBanditMeanLossRegret_le_log_fixedGap
116     {K : Nat} {Env : Type u}
117     [MeasurableSpace Env] [StandardBorelSpace Env]
118     (prior : Measure Env) [IsProbabilityMeasure prior]
119     (model : FiniteBanditModel K)
120     (hmean : forall arm,
121       ((model.mean arm : Rat) : Real) ∈ Set.Icc (0 : Real) 1)
122     (hgapPos : forall arm, arm ≠ model.bestArm ->
123       0 < ((model.gap arm : Rat) : Real))
124     (horizon : Nat) (corruption : Real) (hcorruption : 0 <= corruption) :
125     letI : Nonempty (Fin K) := ⟨model.bestArm⟩
126     let loss := Exp3.finiteBanditMeanLoss (Env := Env) model hmean
127     let selector :=
128       canonicalHalfTsallisScheduleGeneratedSelectorMeasurability
129         (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
130         sampledScheduledHalfTsallisSqrtSchedule loss
131     let mu := prior ⊗ₘ sampledScheduledHalfTsallisTrajectoryKernel
132       (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
133       sampledScheduledHalfTsallisSqrtSchedule
134       selector.finiteHistory loss.environment
135     integral mu (sampledScheduledHalfTsallisPredictableEnvironmentRegret
136         (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
137         sampledScheduledHalfTsallisSqrtSchedule loss
138         (pointMass model.bestArm) horizon) <=
139       (1 + Real.log (((horizon + 1 : Nat) : Real))) *
140         (1 + 25 * ((Finset.univ : Finset (Fin K)).erase model.bestArm).sum
141           (fun arm => 1 / ((model.gap arm : Rat) : Real))) +
142         corruption := by
143   letI : Nonempty (Fin K) := ⟨model.bestArm⟩
144   dsimp only
145   exact

147       prior (Finset.univ : Finset (Fin K)) Finset.univ_nonempty
