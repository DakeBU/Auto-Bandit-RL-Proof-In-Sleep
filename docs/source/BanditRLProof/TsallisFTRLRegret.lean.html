5 # Finite-horizon FTRL stability and Tsallis penalty decomposition
6 
7 This module lifts the one-step explicit-minimizer API to a finite-horizon
8 regularized be-the-leader theorem and the standard FTRL stability/penalty
9 regret decomposition. It then specializes the regularizer to negative Tsallis
10 entropy and exposes the penalty as a difference of finite power sums.
11 
12 The route is deterministic. It consumes explicit minimizer certificates and
13 does not prove minimizer existence, a Tsallis stability bound, unbiased loss
14 estimation, self-bounding conversion, or a final Tsallis-INF regret theorem.
15 -/

22 /-- Coordinatewise cumulative loss before round `t`. -/
23 noncomputable def cumulativeLoss {Action : Type u}
24     (loss : Nat -> Action -> Real) (t : Nat) : Action -> Real :=
25   fun action => (Finset.range t).sum (fun s => loss s action)
26 
27 @[simp] theorem cumulativeLoss_zero {Action : Type u}
28     (loss : Nat -> Action -> Real) :
29     cumulativeLoss loss 0 = 0 := by
30   funext action
31   simp [cumulativeLoss]
32 
33 /-- Adding one round appends its loss vector to the cumulative loss. -/
34 theorem cumulativeLoss_succ {Action : Type u}
35     (loss : Nat -> Action -> Real) (t : Nat) :
36     cumulativeLoss loss (t + 1) =
37       fun action => cumulativeLoss loss t action + loss t action := by
38   funext action
39   simp [cumulativeLoss, Finset.sum_range_succ]
40 
41 /-- Finite-action linear loss is additive in its loss-vector argument. -/
42 theorem linearLoss_add_right {Action : Type u}
43     (arms : Finset Action) (p x y : Action -> Real) :
44     linearLoss arms p (fun action => x action + y action) =
45       linearLoss arms p x + linearLoss arms p y := by
46   simp [linearLoss, mul_add, Finset.sum_add_distrib]
47 
48 /-- A linear loss against the cumulative vector is the sum of round losses. -/
49 theorem linearLoss_cumulativeLoss {Action : Type u}
50     (arms : Finset Action) (p : Action -> Real)
51     (loss : Nat -> Action -> Real) (T : Nat) :
52     linearLoss arms p (cumulativeLoss loss T) =
53       (Finset.range T).sum (fun t => linearLoss arms p (loss t)) := by
54   simp only [linearLoss, cumulativeLoss, Finset.mul_sum]
55   rw [Finset.sum_comm]
56 
57 /-- The cumulative regularized objective has the expected successor update. -/
58 theorem regularizedObjective_cumulativeLoss_succ {Action : Type u}
59     (arms : Finset Action) (eta : Real)
60     (regularizer : (Action -> Real) -> Real)
61     (loss : Nat -> Action -> Real) (p : Action -> Real) (t : Nat) :
62     regularizedObjective arms eta regularizer (cumulativeLoss loss (t + 1)) p =
63       regularizedObjective arms eta regularizer (cumulativeLoss loss t) p +
64         eta * linearLoss arms p (loss t) := by
65   rw [cumulativeLoss_succ, regularizedObjective, regularizedObjective,
66     linearLoss_add_right]
67   ring

69 /--
70 Scaled regularized be-the-leader inequality for cumulative-loss minimizers.
71 
72 The point `p t` minimizes the objective built from losses before round `t`.
73 Consequently the shifted choices `p (t + 1)` can be compared to the terminal
74 cumulative objective while retaining the initial regularizer value.
75 -/
76 theorem eta_mul_sum_next_linearLoss_add_regularizer_zero_le_objective
77     {Action : Type u}
78     (feasible : (Action -> Real) -> Prop)
79     (arms : Finset Action) (eta : Real)
80     (regularizer : (Action -> Real) -> Real)
81     (loss : Nat -> Action -> Real) (p : Nat -> Action -> Real)
82     (T : Nat)
83     (hp : forall t, t <= T ->
84       IsRegularizedMinimizer feasible arms eta regularizer
85         (cumulativeLoss loss t) (p t)) :
86     eta * (Finset.range T).sum
87         (fun t => linearLoss arms (p (t + 1)) (loss t)) +
88       regularizer (p 0) <=
89         regularizedObjective arms eta regularizer
90           (cumulativeLoss loss T) (p T) := by
91   induction T with
92   | zero =>
93       simp [regularizedObjective, cumulativeLoss, linearLoss]
94   | succ T ih =>
95       have hprefix : forall t, t <= T ->
96           IsRegularizedMinimizer feasible arms eta regularizer
97             (cumulativeLoss loss t) (p t) := by
98         intro t ht
99         exact hp t (ht.trans (Nat.le_succ T))
100       have hih := ih hprefix
101       have hpT := hp T (Nat.le_succ T)
102       have hpNext := hp (T + 1) (by rfl)
103       have hmin :
104           regularizedObjective arms eta regularizer
105               (cumulativeLoss loss T) (p T) <=
106             regularizedObjective arms eta regularizer
107               (cumulativeLoss loss T) (p (T + 1)) :=
108         hpT.2 (p (T + 1)) hpNext.1
109       calc
110         eta * (Finset.range (T + 1)).sum
111               (fun t => linearLoss arms (p (t + 1)) (loss t)) +
112             regularizer (p 0) =
113             (eta * (Finset.range T).sum
114                 (fun t => linearLoss arms (p (t + 1)) (loss t)) +
115               regularizer (p 0)) +
116                 eta * linearLoss arms (p (T + 1)) (loss T) := by
117           rw [Finset.sum_range_succ]
118           ring
119         _ <= regularizedObjective arms eta regularizer
120               (cumulativeLoss loss T) (p (T + 1)) +
121                 eta * linearLoss arms (p (T + 1)) (loss T) :=
122           by
123             simpa [add_comm] using
124               add_le_add_right (hih.trans hmin)
125                 (eta * linearLoss arms (p (T + 1)) (loss T))
126         _ = regularizedObjective arms eta regularizer
127               (cumulativeLoss loss (T + 1)) (p (T + 1)) :=
128           (regularizedObjective_cumulativeLoss_succ
129             arms eta regularizer loss (p (T + 1)) T).symm

131 /--
132 Regularized be-the-leader bound against any feasible comparator.
133 -/
134 theorem sum_next_linearLoss_sub_comparator_le_regularizer_penalty
135     {Action : Type u}
136     (feasible : (Action -> Real) -> Prop)
137     (arms : Finset Action) (eta : Real)
138     (regularizer : (Action -> Real) -> Real)
139     (loss : Nat -> Action -> Real) (p : Nat -> Action -> Real)
140     (q : Action -> Real) (T : Nat)
141     (heta : 0 < eta)
142     (hp : forall t, t <= T ->
143       IsRegularizedMinimizer feasible arms eta regularizer
144         (cumulativeLoss loss t) (p t))
145     (hq : feasible q) :
146     (Finset.range T).sum (fun t =>
147         linearLoss arms (p (t + 1)) (loss t) -
148           linearLoss arms q (loss t)) <=
149       (regularizer q - regularizer (p 0)) / eta := by
150   have hleader :=
151     eta_mul_sum_next_linearLoss_add_regularizer_zero_le_objective
152       feasible arms eta regularizer loss p T hp
153   have hpT := hp T (by rfl)
154   have hcompare :
155       regularizedObjective arms eta regularizer
156           (cumulativeLoss loss T) (p T) <=
157         regularizedObjective arms eta regularizer
158           (cumulativeLoss loss T) q :=
159     hpT.2 q hq
160   have htotal := hleader.trans hcompare
161   rw [Finset.sum_sub_distrib, le_div_iff₀ heta]
162   rw [regularizedObjective, linearLoss_cumulativeLoss] at htotal
163   calc
164     ((Finset.range T).sum
165         (fun t => linearLoss arms (p (t + 1)) (loss t)) -
166       (Finset.range T).sum
167         (fun t => linearLoss arms q (loss t))) * eta =
168         eta * (Finset.range T).sum
169             (fun t => linearLoss arms (p (t + 1)) (loss t)) -
170           eta * (Finset.range T).sum
171             (fun t => linearLoss arms q (loss t)) := by ring
172     _ <= regularizer q - regularizer (p 0) := by linarith

174 /--
175 Finite-horizon FTRL stability/penalty regret decomposition.
176 
177 The first sum on the right is the stability term. The second term is the
178 regularizer penalty. No convexity or minimizer-existence theorem is hidden:
179 all cumulative minimizer certificates are explicit inputs.
180 -/
181 theorem cumulativeLinearLoss_sub_comparator_le_stability_add_penalty
182     {Action : Type u}
183     (feasible : (Action -> Real) -> Prop)
184     (arms : Finset Action) (eta : Real)
185     (regularizer : (Action -> Real) -> Real)
186     (loss : Nat -> Action -> Real) (p : Nat -> Action -> Real)
187     (q : Action -> Real) (T : Nat)
188     (heta : 0 < eta)
189     (hp : forall t, t <= T ->
190       IsRegularizedMinimizer feasible arms eta regularizer
191         (cumulativeLoss loss t) (p t))
192     (hq : feasible q) :
193     (Finset.range T).sum (fun t =>
194         linearLoss arms (p t) (loss t) - linearLoss arms q (loss t)) <=
195       (Finset.range T).sum (fun t =>
196           linearLoss arms (p t) (loss t) -
197             linearLoss arms (p (t + 1)) (loss t)) +
198         (regularizer q - regularizer (p 0)) / eta := by
199   have hleader :=
200     sum_next_linearLoss_sub_comparator_le_regularizer_penalty
201       feasible arms eta regularizer loss p q T heta hp hq
202   calc
203     (Finset.range T).sum (fun t =>
204         linearLoss arms (p t) (loss t) - linearLoss arms q (loss t)) =
205       (Finset.range T).sum (fun t =>
206           linearLoss arms (p t) (loss t) -
207             linearLoss arms (p (t + 1)) (loss t)) +
208         (Finset.range T).sum (fun t =>
209           linearLoss arms (p (t + 1)) (loss t) -
210             linearLoss arms q (loss t)) := by
211       rw [← Finset.sum_add_distrib]
212       apply Finset.sum_congr rfl
213       intro t ht
214       ring
215     _ <= (Finset.range T).sum (fun t =>
216           linearLoss arms (p t) (loss t) -
217             linearLoss arms (p (t + 1)) (loss t)) +
218         (regularizer q - regularizer (p 0)) / eta :=
219       by
220         simpa [add_comm] using
221           add_le_add_right hleader
222             ((Finset.range T).sum (fun t =>
223               linearLoss arms (p t) (loss t) -
224                 linearLoss arms (p (t + 1)) (loss t)))

253 /-- Difference of negative Tsallis entropies as a power-sum difference. -/
254 theorem negEntropyRegularizer_sub_eq_powerSum_sub_div
255     {Action : Type u}
256     (arms : Finset Action) (alpha : Real) (p q : Action -> Real)
257     (halpha : Ne alpha 1) :
258     negEntropyRegularizer arms alpha q -
259         negEntropyRegularizer arms alpha p =
260       (powerSum arms alpha p - powerSum arms alpha q) / (1 - alpha) := by
261   have hdenom : Ne (1 - alpha) 0 := one_sub_exponent_ne_zero halpha
262   simp only [negEntropyRegularizer, entropy]
263   field_simp [hdenom]
264   ring

266 /--
267 Finite-horizon Tsallis-FTRL stability/penalty regret decomposition.
268 
269 The penalty is explicit in the finite power sums. The remaining algorithmic
270 obligation is to bound the stability sum for the chosen Tsallis exponent and
271 loss estimator, then supply minimizer existence and any stochastic contracts.
272 -/
273 theorem cumulativeLinearLoss_sub_comparator_le_stability_add_powerSumPenalty
274     {Action : Type u}
275     (arms : Finset Action) (alpha eta : Real)
276     (loss : Nat -> Action -> Real) (p : Nat -> Action -> Real)
277     (q : Action -> Real) (T : Nat)
278     (halpha : Ne alpha 1) (heta : 0 < eta)
279     (hp : forall t, t <= T ->
280       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms) arms eta
281         (negEntropyRegularizer arms alpha)
282         (FTRL.cumulativeLoss loss t) (p t))
283     (hq : FTRL.finiteSimplex arms q) :
284     (Finset.range T).sum (fun t =>
285         FTRL.linearLoss arms (p t) (loss t) -
286           FTRL.linearLoss arms q (loss t)) <=
287       (Finset.range T).sum (fun t =>
288           FTRL.linearLoss arms (p t) (loss t) -
289             FTRL.linearLoss arms (p (t + 1)) (loss t)) +
290         ((powerSum arms alpha (p 0) - powerSum arms alpha q) /
291           (1 - alpha)) / eta := by
292   have hregret :=
293     FTRL.cumulativeLinearLoss_sub_comparator_le_stability_add_penalty_simplex
294       arms eta (negEntropyRegularizer arms alpha) loss p q T heta hp hq
295   rw [negEntropyRegularizer_sub_eq_powerSum_sub_div
296     arms alpha (p 0) q halpha] at hregret
297   exact hregret

299 end Tsallis
