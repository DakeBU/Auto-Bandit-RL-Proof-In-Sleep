9 # Half-Tsallis FTRL one-step stability
10 
11 This module proves the deterministic one-step stability estimate used by the
12 `alpha = 1 / 2` Tsallis-INF route.  It consumes explicit interior stationarity
13 certificates for the current and importance-weighted updated distributions.
14 The certificates expose exactly the KKT equation for the local objective
15 `eta * <p, score> + negEntropyRegularizer arms (1 / 2) p`.
16 
17 The final theorem averages the pathwise stability term over the current finite
18 simplex and bounds it by `2 * eta * sum_a sqrt (p a)`.  It does not derive the
19 stationarity certificates from minimizer certificates, prove minimizer
20 interiority/existence, or perform a conditional-expectation transport.
21 
22 This is a valid but looser route for the stability term in the local FTRL
23 decomposition, not a literal port of the conjugate-potential stability bound in
24 Tsallis-INF Lemmas 11/19.  Matching the paper's half-Tsallis regularizer scaling
25 uses `eta_local = eta_paper / 2` up to simplex-constant terms; the bound here is
26 then `eta_paper * sum_a sqrt (p a)`, twice the comparable Lemma 11 coefficient.
27 -/

34 /--
35 Interior first-order stationarity for the half-Tsallis regularizer.
36 
37 For `alpha = 1 / 2`, the coordinate derivative of the local negative Tsallis
38 entropy is `-p_a^(-1/2)`.  The common multiplier records the simplex equality
39 constraint; positivity and normalization are kept as separate theorem inputs.
40 -/
41 def HalfTsallisInteriorStationary {Action : Type u}
42     (arms : Finset Action) (eta : Real)
43     (score p : Action -> Real) (multiplier : Real) : Prop :=
44   forall action, action ∈ arms ->
45     eta * score action - (p action) ^ (-(1 / 2 : Real)) = multiplier
46 
47 /-- Subtracting two half-Tsallis stationarity equations isolates the update. -/
48 theorem halfTsallisInteriorStationary_rpow_sub_rpow_eq
49     {Action : Type u}
50     (arms : Finset Action) (eta : Real)
51     (score increment p q : Action -> Real)
52     (multiplier nextMultiplier : Real)
53     (hp : HalfTsallisInteriorStationary
54       arms eta score p multiplier)
55     (hq : HalfTsallisInteriorStationary
56       arms eta (fun action => score action + increment action)
57         q nextMultiplier)
58     {action : Action} (haction : action ∈ arms) :
59     (q action) ^ (-(1 / 2 : Real)) -
60         (p action) ^ (-(1 / 2 : Real)) =
61       eta * increment action - (nextMultiplier - multiplier) := by
62   have hpAction := hp action haction
63   have hqAction := hq action haction
64   dsimp only at hqAction
65   linarith

67 /--
68 Scalar half-Tsallis curvature inequality on the positive cone.
69 
70 This is the one-dimensional inequality that converts a negative-half-power
71 gradient displacement into a displacement of probability mass.
72 -/
73 theorem sub_le_two_mul_rpow_three_halves_mul_neg_half_rpow_sub
74     {p q : Real} (hp : 0 < p) (hq : 0 < q) (hqp : q <= p) :
75     p - q <=
76       2 * p ^ (3 / 2 : Real) *
77         (q ^ (-(1 / 2 : Real)) - p ^ (-(1 / 2 : Real))) := by
78   have hsqrtPPos : 0 < Real.sqrt p := Real.sqrt_pos.2 hp
79   have hsqrtQPos : 0 < Real.sqrt q := Real.sqrt_pos.2 hq
80   have hsqrtLe : Real.sqrt q <= Real.sqrt p := Real.sqrt_le_sqrt hqp
81   have hsqrtPSq : (Real.sqrt p) ^ 2 = p := Real.sq_sqrt hp.le
82   have hsqrtQSq : (Real.sqrt q) ^ 2 = q := Real.sq_sqrt hq.le
83   have hdiff :
84       q ^ (-(1 / 2 : Real)) - p ^ (-(1 / 2 : Real)) =
85         (Real.sqrt p - Real.sqrt q) /
86           (Real.sqrt p * Real.sqrt q) := by
87     rw [Real.rpow_neg hq.le, Real.rpow_neg hp.le]
88     rw [← Real.sqrt_eq_rpow, ← Real.sqrt_eq_rpow]
89     field_simp [ne_of_gt hsqrtPPos, ne_of_gt hsqrtQPos]
90   have hdiffNonneg :
91       0 <= q ^ (-(1 / 2 : Real)) - p ^ (-(1 / 2 : Real)) := by
92     rw [hdiff]
93     exact div_nonneg (sub_nonneg.2 hsqrtLe)
94       (mul_nonneg hsqrtPPos.le hsqrtQPos.le)
95   have hfactorLe :
96       Real.sqrt p * Real.sqrt q * (Real.sqrt p + Real.sqrt q) <=
97         2 * p ^ (3 / 2 : Real) := by
98     have hsumLe :
99         Real.sqrt p + Real.sqrt q <=
100           Real.sqrt p + Real.sqrt p := by
101       linarith
102     have hinner :
103         Real.sqrt q * (Real.sqrt p + Real.sqrt q) <=
104           Real.sqrt p * (Real.sqrt p + Real.sqrt p) :=
105       mul_le_mul hsqrtLe hsumLe
106         (add_nonneg hsqrtPPos.le hsqrtQPos.le) hsqrtPPos.le
107     have hscaled := mul_le_mul_of_nonneg_left hinner hsqrtPPos.le
108     have hrpow :
109         p ^ (3 / 2 : Real) = p * Real.sqrt p := by
110       rw [show (3 / 2 : Real) = 1 + 1 / 2 by norm_num,
111         Real.rpow_add hp, Real.rpow_one, ← Real.sqrt_eq_rpow]
112     rw [hrpow]
113     nlinarith
114   have hfactor :
115       p - q =
116         (q ^ (-(1 / 2 : Real)) - p ^ (-(1 / 2 : Real))) *
117           (Real.sqrt p * Real.sqrt q *
118             (Real.sqrt p + Real.sqrt q)) := by
119     rw [hdiff]
120     field_simp [ne_of_gt hsqrtPPos, ne_of_gt hsqrtQPos]
121     nlinarith
122   rw [hfactor]
123   simpa [mul_comm] using
124     mul_le_mul_of_nonneg_left hfactorLe hdiffNonneg

126 /--
127 Pathwise half-Tsallis FTRL stability for one importance-weighted observation.
128 
129 The current and updated distributions are normalized and strictly positive on
130 `arms`.  Their stationarity certificates force the multiplier displacement to
131 lie between zero and the selected coordinate update; the scalar curvature
132 lemma then controls the one-step linear-loss difference.
133 -/
134 theorem linearLoss_sub_next_importanceWeightedLoss_le
135     {Action : Type u} [DecidableEq Action]
136     (arms : Finset Action) (eta : Real)
137     (score prob next loss : Action -> Real) (chosen : Action)
138     (multiplier nextMultiplier : Real)
139     (hchosen : chosen ∈ arms) (heta : 0 < eta)
140     (hprobSimplex : FTRL.finiteSimplex arms prob)
141     (hnextSimplex : FTRL.finiteSimplex arms next)
142     (hprobPos : forall action, action ∈ arms -> 0 < prob action)
143     (hnextPos : forall action, action ∈ arms -> 0 < next action)
144     (hlossNonneg : forall action, action ∈ arms -> 0 <= loss action)
145     (hprobStationary : HalfTsallisInteriorStationary
146       arms eta score prob multiplier)
147     (hnextStationary : HalfTsallisInteriorStationary
148       arms eta
149         (fun action => score action +
150           Exp3.importanceWeightedLoss prob loss chosen action)
151         next nextMultiplier) :
152     FTRL.linearLoss arms prob
153           (Exp3.importanceWeightedLoss prob loss chosen) -
154         FTRL.linearLoss arms next
155           (Exp3.importanceWeightedLoss prob loss chosen) <=
156       2 * eta *
157         powerWeightedSquaredImportanceWeightedLoss
158           arms (1 / 2 : Real) prob loss chosen := by
159   let shift := nextMultiplier - multiplier
160   let observed := loss chosen / prob chosen
161   have hnegHalf : (-(1 / 2 : Real)) < 0 := by norm_num
162   have hprobChosen : 0 < prob chosen := hprobPos chosen hchosen
163   have hnextChosen : 0 < next chosen := hnextPos chosen hchosen
164   have hlossChosen : 0 <= loss chosen := hlossNonneg chosen hchosen
165   have hobservedNonneg : 0 <= observed :=
166     div_nonneg hlossChosen hprobChosen.le
167   have hdiff : forall action, action ∈ arms ->
168       (next action) ^ (-(1 / 2 : Real)) -
169           (prob action) ^ (-(1 / 2 : Real)) =
170         eta * Exp3.importanceWeightedLoss prob loss chosen action - shift := by
171     intro action haction
172     exact halfTsallisInteriorStationary_rpow_sub_rpow_eq
173       arms eta score
174       (Exp3.importanceWeightedLoss prob loss chosen)
175       prob next multiplier nextMultiplier
176       hprobStationary hnextStationary haction
177   have hshiftNonneg : 0 <= shift := by
178     by_contra hnot
179     have hshiftNeg : shift < 0 := lt_of_not_ge hnot
180     have hsumLt : arms.sum next < arms.sum prob :=
181       Finset.sum_lt_sum_of_nonempty ⟨chosen, hchosen⟩ (by
182         intro action haction
183         have hiwNonneg :
184             0 <= Exp3.importanceWeightedLoss prob loss chosen action :=
185           Exp3.importanceWeightedLoss_nonneg
186             (hprobSimplex.1 action haction)
187             (hlossNonneg action haction)
188         have hpowerLt :
189             (prob action) ^ (-(1 / 2 : Real)) <
190               (next action) ^ (-(1 / 2 : Real)) := by
191           have := hdiff action haction
192           nlinarith [mul_nonneg heta.le hiwNonneg]
193         exact (Real.rpow_lt_rpow_iff_of_neg
194           (hprobPos action haction) (hnextPos action haction)
195           hnegHalf).1 hpowerLt)
196     rw [hprobSimplex.2, hnextSimplex.2] at hsumLt
197     exact (lt_irrefl 1 hsumLt)
198   have hshiftLe : shift <= eta * observed := by
199     by_contra hnot
200     have hselectedLtShift : eta * observed < shift := lt_of_not_ge hnot
201     have hsumLt : arms.sum prob < arms.sum next :=
202       Finset.sum_lt_sum_of_nonempty ⟨chosen, hchosen⟩ (by
203         intro action haction
204         have hiwLe :
205             Exp3.importanceWeightedLoss prob loss chosen action <= observed := by
206           by_cases heq : chosen = action
207           · subst action
208             simp [Exp3.importanceWeightedLoss, observed]
209           · simp [Exp3.importanceWeightedLoss, heq, hobservedNonneg]
210         have hetaIWLt :
211             eta * Exp3.importanceWeightedLoss prob loss chosen action < shift :=
212           (mul_le_mul_of_nonneg_left hiwLe heta.le).trans_lt hselectedLtShift
213         have hpowerLt :
214             (next action) ^ (-(1 / 2 : Real)) <
215               (prob action) ^ (-(1 / 2 : Real)) := by
216           have := hdiff action haction
217           linarith
218         exact (Real.rpow_lt_rpow_iff_of_neg
219           (hnextPos action haction) (hprobPos action haction)
220           hnegHalf).1 hpowerLt)
221     rw [hprobSimplex.2, hnextSimplex.2] at hsumLt
222     exact (lt_irrefl 1 hsumLt)
223   have hiwChosen :
224       Exp3.importanceWeightedLoss prob loss chosen chosen = observed := by
225     simp [Exp3.importanceWeightedLoss, observed]
226   have hchosenDiff :
227       (next chosen) ^ (-(1 / 2 : Real)) -
228           (prob chosen) ^ (-(1 / 2 : Real)) =
229         eta * observed - shift := by
230     simpa [hiwChosen] using hdiff chosen hchosen
231   have hchosenDiffNonneg :
232       0 <= (next chosen) ^ (-(1 / 2 : Real)) -
233         (prob chosen) ^ (-(1 / 2 : Real)) := by
234     rw [hchosenDiff]
235     linarith
236   have hnextLe : next chosen <= prob chosen := by
237     apply (Real.rpow_le_rpow_iff_of_neg
238       hprobChosen hnextChosen hnegHalf).1
239     linarith
240   have hcurvature :=
241     sub_le_two_mul_rpow_three_halves_mul_neg_half_rpow_sub
242       hprobChosen hnextChosen hnextLe
243   have hdiffLe :
244       (next chosen) ^ (-(1 / 2 : Real)) -
245           (prob chosen) ^ (-(1 / 2 : Real)) <= eta * observed := by
246     rw [hchosenDiff]
247     linarith
248   have hlinear :
249       FTRL.linearLoss arms prob
250             (Exp3.importanceWeightedLoss prob loss chosen) -
251           FTRL.linearLoss arms next
252             (Exp3.importanceWeightedLoss prob loss chosen) =
253         (prob chosen - next chosen) * observed := by
254     unfold FTRL.linearLoss
255     rw [← Finset.sum_sub_distrib, Finset.sum_eq_single chosen]
256     · simp [Exp3.importanceWeightedLoss, observed]
257       ring
258     · intro action haction hne
259       simp [Exp3.importanceWeightedLoss, Ne.symm hne]
260     · exact fun hnotmem => (hnotmem hchosen).elim
261   have hscaledCurvature :
262       (prob chosen - next chosen) * observed <=
263         (2 * (prob chosen) ^ (3 / 2 : Real) *
264           ((next chosen) ^ (-(1 / 2 : Real)) -
265             (prob chosen) ^ (-(1 / 2 : Real)))) * observed :=
266     mul_le_mul_of_nonneg_right hcurvature hobservedNonneg
267   have hscaledDiff :
268       (2 * (prob chosen) ^ (3 / 2 : Real) *
269           ((next chosen) ^ (-(1 / 2 : Real)) -
270             (prob chosen) ^ (-(1 / 2 : Real)))) * observed <=
271         (2 * (prob chosen) ^ (3 / 2 : Real) *
272           (eta * observed)) * observed := by
273     have hcoeffNonneg :
274         0 <= 2 * (prob chosen) ^ (3 / 2 : Real) := by positivity
275     exact mul_le_mul_of_nonneg_right
276       (mul_le_mul_of_nonneg_left hdiffLe hcoeffNonneg)
277       hobservedNonneg
278   have hpow :
279       (prob chosen) ^ (3 / 2 : Real) =
280         (prob chosen) ^ (2 : Real) *
281           (prob chosen) ^ (-(1 / 2 : Real)) := by
282     rw [← Real.rpow_add hprobChosen]
283     congr 1
284     norm_num
285   rw [hlinear]
286   calc
287     (prob chosen - next chosen) * observed <=
288         (2 * (prob chosen) ^ (3 / 2 : Real) *
289           ((next chosen) ^ (-(1 / 2 : Real)) -
290             (prob chosen) ^ (-(1 / 2 : Real)))) * observed :=
291       hscaledCurvature
292     _ <= (2 * (prob chosen) ^ (3 / 2 : Real) *
293           (eta * observed)) * observed := hscaledDiff
294     _ = 2 * eta * ((loss chosen) ^ 2 *
295           (prob chosen) ^ (-(1 / 2 : Real))) := by
296       rw [hpow, Real.rpow_two]
297       dsimp only [observed]
298       field_simp [ne_of_gt hprobChosen]
299     _ = 2 * eta *
300           powerWeightedSquaredImportanceWeightedLoss
301             arms (1 / 2 : Real) prob loss chosen := by
302       rw [powerWeightedSquaredImportanceWeightedLoss_eq_selected
303         arms (1 / 2 : Real) prob loss chosen hchosen hprobChosen]

305 /--
306 Sampling-law finite-sum half-Tsallis stability bound.
307 
308 For every possible sampled action, `next chosen` carries its own updated
309 stationarity certificate.  Averaging the pathwise FTRL stability terms with
310 the current simplex masses is bounded by `2 * eta` times the half-power sum.
311 -/
312 theorem sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half
313     {Action : Type u} [DecidableEq Action]
314     (arms : Finset Action) (eta : Real)
315     (score prob loss : Action -> Real)
316     (next : Action -> Action -> Real)
317     (multiplier : Real) (nextMultiplier : Action -> Real)
318     (heta : 0 < eta)
319     (hprobSimplex : FTRL.finiteSimplex arms prob)
320     (hprobPos : forall action, action ∈ arms -> 0 < prob action)
321     (hnextSimplex : forall chosen, chosen ∈ arms ->
322       FTRL.finiteSimplex arms (next chosen))
323     (hnextPos : forall chosen, chosen ∈ arms ->
324       forall action, action ∈ arms -> 0 < next chosen action)
325     (hloss : forall action, action ∈ arms ->
326       0 <= loss action ∧ loss action <= 1)
327     (hprobStationary : HalfTsallisInteriorStationary
328       arms eta score prob multiplier)
329     (hnextStationary : forall chosen, chosen ∈ arms ->
330       HalfTsallisInteriorStationary
331         arms eta
332           (fun action => score action +
333             Exp3.importanceWeightedLoss prob loss chosen action)
334           (next chosen) (nextMultiplier chosen)) :
335     arms.sum (fun chosen =>
336         prob chosen *
337           (FTRL.linearLoss arms prob
338               (Exp3.importanceWeightedLoss prob loss chosen) -
339             FTRL.linearLoss arms (next chosen)
340               (Exp3.importanceWeightedLoss prob loss chosen))) <=
341       2 * eta * powerSum arms (1 / 2 : Real) prob := by
342   have hpathwise : forall chosen, chosen ∈ arms ->
343       FTRL.linearLoss arms prob
344             (Exp3.importanceWeightedLoss prob loss chosen) -
345           FTRL.linearLoss arms (next chosen)
346             (Exp3.importanceWeightedLoss prob loss chosen) <=
347         2 * eta *
348           powerWeightedSquaredImportanceWeightedLoss
349             arms (1 / 2 : Real) prob loss chosen := by
350     intro chosen hchosen
351     exact linearLoss_sub_next_importanceWeightedLoss_le
352       arms eta score prob (next chosen) loss chosen
353       multiplier (nextMultiplier chosen) hchosen heta
354       hprobSimplex (hnextSimplex chosen hchosen) hprobPos
355       (hnextPos chosen hchosen) (fun action haction =>
356         (hloss action haction).1)
357       hprobStationary (hnextStationary chosen hchosen)
358   have hmoment :=
359     sum_prob_mul_powerWeightedSquaredImportanceWeightedLoss_le_powerSum
360       arms (1 / 2 : Real) prob loss hprobPos hloss
361   calc
362     arms.sum (fun chosen =>
363         prob chosen *
364           (FTRL.linearLoss arms prob
365               (Exp3.importanceWeightedLoss prob loss chosen) -
366             FTRL.linearLoss arms (next chosen)
367               (Exp3.importanceWeightedLoss prob loss chosen))) <=
368         arms.sum (fun chosen =>
369           prob chosen *
370             (2 * eta *
371               powerWeightedSquaredImportanceWeightedLoss
372                 arms (1 / 2 : Real) prob loss chosen)) := by
373       apply Finset.sum_le_sum
374       intro chosen hchosen
375       exact mul_le_mul_of_nonneg_left (hpathwise chosen hchosen)
376         (hprobSimplex.1 chosen hchosen)
377     _ = 2 * eta *
378         arms.sum (fun chosen =>
379           prob chosen *
380             powerWeightedSquaredImportanceWeightedLoss
381               arms (1 / 2 : Real) prob loss chosen) := by
382       rw [Finset.mul_sum]
383       apply Finset.sum_congr rfl
384       intro chosen _hchosen
385       ring
386     _ <= 2 * eta * powerSum arms (1 - (1 / 2 : Real)) prob := by
387       exact mul_le_mul_of_nonneg_left hmoment (mul_nonneg (by norm_num) heta.le)
388     _ = 2 * eta * powerSum arms (1 / 2 : Real) prob := by norm_num
