6 /-!
7 # Recursive half-Tsallis trajectories
8 
9 This module builds the pure half-Tsallis finite-history policy and its
10 Ionescu--Tulcea trajectory.  The only selector-specific input is coordinate
11 measurability of the canonical noncomputable minimizer for every measurable
12 finite-history score.  Simplex feasibility, score recursion, policy kernels,
13 and conditional action laws are constructed locally.
14 -/

23 /-- Reusable regularity boundary for the noncomputable half-Tsallis selector.
24 It asks only that measurable finite-history score coordinates produce
25 measurable selected probability coordinates. -/
26 structure HalfTsallisFiniteHistorySelectorMeasurability
27     {Action : Type u} [MeasurableSpace Action] [DecidableEq Action]
28     (arms : Finset Action) (harms : arms.Nonempty)
29     (eta : Real) : Prop where
30   measurable_selector :
31     forall n
32       (score : History.FinitePairHistory Action Real n -> Action -> Real),
33       (forall action, action ∈ arms ->
34         Measurable (fun history => score history action)) ->
35       forall action, action ∈ arms ->
36         Measurable (fun history =>
37           halfTsallisMinimizer arms harms eta (score history) action)

39 /-- Initial pure half-Tsallis law, before any observed pair. -/
40 noncomputable def initialHalfTsallisDistribution
41     {Action : Type u} [DecidableEq Action]
42     (arms : Finset Action) (harms : arms.Nonempty)
43     (eta : Real) : Action -> Real :=
44   halfTsallisMinimizer arms harms eta (fun _ => 0)

46 theorem finiteActionDistribution_initialHalfTsallisDistribution
47     {Action : Type u} [DecidableEq Action]
48     (arms : Finset Action) (harms : arms.Nonempty)
49     (eta : Real) :
50     Exp3.FiniteActionDistribution arms
51       (initialHalfTsallisDistribution arms harms eta) := by
52   have hmin := halfTsallisMinimizer_isRegularizedMinimizer
53     arms harms eta (fun _ => 0)
54   exact {
55     nonneg := hmin.1.1
56     sum_eq_one := hmin.1.2
57   }

59 /-- Cumulative importance-weighted score through an inclusive observed pair
60 history, using the pure half-Tsallis law generated by the previous score. -/
61 noncomputable def sampledHalfTsallisHistoryScore
62     {Action : Type u} [DecidableEq Action]
63     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real) :
64     (n : Nat) -> History.FinitePairHistory Action Real n -> Action -> Real
65   | 0, history, action =>
66       Exp3.importanceWeightedLoss
67         (initialHalfTsallisDistribution arms harms eta)
68         (fun _ => (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).2)
69         (history ⟨0, Finset.mem_Iic.mpr le_rfl⟩).1 action
70   | n + 1, history, action =>
71       let previous := Exp3.previousPairHistory history
72       sampledHalfTsallisHistoryScore arms harms eta n previous action +
73         Exp3.importanceWeightedLoss
74           (halfTsallisMinimizer arms harms eta
75             (sampledHalfTsallisHistoryScore arms harms eta n previous))
76           (fun _ => (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).2)
77           (history ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩).1 action

108 /-- The generic selector contract makes the recursively accumulated score
109 measurable at every supported action coordinate. -/
110 theorem measurable_sampledHalfTsallisHistoryScore
111     {Action : Type u}
112     [MeasurableSpace Action] [MeasurableSingletonClass Action]
113     [DecidableEq Action]
114     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
115     (selector : HalfTsallisFiniteHistorySelectorMeasurability
116       arms harms eta) :
117     forall n action, action ∈ arms ->
118       Measurable (fun history : History.FinitePairHistory Action Real n =>
119         sampledHalfTsallisHistoryScore arms harms eta n history action) := by
120   intro n
121   induction n with
122   | zero =>
123       intro action _haction
124       let zeroIndex : Finset.Iic 0 := ⟨0, Finset.mem_Iic.mpr le_rfl⟩
125       have hchosen : Measurable
126           (fun history : History.FinitePairHistory Action Real 0 =>
127             (history zeroIndex).1) :=
128         measurable_fst.comp (measurable_pi_apply zeroIndex)
129       have hloss : Measurable
130           (fun history : History.FinitePairHistory Action Real 0 =>
131             (history zeroIndex).2) :=
132         measurable_snd.comp (measurable_pi_apply zeroIndex)
133       simpa [sampledHalfTsallisHistoryScore, zeroIndex] using
134         (Exp3.measurable_observedImportanceWeightedLoss
135           (fun _history : History.FinitePairHistory Action Real 0 =>
136             initialHalfTsallisDistribution arms harms eta)
137           (fun history => (history zeroIndex).1)
138           (fun history => (history zeroIndex).2)
139           action measurable_const hchosen hloss)
140   | succ n ih =>
141       intro action haction
142       let lastIndex : Finset.Iic (n + 1) :=
143         ⟨n + 1, Finset.mem_Iic.mpr le_rfl⟩
144       have hprevious : Measurable
145           (Exp3.previousPairHistory (Action := Action) (n := n)) :=
146         Exp3.measurable_previousPairHistory
147       have hscore : Measurable
148           (fun history : History.FinitePairHistory Action Real (n + 1) =>
149             sampledHalfTsallisHistoryScore arms harms eta n
150               (Exp3.previousPairHistory history) action) :=
151         (ih action haction).comp hprevious
152       have hprob : Measurable
153           (fun history : History.FinitePairHistory Action Real (n + 1) =>
154             halfTsallisMinimizer arms harms eta
155               (sampledHalfTsallisHistoryScore arms harms eta n
156                 (Exp3.previousPairHistory history)) action) :=
157         (selector.measurable_selector n
158           (sampledHalfTsallisHistoryScore arms harms eta n)
159           (fun selected hselected => ih selected hselected)
160           action haction).comp hprevious
161       have hchosen : Measurable
162           (fun history : History.FinitePairHistory Action Real (n + 1) =>
163             (history lastIndex).1) :=
164         measurable_fst.comp (measurable_pi_apply lastIndex)
165       have hloss : Measurable
166           (fun history : History.FinitePairHistory Action Real (n + 1) =>
167             (history lastIndex).2) :=
168         measurable_snd.comp (measurable_pi_apply lastIndex)
169       have hincrement : Measurable
170           (fun history : History.FinitePairHistory Action Real (n + 1) =>
171             Exp3.importanceWeightedLoss
172               (halfTsallisMinimizer arms harms eta
173                 (sampledHalfTsallisHistoryScore arms harms eta n
174                   (Exp3.previousPairHistory history)))
175               (fun _ => (history lastIndex).2)
176               (history lastIndex).1 action) :=
177         Exp3.measurable_observedImportanceWeightedLoss
178           (fun history =>
179             halfTsallisMinimizer arms harms eta
180               (sampledHalfTsallisHistoryScore arms harms eta n
181                 (Exp3.previousPairHistory history)))
182           (fun history => (history lastIndex).1)
183           (fun history => (history lastIndex).2)
184           action hprob hchosen hloss
185       simpa [sampledHalfTsallisHistoryScore, lastIndex] using
186         hscore.add hincrement

285 /-- Every successor action has the recursive pure half-Tsallis finite-action
286 law conditional on its visible pair-history prefix. -/
287 theorem sampledHalfTsallisTrajectoryMeasure_condDistrib_action
288     {Env : Type v} {Action : Type u}
289     [MeasurableSpace Env] [StandardBorelSpace Env]
290     [MeasurableSpace Action] [MeasurableSingletonClass Action]
291     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
292     (prior : Measure Env) [IsFiniteMeasure prior]
293     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
294     (selector : HalfTsallisFiniteHistorySelectorMeasurability
295       arms harms eta)
296     (environment : Thompson.MeasurableHistoryEnvironment Env Action Real)
297     (n : Nat) :
298     condDistrib
299         (fun sample : Env × ((k : Nat) -> Action × Real) =>
300           (sample.2 (n + 1)).1)
301         (fun sample => Preorder.frestrictLe n sample.2)
302         (prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
303           arms harms eta selector environment) =ᵐ[
304       (prior ⊗ₘ sampledHalfTsallisTrajectoryKernel
305         arms harms eta selector environment).map
306           (fun sample => Preorder.frestrictLe n sample.2)]
307       Exp3.finiteActionKernel arms
308         (sampledHalfTsallisHistoryDistribution arms harms eta n)
309         (sampledHalfTsallisHistoryDistributionSource
310           arms harms eta selector n) := by
311   simpa [sampledHalfTsallisTrajectoryKernel,
312     sampledHalfTsallisHistoryAlgorithm_policy] using
313     (Thompson.canonicalMeasurableEnvironmentTrajectoryMeasure_condDistrib_action
314       prior (sampledHalfTsallisHistoryAlgorithm arms harms eta selector)
315       environment n)
