10 /-!
11 # Half-Tsallis FTRL minimizer and stationarity transport
12 
13 This module connects the finite-simplex minimizer certificate used by the
14 generic FTRL route to the explicit interior stationarity certificate consumed
15 by the half-Tsallis one-step stability theorem.  The proof perturbs two simplex
16 coordinates in opposite directions, differentiates the resulting scalar
17 objective at an interior minimizer, and uses the vanishing derivative to
18 identify a common multiplier.
19 
20 Strict positivity is an explicit input.  This module does not prove existence
21 or interiority of a half-Tsallis minimizer and does not construct the stochastic
22 Tsallis-INF update.
23 -/

30 /-- The zero-sum direction that transfers mass from `j` to `i`. -/
31 def pairDirection {Action : Type u} [DecidableEq Action]
32     (i j : Action) (a : Action) : Real :=
33   (if a = i then 1 else 0) - (if a = j then 1 else 0)
34 
35 /-- Transfer scalar mass `t` from coordinate `j` to coordinate `i`. -/
36 def simplexPairShift {Action : Type u} [DecidableEq Action]
37     (p : Action -> Real) (i j : Action) (t : Real) : Action -> Real :=
38   fun a => p a + t * pairDirection i j a

41 theorem simplexPairShift_zero {Action : Type u} [DecidableEq Action]
42     (p : Action -> Real) (i j : Action) :
43     simplexPairShift p i j 0 = p := by
44   funext a
45   simp [simplexPairShift]

47 theorem sum_pairDirection_mul {Action : Type u} [DecidableEq Action]

50     arms.sum (fun a => pairDirection i j a * f a) = f i - f j := by

52   simp [hi, hj]

54 theorem sum_simplexPairShift {Action : Type u} [DecidableEq Action]
55     (arms : Finset Action) (p : Action -> Real) (i j : Action) (t : Real)
56     (hi : i ∈ arms) (hj : j ∈ arms) :
57     arms.sum (simplexPairShift p i j t) = arms.sum p := by
58   have hdir := sum_pairDirection_mul arms i j (fun _ => (1 : Real)) hi hj
59   simp only [mul_one, sub_self] at hdir
60   simp only [simplexPairShift, Finset.sum_add_distrib]
61   rw [← Finset.mul_sum, hdir]

64 theorem finiteSimplex_simplexPairShift_of_abs_lt_min
65     {Action : Type u} [DecidableEq Action]
66     (arms : Finset Action) (p : Action -> Real) (i j : Action) (t : Real)
67     (hij : i ≠ j) (hp : FTRL.finiteSimplex arms p)
68     (hi : i ∈ arms) (hj : j ∈ arms)
69     (ht : |t| < min (p i) (p j)) :
70     FTRL.finiteSimplex arms (simplexPairShift p i j t) := by
71   constructor
72   · intro a ha
73     rcases abs_lt.mp (ht.trans_le (min_le_left _ _)) with ⟨htLower, htUpper⟩
74     rcases abs_lt.mp (ht.trans_le (min_le_right _ _)) with ⟨htLower', htUpper'⟩
75     by_cases hai : a = i
76     · subst a
77       simp [simplexPairShift, pairDirection, hij]
78       linarith
79     · by_cases haj : a = j
80       · subst a
81         simp [simplexPairShift, pairDirection, hai]
82         linarith
83       · simp [simplexPairShift, pairDirection, hai, haj, hp.1 a ha]
84   · rw [sum_simplexPairShift arms p i j t hi hj, hp.2]

86 /-- The half-Tsallis objective is a linear term minus twice the square-root sum. -/
87 theorem regularizedObjective_half_eq {Action : Type u}
88     (arms : Finset Action) (eta : Real)
89     (score p : Action -> Real) :
90     FTRL.regularizedObjective arms eta
91         (negEntropyRegularizer arms (1 / 2 : Real)) score p =
92       eta * FTRL.linearLoss arms p score -
93         2 * arms.sum (fun a => Real.sqrt (p a)) + 2 := by
94   simp [FTRL.regularizedObjective, negEntropyRegularizer, entropy, powerSum,
95     Real.sqrt_eq_rpow]
96   ring

107 theorem hasDerivAt_linearLoss_simplexPairShift
108     {Action : Type u} [DecidableEq Action]
109     (arms : Finset Action) (p score : Action -> Real) (i j : Action)
110     (hi : i ∈ arms) (hj : j ∈ arms) :
111     HasDerivAt
112       (fun t => FTRL.linearLoss arms (simplexPairShift p i j t) score)
113       (score i - score j) 0 := by
114   have hsum :
115       HasDerivAt
116         (fun t => arms.sum (fun a => simplexPairShift p i j t a * score a))
117         (arms.sum (fun a => pairDirection i j a * score a)) 0 :=
118     HasDerivAt.fun_sum (fun a _ =>
119       (hasDerivAt_simplexPairShift p i j a).mul_const (score a))
120   rw [sum_pairDirection_mul arms i j score hi hj] at hsum
121   simpa [FTRL.linearLoss] using hsum

124     {Action : Type u} [DecidableEq Action]
125     (arms : Finset Action) (p : Action -> Real) (i j : Action)

136   · rw [Finset.sum_eq_single j]
137     · simp
138     · intro a _ haj
139       simp [haj]
140     · exact fun hnot => (hnot hj).elim

160 theorem hasDerivAt_regularizedObjective_half_simplexPairShift
161     {Action : Type u} [DecidableEq Action]
162     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
163     (i j : Action) (hi : i ∈ arms) (hj : j ∈ arms)
164     (hpPos : forall a, a ∈ arms -> 0 < p a) :
165     HasDerivAt
166       (fun t => FTRL.regularizedObjective arms eta
167         (negEntropyRegularizer arms (1 / 2 : Real)) score
168         (simplexPairShift p i j t))
169       (eta * (score i - score j) -
170         (1 / Real.sqrt (p i) - 1 / Real.sqrt (p j))) 0 := by
171   have hlinear :=
172     (hasDerivAt_linearLoss_simplexPairShift arms p score i j hi hj).const_mul eta
173   have hsqrt :=
174     (hasDerivAt_sum_sqrt_simplexPairShift arms p i j hi hj hpPos).const_mul 2
175   have htotal := (hlinear.sub hsqrt).const_add 2
176   have hfun :
177       (fun t => FTRL.regularizedObjective arms eta
178         (negEntropyRegularizer arms (1 / 2 : Real)) score
179         (simplexPairShift p i j t)) =
180       (fun t => eta * FTRL.linearLoss arms (simplexPairShift p i j t) score -
181         2 * arms.sum (fun a => Real.sqrt (simplexPairShift p i j t a)) + 2) := by
182     funext t
183     exact regularizedObjective_half_eq arms eta score
184       (simplexPairShift p i j t)
185   rw [hfun]
186   convert htotal using 1
187   · funext t
188     simp only [Pi.sub_apply]
189     ring
190   · ring

192 theorem isLocalMin_regularizedObjective_half_simplexPairShift
193     {Action : Type u} [DecidableEq Action]
194     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
195     (i j : Action) (hij : i ≠ j)
196     (hi : i ∈ arms) (hj : j ∈ arms)
197     (hpMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
198       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p)
199     (hpPos : forall a, a ∈ arms -> 0 < p a) :
200     IsLocalMin
201       (fun t => FTRL.regularizedObjective arms eta
202         (negEntropyRegularizer arms (1 / 2 : Real)) score
203         (simplexPairShift p i j t)) 0 := by
204   have hdelta : 0 < min (p i) (p j) :=
205     lt_min (hpPos i hi) (hpPos j hj)
206   show ∀ᶠ t in nhds 0,
207     FTRL.regularizedObjective arms eta
208         (negEntropyRegularizer arms (1 / 2 : Real)) score
209         (simplexPairShift p i j 0) <=
210       FTRL.regularizedObjective arms eta
211         (negEntropyRegularizer arms (1 / 2 : Real)) score
212         (simplexPairShift p i j t)
213   filter_upwards [Metric.ball_mem_nhds (0 : Real) hdelta] with t ht
214   have htAbs : |t| < min (p i) (p j) := by
215     simpa [Real.dist_eq] using ht
216   have hshift := finiteSimplex_simplexPairShift_of_abs_lt_min
217     arms p i j t hij hpMin.1 hi hj htAbs
218   simpa [simplexPairShift] using hpMin.2 (simplexPairShift p i j t) hshift

220 theorem halfTsallis_pairwise_stationary_of_isRegularizedMinimizer
221     {Action : Type u} [DecidableEq Action]
222     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
223     (hpMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
224       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p)
225     (hpPos : forall a, a ∈ arms -> 0 < p a)
226     {i j : Action} (hi : i ∈ arms) (hj : j ∈ arms) :
227     eta * score i - (p i) ^ (-(1 / 2 : Real)) =
228       eta * score j - (p j) ^ (-(1 / 2 : Real)) := by
229   by_cases hij : i = j
230   · subst j
231     rfl
232   have hlocal := isLocalMin_regularizedObjective_half_simplexPairShift
233     arms eta score p i j hij hi hj hpMin hpPos
234   have hderiv := hasDerivAt_regularizedObjective_half_simplexPairShift
235     arms eta score p i j hi hj hpPos
236   have hzero := hlocal.hasDerivAt_eq_zero hderiv
237   have hrpowI :
238       (p i) ^ (-(1 / 2 : Real)) = 1 / Real.sqrt (p i) := by
239     rw [Real.rpow_neg (hpPos i hi).le, ← Real.sqrt_eq_rpow]
240     simp [one_div]
241   have hrpowJ :
242       (p j) ^ (-(1 / 2 : Real)) = 1 / Real.sqrt (p j) := by
243     rw [Real.rpow_neg (hpPos j hj).le, ← Real.sqrt_eq_rpow]
244     simp [one_div]
245   rw [hrpowI, hrpowJ]
246   linarith

248 /--
249 An explicitly interior half-Tsallis simplex minimizer admits the stationarity
250 certificate consumed by the one-step stability theorem.
251 -/
252 theorem exists_halfTsallisInteriorStationary_of_isRegularizedMinimizer
253     {Action : Type u} [DecidableEq Action]
254     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
255     (hpMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
256       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p)
257     (hpPos : forall a, a ∈ arms -> 0 < p a) :
258     exists multiplier,
259       HalfTsallisInteriorStationary arms eta score p multiplier := by
260   have harms : arms.Nonempty := by
261     by_contra hnot
262     have hempty : arms = ∅ := Finset.not_nonempty_iff_eq_empty.mp hnot
263     rw [hempty] at hpMin
264     simp [FTRL.IsRegularizedMinimizer, FTRL.finiteSimplex] at hpMin
265   rcases harms with ⟨anchor, hanchor⟩
266   refine ⟨eta * score anchor - (p anchor) ^ (-(1 / 2 : Real)), ?_⟩
267   intro action haction
268   exact halfTsallis_pairwise_stationary_of_isRegularizedMinimizer
269     arms eta score p hpMin hpPos haction hanchor

271 /-- The supporting-line inequality for the square root at a positive point. -/
272 theorem two_mul_sqrt_sub_sqrt_le_sub_div_sqrt
273     {p q : Real} (hp : 0 < p) (hq : 0 <= q) :
274     2 * (Real.sqrt q - Real.sqrt p) <=
275       (q - p) / Real.sqrt p := by
276   have hsqrtP : 0 < Real.sqrt p := Real.sqrt_pos.2 hp
277   have hsqrtQ : 0 <= Real.sqrt q := Real.sqrt_nonneg q
278   have hsqrtPSq : (Real.sqrt p) ^ 2 = p := Real.sq_sqrt hp.le
279   have hsqrtQSq : (Real.sqrt q) ^ 2 = q := Real.sq_sqrt hq
280   apply (le_div_iff₀ hsqrtP).2
281   nlinarith [sq_nonneg (Real.sqrt q - Real.sqrt p)]

283 /--
284 A positive simplex point satisfying half-Tsallis stationarity globally minimizes
285 the corresponding regularized objective on the finite simplex.
286 -/
287 theorem isRegularizedMinimizer_of_halfTsallisInteriorStationary
288     {Action : Type u} [DecidableEq Action]
289     (arms : Finset Action) (eta : Real) (score p : Action -> Real)
290     (multiplier : Real)
291     (hp : FTRL.finiteSimplex arms p)
292     (hpPos : forall a, a ∈ arms -> 0 < p a)
293     (hstationary : HalfTsallisInteriorStationary
294       arms eta score p multiplier) :
295     FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
296       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p := by
297   refine ⟨hp, ?_⟩
298   intro q hq
299   have hcoord : forall a, a ∈ arms ->
300       multiplier * (q a - p a) <=
301         eta * (q a - p a) * score a -
302           2 * (Real.sqrt (q a) - Real.sqrt (p a)) := by
303     intro a ha
304     have htangent := two_mul_sqrt_sub_sqrt_le_sub_div_sqrt
305       (hpPos a ha) (hq.1 a ha)
306     have hrpow :
307         (p a) ^ (-(1 / 2 : Real)) = 1 / Real.sqrt (p a) := by
308       rw [Real.rpow_neg (hpPos a ha).le, ← Real.sqrt_eq_rpow]
309       simp [one_div]
310     have hstat := hstationary a ha
311     rw [hrpow] at hstat
312     calc
313       multiplier * (q a - p a) =
314           eta * (q a - p a) * score a -
315             (q a - p a) / Real.sqrt (p a) := by
316         rw [← hstat]
317         ring
318       _ <= eta * (q a - p a) * score a -
319             2 * (Real.sqrt (q a) - Real.sqrt (p a)) :=
320         sub_le_sub_left htangent _
321   have hsum := Finset.sum_le_sum hcoord
322   have hleft :
323       arms.sum (fun a => multiplier * (q a - p a)) = 0 := by
324     rw [← Finset.mul_sum]
325     simp only [Finset.sum_sub_distrib, hq.2, hp.2, sub_self, mul_zero]
326   have hright :
327       arms.sum (fun a =>
328         eta * (q a - p a) * score a -
329           2 * (Real.sqrt (q a) - Real.sqrt (p a))) =
330         FTRL.regularizedObjective arms eta
331             (negEntropyRegularizer arms (1 / 2 : Real)) score q -
332           FTRL.regularizedObjective arms eta
333             (negEntropyRegularizer arms (1 / 2 : Real)) score p := by
334     rw [regularizedObjective_half_eq, regularizedObjective_half_eq]
335     unfold FTRL.linearLoss
336     simp only [Finset.sum_sub_distrib]
337     have hlinear :
338         arms.sum (fun a => eta * (q a - p a) * score a) =
339           eta * arms.sum (fun a => q a * score a) -
340             eta * arms.sum (fun a => p a * score a) := by
341       simp_rw [show forall a,
342           eta * (q a - p a) * score a =
343             eta * (q a * score a - p a * score a) by
344           intro a
345           ring]
346       rw [← Finset.mul_sum, Finset.sum_sub_distrib]
347       ring
348     have hsqrt :
349         arms.sum (fun a =>
350           2 * (Real.sqrt (q a) - Real.sqrt (p a))) =
351           2 * arms.sum (fun a => Real.sqrt (q a)) -
352             2 * arms.sum (fun a => Real.sqrt (p a)) := by
353       simp_rw [mul_sub]
354       rw [Finset.sum_sub_distrib, ← Finset.mul_sum, ← Finset.mul_sum]
355     rw [hlinear, hsqrt]
356     ring
357   rw [hleft, hright] at hsum
358   linarith

379 Sampling-law half-Tsallis stability directly from concrete regularized-minimizer
380 certificates.  Common multipliers are constructed internally from interiority.
381 -/
382 theorem sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half_of_positive_minimizers
383     {Action : Type u} [DecidableEq Action]
384     (arms : Finset Action) (eta : Real)
385     (score prob loss : Action -> Real)
386     (next : Action -> Action -> Real)
387     (heta : 0 < eta)
388     (hprobMin : FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
389       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score prob)
390     (hprobPos : forall action, action ∈ arms -> 0 < prob action)
391     (hnextMin : forall chosen, chosen ∈ arms ->
392       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
393         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
394         (fun action => score action +
395           Exp3.importanceWeightedLoss prob loss chosen action)
396         (next chosen))
397     (hnextPos : forall chosen, chosen ∈ arms ->
398       forall action, action ∈ arms -> 0 < next chosen action)
399     (hloss : forall action, action ∈ arms ->
400       0 <= loss action ∧ loss action <= 1) :
401     arms.sum (fun chosen =>
402         prob chosen *
403           (FTRL.linearLoss arms prob
404               (Exp3.importanceWeightedLoss prob loss chosen) -
405             FTRL.linearLoss arms (next chosen)
406               (Exp3.importanceWeightedLoss prob loss chosen))) <=
407       2 * eta * powerSum arms (1 / 2 : Real) prob := by
408   classical
409   rcases exists_halfTsallisInteriorStationary_of_isRegularizedMinimizer
410     arms eta score prob hprobMin hprobPos with
411     ⟨multiplier, hprobStationary⟩
412   have hnextExists : forall chosen, chosen ∈ arms ->
413       exists nextMultiplier,
414         HalfTsallisInteriorStationary arms eta
415           (fun action => score action +
416             Exp3.importanceWeightedLoss prob loss chosen action)
417           (next chosen) nextMultiplier := by
418     intro chosen hchosen
419     exact exists_halfTsallisInteriorStationary_of_isRegularizedMinimizer
420       arms eta
421       (fun action => score action +
422         Exp3.importanceWeightedLoss prob loss chosen action)
423       (next chosen) (hnextMin chosen hchosen) (hnextPos chosen hchosen)
424   let nextMultiplier : Action -> Real := fun chosen =>
425     if hchosen : chosen ∈ arms then
426       Classical.choose (hnextExists chosen hchosen)
427     else 0
428   have hnextStationary : forall chosen, chosen ∈ arms ->
429       HalfTsallisInteriorStationary arms eta
430         (fun action => score action +
431           Exp3.importanceWeightedLoss prob loss chosen action)
432         (next chosen) (nextMultiplier chosen) := by
433     intro chosen hchosen
434     simp only [nextMultiplier, dif_pos hchosen]
435     exact Classical.choose_spec (hnextExists chosen hchosen)
436   exact sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half
437     arms eta score prob loss next multiplier nextMultiplier heta
438     hprobMin.1 hprobPos (fun chosen hchosen => (hnextMin chosen hchosen).1)
439     hnextPos hloss hprobStationary hnextStationary
