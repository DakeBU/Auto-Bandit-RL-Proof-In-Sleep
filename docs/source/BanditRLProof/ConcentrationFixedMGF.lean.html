3 /-!
4 # Fixed-tilt conditional MGF concentration
5 
6 This module isolates the part of a martingale Bernstein/Freedman route that does not depend on
7 the particular one-step exponential inequality.  Unlike `HasSubgaussianMGF`, the upper bound is
8 required at one fixed tilt only.  Exponential integrability at every real multiple is retained
9 because it is the regularity needed to compose kernel laws.
10 -/

23 /-- A kernel-valued MGF upper bound at one fixed tilt.  Integrability is required at every tilt
24 so that successive conditional laws can be composed without adding boundedness assumptions. -/
25 structure HasMGFUpperBoundAt (X : Ω → ℝ) (t ψ : ℝ)
26     (κ : ProbabilityTheory.Kernel Ω' Ω) (ν : Measure Ω' := by volume_tac) : Prop where
27   integrable_exp_mul : ∀ s, Integrable (fun ω ↦ exp (s * X ω)) (κ ∘ₘ ν)
28   mgf_le : ∀ᵐ ω' ∂ν, ProbabilityTheory.mgf X (κ ω') t ≤ exp ψ

45 lemma congr {Y : Ω -> Real} (h : HasMGFUpperBoundAt X t ψ κ ν)
46     (hXY : X =ᵐ[κ ∘ₘ ν] Y) : HasMGFUpperBoundAt Y t ψ κ ν where
47   integrable_exp_mul s := by
48     refine (integrable_congr ?_).mpr (h.integrable_exp_mul s)
49     filter_upwards [hXY] with omega homega

53     filter_upwards [h.mgf_le, hkernel] with omega hmgf heq
54     rw [ProbabilityTheory.mgf_congr (Filter.EventuallyEq.symm heq)]
55     exact hmgf

105 lemma integrable_exp_add_compProd
106     {η : ProbabilityTheory.Kernel (Ω' × Ω) Ω''}
107     [ProbabilityTheory.IsZeroOrMarkovKernel η]
108     (hX : HasMGFUpperBoundAt X t ψ κ ν)
109     (hY : HasMGFUpperBoundAt Y t ψY η (ν ⊗ₘ κ)) (s : ℝ) :
110     Integrable (fun ω ↦ exp (s * (X ω.1 + Y ω.2))) ((κ ⊗ₖ η) ∘ₘ ν) := by
111   by_cases hκ : ProbabilityTheory.IsSFiniteKernel κ
112   swap; · simp [hκ]
113   rcases ProbabilityTheory.eq_zero_or_isMarkovKernel η with rfl | hη
114   · simp
115   simp_rw [mul_add, exp_add]
116   refine MemLp.integrable_mul (p := 2) (q := 2) ?_ ?_
117   · have h := hX.memLp_exp_mul s 2
118     simp only [ENNReal.coe_ofNat] at h
119     have : κ ∘ₘ ν = ((κ ⊗ₖ η) ∘ₘ ν).map Prod.fst := by
120       rw [Measure.map_comp _ _ measurable_fst, ← ProbabilityTheory.Kernel.fst_eq,
121         ProbabilityTheory.Kernel.fst_compProd]
122     rwa [this, memLp_map_measure_iff h.1 measurable_fst.aemeasurable] at h
123   · have h := hY.memLp_exp_mul s 2
124     rwa [ENNReal.coe_ofNat, Measure.comp_compProd_comm, Measure.snd,
125       memLp_map_measure_iff h.1 measurable_snd.aemeasurable] at h

127 theorem add_compProd
128     {η : ProbabilityTheory.Kernel (Ω' × Ω) Ω''}
129     [ProbabilityTheory.IsZeroOrMarkovKernel η]
130     (hX : HasMGFUpperBoundAt X t ψ κ ν)
131     (hY : HasMGFUpperBoundAt Y t ψY η (ν ⊗ₘ κ)) :
132     HasMGFUpperBoundAt (fun p ↦ X p.1 + Y p.2) t (ψ + ψY) (κ ⊗ₖ η) ν := by
133   by_cases hκ : ProbabilityTheory.IsSFiniteKernel κ
134   swap; · simp [hκ]
135   refine ⟨integrable_exp_add_compProd hX hY, ?_⟩
136   filter_upwards [hX.mgf_le, hX.ae_integrable_exp_mul t,
137     Measure.ae_ae_of_ae_compProd hY.mgf_le,
138     Measure.ae_integrable_of_integrable_comp <| integrable_exp_add_compProd hX hY t]
139     with ω' hX_mgf hX_int hY_mgf h_int_mul
140   calc ProbabilityTheory.mgf (fun p ↦ X p.1 + Y p.2) ((κ ⊗ₖ η) ω') t
141   _ = ∫ x, exp (t * X x) * ∫ y, exp (t * Y y) ∂(η (ω', x)) ∂(κ ω') := by
142     simp_rw [ProbabilityTheory.mgf, mul_add, exp_add] at h_int_mul ⊢
143     simp_rw [integral_compProd h_int_mul, integral_const_mul]
144   _ ≤ ∫ x, exp (t * X x) * exp ψY ∂(κ ω') := by
145     refine integral_mono_of_nonneg ?_ (hX_int.mul_const _) ?_
146     · exact ae_of_all _ fun _ ↦ mul_nonneg (by positivity)
147         (integral_nonneg (fun _ ↦ by positivity))
148     · filter_upwards [hY_mgf] with ω hY_mgf
149       gcongr
150       simpa [ProbabilityTheory.mgf] using hY_mgf
151   _ ≤ exp (ψ + ψY) := by
152     rw [integral_mul_const, exp_add]
153     gcongr
154     simpa [ProbabilityTheory.mgf] using hX_mgf

186 lemma of_map {Ω' : Type*} {mΩ' : MeasurableSpace Ω'} {μ' : Measure Ω'}
187     {Z : Ω' → Ω} (hZ : AEMeasurable Z μ')
188     (h : HasMGFUpperBoundAt X t ψ (μ'.map Z)) :
189     HasMGFUpperBoundAt (X ∘ Z) t ψ μ' where
190   integrable_exp_mul s := by
191     have h1 := h.integrable_exp_mul s
192     rwa [integrable_map_measure h1.aestronglyMeasurable (by fun_prop)] at h1
193   mgf_le := by
194     convert h.mgf_le using 1
195     rw [ProbabilityTheory.mgf_map hZ (h.integrable_exp_mul t).1]

197 lemma id_map_iff (hX : AEMeasurable X μ) :
198     HasMGFUpperBoundAt id t ψ (μ.map X) ↔ HasMGFUpperBoundAt X t ψ μ := by

207 /-- Subtracting a deterministic log-MGF budget turns a fixed-tilt bound into
208 a unit-tilt zero-budget bound. This is the algebraic step used to iterate
209 exponential supermartingale increments. -/
210 theorem compensated (h : HasMGFUpperBoundAt X t ψ μ) :
211     HasMGFUpperBoundAt (fun ω => t * X ω - ψ) 1 0 μ := by
212   constructor
213   · intro s
214     have hint := (h.integrable_exp_mul (s * t)).const_mul (Real.exp (-s * ψ))
215     convert hint using 1
216     funext ω
217     calc
218       Real.exp (s * (t * X ω - ψ)) =
219           Real.exp (-s * ψ + (s * t) * X ω) := by congr 1; ring
220       _ = Real.exp (-s * ψ) * Real.exp ((s * t) * X ω) := by
221         rw [Real.exp_add]
222   · rw [ProbabilityTheory.mgf]
223     simp only [one_mul]
224     simp_rw [sub_eq_add_neg, Real.exp_add]
225     rw [integral_mul_const]
226     calc
227       (∫ ω, Real.exp (t * X ω) ∂μ) * Real.exp (-ψ) ≤
228           Real.exp ψ * Real.exp (-ψ) := by
229             gcongr
230             simpa [ProbabilityTheory.mgf] using h.mgf_le
231       _ = Real.exp 0 := by rw [← Real.exp_add]; simp
232 
233 end HasMGFUpperBoundAt

256 theorem add_of_hasCondMGFUpperBoundAt {Y : Ω → ℝ} {ψX ψY : ℝ}
257     (hm : m ≤ mΩ) (hX : HasMGFUpperBoundAt X t ψX (μ.trim hm))
258     (hY : HasCondMGFUpperBoundAt m hm Y t ψY μ) :
259     HasMGFUpperBoundAt (X + Y) t (ψX + ψY) μ := by

261       (@Measure.map Ω (Ω × Ω) mΩ (m.prod mΩ) (fun ω ↦ (id ω, id ω)) μ) by
262     have h_eq : X + Y = (fun p ↦ X p.1 + Y p.2) ∘ (fun ω ↦ (id ω, id ω)) := rfl
263     rw [h_eq]
264     refine HasMGFUpperBoundAt.of_map ?_ this
265     exact @Measurable.aemeasurable _ _ _ (m.prod mΩ) _ _

267   rw [hasMGFUpperBoundAt_iff_kernel] at hX ⊢
268   have hY' : Kernel.HasMGFUpperBoundAt Y t ψY (ProbabilityTheory.condExpKernel μ m)

290 theorem HasMGFUpperBoundAt.sum_of_hasCondMGFUpperBoundAt [IsZeroOrProbabilityMeasure μ]
291     (h_adapted : StronglyAdapted ℱ Y)
292     (h0 : HasMGFUpperBoundAt (Y 0) t (ψY 0) μ) (n : ℕ)
293     (h_mgf : ∀ i < n - 1,
294       HasCondMGFUpperBoundAt (ℱ i) (ℱ.le i) (Y (i + 1)) t (ψY (i + 1)) μ) :
295     HasMGFUpperBoundAt (fun ω ↦ ∑ i ∈ Finset.range n, Y i ω) t
296       (∑ i ∈ Finset.range n, ψY i) μ := by
297   induction n with
298   | zero =>
299       constructor
300       · simp
301       · simpa [ProbabilityTheory.mgf] using (measureReal_le_one (μ := μ))
302   | succ n hn =>
303     induction n with
304     | zero => simpa using h0
305     | succ n =>
306       specialize hn fun i hi ↦ h_mgf i (by omega)
307       simp_rw [Finset.sum_range_succ _ (n + 1)]
308       refine HasMGFUpperBoundAt.add_of_hasCondMGFUpperBoundAt (ℱ.le n) ?_ (h_mgf n (by omega))
309       refine HasMGFUpperBoundAt.trim (ℱ.le n) ?_ hn
310       refine Finset.measurable_fun_sum (Finset.range (n + 1)) fun j hj ↦
311         ((h_adapted j).mono (ℱ.mono ?_)).measurable
312       simp only [Finset.mem_range] at hj
313       omega

326 /--
327 Fixed-tilt tail bound with a random predictable compensator retained in the
328 event.  The conditional MGF hypotheses are imposed on the compensated
329 increments `tilt * Y i - varianceCoeff * V i`; on the event where the
330 cumulative compensator is at most `varianceBudget`, their exponential tail
331 controls the uncompensated sum.
332 -/
333 theorem measure_sum_ge_inter_sum_le_of_compensated_hasCondMGFUpperBoundAt
334     [IsZeroOrProbabilityMeasure μ]
335     (Y V : ℕ → Ω → ℝ) (n : ℕ) (tilt varianceCoeff threshold varianceBudget : ℝ)
336     (h_adapted : StronglyAdapted ℱ
337       (fun i ω => tilt * Y i ω - varianceCoeff * V i ω))
338     (h0 : HasMGFUpperBoundAt
339       (fun ω => tilt * Y 0 ω - varianceCoeff * V 0 ω) 1 0 μ)
340     (h_mgf : ∀ i < n - 1,
341       HasCondMGFUpperBoundAt (ℱ i) (ℱ.le i)
342         (fun ω => tilt * Y (i + 1) ω - varianceCoeff * V (i + 1) ω)
343         1 0 μ)
344     (htilt : 0 ≤ tilt) (hvarianceCoeff : 0 ≤ varianceCoeff) :
345     μ {ω |
346         threshold ≤ ∑ i ∈ Finset.range n, Y i ω ∧
347         (∑ i ∈ Finset.range n, V i ω) ≤ varianceBudget} ≤
348       ENNReal.ofReal (Real.exp
349         (-tilt * threshold + varianceCoeff * varianceBudget)) := by
350   let Z : ℕ → Ω → ℝ := fun i ω =>
351     tilt * Y i ω - varianceCoeff * V i ω
352   let compensatedThreshold := tilt * threshold - varianceCoeff * varianceBudget
353   let targetEvent := {ω : Ω |
354     compensatedThreshold ≤ ∑ i ∈ Finset.range n, Z i ω}
355   let sourceEvent := {ω : Ω |
356     threshold ≤ ∑ i ∈ Finset.range n, Y i ω ∧
357       (∑ i ∈ Finset.range n, V i ω) ≤ varianceBudget}
358   have hreal := measure_sum_ge_le_of_hasCondMGFUpperBoundAt
359     (μ := μ) (ℱ := ℱ) (Y := Z) (t := 1) (ψY := fun _ => 0)
360       h_adapted h0 n h_mgf compensatedThreshold (by norm_num)
361   have hreal' : μ.real targetEvent ≤
362       Real.exp (-tilt * threshold + varianceCoeff * varianceBudget) := by
363     simpa [Z, targetEvent, compensatedThreshold, Finset.sum_sub_distrib,
364       ← Finset.mul_sum, sub_eq_add_neg, add_comm] using hreal
365   have htarget : μ targetEvent ≤
366       ENNReal.ofReal (Real.exp
367         (-tilt * threshold + varianceCoeff * varianceBudget)) := by
368     rw [Measure.real] at hreal'
369     exact (ENNReal.le_ofReal_iff_toReal_le
370       (measure_ne_top μ targetEvent) (Real.exp_pos _).le).2 hreal'
371   have hsubset : sourceEvent ⊆ targetEvent := by
372     intro ω hω
373     change threshold ≤ ∑ i ∈ Finset.range n, Y i ω ∧
374       (∑ i ∈ Finset.range n, V i ω) ≤ varianceBudget at hω
375     change compensatedThreshold ≤
376       ∑ i ∈ Finset.range n, Z i ω
377     have hdeviation := mul_le_mul_of_nonneg_left hω.1 htilt
378     have hvariance := mul_le_mul_of_nonneg_left hω.2 hvarianceCoeff
379     simp only [Z, Finset.sum_sub_distrib, ← Finset.mul_sum]
380     dsimp only [compensatedThreshold]
381     linarith
382   change μ sourceEvent ≤ _
383   exact (measure_mono hsubset).trans htarget
