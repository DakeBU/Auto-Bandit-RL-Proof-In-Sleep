11 def fixedRadiusMeanEvent (X : ℕ → Ω → ℝ) (a ε : ℝ) (s : ℕ) : Set Ω :=
12   {ω | ε ≤ (∑ j ∈ range s, X (j+1) ω)/(s : ℝ)+sqrt (2*a/(s : ℝ))}

14 theorem measure_fixedRadiusMeanEvent_le (X : ℕ → Ω → ℝ)
15     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
16     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0) (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
17     (a ε : ℝ) (ha : 0 < a) (hε : 0 < ε) (s : ℕ) (hs : 2*a/ε^2 < (s : ℝ)) :
18     μ (fixedRadiusMeanEvent X a ε s) ≤ ENNReal.ofReal (occupancyTail a ε s) := by
19   have hsR : 0 < (s : ℝ) := (by positivity : 0 < 2*a/ε^2).trans hs
20   have hsN : 0 < s := Nat.cast_pos.mp hsR
21   have hsq := sq_sqrt hsR.le
22   have haSq := sq_sqrt (by positivity : 0 ≤ 2*a)
23   have hlarge : 2*a < (s : ℝ)*ε^2 := (div_lt_iff₀ (sq_pos_of_pos hε)).mp hs
24   have hpos : 0 < ε*sqrt (s : ℝ)-sqrt (2*a) := by
25     have he : (ε*sqrt (s : ℝ))^2 = ε^2*(s : ℝ) := by rw [mul_pow, hsq]
26     nlinarith [mul_pos hε (sqrt_pos.mpr hsR), sqrt_nonneg (2*a)]
27   let A := sqrt (s : ℝ)*(ε*sqrt (s : ℝ)-sqrt (2*a))
28   have hA : 0 < A := mul_pos (sqrt_pos.mpr hsR) hpos
29   have hr : (s : ℝ)*sqrt (2*a/(s : ℝ)) = sqrt (s : ℝ)*sqrt (2*a) := by
30     rw [sqrt_div (by positivity), div_eq_mul_inv]
31     field_simp
32     nlinarith
33   have hbound := measure_exists_le_independent_partialSum_ge_le_subgaussian
34     X hXm hind hmean 1 (by norm_num) hsubG s hsN A hA
35   have hevent : fixedRadiusMeanEvent X a ε s ⊆
36       {ω | ∃ i, i ≤ s ∧ A ≤ ∑ j ∈ range i, X (j+1) ω} := by
37     intro ω hω
38     refine ⟨s, le_rfl, ?_⟩
39     have h := mul_le_mul_of_nonneg_left hω hsR.le
40     change (s : ℝ)*ε ≤ (s : ℝ)*((∑ j ∈ range s, X (j+1) ω)/(s : ℝ)+sqrt (2*a/(s : ℝ))) at h
41     rw [mul_add, mul_div_cancel₀ _ hsR.ne', hr] at h
42     dsimp [A]
43     nlinarith
44   apply (measure_mono hevent).trans
45   convert hbound using 1
46   unfold occupancyTail
47   congr 2
48   dsimp [A]
49   rw [mul_pow, hsq]
50   field_simp

52 theorem sum_measureReal_fixedRadiusMeanEvent_le (X : ℕ → Ω → ℝ)
53     (hXm : ∀ i, StronglyMeasurable (X i)) (hind : iIndepFun X μ)
54     (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0) (hsubG : ∀ i, HasSubgaussianMGF (X i) 1 μ)
55     (a ε : ℝ) (ha : 0 < a) (hε : 0 < ε) (n : ℕ) :
56     (∑ i ∈ range n, μ.real (fixedRadiusMeanEvent X a ε (i+1))) ≤
57       1+(2/ε^2)*(a+sqrt (Real.pi*a)+1) := by
58   apply sum_le_occupancy_bound _ a ε ha hε (fun _ => measureReal_le_one) _ n
59   intro s hs
60   have h := ENNReal.toReal_mono ENNReal.ofReal_ne_top
61     (measure_fixedRadiusMeanEvent_le X hXm hind hmean hsubG a ε ha hε s hs)
62   have hn : 0 ≤ occupancyTail a ε s := (exp_pos _).le
63   simpa only [ENNReal.toReal_ofReal hn] using h

66     (hXm : ∀ i, StronglyMeasurable (X i)) (a ε : ℝ) (s : ℕ) :

93       ∑ i ∈ range n, μ.real (fixedRadiusMeanEvent X a ε (i+1)) := by
