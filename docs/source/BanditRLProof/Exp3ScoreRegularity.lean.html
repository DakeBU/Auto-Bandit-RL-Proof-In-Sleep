3 /-!
4 # Regularity of one-round EXP3 importance-weighted scores
5 
6 This module discharges the measurable-score and integrability premises of the
7 generated one-round EXP3 action process.  A uniform positive probability floor
8 and measurable losses in `[0, 1]` give explicit pointwise bounds for the
9 armwise, mixed first-moment, and mixed second-moment scores.
10 -/

19 /-- Measurable bounded losses and a uniform exploration floor on the finite support. -/
20 structure BoundedMeasurableLossWithProbabilityFloor
21     {History : Type u} {Action : Type v}
22     [MeasurableSpace History]
23     (arms : Finset Action) (prob loss : History -> Action -> Real)
24     (epsilon : Real) : Prop where
25   epsilon_pos : 0 < epsilon
26   prob_floor : forall history action, action ∈ arms ->
27     epsilon <= prob history action
28   measurable_loss : forall action, action ∈ arms ->
29     Measurable (fun history => loss history action)
30   loss_mem_Icc : forall history action, action ∈ arms ->
31     loss history action ∈ Set.Icc (0 : Real) 1
32 
33 theorem BoundedMeasurableLossWithProbabilityFloor.prob_pos
34     {History : Type u} {Action : Type v}
35     [MeasurableSpace History]
36     {arms : Finset Action} {prob loss : History -> Action -> Real}
37     {epsilon : Real}
38     (regularity : BoundedMeasurableLossWithProbabilityFloor
39       arms prob loss epsilon)
40     (history : History) (action : Action) (haction : action ∈ arms) :
41     0 < prob history action :=
42   regularity.epsilon_pos.trans_le
43     (regularity.prob_floor history action haction)

45 /-- A fixed-arm importance-weighted score is measurable on history/action pairs. -/
46 theorem measurable_importanceWeightedLoss_score
47     {History : Type u} {Action : Type v}
48     [MeasurableSpace History] [MeasurableSpace Action]
49     [MeasurableSingletonClass Action] [DecidableEq Action]
50     (arms : Finset Action) (prob loss : History -> Action -> Real)
51     (source : MeasurableFiniteActionDistribution arms prob)
52     (epsilon : Real)
53     (regularity : BoundedMeasurableLossWithProbabilityFloor
54       arms prob loss epsilon)
55     (action : Action) (haction : action ∈ arms) :
56     Measurable (fun sample : History × Action =>
57       importanceWeightedLoss (prob sample.1) (loss sample.1)
58         sample.2 action) := by
59   unfold importanceWeightedLoss
60   refine Measurable.ite ?_ ?_ measurable_const
61   · simpa only [Set.mem_setOf_eq] using
62       (measurable_snd (measurableSet_singleton action))
63   · exact
64       ((regularity.measurable_loss action haction).comp measurable_fst).div
65         ((source.measurable_prob action haction).comp measurable_fst)

125 /-- A fixed-arm importance-weighted score is bounded by the reciprocal floor. -/
126 theorem norm_importanceWeightedLoss_score_le_inv_floor
127     {History : Type u} {Action : Type v}
128     [MeasurableSpace History] [DecidableEq Action]
129     (arms : Finset Action) (prob loss : History -> Action -> Real)
130     (epsilon : Real)
131     (regularity : BoundedMeasurableLossWithProbabilityFloor
132       arms prob loss epsilon)
133     (history : History) (chosen action : Action) (haction : action ∈ arms) :
134     ‖importanceWeightedLoss (prob history) (loss history) chosen action‖ <=
135       1 / epsilon := by
136   have hprob_pos := regularity.prob_pos history action haction
137   have hloss := regularity.loss_mem_Icc history action haction
138   unfold importanceWeightedLoss
139   split_ifs
140   · rw [Real.norm_eq_abs, abs_of_nonneg (div_nonneg hloss.1 hprob_pos.le)]
141     exact (div_le_div_of_nonneg_right hloss.2 hprob_pos.le).trans
142       (one_div_le_one_div_of_le regularity.epsilon_pos
143         (regularity.prob_floor history action haction))
144   · simpa only [norm_zero] using
145       (one_div_pos.mpr regularity.epsilon_pos).le

147 /-- The mixed first-moment score is bounded by the reciprocal floor. -/
148 theorem norm_mixedImportanceWeightedLoss_score_le_inv_floor
149     {History : Type u} {Action : Type v}
150     [MeasurableSpace History] [DecidableEq Action]
151     (arms : Finset Action) (prob loss : History -> Action -> Real)
152     (source : MeasurableFiniteActionDistribution arms prob)
153     (epsilon : Real)
154     (regularity : BoundedMeasurableLossWithProbabilityFloor
155       arms prob loss epsilon)
156     (history : History) (chosen : Action) :
157     ‖mixedImportanceWeightedLoss arms (prob history) (loss history) chosen‖ <=
158       1 / epsilon := by
159   rw [Real.norm_eq_abs]
160   unfold mixedImportanceWeightedLoss
161   calc
162     |arms.sum (fun action =>
163         prob history action *
164           importanceWeightedLoss (prob history) (loss history) chosen action)| <=
165         arms.sum (fun action =>
166           |prob history action *
167             importanceWeightedLoss (prob history) (loss history) chosen action|) :=
168       Finset.abs_sum_le_sum_abs _ arms
169     _ <= arms.sum (fun action => prob history action * (1 / epsilon)) := by
170       apply Finset.sum_le_sum
171       intro action haction
172       have hprob_nonneg := (source.distribution history).nonneg action haction
173       have hiw_bound :=
174         norm_importanceWeightedLoss_score_le_inv_floor arms prob loss epsilon
175           regularity history chosen action haction
176       rw [Real.norm_eq_abs] at hiw_bound
177       rw [abs_mul, abs_of_nonneg hprob_nonneg]
178       exact mul_le_mul_of_nonneg_left hiw_bound hprob_nonneg
179     _ = arms.sum (prob history) * (1 / epsilon) := by
180       rw [Finset.sum_mul]
181     _ = 1 / epsilon := by
182       rw [(source.distribution history).sum_eq_one, one_mul]

223 /-- The mixed second-moment score is bounded by the square reciprocal floor. -/
224 theorem norm_mixedSquaredImportanceWeightedLoss_score_le_inv_floor_sq
225     {History : Type u} {Action : Type v}
226     [MeasurableSpace History] [DecidableEq Action]
227     (arms : Finset Action) (prob loss : History -> Action -> Real)
228     (source : MeasurableFiniteActionDistribution arms prob)
229     (epsilon : Real)
230     (regularity : BoundedMeasurableLossWithProbabilityFloor
231       arms prob loss epsilon)
232     (history : History) (chosen : Action) :
233     ‖mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
234         chosen‖ <= (1 / epsilon) ^ 2 := by
235   have hbound_nonneg : 0 <= 1 / epsilon :=
236     (one_div_pos.mpr regularity.epsilon_pos).le
237   rw [Real.norm_eq_abs]
238   unfold mixedSquaredImportanceWeightedLoss
239   rw [abs_of_nonneg]
240   · calc
241       arms.sum (fun action =>
242           prob history action *
243             (importanceWeightedLoss (prob history) (loss history)
244               chosen action) ^ 2) <=
245           arms.sum (fun action => prob history action * (1 / epsilon) ^ 2) := by
246         apply Finset.sum_le_sum
247         intro action haction
248         have hprob_nonneg := (source.distribution history).nonneg action haction
249         have hiw_nonneg := importanceWeightedLoss_nonneg (chosen := chosen)
250           hprob_nonneg (regularity.loss_mem_Icc history action haction).1
251         have hiw_le :
252             importanceWeightedLoss (prob history) (loss history) chosen action <=
253               1 / epsilon := by
254           simpa [Real.norm_eq_abs, abs_of_nonneg hiw_nonneg] using
255             norm_importanceWeightedLoss_score_le_inv_floor arms prob loss
256               epsilon regularity history chosen action haction
257         exact mul_le_mul_of_nonneg_left
258           ((sq_le_sq₀ hiw_nonneg hbound_nonneg).2 hiw_le) hprob_nonneg
259       _ = arms.sum (prob history) * (1 / epsilon) ^ 2 := by
260         rw [Finset.sum_mul]
261       _ = (1 / epsilon) ^ 2 := by
262         rw [(source.distribution history).sum_eq_one, one_mul]
263   · exact Finset.sum_nonneg fun action haction =>
264       mul_nonneg ((source.distribution history).nonneg action haction)
265         (sq_nonneg _)
