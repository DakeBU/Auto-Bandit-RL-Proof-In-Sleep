10 /-- Each selected round visits exactly one successive pre-pull count. -/
11 theorem sum_selected_pullCount {Action : Type*} [DecidableEq Action]
12     (action : ActionTrace Action) (a : Action) (f : ℕ → ℝ) (T : ℕ) :
13     (∑ t ∈ range T, if action t = a then f (pullCount action a t) else 0) =
14       ∑ s ∈ range (pullCount action a T), f s := by
15   induction T with
16   | zero => simp
17   | succ T ih =>
18       rw [sum_range_succ, ih, pullCount_succ]
19       by_cases h : action T = a
20       · simp [h, sum_range_succ]
21       · simp [h]

23 /-- Event-count transport retains the first pull, whose pre-pull count is zero.
24 The event premise must be supplied separately by the algorithm analysis. -/
25 theorem pullCount_le_one_add_eventCount {Action : Type*} [DecidableEq Action]
26     (action : ActionTrace Action) (a : Action) (P : ℕ → Prop) [DecidablePred P]
27     (T : ℕ)
28     (hselected : ∀ t < T, action t = a → 0 < pullCount action a t →
29       P (pullCount action a t)) :
30     (pullCount action a T : ℝ) ≤
31       1 + ∑ s ∈ range T, if P (s + 1) then (1 : ℝ) else 0 := by
32   let f : ℕ → ℝ := fun s => if s = 0 then 1 else if P s then 1 else 0
33   have hsum : (pullCount action a T : ℝ) = ∑ s ∈ range (pullCount action a T), f s := by
34     have hconst := sum_selected_pullCount action a (fun _ => (1 : ℝ)) T
35     have hf := sum_selected_pullCount action a f T
36     have heq : (∑ t ∈ range T, if action t = a then (1 : ℝ) else 0) =
37         ∑ t ∈ range T, if action t = a then f (pullCount action a t) else 0 := by
38       apply sum_congr rfl
39       intro t ht
40       by_cases h : action t = a
41       · by_cases hz : pullCount action a t = 0
42         · simp [h, f, hz]
43         · have hp := hselected t (mem_range.mp ht) h (Nat.pos_of_ne_zero hz)
44           simp [h, f, hz, hp]
45       · simp [h]
46     simpa using hconst.symm.trans (heq.trans hf)
47   rw [hsum]
48   have hle : ∑ s ∈ range (pullCount action a T), f s ≤ ∑ s ∈ range (T + 1), f s := by
49     apply sum_le_sum_of_subset_of_nonneg
50     · exact range_mono (Nat.le_trans (pullCount_le_time action a T) (Nat.le_succ T))
51     · intro s _ _
52       dsimp [f]
53       split_ifs <;> norm_num
54   calc
55     _ ≤ ∑ s ∈ range (T + 1), f s := hle
56     _ = _ := by rw [sum_range_succ']; simp [f, add_comm]
