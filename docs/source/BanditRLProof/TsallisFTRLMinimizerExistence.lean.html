5 /-!
6 # Existence of half-Tsallis finite-simplex minimizers
7 
8 The project simplex only constrains coordinates in an explicit `Finset`, so it
9 is not compact as a subset of the full function space when the ambient action
10 type is infinite.  We minimize instead on Mathlib's compact standard simplex
11 over the finite subtype `↥arms`, then extend the minimizer by zero.
12 -/
13

19 /-- Extend finite-subtype weights by zero outside the explicit arm set. -/
20 def extendFiniteWeights {Action : Type u} [DecidableEq Action]
21     (arms : Finset Action) (p : ↥arms -> Real) : Action -> Real :=
22   fun action => if haction : action ∈ arms then p ⟨action, haction⟩ else 0
23 
24 @[simp]
25 theorem extendFiniteWeights_apply_of_mem
26     {Action : Type u} [DecidableEq Action]
27     (arms : Finset Action) (p : ↥arms -> Real)
28     {action : Action} (haction : action ∈ arms) :
29     extendFiniteWeights arms p action = p ⟨action, haction⟩ := by
30   simp [extendFiniteWeights, haction]
31 
32 @[simp]
33 theorem extendFiniteWeights_apply_of_not_mem
34     {Action : Type u} [DecidableEq Action]
35     (arms : Finset Action) (p : ↥arms -> Real)
36     {action : Action} (haction : action ∉ arms) :
37     extendFiniteWeights arms p action = 0 := by
38   simp [extendFiniteWeights, haction]

40 theorem sum_extendFiniteWeights
41     {Action : Type u} [DecidableEq Action]
42     (arms : Finset Action) (p : ↥arms -> Real) :
43     arms.sum (extendFiniteWeights arms p) = ∑ action : ↥arms, p action := by
44   rw [← arms.sum_attach]
45   simp [extendFiniteWeights]

47 theorem finiteSimplex_extendFiniteWeights
48     {Action : Type u} [DecidableEq Action]
49     (arms : Finset Action) {p : ↥arms -> Real}
50     (hp : p ∈ stdSimplex Real ↥arms) :
51     FTRL.finiteSimplex arms (extendFiniteWeights arms p) := by
52   constructor
53   · intro action haction
54     rw [extendFiniteWeights_apply_of_mem arms p haction]
55     exact hp.1 ⟨action, haction⟩
56   · rw [sum_extendFiniteWeights arms p, hp.2]

58 theorem restrict_mem_stdSimplex_of_finiteSimplex
59     {Action : Type u} [DecidableEq Action]
60     (arms : Finset Action) {p : Action -> Real}
61     (hp : FTRL.finiteSimplex arms p) :
62     Finset.restrict arms p ∈ stdSimplex Real ↥arms := by
63   constructor
64   · intro action
65     exact hp.1 action action.2
66   · change (∑ action : ↥arms, p action) = 1
67     rw [← Finset.attach_eq_univ, Finset.sum_attach]
68     exact hp.2

70 theorem linearLoss_extendFiniteWeights_eq
71     {Action : Type u} [DecidableEq Action]
72     (arms : Finset Action) (p : ↥arms -> Real) (score : Action -> Real) :
73     FTRL.linearLoss arms (extendFiniteWeights arms p) score =
74       FTRL.linearLoss Finset.univ p (fun action => score action) := by

133 /-- The half-Tsallis objective is continuous on the finite subtype function
134 space. -/
135 theorem continuous_regularizedObjective_half_univ
136     {Action : Type u} [Fintype Action]
137     (eta : Real) (score : Action -> Real) :
138     Continuous (fun p : Action -> Real =>
139       FTRL.regularizedObjective Finset.univ eta
140         (negEntropyRegularizer Finset.univ (1 / 2 : Real)) score p) := by
141   simp_rw [regularizedObjective_half_eq, FTRL.linearLoss]

144 /-- Every nonempty explicit finite arm set admits a half-Tsallis regularized
145 minimizer.  The learning rate and finite score coordinates may be arbitrary
146 real numbers. -/
147 theorem exists_isRegularizedMinimizer_half
148     {Action : Type u} [DecidableEq Action]
149     (arms : Finset Action) (harms : arms.Nonempty)
150     (eta : Real) (score : Action -> Real) :
151     exists p,
152       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
153         arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p := by
154   let restrictedScore : ↥arms -> Real := fun action => score action
155   let objective : (↥arms -> Real) -> Real := fun p =>
156     FTRL.regularizedObjective arms.attach eta
157       (negEntropyRegularizer arms.attach (1 / 2 : Real))
158       restrictedScore p
159   have hsimplexNonempty : (stdSimplex Real ↥arms).Nonempty := by
160     let anchor : ↥arms := ⟨harms.choose, harms.choose_spec⟩
161     exact ⟨Pi.single anchor 1, single_mem_stdSimplex Real anchor⟩
162   have hcontinuous : Continuous objective := by
163     dsimp [objective, restrictedScore]
164     exact continuous_regularizedObjective_half_univ eta
165       (fun action : ↥arms => score action)
166   obtain ⟨p, hpSimplex, hpMin⟩ :=
167     (isCompact_stdSimplex Real ↥arms).exists_isMinOn
168       hsimplexNonempty hcontinuous.continuousOn
169   refine ⟨extendFiniteWeights arms p,
170     finiteSimplex_extendFiniteWeights arms hpSimplex, ?_⟩
171   intro q hq
172   have hqSimplex : Finset.restrict arms q ∈ stdSimplex Real ↥arms :=
173     restrict_mem_stdSimplex_of_finiteSimplex arms hq
174   have hmin := hpMin hqSimplex
175   dsimp [objective, restrictedScore] at hmin
176   rw [← regularizedObjective_half_extendFiniteWeights_eq arms eta score p,
177     regularizedObjective_half_restrict_eq arms eta score q] at hmin
178   exact hmin

180 /-- A fixed choice of half-Tsallis minimizer on a nonempty explicit arm set. -/
181 noncomputable def halfTsallisMinimizer
182     {Action : Type u} [DecidableEq Action]
183     (arms : Finset Action) (harms : arms.Nonempty)
184     (eta : Real) (score : Action -> Real) : Action -> Real :=
185   Classical.choose (exists_isRegularizedMinimizer_half arms harms eta score)

187 theorem halfTsallisMinimizer_isRegularizedMinimizer
188     {Action : Type u} [DecidableEq Action]
189     (arms : Finset Action) (harms : arms.Nonempty)
190     (eta : Real) (score : Action -> Real) :
191     FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
192       arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score
193       (halfTsallisMinimizer arms harms eta score) :=
194   Classical.choose_spec (exists_isRegularizedMinimizer_half arms harms eta score)

196 /-- Canonical half-Tsallis update after observing one importance-weighted loss. -/
197 noncomputable def halfTsallisUpdatedMinimizer
198     {Action : Type u} [DecidableEq Action]
199     (arms : Finset Action) (harms : arms.Nonempty)
200     (eta : Real) (score loss : Action -> Real) (chosen : Action) :
201     Action -> Real :=
202   let prob := halfTsallisMinimizer arms harms eta score
203   halfTsallisMinimizer arms harms eta (fun action =>
204     score action + Exp3.importanceWeightedLoss prob loss chosen action)
205 
206 theorem halfTsallisUpdatedMinimizer_isRegularizedMinimizer
207     {Action : Type u} [DecidableEq Action]
208     (arms : Finset Action) (harms : arms.Nonempty)
209     (eta : Real) (score loss : Action -> Real) (chosen : Action) :
210     FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
211       arms eta (negEntropyRegularizer arms (1 / 2 : Real))
212       (fun action => score action +
213         Exp3.importanceWeightedLoss
214           (halfTsallisMinimizer arms harms eta score) loss chosen action)
215       (halfTsallisUpdatedMinimizer arms harms eta score loss chosen) := by
216   exact halfTsallisMinimizer_isRegularizedMinimizer arms harms eta
217     (fun action => score action +
218       Exp3.importanceWeightedLoss
219         (halfTsallisMinimizer arms harms eta score) loss chosen action)

221 /-- The sampling-law one-step stability endpoint with both current and updated
222 minimizers selected internally. -/
223 theorem sum_halfTsallisMinimizer_mul_linearLoss_sub_updated_le_powerSum_half
224     {Action : Type u} [DecidableEq Action]
225     (arms : Finset Action) (harms : arms.Nonempty)
226     (eta : Real) (score loss : Action -> Real)
227     (heta : 0 < eta)
228     (hloss : forall action, action ∈ arms ->
229       0 <= loss action ∧ loss action <= 1) :
230     let prob := halfTsallisMinimizer arms harms eta score
231     let next := fun chosen =>
232       halfTsallisUpdatedMinimizer arms harms eta score loss chosen
233     arms.sum (fun chosen =>
234         prob chosen *
235           (FTRL.linearLoss arms prob
236               (Exp3.importanceWeightedLoss prob loss chosen) -
237             FTRL.linearLoss arms (next chosen)
238               (Exp3.importanceWeightedLoss prob loss chosen))) <=
239       2 * eta * powerSum arms (1 / 2 : Real) prob := by
240   dsimp only
241   exact
242     sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half_of_minimizers
243       arms eta score (halfTsallisMinimizer arms harms eta score) loss
244       (fun chosen =>
245         halfTsallisUpdatedMinimizer arms harms eta score loss chosen)
246       heta
247       (halfTsallisMinimizer_isRegularizedMinimizer arms harms eta score)
248       (fun chosen _ =>
249         halfTsallisUpdatedMinimizer_isRegularizedMinimizer
250           arms harms eta score loss chosen)
251       hloss

253 /-- A nonempty finite arm set admits a strictly positive half-Tsallis
254 minimizer together with its common stationarity multiplier. -/
255 theorem exists_halfTsallisInteriorStationary_minimizer
256     {Action : Type u} [DecidableEq Action]
257     (arms : Finset Action) (harms : arms.Nonempty)
258     (eta : Real) (score : Action -> Real) :
259     exists p multiplier,
260       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
261           arms eta (negEntropyRegularizer arms (1 / 2 : Real)) score p ∧
262         (forall action, action ∈ arms -> 0 < p action) ∧
263         HalfTsallisInteriorStationary arms eta score p multiplier := by
264   obtain ⟨p, hpMin⟩ := exists_isRegularizedMinimizer_half arms harms eta score
265   obtain ⟨multiplier, hstationary⟩ :=
266     exists_halfTsallisInteriorStationary_of_isRegularizedMinimizer_auto
267       arms eta score p hpMin
268   exact ⟨p, multiplier, hpMin,
269     isRegularizedMinimizer_pos arms eta score p hpMin, hstationary⟩
