9 theorem integral_positive_tail_le_two_sqrt (f : ℝ → ℝ) (hf : Measurable f)
10     (hn : ∀ t, 0 ≤ f t) (h1 : ∀ t, f t ≤ 1)
11     (c : ℝ) (hc : 0 < c) (ht : ∀ t, 0 < t → f t ≤ c/t^2) :
12     ∫ t in Ioi 0, f t ≤ 2*sqrt c := by
13   let a := sqrt c
14   have ha : 0 < a := sqrt_pos.mpr hc
15   have heq (t : ℝ) (ht : 0 ≤ t) : c / t^2 = c * t^(-2 : ℝ) := by
16     rw [Real.rpow_neg ht, Real.rpow_two]
17     rfl
18   have hpow : IntegrableOn (fun t : ℝ => c*t^(-2 : ℝ)) (Ioi a) :=
19     (integrableOn_Ioi_rpow_of_lt (by norm_num : (-2 : ℝ) < -1) ha).const_mul c
20   have hsmall : IntegrableOn f (Ioc 0 a) := by
21     apply (integrableOn_const (C := (1 : ℝ)) (hs := by simp)).mono' hf.aestronglyMeasurable
22     exact Eventually.of_forall (fun t => by simpa [abs_of_nonneg (hn t)] using h1 t)
23   have hlarge : IntegrableOn f (Ioi a) := by
24     apply hpow.mono' hf.aestronglyMeasurable
25     filter_upwards [ae_restrict_mem measurableSet_Ioi] with t htmem
26     simpa only [Real.norm_eq_abs, abs_of_nonneg (hn t), ← heq t (ha.trans htmem).le] using ht t (ha.trans htmem)
27   have hall : IntegrableOn f (Ioi 0) := by
28     rw [← Ioc_union_Ioi_eq_Ioi ha.le]
29     exact hsmall.union hlarge
30   have hs : ∫ t in Ioc 0 a, f t ≤ a := by
31     calc
32       _ ≤ ∫ t in Ioc 0 a, (1 : ℝ) := integral_mono hsmall (integrableOn_const (hs := by simp))
33         (fun t => h1 t)
34       _ = a := by simp [ha.le]
35   have hl : ∫ t in Ioi a, f t ≤ c/a := by
36     calc
37       _ ≤ ∫ t in Ioi a, c*t^(-2 : ℝ) := integral_mono_ae hlarge hpow (by
38         filter_upwards [ae_restrict_mem measurableSet_Ioi] with t htmem
39         simpa only [← heq t (ha.trans htmem).le] using ht t (ha.trans htmem))
40       _ = c/a := by
41         rw [integral_const_mul, integral_Ioi_rpow_of_lt (by norm_num : (-2 : ℝ) < -1) ha]
42         norm_num [Real.rpow_neg_one, div_eq_mul_inv]
43   have hsplit : (∫ t in Ioc 0 a, f t) + (∫ t in Ioi a, f t) = ∫ t in Ioi 0, f t := by
44     rw [← setIntegral_union Ioc_disjoint_Ioi_same measurableSet_Ioi hsmall hlarge,
45       Ioc_union_Ioi_eq_Ioi ha.le]
46   rw [← hsplit]
47   have hsq : a^2 = c := sq_sqrt hc.le
48   have hdiv : c/a = a := (div_eq_iff ha.ne').mpr (by nlinarith)
49   rw [hdiv] at hl
50   dsimp [a] at hs hl
51   linarith
