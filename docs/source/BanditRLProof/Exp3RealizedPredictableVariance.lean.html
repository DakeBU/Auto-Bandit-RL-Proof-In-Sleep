18 /-- Exact centered second moment of a bounded loss under a finite action law. -/
19 noncomputable def selectedLossCenteredSecondMoment
20     {History : Type u} {Action : Type v} [DecidableEq Action]
21     (arms : Finset Action) (prob loss : History → Action → Real)
22     (history : History) : Real :=
23   let mean := arms.sum fun action => prob history action * loss history action
24   arms.sum fun selected =>
25     prob history selected * (loss history selected - mean) ^ 2

27 theorem measurable_selectedLossCenteredSecondMoment
28     {History : Type u} {Action : Type v}
29     [MeasurableSpace History]
30     [MeasurableSpace Action] [MeasurableSingletonClass Action]
31     [DecidableEq Action]
32     (arms : Finset Action) (prob loss : History → Action → Real)
33     (source : MeasurableFiniteActionDistribution arms prob)
34     (epsilon : Real)
35     (regularity : BoundedMeasurableLossWithProbabilityFloor
36       arms prob loss epsilon) :
37     Measurable (selectedLossCenteredSecondMoment arms prob loss) := by
38   let mean := fun history : History =>
39     arms.sum fun action => prob history action * loss history action
40   have hmean : Measurable mean := by
41     refine Finset.measurable_sum arms fun action haction => ?_
42     exact (source.measurable_prob action haction).mul
43       (regularity.measurable_loss action haction)
44   refine Finset.measurable_sum arms fun selected hselected => ?_
45   simpa [selectedLossCenteredSecondMoment, mean] using
46     (source.measurable_prob selected hselected).mul
47       (((regularity.measurable_loss selected hselected).sub hmean).pow_const 2)

58 /-- A probability-weighted centered second moment of `[0,1]` losses is at
59 most one. -/
60 theorem selectedLossCenteredSecondMoment_le_one
61     {History : Type u} {Action : Type v} [DecidableEq Action]
62     (arms : Finset Action) (prob loss : History → Action → Real)
63     (history : History) (hdist : FiniteActionDistribution arms (prob history))
64     (hloss : ∀ action ∈ arms, loss history action ∈ Set.Icc (0 : Real) 1) :
65     selectedLossCenteredSecondMoment arms prob loss history ≤ 1 := by
66   let mean := arms.sum fun action => prob history action * loss history action
67   have hmean_nonneg : 0 ≤ mean := by
68     exact Finset.sum_nonneg fun action haction =>
69       mul_nonneg (hdist.nonneg action haction) (hloss action haction).1
70   have hmean_le_one : mean ≤ 1 := by
71     calc
72       mean ≤ arms.sum (fun action => prob history action * 1) := by
73         exact Finset.sum_le_sum fun action haction =>
74           mul_le_mul_of_nonneg_left (hloss action haction).2
75             (hdist.nonneg action haction)
76       _ = 1 := by simpa using hdist.sum_eq_one
77   unfold selectedLossCenteredSecondMoment
78   change
79     (∑ selected ∈ arms,
80       prob history selected * (loss history selected - mean) ^ 2) ≤ 1
81   calc
82     (∑ selected ∈ arms,
83         prob history selected * (loss history selected - mean) ^ 2) ≤
84         ∑ selected ∈ arms, prob history selected * 1 := by
85       exact Finset.sum_le_sum fun selected hselected => by
86         have hl := hloss selected hselected
87         have hl_nonneg : 0 ≤ loss history selected := hl.1
88         have hl_le_one : loss history selected ≤ 1 := hl.2
89         have hlower : -1 ≤ loss history selected - mean := by linarith
90         have hupper : loss history selected - mean ≤ 1 := by linarith
91         have honeplus :
92             0 ≤ 1 + (loss history selected - mean) := by
93           linarith
94         have hproduct :
95             0 ≤ (1 - (loss history selected - mean)) *
96               (1 + (loss history selected - mean)) :=
97           mul_nonneg (sub_nonneg.mpr hupper) honeplus
98         have hsquare : (loss history selected - mean) ^ 2 ≤ 1 := by
99           nlinarith
100         exact mul_le_mul_of_nonneg_left hsquare
101           (hdist.nonneg selected hselected)
102     _ = 1 := by simpa using hdist.sum_eq_one

104 /-- For `[0,1]` losses, the exact selected-loss variance is at most the
105 unweighted armwise loss mass. -/
106 theorem selectedLossCenteredSecondMoment_le_lossMass
107     {History : Type u} {Action : Type v} [DecidableEq Action]
108     (arms : Finset Action) (prob loss : History → Action → Real)
109     (history : History) (hdist : FiniteActionDistribution arms (prob history))
110     (hloss : ∀ action ∈ arms, loss history action ∈ Set.Icc (0 : Real) 1) :
111     selectedLossCenteredSecondMoment arms prob loss history ≤
112       arms.sum fun action => loss history action := by
113   let mean := arms.sum fun action => prob history action * loss history action
114   have hmean_nonneg : 0 ≤ mean := by
115     exact Finset.sum_nonneg fun action haction =>
116       mul_nonneg (hdist.nonneg action haction) (hloss action haction).1
117   have hmean_le_one : mean ≤ 1 := by
118     calc
119       mean ≤ arms.sum (fun action => prob history action * 1) := by
120         exact Finset.sum_le_sum fun action haction =>
121           mul_le_mul_of_nonneg_left (hloss action haction).2
122             (hdist.nonneg action haction)
123       _ = 1 := by simpa using hdist.sum_eq_one
124   have hcenter :
125       selectedLossCenteredSecondMoment arms prob loss history =
126         arms.sum (fun action =>
127           prob history action * (loss history action) ^ 2) - mean ^ 2 := by
128     have hterm (action : Action) :
129         prob history action * (loss history action - mean) ^ 2 =
130           (prob history action * (loss history action) ^ 2 -
131             2 * mean * (prob history action * loss history action)) +
132             mean ^ 2 * prob history action := by
133       ring
134     change
135       (∑ action ∈ arms,
136         prob history action * (loss history action - mean) ^ 2) =
137       (∑ action ∈ arms, prob history action * loss history action ^ 2) -
138         mean ^ 2
139     simp_rw [hterm]
140     rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
141       ← Finset.mul_sum, ← Finset.mul_sum, hdist.sum_eq_one]
142     change
143       (∑ action ∈ arms, prob history action * loss history action ^ 2) -
144           2 * mean * mean + mean ^ 2 * 1 =
145         (∑ action ∈ arms, prob history action * loss history action ^ 2) -
146           mean ^ 2
147     ring
148   rw [hcenter]
149   calc
150     (∑ action ∈ arms, prob history action * loss history action ^ 2) -
151           mean ^ 2 ≤
152         ∑ action ∈ arms, prob history action * loss history action ^ 2 := by
153       nlinarith [sq_nonneg mean]
154     _ ≤ ∑ action ∈ arms, loss history action := by
155       exact Finset.sum_le_sum fun action haction => by
156         have hp_nonneg := hdist.nonneg action haction
157         have hp_le_one : prob history action ≤ 1 := by
158           calc
159             prob history action ≤ arms.sum (prob history) :=
160               Finset.single_le_sum
161                 (fun candidate hcandidate => hdist.nonneg candidate hcandidate)
162                 haction
163             _ = 1 := hdist.sum_eq_one
164         have hl := hloss action haction
165         have hl_sq_le : (loss history action) ^ 2 ≤ loss history action := by
166           nlinarith [mul_nonneg hl.1 (sub_nonneg.mpr hl.2)]
167         calc
168           prob history action * loss history action ^ 2 ≤
169               1 * loss history action ^ 2 :=
170             mul_le_mul_of_nonneg_right hp_le_one (sq_nonneg _)
171           _ ≤ loss history action := by simpa using hl_sq_le

174 theorem finiteActionSelectedLossDeviation_hasMGFUpperBoundAt_variance
175     {History : Type u} {Action : Type v}

181     (epsilon : Real)

183       arms prob loss epsilon)

185     let mean := arms.sum fun action =>
186       prob history action * loss history action
187     Concentration.HasMGFUpperBoundAt
188       (fun selected => loss history selected - mean)
189       tilt
190       (tilt ^ 2 * selectedLossCenteredSecondMoment arms prob loss history)
191       (finiteActionMeasure arms (prob history)) := by

194     prob history action * loss history action

196   let variance := selectedLossCenteredSecondMoment arms prob loss history

199   have hmean_nonneg : 0 ≤ mean := by
200     exact Finset.sum_nonneg fun action haction =>
201       mul_nonneg (hdist.nonneg action haction)
202         (regularity.loss_mem_Icc history action haction).1
203   have hmean_le_one : mean ≤ 1 := by
204     calc
205       mean ≤ arms.sum (fun action => prob history action * 1) := by
206         exact Finset.sum_le_sum fun action haction =>
207           mul_le_mul_of_nonneg_left
208             (regularity.loss_mem_Icc history action haction).2
209             (hdist.nonneg action haction)
210       _ = 1 := by simpa using hdist.sum_eq_one
211   have hX_bound (selected : Action) (hselected : selected ∈ arms) :
212       |X selected| ≤ 1 := by
213     rw [abs_le]
214     constructor
215     · have hloss_nonneg :=
216         (regularity.loss_mem_Icc history selected hselected).1
217       linarith
218     · have hloss_le :=
219         (regularity.loss_mem_Icc history selected hselected).2
220       linarith
221   constructor
222   · intro s
223     rw [finiteActionMeasure]
224     refine integrable_finset_sum_measure.2 fun action haction => ?_
225     exact (integrable_dirac (by simp)).smul_measure ENNReal.ofReal_ne_top
226   · rw [ProbabilityTheory.mgf,
227       integral_finiteActionMeasure_eq_sum arms (prob history) hdist]
228     have hexp (selected : Action) (hselected : selected ∈ arms) :
229         Real.exp (tilt * X selected) ≤
230           1 + tilt * X selected + (tilt * X selected) ^ 2 := by
231       apply Concentration.exp_le_one_add_self_add_sq_of_abs_le_one
232       rw [abs_mul, abs_of_nonneg htilt_nonneg]
233       calc
234         tilt * |X selected| ≤ tilt * 1 :=
235           mul_le_mul_of_nonneg_left (hX_bound selected hselected) htilt_nonneg
236         _ ≤ 1 := by simpa using htilt_le_one
237     calc
238       arms.sum (fun selected =>
239           prob history selected * Real.exp (tilt * X selected)) ≤
240           arms.sum (fun selected =>
241             prob history selected *
242               (1 + tilt * X selected + (tilt * X selected) ^ 2)) := by
243         exact Finset.sum_le_sum fun selected hselected =>
244           mul_le_mul_of_nonneg_left (hexp selected hselected)
245             (hdist.nonneg selected hselected)
246       _ = 1 +
247           tilt * arms.sum (fun selected => prob history selected * X selected) +
248           tilt ^ 2 *
249             arms.sum (fun selected => prob history selected * X selected ^ 2) := by
250         have hterm (selected : Action) :
251             prob history selected *
252                 (1 + tilt * X selected + (tilt * X selected) ^ 2) =
253               (prob history selected +
254                 tilt * (prob history selected * X selected)) +
255                 tilt ^ 2 *
256                   (prob history selected * X selected ^ 2) := by
257           ring
258         simp_rw [hterm, Finset.sum_add_distrib, ← Finset.mul_sum,
259           hdist.sum_eq_one]
260       _ = 1 + tilt ^ 2 * variance := by
261         have hmean_zero :
262             arms.sum (fun selected => prob history selected * X selected) = 0 := by
263           dsimp only [X]
264           simp_rw [mul_sub]
265           rw [Finset.sum_sub_distrib, ← Finset.sum_mul, hdist.sum_eq_one]
266           change mean - 1 * mean = 0
267           ring
268         rw [hmean_zero, mul_zero, add_zero]
269         rfl
270       _ ≤ Real.exp (tilt ^ 2 * variance) := by
271         simpa [add_comm] using Real.add_one_le_exp (tilt ^ 2 * variance)
