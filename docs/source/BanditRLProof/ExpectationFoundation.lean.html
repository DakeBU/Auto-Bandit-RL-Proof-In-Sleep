5 # Minimal expectation and integration foundations
6 
7 This module introduces the first integration canary through the lower Lebesgue
8 integral of a pull-event indicator.  It deliberately avoids Bochner
9 expectation, probability measures, conditional expectation, filtrations,
10 kernels, and concentration assumptions.
11 -/

20 /--
21 The lower integral of the indicator of a measurable pull event is the measure
22 of that event.
23 
24 This is the `EXP-INDICATOR-PULL` canary.  It uses an arbitrary measure and an
25 `ENNReal` indicator, so it does not choose a Bochner expectation or probability
26 measure interface yet.
27 -/
28 theorem lintegral_actionTrace_eval_eq_indicator_one
29     {Omega : Type u} {Action : Type v}
30     [MeasurableSpace Omega] [MeasurableSpace Action]
31     [MeasurableSingletonClass Action]
32     (mu : Measure Omega)
33     (action : Omega -> ActionTrace Action)
34     (haction : forall t : Nat,
35       Measurable (fun omega : Omega => action omega t))
36     (a : Action) (t : Nat) :
37     MeasureTheory.lintegral mu
38       (fun omega : Omega =>
39         (({omega' : Omega | action omega' t = a} : Set Omega).indicator
40           (1 : Omega -> ENNReal)) omega)
41       =
42     mu {omega : Omega | action omega t = a} := by
43   simpa using
44     (@MeasureTheory.lintegral_indicator_one
45       Omega _ mu {omega : Omega | action omega t = a}
46       (measurableSet_actionTrace_eval_eq action haction a t))
47 
48 end BanditRLProof
