5 /-!
6 # Obstruction to refining the current symmetrized half-Tsallis stability term
7 
8 The shifted ordinary-importance-weighted moments have the paper-shaped
9 sampled-action average, but the current local stability quantity
10 `<p - p_next, hatLoss>` is not the conjugate-potential stability used in the
11 Tsallis-INF proof.  This module gives a fully rational two-arm counterexample:
12 the current and sampled-update distributions are strict simplex minimizers and
13 the losses lie in `[0,1]`, yet the proposed refined averaged upper bound fails.
14 
15 The counterexample closes the direct averaged-stability diagnostic.  It does
16 not obstruct the paper's conjugate-potential route.
17 -/

183 /--
184 Even with strict positive simplex minimizers and `[0,1]` losses, the current
185 sampled-action average of `<p - p_next, hatLoss>` can exceed the locally scaled
186 paper coefficient `eta * sum sqrt(p) * (1-p) + 2 * eta^2`.
187 -/
188 theorem exists_minimizer_counterexample_to_refinedAveragedStability :
189     ∃ (eta : Real) (score prob loss : Fin 2 -> Real)
190       (next : Fin 2 -> Fin 2 -> Real),
191       0 < eta ∧
192       eta <= 1 ∧
193       FTRL.finiteSimplex Finset.univ prob ∧
194       (∀ action ∈ (Finset.univ : Finset (Fin 2)), 0 < prob action) ∧
195       (∀ action ∈ (Finset.univ : Finset (Fin 2)),
196         0 <= loss action ∧ loss action <= 1) ∧
197       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex Finset.univ)
198         Finset.univ eta (negEntropyRegularizer Finset.univ (1 / 2 : Real))
199         score prob ∧
200       (∀ chosen ∈ (Finset.univ : Finset (Fin 2)),
201         ∀ action ∈ (Finset.univ : Finset (Fin 2)),
202           0 < next chosen action) ∧
203       (∀ chosen ∈ (Finset.univ : Finset (Fin 2)),
204         FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex Finset.univ)
205           Finset.univ eta (negEntropyRegularizer Finset.univ (1 / 2 : Real))
206           (fun action => score action +
207             Exp3.importanceWeightedLoss prob loss chosen action)
208           (next chosen)) ∧
209       eta * (Finset.univ : Finset (Fin 2)).sum (fun action =>
210           Real.sqrt (prob action) * (1 - prob action)) + 2 * eta ^ 2 <
211         (Finset.univ : Finset (Fin 2)).sum (fun chosen =>
212           prob chosen *
213             (FTRL.linearLoss Finset.univ prob
214                 (Exp3.importanceWeightedLoss prob loss chosen) -
215               FTRL.linearLoss Finset.univ (next chosen)
216                 (Exp3.importanceWeightedLoss prob loss chosen))) := by
217   refine ⟨counterEta, counterScore, counterProb, counterLoss, counterNext,
218     counterEta_pos, counterEta_le_one, counterProb_simplex, counterProb_pos,
219     counterLoss_mem_Icc, counterProb_minimizer, counterNext_pos,
220     counterNext_minimizer, ?_⟩
221   norm_num [counterArms, counterEta, counterProb, counterLoss, counterNext,
222     FTRL.linearLoss, Exp3.importanceWeightedLoss, Fin.sum_univ_two,
223     sqrt_49, sqrt_576, sqrt_625]
