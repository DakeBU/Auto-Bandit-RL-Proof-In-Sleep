6 # L2 control of an indicator-restricted integral
7 
8 This is the nonnegative `2,2` Holder specialization used to turn an event
9 probability bound and an exact second moment into an expected overflow bound.
10 -/

13 open scoped ENNReal

19 /-- A nonnegative `L2` function restricted to a measurable event is controlled
20 by its exact second moment and the square root of the event's real mass. -/
21 theorem integral_indicator_le_sqrt_secondMoment_mul_sqrt_real_measure
22     {Omega : Type u} [MeasurableSpace Omega]
23     (mu : Measure Omega) [IsFiniteMeasure mu]
24     (f : Omega -> Real)
25     (hf_nonneg : forall omega, 0 <= f omega)
26     (hf : MemLp f 2 mu)
27     (event : Set Omega) (hevent : MeasurableSet event) :
28     integral mu (event.indicator f) <=
29       Real.sqrt (integral mu (fun omega => f omega ^ 2)) *
30         Real.sqrt (mu.real event) := by
31   let indicatorOne : Omega -> Real := event.indicator (fun _ => 1)
32   have hpq : (2 : Real).HolderConjugate 2 := by
33     rw [Real.holderConjugate_iff]
34     norm_num
35   have hindicatorOne_nonneg : forall omega, 0 <= indicatorOne omega := by
36     intro omega
37     by_cases homega : omega ∈ event <;> simp [indicatorOne, homega]
38   have hindicatorOne : MemLp indicatorOne 2 mu := by
39     exact memLp_indicator_const 2 hevent 1
40       (Or.inr (measure_lt_top mu event).ne)
41   have hf_realTwo : MemLp f (ENNReal.ofReal (2 : Real)) mu := by
42     norm_num
43     exact hf
44   have hindicatorOne_realTwo :
45       MemLp indicatorOne (ENNReal.ofReal (2 : Real)) mu := by
46     norm_num
47     exact hindicatorOne
48   have hholder :=
49     integral_mul_le_Lp_mul_Lq_of_nonneg hpq
50       (Filter.Eventually.of_forall hf_nonneg)
51       (Filter.Eventually.of_forall hindicatorOne_nonneg)
52       hf_realTwo hindicatorOne_realTwo
53   have hmul :
54       (fun omega => f omega * indicatorOne omega) = event.indicator f := by
55     funext omega
56     by_cases homega : omega ∈ event <;> simp [indicatorOne, homega]
57   have hpow :
58       (fun omega => indicatorOne omega ^ (2 : Real)) =
59         event.indicator (fun _ => (1 : Real)) := by
60     funext omega
61     by_cases homega : omega ∈ event <;> simp [indicatorOne, homega]
62   have hintegralIndicatorOne :
63       integral mu (event.indicator (fun _ => (1 : Real))) = mu.real event := by
64     simpa using (integral_indicator_one (μ := mu) hevent)
65   rw [hmul, hpow, hintegralIndicatorOne] at hholder
66   simp_rw [Real.rpow_two] at hholder
67   simpa only [Real.sqrt_eq_rpow] using hholder
68 
69 end BanditRLProof
