23 /-- Conditional Jensen turns an exponentially integrable real martingale
24 into a nonnegative exponential submartingale. -/
25 theorem submartingale_exp_of_martingale (hS : Martingale S F μ)
26     (hint : ∀ i, Integrable (fun ω => exp (S i ω)) μ) :
27     Submartingale (fun i ω => exp (S i ω)) F μ := by
28   refine ⟨fun i => continuous_exp.comp_stronglyMeasurable (hS.stronglyAdapted i),
29     fun i j hij => ?_, hint⟩
30   have hj := convexOn_exp.map_condExp_le_univ (F.le i)
31     continuous_exp.lowerSemicontinuous (hS.integrable j) (hint j)
32   filter_upwards [hj, hS.condExp_ae_eq hij] with ω hj hEq
33   simpa only [Function.comp_apply, hEq] using hj

35 /-- Finite-time maximal Chernoff bound with no union-bound cardinality factor.
36 The terminal MGF supplies the variance budget; all-time exponential
37 integrability is explicit for the Jensen producer. -/
38 theorem measure_exists_le_martingale_ge_le_exp (hS : Martingale S F μ)
39     (hint : ∀ i t, Integrable (fun ω => exp (t * S i ω)) μ)
40     (n : ℕ) (c : ℝ≥0) (hmgf : HasSubgaussianMGF (S n) c μ)
41     (ε t : ℝ) (ht : 0 < t) :
42     μ {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω} ≤
43       ENNReal.ofReal (exp (-t * ε + (c : ℝ) * t ^ 2 / 2)) := by
44   have hscaled : Martingale (fun i ω => t * S i ω) F μ := hS.smul t
45   have hexp := submartingale_exp_of_martingale hscaled (fun i => hint i t)
46   let a : ℝ≥0 := ⟨exp (t * ε), (exp_pos _).le⟩
47   let B : Set Ω := {ω | (a : ℝ) ≤
48     (range (n + 1)).sup' nonempty_range_add_one (fun i => exp (t * S i ω))}
49   have hsub : {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω} ⊆ B := by
50     rintro ω ⟨i, hi, hε⟩
51     exact (exp_le_exp.mpr (mul_le_mul_of_nonneg_left hε ht.le)).trans
52       (le_sup' (f := fun j => exp (t * S j ω)) (mem_range.mpr (Nat.lt_succ_of_le hi)))
53   have hd : (a : ℝ≥0∞) * μ B ≤ ENNReal.ofReal (exp ((c : ℝ) * t ^ 2 / 2)) := by
54     refine (maximal_ineq hexp (fun i ω => (exp_pos _).le) n).trans ?_
55     apply ENNReal.ofReal_le_ofReal
56     exact (setIntegral_le_integral (hint n t)
57       (Filter.Eventually.of_forall fun ω => (exp_pos _).le)).trans (hmgf.mgf_le t)
58   have hbound := (mul_le_mul_left' (measure_mono hsub) (a : ℝ≥0∞)).trans hd
59   have hr := ENNReal.toReal_mono (ENNReal.ofReal_ne_top) hbound
60   have hr' : exp (t * ε) * (μ {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω}).toReal ≤
61       exp ((c : ℝ) * t ^ 2 / 2) := by
62     simpa [ENNReal.toReal_mul, a, (exp_pos _).le] using hr
63   have hreal : (μ {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω}).toReal ≤
64       exp (-t * ε + (c : ℝ) * t ^ 2 / 2) := by
65     have hdiv : (μ {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω}).toReal ≤
66         exp ((c : ℝ) * t ^ 2 / 2) / exp (t * ε) :=
67       (le_div_iff₀ (exp_pos (t * ε))).2
68       (by simpa [mul_comm] using hr')
69     rw [← exp_sub] at hdiv
70     convert hdiv using 1 <;> congr 1 <;> ring
71   simpa using ENNReal.ofReal_le_ofReal hreal

73 /-- Optimized finite maximal subgaussian bound. -/
74 theorem measure_exists_le_martingale_ge_le_subgaussian (hS : Martingale S F μ)
75     (hint : ∀ i t, Integrable (fun ω => exp (t * S i ω)) μ)
76     (n : ℕ) (c : ℝ≥0) (hc : 0 < (c : ℝ))
77     (hmgf : HasSubgaussianMGF (S n) c μ) (ε : ℝ) (hε : 0 < ε) :
78     μ {ω | ∃ i, i ≤ n ∧ ε ≤ S i ω} ≤
79       ENNReal.ofReal (exp (-(ε ^ 2) / (2 * (c : ℝ)))) := by
80   have h := measure_exists_le_martingale_ge_le_exp hS hint n c hmgf ε
81     (ε / (c : ℝ)) (div_pos hε hc)
82   have heq : -(ε / (c : ℝ)) * ε + (c : ℝ) * (ε / (c : ℝ)) ^ 2 / 2 =
83       -(ε ^ 2) / (2 * (c : ℝ)) := by
84     field_simp [ne_of_gt hc]
85     <;> ring
86   simpa only [heq] using h

88 /-- Source Theorem 9.2 shape for independent centered subgaussian increments.
89 The source variance is `c = σ²`; the partial sum uses X1 through Xn.
90 Centering and coordinate measurability are explicit model contracts. -/
91 theorem measure_exists_le_independent_partialSum_ge_le_subgaussian
92     (X : ℕ → Ω → ℝ) (hXm : ∀ i, StronglyMeasurable (X i))
93     (hind : iIndepFun X μ) (hmean : ∀ i, ∫ ω, X i ω ∂μ = 0)
94     (c : ℝ≥0) (hc : 0 < (c : ℝ))
95     (hsubG : ∀ i, HasSubgaussianMGF (X i) c μ)
96     (n : ℕ) (hn : 0 < n) (ε : ℝ) (hε : 0 < ε) :
97     μ {ω | ∃ i, i ≤ n ∧ ε ≤ ∑ j ∈ range i, X (j + 1) ω} ≤
98       ENNReal.ofReal (exp (-(ε ^ 2) / (2 * (n : ℝ) * (c : ℝ)))) := by
99   have hdiff : MartingaleDiff.SuccMartingaleDifference μ (Filtration.natural X hXm) X := {
100     stronglyAdapted := Filtration.stronglyAdapted_natural hXm
101     integrable := fun i => (hsubG i).integrable
102     condExp_succ_eq_zero := fun i => by
103       simpa only [hmean] using hind.condExp_natural_ae_eq_of_lt hXm (Nat.lt_succ_self i)
104   }
105   have hmart := MartingaleDiff.martingale_partialSumsSucc_of_succMartingaleDifference μ hdiff
106   have hsum (m : ℕ) : HasSubgaussianMGF (MartingaleDiff.partialSumsSucc X m)
107       ((m : ℝ≥0) * c) μ := by
108     have h := HasSubgaussianMGF.sum_of_iIndepFun
109       (hind.precomp Nat.succ_injective) (s := range m) (c := fun _ => c)
110       (fun i _ => hsubG (i + 1))
111     have heq : MartingaleDiff.partialSumsSucc X m =
112         (fun ω => ∑ i ∈ range m, X (i + 1) ω) := by
113       funext ω
114       simp [MartingaleDiff.partialSumsSucc, sum_apply]
115     rw [heq]
116     simpa using h
117   have h := measure_exists_le_martingale_ge_le_subgaussian hmart
118     (fun i t => (hsum i).integrable_exp_mul t) n ((n : ℝ≥0) * c)
119     (by simpa using mul_pos (Nat.cast_pos.mpr hn : 0 < (n : ℝ)) hc)
120     (hsum n) ε hε
121   simpa [MartingaleDiff.partialSumsSucc, sum_apply, mul_assoc] using h
