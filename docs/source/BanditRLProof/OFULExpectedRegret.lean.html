227 /--
228 Generic expectation assembly: the all-round gap is charged by the standard
229 budget off the violation set and by the deterministic envelope on it.
230 -/
231 theorem integral_canonicalHistoryTrajectorySumRangeAllGap_le_standard_add_envelope_mul_real_measure
232     {K : Nat} {Feature : Type u}
233     [Fintype Feature]
234     (mu : Measure (Nat -> Fin K × Real)) [IsProbabilityMeasure mu]
235     (lambda : Real)
236     (thetaStar : Feature -> Real)
237     (actionFeature : Fin K -> Feature -> Real)
238     (R delta S : Real) (hS : 0 <= S)
239     (horizon : Nat) (L2 : Real)
240     (hactionFeatureBound : forall action,
241       dotProduct (actionFeature action) (actionFeature action) <= L2)
242     (comparator : Nat -> Fin K)
243     (htheta : euclideanLength thetaStar <= S) :
244     integral mu
245         (canonicalHistoryTrajectorySumRangeAllGap
246           thetaStar actionFeature horizon comparator) <=
247       standardScalarAllRoundGapBound
248           (Feature := Feature) R delta lambda S horizon L2 +
249         standardScalarAllRoundGapEnvelope S horizon L2 *
250           mu.real
251             (canonicalHistoryTrajectorySumRangeAllGapStandardViolationSet
252               lambda thetaStar actionFeature R delta S horizon L2
253                 comparator) := by
254   let gap :=
255     canonicalHistoryTrajectorySumRangeAllGap
256       thetaStar actionFeature horizon comparator
257   let budget :=
258     standardScalarAllRoundGapBound
259       (Feature := Feature) R delta lambda S horizon L2
260   let envelope := standardScalarAllRoundGapEnvelope S horizon L2
261   let bad :=
262     canonicalHistoryTrajectorySumRangeAllGapStandardViolationSet
263       lambda thetaStar actionFeature R delta S horizon L2 comparator
264   let overflow : (Nat -> Fin K × Real) -> Real :=
265     bad.indicator (fun _trajectory => envelope)
266   have hbad : MeasurableSet bad := by
267     exact
268       measurableSet_canonicalHistoryTrajectorySumRangeAllGapStandardViolationSet
269         lambda thetaStar actionFeature R delta S horizon L2 comparator
270   have hgap : Integrable gap mu := by
271     exact
272       integrable_canonicalHistoryTrajectorySumRangeAllGap
273         mu thetaStar actionFeature S hS horizon L2 hactionFeatureBound
274           comparator htheta
275   have hoverflow : Integrable overflow mu := by
276     exact (integrable_const envelope).indicator hbad
277   have hbudget : 0 <= budget := by
278     exact
279       standardScalarAllRoundGapBound_nonneg
280         (Feature := Feature) R delta lambda S hS horizon L2
281   have hpoint : forall trajectory, gap trajectory <= budget + overflow trajectory := by
282     intro trajectory
283     by_cases htrajectory : trajectory ∈ bad
284     · have hgapEnvelope : gap trajectory <= envelope := by
285         exact (le_abs_self _).trans
286           (abs_canonicalHistoryTrajectorySumRangeAllGap_le_envelope
287             thetaStar actionFeature S hS horizon L2 hactionFeatureBound
288               comparator htheta trajectory)
289       calc
290         gap trajectory <= envelope := hgapEnvelope
291         _ <= budget + envelope := le_add_of_nonneg_left hbudget
292         _ = budget + overflow trajectory := by
293           simp [overflow, Set.indicator_of_mem htrajectory]
294     · have hgapBudget : gap trajectory <= budget := by
295         apply le_of_not_gt
296         simpa [bad, budget, gap,
297           canonicalHistoryTrajectorySumRangeAllGapStandardViolationSet,
298           canonicalHistoryTrajectorySumRangeAllGap] using htrajectory
299       calc
300         gap trajectory <= budget := hgapBudget
301         _ = budget + overflow trajectory := by
302           simp [overflow, Set.indicator_of_notMem htrajectory]
303   calc
304     integral mu gap <=
305         integral mu (fun trajectory => budget + overflow trajectory) := by
306       exact integral_mono hgap ((integrable_const budget).add hoverflow) hpoint
307     _ = budget + integral mu overflow := by
308       rw [integral_add (integrable_const budget) hoverflow, integral_const]
309       simp [MeasureTheory.probReal_univ]
310     _ = budget + envelope * mu.real bad := by
311       congr 1
312       change integral mu (bad.indicator (fun _trajectory => envelope)) =
313         envelope * mu.real bad
314       rw [integral_indicator hbad, setIntegral_const]
315       simp [Measure.real, smul_eq_mul, mul_comm]
