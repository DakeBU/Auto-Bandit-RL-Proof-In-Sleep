4 /-!
5 # One-round conditional transport for half-Tsallis stability
6 
7 This module transports the compiled finite sampling-law half-Tsallis stability
8 bound through an identified conditional action law.  It also exposes a
9 canonical generated-action consumer for the fixed half-Tsallis minimizer.
10 
11 The canonical minimizer is selected with `Classical.choose`.  Its coordinate
12 measurability, and measurability/integrability of the score involving the
13 chosen updated minimizer, remain explicit inputs rather than hidden claims.
14 -/

23 /-- The one-round FTRL stability score generated by an importance-weighted
24 loss estimate. -/
25 noncomputable def importanceWeightedStabilityScore
26     {History : Type u} {Action : Type v}
27     (arms : Finset Action)
28     (prob loss : History -> Action -> Real)
29     (next : History -> Action -> Action -> Real)
30     (sample : History × Action) : Real :=
31   FTRL.linearLoss arms (prob sample.1)
32       (Exp3.importanceWeightedLoss
33         (prob sample.1) (loss sample.1) sample.2) -
34     FTRL.linearLoss arms (next sample.1 sample.2)
35       (Exp3.importanceWeightedLoss
36         (prob sample.1) (loss sample.1) sample.2)
37 
38 /-- The pointwise half-Tsallis upper bound for the sampling-law averaged
39 one-round stability score. -/
40 noncomputable def halfPowerStabilityBound
41     {History : Type u} {Action : Type v}
42     (arms : Finset Action) (eta : Real)
43     (prob : History -> Action -> Real) (history : History) : Real :=
44   2 * eta * powerSum arms (1 / 2 : Real) (prob history)

46 /--
47 An identified finite conditional action law transports the pointwise
48 half-Tsallis minimizer stability theorem to a one-round integral inequality.
49 
50 The score and bound integrability hypotheses are the exact analytic boundary
51 needed by Bochner integral monotonicity.  No measurability of a particular
52 minimizer selector is inferred here.
53 -/
54 theorem integral_importanceWeightedStabilityScore_le_integral_halfPowerStabilityBound_of_condDistrib_of_minimizers
55     {Omega : Type u} {History : Type v} {Action : Type w}
56     [MeasurableSpace Omega] [MeasurableSpace History]
57     [MeasurableSpace Action] [MeasurableSingletonClass Action]
58     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
59     (mu : Measure Omega) [IsFiniteMeasure mu]
60     (history : Omega -> History) (hhistory : Measurable history)
61     (action : Omega -> Action) (haction : Measurable action)
62     (arms : Finset Action) (eta : Real)
63     (score prob loss : History -> Action -> Real)
64     (next : History -> Action -> Action -> Real)
65     (policy : Kernel History Action) [IsMarkovKernel policy]
66     (hpolicy : policy =ᵐ[mu.map history]
67       fun h => Exp3.finiteActionMeasure arms (prob h))
68     (hcond : condDistrib action history mu =ᵐ[mu.map history] policy)
69     (heta : 0 < eta)
70     (hprobMin : forall h,
71       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
72         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
73         (score h) (prob h))
74     (hnextMin : forall h chosen, chosen ∈ arms ->
75       FTRL.IsRegularizedMinimizer (FTRL.finiteSimplex arms)
76         arms eta (negEntropyRegularizer arms (1 / 2 : Real))
77         (fun candidate => score h candidate +
78           Exp3.importanceWeightedLoss
79             (prob h) (loss h) chosen candidate)
80         (next h chosen))
81     (hloss : forall h candidate, candidate ∈ arms ->
82       0 <= loss h candidate ∧ loss h candidate <= 1)
83     (hscore : Measurable
84       (importanceWeightedStabilityScore arms prob loss next))
85     (hIntegrable : Integrable
86       (importanceWeightedStabilityScore arms prob loss next)
87       (mu.map history ⊗ₘ policy))
88     (hboundIntegrable : Integrable
89       (halfPowerStabilityBound arms eta prob) (mu.map history)) :
90     integral mu (fun omega =>
91         importanceWeightedStabilityScore arms prob loss next
92           (history omega, action omega)) <=
93       integral (mu.map history) (halfPowerStabilityBound arms eta prob) := by
94   have hdist : forall h, Exp3.FiniteActionDistribution arms (prob h) := by
95     intro h
96     exact {
97       nonneg := (hprobMin h).1.1
98       sum_eq_one := (hprobMin h).1.2
99     }
100   have htransport :=
101     Exp3.integral_historyAction_eq_integral_sum_of_condDistrib_ae_eq_finiteActionMeasure
102       mu history hhistory action haction arms prob hdist policy hpolicy hcond
103       (importanceWeightedStabilityScore arms prob loss next)
104       hscore hIntegrable
105   have hsumEq :
106       (fun h => integral (policy h) (fun chosen =>
107         importanceWeightedStabilityScore arms prob loss next (h, chosen))) =ᵐ[
108         mu.map history]
109       fun h => arms.sum (fun chosen =>
110         prob h chosen *
111           importanceWeightedStabilityScore arms prob loss next (h, chosen)) := by
112     filter_upwards [hpolicy] with h hh
113     rw [hh, Exp3.integral_finiteActionMeasure_eq_sum
114       arms (prob h) (hdist h)]
115   have hsumIntegrable : Integrable (fun h => arms.sum (fun chosen =>
116       prob h chosen *
117         importanceWeightedStabilityScore arms prob loss next (h, chosen)))
118       (mu.map history) :=
119     hIntegrable.integral_compProd.congr hsumEq
120   have hpoint : forall h,
121       arms.sum (fun chosen => prob h chosen *
122           importanceWeightedStabilityScore arms prob loss next (h, chosen)) <=
123         halfPowerStabilityBound arms eta prob h := by
124     intro h
125     simpa [importanceWeightedStabilityScore, halfPowerStabilityBound] using
126       sum_prob_mul_linearLoss_sub_next_importanceWeightedLoss_le_powerSum_half_of_minimizers
127         arms eta (score h) (prob h) (loss h) (next h) heta
128         (hprobMin h) (hnextMin h) (hloss h)
129   calc
130     integral mu (fun omega =>
131         importanceWeightedStabilityScore arms prob loss next
132           (history omega, action omega)) =
133         integral (mu.map history) (fun h => arms.sum (fun chosen =>
134           prob h chosen *
135             importanceWeightedStabilityScore arms prob loss next
136               (h, chosen))) := htransport
137     _ <= integral (mu.map history)
138         (halfPowerStabilityBound arms eta prob) :=
139       integral_mono hsumIntegrable hboundIntegrable hpoint

196 /-- The fixed half-Tsallis minimizer viewed as a history-indexed action
197 distribution. -/
198 noncomputable def halfTsallisHistoryMinimizer
199     {History : Type u} {Action : Type v} [DecidableEq Action]
200     (arms : Finset Action) (harms : arms.Nonempty)
201     (eta : Real) (score : History -> Action -> Real) :
202     History -> Action -> Real :=
203   fun history => halfTsallisMinimizer arms harms eta (score history)
204 
205 /-- The fixed importance-weighted half-Tsallis update viewed as a
206 history/action-indexed family. -/
207 noncomputable def halfTsallisHistoryUpdatedMinimizer
208     {History : Type u} {Action : Type v} [DecidableEq Action]
209     (arms : Finset Action) (harms : arms.Nonempty)
210     (eta : Real) (score loss : History -> Action -> Real) :
211     History -> Action -> Action -> Real :=
212   fun history chosen =>
213     halfTsallisUpdatedMinimizer arms harms eta
214       (score history) (loss history) chosen

216 /-- Coordinate measurability is the only missing contract for turning the
217 fixed history-indexed minimizer into the project's finite-action kernel
218 source.  Distribution feasibility follows from the minimizer certificate. -/
219 noncomputable def measurableFiniteActionDistribution_halfTsallisHistoryMinimizer
220     {History : Type u} {Action : Type v}
221     [MeasurableSpace History] [DecidableEq Action]
222     (arms : Finset Action) (harms : arms.Nonempty)
223     (eta : Real) (score : History -> Action -> Real)
224     (hmeasurable : forall action, action ∈ arms -> Measurable (fun history =>
225       halfTsallisHistoryMinimizer arms harms eta score history action)) :
226     Exp3.MeasurableFiniteActionDistribution arms
227       (halfTsallisHistoryMinimizer arms harms eta score) where
228   distribution history := by
229     have hmin := halfTsallisMinimizer_isRegularizedMinimizer
230       arms harms eta (score history)
231     exact {
232       nonneg := hmin.1.1
233       sum_eq_one := hmin.1.2
234     }
235   measurable_prob := hmeasurable

237 /--
238 Canonical generated-action one-round stability endpoint.
239 
240 The minimizer certificates and finite-distribution laws are internal.  The
241 remaining caller contracts expose exactly what is not yet proved for the
242 `Classical.choose` selector: coordinate measurability and regularity of the
243 updated stability score.
244 -/
245 theorem actionProcess_integral_halfTsallisHistoryStability_le_integral_halfPowerStabilityBound
246     {History : Type u} {Action : Type v}
247     [MeasurableSpace History] [MeasurableSpace Action]
248     [MeasurableSingletonClass Action]
249     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
250     (historyMu : Measure History) [IsFiniteMeasure historyMu]
251     (arms : Finset Action) (harms : arms.Nonempty)
252     (eta : Real) (score loss : History -> Action -> Real)
253     (hmeasurable : forall action, action ∈ arms -> Measurable (fun history =>
254       halfTsallisHistoryMinimizer arms harms eta score history action))
255     (heta : 0 < eta)
256     (hloss : forall history candidate, candidate ∈ arms ->
257       0 <= loss history candidate ∧ loss history candidate <= 1)
258     (hscore : Measurable (importanceWeightedStabilityScore arms
259       (halfTsallisHistoryMinimizer arms harms eta score) loss
260       (halfTsallisHistoryUpdatedMinimizer arms harms eta score loss)))
261     (hIntegrable : Integrable (importanceWeightedStabilityScore arms
262       (halfTsallisHistoryMinimizer arms harms eta score) loss
263       (halfTsallisHistoryUpdatedMinimizer arms harms eta score loss))

266         (measurableFiniteActionDistribution_halfTsallisHistoryMinimizer
267           arms harms eta score hmeasurable)))

269       (halfTsallisHistoryMinimizer arms harms eta score)) historyMu) :

275         (halfTsallisHistoryMinimizer arms harms eta score) loss
