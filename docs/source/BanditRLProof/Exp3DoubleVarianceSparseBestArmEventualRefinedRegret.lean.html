3 /-!
4 # Eventual refined best-arm sparse EXP3 tail
5 
6 For fixed model parameters, the four deterministic large-horizon inequalities
7 used by the exact double-predictable-variance sparse EXP3 schedule eventually
8 hold automatically. This module therefore removes the coarse `T + 1` fallback
9 eventually, identifies the best-arm threshold with `16 * gamma_T * T`, and
10 reuses the existing off-sparsity, residual, and practical tails under the same
11 horizon-indexed generated trajectory measures.
12 -/

26 private theorem eventually_const_le_natCast_pow_three (c : Real) :
27     ∀ᶠ n : Nat in atTop, c <= (n : Real) ^ 3 := by
28   obtain ⟨N, hN⟩ := exists_nat_ge c
29   filter_upwards [eventually_ge_atTop (max N 1)] with n hn
30   have hNn : N <= n := (Nat.le_max_left N 1).trans hn
31   have h1n : 1 <= n := (Nat.le_max_right N 1).trans hn
32   have hcast : c <= (n : Real) := hN.trans (by exact_mod_cast hNn)
33   have hn0 : 0 <= (n : Real) := by positivity
34   have hn1 : (1 : Real) <= n := by exact_mod_cast h1n
35   have hone_le_sq : (1 : Real) <= (n : Real) * n := by nlinarith
36   calc
37     c <= (n : Real) := hcast
38     _ = (n : Real) * 1 := by ring
39     _ <= (n : Real) * ((n : Real) * n) :=
40       mul_le_mul_of_nonneg_left hone_le_sq hn0
41     _ = (n : Real) ^ 3 := by ring
