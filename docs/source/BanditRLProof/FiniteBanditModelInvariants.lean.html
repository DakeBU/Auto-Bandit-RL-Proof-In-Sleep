7 # Finite-bandit model invariants
8 
9 This module contains model-semantic facts about the local
10 `FiniteBanditModel.bestArm` selector.  It stays below regret, expectation,
11 filtrations, kernels, and concentration.
12 -/
13 
14 namespace BanditRLProof

16 namespace FiniteBanditModel

18 private theorem mean_le_foldl_select
19     {K : Nat} (mean : Fin K -> Rat) (init : Fin K) :
20     forall l : List (Fin K),
21       (forall a : Fin K, a ∈ l ->
22         mean a <=
23           mean
24             (l.foldl
25               (fun best arm : Fin K =>
26                 if mean best < mean arm then arm else best)
27               init)) /\
28       mean init <=
29         mean
30           (l.foldl
31             (fun best arm : Fin K =>
32               if mean best < mean arm then arm else best)
33             init)
34   | [] => by
35       simp
36   | arm :: rest => by
37       let select :=
38         fun best arm : Fin K =>
39           if mean best < mean arm then arm else best
40       let next := select init arm
41       have ih := mean_le_foldl_select (mean := mean) next rest
42       have harm_next : mean arm <= mean next := by
43         by_cases hlt : mean init < mean arm
44         · simp [next, select, hlt]
45         · simp [next, select, hlt, le_of_not_gt hlt]
46       constructor
47       · intro a ha
48         simp only [List.mem_cons] at ha
49         rcases ha with ha | ha
50         · subst ha
51           exact le_trans harm_next ih.2
52         · exact ih.1 a ha
53       · have hinit_next : mean init <= mean next := by
54           by_cases hlt : mean init < mean arm
55           · simpa [next, select, hlt] using le_of_lt hlt
56           · simp [next, select, hlt]
57         exact le_trans hinit_next ih.2
58 
59 /--
60 The mean of every arm is at most the mean of the model's selected best arm.
61 
62 This is the `FINITE-BANDIT-BESTARM-DOMINATES` model-invariant leaf.  It is a
63 semantic fact about the local finite-arm selector only; it does not prove gap
64 nonnegativity or any expectation/concentration statement.
65 -/
66 theorem mean_le_bestArm_mean
67     {K : Nat}
68     (model : FiniteBanditModel K) (a : Fin K) :
69     model.mean a <= model.mean model.bestArm := by
70   unfold bestArm
71   exact
72     (mean_le_foldl_select
73       (mean := model.mean)
74       (init := ⟨0, model.hK⟩)
75       (List.finRange K)).1 a (List.mem_finRange a)
76 
77 /--
78 Every local model gap is nonnegative.
79 
80 This is the `FINITE-BANDIT-GAP-NONNEG` model-invariant leaf.  It consumes only
81 `FiniteBanditModel.mean_le_bestArm_mean` and the local `gap` definition; it does
82 not prove or use any expectation, filtration, or concentration statement.
83 -/
84 theorem gap_nonneg
85     {K : Nat}
86     (model : FiniteBanditModel K) (a : Fin K) :
87     (0 : Rat) <= model.gap a := by
88   by_cases h : a = model.bestArm
89   · simp [gap, h]
90   · simp [gap, h, bestMean,
91       sub_nonneg.mpr (mean_le_bestArm_mean model a)]
92 
93 /--
94 The maximum local arm gap over the finite arm set.
95 
96 This is a deterministic finite-model constant only. It does not introduce
97 probability, expectation, concentration, or algorithmic behavior.
98 -/
99 noncomputable def maxGap
100     {K : Nat}
101     (model : FiniteBanditModel K) : Rat := by
102   classical
103   refine Finset.sup'
104     (s := (Finset.univ : Finset (Fin K))) ?_
105     (fun a : Fin K => model.gap a)
106   use model.bestArm
107   exact Finset.mem_univ model.bestArm
108 
109 /--
110 Every local model gap is bounded by `FiniteBanditModel.maxGap`.
111 
112 This is the finite max-gap adapter used by sharper ETC suffix bounds.
113 -/
114 theorem gap_le_maxGap
115     {K : Nat}
116     (model : FiniteBanditModel K) (a : Fin K) :
117     model.gap a <= model.maxGap := by
118   unfold maxGap
119   exact Finset.le_sup' (f := fun a : Fin K => model.gap a)
120     (Finset.mem_univ a)
121 
122 /-- The finite maximum gap is nonnegative. -/
123 theorem maxGap_nonneg
124     {K : Nat}
125     (model : FiniteBanditModel K) :
126     (0 : Rat) <= model.maxGap := by
127   simpa [FiniteBanditModel.gap_bestArm] using
128     (FiniteBanditModel.gap_le_maxGap model model.bestArm)
