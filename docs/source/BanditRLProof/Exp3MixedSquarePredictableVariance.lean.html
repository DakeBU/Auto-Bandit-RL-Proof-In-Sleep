19 /-- Exact centered second moment of the mixed importance-weighted square under
20 a finite sampling distribution. -/
21 noncomputable def mixedSquaredEstimatorCenteredSecondMoment
22     {History : Type u} {Action : Type v} [DecidableEq Action]
23     (arms : Finset Action) (prob loss : History -> Action -> Real)
24     (history : History) : Real :=
25   arms.sum (fun chosen =>
26     prob history chosen *
27       (mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
28           chosen -
29         arms.sum (fun action => (loss history action) ^ 2)) ^ 2)

34     [MeasurableSpace Action] [MeasurableSingletonClass Action]

38     (epsilon : Real)
39     (regularity : BoundedMeasurableLossWithProbabilityFloor
40       arms prob loss epsilon) :
41     Measurable (mixedSquaredEstimatorCenteredSecondMoment arms prob loss) := by
42   let mean := fun history : History =>
43     arms.sum (fun action => (loss history action) ^ 2)
44   have hmean : Measurable mean := by
45     refine Finset.measurable_sum arms fun action haction => ?_
46     exact (regularity.measurable_loss action haction).pow_const 2
47   have hrawPair := measurable_mixedSquaredImportanceWeightedLoss_score
48     arms prob loss source epsilon regularity
49   refine Finset.measurable_sum arms fun chosen hchosen => ?_
50   have hraw : Measurable (fun history =>
51       mixedSquaredImportanceWeightedLoss arms (prob history) (loss history)
52         chosen) :=
53     hrawPair.comp (measurable_id.prodMk measurable_const)
54   simpa [mixedSquaredEstimatorCenteredSecondMoment, mean] using
55     (source.measurable_prob chosen hchosen).mul
56       ((hraw.sub hmean).pow_const 2)
