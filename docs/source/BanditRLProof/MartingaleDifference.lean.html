4 # Martingale-difference witness surface
5 
6 This module exposes a small finite-prefix contract for martingale-difference
7 increments.  It deliberately stays below optional stopping or final regret
8 theorems: the witness records integrability, adaptedness, and succ-indexed
9 conditional mean-zero facts in the shape used by the local bandit
10 filtration/concentration route.
11 -/

18 /--
19 Global succ-indexed martingale-difference witness.
20 
21 This is the all-time version of `SuccMartingaleDifferencePrefix`.  It records
22 the local hypotheses needed to build the Mathlib martingale of partial sums.
23 -/
24 structure SuccMartingaleDifference
25     {Omega : Type u} [mOmega : MeasurableSpace Omega]
26     (mu : Measure Omega)
27     (F : Filtration Nat mOmega)
28     (Y : Nat -> Omega -> Real) : Prop where
29   stronglyAdapted : StronglyAdapted F Y
30   integrable : forall t, Integrable (Y t) mu
31   condExp_succ_eq_zero :
32     forall i,
33       Filter.EventuallyEq (ae mu)
34         (condExp (F i) mu (Y (i + 1)))
35         (fun _omega => (0 : Real))

37 /--
38 Finite-prefix succ-indexed martingale-difference witness.
39 
40 For a process `Y`, this records that the prefix up to `n` is integrable and
41 adapted, and that each later increment `Y (i + 1)` has zero conditional
42 expectation against filtration level `i`.  This is the local bridge between the
43 compiled conditional-mean-zero leaves and later martingale or concentration
44 consumers.
45 -/
46 structure SuccMartingaleDifferencePrefix
47     {Omega : Type u} [mOmega : MeasurableSpace Omega]
48     (mu : Measure Omega)
49     (F : Filtration Nat mOmega)
50     (Y : Nat -> Omega -> Real)
51     (n : Nat) : Prop where
52   stronglyAdapted : StronglyAdapted F Y
53   integrable : forall t, t < n -> Integrable (Y t) mu
54   condExp_succ_eq_zero :
55     forall i, i + 1 < n ->
56       Filter.EventuallyEq (ae mu)
57         (condExp (F i) mu (Y (i + 1)))
58         (fun _omega => (0 : Real))

145 /--
146 Centered reward process obtained by subtracting a predictable or otherwise
147 chosen baseline from a raw reward process.
148 
149 The baseline is allowed to be random; the martingale-difference builder below
150 therefore asks directly for adaptedness, integrability, and conditional
151 mean-zero of this centered process.
152 -/
153 def centeredRewardProcess
154     {Omega : Type u}
155     (reward baseline : Nat -> Omega -> Real) :
156     Nat -> Omega -> Real :=
157   fun t omega => reward t omega - baseline t omega
158

212 /--
213 Partial sums of a succ-indexed martingale-difference process.
214 
215 The sum starts at `Y 1`, so the increment from time `i` to `i + 1` is
216 definitionally `Y (i + 1)`, matching Mathlib's martingale theorem
217 `martingale_of_condExp_sub_eq_zero_nat`.
218 -/
219 noncomputable def partialSumsSucc
220     {Omega : Type u}
221     (Y : Nat -> Omega -> Real) :
222     Nat -> Omega -> Real :=
223   fun n => (Finset.range n).sum (fun i => Y (i + 1))
224 
225 /--
226 The partial sums of a global succ-indexed martingale-difference process form a
227 Mathlib martingale.
228 
229 This is a thin wrapper around
230 `MeasureTheory.martingale_of_condExp_sub_eq_zero_nat`.  It does not assert
231 optional stopping, concentration, or final regret; it only upgrades the local
232 conditional-mean-zero increment contract to Mathlib's martingale API.
233 -/
234 theorem martingale_partialSumsSucc_of_succMartingaleDifference
235     {Omega : Type u} [mOmega : MeasurableSpace Omega]
236     (mu : Measure Omega) [IsFiniteMeasure mu]
237     {F : Filtration Nat mOmega}
238     {Y : Nat -> Omega -> Real}
239     (h : SuccMartingaleDifference mu F Y) :
240     Martingale (partialSumsSucc Y) F mu := by
241   have h_adapted : StronglyAdapted F (partialSumsSucc Y) := by
242     intro n
243     change StronglyMeasurable[F n]
244       ((Finset.range n).sum (fun i => Y (i + 1)))
245     refine (Finset.range n).stronglyMeasurable_sum fun i hi => ?_
246     rw [Finset.mem_range] at hi
247     exact h.stronglyAdapted.stronglyMeasurable_le (Nat.succ_le_of_lt hi)
248   have h_integrable : forall n, Integrable (partialSumsSucc Y n) mu := by
249     intro n
250     change Integrable
251       ((Finset.range n).sum (fun i => Y (i + 1))) mu
252     exact
253       MeasureTheory.integrable_finset_sum' (Finset.range n) fun i _hi =>
254         h.integrable (i + 1)
255   refine
256     MeasureTheory.martingale_of_condExp_sub_eq_zero_nat
257       h_adapted h_integrable ?_
258   intro i
259   have hdiff :
260       (partialSumsSucc Y (i + 1) - partialSumsSucc Y i) =
261         Y (i + 1) := by
262     funext omega
263     simp [partialSumsSucc, Pi.sub_apply, Finset.sum_range_succ]
264   simpa [hdiff] using h.condExp_succ_eq_zero i
265
