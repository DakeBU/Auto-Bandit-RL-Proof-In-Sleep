4 /-!
5 # All-horizon sparse EXP3 with two predictable variances
6 
7 The exact double-variance schedule has the refined `16 * gamma * T`
8 threshold when four horizon inequalities make clipping inactive. Outside that
9 regime this module uses the strict `T + 1` zero-probability threshold under
10 the identical internal eta, gamma, and generated trajectory measure.
11 
12 The common support-sparsity failure event is removed once in the off-bad
13 theorem and added once in the residual theorem. The practical endpoint
14 consumes its same-measure `ofReal epsilon` bound.
15 -/

23 /-- Regime in which every component of the exact double-variance exploration
24 schedule is at most one half. -/
25 def doubleVarianceProbabilisticSparseLossLargeHorizonCondition
26     (K S T delta : Real) : Prop :=
27   4 * (S * Real.log K) <= T ∧
28     32 * (K * S * Real.log K ^ 2 * Real.log (4 / delta)) <= T ^ 3 ∧
29     8 * (K * Real.log (4 / delta)) <= T ∧
30     4 * (S * Real.log (4 / delta)) <= T
31 
32 /-- All-horizon threshold for the exact double-variance sparse route: use
33 the refined threshold in its valid regime and strict `T + 1` otherwise. -/
34 noncomputable def doubleVarianceProbabilisticSparseLossAllHorizonRegretThreshold
35     {Action : Type v} (arms : Finset Action) (horizon sparsity : Nat)
36     (delta : Real) : Real := by
37   classical
38   let gamma :=
39     doubleVarianceProbabilisticSparseLossClippedExplorationRate
40       (arms.card : Real) (sparsity : Real) (horizon : Real) delta
41   exact if doubleVarianceProbabilisticSparseLossLargeHorizonCondition
42       (arms.card : Real) (sparsity : Real) (horizon : Real) delta then
43     pathwiseVarianceProbabilisticSparseLossDoubleVarianceRealizedExplicitThreshold
44       arms gamma horizon sparsity delta
45   else
46     (horizon : Real) + 1
