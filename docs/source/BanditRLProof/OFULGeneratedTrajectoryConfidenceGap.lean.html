4 /-!
5 # Generated-trajectory confidence and gap transport for OFUL
6 
7 This module identifies the scalar-ridge state reconstructed from an inclusive
8 finite pair history with the generic finite-horizon state on the underlying
9 trajectory. It then transports the measurable strict-fold selector's score
10 maximality to one-step and finite successor-window optimism-gap bounds.
11 -/

21 /--
22 Any action whose optimistic score dominates a comparator satisfies the usual
23 two-bonus OFUL gap bound on the confidence ellipsoid.
24 -/
25 theorem linearValue_sub_selected_le_two_mul_bonus_of_score_max
26     {Feature Action : Type*}
27     [Fintype Feature] [DecidableEq Feature]
28     (V : Matrix Feature Feature Real) (hV : V.PosDef)
29     (thetaHat thetaStar : Feature -> Real) (beta : Real)
30     (actionFeature : Action -> Feature -> Real)
31     (selected comparator : Action)
32     (hconfidence : matrixNorm V (thetaHat - thetaStar) <= beta)
33     (hscoreMax :
34       optimisticScore thetaHat V beta (actionFeature comparator) <=
35         optimisticScore thetaHat V beta (actionFeature selected)) :
36     linearValue thetaStar (actionFeature comparator) -
37         linearValue thetaStar (actionFeature selected) <=
38       2 * beta * confidenceWidth V (actionFeature selected) := by
39   have hoptimistic :
40       linearValue thetaStar (actionFeature comparator) <=
41         optimisticScore thetaHat V beta (actionFeature comparator) :=
42     linearValue_le_optimisticScore_of_matrixNorm_sub_le
43       V hV thetaHat thetaStar beta (actionFeature comparator) hconfidence
44   have hselected :
45       optimisticScore thetaHat V beta (actionFeature selected) <=
46         linearValue thetaStar (actionFeature selected) +
47           2 * beta * confidenceWidth V (actionFeature selected) :=
48     optimisticScore_le_linearValue_add_two_mul_bonus_of_matrixNorm_sub_le
49       V hV thetaHat thetaStar beta (actionFeature selected) hconfidence
50   linarith

53 The concrete strict-fold history selector has the OFUL one-step gap
54 certificate. No identification with the nonconstructive finite argmax is used.

56 theorem finiteHistoryScalarRidgeOptimisticAction_gap_le
57     {K : Nat} {Feature : Type u}
58     [Fintype Feature] [DecidableEq Feature]
59     (hK : 0 < K)
60     (lambda : Real) (hlambda : 0 < lambda)
61     (thetaStar : Feature -> Real)
62     (actionFeature : Fin K -> Feature -> Real)
63     (R delta S : Real)
64     (n : Nat) (history : History.FinitePairHistory (Fin K) Real n)
65     (hconfidence :
66       matrixNorm
67           (finiteHistoryScalarRidgeDesign lambda actionFeature n history)
68           (finiteHistoryScalarRidgeEstimate lambda actionFeature n history -
69             thetaStar) <=
70         finiteHistoryScalarRidgeRadius
71           actionFeature R delta lambda S n history)
72     (comparator : Fin K) :
73     linearValue thetaStar (actionFeature comparator) -
74         linearValue thetaStar
75           (actionFeature
76             (finiteHistoryScalarRidgeOptimisticAction
77               hK lambda actionFeature R delta S n history)) <=
78       2 *
79         finiteHistoryScalarRidgeRadius
80           actionFeature R delta lambda S n history *
81         confidenceWidth
82           (finiteHistoryScalarRidgeDesign lambda actionFeature n history)
83           (actionFeature
84             (finiteHistoryScalarRidgeOptimisticAction
85               hK lambda actionFeature R delta S n history)) := by
86   have hV :
87       (finiteHistoryScalarRidgeDesign
88         lambda actionFeature n history).PosDef := by
89     unfold finiteHistoryScalarRidgeDesign
90     exact
91       (scalarIdentity_posDef lambda hlambda).add_posSemidef
92         (finiteHorizonFeatureGram_posSemidef
93           (fun t historyValue =>
94             finiteHistoryObservedFeature actionFeature n historyValue t)
95           (n + 1) history)
96   have hscoreMax :=
97     finiteHistoryScalarRidgeOptimisticAction_score_max
98       hK lambda actionFeature R delta S n history comparator
99   simpa only [finiteHistoryScalarRidgeOptimisticScore,
100     finiteHistoryFixedActionFeature] using
101     (linearValue_sub_selected_le_two_mul_bonus_of_score_max
102       (finiteHistoryScalarRidgeDesign lambda actionFeature n history) hV
103       (finiteHistoryScalarRidgeEstimate lambda actionFeature n history)
104       thetaStar
105       (finiteHistoryScalarRidgeRadius
106         actionFeature R delta lambda S n history)
107       actionFeature
108       (finiteHistoryScalarRidgeOptimisticAction
109         hK lambda actionFeature R delta S n history)
110       comparator hconfidence hscoreMax)
