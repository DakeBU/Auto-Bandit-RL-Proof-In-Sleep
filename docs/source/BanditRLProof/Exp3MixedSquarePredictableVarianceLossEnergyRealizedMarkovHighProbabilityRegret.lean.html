17 /-- A finite-action centered mixed-square estimator has variance at most the
18 inverse probability floor times the armwise loss-square energy. -/
19 theorem sum_prob_mul_sq_mixedSquaredEstimatorDeviation_le_inv_floor_mul_sum_loss_sq
20     {Action : Type u} [DecidableEq Action]
21     (arms : Finset Action) (prob loss : Action → Real)
22     (hdist : FiniteActionDistribution arms prob)
23     (epsilon : Real) (hepsilon : 0 < epsilon)
24     (hfloor : ∀ action, action ∈ arms → epsilon ≤ prob action)
25     (hloss : ∀ action, action ∈ arms → loss action ∈ Set.Icc (0 : Real) 1) :
26     arms.sum (fun chosen =>
27         prob chosen *
28           (mixedSquaredImportanceWeightedLoss arms prob loss chosen -
29             arms.sum (fun action => (loss action) ^ 2)) ^ 2) ≤
30       (1 / epsilon) * arms.sum (fun action => (loss action) ^ 2) := by
31   let raw := fun chosen =>
32     mixedSquaredImportanceWeightedLoss arms prob loss chosen
33   let mean := arms.sum (fun action => (loss action) ^ 2)
34   have hprob_pos (action : Action) (haction : action ∈ arms) :
35       0 < prob action := hepsilon.trans_le (hfloor action haction)
36   have hfirst : arms.sum (fun chosen => prob chosen * raw chosen) = mean := by
37     simpa [raw, mean] using
38       sum_prob_mul_mixedSquaredImportanceWeightedLoss_eq_sum_loss_sq
39         arms prob loss (fun action haction => (hprob_pos action haction).ne')
40   have hsecond : arms.sum (fun chosen => prob chosen * (raw chosen) ^ 2) =
41       arms.sum (fun action => (loss action) ^ 4 / prob action) := by
42     simpa [raw] using
43       sum_prob_mul_sq_mixedSquaredImportanceWeightedLoss_eq
44         arms prob loss (fun action haction => (hprob_pos action haction).ne')
45   have hcenter (chosen : Action) :
46       prob chosen * (raw chosen - mean) ^ 2 =
47         (prob chosen * (raw chosen) ^ 2 -
48           2 * mean * (prob chosen * raw chosen)) +
49           mean ^ 2 * prob chosen := by
50     ring
51   have hvariance :
52       arms.sum (fun chosen => prob chosen * (raw chosen - mean) ^ 2) =
53         arms.sum (fun action => (loss action) ^ 4 / prob action) - mean ^ 2 := by
54     simp_rw [hcenter]
55     rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
56       ← Finset.mul_sum, ← Finset.mul_sum, hsecond, hfirst, hdist.sum_eq_one]
57     ring
58   have hfourth_le_sq (action : Action) (haction : action ∈ arms) :
59       (loss action) ^ 4 ≤ (loss action) ^ 2 := by
60     have hsq_nonneg : 0 ≤ (loss action) ^ 2 := sq_nonneg _
61     have hsq_le_one : (loss action) ^ 2 ≤ 1 := by
62       simpa using (sq_le_sq₀ (hloss action haction).1 zero_le_one).2
63         (hloss action haction).2
64     nlinarith [sq_nonneg ((loss action) ^ 2)]
65   have hterm (action : Action) (haction : action ∈ arms) :
66       (loss action) ^ 4 / prob action ≤
67         (1 / epsilon) * (loss action) ^ 2 := by
68     have hinv : 1 / prob action ≤ 1 / epsilon :=
69       one_div_le_one_div_of_le hepsilon (hfloor action haction)
70     calc
71       (loss action) ^ 4 / prob action ≤
72           (loss action) ^ 2 / prob action :=
73         div_le_div_of_nonneg_right (hfourth_le_sq action haction)
74           (hprob_pos action haction).le
75       _ = (loss action) ^ 2 * (1 / prob action) := by ring
76       _ ≤ (loss action) ^ 2 * (1 / epsilon) :=
77         mul_le_mul_of_nonneg_left hinv (sq_nonneg _)
78       _ = (1 / epsilon) * (loss action) ^ 2 := by ring
79   have hsecond_le :
80       arms.sum (fun action => (loss action) ^ 4 / prob action) ≤
81         (1 / epsilon) * arms.sum (fun action => (loss action) ^ 2) := by
82     calc
83       arms.sum (fun action => (loss action) ^ 4 / prob action) ≤
84           arms.sum (fun action => (1 / epsilon) * (loss action) ^ 2) :=
85         Finset.sum_le_sum fun action haction => hterm action haction
86       _ = (1 / epsilon) * arms.sum (fun action => (loss action) ^ 2) := by
87         rw [Finset.mul_sum]
88   change arms.sum (fun chosen => prob chosen * (raw chosen - mean) ^ 2) ≤ _
89   rw [hvariance]
90   nlinarith [sq_nonneg mean]

134 theorem sampledPredictableMixedSquaredVarianceSum_le_inv_floor_mul_lossSquaredSum
135     {Env : Type u} {Action : Type v}
136     [MeasurableSpace Env]
137     [MeasurableSpace Action] [MeasurableSingletonClass Action]
138     [DecidableEq Action]
139     (arms : Finset Action) (harms : arms.Nonempty)
140     (eta gamma : Real) (hgamma_pos : 0 < gamma)
141     (hgamma_le_one : gamma ≤ 1)
142     (loss : PredictableLossVector Env Action) (horizon : Nat)
143     (sample : Env × ((k : Nat) → Action × Real)) :
144     sampledPredictableMixedSquaredVarianceSum
145         arms eta gamma loss horizon sample ≤
146       (1 / (gamma / (arms.card : Real))) *
147         sampledPredictableLossSquaredSum arms loss horizon sample := by
148   unfold sampledPredictableMixedSquaredVarianceSum
149   unfold sampledPredictableLossSquaredSum
150   calc
151     (Finset.range horizon).sum (fun i =>
152         sampledTrajectoryPredictableMixedSquaredVarianceAt
153           arms eta gamma loss i sample) ≤
154       (Finset.range horizon).sum (fun i =>
155         (1 / (gamma / (arms.card : Real))) *
156           arms.sum (fun action =>
157             (predictableLossAt loss i sample action) ^ 2)) :=
158       Finset.sum_le_sum fun i _hi =>
159         sampledTrajectoryPredictableMixedSquaredVarianceAt_le_inv_floor_mul_lossSquaredAt
160           arms harms eta gamma hgamma_pos hgamma_le_one loss i sample
161     _ = (1 / (gamma / (arms.card : Real))) *
162         (Finset.range horizon).sum (fun i =>
163           arms.sum (fun action =>
164             (predictableLossAt loss i sample action) ^ 2)) := by
165       rw [Finset.mul_sum]

167 /-- A pathwise predictable loss-square budget yields the generated
168 cumulative-variance `lintegral` budget required by the Markov route. -/
169 theorem sampledPredictableMixedSquaredVarianceLIntegral_le_of_lossSquaredSum_le
170     {Env : Type u} {Action : Type v}
171     [MeasurableSpace Env]
172     [MeasurableSpace Action] [MeasurableSingletonClass Action]
173     [DecidableEq Action]
174     (mu : Measure (Env × ((k : Nat) → Action × Real)))
175     [IsProbabilityMeasure mu]
176     (arms : Finset Action) (harms : arms.Nonempty)
177     (eta gamma : Real) (hgamma_pos : 0 < gamma)
178     (hgamma_le_one : gamma ≤ 1)
179     (loss : PredictableLossVector Env Action) (horizon : Nat)
180     (lossSquaredBudget : Real)
181     (henergy : ∀ sample,
182       sampledPredictableLossSquaredSum arms loss horizon sample ≤
183         lossSquaredBudget) :
184     sampledPredictableMixedSquaredVarianceLIntegral
185         mu arms eta gamma loss horizon ≤
186       ENNReal.ofReal
187         ((1 / (gamma / (arms.card : Real))) * lossSquaredBudget) := by
188   let varianceSum := sampledPredictableMixedSquaredVarianceSum
189     arms eta gamma loss horizon
190   have hfloor_pos : 0 < gamma / (arms.card : Real) := by
191     exact div_pos hgamma_pos (Nat.cast_pos.2 (Finset.card_pos.mpr harms))
192   have hinv_nonneg : 0 ≤ 1 / (gamma / (arms.card : Real)) :=
193     one_div_nonneg.2 hfloor_pos.le
194   have hpoint (sample : Env × ((k : Nat) → Action × Real)) :
195       varianceSum sample ≤
196         (1 / (gamma / (arms.card : Real))) * lossSquaredBudget := by
197     exact
198       (sampledPredictableMixedSquaredVarianceSum_le_inv_floor_mul_lossSquaredSum
199         arms harms eta gamma hgamma_pos hgamma_le_one loss horizon sample).trans
200       (mul_le_mul_of_nonneg_left (henergy sample) hinv_nonneg)
201   calc
202     sampledPredictableMixedSquaredVarianceLIntegral
203         mu arms eta gamma loss horizon =
204       ∫⁻ sample, ENNReal.ofReal (varianceSum sample) ∂mu := by rfl
205     _ ≤ ∫⁻ _sample,
206         ENNReal.ofReal
207           ((1 / (gamma / (arms.card : Real))) * lossSquaredBudget) ∂mu := by
208       exact lintegral_mono fun sample => ENNReal.ofReal_le_ofReal (hpoint sample)
209     _ = ENNReal.ofReal
210         ((1 / (gamma / (arms.card : Real))) * lossSquaredBudget) := by
211       simp
