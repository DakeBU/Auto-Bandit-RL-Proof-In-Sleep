18 @[simp] theorem pullCount_one :
19     pullCount action a 1 = if action 0 = a then 1 else 0 := by
20   by_cases h : action 0 = a <;> simp [pullCount, h]
21 
22 theorem pullCount_succ_of_eq (h : action t = a) :
23     pullCount action a (t + 1) = pullCount action a t + 1 := by
24   simp [pullCount_succ, h]
25 
26 theorem pullCount_succ_of_ne (h : action t ≠ a) :
27     pullCount action a (t + 1) = pullCount action a t := by
28   simp [pullCount_succ, h]
29 
30 theorem pullCount_le_succ :
31     pullCount action a t ≤ pullCount action a (t + 1) := by
32   rw [pullCount_succ]
33   exact Nat.le_add_right _ _
34 
35 theorem pullCount_succ_le_succ :
36     pullCount action a (t + 1) ≤ pullCount action a t + 1 := by
37   rw [pullCount_succ]
38   by_cases h : action t = a <;> simp [h]

40 /-- If a positive count level is never hit at a successor time, every finite
41 pull count stays strictly below that level.  The proof uses only that
42 `pullCount` starts at zero and grows by at most one per round. -/
43 theorem pullCount_lt_of_forall_succ_ne
44     (target : Nat) (htarget : 0 < target)
45     (hnever : ∀ chron, pullCount action a (chron + 1) ≠ target) :
46     pullCount action a t < target := by
47   induction t with
48   | zero => simpa using htarget
49   | succ t ih =>
50       have hle : pullCount action a (t + 1) ≤ target :=
51         Nat.le_trans (pullCount_succ_le_succ action a t)
52           (Nat.succ_le_iff.mpr ih)
53       exact Nat.lt_of_le_of_ne hle (hnever t)
54 
55 theorem pullCount_mono {s t : Nat} (h : s ≤ t) :
56     pullCount action a s ≤ pullCount action a t := by
57   induction h with
58   | refl => exact Nat.le_refl _
59   | step h ih => exact Nat.le_trans ih (pullCount_le_succ action a _)
60 
61 theorem pullCount_le_time :
62     pullCount action a t ≤ t := by
63   induction t with
64   | zero => simp
65   | succ t ih =>
66       rw [pullCount_succ]
67       by_cases h : action t = a
68       · simp [h]
69         exact ih
70       · simp [h]
71         exact Nat.le_trans ih (Nat.le_succ t)
72 
73 theorem pullCount_add_le (n : Nat) :
74     pullCount action a (t + n) ≤ pullCount action a t + n := by
75   induction n with
76   | zero => simp
77   | succ n ih =>
78       rw [Nat.add_succ]
79       exact Nat.le_trans (pullCount_succ_le_succ action a (t + n))
80         (Nat.add_le_add_right ih 1)
81 
82 theorem pullCount_le_add :
83     pullCount action a t ≤ pullCount action a (t + n) := by
84   exact pullCount_mono action a (Nat.le_add_right t n)
85 
86 theorem pullCount_eq_zero_of_forall_ne
87     (h : ∀ s, s < t → action s ≠ a) :
88     pullCount action a t = 0 := by
89   induction t with
90   | zero => simp
91   | succ t ih =>
92       rw [pullCount_succ_of_ne action a t (h t (Nat.lt_succ_self t))]
93       exact ih (fun s hs => h s (Nat.lt_trans hs (Nat.lt_succ_self t)))
94 
95 theorem pullCount_eq_time_of_forall_eq
96     (h : ∀ s, s < t → action s = a) :
97     pullCount action a t = t := by
98   induction t with
99   | zero => simp
100   | succ t ih =>
101       rw [pullCount_succ_of_eq action a t (h t (Nat.lt_succ_self t))]
102       rw [ih (fun s hs => h s (Nat.lt_trans hs (Nat.lt_succ_self t)))]
103 
104 theorem pullCount_pos_of_eq_before {s t : Nat}
105     (hst : s < t) (h : action s = a) :
106     0 < pullCount action a t := by
107   have hstep :
108       pullCount action a (s + 1) = pullCount action a s + 1 :=
109     pullCount_succ_of_eq action a s h
110   have hpos : 0 < pullCount action a (s + 1) := by
111     rw [hstep]
112     exact Nat.succ_pos _
113   exact Nat.lt_of_lt_of_le hpos
114     (pullCount_mono action a (Nat.succ_le_of_lt hst))

116 /--
117 Pull counts depend only on the half-open action prefix `0, ..., t - 1`.
118 
119 This keeps later adaptive-policy wrappers from reproving the same induction
120 when a history-generated trace is known to agree pointwise with an index-policy
121 trace up to a finite horizon.
122 -/
123 theorem pullCount_eq_of_forall_lt
124     (action action' : ActionTrace Action) (a : Action) :
125     forall t : Nat,
126       (forall s : Nat, s < t -> action s = action' s) ->
127         pullCount action a t = pullCount action' a t := by
128   intro t
129   induction t with
130   | zero =>
131       intro _h
132       simp
133   | succ t ih =>
134       intro h
135       rw [pullCount_succ, pullCount_succ]
136       rw [ih (fun s hs => h s (Nat.lt_trans hs (Nat.lt_succ_self t)))]
137       rw [h t (Nat.lt_succ_self t)]
138 
139 @[simp] theorem pullCount_const_self (a : Action) (t : Nat) :
140     pullCount (fun _ => a) a t = t := by
141   apply pullCount_eq_time_of_forall_eq
142   intro _s _hs
143   rfl
144 
145 theorem pullCount_const_of_ne (b : Action) (h : b ≠ a) (t : Nat) :
146     pullCount (fun _ => b) a t = 0 := by
147   apply pullCount_eq_zero_of_forall_ne
148   intro _s _hs
149   exact h

151 theorem pullCount_add_eq_of_forall_ne_between (n : Nat)
152     (h : ∀ s, t ≤ s → s < t + n → action s ≠ a) :
153     pullCount action a (t + n) = pullCount action a t := by
154   induction n with
155   | zero => simp
156   | succ n ih =>
157       rw [Nat.add_succ]
158       rw [pullCount_succ_of_ne action a (t + n)
159         (h (t + n) (Nat.le_add_right t n) (Nat.lt_succ_self (t + n)))]
160       exact ih (fun s hts hsn =>
161         h s hts (Nat.lt_trans hsn (Nat.lt_succ_self (t + n))))
162 
163 theorem pullCount_add_eq_add_of_forall_eq_between (n : Nat)
164     (h : ∀ s, t ≤ s → s < t + n → action s = a) :
165     pullCount action a (t + n) = pullCount action a t + n := by
166   induction n with
167   | zero => simp
168   | succ n ih =>
169       rw [Nat.add_succ]
170       rw [pullCount_succ_of_eq action a (t + n)
171         (h (t + n) (Nat.le_add_right t n) (Nat.lt_succ_self (t + n)))]
172       rw [ih (fun s hts hsn =>
173         h s hts (Nat.lt_trans hsn (Nat.lt_succ_self (t + n))))]
174       rw [Nat.add_assoc]

176 /--
177 The recursive pull count equals the number of matching actions in the
178 half-open time prefix `0, ..., t - 1`.
179 
180 This is intentionally a dependency-light `List.range` bridge.  The Mathlib
181 `Finset.range` cardinality wrapper is a separate downstream leaf.
182 -/
183 theorem pullCount_eq_list_filter_length :
184     pullCount action a t =
185       ((List.range t).filter (fun s : Nat => decide (action s = a))).length := by
186   induction t with
187   | zero =>
188       simp [pullCount]
189   | succ t ih =>
190       rw [pullCount_succ, ih]
191       by_cases h : action t = a
192       · simp [List.range_succ, h]
193       · simp [List.range_succ, h]
194 
195 end PullCount

199 variable {Action Reward : Type u} [DecidableEq Action]
200 variable [OfNat Reward 0] [HAdd Reward Reward Reward]
201 variable (action : ActionTrace Action) (reward : RewardTrace Reward)
202 variable (a : Action) (t : Nat)
203 
204 theorem sumRewards_succ_of_eq (h : action t = a) :
205     sumRewards action reward a (t + 1) =
206       sumRewards action reward a t + reward t := by
207   rw [sumRewards_succ, if_pos h]
208 
209 theorem sumRewards_succ_of_ne (hzero : ∀ x : Reward, x + 0 = x)
210     (h : action t ≠ a) :
211     sumRewards action reward a (t + 1) =
212       sumRewards action reward a t := by
213   rw [sumRewards_succ, if_neg h, hzero]
214 
215 theorem sumRewards_eq_zero_of_forall_ne (hzero : ∀ x : Reward, x + 0 = x)
216     (h : ∀ s, s < t → action s ≠ a) :
217     sumRewards action reward a t = 0 := by
218   induction t with
219   | zero => simp
220   | succ t ih =>
221       rw [sumRewards_succ_of_ne action reward a t hzero (h t (Nat.lt_succ_self t))]
222       exact ih (fun s hs => h s (Nat.lt_trans hs (Nat.lt_succ_self t)))
223 
224 theorem sumRewards_const_of_ne (hzero : ∀ x : Reward, x + 0 = x)
225     (b : Action) (h : b ≠ a) (t : Nat) :
226     sumRewards (fun _ => b) reward a t = 0 := by
227   apply sumRewards_eq_zero_of_forall_ne
228   · exact hzero
229   · intro _s _hs
230     exact h
231 
232 theorem sumRewards_add_eq_of_forall_ne_between (hzero : ∀ x : Reward, x + 0 = x)
233     (n : Nat) (h : ∀ s, t ≤ s → s < t + n → action s ≠ a) :
234     sumRewards action reward a (t + n) = sumRewards action reward a t := by
235   induction n with
236   | zero => simp
237   | succ n ih =>
238       rw [Nat.add_succ]
239       rw [sumRewards_succ_of_ne action reward a (t + n) hzero
240         (h (t + n) (Nat.le_add_right t n) (Nat.lt_succ_self (t + n)))]
241       exact ih (fun s hts hsn =>
242         h s hts (Nat.lt_trans hsn (Nat.lt_succ_self (t + n))))
243 
244 /--
245 The recursive reward sum equals a left fold over the half-open time prefix.
246 
247 This bridge deliberately keeps the false-branch `+ 0` steps in the fold, so it
248 does not require additive laws beyond the weak assumptions used by `sumRewards`.
249 -/
250 theorem sumRewards_eq_list_range_foldl :
251     sumRewards action reward a t =
252       (List.range t).foldl
253         (fun acc s => acc + if action s = a then reward s else 0)
254         0 := by
255   induction t with
256   | zero =>
257       simp [sumRewards]
258   | succ t ih =>
259       simp [sumRewards, List.range_succ, List.foldl_append, ih]
260 
261 /--
262 The reward sum can also drop nonmatching time steps from the list fold when
263 the accumulator has a right-zero law.
264 
265 This is still dependency-light: it uses `List.range` and `List.filter`, not a
266 Mathlib `Finset` sum.
267 -/
268 theorem sumRewards_eq_list_range_filter_foldl
269     (hzero : ∀ x : Reward, x + 0 = x) :
270     sumRewards action reward a t =
271       ((List.range t).filter (fun s : Nat => decide (action s = a))).foldl
272         (fun acc s => acc + reward s)
273         0 := by
274   induction t with
275   | zero =>
276       simp [sumRewards]
277   | succ t ih =>
278       rw [sumRewards_succ, ih]
279       by_cases h : action t = a
280       · simp [List.range_succ, List.foldl_append, h]
281       · simp [List.range_succ, h, hzero]
282 
283 end SumRewards
