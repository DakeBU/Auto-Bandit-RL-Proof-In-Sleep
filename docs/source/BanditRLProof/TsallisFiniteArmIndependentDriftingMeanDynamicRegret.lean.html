6 # Dynamic regret for independent reward laws with drifting means
7 
8 This module upgrades the fixed-baseline all-regimes theorem to the comparator
9 that maximizes the actual reward mean at every round. The dynamic regret is
10 decomposed into fixed-`model.bestArm` regret and the actual mean advantage of
11 the moving comparator. The supplied armwise mean-deviation envelope controls
12 that second term.
13 -/

23 /-- Predictable environment regret against a deterministic comparator that may
24 change with the round. -/
25 noncomputable def sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret
26     {Env : Type u} {Action : Type v}
27     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
28     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
29     (loss : Exp3.PredictableLossVector Env Action)
30     (comparator : Nat -> Action) (horizon : Nat)
31     (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
32   (Finset.range (horizon + 1)).sum (fun t =>
33     FTRL.linearLoss arms
34         (sampledScheduledHalfTsallisProbabilityAtTime
35           arms harms eta t sample)
36         (Exp3.predictableLossAt loss t sample) -
37       Exp3.predictableLossAt loss t sample (comparator t))

39 /-- Moving-comparator regret is fixed-comparator regret plus the cumulative
40 loss advantage of the moving comparator over the fixed arm. -/
41 theorem sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret_eq_fixed_add

43     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
44     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
45     (loss : Exp3.PredictableLossVector Env Action)
46     {best : Action} (hbest : best ∈ arms)
47     (comparator : Nat -> Action) (horizon : Nat)
48     (sample : Env × ((k : Nat) -> Action × Real)) :
49     sampledScheduledHalfTsallisPredictableMovingComparatorEnvironmentRegret
50         arms harms eta loss comparator horizon sample =
51       sampledScheduledHalfTsallisPredictableEnvironmentRegret
52           arms harms eta loss (pointMass best) horizon sample +
53         (Finset.range (horizon + 1)).sum (fun t =>
54           Exp3.predictableLossAt loss t sample best -
55             Exp3.predictableLossAt loss t sample (comparator t)) := by

64 /-- An actual-mean maximizing arm at round `t`. The finite action space is
65 nonempty because it comes from a `FiniteBanditModel`. -/
66 noncomputable def finiteArmIndependentBestArmAt {K : Nat}
67     (model : FiniteBanditModel K)
68     (armLaw : Nat -> Fin K -> Measure Rat) (t : Nat) : Fin K := by
69   letI : Nonempty (Fin K) := ⟨model.bestArm⟩
70   exact Classical.choose
71     (Finset.exists_max_image (Finset.univ : Finset (Fin K))
72       (finiteArmIndependentRewardMean armLaw t) Finset.univ_nonempty)

74 /-- The selected dynamic arm maximizes the actual roundwise reward mean. -/
75 theorem finiteArmIndependentRewardMean_le_bestArmAt {K : Nat}
76     (model : FiniteBanditModel K)
77     (armLaw : Nat -> Fin K -> Measure Rat) (t : Nat) (arm : Fin K) :
78     finiteArmIndependentRewardMean armLaw t arm <=
79       finiteArmIndependentRewardMean armLaw t
80         (finiteArmIndependentBestArmAt model armLaw t) := by
81   letI : Nonempty (Fin K) := ⟨model.bestArm⟩
82   have hspec :=
83     Classical.choose_spec
84       (Finset.exists_max_image (Finset.univ : Finset (Fin K))
85         (finiteArmIndependentRewardMean armLaw t) Finset.univ_nonempty)
86   exact hspec.2 arm (Finset.mem_univ arm)

88 /-- Extra envelope budget needed to move from the fixed model comparator to
89 the actual-mean maximizing arm at every included round. -/
90 noncomputable def finiteArmIndependentDynamicComparatorPenalty {K : Nat}
91     (model : FiniteBanditModel K)
92     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat)
93     (meanDeviation : Nat -> Fin K -> Real) : Real :=
94   (Finset.range (horizon + 1)).sum (fun t =>
95     if finiteArmIndependentBestArmAt model armLaw t = model.bestArm then
96       0
97     else
98       meanDeviation t (finiteArmIndependentBestArmAt model armLaw t) +
99         meanDeviation t model.bestArm)

115 /-- The actual-mean advantage of any arm over the fixed baseline best arm is
116 controlled by the two corresponding mean-deviation envelopes. -/
117 theorem finiteArmIndependentRewardMean_sub_bestArm_le_meanDeviation
118     {K : Nat} (model : FiniteBanditModel K)
119     (armLaw : Nat -> Fin K -> Measure Rat)
120     (meanDeviation : Nat -> Fin K -> Real)
121     (hmeanDeviation : forall t arm,
122       |finiteArmIndependentRewardMean armLaw t arm -
123           ((model.mean arm : Rat) : Real)| <= meanDeviation t arm)
124     (t : Nat) (arm : Fin K) :
125     finiteArmIndependentRewardMean armLaw t arm -
126         finiteArmIndependentRewardMean armLaw t model.bestArm <=
127       meanDeviation t arm + meanDeviation t model.bestArm := by
128   have hmodel :
129       ((model.mean arm : Rat) : Real) <=
130         ((model.mean model.bestArm : Rat) : Real) := by
131     exact_mod_cast FiniteBanditModel.mean_le_bestArm_mean model arm
132   have harm :
133       finiteArmIndependentRewardMean armLaw t arm -
134           ((model.mean arm : Rat) : Real) <= meanDeviation t arm :=
135     (le_abs_self _).trans (hmeanDeviation t arm)
136   have hbest :
137       ((model.mean model.bestArm : Rat) : Real) -
138           finiteArmIndependentRewardMean armLaw t model.bestArm <=
139         meanDeviation t model.bestArm := by
140     calc
141       ((model.mean model.bestArm : Rat) : Real) -
142           finiteArmIndependentRewardMean armLaw t model.bestArm <=
143           |((model.mean model.bestArm : Rat) : Real) -
144             finiteArmIndependentRewardMean armLaw t model.bestArm| :=
145         le_abs_self _
146       _ = |finiteArmIndependentRewardMean armLaw t model.bestArm -
147           ((model.mean model.bestArm : Rat) : Real)| := abs_sub_comm _ _
148       _ <= meanDeviation t model.bestArm :=
149         hmeanDeviation t model.bestArm
150   linarith

306 /-- Dynamic all-regimes bound: fixed-baseline all-regimes regret plus the
307 explicit envelope cost of following the actual-mean maximizing arm. -/
308 noncomputable def finiteArmIndependentDriftingMeanDynamicAllRegimeBound
309     {K : Nat} (model : FiniteBanditModel K)
310     (armLaw : Nat -> Fin K -> Measure Rat) (horizon : Nat)
311     (meanDeviation : Nat -> Fin K -> Real) : Real :=
312   finiteArmIndependentDriftingMeanAllRegimeBound
313       model horizon meanDeviation +
314     finiteArmIndependentDynamicComparatorPenalty
315       model armLaw horizon meanDeviation
