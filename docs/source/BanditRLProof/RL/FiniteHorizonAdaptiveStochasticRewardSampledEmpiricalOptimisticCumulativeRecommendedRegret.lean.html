72     (defaultState : State) (rewardBudget transitionBudget : Real)
73     (hround : forall round : Fin rounds,
74       let plan := adaptiveStochasticSampledEmpiricalOptimisticPlanAt trajectory

77           mdp.optimalValueRemaining mdp.horizon le_rfl state <=
78             plan.upperValueRemaining mdp.horizon le_rfl state) ∧
79         plan.optimisticPolicy.expectedRegret initialState <=

81             (fun remaining hremaining state =>
82               2 * plan.selectedRadiusRemaining remaining hremaining state)
83             mdp.horizon le_rfl initialState) :
84     (forall round : Fin rounds, forall state,
85       mdp.optimalValueRemaining mdp.horizon le_rfl state <=
86         (adaptiveStochasticSampledEmpiricalOptimisticPlanAt trajectory
87           defaultState rewardBudget transitionBudget round).upperValueRemaining
88             mdp.horizon le_rfl state) ∧
89       adaptiveStochasticSampledEmpiricalOptimisticRecommendedExpectedRegret
90           (mdp := mdp) (initialState := initialState) (episodes := episodes)
91           trajectory defaultState rewardBudget transitionBudget rounds <=
92         adaptiveStochasticSampledEmpiricalOptimisticOccupancyRadiusSum
93           (mdp := mdp) (initialState := initialState) (episodes := episodes)
