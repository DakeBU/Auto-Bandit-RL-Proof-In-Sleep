4 /-!
5 # Finite same-source confidence family for cumulative UCBVI-CH
6 
7 The family contains two genuinely generated transition coordinates:
8 
9 * every next-state singleton, with its Bernoulli variance; and
10 * the normalized optimal continuation value at every stage.
11 
12 The second coordinate is not postulated as an external confidence premise.  It
13 is proved from the same recursive episode source and is linked to the singleton
14 residual family by the exact finite linear identity in
15 `FiniteHorizonAdaptiveCumulativeUCBVITransitionValueConfidence`.
16 -/

55 noncomputable def bernsteinCoordinateFailureEvent
56     {mdp : MDP State Action} {initialState : Measure State}
57     [IsProbabilityMeasure initialState]
58     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
59     (logBudget : Real) (index : BernsteinCoordinateIndex mdp episodes) :
60     Set (EpisodeBatchTrajectory mdp 1) :=
61   let varianceBudget := mdp.transitionCoordinateVariance
62       index.state index.action index.nextState * (index.count + 1 : Nat)
63   {trajectory |
64     bernsteinCoordinateThreshold logBudget varianceBudget <=
65         |∑ i ∈ Finset.range (index.round + 1),
66           source.aggregateTransitionResidualIncrement
67             index.state index.action index.nextState i trajectory| ∧
68       (∑ i ∈ Finset.range (index.round + 1),
69           source.aggregateVisitIncrement index.state index.action i trajectory) <=
70         (index.count + 1 : Nat)}

100 theorem trajectoryMeasure_bernsteinCoordinateFailureEvent_le
101     {mdp : MDP State Action} {initialState : Measure State}
102     [IsProbabilityMeasure initialState]
103     [StandardBorelSpace (EpisodeBatch mdp 1)]
104     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]
105     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
106     (logBudget : Real) (hlog : 0 < logBudget)
107     (index : BernsteinCoordinateIndex mdp episodes) :
108     source.trajectoryMeasure
109         (bernsteinCoordinateFailureEvent source logBudget index) <=
110       2 * ENNReal.ofReal (Real.exp (-2 * logBudget)) := by
111   let varianceBudget : Real := mdp.transitionCoordinateVariance
112       index.state index.action index.nextState * (index.count + 1 : Nat)
113   let tilt := bernsteinTilt logBudget varianceBudget
114   have hvariance : 0 <= varianceBudget := by
115     exact mul_nonneg
116       (mdp.transitionCoordinateVariance_nonneg _ _ _)
117       (Nat.cast_nonneg _)
118   have htail := source.measure_abs_aggregateTransitionResidualSum_ge_inter_visitSum_le_variance
119     index.state index.action index.nextState (index.round + 1)
120     tilt (bernsteinCoordinateThreshold logBudget varianceBudget)
121     (index.count + 1 : Nat)
122     (bernsteinTilt_pos hlog hvariance)
123     (bernsteinTilt_le_one _ _)
124   have hexponent := bernsteinTilt_exponent_le hlog hvariance
125   have hexponent' :
126       -tilt * bernsteinCoordinateThreshold logBudget varianceBudget +
127           tilt ^ 2 * varianceBudget <= -2 * logBudget := by
128     simpa [tilt] using hexponent
129   have harg :
130       -tilt * bernsteinCoordinateThreshold logBudget varianceBudget +
131           tilt ^ 2 * mdp.transitionCoordinateVariance
132             index.state index.action index.nextState * (index.count + 1 : Nat) <=
133         -2 * logBudget := by
134     rw [show tilt ^ 2 * mdp.transitionCoordinateVariance
135           index.state index.action index.nextState * (index.count + 1 : Nat) =
136         tilt ^ 2 * varianceBudget by simp [varianceBudget, mul_assoc]]
137     exact hexponent'
138   calc
139     source.trajectoryMeasure
140         (bernsteinCoordinateFailureEvent source logBudget index) <=
141       2 * ENNReal.ofReal
142         (Real.exp (-tilt * bernsteinCoordinateThreshold logBudget varianceBudget +
143           tilt ^ 2 * mdp.transitionCoordinateVariance
144             index.state index.action index.nextState * (index.count + 1 : Nat))) := by
145       simpa [bernsteinCoordinateFailureEvent, varianceBudget, tilt, mul_assoc] using htail
146     _ <= 2 * ENNReal.ofReal (Real.exp (-2 * logBudget)) := by
147       apply mul_le_mul_left'
148       apply ENNReal.ofReal_le_ofReal
149       apply Real.exp_le_exp.mpr
150       exact harg

152 noncomputable def functionalTilt (logBudget visitBudget : Real) : Real :=
153   4 * logBudget / Real.sqrt visitBudget
154 
155 theorem functionalTilt_pos
156     {logBudget visitBudget : Real}
157     (hlog : 0 < logBudget) (hvisit : 0 < visitBudget) :
158     0 < functionalTilt logBudget visitBudget := by
159   unfold functionalTilt
160   exact div_pos (mul_pos (by norm_num) hlog) (Real.sqrt_pos.2 hvisit)
161 
162 theorem functionalTilt_exponent_eq
163     {logBudget visitBudget : Real}
164     (hvisit : 0 < visitBudget) :
165     -functionalTilt logBudget visitBudget *
166           (logBudget * Real.sqrt visitBudget) +
167         (functionalTilt logBudget visitBudget ^ 2 / 8) * visitBudget =
168       -2 * logBudget ^ 2 := by
169   have hsqrt : Real.sqrt visitBudget ≠ 0 :=
170     ne_of_gt (Real.sqrt_pos.2 hvisit)
171   have hsqrtSq : Real.sqrt visitBudget ^ 2 = visitBudget :=
172     Real.sq_sqrt hvisit.le
173   unfold functionalTilt
174   field_simp
175   nlinarith

177 theorem trajectoryMeasure_optimalTailFailureEvent_le
178     {mdp : MDP State Action} {initialState : Measure State}
179     [IsProbabilityMeasure initialState]
180     [StandardBorelSpace (EpisodeBatch mdp 1)]
181     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]
182     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
183     (hreward : ∀ state action, |mdp.reward state action| <= 1)
184     (logBudget : Real) (hlog : 0 < logBudget)
185     (index : OptimalTailIndex mdp episodes) :
186     source.trajectoryMeasure
187         (optimalTailFailureEvent source logBudget index) <=
188       2 * ENNReal.ofReal (Real.exp (-2 * logBudget ^ 2)) := by
189   let visitBudget : Real := (index.count + 1 : Nat)
190   let tilt := functionalTilt logBudget visitBudget
191   have hvisit : 0 < visitBudget := by
192     change 0 < ((index.count + 1 : Nat) : Real)
193     positivity
194   have hprobe : ∀ nextState,
195       mdp.optimalTailProbe index.stage nextState ∈ Set.Icc (0 : Real) 1 :=
196     mdp.optimalTailProbe_mem_Icc hreward index.stage
197   have htail := source.measure_abs_aggregateTransitionFunctionalResidualSum_ge_inter_visitSum_le
198     (mdp.optimalTailProbe index.stage) hprobe index.state index.action
199     (index.round + 1) tilt (logBudget * Real.sqrt visitBudget) visitBudget
200     (functionalTilt_pos hlog hvisit)
201   calc
202     source.trajectoryMeasure (optimalTailFailureEvent source logBudget index) <=
203       2 * ENNReal.ofReal
204         (Real.exp (-tilt * (logBudget * Real.sqrt visitBudget) +
205           (tilt ^ 2 / 8) * visitBudget)) := by
206       simpa [optimalTailFailureEvent, visitBudget, tilt] using htail
207     _ = 2 * ENNReal.ofReal (Real.exp (-2 * logBudget ^ 2)) := by
208       rw [functionalTilt_exponent_eq hvisit]

210 /-- A transparent finite-union bound.  The later UCBVI specialization proves
211 that the displayed right hand side is at most its allotted fraction of
212 `delta`; keeping this intermediate theorem exact makes the event accounting
213 auditable. -/
214 theorem trajectoryMeasure_simultaneousTransitionFailureEvent_le
215     {mdp : MDP State Action} {initialState : Measure State}
216     [IsProbabilityMeasure initialState]
217     [StandardBorelSpace (EpisodeBatch mdp 1)]
218     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]
219     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
220     (hreward : ∀ state action, |mdp.reward state action| <= 1)
221     (episodes : Nat) (logBudget : Real) (hlog : 0 < logBudget) :
222     source.trajectoryMeasure
223         (simultaneousTransitionFailureEvent source episodes logBudget) <=
224       (Fintype.card (BernsteinCoordinateIndex mdp episodes) : ENNReal) *
225           (2 * ENNReal.ofReal (Real.exp (-2 * logBudget))) +
226         (Fintype.card (OptimalTailIndex mdp episodes) : ENNReal) *
227           (2 * ENNReal.ofReal (Real.exp (-2 * logBudget ^ 2))) := by
228   unfold simultaneousTransitionFailureEvent
229   calc
230     source.trajectoryMeasure
231         ((⋃ index : BernsteinCoordinateIndex mdp episodes,
232             bernsteinCoordinateFailureEvent source logBudget index) ∪
233           (⋃ index : OptimalTailIndex mdp episodes,
234             optimalTailFailureEvent source logBudget index)) <=
235       source.trajectoryMeasure
236           (⋃ index : BernsteinCoordinateIndex mdp episodes,
237             bernsteinCoordinateFailureEvent source logBudget index) +
238         source.trajectoryMeasure
239           (⋃ index : OptimalTailIndex mdp episodes,
240             optimalTailFailureEvent source logBudget index) := measure_union_le _ _
241     _ <= (∑ _index : BernsteinCoordinateIndex mdp episodes,
242           2 * ENNReal.ofReal (Real.exp (-2 * logBudget))) +
243         (∑ _index : OptimalTailIndex mdp episodes,
244           2 * ENNReal.ofReal (Real.exp (-2 * logBudget ^ 2))) := by
245       gcongr
246       · refine (ProbabilityUnionBound.measure_iUnion_fintype_le_sum
247           source.trajectoryMeasure _).trans ?_
248         exact Finset.sum_le_sum fun index _ =>
249           trajectoryMeasure_bernsteinCoordinateFailureEvent_le
250             source logBudget hlog index
251       · refine (ProbabilityUnionBound.measure_iUnion_fintype_le_sum
252           source.trajectoryMeasure _).trans ?_
253         exact Finset.sum_le_sum fun index _ =>
254           trajectoryMeasure_optimalTailFailureEvent_le
255             source hreward logBudget hlog index
256     _ = _ := by simp [nsmul_eq_mul]

258 /-- Outside the joint event, every peeled singleton residual is strictly below
259 its variance-sensitive threshold whenever the peel equals the actual count. -/
260 theorem abs_coordinateResidual_lt_of_not_mem_simultaneousTransitionFailureEvent
261     {mdp : MDP State Action} {initialState : Measure State}
262     [IsProbabilityMeasure initialState]
263     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
264     {episodes : Nat} {logBudget : Real}
265     {trajectory : EpisodeBatchTrajectory mdp 1}
266     (htrajectory : trajectory ∉
267       simultaneousTransitionFailureEvent source episodes logBudget)
268     (index : BernsteinCoordinateIndex mdp episodes)
269     (hcount :
270       (∑ i ∈ Finset.range (index.round + 1),
271           source.aggregateVisitIncrement index.state index.action i trajectory) =
272         (index.count + 1 : Nat)) :
273     |∑ i ∈ Finset.range (index.round + 1),
274         source.aggregateTransitionResidualIncrement
275           index.state index.action index.nextState i trajectory| <
276       bernsteinCoordinateThreshold logBudget
277         (mdp.transitionCoordinateVariance
278           index.state index.action index.nextState * (index.count + 1 : Nat)) := by
279   apply lt_of_not_ge
280   intro hge
281   apply htrajectory
282   apply Set.mem_union_left
283   exact Set.mem_iUnion.2 ⟨index, hge, hcount.le⟩
