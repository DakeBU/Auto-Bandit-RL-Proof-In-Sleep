6 # Adaptive sampled-reward empirical optimistic source
7 
8 This module constructs a genuinely sampled-reward adaptive optimistic source.
9 The successor table is computed from the latest complete stochastic batch,
10 including its observed rewards, rather than from the known-mean projection.
11 
12 The main regularity obligation is measurability of the finite-horizon dynamic
13 program as a function of the sampled batch.  The explicit finite argmax and a
14 batch-indexed empirical transition kernel discharge that obligation.  The
15 resulting source has the exact selected-policy iid stochastic batch law in
16 every history fiber by construction.
17 -/

37 /-- A fixed sampled-reward sum is measurable on raw episode batches. -/
38 theorem measurable_rewardSum
39     {mdp : MDP State Action} {episodes : Nat}
40     (stage : Fin mdp.horizon) (state : State) (action : Action) :
41     Measurable fun batch : EpisodeBatch mdp episodes =>
42       batch.rewardSum stage state action := by
43   classical
44   refine Finset.measurable_sum Finset.univ fun episode _ => ?_
45   let step : EpisodeBatch mdp episodes -> EpisodeStep State Action :=
46     fun batch => batch episode stage
47   have hstep : Measurable step :=
48     (measurable_pi_apply stage).comp (measurable_pi_apply episode)
49   have hstate : Measurable fun batch : EpisodeBatch mdp episodes =>
50       (step batch).state := EpisodeStep.measurable_state.comp hstep
51   have haction : Measurable fun batch : EpisodeBatch mdp episodes =>
52       (step batch).action := EpisodeStep.measurable_action.comp hstep
53   have hreward : Measurable fun batch : EpisodeBatch mdp episodes =>
54       (step batch).reward := EpisodeStep.measurable_reward.comp hstep
55   exact Measurable.ite
56     ((measurableSet_eq_fun hstate measurable_const).inter
57       (measurableSet_eq_fun haction measurable_const))
58     hreward measurable_const

139 /-- Finite-state coordinatewise measurability yields joint measurability. -/
140 theorem measurable_uncurry_of_forall_measurable
141     {Omega : Type*} [MeasurableSpace Omega]
142     (value : Omega -> State -> Real)
143     (hvalue : forall nextState : State,
144       Measurable fun omega => value omega nextState) :
145     Measurable (Function.uncurry value) := by
146   classical
147   have hsum : Measurable fun pair : Omega × State =>
148       ∑ nextState : State,
149         if pair.2 = nextState then value pair.1 nextState else 0 := by
150     refine Finset.measurable_sum Finset.univ fun nextState _ => ?_
151     exact Measurable.ite
152       (measurableSet_eq_fun measurable_snd measurable_const)
153       ((hvalue nextState).comp measurable_fst) measurable_const
154   convert hsum using 1
155   funext pair
156   simp [Function.uncurry]
