5 /-!
6 # Common-space episodewise realized behavior consistency
7 
8 The preceding episodewise route gives one sharp certificate for every scheduled
9 finite window, but those windows have different trajectory types.  This module
10 places the finite-window laws on one dependent infinite product.  Coordinate
11 `n` therefore has exactly the compiled adaptive trajectory law for schedule
12 `n`, and the scheduled realized-regret coordinates form one random process.
13 
14 The coupling is intentionally the independent-coordinate product coupling.  It
15 is sufficient for a mathematically literal convergence-in-probability theorem,
16 but it is not a nested coupling of one online algorithm across schedules and
17 does not imply pathwise, almost-sure, or anytime behavior.
18 -/

88 /--
89 An expected-regret upper bound and a two-sided return-deviation bound control
90 the absolute realized average regret.
91 -/
92 theorem abs_realizedSuccessorAverageRegret_le_of_expected_le_of_deviation_abs_le
93     {mdp : MDP State Action} {initialState : Measure State}
94     [IsProbabilityMeasure initialState] {episodes : Nat}
95     (source : AdaptiveEpisodeBatchSource mdp initialState episodes)
96     (trajectory : EpisodeBatchTrajectory mdp episodes)
97     (rounds : Nat) (hrounds : 0 < rounds) (hepisodes : 0 < episodes)
98     (expectedBound deviationBound : Real)
99     (hexpected :
100       source.successorExpectedAverageRegret trajectory rounds <= expectedBound)
101     (hdeviation :
102       |source.cumulativeSuccessorReturnDeviation rounds trajectory| <= deviationBound) :
103     |source.realizedSuccessorAverageRegret trajectory rounds| <=
104       expectedBound + deviationBound / ((episodes : Real) * (rounds : Real)) := by
105   have hdenom : 0 < (episodes : Real) * (rounds : Real) := by positivity
106   rw [source.realizedSuccessorAverageRegret_eq_expected_sub_deviation
107     trajectory rounds hrounds hepisodes]
108   calc
109     |source.successorExpectedAverageRegret trajectory rounds -
110         source.cumulativeSuccessorReturnDeviation rounds trajectory /
111           ((episodes : Real) * (rounds : Real))| <=
112         |source.successorExpectedAverageRegret trajectory rounds| +
113           |source.cumulativeSuccessorReturnDeviation rounds trajectory /
114             ((episodes : Real) * (rounds : Real))| :=
115       abs_sub _ _
116     _ = source.successorExpectedAverageRegret trajectory rounds +
117           |source.cumulativeSuccessorReturnDeviation rounds trajectory| /
118             ((episodes : Real) * (rounds : Real)) := by
119       rw [abs_of_nonneg
120         (source.successorExpectedAverageRegret_nonneg trajectory rounds),
121         abs_div, abs_of_pos hdenom]
122     _ <= expectedBound +
123           deviationBound / ((episodes : Real) * (rounds : Real)) :=
124       add_le_add hexpected
125         (div_le_div_of_nonneg_right hdeviation hdenom.le)
