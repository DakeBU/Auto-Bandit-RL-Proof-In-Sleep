4 # Deterministic-tail high-probability stopped-return optimality
5 
6 This module extracts a deterministic tail start from the qualitative
7 simultaneous stopped-return probability limit. At every later schedule index,
8 the complement of the existing six-way violation event has real probability
9 strictly greater than `1 - delta` and is exactly the event on which all six
10 literal errors are strictly below `epsilon`.
11 
12 The tail start is existential and noncomputable. No convergence rate,
13 independence, or optional-stopping argument is introduced.
14 -/

23 /-- A strict `ENNReal` upper bound on a measurable bad event gives the
24 corresponding strict real-probability lower bound on its complement. -/
25 theorem probReal_compl_gt_one_sub_of_measure_lt_ofReal
26     {Omega : Type w} [MeasurableSpace Omega]
27     (mu : Measure Omega) [IsProbabilityMeasure mu]
28     {event : Set Omega} (hevent : MeasurableSet event)
29     {delta : Real} (hdelta : 0 < delta)
30     (htail : mu event < ENNReal.ofReal delta) :
31     1 - delta < mu.real event.compl := by
32   have htailReal : mu.real event < delta := by
33     change (mu event).toReal < delta
34     have htoReal :
35         (mu event).toReal < (ENNReal.ofReal delta).toReal :=
36       (ENNReal.toReal_lt_toReal (measure_ne_top mu event)
37         ENNReal.ofReal_ne_top).2 htail
38     simpa [ENNReal.toReal_ofReal hdelta.le] using htoReal
39   change 1 - delta < mu.real (eventᶜ)
40   rw [MeasureTheory.probReal_compl_eq_one_sub hevent]
41   linarith
