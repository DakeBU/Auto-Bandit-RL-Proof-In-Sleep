3 /-!
4 # Initial-law finite-horizon sampled-return concentration
5 
6 This module lifts the statewise sampled-return concentration theorem to the
7 full stochastic trajectory measure generated from a finite initial-state law.
8 The random variable remains centered by the recursive value of its own initial
9 state; no concentration claim is made for mixing those state-dependent values.
10 -/

21 /--
22 A common sub-Gaussian proxy on every fiber of a Markov kernel is preserved by
23 mixing the fibers with a probability measure on a finite index type.
24 -/
25 theorem hasSubgaussianMGF_compProd_of_forall_fintype
26     {Index Omega : Type*}
27     [MeasurableSpace Index] [MeasurableSpace Omega] [Fintype Index]
28     (mu : Measure Index) [IsProbabilityMeasure mu]
29     (kappa : ProbabilityTheory.Kernel Index Omega)
30     [ProbabilityTheory.IsMarkovKernel kappa]
31     (X : Index × Omega -> Real) (hX : Measurable X) (c : NNReal)
32     (hfiber : forall index,
33       ProbabilityTheory.HasSubgaussianMGF
34         (fun omega => X (index, omega)) c (kappa index)) :
35     ProbabilityTheory.HasSubgaussianMGF X c (mu.compProd kappa) := by
36   constructor
37   · intro t
38     have hfullMeas : Measurable (fun p => Real.exp (t * X p)) :=
39       Real.measurable_exp.comp (measurable_const.mul hX)
40     have hfullAE : AEStronglyMeasurable
41         (fun p => Real.exp (t * X p)) (mu.compProd kappa) :=
42       hfullMeas.aestronglyMeasurable
43     apply (Measure.integrable_compProd_iff hfullAE).2
44     constructor
45     · exact Filter.Eventually.of_forall fun index =>
46         (hfiber index).integrable_exp_mul t
47     · exact FiniteHorizonRL.integrable_of_fintype_aestronglyMeasurable mu _
48         hfullAE.norm.integral_kernel_compProd
49   · intro t
50     have hfullMeas : Measurable (fun p => Real.exp (t * X p)) :=
51       Real.measurable_exp.comp (measurable_const.mul hX)
52     have hfullAE : AEStronglyMeasurable
53         (fun p => Real.exp (t * X p)) (mu.compProd kappa) :=
54       hfullMeas.aestronglyMeasurable
55     have hfullInt : Integrable (fun p => Real.exp (t * X p))
56         (mu.compProd kappa) := by
57       apply (Measure.integrable_compProd_iff hfullAE).2
58       constructor
59       · exact Filter.Eventually.of_forall fun index =>
60           (hfiber index).integrable_exp_mul t
61       · exact FiniteHorizonRL.integrable_of_fintype_aestronglyMeasurable mu _
62           hfullAE.norm.integral_kernel_compProd
63     have hinnerAE : AEStronglyMeasurable
64         (fun index => integral (kappa index)
65           (fun omega => Real.exp (t * X (index, omega)))) mu :=
66       hfullAE.integral_kernel_compProd
67     have hinnerInt : Integrable
68         (fun index => integral (kappa index)
69           (fun omega => Real.exp (t * X (index, omega)))) mu :=
70       FiniteHorizonRL.integrable_of_fintype_aestronglyMeasurable mu _ hinnerAE
71     rw [ProbabilityTheory.mgf, Measure.integral_compProd hfullInt]
72     have hmono := integral_mono_ae hinnerInt (integrable_const _)
73       (Filter.Eventually.of_forall fun index => by
74         simpa [ProbabilityTheory.mgf] using (hfiber index).mgf_le t)
75     simpa using hmono

108 /--
109 Initial-law sampled-return MGF bound. The centering is state dependent, so the
110 common statewise proxy passes unchanged through the finite initial-state mix.
111 -/
112 theorem stochasticTrajectoryMeasure_sampledCumulativeReturnDeviation_hasSubgaussianMGF
113     [StandardBorelSpace State] [StandardBorelSpace Action]
114     [Nonempty Action]
115     (source : MeanCompatibleRewardKernel mdp)
116     (policy : MarkovPolicy mdp) (initialState : Measure State)
117     [IsProbabilityMeasure initialState]
118     (rewardBound rewardVarianceProxy : NNReal)
119     (hrewardBound : forall state action,
120       |mdp.reward state action| <= (rewardBound : Real))
121     (law : source.UniformSubgaussianRewardLaw rewardVarianceProxy) :
122     ProbabilityTheory.HasSubgaussianMGF
123       (mdp.sampledCumulativeReturnDeviation policy)
124       ((mdp.horizon : NNReal) * rewardVarianceProxy +
125         meanBellmanInnovationVarianceProxy rewardBound mdp.horizon)
126       (source.stochasticTrajectoryMeasure policy initialState) := by
127   unfold stochasticTrajectoryMeasure
128   apply Concentration.hasSubgaussianMGF_compProd_of_forall_fintype
129     initialState
130     (source.stochasticTrajectoryKernelRemaining policy mdp.horizon le_rfl)
131     (mdp.sampledCumulativeReturnDeviation policy)
132     (mdp.measurable_sampledCumulativeReturnDeviation policy)
133   intro state
134   simpa [MDP.sampledCumulativeReturnDeviation] using
135     source.stochasticTrajectoryKernelRemaining_sampledCumulativeReturnDeviationFrom_hasSubgaussianMGF
136       policy rewardBound rewardVarianceProxy hrewardBound law
137       mdp.horizon le_rfl state

140 Fixed-horizon two-sided delta tail under an arbitrary finite initial-state law.

142 theorem stochasticTrajectoryMeasure_sampledCumulativeReturnDeviation_abs_tail_le
143     [StandardBorelSpace State] [StandardBorelSpace Action]
144     [Nonempty Action]
145     (source : MeanCompatibleRewardKernel mdp)
146     (policy : MarkovPolicy mdp) (initialState : Measure State)
147     [IsProbabilityMeasure initialState]
148     (rewardBound rewardVarianceProxy : NNReal)
149     (hrewardBound : forall state action,
150       |mdp.reward state action| <= (rewardBound : Real))
151     (law : source.UniformSubgaussianRewardLaw rewardVarianceProxy)
152     (htotal : 0 <
153       ((((mdp.horizon : NNReal) * rewardVarianceProxy +
154         meanBellmanInnovationVarianceProxy rewardBound mdp.horizon : NNReal) : Real)))
155     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
156     (source.stochasticTrajectoryMeasure policy initialState)
157       {trajectory |
158         Concentration.subGaussianSumConfidenceRadius
159             ((mdp.horizon : NNReal) * rewardVarianceProxy +
160               meanBellmanInnovationVarianceProxy rewardBound mdp.horizon) delta <=
161           |mdp.sampledCumulativeReturnDeviation policy trajectory|} <=
162       ENNReal.ofReal delta := by
163   let mu := source.stochasticTrajectoryMeasure policy initialState
164   let F : Filtration Nat
165       (inferInstance : MeasurableSpace
166         (State × RewardStepTrace Action State mdp.horizon)) :=
167     Filtration.const Nat inferInstance le_rfl
168   let Y : Nat -> State × RewardStepTrace Action State mdp.horizon -> Real :=
169     fun _ => mdp.sampledCumulativeReturnDeviation policy
170   let cY : Nat -> NNReal := fun _ =>
171     (mdp.horizon : NNReal) * rewardVarianceProxy +
172       meanBellmanInnovationVarianceProxy rewardBound mdp.horizon
173   have hadapted : StronglyAdapted F Y := by
174     intro i
175     simpa [F, Y] using
176       (mdp.measurable_sampledCumulativeReturnDeviation policy).stronglyMeasurable
177   have hzero : ProbabilityTheory.HasSubgaussianMGF (Y 0) (cY 0) mu := by
178     simpa [Y, cY, mu] using
179       source.stochasticTrajectoryMeasure_sampledCumulativeReturnDeviation_hasSubgaussianMGF
180         policy initialState rewardBound rewardVarianceProxy hrewardBound law
181   have hsucc : forall i, i < 1 - 1 ->
182       ProbabilityTheory.HasCondSubgaussianMGF
183         (F i) (F.le i) (Y (i + 1)) (cY (i + 1)) mu := by
184     intro i hi
185     omega
186   have hvarianceSum :
187       0 < ((((Finset.range 1).sum cY : NNReal) : Real)) := by
188     simpa [cY] using htotal
189   simpa [mu, Y, cY] using
190     (Concentration.condSubGaussian_sum_abs_tail_ennreal_delta_of_stronglyAdapted
191       hadapted hzero 1 hsucc hvarianceSum delta hdelta hdelta_le_one)
