122 /-- A successor event inherits a uniform bound on every history fiber. -/
123 theorem trajectoryMeasure_successorBadEvent_le
124     {mdp : MDP State Action} {initialState : Measure State}
125     [IsProbabilityMeasure initialState] {episodes : Nat}
126     (source : AdaptiveStochasticEpisodeBatchSource mdp initialState episodes)
127     (n : Nat)
128     {bad : Set (StochasticEpisodeBatchPrefix mdp episodes n ×
129       StochasticEpisodeBatch mdp episodes)}
130     (hbad : MeasurableSet bad) (budget : ENNReal)
131     (hfiber : forall history,
132       source.batchKernel n history (Prod.mk history ⁻¹' bad) <= budget) :
133     source.trajectoryMeasure (successorBadEvent n bad) <= budget := by
134   let prefixMeasure :=
135     source.trajectoryMeasure.map (Preorder.frestrictLe n)
136   let pairMap : StochasticEpisodeBatchTrajectory mdp episodes ->
137       StochasticEpisodeBatchPrefix mdp episodes n ×
138         StochasticEpisodeBatch mdp episodes :=
139     fun trajectory =>
140       (Preorder.frestrictLe n trajectory, trajectory (n + 1))
141   have hpairMap : Measurable pairMap := by fun_prop
142   haveI : IsProbabilityMeasure prefixMeasure := by
143     dsimp only [prefixMeasure]
144     exact Measure.isProbabilityMeasure_map (by fun_prop)
145   calc
146     source.trajectoryMeasure (successorBadEvent n bad) =
147         source.trajectoryMeasure.map pairMap bad := by
148       rw [Measure.map_apply hpairMap hbad]
149       rfl
150     _ = (prefixMeasure ⊗ₘ source.batchKernel n) bad := by
151       rw [source.trajectoryMeasure_prefix_compProd n]
152     _ = ∫⁻ history, source.batchKernel n history
153           (Prod.mk history ⁻¹' bad) ∂prefixMeasure :=
154       Measure.compProd_apply hbad

318 /-- The two local shares are the local share of their sum. -/
319 theorem ofReal_multiBatchLocalDelta_add
320     (rounds : Nat) {countDelta rewardDelta : Real}
321     (hcountDelta : 0 <= countDelta) (hrewardDelta : 0 <= rewardDelta) :
322     ENNReal.ofReal (multiBatchLocalDelta rounds countDelta) +
323         ENNReal.ofReal (multiBatchLocalDelta rounds rewardDelta) =
324       ENNReal.ofReal
325         (multiBatchLocalDelta rounds (countDelta + rewardDelta)) := by
326   have hrounds : (0 : Real) <= (rounds : Real) := Nat.cast_nonneg _
327   have hcountLocal : 0 <= multiBatchLocalDelta rounds countDelta := by
328     exact div_nonneg hcountDelta hrounds
329   have hrewardLocal : 0 <= multiBatchLocalDelta rounds rewardDelta := by
330     exact div_nonneg hrewardDelta hrounds
331   calc
332     ENNReal.ofReal (multiBatchLocalDelta rounds countDelta) +
333         ENNReal.ofReal (multiBatchLocalDelta rounds rewardDelta) =
334       ENNReal.ofReal
335         (multiBatchLocalDelta rounds countDelta +
336           multiBatchLocalDelta rounds rewardDelta) :=
337       (ENNReal.ofReal_add hcountLocal hrewardLocal).symm
338     _ = ENNReal.ofReal
339         (multiBatchLocalDelta rounds (countDelta + rewardDelta)) := by
340       congr 1
341       unfold multiBatchLocalDelta
342       ring

348 theorem trajectoryMeasure_adaptiveAllCoordinateEmpiricalModelBadEvent_le
349     [StandardBorelSpace State] [StandardBorelSpace Action]
350     {mdp : MDP State Action} {initialState : Measure State}
351     [IsProbabilityMeasure initialState] {episodes : Nat}
352     (source : AdaptiveStochasticEpisodeBatchSource mdp initialState episodes)
353     (rounds : Nat) (hrounds : 0 < rounds) (hepisodes : 0 < episodes)
354     (varianceProxy : NNReal)
355     (law : source.rewardSource.UniformSubgaussianRewardLaw varianceProxy)
356     (htotal : 0 < ((((episodes : NNReal) * varianceProxy : NNReal) : Real)))
357     (countDelta : Real) (hcountDelta : 0 < countDelta)
358     (hcountDelta_le_one : countDelta <= 1)
359     (rewardDelta : Real) (hrewardDelta : 0 < rewardDelta)
360     (hrewardDelta_le_one : rewardDelta <= 1)
361     (hsuccessor : forall n, n + 1 < rounds ->
362       MeasurableSet (source.successorAllCoordinateEmpiricalModelBadEvent
363         rounds varianceProxy countDelta rewardDelta n)) :
364     source.trajectoryMeasure
365         (source.adaptiveAllCoordinateEmpiricalModelBadEvent
366           rounds varianceProxy countDelta rewardDelta) <=
367       ENNReal.ofReal countDelta + ENNReal.ofReal rewardDelta := by
368   let initialBad := source.initialAllCoordinateEmpiricalModelBadEvent
369     rounds varianceProxy countDelta rewardDelta
370   let successorBad := source.successorAllCoordinateEmpiricalModelBadEvent
371     rounds varianceProxy countDelta rewardDelta
372   have hinitialMeasurable : MeasurableSet initialBad := by
373     exact source.measurableSet_initialAllCoordinateEmpiricalModelBadEvent
374       rounds varianceProxy countDelta rewardDelta
375   have hlocal :
376       ENNReal.ofReal (multiBatchLocalDelta rounds countDelta) +
377           ENNReal.ofReal (multiBatchLocalDelta rounds rewardDelta) =
378         ENNReal.ofReal
379           ((countDelta + rewardDelta) / ((Finset.univ : Finset (Fin rounds)).card : Real)) := by
380     rw [ofReal_multiBatchLocalDelta_add rounds hcountDelta.le hrewardDelta.le]
381     simp [multiBatchLocalDelta]
382   have hroundTail : forall round : Fin rounds,
383       source.trajectoryMeasure
384           (roundBadEvent initialBad successorBad round) <=
385         ENNReal.ofReal
386           ((countDelta + rewardDelta) /
387             ((Finset.univ : Finset (Fin rounds)).card : Real)) := by
388     intro round
389     cases round with
390     | mk value hvalue =>
391         cases value with
392         | zero =>
393             rw [roundBadEvent,
394               source.trajectoryMeasure_initialBadEvent hinitialMeasurable]
395             exact (source.initialAllCoordinateEmpiricalModelBadEvent_le
396               rounds hrounds hepisodes varianceProxy law htotal
397                 countDelta hcountDelta hcountDelta_le_one
398                 rewardDelta hrewardDelta hrewardDelta_le_one).trans_eq hlocal
399         | succ n =>
400             exact (source.trajectoryMeasure_successorBadEvent_le n
401               (hsuccessor n hvalue)
402               (ENNReal.ofReal
403                 ((countDelta + rewardDelta) /
404                   ((Finset.univ : Finset (Fin rounds)).card : Real)))
405               (fun history =>
406                 (source.successorAllCoordinateEmpiricalModelBadEvent_fiber_le
407                   rounds hrounds hepisodes varianceProxy law htotal
408                     countDelta hcountDelta hcountDelta_le_one
409                     rewardDelta hrewardDelta hrewardDelta_le_one n history).trans_eq
410                       hlocal)).trans_eq rfl
411   letI : Nonempty (Fin rounds) := ⟨⟨0, hrounds⟩⟩
412   have hglobal :=
413     ProbabilityUnionBound.measure_biUnion_finset_le_of_uniform
414       source.trajectoryMeasure (Finset.univ : Finset (Fin rounds))
415       Finset.univ_nonempty (countDelta + rewardDelta)
416       (fun round => roundBadEvent initialBad successorBad round)
417       (fun round _ => hroundTail round)
418   rw [← ENNReal.ofReal_add hcountDelta.le hrewardDelta.le]
419   simpa [adaptiveAllCoordinateEmpiricalModelBadEvent,
420     finiteHorizonBadEvent, initialBad, successorBad] using hglobal
