5 /-!
6 # Fixed-window stopping-time L1 natural causal consistency
7 
8 This module transports the compiled all-prefix L1 consistency theorem through
9 stopping times constrained to a fixed deterministic window around each natural
10 prefix.  The selector is charged to a finite sum of shifted coordinate L1
11 envelopes.  No optional-stopping identity or independence assumption is used.
12 -/

30 /-- A finite `WithTop Nat` window selects one explicit natural offset. -/
31 theorem exists_window_offset_untopA_eq_of_withTop_bounds
32     {Omega : Type*} (tau : Omega -> WithTop Nat) (start window : Nat)
33     (htau_lower : forall omega, (start : WithTop Nat) <= tau omega)
34     (htau_upper : forall omega,
35       tau omega <= ((start + window : Nat) : WithTop Nat))
36     (omega : Omega) :
37     exists offset, offset ∈ Finset.range (window + 1) /\
38       (tau omega).untopA = start + offset := by
39   have hne : tau omega ≠ (⊤ : WithTop Nat) :=
40     ne_top_of_le_ne_top WithTop.coe_ne_top (htau_upper omega)
41   have hcoe : (((tau omega).untopA : Nat) : WithTop Nat) = tau omega := by
42     rw [WithTop.untopA_eq_untop hne]
43     exact WithTop.coe_untop _ hne
44   have hlower : start <= (tau omega).untopA := by
45     have hlowerTop := htau_lower omega
46     rw [← hcoe] at hlowerTop
47     exact WithTop.coe_le_coe.mp hlowerTop
48   have hupper : (tau omega).untopA <= start + window := by
49     have hupperTop := htau_upper omega
50     rw [← hcoe] at hupperTop
51     exact WithTop.coe_le_coe.mp hupperTop
52   obtain ⟨offset, hoffset⟩ := Nat.exists_eq_add_of_le hlower
53   refine ⟨offset, Finset.mem_range.mpr ?_, hoffset⟩
54   rw [Nat.lt_add_one_iff]
55   rw [hoffset] at hupper
56   exact Nat.add_le_add_iff_left.mp hupper
