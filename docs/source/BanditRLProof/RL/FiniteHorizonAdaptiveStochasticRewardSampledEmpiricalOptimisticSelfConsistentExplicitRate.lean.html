3 /-!
4 # Explicit finite-window rate for sampled self-consistent optimism
5 
6 This consumer keeps the scheduled source and all its regularity contracts
7 unchanged.  It combines the compiled bounds
8 
9 * `rewardBudget < scale^-2`,
10 * `q < 4 * card State * horizon / scale^2`, and
11 * `q <= 1 / 2`
12 
13 to expose closed finite-window failure and realized-regret envelopes.  The
14 proof route uses only the local self-consistent schedule, the explicit
15 exploration charge, the normalized return-radius envelope, and ordered-field
16 algebra (`MLIB-ORDER-ALGEBRA`, `MLIB-REAL-LOG-SQRT`, and
17 `MLIB-ASYMPTOTICS`).  No theorem-card or proof-weapon statement is used as a
18 Lean proof.
19 
20 Failure policy: preserve actual sampled rewards, three separate confidence
21 shares, global return centering, and changing finite-window sample spaces.  A
22 failure here must remain a rate-algebra or local-API issue; it must not be
23 repaired by weakening the source law, replacing sampled rewards by means, or
24 claiming a common process, anytime control, minimax regret, or complete UCB-VI.
25 -/

61 /-- The exact fixed-point transition budget has an explicit scale-squared rate. -/
62 theorem selfConsistentScheduledTransitionBudget_le_rateEnvelope
63     (mdp : MDP State Action) (varianceProxy : NNReal)
64     {baseVisitFloor : Real} (hhorizon : 0 < mdp.horizon)
65     (hbaseVisitFloor : 0 < baseVisitFloor) (n : Nat) :
66     selfConsistentScheduledTransitionBudget mdp varianceProxy baseVisitFloor n <=
67       selfConsistentScheduledTransitionBudgetRateEnvelope mdp n := by
68   let witnessState : State := Classical.choice inferInstance
69   let scale : Real := AdaptiveEpisodeBatchSource.decayingExplorationScale n
70   let rewardBudget :=
71     selfConsistentScheduledRewardBudget mdp varianceProxy baseVisitFloor n
72   let contraction :=
73     selfConsistentScheduledTransitionContraction mdp varianceProxy
74       baseVisitFloor n
75   let contractionEnvelope :=
76     selfConsistentScheduledTransitionContractionEnvelope mdp n
77   have hscalePos : 0 < scale := by
78     dsimp [scale]
79     exact_mod_cast AdaptiveEpisodeBatchSource.decayingExplorationScale_pos n
80   have hscaleFour : 4 <= scale ^ 2 := by
81     have hscaleTwo : 2 <= scale := by
82       simp [scale, AdaptiveEpisodeBatchSource.decayingExplorationScale]
83     nlinarith
84   have hinvQuarter : 1 / scale ^ 2 <= (1 : Real) / 4 :=
85     one_div_le_one_div_of_le (by norm_num) hscaleFour
86   have hrewardNonneg : 0 <= rewardBudget := by
87     dsimp [rewardBudget]
88     exact selfConsistentScheduledRewardBudget_nonneg mdp witnessState
89       varianceProxy hhorizon hbaseVisitFloor n
90   have hrewardScale : rewardBudget <= 1 / scale ^ 2 := by
91     dsimp [rewardBudget, scale]
92     exact (selfConsistentScheduledRewardBudget_lt_inv_scale_sq mdp
93       varianceProxy hhorizon hbaseVisitFloor n).le
94   have hrewardQuarter : rewardBudget <= (1 : Real) / 4 :=
95     hrewardScale.trans hinvQuarter
96   have hcontractionNonneg : 0 <= contraction := by
97     dsimp [contraction]
98     exact selfConsistentScheduledTransitionContraction_nonneg mdp witnessState
99       varianceProxy hhorizon hbaseVisitFloor n
100   have hcontractionHalf : contraction <= (1 : Real) / 2 := by
101     have hhalf :=
102       (selfConsistentScheduled_countMargin_and_halfContraction mdp witnessState
103         varianceProxy hhorizon hbaseVisitFloor n).2
104     simpa [contraction, selfConsistentScheduledTransitionContraction] using hhalf
105   have hcontractionEnvelope : contraction <= contractionEnvelope := by
106     dsimp [contraction, contractionEnvelope]
107     exact (selfConsistentScheduledTransitionContraction_lt_envelope mdp
108       varianceProxy hhorizon hbaseVisitFloor n).le
109   have hdenominatorPos : 0 < 1 - contraction := by linarith
110   have hrewardProduct :
111       0 <= contraction * ((1 : Real) / 4 - rewardBudget) :=
112     mul_nonneg hcontractionNonneg (sub_nonneg.mpr hrewardQuarter)
113   have hcontractionProduct :
114       0 <= contraction * ((1 : Real) / 2 - contraction) :=
115     mul_nonneg hcontractionNonneg (sub_nonneg.mpr hcontractionHalf)
116   have hfixedPoint :
117       contraction * (1 + 2 * rewardBudget) / (1 - contraction) <=
118         3 * contraction := by
119     rw [div_le_iff₀ hdenominatorPos]
120     nlinarith
121   change contraction * (1 + 2 * rewardBudget) / (1 - contraction) <=
122     3 * contractionEnvelope
123   exact hfixedPoint.trans
124     (mul_le_mul_of_nonneg_left hcontractionEnvelope (by norm_num))
