3 /-!
4 # Adaptive cumulative Hoeffding UCB-VI foundation
5 
6 This module records the first generated-process leaves for the Hoeffding
7 UCB-VI route.  It adds the reward-sum component that was absent from the
8 existing cumulative transition-count state, proves exact prefix updates and
9 measurability, and defines a one-episode-at-a-time adaptive source whose next
10 policy is the cumulative empirical optimistic policy with a clipped
11 inverse-square-root Hoeffding radius.
12 
13 The source is genuinely generated by `AdaptiveEpisodeBatchSource.trajectoryMeasure`.
14 It has no forced uniform-exploration mixture and its policy depends on the
15 complete generated prefix.  The definitions are total at zero counts and at
16 invalid confidence inputs; under the eventual task contract (`0 < delta <= 1`,
17 positive horizon and episode budget) the logarithmic factor agrees with the
18 usual UCBVI-CH factor.
19 
20 This is not yet the high-probability UCB-VI regret theorem.  In particular,
21 the existing `EstimatedModelPlan` recursion does not yet implement the
22 paper's cross-episode `min (Q_previous, H, backup)` clipping, and the adaptive
23 self-normalized transition-value confidence and cumulative martingale/bonus
24 assembly remain downstream obligations.
25 -/

87 def cumulativeRewardSumSummary
88     {mdp : MDP State Action} {episodes n : Nat}
89     (history : EpisodeBatchPrefix mdp episodes n) : RewardSumSummary mdp :=
90   fun stage state action =>
91     ∑ i : Fin (n + 1),

93         stage state action

95 omit [Nonempty State] [Nonempty Action] in
96 /-- A fixed batch reward sum is measurable. -/
97 theorem measurable_rewardSum_coordinate
98     {mdp : MDP State Action} {episodes : Nat}
99     (stage : Fin mdp.horizon) (state : State) (action : Action) :
100     Measurable fun batch : EpisodeBatch mdp episodes =>
101       batch.rewardSum stage state action := by
102   classical
103   refine Finset.measurable_sum Finset.univ fun episode _ => ?_
104   let step : EpisodeBatch mdp episodes -> EpisodeStep State Action :=
105     fun batch => batch episode stage
106   have hstep : Measurable step :=
107     (measurable_pi_apply stage).comp (measurable_pi_apply episode)
108   have hstate : Measurable fun batch : EpisodeBatch mdp episodes =>
109       (step batch).state := EpisodeStep.measurable_state.comp hstep
110   have haction : Measurable fun batch : EpisodeBatch mdp episodes =>
111       (step batch).action := EpisodeStep.measurable_action.comp hstep
112   have hreward : Measurable fun batch : EpisodeBatch mdp episodes =>
113       (step batch).reward := EpisodeStep.measurable_reward.comp hstep
114   exact Measurable.ite
115     ((measurableSet_eq_fun hstate measurable_const).inter
116       (measurableSet_eq_fun haction measurable_const))
117     hreward measurable_const

304 /-- Total number of environment time steps after `episodes` episodes. -/

308 /-- Integer numerator used in the UCBVI-CH logarithmic factor. -/
309 def confidenceNumerator
310     (mdp : MDP State Action) (episodes : Nat) : Nat :=
311   5 * mdp.horizon * Fintype.card State * Fintype.card Action *
312     totalSteps mdp episodes

314 /--
315 Safe total version of the UCBVI-CH logarithmic factor.  The `max 1` only
316 specifies invalid/degenerate inputs and disappears under the positive task
317 contract.
318 -/
319 noncomputable def logFactor
320     (mdp : MDP State Action) (episodes : Nat) (delta : Real) : Real :=
321   Real.log (max 1
322     (((confidenceNumerator (State := State) (Action := Action)
323       mdp episodes : Nat) : Real) / delta))

325 omit [DecidableEq State] [DecidableEq Action]
326     [MeasurableSingletonClass State] [MeasurableSingletonClass Action] in
327 theorem one_le_confidenceNumerator_div
328     (mdp : MDP State Action) (episodes : Nat) (delta : Real)
329     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
330     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
331     1 <= ((confidenceNumerator (State := State) (Action := Action)
332       mdp episodes : Nat) : Real) / delta := by
333   have hnumeratorPos :
334       0 < confidenceNumerator (State := State) (Action := Action)
335         mdp episodes := by
336     unfold confidenceNumerator totalSteps
337     positivity
338   have honeNumerator :
339       (1 : Real) <=
340         (confidenceNumerator (State := State) (Action := Action)
341           mdp episodes : Nat) := by
342     exact_mod_cast hnumeratorPos
343   apply (le_div_iff₀ hdelta).2
344   simpa using hdelta_le_one.trans honeNumerator
