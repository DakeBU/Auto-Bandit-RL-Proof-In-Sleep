22 theorem card_bernsteinCoordinateIndex
23     (mdp : MDP State Action) (episodes : Nat) :
24     Fintype.card (BernsteinCoordinateIndex mdp episodes) =
25       episodes * Fintype.card State * Fintype.card Action *
26         Fintype.card State * (episodes * mdp.horizon) := by
27   let e : BernsteinCoordinateIndex mdp episodes ≃
28       Fin episodes × State × Action × State × Fin (episodes * mdp.horizon) := {
29     toFun := fun i => (i.round, i.state, i.action, i.nextState, i.count)
30     invFun := fun i => ⟨i.1, i.2.1, i.2.2.1, i.2.2.2.1, i.2.2.2.2⟩
31     left_inv := by intro i; cases i; rfl
32     right_inv := by
33       intro i
34       rcases i with ⟨round, state, action, nextState, count⟩
35       rfl }
36   rw [Fintype.card_congr e]
37   simp only [Fintype.card_prod, Fintype.card_fin]
38   ring

59 /-- Exact exponential simplification at the paper logarithmic factor. -/
60 theorem exp_neg_two_mul_logFactor_eq
61     (mdp : MDP State Action) (episodes : Nat) (delta : Real)
62     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
63     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
64     Real.exp (-2 * logFactor (State := State) (Action := Action)
65         mdp episodes delta) =
66       (delta /
67         (confidenceNumerator (State := State) (Action := Action)
68           mdp episodes : Nat)) ^ 2 := by
69   let Q := (confidenceNumerator (State := State) (Action := Action)
70     mdp episodes : Nat) / delta
71   have hQ : 0 < Q := by
72     dsimp [Q]
73     apply div_pos _ hdelta
74     exact_mod_cast (show 0 < confidenceNumerator
75       (State := State) (Action := Action) mdp episodes by
76         unfold confidenceNumerator totalSteps
77         positivity)
78   rw [logFactor_eq_paper mdp episodes delta hhorizon hepisodes
79     hdelta hdelta_le_one]
80   change Real.exp (-2 * Real.log Q) = _
81   rw [show -2 * Real.log Q = -(2 * Real.log Q) by ring]
82   rw [Real.exp_neg]
83   rw [show (2 : Real) * Real.log Q = (2 : Nat) * Real.log Q by norm_num]
84   rw [Real.exp_nat_mul, Real.exp_log hQ]
85   dsimp [Q]
86   field_simp

94     Real.exp (-2 * logFactor (State := State) (Action := Action)
95         mdp episodes delta ^ 2) <=
96       Real.exp (-2 * logFactor (State := State) (Action := Action)

116 /-- The complete coordinate-plus-optimal-tail confidence family consumes at
117 most one fifth of `delta`. -/
118 theorem simultaneousTransitionTailBudget_le_fifth
119     (mdp : MDP State Action) (episodes : Nat) (delta : Real)
120     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
121     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
122     (Fintype.card (BernsteinCoordinateIndex mdp episodes) : ENNReal) *
123           (2 * ENNReal.ofReal (Real.exp (-2 *
124             logFactor (State := State) (Action := Action)
125               mdp episodes delta))) +
126         (Fintype.card (OptimalTailIndex mdp episodes) : ENNReal) *
127           (2 * ENNReal.ofReal (Real.exp (-2 *
128             logFactor (State := State) (Action := Action)
129               mdp episodes delta ^ 2))) <=
130       ENNReal.ofReal (delta / 5) := by
131   let H : Real := mdp.horizon
132   let K : Real := episodes
133   let S : Real := Fintype.card State
134   let A : Real := Fintype.card Action
135   let B : Real := confidenceNumerator (State := State) (Action := Action)
136     mdp episodes
137   let L := logFactor (State := State) (Action := Action) mdp episodes delta
138   have hH : 1 <= H := by
139     dsimp [H]
140     exact_mod_cast Nat.one_le_iff_ne_zero.mpr hhorizon.ne'
141   have hK : 1 <= K := by
142     dsimp [K]
143     exact_mod_cast Nat.one_le_iff_ne_zero.mpr hepisodes.ne'
144   have hS : 1 <= S := by
145     dsimp [S]
146     exact_mod_cast Nat.one_le_iff_ne_zero.mpr (Fintype.card_ne_zero (α := State))
147   have hA : 1 <= A := by
148     dsimp [A]
149     exact_mod_cast Nat.one_le_iff_ne_zero.mpr (Fintype.card_ne_zero (α := Action))
150   have hB : B = 5 * H * S * A * (K * H) := by
151     dsimp [B, H, K, S, A]
152     norm_num [confidenceNumerator, totalSteps]
153   have hexp : Real.exp (-2 * L) = (delta / B) ^ 2 := by
154     dsimp [L, B]
155     exact exp_neg_two_mul_logFactor_eq mdp episodes delta
156       hhorizon hepisodes hdelta hdelta_le_one
157   have hexpSq : Real.exp (-2 * L ^ 2) <= (delta / B) ^ 2 := by
158     rw [← hexp]
159     dsimp [L]
160     exact exp_neg_two_mul_logFactor_sq_le mdp episodes delta
161       hhorizon hepisodes hdelta hdelta_le_one
162   have hcoordCard :
163       (Fintype.card (BernsteinCoordinateIndex mdp episodes) : Real) =
164         K * S * A * S * (K * H) := by
165     rw [card_bernsteinCoordinateIndex]
166     dsimp [H, K, S, A]
167     norm_num only [Nat.cast_mul]
168   have htailCard :
169       (Fintype.card (OptimalTailIndex mdp episodes) : Real) =
170         (K * H) * S * A * (K * H) := by
171     rw [card_optimalTailIndex]
172     dsimp [H, K, S, A]
173     norm_num only [Nat.cast_mul]
174   have hdeltaNonneg : 0 <= delta := hdelta.le
175   have hcoordReal :
176       (Fintype.card (BernsteinCoordinateIndex mdp episodes) : Real) *
177           (2 * Real.exp (-2 * L)) <= 2 * delta / 25 := by
178     rw [hcoordCard, hexp, hB]
179     have hdenom : 0 < 5 * H * S * A * (K * H) := by positivity
180     have hscale : delta <= A * H ^ 3 := by
181       have hH3 : 1 <= H ^ 3 := one_le_pow₀ hH
182       have hone : 1 <= A * H ^ 3 := by
183         nlinarith [mul_nonneg (sub_nonneg.mpr hA)
184           (zero_le_one.trans hH3)]
185       exact hdelta_le_one.trans hone
186     field_simp
187     nlinarith
188   have htailReal :
189       (Fintype.card (OptimalTailIndex mdp episodes) : Real) *
190           (2 * Real.exp (-2 * L ^ 2)) <= 2 * delta / 25 := by
191     calc
192       _ <= ((K * H) * S * A * (K * H)) *
193           (2 * (delta / B) ^ 2) := by
194         rw [htailCard]
195         gcongr
196       _ <= 2 * delta / 25 := by
197         rw [hB]
198         have hdenom : 0 < 5 * H * S * A * (K * H) := by positivity
199         have hscale : delta <= H ^ 2 * S * A := by
200           have hH2 : 1 <= H ^ 2 := one_le_pow₀ hH
201           have hHS : 1 <= H ^ 2 * S := by
202             nlinarith [mul_nonneg (sub_nonneg.mpr hS)
203               (zero_le_one.trans hH2)]
204           have hone : 1 <= H ^ 2 * S * A := by
205             nlinarith [mul_nonneg (sub_nonneg.mpr hA)
206               (zero_le_one.trans hHS)]
207           exact hdelta_le_one.trans hone
208         field_simp
209         nlinarith
210   have hcoordENN := ofReal_card_mul_two_mul_exp_le
211     (Fintype.card (BernsteinCoordinateIndex mdp episodes))
212     (Real.exp (-2 * L)) (2 * delta / 25) (Real.exp_pos _).le
213     (div_nonneg (mul_nonneg (by norm_num) hdelta.le) (by norm_num)) hcoordReal
214   have htailENN := ofReal_card_mul_two_mul_exp_le
215     (Fintype.card (OptimalTailIndex mdp episodes))
216     (Real.exp (-2 * L ^ 2)) (2 * delta / 25) (Real.exp_pos _).le
217     (div_nonneg (mul_nonneg (by norm_num) hdelta.le) (by norm_num)) htailReal
218   calc
219     _ <= ENNReal.ofReal (2 * delta / 25) +
220         ENNReal.ofReal (2 * delta / 25) := add_le_add hcoordENN htailENN
221     _ = ENNReal.ofReal (4 * delta / 25) := by
222       have hsmall : 0 <= 2 * delta / 25 :=
223         div_nonneg (mul_nonneg (by norm_num) hdelta.le) (by norm_num)
224       rw [← ENNReal.ofReal_add hsmall hsmall]
225       congr 1
226       ring
227     _ <= ENNReal.ofReal (delta / 5) := by
228       apply ENNReal.ofReal_le_ofReal
229       have hfour : 4 * delta <= 5 * delta :=
230         mul_le_mul_of_nonneg_right (by norm_num) hdelta.le
231       nlinarith
