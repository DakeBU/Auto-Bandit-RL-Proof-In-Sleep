29 /-- The logarithmic factor shared by every queried cumulative prefix. -/
30 noncomputable def cumulativeInverseSqrtLogFactor
31     (mdp : MDP State Action) (rounds : Nat) (delta : Real) : Real :=
32   Real.log (2 / cumulativeCountLocalDelta mdp rounds delta)
33 
34 /-- The deterministic coefficient multiplying each cumulative count radius. -/
35 noncomputable def cumulativeInverseSqrtCoverCoefficient
36     (mdp : MDP State Action) (rewardBound budget : Real) : Real :=
37   2 * (Fintype.card State : Real) * (mdp.horizon : Real) *
38     (rewardBound + budget)

40 /--
41 A sufficient episode threshold for both a half expected-visit margin and the
42 budget branch of the capped transition-radius cover.
43 -/
44 noncomputable def cumulativeInverseSqrtCalibrationEpisodeThreshold
45     (mdp : MDP State Action) (rounds : Nat)
46     (delta visitFloor rewardBound budget : Real) : Real :=
47   max
48     (2 * cumulativeInverseSqrtLogFactor mdp rounds delta / visitFloor ^ 2)
49     (2 * (cumulativeInverseSqrtCoverCoefficient mdp rewardBound budget) ^ 2 *
50       cumulativeInverseSqrtLogFactor mdp rounds delta /
51         (budget ^ 2 * visitFloor ^ 2))

53 /-- Exact square of one cumulative coordinate confidence radius. -/
54 theorem cumulativeCoordinateConfidenceRadius_sq_eq
55     (mdp : MDP State Action) {episodes rounds prefixRounds : Nat}
56     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
57     {delta : Real} (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
58     (cumulativeCoordinateConfidenceRadius episodes prefixRounds
59         (cumulativeCountLocalDelta mdp rounds delta)) ^ 2 =
60       (prefixRounds : Real) * (episodes : Real) / 2 *
61         cumulativeInverseSqrtLogFactor mdp rounds delta := by
62   letI : Nonempty (Fin rounds × CountCoordinate mdp) :=
63     cumulativeCountIndex_nonempty hhorizon hrounds
64   have hlocalPos : 0 < cumulativeCountLocalDelta mdp rounds delta :=
65     cumulativeCountLocalDelta_pos inferInstance hdelta
66   have hlocalLeOne : cumulativeCountLocalDelta mdp rounds delta <= 1 :=
67     cumulativeCountLocalDelta_le_one inferInstance hdelta hdelta_le_one
68   rw [cumulativeCoordinateConfidenceRadius,
69     Concentration.subGaussianSumConfidenceRadius_sq _ _ hlocalPos hlocalLeOne]
70   have hvariance :
71       (((cumulativeCoordinateVarianceProxy episodes prefixRounds : NNReal) : Real)) =
72         (prefixRounds : Real) * (episodes : Real) / 4 := by
73     rw [cumulativeCoordinateVarianceProxy]
74     simp only [Finset.sum_const, Finset.card_range]
75     rw [MarkovPolicy.iidBernoulliVarianceProxy_eq]
76     push_cast
77     ring
78   rw [hvariance]
79   unfold cumulativeInverseSqrtLogFactor
80   ring

103 theorem half_cumulativePathVisitExpectedFloor_lt_lowerMargin_of_episodeThreshold

161       _ = ((prefixRounds : Real) * (episodes : Real) * visitFloor / 2) ^ 2 := by
162         ring
163   have hradiusLt : radius <

165     (sq_lt_sq₀ hradiusNonneg hhalfPos.le).mp hsquare

359 The round-indexed capped inverse-square-root envelopes sum to the minimum of

362 theorem sum_cumulativeInverseSqrtRadiusEnvelope_le_explicit
363     (mdp : MDP State Action) {episodes rounds : Nat}
364     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
365     (hepisodes : 0 < episodes) {delta visitFloor rewardBound budget scale : Real}
366     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
367     (hvisitFloor : 0 < visitFloor) (hscale : 0 <= scale)
368     (hthreshold :
369       cumulativeInverseSqrtCalibrationEpisodeThreshold mdp rounds delta
370           visitFloor rewardBound budget < (episodes : Real)) :
371     (∑ round : Fin rounds,
372         cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta
373           visitFloor budget scale round) <=
374       cumulativeInverseSqrtEnvelopeSumBound episodes rounds visitFloor budget scale := by
375   let baseFloor := (episodes : Real) * visitFloor / 2
376   have hepisodesReal : (0 : Real) < episodes := by exact_mod_cast hepisodes
377   have hbaseFloor : 0 < baseFloor := by
378     dsimp [baseFloor]
379     positivity
380   have hlinear :
381       (∑ round : Fin rounds,
382           cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta

384         (rounds : Real) * budget := by
385     calc
386       (∑ round : Fin rounds,
387           cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta
388             visitFloor budget scale round) <=
389           ∑ _round : Fin rounds, budget := by
390             apply Finset.sum_le_sum
391             intro round _hround
392             exact min_le_left _ _
393       _ = (rounds : Real) * budget := by simp
394   have hsqrt :
395       (∑ round : Fin rounds,
396           cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta
397             visitFloor budget scale round) <=
398         2 * scale * Real.sqrt (rounds : Real) / Real.sqrt baseFloor := by
399     calc
400       (∑ round : Fin rounds,
401           cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta
402             visitFloor budget scale round) <=
403           ∑ round : Fin rounds,
404             (scale / Real.sqrt baseFloor) *
405               (1 / Real.sqrt (((round.val + 1 : Nat) : Real))) := by
406         apply Finset.sum_le_sum
407         intro round _hround
408         let prefixRounds : Nat := round.val + 1
409         let halfFloor := (prefixRounds : Real) * (episodes : Real) * visitFloor / 2
410         let margin := cumulativePathVisitLowerMargin
411           mdp episodes rounds delta visitFloor round
412         have hmargin : halfFloor < margin := by
413           simpa [halfFloor, margin, prefixRounds] using
414             half_cumulativePathVisitExpectedFloor_lt_lowerMargin_of_episodeThreshold
415               mdp hhorizon hrounds hepisodes hdelta hdelta_le_one hvisitFloor
416               hthreshold round
417         have hprefixPos : 0 < prefixRounds := by
418           dsimp [prefixRounds]
419           omega
420         have hhalfFloor : 0 < halfFloor := by
421           dsimp [halfFloor]
422           positivity
423         have hsqrtLe : Real.sqrt halfFloor <= Real.sqrt margin :=
424           Real.sqrt_le_sqrt hmargin.le
425         have hdivLe : scale / Real.sqrt margin <= scale / Real.sqrt halfFloor :=
426           div_le_div_of_nonneg_left hscale (Real.sqrt_pos.2 hhalfFloor) hsqrtLe
427         calc
428           cumulativeInverseSqrtRadiusEnvelope mdp episodes rounds delta
429               visitFloor budget scale round <=
430             scale / Real.sqrt margin := by
431               exact min_le_right _ _
432           _ <= scale / Real.sqrt halfFloor := hdivLe
433           _ = (scale / Real.sqrt baseFloor) *
434               (1 / Real.sqrt (((round.val + 1 : Nat) : Real))) := by
435             have hsqrtPrefix : 0 < Real.sqrt (prefixRounds : Real) := by positivity
436             have hsqrtBase : 0 < Real.sqrt baseFloor := Real.sqrt_pos.2 hbaseFloor
437             have hhalfEq : halfFloor = (prefixRounds : Real) * baseFloor := by
438               dsimp [halfFloor, baseFloor]
439               ring
440             rw [hhalfEq, Real.sqrt_mul (by positivity : (0 : Real) <= prefixRounds)]
441             dsimp [prefixRounds]
442             field_simp
443     _ = (scale / Real.sqrt baseFloor) *
444           (∑ round : Fin rounds,
445             1 / Real.sqrt (((round.val + 1 : Nat) : Real))) := by
446       rw [Finset.mul_sum]
447     _ = (scale / Real.sqrt baseFloor) *
448           (Finset.range rounds).sum (fun round =>
449             1 / Real.sqrt (((round + 1 : Nat) : Real))) := by
450       rw [Fin.sum_univ_eq_sum_range (fun round : Nat =>
451         1 / Real.sqrt (((round + 1 : Nat) : Real)))]
452     _ <= (scale / Real.sqrt baseFloor) *
453           (2 * Real.sqrt (rounds : Real)) := by
