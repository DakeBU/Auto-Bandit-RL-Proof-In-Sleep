6 # Finite-horizon Markov policy evaluation
7 
8 This module defines a Markov action kernel at every valid decision stage of a
9 finite-horizon MDP.  It constructs the induced next-state kernel, the policy
10 Bellman operator, and the backward finite-horizon policy value.  The terminal
11 and Bellman recursion theorems are policy-evaluation facts; no maximization,
12 optimality, occupancy, or regret statement is made here.
13 -/
14 
15 open MeasureTheory
16 
17 universe u v
18 
19 namespace BanditRLProof
20 namespace FiniteHorizonRL
21 
22 variable {State : Type u} {Action : Type v}
23     [MeasurableSpace State] [MeasurableSpace Action]
24     [Fintype State] [Fintype Action]

26 /-- A Markov action kernel for every decision stage before `mdp.horizon`. -/
27 structure MarkovPolicy (mdp : MDP State Action) where
28   actionKernel : (stage : Fin mdp.horizon) ->
29     ProbabilityTheory.Kernel State Action
30   actionKernel_isMarkov : forall stage,
31     ProbabilityTheory.IsMarkovKernel (actionKernel stage)
32 
33 namespace MarkovPolicy
34 
35 instance instActionKernelIsMarkovKernel
36     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
37     (stage : Fin mdp.horizon) :
38     ProbabilityTheory.IsMarkovKernel (policy.actionKernel stage) :=
39   policy.actionKernel_isMarkov stage
40 
41 /-- State transition kernel induced by sampling the policy action and then the MDP transition. -/
42 noncomputable def inducedStateKernel
43     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
44     (stage : Fin mdp.horizon) :
45   ProbabilityTheory.Kernel State State :=
46   (ProbabilityTheory.Kernel.compProd
47       (policy.actionKernel stage) mdp.transition).map Prod.snd
48 
49 instance instInducedStateKernelIsMarkovKernel
50     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
51     (stage : Fin mdp.horizon) :
52     ProbabilityTheory.IsMarkovKernel (policy.inducedStateKernel stage) := by
53   unfold inducedStateKernel
54   exact ProbabilityTheory.Kernel.IsMarkovKernel.map _ measurable_snd
55 
56 /-- Bellman expectation under one stage of the supplied Markov policy. -/
57 noncomputable def bellman
58     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
59     (stage : Fin mdp.horizon) (value : State -> Real)
60     (state : State) : Real :=
61   ∫ action, mdp.bellmanQ value state action
62     ∂policy.actionKernel stage state
63 
64 /-- A measurable continuation value gives a measurable policy Bellman value. -/
65 theorem measurable_bellman
66     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
67     (stage : Fin mdp.horizon) {value : State -> Real}
68     (hvalue : Measurable value) :
69     Measurable (policy.bellman stage value) := by
70   exact (mdp.measurable_bellmanQ hvalue).stronglyMeasurable
71     |>.integral_kernel_prod_right.measurable
72 
73 /--
74 Backward policy value indexed by the number of decisions remaining.  The first
75 kernel used at `remaining` is chronological stage `horizon - remaining`.
76 -/
77 noncomputable def valueRemaining
78     {mdp : MDP State Action} (policy : MarkovPolicy mdp) :
79     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
80   | 0, _ => fun _ => 0
81   | remaining + 1, hremaining =>
82       policy.bellman
83         ⟨mdp.horizon - (remaining + 1), by omega⟩
84         (policy.valueRemaining remaining (by omega))
85 
86 /-- The backward policy value is measurable for every valid remaining horizon. -/
87 theorem measurable_valueRemaining
88     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
89     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
90     Measurable (policy.valueRemaining remaining hremaining) := by
91   induction remaining with
92   | zero =>
93       simp [valueRemaining]
94   | succ remaining ih =>
95       rw [valueRemaining]
96       exact policy.measurable_bellman _ (ih (by omega))
97 
98 /-- Policy value at a chronological stage `stage <= horizon`. -/
99 noncomputable def valueAt
100     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
101     (stage : Nat) (_hstage : stage <= mdp.horizon) : State -> Real :=
102   policy.valueRemaining (mdp.horizon - stage) (Nat.sub_le _ _)
103 
104 /-- Every chronological policy-value surface is measurable. -/
105 theorem measurable_valueAt
106     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
107     (stage : Nat) (hstage : stage <= mdp.horizon) :
108     Measurable (policy.valueAt stage hstage) :=
109   policy.measurable_valueRemaining _ _
110 
111 /-- Transport `valueRemaining` across equality of the remaining horizon. -/
112 theorem valueRemaining_eq_of_eq
113     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
114     {left right : Nat} (hleft : left <= mdp.horizon)
115     (hright : right <= mdp.horizon) (h : left = right) :
116     policy.valueRemaining left hleft =
117       policy.valueRemaining right hright := by
118   subst right
119   rfl
120 
121 /-- The finite-horizon policy value is zero at the terminal stage. -/
122 @[simp]
123 theorem valueAt_horizon
124     {mdp : MDP State Action} (policy : MarkovPolicy mdp) :
125     policy.valueAt mdp.horizon le_rfl = fun _ => 0 := by
126   unfold valueAt
127   calc
128     policy.valueRemaining (mdp.horizon - mdp.horizon) _ =
129         policy.valueRemaining 0 (Nat.zero_le _) := by
130       apply policy.valueRemaining_eq_of_eq
131       exact Nat.sub_self mdp.horizon
132     _ = (fun _ => 0) := rfl
133 
134 /--
135 Finite-horizon policy evaluation satisfies the Bellman recursion at every
136 decision stage.
137 -/
138 theorem valueAt_bellman
139     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
140     (stage : Nat) (hstage : stage < mdp.horizon) :
141     policy.valueAt stage (Nat.le_of_lt hstage) =
142       policy.bellman ⟨stage, hstage⟩
143         (policy.valueAt (stage + 1) (by omega)) := by
144   have hremaining : mdp.horizon - stage =
145       (mdp.horizon - (stage + 1)) + 1 := by
146     omega
147   have hstageIndex :
148       (⟨mdp.horizon - ((mdp.horizon - (stage + 1)) + 1), by omega⟩ :
149           Fin mdp.horizon) = ⟨stage, hstage⟩ := by
150     apply Fin.ext
151     simp
152     omega
153   calc
154     policy.valueAt stage (Nat.le_of_lt hstage) =
155         policy.valueRemaining
156           ((mdp.horizon - (stage + 1)) + 1) (by omega) := by
157       apply policy.valueRemaining_eq_of_eq
158       exact hremaining
159     _ = policy.bellman
160           ⟨mdp.horizon - ((mdp.horizon - (stage + 1)) + 1), by omega⟩
161           (policy.valueRemaining (mdp.horizon - (stage + 1)) (by omega)) := by
162       rw [valueRemaining]
163     _ = policy.bellman ⟨stage, hstage⟩
164           (policy.valueAt (stage + 1) (by omega)) := by
165       rw [hstageIndex]
166       rfl
