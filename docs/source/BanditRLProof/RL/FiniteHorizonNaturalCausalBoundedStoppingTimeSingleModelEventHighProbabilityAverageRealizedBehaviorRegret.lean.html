3 /-!
4 # Bounded stopping-time regret with one horizon model event
5 
6 The finite stopped-prefix route is sharpened by charging model-confidence
7 failures only once at the deterministic horizon. Return deviations still use
8 an equal-share finite union over the possible positive stopped prefixes.
9 
10 Stopping-time regularity is used only for filtered measurability. The tail
11 proof is event containment and finite subadditivity, without optional stopping
12 or independence between prefix events.
13 -/

31 /-- A finite-horizon heterogeneous bad-event union is monotone in its outer
32 horizon when the underlying coordinate event family is unchanged. -/
33 theorem finiteHorizonBadEvent_mono
34     {mdp : MDP State Action} {episodes : Nat -> Nat}
35     {rounds maxRounds : Nat}
36     {initialBad : Set (StochasticEpisodeBatch mdp (episodes 0))}
37     {successorBad : (n : Nat) ->
38       Set (HeterogeneousStochasticEpisodeBatchPrefix mdp episodes n ×
39         StochasticEpisodeBatch mdp (episodes (n + 1)))}
40     (hrounds : rounds <= maxRounds) :
41     finiteHorizonBadEvent rounds initialBad successorBad ⊆
42       finiteHorizonBadEvent maxRounds initialBad successorBad := by
43   intro trajectory htrajectory
44   rw [finiteHorizonBadEvent] at htrajectory ⊢
45   rcases Set.mem_iUnion.mp htrajectory with ⟨round, hround⟩
46   refine Set.mem_iUnion.mpr ⟨⟨round, lt_of_lt_of_le round.isLt hrounds⟩, ?_⟩
47   simpa using hround
