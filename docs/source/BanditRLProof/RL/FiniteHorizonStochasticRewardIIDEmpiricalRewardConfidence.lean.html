5 /-!
6 # IID empirical reward confidence for finite-horizon stochastic rewards
7 
8 This module retains the sampled reward in every generated episode record.  It
9 proves that a fixed stage/state/action reward sum, centered by its stored MDP
10 mean and masked by the visit event, is sub-Gaussian across iid complete
11 trajectories.  The total proxy is the conservative episode-linear proxy
12 `episodes * varianceProxy`; no bounded sampled-reward or exact-count claim is
13 used.
14 -/

25 /-- A uniform fiberwise sub-Gaussian proxy survives an arbitrary probability mixture. -/
26 theorem hasSubgaussianMGF_compProd_of_forall
27     {Index : Type u} {Omega : Type v}
28     [MeasurableSpace Index] [MeasurableSpace Omega]
29     (mu : Measure Index) [IsProbabilityMeasure mu]
30     (kappa : ProbabilityTheory.Kernel Index Omega)
31     [ProbabilityTheory.IsMarkovKernel kappa]
32     (X : Index × Omega -> Real) (hX : Measurable X) (c : NNReal)
33     (hfiber : forall index,
34       ProbabilityTheory.HasSubgaussianMGF
35         (fun omega => X (index, omega)) c (kappa index)) :
36     ProbabilityTheory.HasSubgaussianMGF X c (mu.compProd kappa) := by
37   constructor
38   · intro t
39     have hfullMeas : Measurable (fun p => Real.exp (t * X p)) :=
40       Real.measurable_exp.comp (measurable_const.mul hX)
41     have hfullAE : AEStronglyMeasurable
42         (fun p => Real.exp (t * X p)) (mu.compProd kappa) :=
43       hfullMeas.aestronglyMeasurable
44     apply (Measure.integrable_compProd_iff hfullAE).2
45     constructor
46     · exact Filter.Eventually.of_forall fun index =>
47         (hfiber index).integrable_exp_mul t
48     · have hinnerAE : AEStronglyMeasurable
49           (fun index => integral (kappa index)
50             (fun omega => ‖Real.exp (t * X (index, omega))‖)) mu :=
51         hfullAE.norm.integral_kernel_compProd
52       refine Integrable.of_bound hinnerAE
53         (Real.exp ((c : Real) * t ^ 2 / 2)) ?_
54       exact Filter.Eventually.of_forall fun index => by
55         have heq :
56             (integral (kappa index)
57                 (fun omega => ‖Real.exp (t * X (index, omega))‖)) =
58               ProbabilityTheory.mgf
59                 (fun omega => X (index, omega)) (kappa index) t := by
60           simp [ProbabilityTheory.mgf, Real.norm_eq_abs]
61         rw [Real.norm_of_nonneg (integral_nonneg fun _ => norm_nonneg _), heq]
62         exact (hfiber index).mgf_le t
63   · intro t
64     have hfullMeas : Measurable (fun p => Real.exp (t * X p)) :=
65       Real.measurable_exp.comp (measurable_const.mul hX)
66     have hfullAE : AEStronglyMeasurable
67         (fun p => Real.exp (t * X p)) (mu.compProd kappa) :=
68       hfullMeas.aestronglyMeasurable
69     have hfullInt : Integrable (fun p => Real.exp (t * X p))
70         (mu.compProd kappa) := by
71       apply (Measure.integrable_compProd_iff hfullAE).2
72       constructor
73       · exact Filter.Eventually.of_forall fun index =>
74           (hfiber index).integrable_exp_mul t
75       · have hinnerAE : AEStronglyMeasurable
76             (fun index => integral (kappa index)
77               (fun omega => ‖Real.exp (t * X (index, omega))‖)) mu :=
78           hfullAE.norm.integral_kernel_compProd
79         refine Integrable.of_bound hinnerAE
80           (Real.exp ((c : Real) * t ^ 2 / 2)) ?_
81         exact Filter.Eventually.of_forall fun index => by
82           have heq :
83               (integral (kappa index)
84                   (fun omega => ‖Real.exp (t * X (index, omega))‖)) =
85                 ProbabilityTheory.mgf
86                   (fun omega => X (index, omega)) (kappa index) t := by
87             simp [ProbabilityTheory.mgf, Real.norm_eq_abs]
88           rw [Real.norm_of_nonneg (integral_nonneg fun _ => norm_nonneg _), heq]
89           exact (hfiber index).mgf_le t
90     rw [ProbabilityTheory.mgf, Measure.integral_compProd hfullInt]
91     have hinnerAE : AEStronglyMeasurable
92         (fun index => integral (kappa index)
93           (fun omega => Real.exp (t * X (index, omega)))) mu :=
94       hfullAE.integral_kernel_compProd
95     have hinnerInt : Integrable
96         (fun index => integral (kappa index)
97           (fun omega => Real.exp (t * X (index, omega)))) mu := by
98       refine Integrable.of_bound hinnerAE
99         (Real.exp ((c : Real) * t ^ 2 / 2)) ?_
100       exact Filter.Eventually.of_forall fun index => by
101         rw [Real.norm_of_nonneg (integral_nonneg fun _ => Real.exp_nonneg _)]
102         exact (hfiber index).mgf_le t
103     have hmono := integral_mono_ae hinnerInt (integrable_const _)
104       (Filter.Eventually.of_forall fun index => (hfiber index).mgf_le t)
105     simpa [ProbabilityTheory.mgf] using hmono

216 /-- A sampled reward centered at one target coordinate and masked by its visit event. -/
217 def maskedRewardDeviationAt (mdp : MDP State Action)
218     (initialState : State) {remaining : Nat}
219     (trace : RewardStepTrace Action State remaining)
220     (coordinate : Fin remaining) (state : State) (action : Action) : Real :=
221   if RewardStepTrace.stateAt initialState trace coordinate = state /\
222       (trace coordinate).1 = action then
223     (trace coordinate).2.1 - mdp.reward state action
224   else 0

318 /-- A visit-masked one-step reward deviation retains the common reward proxy. -/
319 theorem actionRewardStateKernel_maskedRewardDeviation_hasSubgaussianMGF
320     [StandardBorelSpace State] [StandardBorelSpace Action]
321     (source : MeanCompatibleRewardKernel mdp)
322     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
323     (initialState state : State) (action : Action) (varianceProxy : NNReal)
324     (law : source.UniformSubgaussianRewardLaw varianceProxy) :
325     ProbabilityTheory.HasSubgaussianMGF
326       (fun head : Action × (Real × State) =>
327         if initialState = state /\ head.1 = action then
328           head.2.1 - mdp.reward state action
329         else 0)
330       varianceProxy
331       (source.actionRewardStateKernel policy stage initialState) := by
332   by_cases hinitial : initialState = state
333   · subst initialState
334     let X : Action × (Real × State) -> Real := fun head =>
335       if head.1 = action then head.2.1 - mdp.reward state action else 0
336     let rewardNextKernel : ProbabilityTheory.Kernel Action (Real × State) :=
337       ProbabilityTheory.Kernel.sectR source.rewardNextStateKernel state
338     have hX : Measurable X := by
339       refine Measurable.ite
340         (measurable_fst (measurableSet_singleton action)) ?_ measurable_const
341       exact measurable_snd.fst.sub measurable_const
342     have hmix := Concentration.hasSubgaussianMGF_compProd_of_forall
343       (policy.actionKernel stage state) rewardNextKernel X hX
344       varianceProxy (fun candidate => by
345         by_cases hcandidate : candidate = action
346         · subst candidate
347           have hbase := law.hasSubgaussianMGF state action
348           have hlift := ProbabilityTheory.HasSubgaussianMGF.of_map
349             (μ := source.rewardNextStateKernel (state, action))
350             (Y := Prod.fst)
351             (X := fun reward : Real => reward - mdp.reward state action)
352             measurable_fst.aemeasurable (by
353               rw [rewardNextStateKernel,
354                 ProbabilityTheory.Kernel.prod_apply,
355                 Measure.map_fst_prod, measure_univ, one_smul]
356               exact hbase)
357           simpa [X, rewardNextKernel, ProbabilityTheory.Kernel.sectR_apply,
358             Function.comp_def] using hlift
359         · simpa [X, hcandidate, rewardNextKernel,
360             ProbabilityTheory.Kernel.sectR_apply] using
361             (Concentration.hasSubgaussianMGF_zero_of_proxy
362               (source.rewardNextStateKernel (state, candidate)) varianceProxy))
363     rw [actionRewardStateKernel,
364       ProbabilityTheory.Kernel.compProd_apply_eq_compProd_sectR]
365     simpa [rewardNextKernel, X] using hmix
366   · simpa [hinitial] using
367       (Concentration.hasSubgaussianMGF_zero_of_proxy
368         (source.actionRewardStateKernel policy stage initialState) varianceProxy)

428               ProbabilityTheory.Kernel.sectR_apply,

431         let consStep : (Action × (Real × State)) ×
432             RewardStepTrace Action State remaining ->
433             RewardStepTrace Action State (remaining + 1) := fun p =>
434           @Fin.cons remaining (fun _ => Action × (Real × State)) p.1 p.2
435         have hcons : Measurable consStep := RewardStepTrace.measurable_cons remaining
436         have htarget := hpair.congr_identDistrib
437           { aemeasurable_fst := hX.aemeasurable
438             aemeasurable_snd :=
439               (MDP.measurable_maskedRewardDeviationAt mdp initialState

441             map_eq := by
442               rw [Measure.map_map
443                 (MDP.measurable_maskedRewardDeviationAt mdp initialState
444                   tailCoordinate.succ state action) hcons]
445               congr 1
446               funext p
447               simp [X, consStep, MDP.maskedRewardDeviationAt] }
448         rw [stochasticTrajectoryKernelRemaining]
449         rw [ProbabilityTheory.Kernel.map_apply _ hcons]
450         rw [ProbabilityTheory.Kernel.compProd_apply_eq_compProd_sectR]
451         simpa [headKernel, tailKernel, tailKernelGlobal, consStep] using htarget
