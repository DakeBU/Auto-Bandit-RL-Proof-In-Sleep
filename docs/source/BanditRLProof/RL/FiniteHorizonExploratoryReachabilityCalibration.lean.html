5 # Exploratory state-reachability calibration
6 
7 This module turns a generated stage-state probability lower envelope into the
8 state/action expected-count margins needed by the adaptive empirical optimistic
9 confidence route.  Uniform exploration supplies only the action factor; state
10 reachability and the transition-bonus cover remain explicit contracts.
11 -/

27 /-- Real probability floor contributed by uniform exploration. -/
28 noncomputable def exploratoryActionProbabilityFloor
29     (Action : Type v) [Fintype Action] (explorationRate : NNReal) : Real :=
30   (explorationRate : Real) * (Fintype.card Action : Real)⁻¹

36 theorem exploratoryActionProbabilityFloor_nonneg
37     (explorationRate : NNReal) :
38     0 <= exploratoryActionProbabilityFloor Action explorationRate := by
39   exact mul_nonneg (NNReal.coe_nonneg explorationRate)
40     (inv_nonneg.mpr (Nat.cast_nonneg (Fintype.card Action)))

81 theorem exploratoryActionProbabilityFloor_le_exploratoryActionPMF_toReal
82     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
83     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
84     (stage : Fin mdp.horizon) (state : State) (action : Action) :
85     exploratoryActionProbabilityFloor Action explorationRate <=
86       (table.exploratoryActionPMF explorationRate hexplorationRate
87         stage state action).toReal := by
88   have hfloor := table.explorationRate_mul_inv_card_le_exploratoryActionPMF
89     explorationRate hexplorationRate stage state action
90   have hreal := ENNReal.toReal_mono
91     ((table.exploratoryActionPMF explorationRate hexplorationRate
92       stage state).apply_ne_top action) hfloor
93   simpa [exploratoryActionProbabilityFloor, ENNReal.toReal_mul] using hreal

114 /-- State reachability times the exploratory action floor lower-bounds visits. -/
115 theorem stateLower_mul_exploratoryActionProbabilityFloor_le_stageVisitProbability
116     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
117     (initialState : Measure State) [IsProbabilityMeasure initialState]
118     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
119     (stateLower : Fin mdp.horizon -> State -> Real)
120     (hstateLower : forall stage state,
121       stateLower stage state <=
122         (table.exploratoryPolicy explorationRate hexplorationRate).stageStateProbability
123           initialState stage state)
124     (stage : Fin mdp.horizon) (state : State) (action : Action) :
125     stateLower stage state *
126         exploratoryActionProbabilityFloor Action explorationRate <=
127       (table.exploratoryPolicy explorationRate hexplorationRate).stageVisitProbability
128         initialState stage state action := by
129   let policy := table.exploratoryPolicy explorationRate hexplorationRate
130   rw [policy.stageVisitProbability_eq_stageStateProbability_mul_action]
131   calc
132     stateLower stage state *
133           exploratoryActionProbabilityFloor Action explorationRate <=
134         policy.stageStateProbability initialState stage state *
135           exploratoryActionProbabilityFloor Action explorationRate :=
136       mul_le_mul_of_nonneg_right (hstateLower stage state)
137         (exploratoryActionProbabilityFloor_nonneg explorationRate)
138     _ <= policy.stageStateProbability initialState stage state *
139           (policy.actionKernel stage state {action}).toReal :=
140       mul_le_mul_of_nonneg_left
141         (table.exploratoryActionProbabilityFloor_le_actionKernel_real
142           explorationRate hexplorationRate stage state action)
143         (policy.stageStateProbability_nonneg initialState stage state)
