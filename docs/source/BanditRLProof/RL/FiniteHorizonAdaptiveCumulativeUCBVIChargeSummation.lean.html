148 /-- The clipped UCBVI-CH charge assigned to one state-action pair in one
149 episode, using the literal strict-prefix count of generated batch records. -/
150 noncomputable def generatedPairLocalCharge
151     (mdp : MDP State Action) (episodes : Nat) (delta : Real)
152     (trajectory : EpisodeBatchTrajectory mdp 1)
153     (state : State) (action : Action) (episode : Nat) : Real :=
154   let count := batchedPrefixCount
155     (generatedBatchAggregateVisitIncrement trajectory state action) episode
156   if count = 0 then
157     (mdp.horizon : Real)
158   else
159     min (mdp.horizon : Real)
160       (9 * (mdp.horizon : Real) *
161           logFactor (State := State) (Action := Action) mdp episodes delta /
162             Real.sqrt count +
163         66 * Fintype.card State * (mdp.horizon : Real) ^ 2 *
164           logFactor (State := State) (Action := Action) mdp episodes delta /
165             count)

222 theorem sum_successorLocalCharge_eq_sum_pairIncrement_mul_charge_of_aligned
223     {mdp : MDP State Action} {defaultState : State}

253           · simp

304 /-- The recursive charge is exactly its chronological finite sum. -/
305 theorem successorWeightedChargeFrom_eq_sum
306     (mdp : MDP State Action) (defaultState : State)
307     (episodes : Nat) (delta : Real)
308     (trajectory : EpisodeBatchTrajectory mdp 1) (n : Nat) :
309     forall (remaining : Nat) (hremaining : remaining <= mdp.horizon)
310       (state : State) (trace : StepTrace Action State remaining),
311       successorWeightedChargeFrom mdp defaultState episodes delta trajectory n
312           remaining hremaining state trace =
313         ∑ stage : Fin remaining,
314           normalizedBellmanChargeWeight mdp
315               (remainingGlobalStage mdp remaining hremaining stage) *
316             successorStageLocalCharge mdp defaultState episodes delta trajectory n
317               (remainingGlobalStage mdp remaining hremaining stage)
318               (StepTrace.stateAt state trace stage) := by
319   intro remaining
320   induction remaining with
321   | zero =>
322       intro hremaining state trace
323       simp [successorWeightedChargeFrom]
324   | succ remaining ih =>
325       intro hremaining state trace
326       rw [successorWeightedChargeFrom, Fin.sum_univ_succ]
327       have hheadStage : remainingGlobalStage mdp (remaining + 1) hremaining 0 =
328           mdp.decisionStageRemaining remaining hremaining := by
329         apply Fin.ext
330         change mdp.horizon - (remaining + 1) + 0 =
331           mdp.horizon - (remaining + 1)
332         omega
333       have hheadLocal : successorLocalCharge mdp defaultState episodes delta
334             trajectory n remaining hremaining state =
335           successorStageLocalCharge mdp defaultState episodes delta trajectory n
336             (mdp.decisionStageRemaining remaining hremaining) state := by
337         unfold successorStageLocalCharge
338         apply successorLocalCharge_eq_of_remaining_eq
339         change remaining = mdp.horizon -
340           (mdp.horizon - (remaining + 1) + 1)
341         omega
342       have hhead : normalizedBellmanChargeWeight mdp
343               (mdp.decisionStageRemaining remaining hremaining) *
344             successorLocalCharge mdp defaultState episodes delta trajectory n
345               remaining hremaining state =
346           normalizedBellmanChargeWeight mdp
347               (remainingGlobalStage mdp (remaining + 1) hremaining 0) *
348             successorStageLocalCharge mdp defaultState episodes delta trajectory n
349               (remainingGlobalStage mdp (remaining + 1) hremaining 0)
350               (StepTrace.stateAt state trace 0) := by
351         rw [hheadStage, hheadLocal]
352         rw [StepTrace.stateAt_zero_apply]
353       rw [hhead]
354       congr 1
355       have htail : remaining <= mdp.horizon := by omega
356       rw [ih htail (trace 0).2 (Fin.tail trace)]
357       apply Finset.sum_congr rfl
358       intro stage _
359       have hglobal : remainingGlobalStage mdp remaining htail stage =
360           remainingGlobalStage mdp (remaining + 1) hremaining stage.succ := by
361         exact Fin.ext (by
362           change mdp.horizon - remaining + stage.val =
363             mdp.horizon - (remaining + 1) + (stage.val + 1)
364           omega)
365       have hstate := StepTrace.stateAt_succ state (trace 0) (Fin.tail trace) stage
366       rw [hglobal]
367       exact congrArg
368         (fun currentState =>
369           normalizedBellmanChargeWeight mdp
370               (remainingGlobalStage mdp (remaining + 1) hremaining stage.succ) *
371             successorStageLocalCharge mdp defaultState episodes delta trajectory n
372               (remainingGlobalStage mdp (remaining + 1) hremaining stage.succ)
373               currentState)
374         (by simpa using hstate.symm)
