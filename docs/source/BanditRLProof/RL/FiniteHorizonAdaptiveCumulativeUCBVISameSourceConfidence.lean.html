6 /-!
7 # Same-source transition residuals for recurrent UCBVI
8 
9 This module starts the statistical producer on the actual finite episode law.
10 For a fixed state/action/next-state coordinate, the residual at a generated
11 stage is
12 
13 `1{X=x,A=a} * (1{Y=y} - P(y | x,a))`.
14 
15 The fixed-tilt exponential budget is charged only when `(x,a)` is really
16 visited.  Thus the compensator is the generated visit count, rather than the
17 episode number or an offline reachability proxy.
18 -/

36 /-- One transition-coordinate residual on a chronological trace head. -/
37 def transitionResidualHead (mdp : MDP State Action)
38     (targetState : State) (targetAction : Action) (targetNextState : State)
39     (currentState : State) (head : Action × State) : Real :=
40   if currentState = targetState ∧ head.1 = targetAction then
41     (if head.2 = targetNextState then 1 else 0) -
42       (mdp.transition (targetState, targetAction)).real {targetNextState}
43   else 0

45 /-- The realized visit charged by one transition-coordinate residual. -/
46 def transitionVisitHead (_mdp : MDP State Action)
47     (targetState : State) (targetAction : Action)
48     (currentState : State) (head : Action × State) : Real :=
49   if currentState = targetState ∧ head.1 = targetAction then 1 else 0

51 /-- Sum of one fixed transition residual over the remaining generated trace. -/
52 def transitionResidualFrom (mdp : MDP State Action)
53     (targetState : State) (targetAction : Action) (targetNextState : State) :
54     (remaining : Nat) -> State -> StepTrace Action State remaining -> Real
55   | 0, _currentState, _trace => 0
56   | remaining + 1, currentState, trace =>
57       mdp.transitionResidualHead targetState targetAction targetNextState
58           currentState (trace 0) +
59         mdp.transitionResidualFrom targetState targetAction targetNextState
60           remaining (trace 0).2 (Fin.tail trace)

69         mdp.transitionVisitFrom targetState targetAction
