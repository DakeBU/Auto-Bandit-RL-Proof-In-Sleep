5 /-!
6 # Explicit weighted rate for the heterogeneous causal sampled source
7 
8 The causal source uses a genuinely round-varying batch size.  Its regret rate
9 therefore cannot reuse the constant-window `episodes * rounds` algebra.  This
10 module keeps the exact successor mass and proves that the corresponding
11 positive-weight average of the coordinatewise scheduled rate tends to zero.
12 
13 The return proxy is also identified exactly as successor episode mass times
14 the one-episode globally centered proxy.  The final consumer keeps a constant
15 return share `1 / 2`; the cumulative model-event budget is the existing exact
16 finite sum and is not claimed to vanish with the prefix.  Consequently this is
17 a causal finite-prefix rate theorem, not convergence in probability, an
18 almost-sure result, an anytime theorem, a minimax rate, or complete UCB-VI.
19 -/

27 noncomputable def natWeightedAverage
28     (weight : Nat -> Nat) (value : Nat -> Real) (rounds : Nat) : Real :=
29   (Finset.range rounds).sum (fun t => (weight t : Real) * value t) /
30     (Finset.range rounds).sum (fun t => (weight t : Real))

32 /-- Positive natural weights preserve a zero limit under finite weighted averaging. -/
33 theorem tendsto_natWeightedAverage_zero
34     (weight : Nat -> Nat) (hweight : forall t, 0 < weight t)
35     (value : Nat -> Real) (hvalue : Tendsto value atTop (nhds 0)) :
36     Tendsto (natWeightedAverage weight value) atTop (nhds 0) := by
37   have hweightNonneg : 0 <= fun t => (weight t : Real) := by
38     intro t
39     positivity
40   have hweightOne : forall t, (1 : Real) <= (weight t : Real) := by
41     intro t
42     exact_mod_cast hweight t
43   have hsumAtTop : Tendsto
44       (fun n => (Finset.range n).sum (fun t => (weight t : Real)))
45       atTop atTop := by
46     apply tendsto_atTop_mono (fun n => ?_) tendsto_natCast_atTop_atTop
47     calc
48       (n : Real) = (Finset.range n).sum (fun _ => (1 : Real)) := by simp
49       _ <= (Finset.range n).sum (fun t => (weight t : Real)) :=
50         Finset.sum_le_sum fun t _ => hweightOne t
51   have hvalueLittle : value =o[atTop] fun _ => (1 : Real) :=
52     (isLittleO_one_iff Real).2 hvalue
53   have hweightedLittle :
54       (fun t => (weight t : Real) * value t) =o[atTop]
55         fun t => (weight t : Real) := by
56     simpa [mul_comm] using hvalueLittle.mul_isBigO
57       (isBigO_refl (fun t => (weight t : Real)) atTop)
58   have hsumLittle := hweightedLittle.sum_range hweightNonneg hsumAtTop
59   simpa [natWeightedAverage] using hsumLittle.tendsto_div_nhds_zero
