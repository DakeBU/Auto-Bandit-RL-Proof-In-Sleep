3 /-!
4 # Polynomial moment envelope for the uncapped inverse-sqrt hitting time
5 
6 This module bounds the explicit ceiling tail start by a model-dependent
7 natural coefficient times the schedule scale. The resulting fourth-power
8 checkpoint square is an explicit degree-eight polynomial in the fixed
9 threshold index.
10 
11 The result remains fixed-index. It does not establish uniform integrability,
12 L1 convergence, or an optional-stopping identity.
13 -/

43 /-- The explicit ceiling start grows at most linearly in the schedule scale. -/
44 theorem
45     inverseSqrtThresholdUnboundedHittingAfterExplicitTailStart_succ_le_polynomialScale
46     (mdp : MDP State Action) (varianceProxy : NNReal)
47     (scheduleIndex : Nat) :
48     inverseSqrtThresholdUnboundedHittingAfterExplicitTailStart
49           mdp varianceProxy scheduleIndex + 1 <=
50       inverseSqrtThresholdUnboundedHittingAfterPolynomialScaleCoefficient
51           mdp varianceProxy *
52         explicitHighProbabilityScale scheduleIndex := by
53   let C : Real :=
54     explicitPolynomialPrefixAverageRealizedBehaviorRegretRateLinearCoefficient
55       mdp varianceProxy
56   let c : Nat := Nat.ceil C
57   let s : Nat := explicitHighProbabilityScale scheduleIndex
58   have hspos : 0 < s := by
59     simpa [s] using explicitHighProbabilityScale_pos scheduleIndex
60   have hsOneReal : (1 : Real) <= (s : Real) := by
61     exact_mod_cast hspos
62   have hsqrtLe : Real.sqrt (s : Real) <= (s : Real) := by
63     rw [Real.sqrt_le_iff]
64     exact ⟨by positivity, by nlinarith⟩
65   have hCceil : C <= (c : Real) := by
66     dsimp [c]
67     exact Nat.le_ceil C
68   have hproduct :
69       C * Real.sqrt (s : Real) <= ((c * s : Nat) : Real) := by
70     calc
71       C * Real.sqrt (s : Real) <=
72           (c : Real) * Real.sqrt (s : Real) :=
73         mul_le_mul_of_nonneg_right hCceil (Real.sqrt_nonneg _)
74       _ <= (c : Real) * (s : Real) :=
75         mul_le_mul_of_nonneg_left hsqrtLe (Nat.cast_nonneg c)
76       _ = ((c * s : Nat) : Real) := by norm_num
77   have hceil : Nat.ceil (C * Real.sqrt (s : Real)) <= c * s := by
78     exact Nat.ceil_le.2 hproduct
79   have hindex : scheduleIndex <= s := by
80     simp [s, explicitHighProbabilityScale]
81   have hsMul : s <= (c + 1) * s :=
82     Nat.le_mul_of_pos_left s (Nat.succ_pos c)
83   have hstart :
84       max scheduleIndex (Nat.ceil (C * Real.sqrt (s : Real))) <=
85         (c + 1) * s := by
86     apply max_le
87     · exact hindex.trans hsMul
88     · exact hceil.trans (Nat.mul_le_mul_right s (Nat.le_succ c))
89   have hfinal :
90       max scheduleIndex (Nat.ceil (C * Real.sqrt (s : Real))) + 1 <=
91         (c + 2) * s := by
92     calc
93       max scheduleIndex (Nat.ceil (C * Real.sqrt (s : Real))) + 1 <=
94           (c + 1) * s + 1 := Nat.add_le_add_right hstart 1
95       _ <= (c + 1) * s + s :=
96         Nat.add_le_add_left (Nat.one_le_iff_ne_zero.2 hspos.ne') _
97       _ = (c + 2) * s := by ring
98   simpa
99       [inverseSqrtThresholdUnboundedHittingAfterExplicitTailStart,
100         inverseSqrtThresholdUnboundedHittingAfterPolynomialScaleCoefficient,
101         C, c, s] using hfinal
