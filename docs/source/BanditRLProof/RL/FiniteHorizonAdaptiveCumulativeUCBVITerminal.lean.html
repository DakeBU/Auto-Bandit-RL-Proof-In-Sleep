28 noncomputable def canonicalRegretBound
29     (mdp : MDP State Action) (episodes : Nat) (delta : Real) : Real :=
30   20 * (mdp.horizon : Real) * Real.sqrt mdp.horizon *
31       logFactor (State := State) (Action := Action) mdp episodes delta *
32       Real.sqrt
33         ((Fintype.card State : Real) * Fintype.card Action * episodes) +
34     250 * (mdp.horizon : Real) ^ 2 *
35       (Fintype.card State : Real) ^ 2 * Fintype.card Action *
36       logFactor (State := State) (Action := Action) mdp episodes delta ^ 2

95 /-- The complete generated terminal failure set costs at most `delta`. -/
96 theorem recurrentSource_trajectoryMeasure_canonicalFailureEvent_le
97     (mdp : MDP State Action) (initialState : Measure State)
98     [IsProbabilityMeasure initialState]
99     [StandardBorelSpace (EpisodeBatch mdp 1)]
100     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]
101     (defaultState : State) (episodes : Nat) (delta : Real)
102     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
103     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
104     (hrewardNonneg : ∀ state action, 0 <= mdp.reward state action)
105     (hrewardOne : ∀ state action, mdp.reward state action <= 1) :
106     (recurrentSource mdp initialState defaultState episodes delta).trajectoryMeasure
107         (canonicalFailureEvent (initialState := initialState)
108           defaultState episodes delta) <= ENNReal.ofReal delta := by
109   let source := recurrentSource mdp initialState defaultState episodes delta
110   let transitionBad :=
111     AdaptiveEpisodeBatchSource.simultaneousTransitionFailureEvent source episodes
112       (logFactor (State := State) (Action := Action) mdp episodes delta)
113   let innovationBad : Set (EpisodeBatchTrajectory mdp 1) :=
114     {trajectory |
115       bellmanInnovationThreshold (State := State) (Action := Action)
116             mdp episodes delta <=
117         (Finset.range episodes).sum (fun round =>
118           recurrentBellmanInnovationProcess mdp defaultState episodes delta
119             round trajectory)}
120   let alignmentBad := alignmentFailureEvent mdp defaultState episodes delta
121   have hreward : ∀ state action, |mdp.reward state action| <= 1 := by
122     intro state action
123     rw [abs_of_nonneg (hrewardNonneg state action)]
124     exact hrewardOne state action
125   have htransition : source.trajectoryMeasure transitionBad <=
126       ENNReal.ofReal (delta / 5) := by
127     exact recurrentSource_trajectoryMeasure_simultaneousTransitionFailureEvent_le_fifth
128       mdp initialState defaultState episodes delta hhorizon hepisodes hdelta
129         hdelta_le_one hreward
130   have hinnovation : source.trajectoryMeasure innovationBad <=
131       ENNReal.ofReal (delta / 5) := by
132     exact recurrentSource_trajectoryMeasure_bellmanInnovation_sum_ge_threshold_le_fifth
133       mdp initialState defaultState episodes delta hhorizon hepisodes
134         hdelta hdelta_le_one
135   have halignment : source.trajectoryMeasure alignmentBad = 0 :=
136     recurrentSource_trajectoryMeasure_alignmentFailureEvent_eq_zero
137       mdp initialState defaultState episodes delta hhorizon
138   unfold canonicalFailureEvent
139   change source.trajectoryMeasure (transitionBad ∪ innovationBad ∪ alignmentBad) <= _
140   calc
141     _ <= source.trajectoryMeasure transitionBad +
142         source.trajectoryMeasure innovationBad +
143         source.trajectoryMeasure alignmentBad := by
144       exact (measure_union_le _ _).trans
145         (add_le_add_left (measure_union_le _ _) _)
146     _ <= ENNReal.ofReal (delta / 5) + ENNReal.ofReal (delta / 5) + 0 := by
147       exact add_le_add (add_le_add htransition hinnovation) halignment.le
148     _ = ENNReal.ofReal (2 * delta / 5) := by
149       have hfifth : 0 <= delta / 5 := div_nonneg hdelta.le (by norm_num)
150       rw [add_zero, ← ENNReal.ofReal_add hfifth hfifth]
151       congr 1
152       ring
153     _ <= ENNReal.ofReal delta := by
154       apply ENNReal.ofReal_le_ofReal
155       nlinarith
