3 /-!
4 # Cross-stage cumulative transition rows for UCBVI-CH
5 
6 UCBVI-CH pools every visit of a state-action pair across stages.  The earlier
7 foundation exposed the aggregate denominator `N_k(x,a)` but retained only the
8 stage-indexed transition numerator.  This module closes that structural gap:
9 it defines `N_k(x,a,y)`, proves that its row sum is exactly the aggregate visit
10 count, proves exact generated-prefix and successor identities, and normalizes
11 the row into a probability kernel with an explicit Dirac fallback at zero.
12 
13 No concentration or regret conclusion is asserted here.
14 -/

32 /-- The UCBVI-CH numerator `N_k(x,a,y)`, pooled across every decision stage. -/
33 def aggregateTransitionCount
34     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
35     (state : State) (action : Action) (nextState : State) : Nat :=
36   ∑ stage : Fin mdp.horizon, summary stage state action nextState

41 /-- The complete pooled transition-count table is measurable. -/
42 theorem measurable_aggregateTransitionCount {mdp : MDP State Action} :
43     Measurable fun summary : TransitionCountSummary mdp =>
44       fun state action nextState =>
45         summary.aggregateTransitionCount state action nextState := by
46   refine measurable_pi_lambda _ fun state => ?_
47   refine measurable_pi_lambda _ fun action => ?_
48   refine measurable_pi_lambda _ fun nextState => ?_
49   refine Finset.measurable_sum Finset.univ fun stage _ => ?_
50   exact (measurable_pi_apply nextState).comp
51     ((measurable_pi_apply action).comp
52       ((measurable_pi_apply state).comp (measurable_pi_apply stage)))

56 /-- Pooling transition destinations gives exactly the pooled visit count. -/
57 theorem sum_aggregateTransitionCount_eq_aggregateVisitCount
58     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
59     (state : State) (action : Action) :
60     (∑ nextState : State,
61         summary.aggregateTransitionCount state action nextState) =
62       summary.aggregateVisitCount state action := by
63   classical
64   unfold aggregateTransitionCount aggregateVisitCount visitCount
65   rw [Finset.sum_comm]

67 /-- The normalized pooled empirical row, with a Dirac fallback at zero visits. -/
68 noncomputable def aggregateEmpiricalTransitionPMF
69     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
70     (defaultState : State) (state : State) (action : Action) : PMF State :=
71   if hzero : summary.aggregateVisitCount state action = 0 then
72     PMF.pure defaultState
73   else
74     PMF.ofFintype
75       (fun nextState =>
76         (summary.aggregateTransitionCount state action nextState : ENNReal) /
77           (summary.aggregateVisitCount state action : ENNReal))
78       (by
79         have sum_div_aggregateVisitCount (states : Finset State) :
80             (∑ nextState ∈ states,
81                 (summary.aggregateTransitionCount state action nextState : ENNReal) /
82                   (summary.aggregateVisitCount state action : ENNReal)) =
83               (∑ nextState ∈ states,
84                   (summary.aggregateTransitionCount state action nextState : ENNReal)) /
85                 (summary.aggregateVisitCount state action : ENNReal) := by
86           classical
87           induction states using Finset.induction_on with
88           | empty => simp
89           | @insert nextState states hnext ih =>
90               simp only [Finset.sum_insert hnext]
91               rw [ih, ENNReal.add_div]
92         have hsum :
93             (∑ nextState : State,
94                 (summary.aggregateTransitionCount state action nextState : ENNReal)) =
95               (summary.aggregateVisitCount state action : ENNReal) := by
96           exact_mod_cast
97             summary.sum_aggregateTransitionCount_eq_aggregateVisitCount state action
98         rw [sum_div_aggregateVisitCount Finset.univ]
99         rw [hsum]
100         exact ENNReal.div_self (by exact_mod_cast hzero) (ENNReal.natCast_ne_top _))
101 
102 @[simp]
103 theorem aggregateEmpiricalTransitionPMF_of_aggregateVisitCount_eq_zero
104     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
105     (defaultState : State) (state : State) (action : Action)
106     (hzero : summary.aggregateVisitCount state action = 0) :
107     summary.aggregateEmpiricalTransitionPMF defaultState state action =
108       PMF.pure defaultState := by

111 theorem aggregateEmpiricalTransitionPMF_apply_of_aggregateVisitCount_pos
112     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
113     (defaultState : State) (state : State) (action : Action) (nextState : State)
114     (hpos : 0 < summary.aggregateVisitCount state action) :
115     summary.aggregateEmpiricalTransitionPMF defaultState state action nextState =
116       (summary.aggregateTransitionCount state action nextState : ENNReal) /
117         (summary.aggregateVisitCount state action : ENNReal) := by
118   simp [aggregateEmpiricalTransitionPMF, Nat.ne_of_gt hpos, PMF.ofFintype_apply]

121 noncomputable def aggregateEmpiricalTransitionKernel
122     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
123     (defaultState : State) : ProbabilityTheory.Kernel (State × Action) State :=
124   ProbabilityTheory.Kernel.ofFunOfCountable fun pair =>
125     (summary.aggregateEmpiricalTransitionPMF

141 /-- The pooled transition numerator through one generated trajectory prefix. -/
142 def adaptiveCumulativeAggregateTransitionCountAt
143     {mdp : MDP State Action} {episodes : Nat}
144     (trajectory : EpisodeBatchTrajectory mdp episodes) (round : Nat)
145     (state : State) (action : Action) (nextState : State) : Nat :=

165 /-- The pooled numerator is the literal episode-by-stage generated count. -/
166 theorem adaptiveCumulativeAggregateTransitionCountAt_eq_sum
167     {mdp : MDP State Action} {episodes : Nat}
168     (trajectory : EpisodeBatchTrajectory mdp episodes) (round : Nat)
169     (state : State) (action : Action) (nextState : State) :
170     adaptiveCumulativeAggregateTransitionCountAt trajectory round
171         state action nextState =
172       ∑ i : Fin (round + 1), ∑ stage : Fin mdp.horizon,
173         (trajectory i).transitionCount stage state action nextState := by
174   classical
175   unfold adaptiveCumulativeAggregateTransitionCountAt
176     TransitionCountSummary.aggregateTransitionCount
177     adaptiveCumulativeEmpiricalModelStateAt
178     EpisodeBatchPrefix.cumulativeEmpiricalModelState
179     EpisodeBatchPrefix.cumulativeTransitionCountSummary
180   rw [Finset.sum_comm]
181   rfl

185 /-- Extending the prefix adds exactly one episode's pooled transition row. -/
186 theorem adaptiveCumulativeAggregateTransitionCountAt_succ
187     {mdp : MDP State Action} {episodes : Nat}
188     (trajectory : EpisodeBatchTrajectory mdp episodes) (round : Nat)
189     (state : State) (action : Action) (nextState : State) :
190     adaptiveCumulativeAggregateTransitionCountAt trajectory (round + 1)
191         state action nextState =
192       adaptiveCumulativeAggregateTransitionCountAt trajectory round
193           state action nextState +
194         ∑ stage : Fin mdp.horizon,
195           (trajectory (round + 1)).transitionCount
196             stage state action nextState := by
197   rw [adaptiveCumulativeAggregateTransitionCountAt_eq_sum,
198     adaptiveCumulativeAggregateTransitionCountAt_eq_sum]
199   rw [Fin.sum_univ_castSucc]
