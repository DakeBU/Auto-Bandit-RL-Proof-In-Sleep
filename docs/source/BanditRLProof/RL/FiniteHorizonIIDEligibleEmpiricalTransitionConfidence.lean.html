6 # Eligible empirical transition confidence for finite-horizon iid batches
7 
8 This module combines the compiled simultaneous visit/joint-count event,
9 eligible positive denominators, and generated population-law factorization.
10 For every eligible state-action-stage coordinate, the empirical singleton
11 transition mass is within `2 * countRadius / visitCount` of the true transition
12 kernel singleton mass. The bundled endpoint reuses the existing global-delta
13 event; it does not spend another failure budget or add reward confidence.
14 -/

34 /-- A single eligible coordinate inherits a strict empirical transition-mass
35 bound from the simultaneous visit and joint-transition count deviations. -/
36 theorem empiricalTransitionMass_abs_sub_transition_lt_of_not_mem_simultaneousCountBadEvent
37     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
38     (initialState : Measure State) [IsProbabilityMeasure initialState]
39     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
40     (hbatch : batch ∉
41       policy.simultaneousCountBadEvent initialState episodes delta)
42     (defaultState : State) (coordinate : VisitCoordinate mdp)
43     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
44       coordinate.expectedCount policy initialState episodes)
45     (nextState : State) :
46     |batch.empiricalTransitionMass defaultState coordinate.stage
47           coordinate.state coordinate.action nextState -
48         (mdp.transition (coordinate.state, coordinate.action)).real {nextState}| <
49       2 * simultaneousCountConfidenceRadius mdp episodes delta /
50         (coordinate.count batch : Real) := by
51   let visit : Real := coordinate.count batch
52   let joint : Real :=
53     batch.transitionCount coordinate.stage coordinate.state
54       coordinate.action nextState
55   let expectedVisit : Real :=
56     coordinate.expectedCount policy initialState episodes
57   let trueMass : Real :=
58     (mdp.transition (coordinate.state, coordinate.action)).real {nextState}
59   let radius : Real := simultaneousCountConfidenceRadius mdp episodes delta
60 
61   have hcountNat : 0 < coordinate.count batch :=
62     policy.visitCoordinate_count_pos_of_not_mem_simultaneousCountBadEvent
63       initialState batch hbatch coordinate hmargin
64   have hvisit : 0 < visit := by
65     dsimp [visit]
66     exact_mod_cast hcountNat
67   have hvisitNe :
68       batch.visitCount coordinate.stage coordinate.state coordinate.action ≠ 0 := by
69     simpa [VisitCoordinate.count] using Nat.ne_of_gt hcountNat
70   have hradius : 0 ≤ radius := by
71     exact Concentration.subGaussianSumConfidenceRadius_nonneg _ _
72   have hmassNonneg : 0 ≤ trueMass := by
73     exact MeasureTheory.measureReal_nonneg
74   have hmassLeOne : trueMass ≤ 1 := by
75     exact MeasureTheory.measureReal_le_one
76 
77   have hvisitDeviation : |visit - expectedVisit| < radius := by
78     simpa [visit, expectedVisit, radius, VisitCoordinate.count,
79       VisitCoordinate.expectedCount] using
80       policy.visitCount_abs_deviation_lt_of_not_mem_simultaneousCountBadEvent
81         initialState batch hbatch coordinate.stage coordinate.state
82           coordinate.action
83   have hjointDeviation : |joint - expectedVisit * trueMass| < radius := by
84     have hdeviation :=
85       policy.transitionCount_abs_deviation_lt_of_not_mem_simultaneousCountBadEvent
86         initialState batch hbatch coordinate.stage coordinate.state
87           coordinate.action nextState
88     rw [policy.stageTransitionJointProbability_eq_stageVisitProbability_mul_transition]
89       at hdeviation
90     simpa [joint, expectedVisit, trueMass, radius,
91       VisitCoordinate.expectedCount, Measure.real, mul_assoc] using hdeviation
92 
93   have hnumerator : |joint - visit * trueMass| < 2 * radius := by
94     calc
95       |joint - visit * trueMass| =
96           |(joint - expectedVisit * trueMass) -
97             trueMass * (visit - expectedVisit)| := by
98               congr 1
99               ring
100       _ ≤ |joint - expectedVisit * trueMass| +
101           |trueMass * (visit - expectedVisit)| := abs_sub _ _
102       _ = |joint - expectedVisit * trueMass| +
103           trueMass * |visit - expectedVisit| := by
104             rw [abs_mul, abs_of_nonneg hmassNonneg]
105       _ < radius + trueMass * radius :=
106         add_lt_add_of_lt_of_le hjointDeviation
107           (mul_le_mul_of_nonneg_left (le_of_lt hvisitDeviation) hmassNonneg)
108       _ ≤ 2 * radius := by nlinarith
109 
110   rw [batch.empiricalTransitionMass_eq_div_of_visitCount_ne_zero
111     defaultState coordinate.stage coordinate.state coordinate.action nextState
112       hvisitNe]
113   change |joint / visit - trueMass| < 2 * radius / visit
114   rw [show joint / visit - trueMass =
115       (joint - visit * trueMass) / visit by field_simp]
116   rw [abs_div, abs_of_pos hvisit]
117   exact (div_lt_div_iff_of_pos_right hvisit).2 hnumerator

120 /--
121 Route endpoint: the existing measurable simultaneous-count event has global
122 delta mass, and outside it all eligible empirical transition singleton masses
123 obey the positive-random-denominator confidence bound simultaneously.
124 -/
125 theorem iidEpisodeBatch_eligible_empiricalTransitionMass_confidence
126     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
127     (initialState : Measure State) [IsProbabilityMeasure initialState]
128     (episodes : Nat) (hepisodes : 0 < episodes)
129     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1)
130     (defaultState : State) (eligible : Finset (VisitCoordinate mdp))
131     (hmargin : ∀ coordinate ∈ eligible,
132       simultaneousCountConfidenceRadius mdp episodes delta <
133         coordinate.expectedCount policy initialState episodes) :
134     MeasurableSet
135         (policy.simultaneousCountBadEvent initialState episodes delta) ∧
136       (policy.iidEpisodeBatchMeasure initialState episodes)
137           (policy.simultaneousCountBadEvent initialState episodes delta) ≤
138         ENNReal.ofReal delta ∧
139       ∀ batch ∉ policy.simultaneousCountBadEvent initialState episodes delta,
140         ∀ coordinate ∈ eligible, ∀ nextState,
141           |batch.empiricalTransitionMass defaultState coordinate.stage
142                 coordinate.state coordinate.action nextState -
143               (mdp.transition
144                 (coordinate.state, coordinate.action)).real {nextState}| <
145             2 * simultaneousCountConfidenceRadius mdp episodes delta /
146               (coordinate.count batch : Real) := by
147   refine ⟨policy.measurableSet_simultaneousCountBadEvent
148       initialState episodes delta,
149     policy.iidEpisodeBatch_simultaneousCountBadEvent_le
150       initialState episodes hepisodes delta hdelta hdelta_le_one, ?_⟩
151   intro batch hbatch coordinate hcoordinate nextState
152   exact policy.empiricalTransitionMass_abs_sub_transition_lt_of_not_mem_simultaneousCountBadEvent
153     initialState batch hbatch defaultState coordinate
154       (hmargin coordinate hcoordinate) nextState
