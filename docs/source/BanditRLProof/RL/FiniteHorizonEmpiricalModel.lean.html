6 # Finite-batch empirical models for finite-horizon RL
7 
8 This module builds a genuine finite-state empirical transition kernel from a
9 finite family of recorded episode steps. Positive visit counts are normalized
10 into a finite PMF; zero visit counts use an explicit default-state Dirac PMF.
11 The resulting empirical reward and transition model is then connected to the
12 compiled coordinate-confidence optimistic-regret route.
13 -/

92 /--
93 A finite table with one record at each valid stage of every episode.  This raw
94 type does not enforce cross-stage state continuity or identify the records with
95 samples from the MDP trajectory law; those are downstream probabilistic laws.
96 -/
97 abbrev EpisodeBatch (mdp : MDP State Action) (episodes : Nat) :=
98   Fin episodes -> Fin mdp.horizon -> EpisodeStep State Action

102 /-- Number of batch episodes visiting a state-action pair at one stage. -/
103 def visitCount {mdp : MDP State Action} {episodes : Nat}
104     (batch : EpisodeBatch mdp episodes) (stage : Fin mdp.horizon)
105     (state : State) (action : Action) : Nat :=
106   ∑ episode, if (batch episode stage).state = state /\
107       (batch episode stage).action = action then 1 else 0
108 
109 /-- Sum of rewards recorded at a state-action pair and stage. -/
110 def rewardSum {mdp : MDP State Action} {episodes : Nat}
111     (batch : EpisodeBatch mdp episodes) (stage : Fin mdp.horizon)
112     (state : State) (action : Action) : Real :=
113   ∑ episode, if (batch episode stage).state = state /\
114       (batch episode stage).action = action then
115         (batch episode stage).reward else 0
116 
117 /-- Empirical reward mean, with the conventional zero value at zero visits. -/
118 noncomputable def empiricalReward {mdp : MDP State Action} {episodes : Nat}
119     (batch : EpisodeBatch mdp episodes) (stage : Fin mdp.horizon)
120     (state : State) (action : Action) : Real :=
121   batch.rewardSum stage state action /
122     (batch.visitCount stage state action : Real)

124 /-- Number of matching transitions to a fixed next state. -/
125 def transitionCount {mdp : MDP State Action} {episodes : Nat}
126     (batch : EpisodeBatch mdp episodes) (stage : Fin mdp.horizon)
127     (state : State) (action : Action) (nextState : State) : Nat :=
128   ∑ episode, if (batch episode stage).state = state /\
129       (batch episode stage).action = action /\
130       (batch episode stage).nextState = nextState then 1 else 0

134 /-- Next-state transition counts partition the state-action visit count. -/
135 theorem sum_transitionCount_eq_visitCount
136     {mdp : MDP State Action} {episodes : Nat}
137     (batch : EpisodeBatch mdp episodes) (stage : Fin mdp.horizon)
138     (state : State) (action : Action) :
139     ∑ nextState, batch.transitionCount stage state action nextState =
140       batch.visitCount stage state action := by
141   classical
142   unfold transitionCount visitCount
143   rw [Finset.sum_comm]
144   apply Finset.sum_congr rfl
145   intro episode _
146   by_cases hstate : (batch episode stage).state = state
147   · by_cases haction : (batch episode stage).action = action
148     · simp [hstate, haction]
149     · simp [haction]
150   · simp [hstate]

152 /-- Empirical next-state PMF with an explicit zero-visit fallback state. -/
153 noncomputable def empiricalTransitionPMF
154     {mdp : MDP State Action} {episodes : Nat}
155     (batch : EpisodeBatch mdp episodes) (defaultState : State)
156     (stage : Fin mdp.horizon) (state : State) (action : Action) : PMF State :=
157   if hzero : batch.visitCount stage state action = 0 then
158     PMF.pure defaultState
159   else
160     PMF.ofFintype
161       (fun nextState =>
162         (batch.transitionCount stage state action nextState : ENNReal) /
163           (batch.visitCount stage state action : ENNReal))
164       (by
165         have sum_div_visitCount (states : Finset State) :
166             (∑ nextState ∈ states,
167                 (batch.transitionCount stage state action nextState : ENNReal) /
168                   (batch.visitCount stage state action : ENNReal)) =
169               (∑ nextState ∈ states,
170                   (batch.transitionCount stage state action nextState : ENNReal)) /
171                 (batch.visitCount stage state action : ENNReal) := by
172           classical
173           induction states using Finset.induction_on with
174           | empty => simp
175           | @insert nextState states hnext ih =>
176               simp only [Finset.sum_insert hnext]
177               rw [ih, ENNReal.add_div]
178         calc
179           ∑ nextState,
180               (batch.transitionCount stage state action nextState : ENNReal) /
181                 (batch.visitCount stage state action : ENNReal) =
182               (∑ nextState,
183                 (batch.transitionCount stage state action nextState : ENNReal)) /
184                 (batch.visitCount stage state action : ENNReal) := by
185             simpa using sum_div_visitCount (Finset.univ : Finset State)
186           _ = (batch.visitCount stage state action : ENNReal) /
187                 (batch.visitCount stage state action : ENNReal) := by
188             congr 1
189             exact_mod_cast batch.sum_transitionCount_eq_visitCount stage state action
190           _ = 1 :=
191             ENNReal.div_self (by exact_mod_cast hzero) (ENNReal.natCast_ne_top _))

244 /-- State-action indexed empirical PMFs form a measurable finite-state kernel. -/

322 /--
323 Raw finite-batch confidence contracts.  Reward errors and singleton-frequency
324 errors are stated directly on the empirical statistics, while the envelope and
325 radius-cover fields discharge the value-dependent coordinate transport.
326 -/
327 structure Confidence {mdp : MDP State Action} {episodes : Nat}
328     (model : FiniteBatchModel mdp episodes) where
329   transitionCoordinateRadius :
330     Fin mdp.horizon -> State -> Action -> State -> Real
331   valueEnvelope : Nat -> Real
332   rewardError_le_radius : forall (stage : Fin mdp.horizon)
333     (state : State) (action : Action),
334     |model.batch.empiricalReward stage state action - mdp.reward state action| <=
335       model.rewardRadius stage state action
336   transitionFrequencyError_le_radius : forall (remaining : Nat)
337     (hremaining : remaining + 1 <= mdp.horizon)
338     (state : State) (action : Action) (nextState : State),
339     |model.batch.empiricalTransitionMass model.defaultState
340           (mdp.decisionStageRemaining remaining hremaining)
341           state action nextState -
342         (mdp.transition (state, action)).real {nextState}| <=
343       transitionCoordinateRadius
344         (mdp.decisionStageRemaining remaining hremaining)
345         state action nextState
346   upperValue_abs_le_envelope : forall (remaining : Nat)
347     (hremaining : remaining + 1 <= mdp.horizon) (state : State),
348     |model.plan.upperValueRemaining remaining (by omega) state| <=
349       valueEnvelope remaining
350   transitionRadius_cover : forall (remaining : Nat)
351     (hremaining : remaining + 1 <= mdp.horizon)
352     (state : State) (action : Action),
353     (∑ nextState,
354         transitionCoordinateRadius
355             (mdp.decisionStageRemaining remaining hremaining)
356             state action nextState *
357           valueEnvelope remaining) <=
358       model.transitionRadius
359         (mdp.decisionStageRemaining remaining hremaining) state action

361 /-- Raw empirical-statistic confidence gives coordinate model confidence. -/
362 noncomputable def Confidence.toCoordinateConfidence
363     {mdp : MDP State Action} {episodes : Nat}
364     {model : FiniteBatchModel mdp episodes}
365     (confidence : model.Confidence) : model.plan.CoordinateConfidence where
366   transitionCoordinateRadius := confidence.transitionCoordinateRadius
367   valueEnvelope := confidence.valueEnvelope
368   rewardError_le_radius stage state action := by
369     exact confidence.rewardError_le_radius stage state action
370   transitionCoordinateError_le_radius remaining hremaining state action nextState := by
371     change
372       |(model.batch.empiricalTransitionKernel model.defaultState
373             (mdp.decisionStageRemaining remaining hremaining)
374             (state, action)).real {nextState} -
375           (mdp.transition (state, action)).real {nextState}| <=
376         confidence.transitionCoordinateRadius
377           (mdp.decisionStageRemaining remaining hremaining)
378           state action nextState
379     rw [EpisodeBatch.empiricalTransitionKernel_real_singleton]
380     exact confidence.transitionFrequencyError_le_radius
381       remaining hremaining state action nextState
382   upperValue_abs_le_envelope remaining hremaining state :=
383     confidence.upperValue_abs_le_envelope remaining hremaining state
384   transitionRadius_cover remaining hremaining state action :=
385     confidence.transitionRadius_cover remaining hremaining state action

388 /--
389 Route endpoint: finite-batch reward and singleton-frequency confidence imply
390 global optimism and the compiled selected-radius expected-regret bound.

392 theorem Confidence.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
393     {mdp : MDP State Action} {episodes : Nat}
394     {model : FiniteBatchModel mdp episodes}
395     (confidence : model.Confidence) (initialState : Measure State)
396     [IsProbabilityMeasure initialState] :
397     (forall state,
398       mdp.optimalValueRemaining mdp.horizon le_rfl state <=
399         model.plan.upperValueRemaining mdp.horizon le_rfl state) /\
400       model.plan.optimisticPolicy.expectedRegret initialState <=
401         model.plan.optimisticPolicy.occupancySumRemaining
402           (fun remaining hremaining state =>
403             2 * model.plan.selectedRadiusRemaining remaining hremaining state)
