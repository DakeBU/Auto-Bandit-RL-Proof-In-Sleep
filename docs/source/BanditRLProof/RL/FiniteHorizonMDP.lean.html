3 /-!
4 # Finite-horizon MDP kernel surface
5 
6 This module fixes the dependency layer for the first finite-horizon RL leaf.
7 A finite MDP uses Mathlib Markov kernels for transitions and a measurable Real
8 reward.  The only derived objects here are the one-step continuation value and
9 Bellman action value.  Policies, trajectories, value recursion, optimality, and
10 regret remain downstream.
11 -/
12 
13 open MeasureTheory
14 
15 universe u v
16 
17 namespace BanditRLProof
18 namespace FiniteHorizonRL
19 
20 /--
21 A finite-state, finite-action, finite-horizon MDP backed by a Mathlib Markov
22 transition kernel.  The reward is allowed to depend on the current state and
23 action; stochastic rewards can be added later through a separate reward kernel.
24 -/
25 structure MDP
26     (State : Type u) (Action : Type v)
27     [MeasurableSpace State] [MeasurableSpace Action]
28     [Fintype State] [Fintype Action] where
29   horizon : Nat
30   transition : ProbabilityTheory.Kernel (State × Action) State
31   transition_isMarkov : ProbabilityTheory.IsMarkovKernel transition
32   reward : State → Action → Real
33   measurable_reward : Measurable (Function.uncurry reward)
34 
35 variable {State : Type u} {Action : Type v}
36     [MeasurableSpace State] [MeasurableSpace Action]
37     [Fintype State] [Fintype Action]
38 
39 namespace MDP
40 
41 instance instTransitionIsMarkovKernel (mdp : MDP State Action) :
42     ProbabilityTheory.IsMarkovKernel mdp.transition :=
43   mdp.transition_isMarkov
44 
45 /-- Expected continuation value after taking `action` in `state`. -/
46 noncomputable def transitionValue
47     (mdp : MDP State Action) (value : State → Real)
48     (state : State) (action : Action) : Real :=
49   ∫ nextState, value nextState ∂mdp.transition (state, action)
50 
51 /-- One-step Bellman action value `r(s,a) + E[V(S') | s,a]`. -/
52 noncomputable def bellmanQ
53     (mdp : MDP State Action) (value : State → Real)
54     (state : State) (action : Action) : Real :=
55   mdp.reward state action + mdp.transitionValue value state action
56 
57 /-- The continuation-value surface is measurable in the state-action pair. -/
58 theorem measurable_transitionValue
59     (mdp : MDP State Action) {value : State → Real}
60     (hvalue : Measurable value) :
61     Measurable (Function.uncurry (mdp.transitionValue value)) := by
62   exact hvalue.stronglyMeasurable.integral_kernel.measurable
63 
64 /-- The one-step Bellman action value is measurable in the state-action pair. -/
65 theorem measurable_bellmanQ
66     (mdp : MDP State Action) {value : State → Real}
67     (hvalue : Measurable value) :
68     Measurable (Function.uncurry (mdp.bellmanQ value)) := by
69   exact mdp.measurable_reward.add (mdp.measurable_transitionValue hvalue)
70 
71 /-- With zero continuation value, the Bellman action value is the reward. -/
72 @[simp]
73 theorem bellmanQ_zero (mdp : MDP State Action)
74     (state : State) (action : Action) :
75     mdp.bellmanQ (fun _ => 0) state action = mdp.reward state action := by
76   simp [bellmanQ, transitionValue]
77 
78 end MDP
79 end FiniteHorizonRL
80 end BanditRLProof
