3 /-!
4 # Diverging random-prefix almost-sure natural causal consistency
5 
6 This module consumes the compiled deterministic all-prefix almost-sure theorem
7 for the exact per-batch-normalized, equal-round-weighted natural average
8 realized behavior-regret process. A countable random-index measurability
9 wrapper makes every random-prefix evaluation measurable, while pathwise
10 composition transports the limit through any measurable random-prefix
11 schedule which diverges almost everywhere.
12 
13 This is random-subsequence transport on one dependent causal measure. It does
14 not use optional stopping and does not prove an anytime confidence sequence,
15 a stopping-time rate, raw single-episode regret, behavior/recommended-policy
16 equality, minimax reachability, or complete UCB-VI.
17 -/

26 /-- A countable random coordinate of a measurable process is measurable. -/
27 theorem measurable_apply_randomNat
28     {Omega : Type w} {Beta : Type x}
29     [MeasurableSpace Omega] [MeasurableSpace Beta]
30     (process : Nat -> Omega -> Beta) (randomIndex : Omega -> Nat)
31     (hprocess : forall n, Measurable (process n))
32     (hrandomIndex : Measurable randomIndex) :
33     Measurable (fun omega => process (randomIndex omega) omega) := by
34   have hjoint : Measurable (fun input : Omega × Nat => process input.2 input.1) := by
35     apply measurable_from_prod_countable_left
36     intro n
37     simpa using hprocess n
38   exact hjoint.comp (measurable_id.prodMk hrandomIndex)

40 /-- An almost-everywhere limit survives an almost-everywhere diverging random prefix. -/
41 theorem ae_tendsto_apply_randomPrefix
42     {Omega : Type w} {Beta : Type x}
43     [MeasurableSpace Omega] [TopologicalSpace Beta]
44     {mu : Measure Omega} {process : Nat -> Omega -> Beta}
45     {randomPrefix : Nat -> Omega -> Nat} {z : Beta}
46     (hprocess : ∀ᵐ omega ∂mu,
47       Tendsto (fun n => process n omega) atTop (nhds z))
48     (hrandomPrefix : ∀ᵐ omega ∂mu,
49       Tendsto (fun n => randomPrefix n omega) atTop atTop) :
50     ∀ᵐ omega ∂mu,
51       Tendsto (fun n => process (randomPrefix n omega) omega) atTop (nhds z) := by
52   filter_upwards [hprocess, hrandomPrefix] with omega hprocessOmega hrandomPrefixOmega
53   exact hprocessOmega.comp hrandomPrefixOmega

55 /-- A pointwise deterministic lower envelope forces a Nat-valued random prefix to diverge. -/
56 theorem tendsto_randomPrefix_atTop_of_nat_le
57     {Omega : Type w} (randomPrefix : Nat -> Omega -> Nat)
58     (hlower : forall n omega, n <= randomPrefix n omega) (omega : Omega) :
59     Tendsto (fun n => randomPrefix n omega) atTop atTop := by
60   refine tendsto_atTop.2 (fun threshold => ?_)
61   filter_upwards [eventually_ge_atTop threshold] with n hn
62   exact hn.trans (hlower n omega)
