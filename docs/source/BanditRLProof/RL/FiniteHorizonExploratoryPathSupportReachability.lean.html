4 # Exploratory path-support state reachability
5 
6 This module derives a policy-independent state-probability lower envelope from
7 explicit initial singleton floors and one chosen predecessor transition for
8 each successor-stage state. Uniform exploration supplies the chosen action's
9 probability. The resulting envelope applies to every exploratory policy table
10 and therefore to every behavior policy in the adaptive empirical optimistic
11 source.
12 -/

64       initialState.real {state} := by

69         Prod.fst ⁻¹' ({state} : Set State) := by

100 /--
101 One explicit predecessor path for each successor-stage state, together with
102 nonnegative singleton-mass floors along those paths.
103 -/
104 structure ExploratoryPathSupport
105     (mdp : MDP State Action) (initialState : Measure State) where
106   initialFloor : State -> Real
107   predecessorState : Fin mdp.horizon -> State -> State
108   predecessorAction : Fin mdp.horizon -> State -> Action
109   transitionFloor : Fin mdp.horizon -> State -> Real
110   initialFloor_nonneg : forall state, 0 <= initialFloor state
111   initialFloor_le : forall state, initialFloor state <= initialState.real {state}
112   transitionFloor_nonneg : forall stage, 0 < stage.val -> forall state,
113     0 <= transitionFloor stage state
114   transitionFloor_le : forall stage, 0 < stage.val -> forall state,
115     transitionFloor stage state <=
116       (mdp.transition
117         (predecessorState stage state, predecessorAction stage state)
118           {state}).toReal
119 
120 /-- Recursive state floor along the selected predecessor paths. -/
121 noncomputable def exploratoryPathStateLowerNat
122     {mdp : MDP State Action} {initialState : Measure State}
123     (support : ExploratoryPathSupport mdp initialState)
124     (explorationRate : NNReal) :
125     (stage : Nat) -> stage < mdp.horizon -> State -> Real
126   | 0, _hstage, state => support.initialFloor state
127   | stage + 1, hstage, state =>
128       exploratoryPathStateLowerNat support explorationRate stage (by omega)
129           (support.predecessorState ⟨stage + 1, hstage⟩ state) *
130         exploratoryActionProbabilityFloor Action explorationRate *
131           support.transitionFloor ⟨stage + 1, hstage⟩ state
132 
133 /-- The selected-path lower envelope indexed by valid chronological stages. -/
134 noncomputable def exploratoryPathStateLower
135     {mdp : MDP State Action} {initialState : Measure State}
136     (support : ExploratoryPathSupport mdp initialState)
137     (explorationRate : NNReal) (stage : Fin mdp.horizon) (state : State) : Real :=
138   exploratoryPathStateLowerNat support explorationRate stage.val stage.isLt state

143 /-- Every recursively constructed path-support floor is nonnegative. -/
144 theorem exploratoryPathStateLowerNat_nonneg
145     {mdp : MDP State Action} {initialState : Measure State}
146     (support : ExploratoryPathSupport mdp initialState)
147     (explorationRate : NNReal) :
148     forall (stage : Nat) (hstage : stage < mdp.horizon) (state : State),
149       0 <= exploratoryPathStateLowerNat support explorationRate stage hstage state := by
150   intro stage
151   induction stage with
152   | zero =>
153       intro hstage state
154       simpa [exploratoryPathStateLowerNat] using support.initialFloor_nonneg state
155   | succ stage ih =>
156       intro hstage state
157       rw [exploratoryPathStateLowerNat]
158       exact mul_nonneg
159         (mul_nonneg
160           (ih (by omega)
161             (support.predecessorState ⟨stage + 1, hstage⟩ state))
162           (exploratoryActionProbabilityFloor_nonneg explorationRate))
163         (support.transitionFloor_nonneg ⟨stage + 1, hstage⟩ (by simp) state)
164

179 /--
180 Every exploratory table policy dominates the same explicit path-support state
181 envelope, independently of the table's deterministic center.
182 -/
183 theorem exploratoryPathStateLower_le_stageStateProbability
184     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
185     (initialState : Measure State) [IsProbabilityMeasure initialState]
186     (support : ExploratoryPathSupport mdp initialState)
187     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
188     (stage : Fin mdp.horizon) (state : State) :
189     exploratoryPathStateLower support explorationRate stage state <=
190       (table.exploratoryPolicy explorationRate hexplorationRate).stageStateProbability
191         initialState stage state := by
192   let policy := table.exploratoryPolicy explorationRate hexplorationRate
193   have go : forall (n : Nat) (hn : n < mdp.horizon) (target : State),
194       exploratoryPathStateLowerNat support explorationRate n hn target <=
195         policy.stageStateProbability initialState ⟨n, hn⟩ target := by
196     intro n
197     induction n with
198     | zero =>
199         intro hn target
200         rw [policy.stageStateProbability_zero initialState hn target]
201         exact support.initialFloor_le target
202     | succ n ih =>
203         intro hn target
204         let successorStage : Fin mdp.horizon := ⟨n + 1, hn⟩
205         let previousStage : Fin mdp.horizon := ⟨n, by omega⟩
206         let previousState := support.predecessorState successorStage target
207         let action := support.predecessorAction successorStage target
208         have hsuccessorStage : 0 < successorStage.val := by
209           simp [successorStage]
210         have hprevious :
211             exploratoryPathStateLowerNat support explorationRate n (by omega)
212                 previousState <=
213               policy.stageStateProbability initialState previousStage
214                 previousState :=
215           ih (by omega) previousState
216         have hvisit :
217             exploratoryPathStateLowerNat support explorationRate n (by omega)
218                   previousState *
219                 exploratoryActionProbabilityFloor Action explorationRate <=
220               policy.stageVisitProbability initialState previousStage
221                 previousState action := by
222           rw [policy.stageVisitProbability_eq_stageStateProbability_mul_action]
223           calc
224             _ <= policy.stageStateProbability initialState previousStage
225                   previousState *
226                 exploratoryActionProbabilityFloor Action explorationRate :=
227               mul_le_mul_of_nonneg_right hprevious
228                 (exploratoryActionProbabilityFloor_nonneg explorationRate)
229             _ <= policy.stageStateProbability initialState previousStage
230                   previousState *
231                 (policy.actionKernel previousStage previousState {action}).toReal :=
232               mul_le_mul_of_nonneg_left
233                 (table.exploratoryActionProbabilityFloor_le_actionKernel_real
234                   explorationRate hexplorationRate previousStage previousState action)
235                 (policy.stageStateProbability_nonneg initialState previousStage
236                   previousState)
237         have hjoint :
238             exploratoryPathStateLowerNat support explorationRate n (by omega)
239                     previousState *
240                   exploratoryActionProbabilityFloor Action explorationRate *
241                 support.transitionFloor successorStage target <=
242               policy.stageTransitionJointProbability initialState previousStage
243                 previousState action target := by
244           rw [policy.stageTransitionJointProbability_eq_stageVisitProbability_mul_transition]
245           calc
246             _ <= policy.stageVisitProbability initialState previousStage
247                   previousState action *
248                 support.transitionFloor successorStage target :=
249               mul_le_mul_of_nonneg_right hvisit
250                 (support.transitionFloor_nonneg successorStage hsuccessorStage target)
251             _ <= policy.stageVisitProbability initialState previousStage
252                   previousState action *
253                 (mdp.transition (previousState, action) {target}).toReal :=
254               mul_le_mul_of_nonneg_left
255                 (support.transitionFloor_le successorStage hsuccessorStage target)
256                 (policy.stageVisitProbability_mem_Icc initialState previousStage
257                   previousState action).1
258         rw [exploratoryPathStateLowerNat]
259         exact le_trans hjoint
260           (MarkovPolicy.stageTransitionJointProbability_le_stageStateProbability_succ
261             policy
262             initialState n hn previousState action target)
263   exact go stage.val stage.isLt state
264 
265 end DeterministicMarkovPolicyTable
