3 /-!
4 # Adaptive stochastic-reward realized behavior regret
5 
6 This module changes the stochastic episode return from centering at each
7 sampled initial state's policy value to centering at the selected policy's
8 global initial-law value. The missing term is the policy-value fluctuation of
9 the sampled initial state. Complete episodes remain the independent units
10 inside one batch; no independence is assumed between the two terms inside an
11 episode or across adaptive rounds.
12 -/

29 noncomputable def initialPolicyValueDeviation

31     (initialState : Measure State)
32     (trajectory : State × RewardStepTrace Action State mdp.horizon) : Real :=

64 /-- The initial-state policy-value fluctuation is sub-Gaussian. -/
65 theorem stochasticTrajectoryMeasure_initialPolicyValueDeviation_hasSubgaussianMGF
66     (source : MeanCompatibleRewardKernel mdp)
67     (policy : MarkovPolicy mdp) (initialState : Measure State)
68     [IsProbabilityMeasure initialState]
69     (rewardBound : NNReal)
70     (hrewardBound : ∀ state action,
71       |mdp.reward state action| ≤ (rewardBound : Real)) :
72     ProbabilityTheory.HasSubgaussianMGF
73       (mdp.initialPolicyValueDeviation policy initialState)
74       (mdp.initialPolicyValueVarianceProxy rewardBound)
75       (source.stochasticTrajectoryMeasure policy initialState) := by
76   let value := policy.valueAt 0 (Nat.zero_le mdp.horizon)
77   have hbound : ∀ᵐ state ∂initialState,
78       Set.Icc
79         (-((mdp.horizon : Real) * (rewardBound : Real)))
80         ((mdp.horizon : Real) * (rewardBound : Real))
81         (value state) := by
82     exact Filter.Eventually.of_forall fun state => by
83       exact abs_le.mp (by
84         simpa [value, MarkovPolicy.valueAt] using
85           policy.valueRemaining_abs_le_of_rewardBound rewardBound hrewardBound
86             mdp.horizon le_rfl state)
87   have hbase : ProbabilityTheory.HasSubgaussianMGF
88       (fun state => value state - integral initialState value)
89       (mdp.initialPolicyValueVarianceProxy rewardBound) initialState := by
90     simpa [MDP.initialPolicyValueVarianceProxy] using
91       (Concentration.boundedCentered_hasSubgaussianMGF_of_mem_Icc_integral_eq
92         (mu := initialState) (X := value)
93         (mean := integral initialState value)
94         (policy.measurable_valueAt 0
95           (Nat.zero_le mdp.horizon)).aemeasurable hbound rfl)
96   have hbaseMap : ProbabilityTheory.HasSubgaussianMGF
97       (fun state => value state - integral initialState value)
98       (mdp.initialPolicyValueVarianceProxy rewardBound)
99       ((source.stochasticTrajectoryMeasure policy initialState).map Prod.fst) := by
100     rw [source.stochasticTrajectoryMeasure_map_fst policy initialState]
101     exact hbase
102   have hlift := ProbabilityTheory.HasSubgaussianMGF.of_map
103     (μ := source.stochasticTrajectoryMeasure policy initialState)
104     (Y := Prod.fst)
105     (X := fun state => value state - integral initialState value)
106     measurable_fst.aemeasurable hbaseMap
107   simpa [MDP.initialPolicyValueDeviation, value, Function.comp_def] using hlift

211 noncomputable def iidGlobalSampledCumulativeReturnDeviationVarianceProxy
