4 /-!
5 # Finite-horizon policy trajectory value identity
6 
7 This module constructs the genuine finite trajectory generated by a Markov
8 policy.  A trace with `n` decisions records the sampled action and resulting
9 next state at each coordinate.  The recursive trajectory kernel is then used
10 to identify expected cumulative reward with the backward policy value.
11 -/

31 /-- Prepending one action-state step is measurable for the product Pi-space. -/
32 theorem measurable_cons (n : Nat) :
33     Measurable
34       (fun p : Prod (Prod Action State) (StepTrace Action State n) =>
35         @Fin.cons n (fun _ => Prod Action State) p.1 p.2) := by
36   apply measurable_pi_lambda
37   intro i
38   refine Fin.cases measurable_fst
39     (fun j => (measurable_pi_apply j).comp measurable_snd) i

42 /-- Removing the first action-state step is measurable. -/
43 theorem measurable_tail (n : Nat) :
44     Measurable
45       (fun trace : StepTrace Action State (n + 1) => Fin.tail trace) := by
46   apply measurable_pi_lambda
47   intro i
48   exact measurable_pi_apply i.succ

52 /-- A measurable Real function on a finite type is integrable under every finite measure. -/
53 theorem integrable_of_fintype
54     {Omega : Type*} [Fintype Omega] [MeasurableSpace Omega]
55     (mu : Measure Omega) [IsFiniteMeasure mu]
56     (f : Omega -> Real) (hf : Measurable f) : Integrable f mu := by
57   apply Integrable.of_bound hf.aestronglyMeasurable
58     (Finset.sum Finset.univ (fun x => |f x|))
59   exact Filter.Eventually.of_forall (fun x => by
60     rw [Real.norm_eq_abs]
61     exact Finset.single_le_sum
62       (fun y _ => abs_nonneg (f y)) (Finset.mem_univ x))

81 /--
82 Kernel of the next `remaining` action-state steps.  Its first chronological
83 stage is `mdp.horizon - remaining`.
84 -/
85 noncomputable def trajectoryKernelRemaining
86     {mdp : MDP State Action} (policy : MarkovPolicy mdp) :
87     (remaining : Nat) -> remaining <= mdp.horizon ->
88       ProbabilityTheory.Kernel State (StepTrace Action State remaining)
89   | 0, _ =>
90       ProbabilityTheory.Kernel.deterministic
91         (fun _ => fun i => Fin.elim0 i) measurable_const
92   | remaining + 1, hremaining =>
93       let stage : Fin mdp.horizon :=
94         ⟨mdp.horizon - (remaining + 1), by omega⟩
95       let tailKernel : ProbabilityTheory.Kernel
96           (Prod State (Prod Action State))
97           (StepTrace Action State remaining) :=
98         (policy.trajectoryKernelRemaining remaining (by omega)).comap
99           (fun p : Prod State (Prod Action State) => p.2.2)
100           (show Measurable
101               (fun p : Prod State (Prod Action State) => p.2.2) from
102             measurable_snd.snd)
103       ((policy.actionStateKernel stage).compProd tailKernel).map
104         (fun p => @Fin.cons remaining (fun _ => Prod Action State) p.1 p.2)
105 
106 instance instTrajectoryKernelRemainingIsMarkovKernel
107     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
108     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
109     ProbabilityTheory.IsMarkovKernel
110       (policy.trajectoryKernelRemaining remaining hremaining) := by
111   induction remaining with
112   | zero =>
113       unfold trajectoryKernelRemaining
114       infer_instance
115   | succ remaining ih =>
116       rw [trajectoryKernelRemaining]
117       apply ProbabilityTheory.Kernel.IsMarkovKernel.map
118       exact StepTrace.measurable_cons remaining

120 end MarkovPolicy

124 /-- Total reward of a finite trace, starting from the supplied current state. -/
125 def cumulativeRewardFrom (mdp : MDP State Action) :
126     (remaining : Nat) -> State -> StepTrace Action State remaining -> Real
127   | 0, _, _ => 0
128   | remaining + 1, state, trace =>
129       mdp.reward state (trace 0).1 +
130         mdp.cumulativeRewardFrom remaining (trace 0).2 (Fin.tail trace)
131 
132 /-- The finite cumulative reward is measurable jointly in its initial state and trace. -/
133 theorem measurable_cumulativeRewardFrom
134     (mdp : MDP State Action) (remaining : Nat) :
135     Measurable
136       (fun p : Prod State (StepTrace Action State remaining) =>
137         mdp.cumulativeRewardFrom remaining p.1 p.2) := by
138   induction remaining with
139   | zero =>
140       simp [cumulativeRewardFrom]
141   | succ remaining ih =>
142       rw [show remaining + 1 = Nat.succ remaining by rfl]
143       simp only [cumulativeRewardFrom]
144       apply Measurable.add
145       · exact mdp.measurable_reward.comp
146           (measurable_fst.prodMk
147             ((measurable_pi_apply (0 : Fin (remaining + 1))).comp measurable_snd).fst)
148       · exact ih.comp
149           (((measurable_pi_apply (0 : Fin (remaining + 1))).comp measurable_snd).snd.prodMk
150             ((StepTrace.measurable_tail remaining).comp measurable_snd))
151 
152 end MDP

156 /-- Every statewise finite-trace cumulative reward is integrable under its trajectory law. -/
157 theorem integrable_cumulativeRewardFrom_trajectoryKernelRemaining
158     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
159     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
160     (state : State) :
161     Integrable (mdp.cumulativeRewardFrom remaining state)
162       (policy.trajectoryKernelRemaining remaining hremaining state) := by
163   apply integrable_of_fintype
164   exact (mdp.measurable_cumulativeRewardFrom remaining).comp
165     (measurable_const.prodMk measurable_id)

167 /--
168 Statewise policy-evaluation identity: integrating the finite generated return
169 over the remaining trajectory gives the backward policy value.
170 -/
171 theorem integral_cumulativeRewardFrom_trajectoryKernelRemaining_eq_valueRemaining
172     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
173     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
174     (state : State) :
175     (∫ trace, mdp.cumulativeRewardFrom remaining state trace
176         ∂policy.trajectoryKernelRemaining remaining hremaining state) =
177       policy.valueRemaining remaining hremaining state := by
178   induction remaining generalizing state with
179   | zero =>
180       rw [trajectoryKernelRemaining, ProbabilityTheory.Kernel.deterministic_apply]
181       rw [integral_dirac'
182         (fun trace => mdp.cumulativeRewardFrom 0 state trace)
183         (fun i => Fin.elim0 i)]
184       · rfl
185       · exact ((mdp.measurable_cumulativeRewardFrom 0).comp
186           ((measurable_const : Measurable
187               (fun _ : StepTrace Action State 0 => state)).prodMk
188             measurable_id)).stronglyMeasurable
189   | succ remaining ih =>
190       let stage : Fin mdp.horizon :=
191         ⟨mdp.horizon - (remaining + 1), by omega⟩
192       let tailKernel : ProbabilityTheory.Kernel
193           (Prod State (Prod Action State))
194           (StepTrace Action State remaining) :=
195         (policy.trajectoryKernelRemaining remaining (by omega)).comap
196           (fun p : Prod State (Prod Action State) => p.2.2)
197           (show Measurable
198               (fun p : Prod State (Prod Action State) => p.2.2) from
199             measurable_snd.snd)
200       let consStep :=
201         fun p : Prod (Prod Action State) (StepTrace Action State remaining) =>
202           @Fin.cons remaining (fun _ => Prod Action State) p.1 p.2
203       have hcons : Measurable consStep := StepTrace.measurable_cons remaining
204       have hreturn : Measurable
205           (mdp.cumulativeRewardFrom (remaining + 1) state) :=
206         (mdp.measurable_cumulativeRewardFrom (remaining + 1)).comp
207           (measurable_const.prodMk measurable_id)
208       have hpair : Integrable
209           (fun p => mdp.cumulativeRewardFrom (remaining + 1) state (consStep p))
210           (((policy.actionStateKernel stage).compProd tailKernel) state) := by
211         apply integrable_of_fintype
212         exact hreturn.comp hcons
213       rw [trajectoryKernelRemaining]
214       rw [ProbabilityTheory.Kernel.map_apply _ hcons]
215       rw [integral_map hcons.aemeasurable hreturn.aestronglyMeasurable]
216       rw [ProbabilityTheory.integral_compProd hpair]
217       have htailIntegral :
218           (fun head : Prod Action State =>
219               ∫ tail, mdp.cumulativeRewardFrom (remaining + 1) state
220                   (consStep (head, tail)) ∂tailKernel (state, head)) =
221             (fun head => mdp.reward state head.1 +
222               policy.valueRemaining remaining (by omega) head.2) := by
223         funext head
224         rw [ProbabilityTheory.Kernel.comap_apply]
225         change (∫ tail,
226             mdp.reward state head.1 +
227               mdp.cumulativeRewardFrom remaining head.2 tail
228             ∂policy.trajectoryKernelRemaining remaining (by omega) head.2) = _
229         rw [integral_add (integrable_const _)
230           (policy.integrable_cumulativeRewardFrom_trajectoryKernelRemaining
231             remaining (by omega) head.2)]
232         rw [ih (by omega) head.2]
233         simp
234       rw [htailIntegral]
235       have hactionPair : Integrable
236           (fun head : Prod Action State =>
237             mdp.reward state head.1 +
238               policy.valueRemaining remaining (by omega) head.2)
239           (policy.actionStateKernel stage state) := by
240         apply integrable_of_fintype
241         exact (mdp.measurable_reward.comp
242             (measurable_const.prodMk measurable_fst)).add
243           ((policy.measurable_valueRemaining remaining (by omega)).comp measurable_snd)
244       unfold actionStateKernel at hactionPair ⊢
245       rw [ProbabilityTheory.integral_compProd hactionPair]
246       rw [valueRemaining]
247       change (∫ action, ∫ nextState,
248           mdp.reward state action +
249               policy.valueRemaining remaining (by omega) nextState
250             ∂mdp.transition (state, action)
251           ∂policy.actionKernel stage state) =
252         policy.bellman stage
253           (policy.valueRemaining remaining (by omega)) state
254       unfold bellman
255       apply integral_congr_ae
256       filter_upwards [] with action
257       rw [integral_add (integrable_const _)
258         (integrable_of_fintype _ _
259           ((policy.measurable_valueRemaining remaining (by omega))))]
260       simp [MDP.bellmanQ, MDP.transitionValue]
