5 /-!
6 # Explicit-schedule almost-sure consistency for natural average realized behavior regret
7 
8 This module keeps the exact process which divides every successor batch by its
9 own positive episode count and then weights rounds equally. Along the
10 deterministic fourth-power prefixes `(n + 1)^4`, its compiled all-prefix L1
11 envelope is dominated by a sum of shifted exponent-three and exponent-two
12 p-series. Markov's inequality and the first Borel-Cantelli lemma then give
13 almost-everywhere convergence on this subsequence.
14 
15 This is not the older total-episode-mass-weighted process and is not an
16 all-prefix, anytime, or stopping-time almost-sure result.
17 -/

58 /-- The scheduled logarithmic average is dominated by a shifted exponent-three term. -/
59 theorem selfConsistentScheduledNaturalCausalLogarithmicAverageIntegratedBehaviorExpectedRegretRate_explicitRounds_le_summableEnvelope
60     (mdp : MDP State Action) (n : Nat) :
61     selfConsistentScheduledNaturalCausalLogarithmicAverageIntegratedBehaviorExpectedRegretRate
62         mdp (explicitHighProbabilityRounds n) <=
63       explicitPolynomialPrefixAverageBehaviorRegretL1SummableEnvelope mdp n := by
64   let s : Real := explicitHighProbabilityScale n
65   let C : Real :=
66     selfConsistentScheduledNaturalCausalLogarithmicRateCoefficient mdp
67   have hs : 0 < s := by
68     dsimp [s]
69     exact_mod_cast explicitHighProbabilityScale_pos n
70   have hC : 0 <= C :=
71     selfConsistentScheduledNaturalCausalLogarithmicRateCoefficient_nonneg mdp
72   have hrounds : (explicitHighProbabilityRounds n : Real) = s ^ 4 := by
73     simp [explicitHighProbabilityRounds, s]
74   have hlog : Real.log s <= s - 1 :=
75     Real.log_le_sub_one_of_pos hs
76   have hlogBound : 1 + Real.log (s ^ 4) <= 4 * s := by
77     rw [Real.log_pow]
78     norm_num
79     linarith
80   have hsPow : 0 < s ^ 4 := pow_pos hs 4
81   unfold selfConsistentScheduledNaturalCausalLogarithmicAverageIntegratedBehaviorExpectedRegretRate
82     selfConsistentScheduledNaturalCausalLogarithmicCumulativeIntegratedBehaviorExpectedRegretRate

85       (explicitHighProbabilityRounds n : Real) <= 4 * C / s ^ 3
86   rw [hrounds]
87   calc
88     C * (1 + Real.log (s ^ 4)) / s ^ 4 <=
89         C * (4 * s) / s ^ 4 :=
90       div_le_div_of_nonneg_right
91         (mul_le_mul_of_nonneg_left hlogBound hC) hsPow.le
92     _ = 4 * C / s ^ 3 := by
93       field_simp [ne_of_gt hs]
