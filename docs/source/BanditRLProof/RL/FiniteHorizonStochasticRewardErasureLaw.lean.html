5 /-!
6 # Finite-horizon stochastic reward erasure laws
7 
8 This module discards sampled Real rewards from generated stochastic-reward
9 trajectories while retaining every action and next state.  The resulting law is
10 exactly the ordinary finite-horizon policy trajectory law.  The equality is
11 then lifted through the initial-state mixture, a finite iid episode family, and
12 the existing known-reward `EpisodeBatch` conversion.
13 
14 The projected batch deliberately reinstates the deterministic mean reward
15 `mdp.reward`.  This is a law transport for transition-learning algorithms with
16 known mean rewards, not a stochastic reward-estimation theorem.
17 -/

33 /-- Discard sampled rewards while retaining every action and next state. -/
34 def eraseRewards (remaining : Nat) :
35     RewardStepTrace Action State remaining -> StepTrace Action State remaining :=
36   fun trace coordinate => ((trace coordinate).1, (trace coordinate).2.2)

39 /-- Coordinatewise reward erasure is measurable on the finite Pi-space. -/
40 theorem measurable_eraseRewards (remaining : Nat) :
41     Measurable (eraseRewards (Action := Action) (State := State) remaining) := by
42   apply measurable_pi_lambda
43   intro coordinate
44   exact (measurable_fst.prodMk measurable_snd.snd).comp
45     (measurable_pi_apply coordinate)

49 /-- Reward erasure commutes with prepending one generated coordinate. -/
50 @[simp]
51 theorem eraseRewards_cons (remaining : Nat)
52     (head : Action × (Real × State))
53     (tail : RewardStepTrace Action State remaining) :
54     eraseRewards (Action := Action) (State := State) (remaining + 1)
55         (@Fin.cons remaining (fun _ => Action × (Real × State)) head tail) =
56       @Fin.cons remaining (fun _ => Action × State)
57         (head.1, head.2.2)
58         (eraseRewards (Action := Action) (State := State) remaining tail) := by
59   funext coordinate
60   refine Fin.cases ?_ (fun tailCoordinate => ?_) coordinate <;> simp [eraseRewards]

66 /--
67 Map both outputs of a composition-product kernel when the mapped head kernel and
68 every mapped tail fiber agree with prescribed target kernels.
69 -/
70 theorem compProd_map_prodMap_of_map_eq
71     {Alpha Beta Beta' Gamma Gamma' : Type*}
72     [MeasurableSpace Alpha] [MeasurableSpace Beta] [MeasurableSpace Beta']
73     [MeasurableSpace Gamma] [MeasurableSpace Gamma']
74     (kappa : ProbabilityTheory.Kernel Alpha Beta)
75       [ProbabilityTheory.IsMarkovKernel kappa]
76     (eta : ProbabilityTheory.Kernel (Alpha × Beta) Gamma)
77       [ProbabilityTheory.IsMarkovKernel eta]
78     (kappa' : ProbabilityTheory.Kernel Alpha Beta')
79       [ProbabilityTheory.IsMarkovKernel kappa']
80     (eta' : ProbabilityTheory.Kernel (Alpha × Beta') Gamma')
81       [ProbabilityTheory.IsMarkovKernel eta']
82     (f : Beta -> Beta') (hf : Measurable f)
83     (g : Gamma -> Gamma') (hg : Measurable g)
84     (hkappa : kappa.map f = kappa')
85     (heta : forall alpha beta,
86       (eta (alpha, beta)).map g = eta' (alpha, f beta)) :
87     (kappa.compProd eta).map (Prod.map f g) =
88       kappa'.compProd eta' := by
89   apply ProbabilityTheory.Kernel.ext
90   intro alpha
91   apply Measure.ext_prod
92   intro betaSet gammaSet hbetaSet hgammaSet
93   have hprodMap : Measurable (Prod.map f g) := hf.prodMap hg
94   rw [ProbabilityTheory.Kernel.map_apply _ hprodMap]
95   rw [Measure.map_apply hprodMap (hbetaSet.prod hgammaSet)]
96   have hpreimage :
97       Prod.map f g ⁻¹' (betaSet ×ˢ gammaSet) =
98         (f ⁻¹' betaSet) ×ˢ (g ⁻¹' gammaSet) := by
99     ext pair
100     simp
101   rw [hpreimage, ProbabilityTheory.Kernel.compProd_apply_prod (hbetaSet.preimage hf)
102     (hgammaSet.preimage hg)]
103   rw [ProbabilityTheory.Kernel.compProd_apply_prod hbetaSet hgammaSet]
104   have htail (beta : Beta) :
105       eta (alpha, beta) (g ⁻¹' gammaSet) =
106         eta' (alpha, f beta) gammaSet := by
107     have h := congrArg (fun mu : Measure Gamma' => mu gammaSet) (heta alpha beta)
108     simpa [Measure.map_apply hg hgammaSet] using h
109   simp_rw [htail]
110   let q : Beta' -> ENNReal := fun beta => eta' (alpha, beta) gammaSet
111   have hq : Measurable q :=
112     (eta'.measurable_coe hgammaSet).comp
113       (measurable_const.prodMk measurable_id)
114   change
115     (∫⁻ beta in f ⁻¹' betaSet, q (f beta) ∂kappa alpha) =
116       ∫⁻ beta in betaSet, q beta ∂kappa' alpha
117   rw [← MeasureTheory.setLIntegral_map hbetaSet hq hf]
118   rw [← ProbabilityTheory.Kernel.map_apply kappa hf alpha, hkappa]
119 
120 end ProbabilityTheory

126 /--
127 Dropping all sampled rewards from a generated finite stochastic trajectory
128 recovers the ordinary action/next-state trajectory kernel exactly.
129 -/
130 theorem stochasticTrajectoryKernelRemaining_map_eraseRewards
131     (source : MeanCompatibleRewardKernel mdp)
132     (policy : MarkovPolicy mdp)
133     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
134     (source.stochasticTrajectoryKernelRemaining policy remaining hremaining).map
135         (RewardStepTrace.eraseRewards
136           (Action := Action) (State := State) remaining) =
137       policy.trajectoryKernelRemaining remaining hremaining := by
138   induction remaining with
139   | zero =>
140       rw [stochasticTrajectoryKernelRemaining, MarkovPolicy.trajectoryKernelRemaining]
141       apply ProbabilityTheory.Kernel.ext
142       intro state
143       rw [ProbabilityTheory.Kernel.map_apply _
144         (RewardStepTrace.measurable_eraseRewards 0)]
145       rw [ProbabilityTheory.Kernel.deterministic_apply,
146         ProbabilityTheory.Kernel.deterministic_apply]
147       rw [Measure.map_dirac]
148       congr 1
149       funext coordinate
150       exact Fin.elim0 coordinate
151   | succ remaining ih =>
152       let dropReward : Action × (Real × State) -> Action × State :=
153         fun head => (head.1, head.2.2)
154       let eraseTail : RewardStepTrace Action State remaining ->
155           StepTrace Action State remaining :=
156         RewardStepTrace.eraseRewards remaining
157       let stochasticTail : ProbabilityTheory.Kernel
158           (State × (Action × (Real × State)))
159           (RewardStepTrace Action State remaining) :=
160         (source.stochasticTrajectoryKernelRemaining policy remaining (by omega)).comap
161           (fun pair : State × (Action × (Real × State)) => pair.2.2.2)
162           measurable_snd.snd.snd
163       let deterministicTail : ProbabilityTheory.Kernel
164           (State × (Action × State)) (StepTrace Action State remaining) :=
165         (policy.trajectoryKernelRemaining remaining (by omega)).comap
166           (fun pair : State × (Action × State) => pair.2.2)
167           measurable_snd.snd
168       let stochasticCons :
169           (Action × (Real × State)) × RewardStepTrace Action State remaining ->
170             RewardStepTrace Action State (remaining + 1) :=
171         fun pair => @Fin.cons remaining
172           (fun _ => Action × (Real × State)) pair.1 pair.2
173       let deterministicCons :
174           (Action × State) × StepTrace Action State remaining ->
175             StepTrace Action State (remaining + 1) :=
176         fun pair => @Fin.cons remaining (fun _ => Action × State) pair.1 pair.2
177       have hdropReward : Measurable dropReward :=
178         measurable_fst.prodMk measurable_snd.snd
179       have heraseTail : Measurable eraseTail :=
180         RewardStepTrace.measurable_eraseRewards remaining
181       have hstochasticCons : Measurable stochasticCons :=
182         RewardStepTrace.measurable_cons remaining
183       have hdeterministicCons : Measurable deterministicCons :=
184         StepTrace.measurable_cons remaining
185       have htail : forall state head,
186           (stochasticTail (state, head)).map eraseTail =
187             deterministicTail (state, dropReward head) := by
188         intro state head
189         simp only [stochasticTail, deterministicTail,
190           ProbabilityTheory.Kernel.comap_apply]
191         change
192           Measure.map eraseTail
193               ((source.stochasticTrajectoryKernelRemaining policy remaining
194                 (by omega)) head.2.2) =
195             (policy.trajectoryKernelRemaining remaining (by omega)) head.2.2
196         rw [← ProbabilityTheory.Kernel.map_apply _ heraseTail]
197         exact congrArg
198           (fun kernel : ProbabilityTheory.Kernel State
199               (StepTrace Action State remaining) => kernel head.2.2)
200           (ih (by omega))
201       have hpairMap :
202           ((source.actionRewardStateKernel policy
203               ⟨mdp.horizon - (remaining + 1), by omega⟩).compProd
204             stochasticTail).map (Prod.map dropReward eraseTail) =
205           (policy.actionStateKernel
206               ⟨mdp.horizon - (remaining + 1), by omega⟩).compProd
207             deterministicTail := by
208         apply ProbabilityTheory.compProd_map_prodMap_of_map_eq
209           _ _ _ _ dropReward hdropReward eraseTail heraseTail
210         · exact source.actionRewardStateKernel_map_dropReward policy _
211         · exact htail
212       rw [stochasticTrajectoryKernelRemaining,
213         MarkovPolicy.trajectoryKernelRemaining]
214       change
215         (((source.actionRewardStateKernel policy
216             ⟨mdp.horizon - (remaining + 1), by omega⟩).compProd
217           stochasticTail).map stochasticCons).map
218             (RewardStepTrace.eraseRewards (remaining + 1)) =
219           ((policy.actionStateKernel
220             ⟨mdp.horizon - (remaining + 1), by omega⟩).compProd
221           deterministicTail).map deterministicCons
222       rw [← ProbabilityTheory.Kernel.map_comp_right _ hstochasticCons
223         (RewardStepTrace.measurable_eraseRewards (remaining + 1))]
224       have hcommute :
225           RewardStepTrace.eraseRewards (remaining + 1) ∘ stochasticCons =
226             deterministicCons ∘ Prod.map dropReward eraseTail := by
227         funext pair
228         exact RewardStepTrace.eraseRewards_cons remaining pair.1 pair.2
229       rw [hcommute]
230       rw [ProbabilityTheory.Kernel.map_comp_right _
231         (hdropReward.prodMap heraseTail) hdeterministicCons]
232       rw [hpairMap]

234 /-- Erase sampled rewards from a full stochastic trajectory, retaining its initial state. -/
235 def eraseTrajectory
236     (trajectory : State × RewardStepTrace Action State mdp.horizon) :
237     State × StepTrace Action State mdp.horizon :=
238   (trajectory.1, RewardStepTrace.eraseRewards mdp.horizon trajectory.2)
239 
240 /-- Full-trajectory reward erasure is measurable. -/
241 theorem measurable_eraseTrajectory :
242     Measurable (eraseTrajectory (mdp := mdp) (Action := Action)) :=
243   measurable_fst.prodMk
244     ((RewardStepTrace.measurable_eraseRewards mdp.horizon).comp measurable_snd)
245 
246 /-- The full stochastic trajectory law maps exactly to the ordinary trajectory law. -/
247 theorem stochasticTrajectoryMeasure_map_eraseTrajectory
248     (source : MeanCompatibleRewardKernel mdp)
249     (policy : MarkovPolicy mdp) (initialState : Measure State)
250     [IsProbabilityMeasure initialState] :
251     (source.stochasticTrajectoryMeasure policy initialState).map
252         (eraseTrajectory (mdp := mdp) (Action := Action)) =
253       policy.trajectoryMeasure initialState := by
254   unfold stochasticTrajectoryMeasure MarkovPolicy.trajectoryMeasure
255   have hkernel := source.stochasticTrajectoryKernelRemaining_map_eraseRewards
256     policy mdp.horizon le_rfl
257   rw [← hkernel]
258   simpa only [eraseTrajectory] using
259     (Measure.compProd_map
260       (μ := initialState)
261       (κ := source.stochasticTrajectoryKernelRemaining
262         policy mdp.horizon le_rfl)
263       (f := RewardStepTrace.eraseRewards mdp.horizon)
264       (RewardStepTrace.measurable_eraseRewards mdp.horizon)).symm

266 /-- Erase sampled rewards coordinatewise from a finite iid trajectory family. -/
267 def eraseTrajectoryFamily (episodes : Nat)
268     (trajectories : Fin episodes ->
269       State × RewardStepTrace Action State mdp.horizon) :
270     Fin episodes -> State × StepTrace Action State mdp.horizon :=
271   fun episode => eraseTrajectory (mdp := mdp) (trajectories episode)
272 
273 /-- Finite-family reward erasure is measurable. -/
274 theorem measurable_eraseTrajectoryFamily (episodes : Nat) :
275     Measurable (eraseTrajectoryFamily
276       (mdp := mdp) (Action := Action) episodes) := by
277   apply measurable_pi_lambda
278   intro episode
279   exact measurable_eraseTrajectory.comp (measurable_pi_apply episode)

281 /-- Finite iid stochastic trajectories map to the deterministic iid family law. -/
282 theorem iidStochasticTrajectoryFamilyMeasure_map_eraseTrajectoryFamily
283     (source : MeanCompatibleRewardKernel mdp)
284     (policy : MarkovPolicy mdp) (initialState : Measure State)
285     [IsProbabilityMeasure initialState] (episodes : Nat) :
286     (source.iidStochasticTrajectoryFamilyMeasure
287         policy initialState episodes).map
288         (eraseTrajectoryFamily (mdp := mdp) (Action := Action) episodes) =
289       policy.iidTrajectoryFamilyMeasure initialState episodes := by
290   unfold iidStochasticTrajectoryFamilyMeasure
291     MarkovPolicy.iidTrajectoryFamilyMeasure eraseTrajectoryFamily
292   rw [Measure.pi_map_pi (fun _ => measurable_eraseTrajectory.aemeasurable)]
293   congr 1
294   funext episode
295   exact source.stochasticTrajectoryMeasure_map_eraseTrajectory
296     policy initialState

308 /--
309 Convert a stochastic trajectory family to the existing empirical batch after
310 erasing rewards.  Batch rewards are the known means `mdp.reward`.
311 -/
312 def knownRewardEpisodeBatchOfStochasticTrajectories (episodes : Nat)
313     (trajectories : Fin episodes ->
314       State × RewardStepTrace Action State mdp.horizon) :
315     EpisodeBatch mdp episodes :=
316   mdp.episodeBatchOfTrajectories episodes
317     (eraseTrajectoryFamily (mdp := mdp) episodes trajectories)
