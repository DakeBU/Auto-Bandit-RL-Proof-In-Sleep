30 /-- L1 convergence to zero gives probability-theory uniform integrability.
31 
32 The `UnifIntegrable` component is Mathlib's Lp convergence theorem. The
33 additional uniform L1 bound is obtained from boundedness of the convergent
34 sequence in `Lp` and transported back through `MemLp.toLp`. -/
35 theorem uniformIntegrable_one_of_memLp_and_tendsto_eLpNorm_sub_zero
36     {Omega : Type w} [MeasurableSpace Omega] {mu : Measure Omega}
37     {f : Nat -> Omega -> Real}
38     (hf : forall n, MemLp (f n) 1 mu)
39     (hfg : Tendsto
40       (fun n => eLpNorm (f n - (fun _ => 0)) 1 mu)
41       atTop (nhds 0)) :
42     UniformIntegrable f 1 mu := by
43   constructor
44   case left => exact fun n => (hf n).aestronglyMeasurable
45   case right =>
46     constructor
47     case left =>
48       exact unifIntegrable_of_tendsto_Lp le_rfl ENNReal.one_ne_top hf
49         MemLp.zero' hfg
50     case right =>
51       let F : Nat -> Lp Real 1 mu := fun n => (hf n).toLp (f n)
52       have hF : Tendsto F atTop (nhds 0) :=
53         (Lp.tendsto_Lp_iff_tendsto_eLpNorm'' f hf
54           (fun _ => (0 : Real)) MemLp.zero').2 hfg
55       have hnn : Tendsto (fun n => nnnorm (F n)) atTop (nhds 0) := by
56         simpa only [Function.comp_apply, nnnorm_zero] using
57           (continuous_nnnorm.tendsto 0).comp hF
58       have hbdd :=
59         Metric.isBounded_range_of_tendsto (fun n => nnnorm (F n)) hnn
60       let C : NNReal := hbdd.bddAbove.choose
61       have hC := hbdd.bddAbove.choose_spec
62       refine Exists.intro C ?_
63       intro n
64       calc
65         eLpNorm (f n) 1 mu = enorm ((hf n).toLp (f n)) :=
66           (Lp.enorm_toLp (hf n)).symm
67         _ = (nnnorm (F n) : ENNReal) := rfl
68         _ <= (C : ENNReal) :=
69           ENNReal.coe_le_coe.2 (hC (Set.mem_range_self n))
