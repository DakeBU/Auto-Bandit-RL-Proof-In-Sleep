33 /-- Hoeffding proxy for the mean Bellman return with `remaining` decisions. -/
34 noncomputable def meanBellmanInnovationStepVarianceProxy
35     (rewardBound : NNReal) (remaining : Nat) : NNReal :=
36   (((remaining : Nat) : NNReal) * rewardBound) ^ 2
37 
38 /-- Sum of the stage-dependent mean Bellman innovation proxies. -/
39 noncomputable def meanBellmanInnovationVarianceProxy
40     (rewardBound : NNReal) (remaining : Nat) : NNReal :=
41   ∑ k ∈ Finset.range remaining,
42     meanBellmanInnovationStepVarianceProxy rewardBound (k + 1)

44 @[simp]
45 theorem meanBellmanInnovationVarianceProxy_zero (rewardBound : NNReal) :
46     meanBellmanInnovationVarianceProxy rewardBound 0 = 0 := by
47   simp [meanBellmanInnovationVarianceProxy]

49 theorem meanBellmanInnovationVarianceProxy_succ
50     (rewardBound : NNReal) (remaining : Nat) :
51     meanBellmanInnovationVarianceProxy rewardBound (remaining + 1) =
52       meanBellmanInnovationStepVarianceProxy rewardBound (remaining + 1) +
53         meanBellmanInnovationVarianceProxy rewardBound remaining := by
54   simp [meanBellmanInnovationVarianceProxy, Finset.sum_range_succ, add_comm]

56 theorem meanBellmanInnovationVarianceProxy_pos
57     {rewardBound : NNReal} {remaining : Nat}
58     (hrewardBound : 0 < rewardBound) (hremaining : 0 < remaining) :
59     0 < meanBellmanInnovationVarianceProxy rewardBound remaining := by
60   obtain ⟨remaining, rfl⟩ := Nat.exists_eq_succ_of_ne_zero (Nat.ne_of_gt hremaining)
61   rw [meanBellmanInnovationVarianceProxy_succ]

63     simp [meanBellmanInnovationStepVarianceProxy]
64     positivity
65   exact add_pos_of_pos_of_nonneg hstep (zero_le _)

69 /-- Every bounded-mean-reward policy value lies in its remaining reward envelope. -/
70 theorem valueRemaining_abs_le_of_rewardBound
71     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
72     (rewardBound : NNReal)
73     (hrewardBound : ∀ state action,
74       |mdp.reward state action| ≤ (rewardBound : Real))
75     (remaining : Nat) (hremaining : remaining ≤ mdp.horizon) (state : State) :
76     |policy.valueRemaining remaining hremaining state| ≤
77       (remaining : Real) * (rewardBound : Real) := by
78   induction remaining generalizing state with
79   | zero =>
80       simp [valueRemaining]
81   | succ remaining ih =>
82       rw [valueRemaining]
83       rw [← Real.norm_eq_abs]
84       have hnorm := norm_integral_le_of_norm_le_const
85         (μ := policy.actionKernel
86           ⟨mdp.horizon - (remaining + 1), by omega⟩ state)
87         (C := ((remaining + 1 : Nat) : Real) * (rewardBound : Real))
88         (Filter.Eventually.of_forall fun action => by
89           rw [Real.norm_eq_abs]
90           have htransition :
91               |mdp.transitionValue
92                   (policy.valueRemaining remaining (by omega)) state action| ≤
93                 (remaining : Real) * (rewardBound : Real) := by
94             rw [← Real.norm_eq_abs]
95             have hinner := norm_integral_le_of_norm_le_const
96               (μ := mdp.transition (state, action))
97               (C := (remaining : Real) * (rewardBound : Real))
98               (Filter.Eventually.of_forall fun nextState => by
99                 rw [Real.norm_eq_abs]
100                 exact ih (by omega) nextState)
101             simpa [MDP.transitionValue] using hinner
102           calc
103             |mdp.bellmanQ
104                 (policy.valueRemaining remaining (by omega)) state action| ≤
105                 |mdp.reward state action| +
106                   |mdp.transitionValue
107                     (policy.valueRemaining remaining (by omega)) state action| :=
108               abs_add_le _ _
109             _ ≤ (rewardBound : Real) +
110                 (remaining : Real) * (rewardBound : Real) :=
111               add_le_add (hrewardBound state action) htransition
112             _ = ((remaining + 1 : Nat) : Real) * (rewardBound : Real) := by
113               push_cast
114               ring)
115       simpa [MarkovPolicy.bellman] using hnorm

121 /-- Sum of policy-action and transition innovations in the mean Bellman return. -/
122 noncomputable def sampledCumulativeMeanBellmanInnovationFrom
123     (mdp : MDP State Action) (policy : MarkovPolicy mdp) :
124     (remaining : Nat) → remaining ≤ mdp.horizon → State →
125       RewardStepTrace Action State remaining → Real
126   | 0, _, _, _ => 0
127   | remaining + 1, hremaining, state, trace =>
128       (mdp.reward state (trace 0).1 +
129           policy.valueRemaining remaining (by omega) (trace 0).2.2 -
130             policy.valueRemaining (remaining + 1) hremaining state) +
131         sampledCumulativeMeanBellmanInnovationFrom mdp policy remaining
132           (by omega) (trace 0).2.2 (Fin.tail trace)

134 /-- The recursive mean Bellman innovation is jointly measurable in start state and trace. -/
135 theorem measurable_sampledCumulativeMeanBellmanInnovationFrom
136     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
137     (remaining : Nat) (hremaining : remaining ≤ mdp.horizon) :
138     Measurable (fun p : State × RewardStepTrace Action State remaining =>
139       mdp.sampledCumulativeMeanBellmanInnovationFrom policy remaining
140         hremaining p.1 p.2) := by
141   induction remaining with
142   | zero =>
143       simp [sampledCumulativeMeanBellmanInnovationFrom]
144   | succ remaining ih =>
145       let head : State × RewardStepTrace Action State (remaining + 1) →
146           Action × (Real × State) := fun p => p.2 0
147       have hhead : Measurable head := (measurable_pi_apply 0).comp measurable_snd
148       have htail : Measurable
149           (fun p : State × RewardStepTrace Action State (remaining + 1) =>
150             ((head p).2.2, Fin.tail p.2)) :=
151         hhead.snd.snd.prodMk
152           ((RewardStepTrace.measurable_tail remaining).comp measurable_snd)
153       have hheadInnovation : Measurable
154           (fun p : State × RewardStepTrace Action State (remaining + 1) =>
155             mdp.reward p.1 (head p).1 +
156                 policy.valueRemaining remaining (by omega) (head p).2.2 -
157               policy.valueRemaining (remaining + 1) hremaining p.1) :=
158         ((mdp.measurable_reward.comp (measurable_fst.prodMk hhead.fst)).add
159           ((policy.measurable_valueRemaining remaining (by omega)).comp
160             hhead.snd.snd)).sub
161           ((policy.measurable_valueRemaining (remaining + 1) hremaining).comp
162             measurable_fst)
163       simpa only [sampledCumulativeMeanBellmanInnovationFrom] using
164         hheadInnovation.add ((ih (by omega)).comp htail)

243 /-- A one-step mean Bellman innovation is sub-Gaussian at its stage envelope. -/
244 theorem actionStateKernel_meanBellmanInnovation_hasSubgaussianMGF
245     (policy : MarkovPolicy mdp)
246     (rewardBound : NNReal)
247     (hrewardBound : ∀ state action,
248       |mdp.reward state action| ≤ (rewardBound : Real))
249     (remaining : Nat) (hremaining : remaining + 1 ≤ mdp.horizon)
250     (state : State) :
251     ProbabilityTheory.HasSubgaussianMGF
252       (fun head : Action × State =>
253         mdp.reward state head.1 +
254           policy.valueRemaining remaining (by omega) head.2 -
255             policy.valueRemaining (remaining + 1) hremaining state)
256       (meanBellmanInnovationStepVarianceProxy rewardBound (remaining + 1))
257       (policy.actionStateKernel
258         ⟨mdp.horizon - (remaining + 1), by omega⟩ state) := by
259   let bound : NNReal := ((remaining + 1 : Nat) : NNReal) * rewardBound
260   let X : Action × State → Real := fun head =>
261     mdp.reward state head.1 +
262       policy.valueRemaining remaining (by omega) head.2
263   have hX : Measurable X :=
264     (mdp.measurable_reward.comp
265       (measurable_const.prodMk measurable_fst)).add
266       ((policy.measurable_valueRemaining remaining (by omega)).comp measurable_snd)
267   have hbound : ∀ head, |X head| ≤ (bound : Real) := by
268     intro head
269     calc
270       |X head| ≤ |mdp.reward state head.1| +
271           |policy.valueRemaining remaining (by omega) head.2| := abs_add_le _ _
272       _ ≤ (rewardBound : Real) +
273           (remaining : Real) * (rewardBound : Real) :=
274         add_le_add (hrewardBound state head.1)
275           (policy.valueRemaining_abs_le_of_rewardBound rewardBound
276             hrewardBound remaining (by omega) head.2)
277       _ = (bound : Real) := by
278         simp [bound]
279         ring
280   have hsub := Concentration.boundedCentered_hasSubgaussianMGF_of_mem_Icc_integral_eq
281     (policy.actionStateKernel
282       ⟨mdp.horizon - (remaining + 1), by omega⟩ state)
283     hX.aemeasurable
284     (Filter.Eventually.of_forall fun head => (abs_le.mp (hbound head)))
285     (integral_meanBellmanReturn_actionStateKernel_eq_valueRemaining
286       policy remaining hremaining state)
287   have hproxy :
288       Concentration.intervalVarianceProxy (-(bound : Real)) (bound : Real) =
289         meanBellmanInnovationStepVarianceProxy rewardBound (remaining + 1) := by
290     rw [Concentration.intervalVarianceProxy_neg_coe]
291     rfl
292   rw [hproxy] at hsub
293   simpa [X] using hsub

447               (source.stochasticTrajectoryKernelRemaining policy remaining

449             dsimp [tailStateKernel]
