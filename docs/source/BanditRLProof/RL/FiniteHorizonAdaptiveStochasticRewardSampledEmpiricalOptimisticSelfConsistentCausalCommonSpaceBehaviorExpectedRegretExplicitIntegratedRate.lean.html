4 # Natural-causal behavior expected-regret explicit integrated rate
5 
6 This module turns the actual exploratory `source.successorPolicyAt` pointwise
7 planning certificate into a finite-coordinate expectation bound on the same
8 genuine heterogeneous dependent causal trajectory measure. A generic event
9 split integrates a local bound off one measurable bad event and a global bound
10 on it. For the natural causal source, the local term is the compiled planning
11 rate, the global term is `2 * horizon`, and the event has the compiled two-share
12 coordinate model-confidence budget.
13 
14 The resulting envelope is explicit: the model-event fallback is
15 `4 * horizon * selfConsistentScheduledLocalDelta mdp t`, and a closed-form
16 theorem expands both this confidence term and the existing planning rate. The
17 envelope tends to zero, yielding a quantitative squeeze proof of expected-
18 absolute behavior-regret convergence without using the realized-return MGF.
19 
20 Regularity: the generic integral lemma needs a probability measure, a
21 measurable bad event, integrability, and pointwise local/global bounds. The
22 causal consumer retains finite nonempty Standard Borel State/Action,
23 probability initial law, positive horizon/base floor/reward proxy, bounded
24 means, uniform mean-compatible selected-reward sub-Gaussianity, and path
25 support. Failure policy preserves the actual exploratory policy, one dependent
26 source, actual samples, `n`-prefix to `n+1` selection, scheduled budgets,
27 initial exclusion, and behavior/recommendation separation. This is a
28 per-coordinate integrated rate, not cumulative/average regret, anytime,
29 reachability, minimax/optimal rate, or complete UCB-VI.
30 -/

37 /-- Integrate a local bound off one bad event and a global bound on it. -/
38 theorem integral_le_add_const_mul_measureReal_of_le_on_compl
39     {Omega : Type*} [MeasurableSpace Omega]
40     (mu : Measure Omega) [IsProbabilityMeasure mu]
41     (f : Omega -> Real) (bad : Set Omega) (hbad : MeasurableSet bad)
42     (hf : Integrable f mu) (rate envelope : Real)
43     (hrate : 0 <= rate)
44     (hglobal : forall omega, f omega <= envelope)
45     (hgood : forall omega, omega ∉ bad -> f omega <= rate) :
46     integral mu f <= rate + envelope * (mu bad).toReal := by
47   let penalty : Omega -> Real := bad.indicator (fun _ => envelope)
48   have hpenalty : Integrable penalty mu :=
49     (integrable_const envelope).indicator hbad
50   have hmajorant : Integrable (fun omega => rate + penalty omega) mu :=
51     (integrable_const rate).add hpenalty
52   calc
53     integral mu f <= integral mu (fun omega => rate + penalty omega) := by
54       apply integral_mono hf hmajorant
55       intro omega
56       by_cases hmem : omega ∈ bad
57       · simp [penalty, hmem]
58         exact hglobal omega |>.trans (le_add_of_nonneg_left hrate)
59       · simp [penalty, hmem]
60         exact hgood omega hmem
61     _ = rate + envelope * (mu bad).toReal := by
62       rw [integral_add (integrable_const rate) hpenalty]
63       simp [penalty, hbad, Measure.real_def, mul_comm]
64 
65 end BanditRLProof.FiniteHorizonRL
