4 /-!
5 # Generated empirical reward exactness for finite-horizon iid batches
6 
7 The finite-horizon MDP surface has a deterministic reward function. Consequently,
8 records extracted from genuine trajectories have exact empirical rewards at every
9 visited coordinate; no reward concentration or additional failure budget is
10 needed. This module records that structural fact and combines it with the existing
11 eligible empirical-transition confidence event.
12 -/

28 /-- Every recorded reward agrees with the MDP reward at its recorded state and action. -/
29 def RewardConsistent {mdp : MDP State Action} {episodes : Nat}
30     (batch : EpisodeBatch mdp episodes) : Prop :=
31   ∀ episode stage,
32     (batch episode stage).reward =
33       mdp.reward (batch episode stage).state (batch episode stage).action
34 
35 /-- Reward consistency is a measurable property of finite episode batches. -/
36 theorem measurableSet_rewardConsistent
37     {mdp : MDP State Action} {episodes : Nat} :
38     MeasurableSet {batch : EpisodeBatch mdp episodes | batch.RewardConsistent} := by
39   simp only [RewardConsistent, Set.setOf_forall]
40   apply MeasurableSet.iInter
41   intro episode
42   apply MeasurableSet.iInter
43   intro stage
44   have hstep : Measurable
45       (fun batch : EpisodeBatch mdp episodes => batch episode stage) :=
46     (measurable_pi_apply stage).comp (measurable_pi_apply episode)
47   have hstate : Measurable
48       (fun batch : EpisodeBatch mdp episodes => (batch episode stage).state) :=
49     EpisodeStep.measurable_state.comp hstep
50   have haction : Measurable
51       (fun batch : EpisodeBatch mdp episodes => (batch episode stage).action) :=
52     EpisodeStep.measurable_action.comp hstep
53   exact measurableSet_eq_fun (EpisodeStep.measurable_reward.comp hstep)
54     (mdp.measurable_reward.comp (hstate.prodMk haction))

56 /-- In a reward-consistent batch, the reward sum is visit count times the true reward. -/
57 theorem rewardSum_eq_visitCount_mul_reward_of_rewardConsistent
58     [DecidableEq State] [DecidableEq Action]
59     {mdp : MDP State Action} {episodes : Nat}
60     (batch : EpisodeBatch mdp episodes) (hbatch : batch.RewardConsistent)
61     (stage : Fin mdp.horizon) (state : State) (action : Action) :
62     batch.rewardSum stage state action =
63       (batch.visitCount stage state action : Real) * mdp.reward state action := by
64   classical
65   unfold rewardSum visitCount
66   rw [Nat.cast_sum, Finset.sum_mul]
67   apply Finset.sum_congr rfl
68   intro episode _
69   by_cases hvisit :
70       (batch episode stage).state = state ∧
71         (batch episode stage).action = action
72   · simp only [hvisit]
73     simpa [hvisit.1, hvisit.2] using hbatch episode stage
74   · simp [hvisit]
75 
76 /-- Positive visits cancel the denominator, so empirical reward is exact. -/
77 theorem empiricalReward_eq_reward_of_rewardConsistent
78     [DecidableEq State] [DecidableEq Action]
79     {mdp : MDP State Action} {episodes : Nat}
80     (batch : EpisodeBatch mdp episodes) (hbatch : batch.RewardConsistent)
81     (stage : Fin mdp.horizon) (state : State) (action : Action)
82     (hcount : batch.visitCount stage state action ≠ 0) :
83     batch.empiricalReward stage state action = mdp.reward state action := by
84   have hcountReal : (batch.visitCount stage state action : Real) ≠ 0 := by
85     exact_mod_cast hcount
86   rw [empiricalReward,
87     batch.rewardSum_eq_visitCount_mul_reward_of_rewardConsistent hbatch]
88   exact mul_div_cancel_left₀ _ hcountReal
89 
90 end EpisodeBatch

156 /--
157 On the simultaneous good event, reward consistency gives exact empirical reward
158 and the existing eligible margin gives every next-state transition bound.
159 -/
160 theorem empiricalReward_eq_and_transition_lt_of_not_mem_simultaneousCountBadEvent
161     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
162     (initialState : Measure State) [IsProbabilityMeasure initialState]
163     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
164     (hbatch : batch ∉
165       policy.simultaneousCountBadEvent initialState episodes delta)
166     (hreward : batch.RewardConsistent) (defaultState : State)
167     (coordinate : VisitCoordinate mdp)
168     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
169       coordinate.expectedCount policy initialState episodes) :
170     batch.empiricalReward coordinate.stage coordinate.state coordinate.action =
171         mdp.reward coordinate.state coordinate.action ∧
172       ∀ nextState,
173         |batch.empiricalTransitionMass defaultState coordinate.stage
174               coordinate.state coordinate.action nextState -
175             (mdp.transition
176               (coordinate.state, coordinate.action)).real {nextState}| <
177           2 * simultaneousCountConfidenceRadius mdp episodes delta /
178             (coordinate.count batch : Real) := by
179   have hcount : 0 < coordinate.count batch :=
180     policy.visitCoordinate_count_pos_of_not_mem_simultaneousCountBadEvent
181       initialState batch hbatch coordinate hmargin
182   constructor
183   · exact batch.empiricalReward_eq_reward_of_rewardConsistent hreward
184       coordinate.stage coordinate.state coordinate.action
185         (Nat.ne_of_gt hcount)
186   · intro nextState
187     exact policy.empiricalTransitionMass_abs_sub_transition_lt_of_not_mem_simultaneousCountBadEvent
188       initialState batch hbatch defaultState coordinate hmargin nextState

218 /--
219 Route endpoint: the existing global-delta event simultaneously controls every
220 eligible transition coordinate, while reward-consistent batches have zero reward
221 error at those same positive-count coordinates.
222 -/
223 theorem iidEpisodeBatch_eligible_empiricalReward_exact_and_transition_confidence
224     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
225     (initialState : Measure State) [IsProbabilityMeasure initialState]
226     (episodes : Nat) (hepisodes : 0 < episodes)
227     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1)
228     (defaultState : State) (eligible : Finset (VisitCoordinate mdp))
229     (hmargin : ∀ coordinate ∈ eligible,
230       simultaneousCountConfidenceRadius mdp episodes delta <
231         coordinate.expectedCount policy initialState episodes) :
232     MeasurableSet
233         (policy.simultaneousCountBadEvent initialState episodes delta) ∧
234       (policy.iidEpisodeBatchMeasure initialState episodes)
235           (policy.simultaneousCountBadEvent initialState episodes delta) ≤
236         ENNReal.ofReal delta ∧
237       ∀ᵐ batch ∂policy.iidEpisodeBatchMeasure initialState episodes,
238         batch ∉ policy.simultaneousCountBadEvent initialState episodes delta ->
239           ∀ coordinate ∈ eligible,
240             batch.empiricalReward coordinate.stage coordinate.state
241                   coordinate.action =
242                 mdp.reward coordinate.state coordinate.action ∧
243               ∀ nextState,
244                 |batch.empiricalTransitionMass defaultState coordinate.stage
245                       coordinate.state coordinate.action nextState -
246                     (mdp.transition
247                       (coordinate.state, coordinate.action)).real {nextState}| <
248                   2 * simultaneousCountConfidenceRadius mdp episodes delta /
249                     (coordinate.count batch : Real) := by
250   refine ⟨policy.measurableSet_simultaneousCountBadEvent
251       initialState episodes delta,
252     policy.iidEpisodeBatch_simultaneousCountBadEvent_le
253       initialState episodes hepisodes delta hdelta hdelta_le_one, ?_⟩
254   filter_upwards [policy.iidEpisodeBatchMeasure_rewardConsistent_ae
255     initialState episodes] with batch hreward
256   intro hbatch coordinate hcoordinate
257   exact policy.empiricalReward_eq_and_transition_lt_of_not_mem_simultaneousCountBadEvent
258     initialState batch hbatch hreward defaultState coordinate
259       (hmargin coordinate hcoordinate)
