4 /-!
5 # Estimated finite-horizon models produce optimistic certificates
6 
7 This module connects a stage-indexed estimated reward/transition model to the
8 compiled deterministic optimistic-certificate route.  Separate two-sided
9 reward and transition-expectation radii are required only on the recursively
10 generated tail upper values.  They produce a true Bellman certificate.  The
11 deterministic policy greedy for the estimated optimistic backup then has true
12 Bellman residual at most twice its selected reward-plus-transition radius, so
13 the existing occupancy theorem gives a single-episode expected-regret bound.
14 -/

31 /-- Chronological stage corresponding to a successor remaining-horizon index. -/
32 def decisionStageRemaining (mdp : MDP State Action) (remaining : Nat)
33     (hremaining : remaining + 1 <= mdp.horizon) : Fin mdp.horizon :=
34   ⟨mdp.horizon - (remaining + 1), by omega⟩

36 /--
37 A stage-indexed estimated MDP together with separate reward and transition
38 confidence radii.  Confidence itself is a downstream proposition because the
39 transition error is evaluated on the recursively generated upper value.
40 -/
41 structure EstimatedModelPlan (mdp : MDP State Action) where
42   estimatedReward : Fin mdp.horizon -> State -> Action -> Real
43   measurable_estimatedReward : forall stage,
44     Measurable (Function.uncurry (estimatedReward stage))
45   estimatedTransition : Fin mdp.horizon ->
46     ProbabilityTheory.Kernel (State × Action) State
47   estimatedTransition_isMarkov : forall stage,
48     ProbabilityTheory.IsMarkovKernel (estimatedTransition stage)
49   rewardRadius : Fin mdp.horizon -> State -> Action -> Real
50   measurable_rewardRadius : forall stage,
51     Measurable (Function.uncurry (rewardRadius stage))
52   transitionRadius : Fin mdp.horizon -> State -> Action -> Real
53   measurable_transitionRadius : forall stage,
54     Measurable (Function.uncurry (transitionRadius stage))

64 /-- The true MDP with zero radii is the canonical exact estimated plan. -/
65 def exactModelPlan (mdp : MDP State Action) : EstimatedModelPlan mdp where
66   estimatedReward _ := mdp.reward
67   measurable_estimatedReward _ := mdp.measurable_reward
68   estimatedTransition _ := mdp.transition
69   estimatedTransition_isMarkov _ := mdp.transition_isMarkov
70   rewardRadius _ := fun _ _ => 0
71   measurable_rewardRadius _ := measurable_const
72   transitionRadius _ := fun _ _ => 0
73   measurable_transitionRadius _ := measurable_const

75 /-- Estimated next-state expectation of a continuation value. -/
76 noncomputable def transitionValue
77     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
78     (stage : Fin mdp.horizon) (value : State -> Real)
79     (state : State) (action : Action) : Real :=
80   ∫ nextState, value nextState ∂plan.estimatedTransition stage (state, action)
81 
82 /-- Estimated one-step reward plus continuation value. -/
83 noncomputable def bellmanQ
84     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
85     (stage : Fin mdp.horizon) (value : State -> Real)
86     (state : State) (action : Action) : Real :=
87   plan.estimatedReward stage state action +
88     plan.transitionValue stage value state action
89 
90 /-- Estimated Bellman action value enlarged by both confidence radii. -/
91 noncomputable def optimisticQ
92     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
93     (stage : Fin mdp.horizon) (value : State -> Real)
94     (state : State) (action : Action) : Real :=
95   plan.bellmanQ stage value state action +
96     plan.rewardRadius stage state action +
97     plan.transitionRadius stage state action

121 /-- A finite action maximizing the estimated optimistic action value. -/
122 noncomputable def optimisticAction
123     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
124     (stage : Fin mdp.horizon) (value : State -> Real) (state : State) : Action :=
125   FiniteRealArgmax.choose

154 /-- Recursive estimated optimistic value indexed by decisions remaining. -/
155 noncomputable def upperValueRemaining
156     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp) :
157     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
158   | 0, _ => fun _ => 0

181 /--
182 Two-sided model confidence on the recursive upper-value route.  The transition
183 contract is deliberately not quantified over arbitrary unbounded values.
184 -/
185 structure Confidence
186     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp) : Prop where
187   rewardError_le_radius : forall (stage : Fin mdp.horizon)
188     (state : State) (action : Action),
189     |plan.estimatedReward stage state action - mdp.reward state action| <=
190       plan.rewardRadius stage state action
191   transitionError_le_radius : forall (remaining : Nat)
192     (hremaining : remaining + 1 <= mdp.horizon)
193     (state : State) (action : Action),
194     |plan.transitionValue (mdp.decisionStageRemaining remaining hremaining)
195           (plan.upperValueRemaining remaining (by omega)) state action -
196         mdp.transitionValue
197           (plan.upperValueRemaining remaining (by omega)) state action| <=
198       plan.transitionRadius
199         (mdp.decisionStageRemaining remaining hremaining) state action

212 /-- Two-sided confidence makes every true action value optimistic. -/
213 theorem Confidence.trueBellmanQ_le_optimisticQ
214     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
215     (confidence : plan.Confidence) (remaining : Nat)
216     (hremaining : remaining + 1 <= mdp.horizon)
217     (state : State) (action : Action) :
218     mdp.bellmanQ (plan.upperValueRemaining remaining (by omega)) state action <=
219       plan.optimisticQ (mdp.decisionStageRemaining remaining hremaining)
220         (plan.upperValueRemaining remaining (by omega)) state action := by
221   have hreward :=
222     (abs_le.mp (confidence.rewardError_le_radius
223       (mdp.decisionStageRemaining remaining hremaining) state action)).1
224   have htransition :=
225     (abs_le.mp (confidence.transitionError_le_radius
226       remaining hremaining state action)).1
227   unfold MDP.bellmanQ optimisticQ bellmanQ
228   linarith

230 /-- The recursive estimated optimistic values form a true Bellman certificate. -/
231 noncomputable def certificate
232     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
233     (confidence : plan.Confidence) : mdp.OptimisticBellmanCertificate where
234   upperValueRemaining := plan.upperValueRemaining
235   upperValueRemaining_zero := rfl
236   optimalBellman_le_upperValueRemaining_succ := by
237     intro remaining hremaining state
238     calc
239       mdp.optimalBellman
240           (plan.upperValueRemaining remaining (by omega)) state =
241           mdp.bellmanQ (plan.upperValueRemaining remaining (by omega)) state
242             (mdp.optimalAction
243               (plan.upperValueRemaining remaining (by omega)) state) := rfl
244       _ <= plan.optimisticQ
245           (mdp.decisionStageRemaining remaining hremaining)
246           (plan.upperValueRemaining remaining (by omega)) state
247           (mdp.optimalAction
248             (plan.upperValueRemaining remaining (by omega)) state) :=
249         confidence.trueBellmanQ_le_optimisticQ remaining hremaining state _
250       _ <= plan.optimisticBellman
251           (mdp.decisionStageRemaining remaining hremaining)
252           (plan.upperValueRemaining remaining (by omega)) state :=
253         plan.optimisticQ_le_optimisticAction _ _ _ _
254       _ = plan.upperValueRemaining (remaining + 1) hremaining state := by
255         rw [upperValueRemaining]

310 /-- The deterministic estimated-greedy policy Bellman integral selects its action. -/
311 theorem optimisticPolicy_bellman_eq_bellmanQ
312     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
313     (stage : Fin mdp.horizon) (value : State -> Real) (state : State) :
314     plan.optimisticPolicy.bellman stage value state =
315       mdp.bellmanQ value state (plan.optimisticActionAt stage state) := by

323 /-- Reward-plus-transition radius selected by the estimated optimistic policy. -/
324 noncomputable def selectedRadiusRemaining
325     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp)
326     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
327     (state : State) : Real :=
328   let stage := mdp.decisionStageRemaining remaining hremaining
329   let action := plan.optimisticAction stage

335 /-- Every selected radius is nonnegative under the two-sided confidence contract. -/
336 theorem Confidence.selectedRadiusRemaining_nonneg
337     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
338     (confidence : plan.Confidence) (remaining : Nat)
339     (hremaining : remaining + 1 <= mdp.horizon) (state : State) :
340     0 <= plan.selectedRadiusRemaining remaining hremaining state := by
341   let stage := mdp.decisionStageRemaining remaining hremaining
342   let action := plan.optimisticAction stage
343     (plan.upperValueRemaining remaining (by omega)) state
344   have hreward := confidence.rewardError_le_radius stage state action
345   have htransition := confidence.transitionError_le_radius
346     remaining hremaining state action
347   unfold selectedRadiusRemaining
348   dsimp only
349   exact add_nonneg (abs_nonneg _ |>.trans hreward)
350     (abs_nonneg _ |>.trans htransition)

352 /--
353 The estimated-greedy policy residual is at most twice its selected model
354 confidence radius.  One side of each absolute-error bound produced optimism;
355 the other side controls the remaining selected-action overestimate.
356 -/
357 theorem Confidence.policyBellmanResidual_le_two_selectedRadiusRemaining
358     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
359     (confidence : plan.Confidence) (remaining : Nat)
360     (hremaining : remaining + 1 <= mdp.horizon) (state : State) :
361     (plan.certificate confidence).policyBellmanResidual
362         plan.optimisticPolicy remaining hremaining state <=
363       2 * plan.selectedRadiusRemaining remaining hremaining state := by
364   change
365     plan.upperValueRemaining (remaining + 1) hremaining state -
366         plan.optimisticPolicy.bellman
367           (mdp.decisionStageRemaining remaining hremaining)
368           (plan.upperValueRemaining remaining (by omega)) state <=
369       2 * plan.selectedRadiusRemaining remaining hremaining state
370   rw [upperValueRemaining]
371   rw [plan.optimisticPolicy_bellman_eq_bellmanQ]
372   rw [congrFun
373     (plan.optimisticActionAt_decisionStageRemaining remaining hremaining) state]
374   have hreward :=
375     (abs_le.mp (confidence.rewardError_le_radius
376       (mdp.decisionStageRemaining remaining hremaining) state
377       (plan.optimisticAction
378         (mdp.decisionStageRemaining remaining hremaining)
379         (plan.upperValueRemaining remaining (by omega)) state))).2
380   have htransition :=
381     (abs_le.mp (confidence.transitionError_le_radius
382       remaining hremaining state
383       (plan.optimisticAction
384         (mdp.decisionStageRemaining remaining hremaining)
385         (plan.upperValueRemaining remaining (by omega)) state))).2
386   unfold optimisticBellman selectedRadiusRemaining optimisticQ bellmanQ
387     MDP.bellmanQ
388   dsimp only
389   linarith

421 /--
422 Route endpoint: estimated-model confidence simultaneously gives global
423 optimism and the selected-radius single-episode expected-regret bound.
424 -/
425 theorem Confidence.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
426     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
427     (confidence : plan.Confidence) (initialState : Measure State)
428     [IsProbabilityMeasure initialState] :
