4 /-!
5 # Simultaneous iid count confidence for finite-horizon RL
6 
7 This module puts every finite visit and joint-transition count coordinate into
8 one index type and applies an equal-share finite union bound to the compiled
9 fixed-coordinate tails. The route remains fixed-policy iid. It does not form
10 visit-conditioned ratios or claim adaptive, anytime, or cumulative regret.
11 -/

28 /--
29 Finite index of all visit and joint-transition count coordinates of an MDP.
30 -/
31 inductive CountCoordinate (mdp : MDP State Action) where
32   | visit (stage : Fin mdp.horizon) (state : State) (action : Action)
33   | transition (stage : Fin mdp.horizon) (state : State) (action : Action)
34       (nextState : State)
35   deriving DecidableEq, Fintype

39 /-- Explicit finite-sum presentation of the two coordinate families. -/
40 def equivVisitSumTransition (mdp : MDP State Action) :
41     CountCoordinate mdp ≃
42       (Fin mdp.horizon × State × Action) ⊕
43         (Fin mdp.horizon × State × Action × State) where
44   toFun
45     | .visit stage state action => Sum.inl (stage, (state, action))
46     | .transition stage state action nextState =>
47         Sum.inr (stage, (state, (action, nextState)))
48   invFun
49     | Sum.inl (stage, (state, action)) => .visit stage state action
50     | Sum.inr (stage, (state, (action, nextState))) =>
51         .transition stage state action nextState
52   left_inv coordinate := by cases coordinate <;> rfl
53   right_inv coordinate := by
54     rcases coordinate with coordinate | coordinate
55     · rcases coordinate with ⟨stage, state, action⟩
56       rfl
57     · rcases coordinate with ⟨stage, state, action, nextState⟩
58       rfl
59 
60 end CountCoordinate

62 /-- Number of visit and joint-transition coordinates in the simultaneous family. -/
63 def countCoordinateCard (mdp : MDP State Action) : Nat :=
64   Fintype.card (CountCoordinate mdp)

69 /-- The simultaneous family contains `H*S*A` visits and `H*S*A*S` transitions. -/
70 theorem countCoordinateCard_eq (mdp : MDP State Action) :
71     countCoordinateCard mdp =
72       mdp.horizon * Fintype.card State * Fintype.card Action +
73         mdp.horizon * Fintype.card State * Fintype.card Action *
74           Fintype.card State := by
75   unfold countCoordinateCard
76   rw [Fintype.card_congr (CountCoordinate.equivVisitSumTransition mdp)]
77   simp [Nat.mul_assoc]

79 /-- Equal confidence share assigned to each count coordinate. -/
80 noncomputable def simultaneousCountDelta
81     (mdp : MDP State Action) (delta : Real) : Real :=
82   delta / (countCoordinateCard mdp : Real)
83 
84 /-- Common count radius after allocating the global confidence budget equally. -/
85 noncomputable def simultaneousCountConfidenceRadius
86     (mdp : MDP State Action) (episodes : Nat) (delta : Real) : Real :=
87   Concentration.subGaussianSumConfidenceRadius
88     (MarkovPolicy.iidBernoulliVarianceProxy episodes)
89     (simultaneousCountDelta mdp delta)

93 /-- Real count deviation selected by a visit or joint-transition coordinate. -/
94 noncomputable def deviation
95     {mdp : MDP State Action} (coordinate : CountCoordinate mdp)
96     (policy : MarkovPolicy mdp)
97     (initialState : Measure State) [IsProbabilityMeasure initialState]
98     {episodes : Nat} (batch : EpisodeBatch mdp episodes) : Real :=
99   match coordinate with
100   | .visit stage state action =>
101       (batch.visitCount stage state action : Real) -
102         (episodes : Real) *
103           policy.stageVisitProbability initialState stage state action
104   | .transition stage state action nextState =>
105       (batch.transitionCount stage state action nextState : Real) -
106         (episodes : Real) *
107           policy.stageTransitionJointProbability initialState stage state action nextState
108 
109 omit [Nonempty State] [Nonempty Action] in
110 /-- Every selected count deviation is measurable on the mapped batch space. -/
111 theorem measurable_deviation
112     {mdp : MDP State Action} (coordinate : CountCoordinate mdp)
113     (policy : MarkovPolicy mdp)
114     (initialState : Measure State) [IsProbabilityMeasure initialState]
115     {episodes : Nat} :
116     Measurable (coordinate.deviation policy initialState :
117       EpisodeBatch mdp episodes → Real) := by
118   cases coordinate with
119   | visit stage state action =>
120       exact policy.measurable_visitCountDeviation initialState stage state action
121   | transition stage state action nextState =>
122       exact policy.measurable_transitionCountDeviation
123         initialState stage state action nextState

125 /-- Two-sided bad event for one selected coordinate at a supplied delta. -/
126 noncomputable def badEvent
127     {mdp : MDP State Action} (coordinate : CountCoordinate mdp)
128     (policy : MarkovPolicy mdp)
129     (initialState : Measure State) [IsProbabilityMeasure initialState]
130     (episodes : Nat) (coordinateDelta : Real) : Set (EpisodeBatch mdp episodes) :=
131   {batch |
132     Concentration.subGaussianSumConfidenceRadius
133         (MarkovPolicy.iidBernoulliVarianceProxy episodes) coordinateDelta ≤
134       |coordinate.deviation policy initialState batch|}

152 /-- The compiled marginal tail dispatches over the finite coordinate type. -/
153 theorem measure_countCoordinate_badEvent_le
154     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
155     (initialState : Measure State) [IsProbabilityMeasure initialState]
156     (episodes : Nat) (hepisodes : 0 < episodes)
157     (coordinate : CountCoordinate mdp)
158     (coordinateDelta : Real) (hdelta : 0 < coordinateDelta)
159     (hdelta_le_one : coordinateDelta ≤ 1) :
160     (policy.iidEpisodeBatchMeasure initialState episodes)
161         (coordinate.badEvent policy initialState episodes coordinateDelta) ≤
162       ENNReal.ofReal coordinateDelta := by
163   cases coordinate with
164   | visit stage state action =>
165       simpa [CountCoordinate.badEvent, CountCoordinate.deviation] using
166         policy.iidEpisodeBatch_visitCount_abs_tail_le
167           initialState episodes hepisodes stage state action
168           coordinateDelta hdelta hdelta_le_one
169   | transition stage state action nextState =>
170       simpa [CountCoordinate.badEvent, CountCoordinate.deviation] using
171         policy.iidEpisodeBatch_transitionCount_abs_tail_le
172           initialState episodes hepisodes stage state action nextState
173           coordinateDelta hdelta hdelta_le_one

199 /-- A nonempty coordinate family receives a positive equal delta share. -/
200 theorem simultaneousCountDelta_pos
201     {mdp : MDP State Action} (hcoordinate : Nonempty (CountCoordinate mdp))
202     {delta : Real} (hdelta : 0 < delta) :
203     0 < simultaneousCountDelta mdp delta := by
204   unfold simultaneousCountDelta countCoordinateCard
205   exact div_pos hdelta (by
206     exact_mod_cast Fintype.card_pos_iff.mpr hcoordinate)

211 /-- A global delta at most one gives every nonempty-family share at most one. -/
212 theorem simultaneousCountDelta_le_one
213     {mdp : MDP State Action} (hcoordinate : Nonempty (CountCoordinate mdp))
214     {delta : Real} (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1) :
215     simultaneousCountDelta mdp delta ≤ 1 := by
216   have hcardNat : 0 < Fintype.card (CountCoordinate mdp) :=
217     Fintype.card_pos_iff.mpr hcoordinate
218   have hcardReal : (1 : Real) ≤ Fintype.card (CountCoordinate mdp) := by
219     exact_mod_cast hcardNat
220   exact (div_le_self (le_of_lt hdelta) hcardReal).trans hdelta_le_one

223 /--
224 All finite visit and joint-transition count deviations share one global delta
225 budget. When the coordinate family is empty (in particular at horizon zero),
226 the bad union is empty and no positive-horizon premise is needed.
227 -/
228 theorem iidEpisodeBatch_simultaneousCountBadEvent_le
229     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
230     (initialState : Measure State) [IsProbabilityMeasure initialState]
231     (episodes : Nat) (hepisodes : 0 < episodes)
232     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1) :
233     (policy.iidEpisodeBatchMeasure initialState episodes)
234         (policy.simultaneousCountBadEvent initialState episodes delta) ≤
235       ENNReal.ofReal delta := by
236   classical
237   by_cases hcoordinate : Nonempty (CountCoordinate mdp)
238   · letI : Nonempty (CountCoordinate mdp) := hcoordinate
239     have htail : ∀ coordinate : CountCoordinate mdp,
240         coordinate ∈ (Finset.univ : Finset (CountCoordinate mdp)) →
241           (policy.iidEpisodeBatchMeasure initialState episodes)
242               (coordinate.badEvent policy initialState episodes
243                 (simultaneousCountDelta mdp delta)) ≤
244             ENNReal.ofReal
245               (delta / ((Finset.univ : Finset (CountCoordinate mdp)).card : Real)) := by
246       intro coordinate _hcoordinate
247       simpa [simultaneousCountDelta, countCoordinateCard] using
248         policy.measure_countCoordinate_badEvent_le
249           initialState episodes hepisodes coordinate
250           (simultaneousCountDelta mdp delta)
251           (simultaneousCountDelta_pos hcoordinate hdelta)
252           (simultaneousCountDelta_le_one
253             hcoordinate hdelta hdelta_le_one)
254     simpa [simultaneousCountBadEvent] using
255       (ProbabilityUnionBound.measure_biUnion_finset_le_of_uniform
256         (policy.iidEpisodeBatchMeasure initialState episodes)
257         (Finset.univ : Finset (CountCoordinate mdp))
258         Finset.univ_nonempty delta
259         (fun coordinate => coordinate.badEvent policy initialState episodes
260           (simultaneousCountDelta mdp delta))
261         htail)
262   · have hcard : Fintype.card (CountCoordinate mdp) = 0 :=
263       Nat.eq_zero_of_not_pos fun hpos =>
264         hcoordinate (Fintype.card_pos_iff.mp hpos)
265     letI : IsEmpty (CountCoordinate mdp) := Fintype.card_eq_zero_iff.mp hcard
266     simp [simultaneousCountBadEvent]

270 /-- Outside the simultaneous union, every indexed deviation is below its radius. -/
271 theorem countCoordinate_abs_deviation_lt_of_not_mem_simultaneousCountBadEvent
272     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
273     (initialState : Measure State) [IsProbabilityMeasure initialState]
274     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
275     (hbatch : batch ∉ policy.simultaneousCountBadEvent initialState episodes delta)
276     (coordinate : CountCoordinate mdp) :
277     |coordinate.deviation policy initialState batch| <
278       simultaneousCountConfidenceRadius mdp episodes delta := by
279   have hnot : batch ∉ coordinate.badEvent policy initialState episodes
280       (simultaneousCountDelta mdp delta) := by
281     intro hmem
282     apply hbatch
283     exact Set.mem_iUnion.2 ⟨coordinate, hmem⟩
284   exact lt_of_not_ge (by
285     simpa [CountCoordinate.badEvent, simultaneousCountConfidenceRadius] using hnot)

324 /--
325 Route endpoint: one global-delta bad union and all coordinatewise good-side
326 bounds under the same mapped fixed-policy iid episode-batch law.
327 -/
328 theorem iidEpisodeBatch_simultaneous_count_confidence
329     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
330     (initialState : Measure State) [IsProbabilityMeasure initialState]
331     (episodes : Nat) (hepisodes : 0 < episodes)
332     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1) :
333     (policy.iidEpisodeBatchMeasure initialState episodes)
334         (policy.simultaneousCountBadEvent initialState episodes delta) ≤
335         ENNReal.ofReal delta ∧
336       ∀ batch ∉ policy.simultaneousCountBadEvent initialState episodes delta,
337         ∀ coordinate : CountCoordinate mdp,
338           |coordinate.deviation policy initialState batch| <
339             simultaneousCountConfidenceRadius mdp episodes delta := by
340   refine ⟨policy.iidEpisodeBatch_simultaneousCountBadEvent_le
341     initialState episodes hepisodes delta hdelta hdelta_le_one, ?_⟩
342   intro batch hbatch coordinate
343   exact policy.countCoordinate_abs_deviation_lt_of_not_mem_simultaneousCountBadEvent
344     initialState batch hbatch coordinate
