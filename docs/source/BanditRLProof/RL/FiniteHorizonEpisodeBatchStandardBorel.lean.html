5 # Standard Borel instances for finite-horizon episode batches
6 
7 `EpisodeStep` uses an explicit product-coordinate `MeasurableSpace.comap`, so
8 the generic product instance is not visible to typeclass search.  This module
9 transports a Polish topology through the coordinate equivalence and then
10 installs the corresponding Standard Borel instance.  The deterministic and
11 stochastic infinite batch trajectories are countable products of their finite
12 batch spaces.
13 
14 The episode-step instance needs only measurable Standard Borel state and action
15 spaces.  The stochastic batch-trajectory wrapper additionally inherits finite
16 state and action spaces from the finite-horizon aliases.  These instances close
17 a regularity boundary only.  They do not supply any probability law,
18 independence, support, concentration, or regret property.
19 -/
20 
21 namespace BanditRLProof.FiniteHorizonRL
22 
23 universe u v

25 namespace EpisodeStep

27 variable {State : Type u} {Action : Type v}
28 
29 /-- Product coordinates underlying the measurable structure on `EpisodeStep`. -/
30 def toProdEquiv : EpisodeStep State Action ≃ State × Action × Real × State where
31   toFun step := (step.state, step.action, step.reward, step.nextState)
32   invFun data :=
33     { state := data.1
34       action := data.2.1
35       reward := data.2.2.1
36       nextState := data.2.2.2 }
37   left_inv step := by cases step; rfl
38   right_inv data := by rcases data with ⟨state, action, reward, nextState⟩; rfl
39 
40 /--
41 The product-coordinate measurable structure on an episode step is Standard
42 Borel whenever the state and action spaces are Standard Borel.
43 -/
44 noncomputable instance instStandardBorelSpace
45     [MeasurableSpace State] [MeasurableSpace Action]
46     [StandardBorelSpace State] [StandardBorelSpace Action] :
47     StandardBorelSpace (EpisodeStep State Action) := by
48   let e := toProdEquiv (State := State) (Action := Action)
49   letI := upgradeStandardBorel (State × Action × Real × State)
50   letI : TopologicalSpace (EpisodeStep State Action) :=
51     TopologicalSpace.induced e inferInstance
52   have hemb : MeasurableEmbedding e := by
53     rw [MeasurableEmbedding.iff_comap_eq]
54     exact ⟨e.injective, rfl, by simp⟩
55   letI : BorelSpace (EpisodeStep State Action) :=
56     hemb.borelSpace (Topology.IsInducing.induced e)
57   letI : PolishSpace (EpisodeStep State Action) := e.polishSpace_induced
58   infer_instance
59 
60 end EpisodeStep

66 /-- A stochastic infinite batch trajectory is a countable Borel product. -/
67 noncomputable instance instStochasticEpisodeBatchTrajectoryStandardBorelSpace
68     {mdp : MDP State Action} {episodes : Nat}
69     [StandardBorelSpace (StochasticEpisodeBatch mdp episodes)] :
70     StandardBorelSpace (StochasticEpisodeBatchTrajectory mdp episodes) := by
71   letI : ∀ _ : Nat, StandardBorelSpace (StochasticEpisodeBatch mdp episodes) :=
72     fun _ => inferInstance
73   exact StandardBorelSpace.pi_countable
74 
75 end BanditRLProof.FiniteHorizonRL
