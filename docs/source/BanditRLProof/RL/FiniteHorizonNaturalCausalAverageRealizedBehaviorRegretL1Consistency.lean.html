3 /-!
4 # All-prefix L1 consistency for natural causal average realized behavior regret
5 
6 This module controls the exact natural process which first normalizes every
7 successor batch by its own positive episode count and then weights rounds
8 equally. It is not the total-episode-mass-weighted process from the earlier L1
9 route. The argument exposes the cumulative normalized-return MGF, obtains a
10 first-moment inverse-square-root bound, and combines it with the compiled
11 logarithmic behavior expected-regret rate.
12 -/

188 /-- The normalized first-moment envelope is bounded by an inverse square root. -/
189 theorem selfConsistentScheduledNaturalCausalAverageReturnFirstMomentBound_le_inverseSqrtEnvelope
190     (mdp : MDP State Action) (varianceProxy : NNReal)
191     (baseVisitFloor : Real) (rounds : Nat) (hrounds : 0 < rounds) :
192     selfConsistentScheduledNaturalCausalAverageReturnFirstMomentBound
193         mdp varianceProxy baseVisitFloor rounds <=
194       selfConsistentScheduledNaturalCausalAverageReturnInverseSqrtEnvelope
195         mdp varianceProxy rounds := by
196   let V : Real :=
197     (selfConsistentScheduledNaturalCausalCumulativeReturnVarianceProxy mdp
198       varianceProxy baseVisitFloor rounds : Real)
199   let C : Real :=
200     (mdp.globalReturnDeviationPerEpisodeVarianceProxy 1 varianceProxy : Real)
201   let r : Real := (rounds : Real)
202   have hr : 0 < r := by
203     dsimp [r]
204     exact_mod_cast hrounds
205   have hproxyNN :=
206     selfConsistentScheduledNaturalCausalCumulativeReturnVarianceProxy_le_rounds_mul
207       mdp varianceProxy baseVisitFloor rounds
208   have hproxy : V <= r * C := by
209     exact_mod_cast hproxyNN
210   have hsqrt : Real.sqrt V <= Real.sqrt (r * C) :=
211     Real.sqrt_le_sqrt hproxy
212   rw [Real.sqrt_mul hr.le C] at hsqrt
213   have hnum :
214       2 * Real.sqrt V * Real.exp (1 / 2 : Real) <=
215         2 * (Real.sqrt r * Real.sqrt C) * Real.exp (1 / 2 : Real) := by
216     gcongr
217   have heq :
218       (2 * (Real.sqrt r * Real.sqrt C) * Real.exp (1 / 2 : Real)) / r =
219         (2 * Real.sqrt C * Real.exp (1 / 2 : Real)) / Real.sqrt r := by
220     have hsqrtPos : 0 < Real.sqrt r := Real.sqrt_pos.2 hr
221     field_simp [ne_of_gt hr, ne_of_gt hsqrtPos]
222     rw [Real.sq_sqrt hr.le]
223     ring
224   unfold selfConsistentScheduledNaturalCausalAverageReturnFirstMomentBound
225     selfConsistentScheduledNaturalCausalAverageReturnInverseSqrtEnvelope
226   change (2 * Real.sqrt V * Real.exp (1 / 2 : Real)) / r <=
227     (2 * Real.sqrt C * Real.exp (1 / 2 : Real)) / Real.sqrt r
228   rw [← heq]
229   exact div_le_div_of_nonneg_right hnum hr.le
