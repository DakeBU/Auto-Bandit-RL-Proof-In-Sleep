4 # Coordinate confidence for finite-horizon estimated models
5 
6 This module turns finite-state singleton transition-mass errors into the
7 transition-expectation confidence consumed by the estimated-model optimistic
8 regret route.  The only value regularity required is an absolute envelope for
9 the recursively generated tail upper value.
10 -/

25 /--
26 On a finite measurable space, coordinate bounds on singleton masses control
27 the expectation error of every pointwise-enveloped real value.
28 -/
29 theorem abs_integral_sub_integral_le_sum_coordinateRadius_mul_envelope
30     (estimated trueMeasure : Measure State)
31     [IsFiniteMeasure estimated] [IsFiniteMeasure trueMeasure]
32     (value coordinateRadius : State -> Real) (envelope : Real)
33     (hcoordinate : forall state,
34       |estimated.real {state} - trueMeasure.real {state}| <=
35         coordinateRadius state)
36     (hvalue : forall state, |value state| <= envelope) :
37     |(∫ state, value state ∂estimated) -
38         ∫ state, value state ∂trueMeasure| <=
39       ∑ state, coordinateRadius state * envelope := by
40   rw [integral_fintype
41     (integrable_of_fintype _ _ (measurable_of_finite value))]
42   rw [integral_fintype
43     (integrable_of_fintype _ _ (measurable_of_finite value))]
44   simp only [smul_eq_mul]
45   calc
46     |(∑ state, estimated.real {state} * value state) -
47         ∑ state, trueMeasure.real {state} * value state| =
48         |∑ state,
49           (estimated.real {state} - trueMeasure.real {state}) *
50             value state| := by
51       congr 1
52       rw [← Finset.sum_sub_distrib]
53       apply Finset.sum_congr rfl
54       intro state _
55       ring
56     _ <= ∑ state,
57         |(estimated.real {state} - trueMeasure.real {state}) *
58           value state| :=
59       Finset.abs_sum_le_sum_abs _ _
60     _ <= ∑ state, coordinateRadius state * envelope := by
61       apply Finset.sum_le_sum
62       intro state _
63       rw [abs_mul]
64       calc
65         |estimated.real {state} - trueMeasure.real {state}| *
66               |value state| <=
67             coordinateRadius state * |value state| :=
68           mul_le_mul_of_nonneg_right (hcoordinate state) (abs_nonneg _)
69         _ <= coordinateRadius state * envelope :=
70           mul_le_mul_of_nonneg_left (hvalue state)
71             ((abs_nonneg _).trans (hcoordinate state))

76 /--
77 Finite-state coordinate confidence sufficient for the recursive optimistic
78 model route.  The transition radius may be any upper bound on the displayed
79 coordinate sum, so later concentration producers can choose their own radii.
80 -/
81 structure CoordinateConfidence
82     {mdp : MDP State Action} (plan : EstimatedModelPlan mdp) where
83   transitionCoordinateRadius :
84     Fin mdp.horizon -> State -> Action -> State -> Real
85   valueEnvelope : Nat -> Real
86   rewardError_le_radius : forall (stage : Fin mdp.horizon)
87     (state : State) (action : Action),
88     |plan.estimatedReward stage state action - mdp.reward state action| <=
89       plan.rewardRadius stage state action
90   transitionCoordinateError_le_radius : forall (remaining : Nat)
91     (hremaining : remaining + 1 <= mdp.horizon)
92     (state : State) (action : Action) (nextState : State),
93     |(plan.estimatedTransition
94           (mdp.decisionStageRemaining remaining hremaining)
95           (state, action)).real {nextState} -
96         (mdp.transition (state, action)).real {nextState}| <=
97       transitionCoordinateRadius
98         (mdp.decisionStageRemaining remaining hremaining)
99         state action nextState
100   upperValue_abs_le_envelope : forall (remaining : Nat)
101     (hremaining : remaining + 1 <= mdp.horizon) (state : State),
102     |plan.upperValueRemaining remaining (by omega) state| <=
103       valueEnvelope remaining
104   transitionRadius_cover : forall (remaining : Nat)
105     (hremaining : remaining + 1 <= mdp.horizon)
106     (state : State) (action : Action),
107     (∑ nextState,
108         transitionCoordinateRadius
109             (mdp.decisionStageRemaining remaining hremaining)
110             state action nextState *
111           valueEnvelope remaining) <=
112       plan.transitionRadius
113         (mdp.decisionStageRemaining remaining hremaining) state action

115 /-- Coordinate transition confidence implies the recursive Bellman error bound. -/
116 theorem CoordinateConfidence.transitionError_le_radius
117     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
118     (confidence : plan.CoordinateConfidence) (remaining : Nat)
119     (hremaining : remaining + 1 <= mdp.horizon)
120     (state : State) (action : Action) :
121     |plan.transitionValue (mdp.decisionStageRemaining remaining hremaining)
122           (plan.upperValueRemaining remaining (by omega)) state action -
123         mdp.transitionValue
124           (plan.upperValueRemaining remaining (by omega)) state action| <=
125       plan.transitionRadius
126         (mdp.decisionStageRemaining remaining hremaining) state action := by
127   unfold transitionValue MDP.transitionValue
128   exact
129     (abs_integral_sub_integral_le_sum_coordinateRadius_mul_envelope
130       (plan.estimatedTransition
131         (mdp.decisionStageRemaining remaining hremaining) (state, action))
132       (mdp.transition (state, action))
133       (plan.upperValueRemaining remaining (by omega))
134       (confidence.transitionCoordinateRadius
135         (mdp.decisionStageRemaining remaining hremaining) state action)
136       (confidence.valueEnvelope remaining)
137       (confidence.transitionCoordinateError_le_radius
138         remaining hremaining state action)
139       (confidence.upperValue_abs_le_envelope remaining hremaining)).trans
140         (confidence.transitionRadius_cover
141           remaining hremaining state action)

143 /-- Package coordinate confidence as the existing estimated-model confidence. -/
144 def CoordinateConfidence.toConfidence
145     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
146     (confidence : plan.CoordinateConfidence) : plan.Confidence where
147   rewardError_le_radius := confidence.rewardError_le_radius
148   transitionError_le_radius := confidence.transitionError_le_radius

150 /--
151 Route endpoint: coordinate model confidence gives global optimism and the
152 compiled selected-radius single-episode expected-regret bound.
153 -/
154 theorem CoordinateConfidence.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
155     {mdp : MDP State Action} {plan : EstimatedModelPlan mdp}
156     (confidence : plan.CoordinateConfidence) (initialState : Measure State)
157     [IsProbabilityMeasure initialState] :
158     (forall state,
159       mdp.optimalValueRemaining mdp.horizon le_rfl state <=
160         plan.upperValueRemaining mdp.horizon le_rfl state) /\
161       plan.optimisticPolicy.expectedRegret initialState <=
162         plan.optimisticPolicy.occupancySumRemaining
163           (fun remaining hremaining state =>
164             2 * plan.selectedRadiusRemaining remaining hremaining state)
165           mdp.horizon le_rfl initialState :=
166   confidence.toConfidence
167     |>.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
168       initialState
