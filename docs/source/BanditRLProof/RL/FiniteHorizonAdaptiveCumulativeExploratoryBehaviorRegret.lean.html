4 /-!
5 # Exploratory behavior regret transport
6 
7 This module transports cumulative expected-regret certificates from deterministic
8 recommended policies to the exploratory behavior policies actually used by the
9 adaptive source.  The explicit charge is linear in the exploration rate and
10 quadratic in the horizon.  For a fixed exploration rate, the vanishing statistical
11 certificate therefore converges to that charge, not to zero.
12 -/

43 theorem integral_exploratoryActionPMF
44     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
45     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
46     (stage : Fin mdp.horizon) (state : State) (f : Action -> Real) :
47     (∫ action, f action ∂
48         (table.exploratoryActionPMF explorationRate hexplorationRate
49           stage state).toMeasure) =
50       (explorationRate : Real) *
51           (∑ action, f action) / (Fintype.card Action : Real) +
52         (1 - (explorationRate : Real)) * f (table stage state) := by
53   rw [PMF.integral_eq_sum]
54   calc
55     (∑ x,
56         (table.exploratoryActionPMF explorationRate hexplorationRate
57           stage state x).toReal • f x) =
58         ∑ x,
59           (((explorationRate : ENNReal) *
60                 (Fintype.card Action : ENNReal)⁻¹).toReal +
61             (((1 - explorationRate : NNReal) : ENNReal) *
62                 if x = table stage state then 1 else 0).toReal) * f x := by
63       apply Finset.sum_congr rfl
64       intro x _hx
65       rw [show table.exploratoryActionPMF explorationRate hexplorationRate
66           stage state x =
67           (explorationRate : ENNReal) *
68               (Fintype.card Action : ENNReal)⁻¹ +
69             (((1 - explorationRate : NNReal) : ENNReal) *
70               if x = table stage state then 1 else 0) by
71         rw [exploratoryActionPMF, PMF.bind_apply]
72         rw [tsum_fintype, Fintype.sum_bool]
73         simp [PMF.bernoulli_apply, PMF.uniformOfFintype_apply, PMF.pure_apply]]
74       rw [ENNReal.toReal_add] <;> try finiteness
75       rfl
76     _ = (explorationRate : Real) *
77           (∑ action, f action) / (Fintype.card Action : Real) +
78         (1 - (explorationRate : Real)) * f (table stage state) := by
79       simp only [ENNReal.toReal_mul, ENNReal.coe_toReal, ENNReal.toReal_inv,
80         ENNReal.toReal_natCast, add_mul]
81       rw [Finset.sum_add_distrib]
82       simp_rw [mul_assoc]
83       rw [← Finset.mul_sum, ← Finset.mul_sum]
84       norm_cast
85       simp only [apply_ite, Nat.cast_one, Nat.cast_zero, ite_mul,
86         mul_zero, one_mul, zero_mul]
87       rw [show (∑ x : Action,
88             if x = table stage state then
89               ((1 - explorationRate : NNReal) : Real) * f x else 0) =
90           ((1 - explorationRate : NNReal) : Real) * f (table stage state) by
91         simpa using
92           (Finset.sum_ite_eq' (Finset.univ : Finset Action)
93             (table stage state)
94             (fun x => ((1 - explorationRate : NNReal) : Real) * f x))]
95       rw [NNReal.coe_sub hexplorationRate]
96       norm_num
97       ring

99 theorem selected_sub_integral_exploratoryActionPMF_le
100     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
101     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
102     (stage : Fin mdp.horizon) (state : State) (f : Action -> Real)
103     (bound : Real) (hbound : forall action, |f action| <= bound) :
104     f (table stage state) -
105         (∫ action, f action ∂
106           (table.exploratoryActionPMF explorationRate hexplorationRate
107             stage state).toMeasure) <=
108       2 * (explorationRate : Real) * bound := by
109   rw [table.integral_exploratoryActionPMF explorationRate hexplorationRate
110     stage state f]
111   have hcardPos : (0 : Real) < Fintype.card Action := by
112     exact_mod_cast Fintype.card_pos
113   have hsumLower :
114       (Fintype.card Action : Real) * (-bound) <= ∑ action, f action := by
115     calc
116       (Fintype.card Action : Real) * (-bound) = ∑ _action : Action, -bound := by
117         simp
118       _ <= ∑ action, f action := by
119         apply Finset.sum_le_sum
120         intro action _haction
121         exact (abs_le.mp (hbound action)).1
122   have haverageLower :
123       -bound <= (∑ action, f action) / (Fintype.card Action : Real) := by
124     rw [le_div_iff₀ hcardPos]
125     nlinarith
126   have hselectedUpper : f (table stage state) <= bound :=
127     (abs_le.mp (hbound (table stage state))).2
128   have hgap :
129       f (table stage state) -
130           (∑ action, f action) / (Fintype.card Action : Real) <=
131         2 * bound := by
132     linarith
133   have hp : 0 <= (explorationRate : Real) := by positivity
134   have hscaled := mul_le_mul_of_nonneg_left hgap hp
135   calc
136     f (table stage state) -
137           ((explorationRate : Real) *
138               (∑ action, f action) / (Fintype.card Action : Real) +
139             (1 - (explorationRate : Real)) * f (table stage state)) =
140         (explorationRate : Real) *
141           (f (table stage state) -
142             (∑ action, f action) / (Fintype.card Action : Real)) := by ring
143     _ <= (explorationRate : Real) * (2 * bound) := hscaled
144     _ = 2 * (explorationRate : Real) * bound := by ring

150 /-- A bounded continuation gives a bounded one-step action value. -/
151 theorem bellmanQ_abs_le
152     (mdp : MDP State Action) (rewardBound continuationBound : Real)
153     (hrewardBound : forall state action, |mdp.reward state action| <= rewardBound)
154     {value : State -> Real}
155     (hvalue : forall state, |value state| <= continuationBound)
156     (state : State) (action : Action) :
157     |mdp.bellmanQ value state action| <= rewardBound + continuationBound := by
158   have htransition :
159       |mdp.transitionValue value state action| <= continuationBound := by
160     rw [← Real.norm_eq_abs]
161     have hnorm := norm_integral_le_of_norm_le_const
162       (μ := mdp.transition (state, action)) (C := continuationBound)
163       (Filter.Eventually.of_forall fun nextState => by
164         rw [Real.norm_eq_abs]
165         exact hvalue nextState)
166     simpa [MDP.transitionValue] using hnorm
167   unfold bellmanQ
168   exact (abs_add_le _ _).trans
169     (add_le_add (hrewardBound state action) htransition)

171 /-- A pointwise continuation difference bound transports through one transition kernel. -/
172 theorem transitionValue_sub_le_const
173     (mdp : MDP State Action) {left right : State -> Real} {bound : Real}
174     (hbound : forall state, left state - right state <= bound)
175     (state : State) (action : Action) :
176     mdp.transitionValue left state action -
177         mdp.transitionValue right state action <= bound := by
178   have hleft : Integrable left (mdp.transition (state, action)) :=
179     integrable_of_fintype _ _ (measurable_of_finite _)
180   have hright : Integrable right (mdp.transition (state, action)) :=
181     integrable_of_fintype _ _ (measurable_of_finite _)
182   unfold transitionValue
183   rw [← integral_sub hleft hright]
184   calc
185     (∫ nextState, left nextState - right nextState ∂
186         mdp.transition (state, action)) <=
187         ∫ _nextState, bound ∂mdp.transition (state, action) := by
188       apply integral_mono
189       · exact hleft.sub hright
190       · exact integrable_const bound
191       · exact hbound
192     _ = bound := by simp

198 /-- Every bounded-reward policy value is bounded by remaining horizon times the reward bound. -/
199 theorem valueRemaining_abs_le
200     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
201     (rewardBound : Real)
202     (hrewardBound : forall state action, |mdp.reward state action| <= rewardBound)
203     (remaining : Nat) (hremaining : remaining <= mdp.horizon) (state : State) :
204     |policy.valueRemaining remaining hremaining state| <=
205       (remaining : Real) * rewardBound := by
206   induction remaining generalizing state with
207   | zero =>
208       simp [valueRemaining]
209   | succ remaining ih =>
210       rw [valueRemaining]
211       rw [← Real.norm_eq_abs]
212       have hnorm := norm_integral_le_of_norm_le_const
213         (μ := policy.actionKernel
214           ⟨mdp.horizon - (remaining + 1), by omega⟩ state)
215         (C := ((remaining + 1 : Nat) : Real) * rewardBound)
216         (Filter.Eventually.of_forall fun action => by
217           rw [Real.norm_eq_abs]
218           calc
219             |mdp.bellmanQ
220                 (policy.valueRemaining remaining (by omega)) state action| <=
221                 rewardBound + (remaining : Real) * rewardBound :=
222               mdp.bellmanQ_abs_le rewardBound ((remaining : Real) * rewardBound)
223                 hrewardBound (fun nextState => ih (by omega) nextState) state action
224             _ = ((remaining + 1 : Nat) : Real) * rewardBound := by
225               push_cast
226               ring)
227       simpa [MarkovPolicy.bellman] using hnorm
228 
229 end MarkovPolicy

265 /--
266 Exploration around a deterministic table loses at most the explicit quadratic-horizon charge.
267 The charge is linear in the exploration rate and reward bound.
268 -/
269 theorem toMarkovPolicy_valueRemaining_sub_exploratoryPolicy_valueRemaining_le
270     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
271     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
272     (rewardBound : Real)
273     (hrewardBound : forall state action, |mdp.reward state action| <= rewardBound)
274     (remaining : Nat) (hremaining : remaining <= mdp.horizon) (state : State) :
275     table.toMarkovPolicy.valueRemaining remaining hremaining state -
276         (table.exploratoryPolicy explorationRate hexplorationRate).valueRemaining
277           remaining hremaining state <=
278       (explorationRate : Real) * rewardBound * (remaining : Real) *
279         ((remaining + 1 : Nat) : Real) := by
280   induction remaining generalizing state with
281   | zero =>
282       simp [MarkovPolicy.valueRemaining]
283   | succ remaining ih =>
284       let stage : Fin mdp.horizon :=
285         ⟨mdp.horizon - (remaining + 1), by omega⟩
286       let deterministicTail :=
287         table.toMarkovPolicy.valueRemaining remaining (by omega)
288       let exploratoryTail :=
289         (table.exploratoryPolicy explorationRate hexplorationRate).valueRemaining
290           remaining (by omega)
291       have hpropagation :
292           table.toMarkovPolicy.bellman stage deterministicTail state -
293               table.toMarkovPolicy.bellman stage exploratoryTail state <=
294             (explorationRate : Real) * rewardBound * (remaining : Real) *
295               ((remaining + 1 : Nat) : Real) := by
296         rw [table.toMarkovPolicy_bellman, table.toMarkovPolicy_bellman]
297         unfold MDP.bellmanQ
298         have htransition := mdp.transitionValue_sub_le_const
299           (left := deterministicTail) (right := exploratoryTail)
300           (bound := (explorationRate : Real) * rewardBound * (remaining : Real) *
301             ((remaining + 1 : Nat) : Real))
302           (fun nextState => ih (by omega) nextState) state (table stage state)
303         linarith
304       have hactionValue (action : Action) :
305           |mdp.bellmanQ exploratoryTail state action| <=
306             ((remaining + 1 : Nat) : Real) * rewardBound := by
307         exact (mdp.bellmanQ_abs_le rewardBound
308           ((remaining : Real) * rewardBound) hrewardBound
309           (fun nextState =>
310             (table.exploratoryPolicy explorationRate hexplorationRate).valueRemaining_abs_le
311               rewardBound hrewardBound remaining (by omega) nextState)
312           state action).trans_eq (by
313             push_cast
314             ring)
315       have hexplorationStep :
316           table.toMarkovPolicy.bellman stage exploratoryTail state -
317               (table.exploratoryPolicy explorationRate hexplorationRate).bellman
318                 stage exploratoryTail state <=
319             2 * (explorationRate : Real) *
320               (((remaining + 1 : Nat) : Real) * rewardBound) := by
321         rw [table.toMarkovPolicy_bellman]
322         unfold MarkovPolicy.bellman exploratoryPolicy
323         change mdp.bellmanQ exploratoryTail state (table stage state) -
324             (∫ action, mdp.bellmanQ exploratoryTail state action ∂
325               (table.exploratoryActionPMF explorationRate hexplorationRate
326                 stage state).toMeasure) <= _
327         exact table.selected_sub_integral_exploratoryActionPMF_le
328           explorationRate hexplorationRate stage state
329           (fun action => mdp.bellmanQ exploratoryTail state action)
330           (((remaining + 1 : Nat) : Real) * rewardBound) hactionValue
331       rw [MarkovPolicy.valueRemaining, MarkovPolicy.valueRemaining]
332       change table.toMarkovPolicy.bellman stage deterministicTail state -
333           (table.exploratoryPolicy explorationRate hexplorationRate).bellman
334             stage exploratoryTail state <= _
335       calc
336         table.toMarkovPolicy.bellman stage deterministicTail state -
337               (table.exploratoryPolicy explorationRate hexplorationRate).bellman
338                 stage exploratoryTail state =
339             (table.toMarkovPolicy.bellman stage deterministicTail state -
340               table.toMarkovPolicy.bellman stage exploratoryTail state) +
341             (table.toMarkovPolicy.bellman stage exploratoryTail state -
342               (table.exploratoryPolicy explorationRate hexplorationRate).bellman
343                 stage exploratoryTail state) := by ring
344         _ <= (explorationRate : Real) * rewardBound * (remaining : Real) *
345               ((remaining + 1 : Nat) : Real) +
346             2 * (explorationRate : Real) *
347               (((remaining + 1 : Nat) : Real) * rewardBound) :=
348           add_le_add hpropagation hexplorationStep
349         _ = (explorationRate : Real) * rewardBound *
350               ((remaining + 1 : Nat) : Real) *
351               ((remaining + 2 : Nat) : Real) := by
352           push_cast
353           ring

355 /-- Exploratory-behavior expected regret is bounded by table regret plus its exploration charge. -/
356 theorem exploratoryPolicy_expectedRegret_le_toMarkovPolicy_expectedRegret_add_charge
357     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
358     (initialState : Measure State) [IsProbabilityMeasure initialState]
359     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1)
360     (rewardBound : Real)
361     (hrewardBound : forall state action, |mdp.reward state action| <= rewardBound) :
362     (table.exploratoryPolicy explorationRate hexplorationRate).expectedRegret
363         initialState <=
364       table.toMarkovPolicy.expectedRegret initialState +
365         (explorationRate : Real) * rewardBound * (mdp.horizon : Real) *
366           ((mdp.horizon + 1 : Nat) : Real) := by
367   let charge := (explorationRate : Real) * rewardBound *
368     (mdp.horizon : Real) * ((mdp.horizon + 1 : Nat) : Real)
369   have hvalueGap (state : State) :
370       table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state -
371           (table.exploratoryPolicy explorationRate hexplorationRate).valueAt
372             0 (Nat.zero_le mdp.horizon) state <= charge := by
373     simpa [MarkovPolicy.valueAt, charge] using
374       table.toMarkovPolicy_valueRemaining_sub_exploratoryPolicy_valueRemaining_le
375         explorationRate hexplorationRate rewardBound hrewardBound
376         mdp.horizon le_rfl state
377   rw [MarkovPolicy.expectedRegret_eq_integral_optimalValueAt_sub_valueAt
378     (table.exploratoryPolicy explorationRate hexplorationRate) initialState]
379   rw [MarkovPolicy.expectedRegret_eq_integral_optimalValueAt_sub_valueAt
380     table.toMarkovPolicy initialState]
381   have hoptimalSubTable : Integrable
382       (fun state =>
383         mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
384           table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state)
385       initialState :=
386     integrable_of_fintype _ _ (measurable_of_finite _)
387   have htableSubExploratory : Integrable
388       (fun state =>
389         table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state -
390           (table.exploratoryPolicy explorationRate hexplorationRate).valueAt
391             0 (Nat.zero_le mdp.horizon) state)
392       initialState :=
393     integrable_of_fintype _ _ (measurable_of_finite _)
394   calc
395     (∫ state,
396         mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
397           (table.exploratoryPolicy explorationRate hexplorationRate).valueAt
398             0 (Nat.zero_le mdp.horizon) state ∂initialState) =
399         (∫ state,
400           (mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
401             table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state) +
402           (table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state -
403             (table.exploratoryPolicy explorationRate hexplorationRate).valueAt
404               0 (Nat.zero_le mdp.horizon) state) ∂initialState) := by
405       apply integral_congr_ae
406       filter_upwards [] with state
407       ring
408     _ = (∫ state,
409           mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
410             table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state
411             ∂initialState) +
412         ∫ state,
413           table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state -
414             (table.exploratoryPolicy explorationRate hexplorationRate).valueAt
415               0 (Nat.zero_le mdp.horizon) state ∂initialState := by
416       rw [integral_add hoptimalSubTable htableSubExploratory]
417     _ <= (∫ state,
418           mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
419             table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state
420             ∂initialState) +
421         ∫ _state, charge ∂initialState := by
422       apply add_le_add (le_refl _)
423       apply integral_mono
424       · exact htableSubExploratory
425       · exact integrable_const charge
426       · exact hvalueGap
427     _ = (∫ state,
428           mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
429             table.toMarkovPolicy.valueAt 0 (Nat.zero_le mdp.horizon) state
430             ∂initialState) +
431         (explorationRate : Real) * rewardBound * (mdp.horizon : Real) *
432           ((mdp.horizon + 1 : Nat) : Real) := by
433       simp [charge]
