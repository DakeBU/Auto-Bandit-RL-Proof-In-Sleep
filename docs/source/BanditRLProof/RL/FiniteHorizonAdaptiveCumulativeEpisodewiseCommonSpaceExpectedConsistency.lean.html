5 /-!
6 # Common-space expected absolute realized consistency
7 
8 This module strengthens the independent-coordinate common-space convergence in
9 probability theorem to convergence of expected absolute realized-behavior
10 regret.  The additional input is an almost-everywhere deterministic envelope:
11 generated adaptive batches are reward-consistent, so every scheduled realized
12 average regret has absolute value at most twice the horizon.
13 
14 The common space remains the product coupling of complete finite-window
15 experiments.  Nothing here constructs a nested causal online stream across
16 schedules or proves pathwise, almost-sure, or anytime consistency.
17 -/

35 /-- A deterministic finite trajectory has return bounded by the horizon. -/
36 theorem abs_cumulativeReward_le_horizon
37     (mdp : MDP State Action)
38     (hrewardBound : forall state action, |mdp.reward state action| <= 1)
39     (trajectory : Prod State (StepTrace Action State mdp.horizon)) :
40     |mdp.cumulativeReward trajectory| <= (mdp.horizon : Real) := by
41   rw [cumulativeReward, cumulativeRewardFrom_eq_sum_traceReward]
42   calc
43     |∑ stage : Fin mdp.horizon,
44         mdp.reward
45           (traceStateAtFrom trajectory.1 trajectory.2 stage)
46           (trajectory.2 stage).1| <=
47         ∑ stage : Fin mdp.horizon,
48           |mdp.reward
49             (traceStateAtFrom trajectory.1 trajectory.2 stage)
50             (trajectory.2 stage).1| :=
51       Finset.abs_sum_le_sum_abs _ _
52     _ <= ∑ _stage : Fin mdp.horizon, (1 : Real) := by
53       apply Finset.sum_le_sum
54       intro stage _hstage
55       exact hrewardBound _ _
56     _ = (mdp.horizon : Real) := by simp

109 /-- Reward-consistent successor batches give a uniform realized-average envelope. -/
110 theorem abs_realizedSuccessorAverageRegret_le_two_mul_horizon
111     {mdp : MDP State Action} {initialState : Measure State}
112     [IsProbabilityMeasure initialState] {episodes : Nat}
113     (source : AdaptiveEpisodeBatchSource mdp initialState episodes)
114     (trajectory : EpisodeBatchTrajectory mdp episodes)
115     (rounds : Nat) (hrounds : 0 < rounds) (hepisodes : 0 < episodes)
116     (hrewardBound : forall state action, |mdp.reward state action| <= 1)
117     (hconsistent : forall round : Fin rounds,
118       (trajectory ((round : Nat) + 1)).RewardConsistent) :
119     |source.realizedSuccessorAverageRegret trajectory rounds| <=
120       2 * (mdp.horizon : Real) := by
121   have hoptimal := abs_optimalInitialExpectedReturn_le_horizon
122     mdp initialState hrewardBound
123   have hepisodes_nonneg : 0 <= (episodes : Real) := by positivity
124   have hterm : forall round : Fin rounds,
125       |(episodes : Real) * optimalInitialExpectedReturn mdp initialState -
126           EpisodeBatch.totalReturn (trajectory ((round : Nat) + 1))| <=
127         2 * (episodes : Real) * (mdp.horizon : Real) := by
128     intro round
129     calc
130       |(episodes : Real) * optimalInitialExpectedReturn mdp initialState -
131           EpisodeBatch.totalReturn (trajectory ((round : Nat) + 1))| <=
132           |(episodes : Real) * optimalInitialExpectedReturn mdp initialState| +
133             |EpisodeBatch.totalReturn
134               (trajectory ((round : Nat) + 1))| := abs_sub _ _
135       _ = (episodes : Real) *
136             |optimalInitialExpectedReturn mdp initialState| +
137             |EpisodeBatch.totalReturn
138               (trajectory ((round : Nat) + 1))| := by
139           rw [abs_mul, abs_of_nonneg hepisodes_nonneg]
140       _ <= (episodes : Real) * (mdp.horizon : Real) +
141             (episodes : Real) * (mdp.horizon : Real) := by
142           exact add_le_add
143             (mul_le_mul_of_nonneg_left hoptimal hepisodes_nonneg)
144             (EpisodeBatch.abs_totalReturn_le_of_rewardConsistent
145               _ (hconsistent round) hrewardBound)
146       _ = 2 * (episodes : Real) * (mdp.horizon : Real) := by ring
147   have hcumulative :
148       |source.realizedSuccessorCumulativeRegret trajectory rounds| <=
149         ∑ _round : Fin rounds,
150           2 * (episodes : Real) * (mdp.horizon : Real) := by
151     unfold realizedSuccessorCumulativeRegret
152     exact (Finset.abs_sum_le_sum_abs _ _).trans
153       (Finset.sum_le_sum fun round _ => hterm round)
154   have hdenom : 0 < (episodes : Real) * (rounds : Real) := by positivity
155   rw [realizedSuccessorAverageRegret, abs_div, abs_of_pos hdenom]
156   calc
157     |source.realizedSuccessorCumulativeRegret trajectory rounds| /
158         ((episodes : Real) * (rounds : Real)) <=
159       (∑ _round : Fin rounds,
160           2 * (episodes : Real) * (mdp.horizon : Real)) /
161         ((episodes : Real) * (rounds : Real)) :=
162       div_le_div_of_nonneg_right hcumulative hdenom.le
163     _ = 2 * (mdp.horizon : Real) := by
164       simp
165       field_simp
