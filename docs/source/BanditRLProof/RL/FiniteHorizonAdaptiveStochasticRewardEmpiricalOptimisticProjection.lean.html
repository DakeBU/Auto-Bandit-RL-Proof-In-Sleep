6 /-!
7 # Adaptive stochastic-reward empirical-optimistic projection
8 
9 This module lifts the existing known-reward exploratory empirical-transition
10 source to stochastic rewards.  Its policy selector reads only the complete
11 known-reward projection of the observed stochastic prefix.  The resulting
12 adaptive stochastic trajectory therefore maps exactly to the deterministic
13 source trajectory, so the compiled count-confidence, optimism, and recommended
14 expected-regret terminal can be pulled back without estimating sampled rewards.
15 
16 The projection preserves every action and next-state coordinate and reinstates
17 only the deterministic mean reward `mdp.reward`.  No realized behavior-regret
18 or stochastic reward-confidence claim is made here.
19 -/

31 /--
32 Map both coordinates of a measure composition product when the base measure
33 and every kernel fiber have the prescribed mapped laws.
34 -/
35 theorem measure_compProd_map_prodMap_of_map_eq
36     {Alpha Alpha' Beta Beta' : Type*}
37     [MeasurableSpace Alpha] [MeasurableSpace Alpha']
38     [MeasurableSpace Beta] [MeasurableSpace Beta']
39     (mu : Measure Alpha) [IsProbabilityMeasure mu]
40     (kappa : ProbabilityTheory.Kernel Alpha Beta)
41       [ProbabilityTheory.IsMarkovKernel kappa]
42     (mu' : Measure Alpha') [IsProbabilityMeasure mu']
43     (kappa' : ProbabilityTheory.Kernel Alpha' Beta')
44       [ProbabilityTheory.IsMarkovKernel kappa']
45     (f : Alpha -> Alpha') (hf : Measurable f)
46     (g : Beta -> Beta') (hg : Measurable g)
47     (hmu : mu.map f = mu')
48     (hkappa : forall alpha, (kappa alpha).map g = kappa' (f alpha)) :
49     (mu ⊗ₘ kappa).map (Prod.map f g) = mu' ⊗ₘ kappa' := by
50   apply Measure.ext_prod
51   intro alphaSet betaSet halphaSet hbetaSet
52   have hprodMap : Measurable (Prod.map f g) := hf.prodMap hg
53   rw [Measure.map_apply hprodMap (halphaSet.prod hbetaSet)]
54   have hpreimage :
55       Prod.map f g ⁻¹' (alphaSet ×ˢ betaSet) =
56         (f ⁻¹' alphaSet) ×ˢ (g ⁻¹' betaSet) := by
57     ext pair
58     simp
59   rw [hpreimage, Measure.compProd_apply_prod (halphaSet.preimage hf)
60     (hbetaSet.preimage hg)]
61   rw [Measure.compProd_apply_prod halphaSet hbetaSet]
62   have hfiber (alpha : Alpha) :
63       kappa alpha (g ⁻¹' betaSet) = kappa' (f alpha) betaSet := by
64     have h := congrArg (fun nu : Measure Beta' => nu betaSet) (hkappa alpha)
65     simpa [Measure.map_apply hg hbetaSet] using h
66   simp_rw [hfiber]
67   let q : Alpha' -> ENNReal := fun alpha => kappa' alpha betaSet
68   have hq : Measurable q := kappa'.measurable_coe hbetaSet
69   change
70     (∫⁻ alpha in f ⁻¹' alphaSet, q (f alpha) ∂mu) =
71       ∫⁻ alpha in alphaSet, q alpha ∂mu'
72   rw [← MeasureTheory.setLIntegral_map halphaSet hq hf]
73   rw [hmu]
74 
75 end ProbabilityTheory
