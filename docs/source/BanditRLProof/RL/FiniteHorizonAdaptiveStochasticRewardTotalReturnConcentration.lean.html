5 /-!
6 # Adaptive stochastic-reward episode-batch concentration
7 
8 This module generates complete reward-bearing episode batches with a policy
9 selected from the preceding batch history.  It retains that history while
10 mapping the next-batch kernel, identifies the resulting dynamic sampled-return
11 law through `condDistrib` and `condExpKernel`, and applies the conditional
12 sub-Gaussian sum theorem across a fixed finite number of adaptive rounds.
13 
14 No independence is assumed across rounds.  Independence is used only inside
15 each conditionally iid episode batch.  The result is a sampled-return
16 deviation bound, not a regret, optimism, or anytime theorem.
17 -/

75           (fun pair => (pair.1, pair)) := by
76             symm
77             simpa [Function.comp_def] using
78               (Measure.map_map
79                 (μ := mu)

81                 (f := fun sample => (condition sample, next sample))
82                 (measurable_fst.prodMk measurable_id)
83                 (hcondition.prodMk hnext))
84     _ = ((mu.map condition).compProd kernel).map
85           (fun pair => (pair.1, pair)) := by rw [hpair]
86     _ = (mu.map condition).compProd (retainedInputKernel kernel) := by
87       ext event hevent

92         (s := (fun pair : Input × Output => (pair.1, pair)) ⁻¹' event)

182 /--
183 An adaptive source of complete stochastic-reward episode batches.
184 
185 The dynamic measurability field is the only additional policy-selection
186 regularity needed by this route.  The exact batch-kernel equality supplies the
187 pointwise conditionally iid law selected by each observed prefix.
188 -/
189 structure AdaptiveStochasticEpisodeBatchSource
190     (mdp : MDP State Action) (initialState : Measure State)
191     [IsProbabilityMeasure initialState] (episodes : Nat) where
192   rewardSource : mdp.MeanCompatibleRewardKernel
193   initialPolicy : MarkovPolicy mdp
194   successorPolicy : (n : Nat) →
195     StochasticEpisodeBatchPrefix mdp episodes n → MarkovPolicy mdp
196   batchKernel : (n : Nat) →
197     ProbabilityTheory.Kernel
198       (StochasticEpisodeBatchPrefix mdp episodes n)
199       (StochasticEpisodeBatch mdp episodes)
200   batchKernel_isMarkov : ∀ n,
201     ProbabilityTheory.IsMarkovKernel (batchKernel n)
202   batchKernel_eq_iidStochasticTrajectoryFamilyMeasure : ∀ n history,
203     batchKernel n history =
204       rewardSource.iidStochasticTrajectoryFamilyMeasure
205         (successorPolicy n history) initialState episodes
206   measurable_successorSampledReturnDeviation : ∀ n,
207     Measurable fun pair :
208         StochasticEpisodeBatchPrefix mdp episodes n ×
209           StochasticEpisodeBatch mdp episodes =>
210       mdp.sampledCumulativeReturnDeviationSum
211         (successorPolicy n pair.1) episodes pair.2
