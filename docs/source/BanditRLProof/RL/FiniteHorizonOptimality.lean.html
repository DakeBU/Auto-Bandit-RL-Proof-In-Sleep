4 # Finite-horizon Bellman optimality
5 
6 This module adds finite-action maximization to the compiled policy-evaluation
7 route.  On a finite discrete state space, the pointwise finite argmax is a
8 measurable deterministic selector.  The resulting greedy Markov policy attains
9 the backward optimal value, which dominates every Markov policy value.
10 -/

26 /-- A finite action maximizing the one-step Bellman action value. -/
27 noncomputable def optimalAction (mdp : MDP State Action)
28     (value : State -> Real) (state : State) : Action :=
29   Classical.choose
30     (Finite.exists_max (fun action => mdp.bellmanQ value state action))

32 omit [MeasurableSingletonClass State] in
33 /-- The selected finite action dominates every Bellman action value. -/
34 theorem bellmanQ_le_optimalAction
35     (mdp : MDP State Action) (value : State -> Real)
36     (state : State) (action : Action) :
37     mdp.bellmanQ value state action <=
38       mdp.bellmanQ value state (mdp.optimalAction value state) := by
39   exact Classical.choose_spec
40     (Finite.exists_max (fun action => mdp.bellmanQ value state action)) action

91 /-- Backward optimal value indexed by the number of decisions remaining. -/
92 noncomputable def optimalValueRemaining (mdp : MDP State Action) :
93     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
94   | 0, _ => fun _ => 0
95   | remaining + 1, hremaining =>
96       mdp.optimalBellman
97         (mdp.optimalValueRemaining remaining (by omega))

130 /-- The optimal value is zero at the terminal stage. -/
131 @[simp]
132 theorem optimalValueAt_horizon (mdp : MDP State Action) :
133     mdp.optimalValueAt mdp.horizon le_rfl = fun _ => 0 := by
134   unfold optimalValueAt
135   calc
136     mdp.optimalValueRemaining (mdp.horizon - mdp.horizon) _ =

143 /-- Finite-action Bellman recursion for the chronological optimal value. -/
144 theorem optimalValueAt_bellman
145     (mdp : MDP State Action) (stage : Nat)
146     (hstage : stage < mdp.horizon) :
147     mdp.optimalValueAt stage (Nat.le_of_lt hstage) =
148       mdp.optimalBellman
149         (mdp.optimalValueAt (stage + 1) (by omega)) := by
150   have hremaining : mdp.horizon - stage =
151       (mdp.horizon - (stage + 1)) + 1 := by
152     omega
153   calc
154     mdp.optimalValueAt stage (Nat.le_of_lt hstage) =
155         mdp.optimalValueRemaining
156           ((mdp.horizon - (stage + 1)) + 1) (by omega) := by
157       apply mdp.optimalValueRemaining_eq_of_eq
158       exact hremaining
159     _ = mdp.optimalBellman
160           (mdp.optimalValueRemaining
161             (mdp.horizon - (stage + 1)) (by omega)) := by
162       rw [optimalValueRemaining]

171 /-- Policy Bellman expectation is monotone in the continuation value. -/
172 theorem bellman_mono
173     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
174     (stage : Fin mdp.horizon) {left right : State -> Real}
175     (hle : forall state, left state <= right state) (state : State) :
176     policy.bellman stage left state <= policy.bellman stage right state := by
177   unfold bellman
178   have hleft : Measurable
179       (fun action => mdp.bellmanQ left state action) :=
180     (mdp.measurable_bellmanQ (measurable_of_finite left)).comp
181       (measurable_const.prodMk measurable_id)
182   have hright : Measurable
183       (fun action => mdp.bellmanQ right state action) :=
184     (mdp.measurable_bellmanQ (measurable_of_finite right)).comp
185       (measurable_const.prodMk measurable_id)
186   apply integral_mono
187   · exact integrable_of_fintype _ _ hleft
188   · exact integrable_of_fintype _ _ hright
189   · exact fun action => mdp.bellmanQ_mono hle state action

191 /-- Every policy Bellman expectation is bounded by the finite-action maximum. -/
192 theorem bellman_le_optimalBellman
193     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
194     (stage : Fin mdp.horizon) (value : State -> Real) (state : State) :
195     policy.bellman stage value state <= mdp.optimalBellman value state := by
196   unfold bellman
197   have hvalue : Measurable
198       (fun action => mdp.bellmanQ value state action) :=
199     (mdp.measurable_bellmanQ (measurable_of_finite value)).comp
200       (measurable_const.prodMk measurable_id)
201   calc
202     (∫ action, mdp.bellmanQ value state action
203         ∂policy.actionKernel stage state) <=
204         ∫ _action, mdp.optimalBellman value state
205           ∂policy.actionKernel stage state := by
206       apply integral_mono
207       · exact integrable_of_fintype _ _ hvalue
208       · exact integrable_const _
209       · exact fun action => mdp.bellmanQ_le_optimalBellman value state action
210     _ = mdp.optimalBellman value state := by simp

212 /-- Every Markov policy value is pointwise bounded by the optimal value. -/
213 theorem valueAt_le_optimalValueAt
214     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
215     (stage : Nat) (hstage : stage <= mdp.horizon) (state : State) :
216     policy.valueAt stage hstage state <=
217       mdp.optimalValueAt stage hstage state := by
218   exact Nat.decreasingInduction
219     (n := mdp.horizon)
220     (motive := fun stage hstage => forall state,
221       policy.valueAt stage hstage state <=
222         mdp.optimalValueAt stage hstage state)
223     (fun current hcurrent ih state => by
224       rw [policy.valueAt_bellman current hcurrent]
225       rw [mdp.optimalValueAt_bellman current hcurrent]
226       exact
227         (policy.bellman_mono ⟨current, hcurrent⟩
228           (fun nextState => ih nextState) state).trans
229         (policy.bellman_le_optimalBellman ⟨current, hcurrent⟩
230           (mdp.optimalValueAt (current + 1) (by omega)) state))
231     (by
232       intro state
233       rw [policy.valueAt_horizon, mdp.optimalValueAt_horizon])
234     hstage state
235 
236 end MarkovPolicy

240 /-- Greedy deterministic Markov policy for the backward optimal value. -/
241 noncomputable def optimalPolicy (mdp : MDP State Action) : MarkovPolicy mdp where
242   actionKernel stage :=
243     ProbabilityTheory.Kernel.deterministic
244       (mdp.optimalAction
245         (mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt)))
246       (mdp.measurable_optimalAction _)
247   actionKernel_isMarkov := by
248     intro stage
249     infer_instance

251 /-- One greedy-policy Bellman step is exactly the finite-action maximum. -/
252 theorem optimalPolicy_bellman_eq_optimalBellman
253     (mdp : MDP State Action) (stage : Fin mdp.horizon) (state : State) :
254     mdp.optimalPolicy.bellman stage
255         (mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt)) state =
256       mdp.optimalBellman
257         (mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt)) state := by
258   unfold MarkovPolicy.bellman optimalPolicy
259   rw [ProbabilityTheory.Kernel.deterministic_apply]
260   rw [integral_dirac']
261   · rfl
262   · exact
263       ((mdp.measurable_bellmanQ
264           (mdp.measurable_optimalValueAt (stage + 1)
265             (Nat.succ_le_of_lt stage.isLt))).comp
266         (measurable_const.prodMk measurable_id)).stronglyMeasurable

268 /-- The measurable greedy deterministic policy attains the optimal value at every stage. -/
269 theorem optimalPolicy_valueAt_eq_optimalValueAt
270     (mdp : MDP State Action) (stage : Nat)
271     (hstage : stage <= mdp.horizon) :
272     mdp.optimalPolicy.valueAt stage hstage =
273       mdp.optimalValueAt stage hstage := by
274   exact Nat.decreasingInduction
275     (n := mdp.horizon)
276     (motive := fun stage hstage =>
277       mdp.optimalPolicy.valueAt stage hstage =
278         mdp.optimalValueAt stage hstage)
279     (fun current hcurrent ih => by
280       change mdp.optimalPolicy.valueAt current _ =
281         mdp.optimalValueAt current _
282       rw [mdp.optimalPolicy.valueAt_bellman current hcurrent]
283       rw [mdp.optimalValueAt_bellman current hcurrent]
284       rw [ih]
285       funext state
286       exact mdp.optimalPolicy_bellman_eq_optimalBellman
287         ⟨current, hcurrent⟩ state)
288     (by
289       change mdp.optimalPolicy.valueAt mdp.horizon _ =
290         mdp.optimalValueAt mdp.horizon _
291       rw [mdp.optimalPolicy.valueAt_horizon, mdp.optimalValueAt_horizon])
292     hstage

294 /--
295 Route endpoint: the backward Bellman value dominates every Markov policy and
296 is attained by the measurable greedy deterministic policy.
297 -/
298 theorem optimalValueAt_dominates_and_is_attained
299     (mdp : MDP State Action) :
300     (forall (policy : MarkovPolicy mdp) (stage : Nat)
301       (hstage : stage <= mdp.horizon) (state : State),
302         policy.valueAt stage hstage state <=
303           mdp.optimalValueAt stage hstage state) /\
304       (exists policy : MarkovPolicy mdp,
305         forall (stage : Nat) (hstage : stage <= mdp.horizon),
306           policy.valueAt stage hstage = mdp.optimalValueAt stage hstage) := by
307   constructor
308   · intro policy stage hstage state
309     exact policy.valueAt_le_optimalValueAt stage hstage state
310   · exact ⟨mdp.optimalPolicy, mdp.optimalPolicy_valueAt_eq_optimalValueAt⟩
