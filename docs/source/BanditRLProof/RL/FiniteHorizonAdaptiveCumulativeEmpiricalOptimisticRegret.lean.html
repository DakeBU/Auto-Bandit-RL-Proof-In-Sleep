3 /-!
4 # Adaptive cumulative empirical optimistic regret contract
5 
6 This module changes the adaptive empirical planner from a latest-batch summary
7 to the sum of every transition count in the observed finite prefix.  A
8 nonnegative antitone count-radius object makes the plan's transition radius a
9 function of the cumulative state-action visit count.  The cumulative selector
10 is measurable, so it also defines a concrete exploratory adaptive batch source.
11 
12 The route terminates at optimism and recommended-policy expected regret under
13 one explicit global cumulative coordinate-confidence contract.  Producing that
14 contract from adaptive cumulative count martingales remains downstream; no
15 such concentration theorem is assumed to have been proved here.
16 -/

40 /-- A finite linear-decay radius, useful as an executable shrinking-radius canary. -/
41 def linearDecay (budget : Nat) : TransitionCountRadius where
42   radius count := (budget - count : Nat)
43   nonneg count := by positivity
44   antitone := by
45     intro left right hle
46     change ((budget - right : Nat) : Real) <= (budget - left : Nat)
47     exact_mod_cast Nat.sub_le_sub_left hle budget

193 /-- Every count-radius optimistic value is controlled by the radius at zero visits. -/
194 theorem countRadiusOptimisticPlan_upperValueRemaining_abs_le
195     (mdp : MDP State Action) (summary : TransitionCountSummary mdp)
196     (defaultState : State) (countRadius : TransitionCountRadius)
197     (rewardBound : Real)
198     (hrewardBound : forall state action, |mdp.reward state action| <= rewardBound) :
199     forall (remaining : Nat) (hremaining : remaining <= mdp.horizon)
200       (state : State),
201       |(summary.countRadiusOptimisticPlan mdp defaultState countRadius).upperValueRemaining
202           remaining hremaining state| <=
203         empiricalFiniteBatchValueEnvelope
204           rewardBound (countRadius.radius 0) remaining := by
205   let plan := summary.countRadiusOptimisticPlan mdp defaultState countRadius
206   intro remaining
207   induction remaining with
208   | zero =>
209       intro hremaining state
210       simp [MDP.EstimatedModelPlan.upperValueRemaining,
211         empiricalFiniteBatchValueEnvelope]
212   | succ remaining ih =>
213       intro hremaining state
214       let stage := mdp.decisionStageRemaining remaining hremaining
215       let tail := plan.upperValueRemaining remaining (by omega)
216       let action := plan.optimisticAction stage tail state
217       have htail : forall nextState,
218           |tail nextState| <=
219             empiricalFiniteBatchValueEnvelope
220               rewardBound (countRadius.radius 0) remaining := by
221         intro nextState
222         exact ih (by omega) nextState

224           |plan.transitionValue stage tail state action| <=
225             empiricalFiniteBatchValueEnvelope
226               rewardBound (countRadius.radius 0) remaining := by
227         rw [← Real.norm_eq_abs]
228         have hnorm := norm_integral_le_of_norm_le_const
229           (μ := plan.estimatedTransition stage (state, action))
230           (C := empiricalFiniteBatchValueEnvelope
231             rewardBound (countRadius.radius 0) remaining)
232           (Filter.Eventually.of_forall fun nextState => by
233             rw [Real.norm_eq_abs]
234             exact htail nextState)
235         simpa [MDP.EstimatedModelPlan.transitionValue] using hnorm
236       have hradius :
237           countRadius.radius (summary.visitCount stage state action) <=
238             countRadius.radius 0 :=
239         countRadius.antitone (Nat.zero_le _)
240       rw [MDP.EstimatedModelPlan.upperValueRemaining]
241       change
242         |mdp.reward state action +
243               plan.transitionValue stage tail state action +
244             0 + countRadius.radius (summary.visitCount stage state action)| <=
245           empiricalFiniteBatchValueEnvelope
246             rewardBound (countRadius.radius 0) (remaining + 1)
247       calc
248         |mdp.reward state action +
249               plan.transitionValue stage tail state action +
250             0 + countRadius.radius (summary.visitCount stage state action)| =
251             |(mdp.reward state action +
252               plan.transitionValue stage tail state action) +
253                 countRadius.radius (summary.visitCount stage state action)| := by
254               ring_nf
255         _ <= |mdp.reward state action +
256               plan.transitionValue stage tail state action| +
257                 |countRadius.radius
258                   (summary.visitCount stage state action)| := abs_add_le _ _
259         _ <= (|mdp.reward state action| +
260               |plan.transitionValue stage tail state action|) +
261                 |countRadius.radius
262                   (summary.visitCount stage state action)| := by
263             have hsum := abs_add_le (mdp.reward state action)
264               (plan.transitionValue stage tail state action)
265             linarith
266         _ <= rewardBound +
267               empiricalFiniteBatchValueEnvelope
268                 rewardBound (countRadius.radius 0) remaining +
269               countRadius.radius 0 := by
270             rw [abs_of_nonneg (countRadius.nonneg _)]
271             linarith [hrewardBound state action, htransition, hradius]
272         _ = empiricalFiniteBatchValueEnvelope
273               rewardBound (countRadius.radius 0) (remaining + 1) := by
274             unfold empiricalFiniteBatchValueEnvelope

276               (remaining : Real) + 1 by norm_num]

423 Roundwise cumulative confidence and a selected-radius envelope imply global
424 optimism and the explicit finite recommendation-regret sum.
425 -/
426 theorem adaptiveCumulativeCoordinateConfidence_optimism_and_recommendedExpectedRegret
427     {mdp : MDP State Action} {initialState : Measure State}
428     [IsProbabilityMeasure initialState] {episodes rounds : Nat}
429     (trajectory : EpisodeBatchTrajectory mdp episodes)
430     (defaultState : State) (countRadius : TransitionCountRadius)
431     (radiusEnvelope : Fin rounds -> Real)
432     (confidence : forall round : Fin rounds,
433       (adaptiveCumulativeEmpiricalOptimisticPlanAt
434         trajectory defaultState countRadius round).CoordinateConfidence)
435     (hradius : forall (round : Fin rounds) (remaining : Nat)
436       (hremaining : remaining + 1 <= mdp.horizon) (state : State),
437       (adaptiveCumulativeEmpiricalOptimisticPlanAt
438         trajectory defaultState countRadius round).selectedRadiusRemaining
439           remaining hremaining state <= radiusEnvelope round) :
440     (forall round : Fin rounds, forall state,
441       mdp.optimalValueRemaining mdp.horizon le_rfl state <=
442         (adaptiveCumulativeEmpiricalOptimisticPlanAt
443           trajectory defaultState countRadius round).upperValueRemaining
444             mdp.horizon le_rfl state) /\
445     adaptiveCumulativeEmpiricalOptimisticRecommendedExpectedRegret
446         (initialState := initialState) trajectory defaultState countRadius rounds <=
447       ∑ round : Fin rounds,
448         (mdp.horizon : Real) * (2 * radiusEnvelope round) := by
449   constructor
450   · intro round state
451     exact
452       ((confidence round).optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
453         initialState).1 state
454   · unfold adaptiveCumulativeEmpiricalOptimisticRecommendedExpectedRegret
455     apply Finset.sum_le_sum
456     intro round _hround
457     let plan := adaptiveCumulativeEmpiricalOptimisticPlanAt
458       trajectory defaultState countRadius round
459     calc
460       plan.optimisticPolicy.expectedRegret initialState <=
461           plan.optimisticPolicy.occupancySumRemaining
462             (fun remaining hremaining state =>
463               2 * plan.selectedRadiusRemaining remaining hremaining state)
464             mdp.horizon le_rfl initialState :=
465         ((confidence round).optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
466           initialState).2
467       _ <= plan.optimisticPolicy.occupancySumRemaining
468             (fun _remaining _hremaining _state => 2 * radiusEnvelope round)
469             mdp.horizon le_rfl initialState := by
470         exact plan.optimisticPolicy.occupancySumRemaining_mono
471           (fun remaining hremaining state =>
472             mul_le_mul_of_nonneg_left
473               (hradius round remaining hremaining state) (by norm_num))
474           mdp.horizon le_rfl initialState
475       _ = (mdp.horizon : Real) * (2 * radiusEnvelope round) :=
476         MarkovPolicy.occupancySumRemaining_const
477           plan.optimisticPolicy (2 * radiusEnvelope round)
478             mdp.horizon le_rfl initialState
