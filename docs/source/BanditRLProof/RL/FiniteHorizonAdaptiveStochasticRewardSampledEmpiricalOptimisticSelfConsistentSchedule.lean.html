6 /-!
7 # Scheduled self-consistent realized regret for actual sampled optimism
8 
9 This module gives the self-consistent actual-sampled route an explicit
10 horizon-indexed batch schedule.  It reuses the decaying exploration, round,
11 visit-floor, and confidence schedules, and chooses one `ceil + 1` episode count
12 above three explicit thresholds: the existing count-calibration threshold, a
13 shrinking count-ratio threshold, and a shrinking sampled-reward threshold.
14 -/

51 /-- The count threshold gives a scale-squared normalized count-radius bound. -/
52 theorem simultaneousCountConfidenceRadius_lt_episodes_mul_visitFloor_div_scale_sq_of_threshold
53     (mdp : MDP State Action) {rounds episodes : Nat}
54     {delta visitFloor scale : Real}
55     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
56     (hepisodes : 0 < episodes) (hdelta : 0 < delta)
57     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor)
58     (hscale : 0 < scale)
59     (hthreshold : selfConsistentCountShrinkEpisodeThreshold mdp rounds delta
60         visitFloor scale < (episodes : Real)) :
61     simultaneousCountConfidenceRadius mdp episodes
62         (multiBatchLocalDelta rounds delta) <
63       (episodes : Real) * visitFloor / scale ^ 2 := by
64   let witnessState : State := Classical.choice inferInstance
65   let witnessAction : Action := Classical.choice inferInstance
66   have hcoordinate : Nonempty (CountCoordinate mdp) :=
67     ⟨CountCoordinate.visit ⟨0, hhorizon⟩ witnessState witnessAction⟩
68   have hroundsReal : (0 : Real) < rounds := by exact_mod_cast hrounds
69   have hlocalPos : 0 < multiBatchLocalDelta rounds delta :=
70     div_pos hdelta hroundsReal
71   have hlocalLeOne : multiBatchLocalDelta rounds delta <= 1 :=
72     (div_le_self hdelta.le (by exact_mod_cast hrounds)).trans hdelta_le_one
73   have hcoordinatePos :
74       0 < simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta) :=
75     MarkovPolicy.simultaneousCountDelta_pos hcoordinate hlocalPos
76   have hcoordinateLeOne :
77       simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta) <= 1 :=
78     MarkovPolicy.simultaneousCountDelta_le_one hcoordinate hlocalPos hlocalLeOne
79   let radius := simultaneousCountConfidenceRadius mdp episodes
80     (multiBatchLocalDelta rounds delta)
81   let logarithm := Real.log
82     (2 / simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta))
83   have hepisodesReal : (0 : Real) < episodes := by exact_mod_cast hepisodes
84   have hradiusNonneg : 0 <= radius :=
85     Concentration.subGaussianSumConfidenceRadius_nonneg _ _
86   have hradiusSq : radius ^ 2 = (episodes : Real) / 2 * logarithm := by
87     change
88       (Concentration.subGaussianSumConfidenceRadius
89           (MarkovPolicy.iidBernoulliVarianceProxy episodes)
90           (simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta))) ^ 2 =
91         (episodes : Real) / 2 * logarithm
92     rw [Concentration.subGaussianSumConfidenceRadius_sq _ _
93       hcoordinatePos hcoordinateLeOne]
94     rw [MarkovPolicy.iidBernoulliVarianceProxy_eq]
95     dsimp [logarithm]
96     push_cast
97     ring
98   have hthresholdProduct :
99       scale ^ 4 * logarithm <
100         (episodes : Real) * (2 * visitFloor ^ 2) := by
101     have hdenominator : 0 < 2 * visitFloor ^ 2 := by positivity
102     exact (div_lt_iff₀ hdenominator).mp (by
103       simpa [selfConsistentCountShrinkEpisodeThreshold, logarithm]
104         using hthreshold)
105   have hsquare :
106       (scale ^ 2 * radius) ^ 2 <
107         ((episodes : Real) * visitFloor) ^ 2 := by
108     rw [mul_pow, hradiusSq]
109     nlinarith
110   have hscaled : scale ^ 2 * radius < (episodes : Real) * visitFloor :=
111     (sq_lt_sq₀ (mul_nonneg (sq_nonneg scale) hradiusNonneg)
112       (mul_nonneg hepisodesReal.le hvisitFloor.le)).mp hsquare
113   exact (lt_div_iff₀ (sq_pos_of_pos hscale)).2 (by
114     simpa [mul_comm] using hscaled)

116 /-- The reward threshold gives a half-scale-squared sampled-reward-sum bound. -/
117 theorem simultaneousRewardSumConfidenceRadius_lt_episodes_mul_visitFloor_div_two_scale_sq_of_threshold
118     (mdp : MDP State Action) {rounds episodes : Nat}
119     (varianceProxy : NNReal) {delta visitFloor scale : Real}
120     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
121     (hepisodes : 0 < episodes) (hdelta : 0 < delta)
122     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor)
123     (hscale : 0 < scale)
124     (hthreshold : selfConsistentRewardShrinkEpisodeThreshold mdp rounds
125         varianceProxy delta visitFloor scale < (episodes : Real)) :
126     MDP.MeanCompatibleRewardKernel.simultaneousRewardSumConfidenceRadius
127         mdp episodes varianceProxy (multiBatchLocalDelta rounds delta) <
128       (episodes : Real) * visitFloor / (2 * scale ^ 2) := by
129   let witnessState : State := Classical.choice inferInstance
130   let witnessAction : Action := Classical.choice inferInstance
131   have hcoordinate : Nonempty (VisitCoordinate mdp) :=
132     ⟨{ stage := ⟨0, hhorizon⟩, state := witnessState, action := witnessAction }⟩
133   have hroundsReal : (0 : Real) < rounds := by exact_mod_cast hrounds
134   have hlocalPos : 0 < multiBatchLocalDelta rounds delta :=
135     div_pos hdelta hroundsReal
136   have hlocalLeOne : multiBatchLocalDelta rounds delta <= 1 :=
137     (div_le_self hdelta.le (by exact_mod_cast hrounds)).trans hdelta_le_one
138   have hcoordinatePos :
139       0 < MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta mdp
140         (multiBatchLocalDelta rounds delta) :=
141     MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta_pos
142       mdp hcoordinate hlocalPos
143   have hcoordinateLeOne :
144       MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta mdp
145         (multiBatchLocalDelta rounds delta) <= 1 :=
146     MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta_le_one
147       mdp hcoordinate hlocalPos hlocalLeOne
148   let radius :=
149     MDP.MeanCompatibleRewardKernel.simultaneousRewardSumConfidenceRadius
150       mdp episodes varianceProxy (multiBatchLocalDelta rounds delta)
151   let logarithm := Real.log
152     (2 / MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta mdp
153       (multiBatchLocalDelta rounds delta))
154   have hepisodesReal : (0 : Real) < episodes := by exact_mod_cast hepisodes
155   have hradiusNonneg : 0 <= radius :=
156     Concentration.subGaussianSumConfidenceRadius_nonneg _ _
157   have hradiusSq :
158       radius ^ 2 = 2 * (episodes : Real) * (varianceProxy : Real) * logarithm := by
159     change
160       (Concentration.subGaussianSumConfidenceRadius
161           ((episodes : NNReal) * varianceProxy)
162           (MDP.MeanCompatibleRewardKernel.simultaneousRewardDelta mdp
163             (multiBatchLocalDelta rounds delta))) ^ 2 =
164         2 * (episodes : Real) * (varianceProxy : Real) * logarithm
165     rw [Concentration.subGaussianSumConfidenceRadius_sq _ _
166       hcoordinatePos hcoordinateLeOne]
167     dsimp [logarithm]
168     push_cast
169     ring
170   have hthresholdProduct :
171       8 * (varianceProxy : Real) * logarithm * scale ^ 4 <
172         (episodes : Real) * visitFloor ^ 2 := by
173     have hdenominator : 0 < visitFloor ^ 2 := sq_pos_of_pos hvisitFloor
174     exact (div_lt_iff₀ hdenominator).mp (by
175       simpa [selfConsistentRewardShrinkEpisodeThreshold, logarithm]
176         using hthreshold)
177   have hsquare :
178       (2 * scale ^ 2 * radius) ^ 2 <
179         ((episodes : Real) * visitFloor) ^ 2 := by
180     rw [mul_pow, mul_pow, hradiusSq]
181     nlinarith
182   have hscaled :
183       2 * scale ^ 2 * radius < (episodes : Real) * visitFloor :=
184     (sq_lt_sq₀
185       (mul_nonneg (mul_nonneg (by norm_num) (sq_nonneg scale)) hradiusNonneg)
186       (mul_nonneg hepisodesReal.le hvisitFloor.le)).mp hsquare
187   exact (lt_div_iff₀ (mul_pos (by norm_num) (sq_pos_of_pos hscale))).2 (by
188     simpa [mul_assoc, mul_comm, mul_left_comm] using hscaled)
