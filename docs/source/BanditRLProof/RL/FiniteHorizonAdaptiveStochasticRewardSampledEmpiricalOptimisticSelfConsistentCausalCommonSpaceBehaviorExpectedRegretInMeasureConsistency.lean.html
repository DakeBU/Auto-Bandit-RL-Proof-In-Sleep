4 # Natural-causal behavior expected-regret convergence in measure
5 
6 This module closes the regularity boundary left by the a.e. behavior route.
7 For a measurable finite policy-table selector, expected regret of the selected
8 exploratory policy is measurable by a finite indicator-sum representation.
9 The heterogeneous trajectory coordinate selector is already measurable, so
10 every actual successor-policy expected-regret coordinate is measurable.
11 Mathlib then transports the compiled a.e. limit to `TendstoInMeasure` on the
12 same genuine dependent causal trajectory measure.
13 
14 Regularity is unchanged from the a.e. parent: finite nonempty Standard Borel
15 State/Action, probability initial law, positive horizon/base visit floor/reward
16 proxy, bounded stored means, uniform mean-compatible selected-reward sub-
17 Gaussianity, and full-exploration path support. The generic selector lemma
18 uses only finite measurable State/Action, measurable singletons, and a
19 probability initial law.
20 
21 Failure policy: preserve the actual exploratory `source.successorPolicyAt`,
22 the sampled-model recommendation/behavior distinction, the one dependent
23 causal measure, `n`-prefix to `n+1` selection, and the existing a.e. route.
24 This module proves coordinate measurability and convergence in measure, but
25 does not derive `Lp`, expected-value, every-trajectory, anytime, reachability,
26 minimax, optimal-rate, or complete-UCB-VI control.
27 -/

46 /-- Expected regret after a measurable finite table selection is measurable. -/
47 theorem measurable_exploratoryPolicy_expectedRegret_comp
48     {Omega : Type w} [MeasurableSpace Omega]
49     {mdp : MDP State Action} (initialState : Measure State)
50     [IsProbabilityMeasure initialState]
51     (selector : Omega -> DeterministicMarkovPolicyTable mdp)
52     (hselector : Measurable selector)
53     (explorationRate : NNReal) (hexplorationRate : explorationRate <= 1) :
54     Measurable fun omega =>
55       ((selector omega).exploratoryPolicy explorationRate
56         hexplorationRate).expectedRegret initialState := by
57   classical
58   let statistic := fun table : DeterministicMarkovPolicyTable mdp =>
59     (table.exploratoryPolicy explorationRate
60       hexplorationRate).expectedRegret initialState
61   have hrepresentation :
62       (fun omega =>
63         ((selector omega).exploratoryPolicy explorationRate
64           hexplorationRate).expectedRegret initialState) =
65         fun omega =>
66           ∑ table : DeterministicMarkovPolicyTable mdp,
67             if selector omega = table then statistic table else 0 := by
68     funext omega
69     simp [statistic]
70   rw [hrepresentation]
71   exact Finset.measurable_sum Finset.univ fun table _ =>
72     Measurable.ite
73       (hselector (measurableSet_singleton table))
74       measurable_const measurable_const
