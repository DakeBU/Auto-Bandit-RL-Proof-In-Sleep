4 # Expected regret of the canonical generated Hoeffding UCBVI-CH source
5 
6 This file first closes the small measurability bridge from the recurrent
7 finite table to policy-value pseudo-regret.  It then integrates the compiled
8 high-probability terminal, charging the deterministic `K H` envelope only on
9 the measurable hull of the proved failure event.  The resulting corollary
10 therefore retains the required `K H delta` term.
11 -/

30     (mdp : MDP State Action) (defaultState : State)

60 theorem measurable_generatedEpisodePseudoRegret
61     (mdp : MDP State Action) (initialState : Measure State)
62     [IsProbabilityMeasure initialState]
63     (defaultState : State) (episodes : Nat) (delta : Real)
64     (episode : Nat) :
65     Measurable (fun trajectory : EpisodeBatchTrajectory mdp 1 =>
66       generatedEpisodePseudoRegret
67         (recurrentSource mdp initialState defaultState episodes delta)
68         defaultState trajectory episode) := by
69   let state : EpisodeBatchTrajectory mdp 1 → State := fun trajectory =>
70     generatedEpisodeInitialState mdp defaultState (trajectory episode)
71   have hstate : Measurable state :=
72     (measurable_generatedEpisodeInitialState mdp defaultState).comp
73       (measurable_pi_apply episode)
74   let regretOf : DeterministicMarkovPolicyTable mdp × State → Real :=
75     fun pair =>
76       mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) pair.2 -
77         pair.1.toMarkovPolicy.valueAt 0
78           (Nat.zero_le mdp.horizon) pair.2
79   have hregretOf : Measurable regretOf := measurable_of_countable _
80   have hpair : Measurable (fun trajectory : EpisodeBatchTrajectory mdp 1 =>
81       (generatedPolicyTable mdp defaultState episodes delta trajectory episode,
82         state trajectory)) :=
83     (measurable_generatedPolicyTable mdp defaultState episodes delta episode).prodMk
84       hstate
85   simpa only [generatedEpisodePseudoRegret,
86     recurrentSource_policyAt_eq_generatedPolicyTable, state, regretOf] using
87     hregretOf.comp hpair

102 /-- Canonical finite-time expected pseudo-regret.  The `K H delta` summand is
103 the explicit contribution of the proved terminal failure event. -/
104 theorem integral_cumulativeEpisodePseudoRegret_recurrentSource_le_canonicalRegretBound_add_failure
105     (mdp : MDP State Action) (initialState : Measure State)
106     [IsProbabilityMeasure initialState]
107     [StandardBorelSpace (EpisodeBatch mdp 1)]
108     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]
109     (defaultState : State) (episodes : Nat) (delta : Real)
110     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
111     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
112     (hrewardNonneg : ∀ state action, 0 <= mdp.reward state action)
113     (hrewardOne : ∀ state action, mdp.reward state action <= 1) :
114     let source := recurrentSource mdp initialState defaultState episodes delta
115     ∫ trajectory, cumulativeEpisodePseudoRegret source defaultState episodes trajectory
116         ∂source.trajectoryMeasure <=
117       canonicalRegretBound (State := State) (Action := Action)
118           mdp episodes delta +
119         (episodes : Real) * mdp.horizon * delta := by
120   dsimp only
121   let source := recurrentSource mdp initialState defaultState episodes delta
122   let μ := source.trajectoryMeasure
123   let regret : EpisodeBatchTrajectory mdp 1 → Real := fun trajectory =>
124     cumulativeEpisodePseudoRegret source defaultState episodes trajectory
125   let bound := canonicalRegretBound (State := State) (Action := Action)
126     mdp episodes delta
127   let envelope : Real := episodes * mdp.horizon
128   let failure := canonicalFailureEvent (mdp := mdp) (initialState := initialState)
129     defaultState episodes delta
130   let bad := toMeasurable μ failure
131   let overflow : EpisodeBatchTrajectory mdp 1 → Real :=
132     bad.indicator (fun _ => envelope)
133   have hregretMeasurable : Measurable regret :=
134     measurable_cumulativeEpisodePseudoRegret mdp initialState defaultState
135       episodes delta
136   have hregretBound : ∀ trajectory, |regret trajectory| <= envelope := by
137     intro trajectory
138     have hmem := cumulativeEpisodePseudoRegret_mem_Icc source defaultState
139       episodes trajectory hrewardNonneg hrewardOne
140     rw [abs_of_nonneg hmem.1]
141     exact hmem.2
142   have hregretIntegrable : Integrable regret μ := by
143     exact (integrable_const envelope).mono'
144       hregretMeasurable.aestronglyMeasurable (ae_of_all μ hregretBound)
145   have hbad : MeasurableSet bad := measurableSet_toMeasurable _ _
146   have hoverflow : Integrable overflow μ :=
147     (integrable_const envelope).indicator hbad
148   have hbound : 0 <= bound := canonicalRegretBound_nonneg mdp episodes delta
149   have hpoint : ∀ trajectory, regret trajectory <= bound + overflow trajectory := by
150     intro trajectory
151     by_cases htrajectory : trajectory ∈ bad
152     · have hupper := (cumulativeEpisodePseudoRegret_mem_Icc source defaultState
153         episodes trajectory hrewardNonneg hrewardOne).2
154       calc
155         regret trajectory <= envelope := hupper
156         _ <= bound + envelope := le_add_of_nonneg_left hbound
157         _ = bound + overflow trajectory := by
158           simp [overflow, Set.indicator_of_mem htrajectory]
159     · have hnotFailure : trajectory ∉ failure := by
160         intro hfailure
161         exact htrajectory (subset_toMeasurable μ failure hfailure)
162       have hgood := cumulativeEpisodePseudoRegret_le_canonicalRegretBound_of_not_mem
163         mdp initialState defaultState episodes delta hhorizon hepisodes
164           hdelta hdelta_le_one hrewardNonneg hrewardOne trajectory hnotFailure
165       calc
166         regret trajectory <= bound := hgood
167         _ = bound + overflow trajectory := by
168           simp [overflow, Set.indicator_of_notMem htrajectory]
169   have htail : μ failure <= ENNReal.ofReal delta :=
170     recurrentSource_trajectoryMeasure_canonicalFailureEvent_le
171       mdp initialState defaultState episodes delta hhorizon hepisodes
172         hdelta hdelta_le_one hrewardNonneg hrewardOne
173   have htailReal : μ.real bad <= delta := by
174     rw [Measure.real, measure_toMeasurable]
175     calc
176       (μ failure).toReal <= (ENNReal.ofReal delta).toReal :=
177         ENNReal.toReal_mono ENNReal.ofReal_ne_top htail
178       _ = delta := ENNReal.toReal_ofReal hdelta.le
179   change integral μ regret <= bound + envelope * delta
180   calc
181     integral μ regret <= integral μ (fun trajectory =>
182         bound + overflow trajectory) := by
183       exact integral_mono hregretIntegrable
184         ((integrable_const bound).add hoverflow) hpoint
185     _ = bound + integral μ overflow := by
186       rw [integral_add (integrable_const bound) hoverflow, integral_const]
187       simp [MeasureTheory.probReal_univ]
188     _ = bound + envelope * μ.real bad := by
189       congr 1
190       change integral μ (bad.indicator (fun _ => envelope)) =
191         envelope * μ.real bad
192       rw [integral_indicator hbad, setIntegral_const]
193       simp [Measure.real, smul_eq_mul, mul_comm]
194     _ <= bound + envelope * delta := by
195       exact add_le_add le_rfl
196         (mul_le_mul_of_nonneg_left htailReal (by positivity))
