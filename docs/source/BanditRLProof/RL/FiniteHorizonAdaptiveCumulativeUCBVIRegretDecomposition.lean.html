60 /-- Per-stage deterministic charge.  A zero previous count is paid by `H`;
61 positive counts pay the `9HL/sqrt N` bonus/projection term plus the explicit
62 `66 S H^2 L/N` Bernstein correction. -/
63 noncomputable def successorLocalCharge
64     (mdp : MDP State Action) (defaultState : State)
65     (episodes : Nat) (delta : Real)
66     (trajectory : EpisodeBatchTrajectory mdp 1) (n remaining : Nat)
67     (hremaining : remaining + 1 <= mdp.horizon) (state : State) : Real :=
68   let stage := mdp.decisionStageRemaining remaining hremaining
69   let action := successorPolicyTable mdp defaultState episodes delta trajectory n
70     stage state
71   let count := (successorSummary trajectory n).aggregateVisitCount state action
72   if count = 0 then
73     (mdp.horizon : Real)
74   else
75     min (mdp.horizon : Real)
76       (9 * (mdp.horizon : Real) *
77           logFactor (State := State) (Action := Action) mdp episodes delta /
78             Real.sqrt count +
79         66 * Fintype.card State * (mdp.horizon : Real) ^ 2 *
80           logFactor (State := State) (Action := Action) mdp episodes delta /
81             count)
