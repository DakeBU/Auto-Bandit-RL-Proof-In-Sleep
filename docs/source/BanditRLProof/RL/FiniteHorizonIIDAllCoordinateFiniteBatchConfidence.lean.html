4 # All-coordinate finite-batch confidence for finite-horizon iid trajectories
5 
6 This module turns the compiled generated reward and transition laws into an
7 actual `MDP.FiniteBatchModel.Confidence` producer. The construction is
8 noncircular: reward radius is zero, transition radius is one fixed external
9 budget, transition coordinate radii use genuine expected-count lower margins,
10 and the recursive value envelope is the explicit linear function
11 `remaining * (rewardBound + transitionBudget)`.
12 -/

29 /-- Explicit noncircular envelope for a fixed reward and transition budget. -/
30 def empiricalFiniteBatchValueEnvelope
31     (rewardBound transitionBudget : Real) (remaining : Nat) : Real :=
32   (remaining : Real) * (rewardBound + transitionBudget)

37 Canonical empirical model for the all-coordinate route: exact generated rewards

40 noncomputable def allCoordinateEmpiricalFiniteBatchModel
41     (mdp : MDP State Action) (episodes : Nat) (batch : EpisodeBatch mdp episodes)
42     (defaultState : State) (transitionBudget : Real) :
43     FiniteBatchModel mdp episodes where

53 /-- Deterministic lower-margin coordinate radius based on the genuine visit mean. -/
54 noncomputable def expectedCountTransitionCoordinateRadius
55     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
56     (initialState : Measure State) [IsProbabilityMeasure initialState]
57     (episodes : Nat) (delta : Real) (stage : Fin mdp.horizon)
58     (state : State) (action : Action) (_nextState : State) : Real :=
59   let coordinate : VisitCoordinate mdp :=
60     { stage := stage, state := state, action := action }
61   2 * simultaneousCountConfidenceRadius mdp episodes delta /
62     (coordinate.expectedCount policy initialState episodes -
63       simultaneousCountConfidenceRadius mdp episodes delta)

67 /-- Outside the simultaneous event, realized count exceeds its deterministic lower margin. -/
68 theorem expectedCount_sub_radius_lt_count_of_not_mem_simultaneousCountBadEvent
69     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
70     (initialState : Measure State) [IsProbabilityMeasure initialState]
71     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
72     (hbatch : batch ∉
73       policy.simultaneousCountBadEvent initialState episodes delta)
74     (coordinate : VisitCoordinate mdp) :
75     coordinate.expectedCount policy initialState episodes -
76         simultaneousCountConfidenceRadius mdp episodes delta <
77       (coordinate.count batch : Real) := by
78   have hdeviation :=
79     policy.visitCount_abs_deviation_lt_of_not_mem_simultaneousCountBadEvent
80       initialState batch hbatch coordinate.stage coordinate.state coordinate.action
81   have hleft := (abs_lt.mp hdeviation).1
82   simpa [VisitCoordinate.count, VisitCoordinate.expectedCount] using (show
83     coordinate.expectedCount policy initialState episodes -
84         simultaneousCountConfidenceRadius mdp episodes delta <
85       (coordinate.count batch : Real) by
86         dsimp [VisitCoordinate.count, VisitCoordinate.expectedCount]
87         linarith)

90 /--
91 The random-denominator transition error is bounded by the deterministic
92 expected-count lower-margin radius.
93 -/
94 theorem empiricalTransitionMass_abs_sub_transition_le_expectedCountRadius_of_not_mem
95     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
96     (initialState : Measure State) [IsProbabilityMeasure initialState]
97     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
98     (hbatch : batch ∉
99       policy.simultaneousCountBadEvent initialState episodes delta)
100     (defaultState : State) (coordinate : VisitCoordinate mdp)
101     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
102       coordinate.expectedCount policy initialState episodes)
103     (nextState : State) :
104     |batch.empiricalTransitionMass defaultState coordinate.stage
105           coordinate.state coordinate.action nextState -
106         (mdp.transition (coordinate.state, coordinate.action)).real {nextState}| ≤
107       policy.expectedCountTransitionCoordinateRadius initialState episodes delta
108         coordinate.stage coordinate.state coordinate.action nextState := by
109   let radius := simultaneousCountConfidenceRadius mdp episodes delta
110   let expected := coordinate.expectedCount policy initialState episodes
111   let count : Real := coordinate.count batch
112   have hradius : 0 ≤ radius := by
113     exact Concentration.subGaussianSumConfidenceRadius_nonneg _ _
114   have hdenominator : 0 < expected - radius := by
115     dsimp [expected, radius]
116     linarith
117   have hlower : expected - radius < count := by
118     simpa [expected, radius, count] using
119       policy.expectedCount_sub_radius_lt_count_of_not_mem_simultaneousCountBadEvent
120         initialState batch hbatch coordinate
121   have hrandom :=
122     policy.empiricalTransitionMass_abs_sub_transition_lt_of_not_mem_simultaneousCountBadEvent
123       initialState batch hbatch defaultState coordinate hmargin nextState
124   calc
125     |batch.empiricalTransitionMass defaultState coordinate.stage
126           coordinate.state coordinate.action nextState -
127         (mdp.transition (coordinate.state, coordinate.action)).real {nextState}| ≤
128         2 * radius / count := by
129           simpa [radius, count] using le_of_lt hrandom
130     _ ≤ 2 * radius / (expected - radius) :=
131       div_le_div_of_nonneg_left (mul_nonneg (by norm_num) hradius)
132         hdenominator (le_of_lt hlower)
133     _ = policy.expectedCountTransitionCoordinateRadius initialState episodes delta
134         coordinate.stage coordinate.state coordinate.action nextState := by
135       rfl

139 /-- Full-coordinate margins make every reward-consistent empirical reward exact. -/
140 theorem empiricalReward_eq_of_not_mem_simultaneousCountBadEvent
141     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
142     (initialState : Measure State) [IsProbabilityMeasure initialState]
143     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
144     (hbatch : batch ∉
145       policy.simultaneousCountBadEvent initialState episodes delta)
146     (hreward : batch.RewardConsistent)
147     (hmargin : ∀ coordinate : VisitCoordinate mdp,
148       simultaneousCountConfidenceRadius mdp episodes delta <
149         coordinate.expectedCount policy initialState episodes)
150     (stage : Fin mdp.horizon) (state : State) (action : Action) :
151     batch.empiricalReward stage state action = mdp.reward state action := by
152   let coordinate : VisitCoordinate mdp :=
153     { stage := stage, state := state, action := action }
154   have hcount : 0 < coordinate.count batch :=
155     policy.visitCoordinate_count_pos_of_not_mem_simultaneousCountBadEvent
156       initialState batch hbatch coordinate (hmargin coordinate)
157   exact batch.empiricalReward_eq_reward_of_rewardConsistent hreward
158     stage state action (by
159       simpa [coordinate, VisitCoordinate.count] using Nat.ne_of_gt hcount)

167 /--
168 Exact empirical rewards and fixed nonnegative budgets give the explicit linear
169 absolute envelope for every recursive optimistic value.
170 -/
171 theorem upperValueRemaining_abs_le
172     (hrewardExact : ∀ stage state action,
173       batch.empiricalReward stage state action = mdp.reward state action)
174     (hrewardBound : ∀ state action, |mdp.reward state action| ≤ rewardBound)
175     (htransitionBudget_nonneg : 0 ≤ transitionBudget) :
176     ∀ (remaining : Nat) (hremaining : remaining ≤ mdp.horizon) (state : State),
177       |(mdp.allCoordinateEmpiricalFiniteBatchModel episodes batch defaultState
178           transitionBudget).plan.upperValueRemaining remaining hremaining state| ≤
179         empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining := by
180   let model := mdp.allCoordinateEmpiricalFiniteBatchModel episodes batch
181     defaultState transitionBudget
182   change ∀ (remaining : Nat) (hremaining : remaining ≤ mdp.horizon) (state : State),
183     |model.plan.upperValueRemaining remaining hremaining state| ≤
184       empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining
185   intro remaining
186   induction remaining with
187   | zero =>
188       intro hremaining state
189       simp [MDP.EstimatedModelPlan.upperValueRemaining,
190         empiricalFiniteBatchValueEnvelope]
191   | succ remaining ih =>
192       intro hremaining state
193       let stage := mdp.decisionStageRemaining remaining hremaining
194       let tail := model.plan.upperValueRemaining remaining (by omega)
195       let action := model.plan.optimisticAction stage tail state
196       have htail : ∀ nextState, |tail nextState| ≤
197           empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining := by
198         intro nextState
199         exact ih (by omega) nextState
200       have htransition :
201           |model.plan.transitionValue stage tail state action| ≤
202             empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining := by
203         rw [← Real.norm_eq_abs]
204         have hnorm := norm_integral_le_of_norm_le_const
205           (μ := model.plan.estimatedTransition stage (state, action))
206           (C := empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining)
207           (Filter.Eventually.of_forall fun nextState => by
208             rw [Real.norm_eq_abs]
209             exact htail nextState)
210         simpa [MDP.EstimatedModelPlan.transitionValue] using hnorm
211       rw [MDP.EstimatedModelPlan.upperValueRemaining]
212       change
213         |batch.empiricalReward stage state action +
214               model.plan.transitionValue stage tail state action +
215             0 + transitionBudget| ≤
216           empiricalFiniteBatchValueEnvelope rewardBound transitionBudget (remaining + 1)
217       rw [hrewardExact stage state action]
218       calc
219         |mdp.reward state action +
220               model.plan.transitionValue stage tail state action +
221             0 + transitionBudget| =
222             |(mdp.reward state action +
223               model.plan.transitionValue stage tail state action) +
224               transitionBudget| := by ring_nf
225         _ ≤ |mdp.reward state action +
226               model.plan.transitionValue stage tail state action| +
227               |transitionBudget| := abs_add_le _ _
228         _ ≤ (|mdp.reward state action| +
229               |model.plan.transitionValue stage tail state action|) +
230               |transitionBudget| := by
231             have hsum := abs_add_le (mdp.reward state action)
232               (model.plan.transitionValue stage tail state action)
233             linarith
234         _ ≤ rewardBound +
235               empiricalFiniteBatchValueEnvelope rewardBound transitionBudget remaining +
236               transitionBudget := by
237             rw [abs_of_nonneg htransitionBudget_nonneg]
238             linarith [hrewardBound state action, htransition]
239         _ = empiricalFiniteBatchValueEnvelope rewardBound transitionBudget (remaining + 1) := by
240             unfold empiricalFiniteBatchValueEnvelope
241             rw [show ((remaining + 1 : Nat) : Real) =
242               (remaining : Real) + 1 by norm_num]
243             ring

245 end AllCoordinateConfidence

247 /--
248 Pathwise producer: full genuine occupancy margins, deterministic radius cover,
249 and reward consistency construct the complete raw finite-batch confidence object.
250 -/
251 noncomputable def allCoordinateEmpiricalFiniteBatchModelConfidence_of_not_mem
252     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
253     (initialState : Measure State) [IsProbabilityMeasure initialState]
254     {episodes : Nat} {delta : Real} (batch : EpisodeBatch mdp episodes)
255     (hbatch : batch ∉
256       policy.simultaneousCountBadEvent initialState episodes delta)
257     (hreward : batch.RewardConsistent) (defaultState : State)
258     (rewardBound transitionBudget : Real)
259     (hrewardBound : ∀ state action, |mdp.reward state action| ≤ rewardBound)
260     (htransitionBudget_nonneg : 0 ≤ transitionBudget)
261     (hmargin : ∀ coordinate : VisitCoordinate mdp,
262       simultaneousCountConfidenceRadius mdp episodes delta <
263         coordinate.expectedCount policy initialState episodes)
264     (hcover : ∀ (remaining : Nat)
265         (hremaining : remaining + 1 ≤ mdp.horizon)
266         (state : State) (action : Action),
267       (∑ nextState,
268           policy.expectedCountTransitionCoordinateRadius
269               initialState episodes delta
270               (mdp.decisionStageRemaining remaining hremaining)
271               state action nextState *
272             empiricalFiniteBatchValueEnvelope
273               rewardBound transitionBudget remaining) ≤
274         transitionBudget) :
275     (mdp.allCoordinateEmpiricalFiniteBatchModel episodes batch defaultState
276       transitionBudget).Confidence where
277   transitionCoordinateRadius :=
278     policy.expectedCountTransitionCoordinateRadius initialState episodes delta
279   valueEnvelope :=
280     empiricalFiniteBatchValueEnvelope rewardBound transitionBudget
281   rewardError_le_radius := by
282     intro stage state action
283     change |batch.empiricalReward stage state action - mdp.reward state action| ≤ 0
284     rw [policy.empiricalReward_eq_of_not_mem_simultaneousCountBadEvent
285       initialState batch hbatch hreward hmargin stage state action]
286     simp
287   transitionFrequencyError_le_radius := by
288     intro remaining hremaining state action nextState
289     let coordinate : VisitCoordinate mdp :=
290       { stage := mdp.decisionStageRemaining remaining hremaining
291         state := state
292         action := action }
293     simpa [coordinate] using
294       policy.empiricalTransitionMass_abs_sub_transition_le_expectedCountRadius_of_not_mem
295         initialState batch hbatch defaultState coordinate
296           (hmargin coordinate) nextState
297   upperValue_abs_le_envelope := by
298     intro remaining hremaining state
299     exact AllCoordinateConfidence.upperValueRemaining_abs_le
300       (mdp := mdp) (batch := batch) (defaultState := defaultState)
301       (transitionBudget := transitionBudget) (rewardBound := rewardBound)
302         (policy.empiricalReward_eq_of_not_mem_simultaneousCountBadEvent
303           initialState batch hbatch hreward hmargin)
304         hrewardBound htransitionBudget_nonneg
305         remaining (by omega) state
306   transitionRadius_cover := by
307     intro remaining hremaining state action
308     simpa [MDP.allCoordinateEmpiricalFiniteBatchModel] using
309       hcover remaining hremaining state action

311 /--
312 Mapped-iid confidence endpoint with the unchanged simultaneous-event failure
313 budget. No additional reward or confidence event is introduced.
314 -/
315 theorem iidEpisodeBatch_allCoordinate_finiteBatchModel_confidence
316     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
317     (initialState : Measure State) [IsProbabilityMeasure initialState]
318     (episodes : Nat) (hepisodes : 0 < episodes)
319     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1)
320     (defaultState : State) (rewardBound transitionBudget : Real)
321     (hrewardBound : ∀ state action, |mdp.reward state action| ≤ rewardBound)
322     (htransitionBudget_nonneg : 0 ≤ transitionBudget)
323     (hmargin : ∀ coordinate : VisitCoordinate mdp,
324       simultaneousCountConfidenceRadius mdp episodes delta <
325         coordinate.expectedCount policy initialState episodes)
326     (hcover : ∀ (remaining : Nat)
327         (hremaining : remaining + 1 ≤ mdp.horizon)
328         (state : State) (action : Action),
329       (∑ nextState,
330           policy.expectedCountTransitionCoordinateRadius
331               initialState episodes delta
332               (mdp.decisionStageRemaining remaining hremaining)
333               state action nextState *
334             empiricalFiniteBatchValueEnvelope
335               rewardBound transitionBudget remaining) ≤
336         transitionBudget) :
337     MeasurableSet
338         (policy.simultaneousCountBadEvent initialState episodes delta) ∧
339       (policy.iidEpisodeBatchMeasure initialState episodes)
340           (policy.simultaneousCountBadEvent initialState episodes delta) ≤
341         ENNReal.ofReal delta ∧
342       ∀ᵐ batch ∂policy.iidEpisodeBatchMeasure initialState episodes,
343         batch ∉ policy.simultaneousCountBadEvent initialState episodes delta →
344           Nonempty
345             (mdp.allCoordinateEmpiricalFiniteBatchModel episodes batch defaultState
346               transitionBudget).Confidence := by
347   refine ⟨policy.measurableSet_simultaneousCountBadEvent
348       initialState episodes delta,
349     policy.iidEpisodeBatch_simultaneousCountBadEvent_le
350       initialState episodes hepisodes delta hdelta hdelta_le_one, ?_⟩
351   filter_upwards [policy.iidEpisodeBatchMeasure_rewardConsistent_ae
352     initialState episodes] with batch hreward
353   intro hbatch
354   exact ⟨policy.allCoordinateEmpiricalFiniteBatchModelConfidence_of_not_mem
355     initialState batch hbatch hreward defaultState rewardBound transitionBudget
356       hrewardBound htransitionBudget_nonneg hmargin hcover⟩

358 /--
359 The produced confidence object immediately yields global optimism and the
360 existing selected-radius single-episode expected-regret bound almost everywhere.
361 -/
362 theorem iidEpisodeBatch_allCoordinate_optimism_and_expectedRegret
363     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
364     (initialState : Measure State) [IsProbabilityMeasure initialState]
365     (episodes : Nat) (hepisodes : 0 < episodes)
366     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta ≤ 1)
367     (defaultState : State) (rewardBound transitionBudget : Real)
368     (hrewardBound : ∀ state action, |mdp.reward state action| ≤ rewardBound)
369     (htransitionBudget_nonneg : 0 ≤ transitionBudget)
370     (hmargin : ∀ coordinate : VisitCoordinate mdp,
371       simultaneousCountConfidenceRadius mdp episodes delta <
372         coordinate.expectedCount policy initialState episodes)
373     (hcover : ∀ (remaining : Nat)
374         (hremaining : remaining + 1 ≤ mdp.horizon)
375         (state : State) (action : Action),
376       (∑ nextState,
377           policy.expectedCountTransitionCoordinateRadius
378               initialState episodes delta
379               (mdp.decisionStageRemaining remaining hremaining)
380               state action nextState *
381             empiricalFiniteBatchValueEnvelope
382               rewardBound transitionBudget remaining) ≤
383         transitionBudget) :
384     MeasurableSet
385         (policy.simultaneousCountBadEvent initialState episodes delta) ∧
386       (policy.iidEpisodeBatchMeasure initialState episodes)
387           (policy.simultaneousCountBadEvent initialState episodes delta) ≤
388         ENNReal.ofReal delta ∧
389       ∀ᵐ batch ∂policy.iidEpisodeBatchMeasure initialState episodes,
390         batch ∉ policy.simultaneousCountBadEvent initialState episodes delta →
391           let model := mdp.allCoordinateEmpiricalFiniteBatchModel
392             episodes batch defaultState transitionBudget
393           (∀ state,
394               mdp.optimalValueRemaining mdp.horizon le_rfl state ≤
395                 model.plan.upperValueRemaining mdp.horizon le_rfl state) ∧
396             model.plan.optimisticPolicy.expectedRegret initialState ≤
397               model.plan.optimisticPolicy.occupancySumRemaining
398                 (fun remaining hremaining state =>
399                   2 * model.plan.selectedRadiusRemaining
400                     remaining hremaining state)
401                 mdp.horizon le_rfl initialState := by
402   obtain ⟨hmeasurable, htail, hconfidence⟩ :=
403     policy.iidEpisodeBatch_allCoordinate_finiteBatchModel_confidence
404       initialState episodes hepisodes delta hdelta hdelta_le_one defaultState
405         rewardBound transitionBudget hrewardBound htransitionBudget_nonneg
406         hmargin hcover
407   refine ⟨hmeasurable, htail, ?_⟩
408   filter_upwards [hconfidence] with batch hbatchConfidence
409   intro hbatch
410   obtain ⟨confidence⟩ := hbatchConfidence hbatch
411   exact confidence
412     |>.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
413       initialState
