20 /-- Nonnegative deterministic rewards give nonnegative finite-horizon policy
21 values. -/
22 theorem valueRemaining_nonneg_of_reward_nonneg
23     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
24     (hreward : ∀ state action, 0 <= mdp.reward state action) :
25     ∀ remaining (hremaining : remaining <= mdp.horizon) state,
26       0 <= policy.valueRemaining remaining hremaining state := by
27   intro remaining
28   induction remaining with
29   | zero => simp [valueRemaining]
30   | succ remaining ih =>
31       intro hremaining state
32       rw [valueRemaining]
33       apply integral_nonneg
34       intro action
35       unfold MDP.bellmanQ MDP.transitionValue
36       exact add_nonneg (hreward state action)
37         (integral_nonneg fun nextState => ih (by omega) nextState)
38 
39 end MarkovPolicy

63 /-- The canonical source policy at every coordinate is definitionally the
64 argmax of `generatedQTable` built from its strict prefix. -/
65 theorem recurrentSource_policyAt_eq_generatedPolicyTable
66     (mdp : MDP State Action) (initialState : Measure State)
67     [IsProbabilityMeasure initialState]
68     (defaultState : State) (episodes : Nat) (delta : Real)
69     (trajectory : EpisodeBatchTrajectory mdp 1) (episode : Nat) :
70     (recurrentSource mdp initialState defaultState episodes delta).policyAt
71         trajectory episode =
72       DeterministicMarkovPolicyTable.toMarkovPolicy
73         (generatedPolicyTable mdp defaultState episodes delta trajectory episode) := by
74   cases episode with
75   | zero => rfl
76   | succ episode => rfl

78 /-- Totalized initial state of one generated single-episode batch.  On the
79 canonical positive-horizon domain it is literally stage zero's state. -/
80 def generatedEpisodeInitialState
81     (mdp : MDP State Action) (defaultState : State)
82     (batch : EpisodeBatch mdp 1) : State :=
83   if hhorizon : 0 < mdp.horizon then
84     (batch 0 ⟨0, hhorizon⟩).state
85   else defaultState

87 /-- Pathwise policy-value pseudo-regret of one generated coordinate. -/
88 noncomputable def generatedEpisodePseudoRegret
89     {mdp : MDP State Action} {initialState : Measure State}
90     [IsProbabilityMeasure initialState]
91     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
92     (defaultState : State) (trajectory : EpisodeBatchTrajectory mdp 1)
93     (episode : Nat) : Real :=
94   let state := generatedEpisodeInitialState mdp defaultState (trajectory episode)
95   mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
96     (source.policyAt trajectory episode).valueAt 0
97       (Nat.zero_le mdp.horizon) state

99 /-- Raw generated cumulative episode pseudo-regret over exactly coordinates
100 `0,...,K-1`; coordinate zero is included and never hidden. -/
101 noncomputable def cumulativeEpisodePseudoRegret
102     {mdp : MDP State Action} {initialState : Measure State}
103     [IsProbabilityMeasure initialState]
104     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
105     (defaultState : State) (episodes : Nat)
106     (trajectory : EpisodeBatchTrajectory mdp 1) : Real :=
107   ∑ episode : Fin episodes,

110 /-- Every generated policy-value pseudo-regret lies in `[0,H]` under rewards
111 in `[0,1]`. -/
112 theorem generatedEpisodePseudoRegret_mem_Icc
113     {mdp : MDP State Action} {initialState : Measure State}
114     [IsProbabilityMeasure initialState]
115     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
116     (defaultState : State) (trajectory : EpisodeBatchTrajectory mdp 1)
117     (episode : Nat)
118     (hrewardNonneg : ∀ state action, 0 <= mdp.reward state action)
119     (hrewardOne : ∀ state action, mdp.reward state action <= 1) :
120     generatedEpisodePseudoRegret source defaultState trajectory episode ∈
121       Set.Icc (0 : Real) mdp.horizon := by
122   let state := generatedEpisodeInitialState mdp defaultState (trajectory episode)
123   let policy := source.policyAt trajectory episode
124   have hpolicyLe := policy.valueAt_le_optimalValueAt 0
125     (Nat.zero_le mdp.horizon) state
126   have hpolicyNonneg : 0 <= policy.valueAt 0
127       (Nat.zero_le mdp.horizon) state := by
128     exact policy.valueRemaining_nonneg_of_reward_nonneg hrewardNonneg
129       mdp.horizon le_rfl state
130   have hoptUpper : mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state <=
131       (mdp.horizon : Real) := by
132     rw [← mdp.optimalPolicy_valueAt_eq_optimalValueAt]
133     exact (le_abs_self _).trans (by
134       simpa using mdp.optimalPolicy.valueRemaining_abs_le_of_rewardBound
135         (1 : NNReal) (by
136           intro state action
137           rw [abs_of_nonneg (hrewardNonneg state action)]
138           exact hrewardOne state action)
139         mdp.horizon le_rfl state)
140   change _ ∈ Set.Icc (0 : Real) mdp.horizon
141   constructor <;> dsimp [generatedEpisodePseudoRegret, state, policy] <;> linarith
142 
143 /-- Consequently the exact `K`-episode raw pseudo-regret lies in `[0,K H]`;
144 this is the envelope later used only on the terminal failure event. -/
145 theorem cumulativeEpisodePseudoRegret_mem_Icc
146     {mdp : MDP State Action} {initialState : Measure State}
147     [IsProbabilityMeasure initialState]
148     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
149     (defaultState : State) (episodes : Nat)
150     (trajectory : EpisodeBatchTrajectory mdp 1)
151     (hrewardNonneg : ∀ state action, 0 <= mdp.reward state action)
152     (hrewardOne : ∀ state action, mdp.reward state action <= 1) :
153     cumulativeEpisodePseudoRegret source defaultState episodes trajectory ∈
154       Set.Icc (0 : Real) ((episodes : Real) * mdp.horizon) := by
155   constructor
156   · exact Finset.sum_nonneg fun episode _ =>
157       (generatedEpisodePseudoRegret_mem_Icc source defaultState trajectory episode
158         hrewardNonneg hrewardOne).1
159   · calc
160       cumulativeEpisodePseudoRegret source defaultState episodes trajectory <=
161           ∑ _episode : Fin episodes, (mdp.horizon : Real) := by
162         exact Finset.sum_le_sum fun episode _ =>
163           (generatedEpisodePseudoRegret_mem_Icc source defaultState trajectory episode
164             hrewardNonneg hrewardOne).2
165       _ = (episodes : Real) * mdp.horizon := by simp
