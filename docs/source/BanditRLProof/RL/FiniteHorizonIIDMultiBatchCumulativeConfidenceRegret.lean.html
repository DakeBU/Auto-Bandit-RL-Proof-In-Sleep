5 # Finite iid multibatch confidence and cumulative expected regret
6 
7 This module takes a finite Mathlib product of the compiled fixed-policy iid
8 episode-batch law. Each product coordinate receives an equal confidence share.
9 Outside the finite union of pulled-back count events, every batch-specific
10 empirical model has a confidence witness, and the resulting one-episode
11 expected-regret bounds sum over the finite product index.
12 
13 The data-generating policy is fixed across product coordinates. The optimistic
14 policy may depend on its batch, but this is not an adaptive online trajectory
15 law and the cumulative quantity is a sum of expected regrets, not realized
16 cumulative regret.
17 -/

63 /-- Pullback of one local simultaneous-count event to a product coordinate. -/
64 def multiBatchRoundBadEvent
65     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
66     (initialState : Measure State) [IsProbabilityMeasure initialState]
67     (rounds episodes : Nat) (delta : Real)
68     (round : Fin rounds) :
69     Set (Fin rounds -> EpisodeBatch mdp episodes) :=

99 /-- Each pulled-back local event has its equal-share probability bound. -/
100 theorem iidEpisodeBatchFamilyMeasure_roundBadEvent_le
101     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
102     (initialState : Measure State) [IsProbabilityMeasure initialState]
103     (rounds : Nat) (hrounds : 0 < rounds)
104     (episodes : Nat) (hepisodes : 0 < episodes)
105     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
106     (round : Fin rounds) :
107     (policy.iidEpisodeBatchFamilyMeasure initialState rounds episodes)
108         (policy.multiBatchRoundBadEvent
109           initialState rounds episodes delta round) <=
110       ENNReal.ofReal (multiBatchLocalDelta rounds delta) := by
111   have hroundsReal : 0 < (rounds : Real) := Nat.cast_pos.mpr hrounds
112   have hroundsOne : (1 : Real) <= (rounds : Real) := by
113     exact_mod_cast hrounds
114   have hlocal_pos : 0 < multiBatchLocalDelta rounds delta :=
115     div_pos hdelta hroundsReal
116   have hlocal_le_one : multiBatchLocalDelta rounds delta <= 1 := by
117     unfold multiBatchLocalDelta
118     exact (div_le_one hroundsReal).2 (hdelta_le_one.trans hroundsOne)
119   let evaluation := MeasureTheory.measurePreserving_eval
120     (fun _round : Fin rounds =>
121       policy.iidEpisodeBatchMeasure initialState episodes) round
122   rw [multiBatchRoundBadEvent]
123   unfold iidEpisodeBatchFamilyMeasure
124   rw [evaluation.measure_preimage
125     (policy.measurableSet_simultaneousCountBadEvent
126       initialState episodes (multiBatchLocalDelta rounds delta)).nullMeasurableSet]
127   exact policy.iidEpisodeBatch_simultaneousCountBadEvent_le
128     initialState episodes hepisodes (multiBatchLocalDelta rounds delta)
129       hlocal_pos hlocal_le_one

191 /-- Sum of the batch-specific selected-radius occupancy bounds. -/
192 noncomputable def multiBatchCumulativeSelectedRadiusOccupancy
193     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
194     (initialState : Measure State) {rounds episodes : Nat}
195     (batches : Fin rounds -> EpisodeBatch mdp episodes)
196     (defaultState : State) (transitionBudget : Real) : Real :=
197   ∑ round,
198     let model := policy.multiBatchEmpiricalModelAt batches defaultState
199       transitionBudget round
200     model.plan.optimisticPolicy.occupancySumRemaining
201       (fun remaining hremaining state =>
202         2 * model.plan.selectedRadiusRemaining remaining hremaining state)

206 Pathwise confidence-family producer outside the finite pulled-back bad-event

209 noncomputable def allCoordinateConfidenceFamily_of_not_mem
210     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
211     (initialState : Measure State) [IsProbabilityMeasure initialState]
212     {rounds episodes : Nat} {delta : Real}
213     (batches : Fin rounds -> EpisodeBatch mdp episodes)
214     (hbatches : batches ∉ policy.multiBatchSimultaneousCountBadEvent
215       initialState rounds episodes delta)
216     (hreward : ∀ round, (batches round).RewardConsistent)
217     (defaultState : State) (rewardBound transitionBudget : Real)
218     (hrewardBound : ∀ state action, |mdp.reward state action| <= rewardBound)
219     (htransitionBudget_nonneg : 0 <= transitionBudget)
220     (hmargin : ∀ coordinate : VisitCoordinate mdp,
221       simultaneousCountConfidenceRadius mdp episodes
222           (multiBatchLocalDelta rounds delta) <
223         coordinate.expectedCount policy initialState episodes)
224     (hcover : ∀ (remaining : Nat)
225         (hremaining : remaining + 1 <= mdp.horizon)
226         (state : State) (action : Action),
227       (∑ nextState,
228           policy.expectedCountTransitionCoordinateRadius
229               initialState episodes (multiBatchLocalDelta rounds delta)
230               (mdp.decisionStageRemaining remaining hremaining)
231               state action nextState *
232             empiricalFiniteBatchValueEnvelope
233               rewardBound transitionBudget remaining) <=
234         transitionBudget) :
235     ∀ round,
236       (policy.multiBatchEmpiricalModelAt batches defaultState
237         transitionBudget round).Confidence := by
238   intro round
239   have hround : batches round ∉ policy.simultaneousCountBadEvent
240       initialState episodes (multiBatchLocalDelta rounds delta) := by
241     intro hbad
242     apply hbatches
243     apply Set.mem_iUnion_of_mem round
244     simpa [multiBatchRoundBadEvent] using hbad
245   exact policy.allCoordinateEmpiricalFiniteBatchModelConfidence_of_not_mem
246     initialState (batches round) hround (hreward round) defaultState
247       rewardBound transitionBudget hrewardBound htransitionBudget_nonneg
248         hmargin hcover

250 /-- Finite sums preserve all roundwise optimism and expected-regret bounds. -/
251 theorem confidenceFamily_optimism_and_cumulativeExpectedRegret
252     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
253     (initialState : Measure State) [IsProbabilityMeasure initialState]
254     {rounds episodes : Nat}
255     (batches : Fin rounds -> EpisodeBatch mdp episodes)
256     (defaultState : State) (transitionBudget : Real)
257     (confidence : ∀ round,
258       (policy.multiBatchEmpiricalModelAt batches defaultState
259         transitionBudget round).Confidence) :
260     (∀ round state,
261         mdp.optimalValueRemaining mdp.horizon le_rfl state <=
262           (policy.multiBatchEmpiricalModelAt batches defaultState
263             transitionBudget round).plan.upperValueRemaining
264               mdp.horizon le_rfl state) ∧
265       policy.multiBatchCumulativeExpectedRegret initialState batches
266           defaultState transitionBudget <=
267         policy.multiBatchCumulativeSelectedRadiusOccupancy initialState batches
268           defaultState transitionBudget := by
269   constructor
270   · intro round state
271     exact
272       (MDP.FiniteBatchModel.Confidence.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
273           (confidence round) initialState).1 state
274   · unfold multiBatchCumulativeExpectedRegret
275       multiBatchCumulativeSelectedRadiusOccupancy
276     apply Finset.sum_le_sum
277     intro round _hround
278     exact
279       (MDP.FiniteBatchModel.Confidence.optimism_and_expectedRegret_le_two_occupancySelectedRadiusRemaining
280           (confidence round) initialState).2

335 /--
336 Route endpoint: the same finite-product event simultaneously yields optimism
337 for every batch model and the cumulative finite sum of expected-regret bounds.
338 -/
339 theorem iidEpisodeBatchFamily_allCoordinate_optimism_and_cumulativeExpectedRegret
340     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
341     (initialState : Measure State) [IsProbabilityMeasure initialState]
342     (rounds : Nat) (hrounds : 0 < rounds)
343     (episodes : Nat) (hepisodes : 0 < episodes)
344     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
345     (defaultState : State) (rewardBound transitionBudget : Real)
346     (hrewardBound : ∀ state action, |mdp.reward state action| <= rewardBound)
347     (htransitionBudget_nonneg : 0 <= transitionBudget)
348     (hmargin : ∀ coordinate : VisitCoordinate mdp,
349       simultaneousCountConfidenceRadius mdp episodes
350           (multiBatchLocalDelta rounds delta) <
351         coordinate.expectedCount policy initialState episodes)
352     (hcover : ∀ (remaining : Nat)
353         (hremaining : remaining + 1 <= mdp.horizon)
354         (state : State) (action : Action),
355       (∑ nextState,
356           policy.expectedCountTransitionCoordinateRadius
357               initialState episodes (multiBatchLocalDelta rounds delta)
358               (mdp.decisionStageRemaining remaining hremaining)
359               state action nextState *
360             empiricalFiniteBatchValueEnvelope
361               rewardBound transitionBudget remaining) <=
362         transitionBudget) :
363     MeasurableSet (policy.multiBatchSimultaneousCountBadEvent
364         initialState rounds episodes delta) ∧
365       (policy.iidEpisodeBatchFamilyMeasure initialState rounds episodes)
366           (policy.multiBatchSimultaneousCountBadEvent
367             initialState rounds episodes delta) <=
368         ENNReal.ofReal delta ∧
369       ∀ᵐ batches ∂policy.iidEpisodeBatchFamilyMeasure
370           initialState rounds episodes,
371         batches ∉ policy.multiBatchSimultaneousCountBadEvent
372             initialState rounds episodes delta ->
373           (∀ round state,
374               mdp.optimalValueRemaining mdp.horizon le_rfl state <=
375                 (policy.multiBatchEmpiricalModelAt batches defaultState
376                   transitionBudget round).plan.upperValueRemaining
377                     mdp.horizon le_rfl state) ∧
378             policy.multiBatchCumulativeExpectedRegret initialState batches
379                 defaultState transitionBudget <=
380               policy.multiBatchCumulativeSelectedRadiusOccupancy initialState
381                 batches defaultState transitionBudget := by
382   obtain ⟨hmeasurable, htail, hconfidence⟩ :=
383     policy.iidEpisodeBatchFamily_allCoordinate_finiteBatchModel_confidence
384       initialState rounds hrounds episodes hepisodes delta hdelta hdelta_le_one
385         defaultState rewardBound transitionBudget hrewardBound
386           htransitionBudget_nonneg hmargin hcover
387   refine ⟨hmeasurable, htail, ?_⟩
388   filter_upwards [hconfidence] with batches hbatchConfidence
389   intro hbatches
390   obtain ⟨confidence⟩ := hbatchConfidence hbatches
391   exact policy.confidenceFamily_optimism_and_cumulativeExpectedRegret
392     initialState batches defaultState transitionBudget confidence
