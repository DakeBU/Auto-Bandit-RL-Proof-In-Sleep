3 /-!
4 # Scheduled average consistency for adaptive cumulative inverse-square-root recommendations
5 
6 This module chooses an explicit natural number of exploratory episodes per
7 batch.  The schedule simultaneously clears the normalized calibration
8 threshold and places the logarithmic factor below one batch's visit mass.
9 The resulting scalar average recommendation-regret bound is at most a fixed
10 constant times `1 / sqrt(rounds)` and therefore tends to zero.
11 
12 The source theorem remains a separate finite-window statement because the
13 episode-batch trajectory type changes with the scheduled batch size.
14 -/

181 /-- The scheduled average bound is controlled by a pure inverse-square-root rate. -/
182 theorem normalizedCumulativeInverseSqrtScheduledAverageBound_le_envelope
183     (mdp : MDP State Action) {rounds : Nat}
184     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
185     {delta visitFloor : Real} (hdelta : 0 < delta)
186     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor) :
187     normalizedCumulativeInverseSqrtAverageRecommendedExpectedRegretBound
188         mdp
189         (normalizedCumulativeInverseSqrtScheduledEpisodes
190           mdp rounds delta visitFloor)
191         rounds delta visitFloor <=
192       normalizedCumulativeInverseSqrtScheduledAverageEnvelope
193         mdp rounds visitFloor := by
194   let episodes := normalizedCumulativeInverseSqrtScheduledEpisodes
195     mdp rounds delta visitFloor
196   have hepisodes : 0 < episodes :=
197     normalizedCumulativeInverseSqrtScheduledEpisodes_pos
198       mdp rounds delta visitFloor
199   have hepisodesReal : 0 < (episodes : Real) := by exact_mod_cast hepisodes
200   have hroundsReal : 0 < (rounds : Real) := by exact_mod_cast hrounds
201   have hlog : 0 < cumulativeInverseSqrtLogFactor mdp rounds delta :=
202     cumulativeInverseSqrtLogFactor_pos mdp hhorizon hrounds
203       hdelta hdelta_le_one
204   have hbatchMass : 0 < (episodes : Real) * visitFloor / 2 := by positivity
205   have hlogLeBatch :
206       cumulativeInverseSqrtLogFactor mdp rounds delta <=
207         (episodes : Real) * visitFloor / 2 := by
208     exact (cumulativeInverseSqrtLogFactor_lt_scheduledEpisodeVisitMass
209       mdp rounds delta hvisitFloor).le
210   have hsqrtLogLeBatch :
211       Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) <=
212         Real.sqrt ((episodes : Real) * visitFloor / 2) :=
213     Real.sqrt_le_sqrt hlogLeBatch
214   have hsqrtVisit : 0 < Real.sqrt visitFloor := Real.sqrt_pos.2 hvisitFloor
215   have hsqrtBatch : 0 < Real.sqrt ((episodes : Real) * visitFloor / 2) :=
216     Real.sqrt_pos.2 hbatchMass
217   have hsqrtRounds : 0 < Real.sqrt (rounds : Real) :=
218     Real.sqrt_pos.2 hroundsReal
219   have htotalMass :
220       (((cumulativeExploratoryEpisodeCount episodes rounds : Nat) : Real) *
221           visitFloor / 2) =
222         ((episodes : Real) * visitFloor / 2) * (rounds : Real) := by
223     unfold cumulativeExploratoryEpisodeCount
224     push_cast
225     ring
226   have hsqrtTotalMass :
227       Real.sqrt
228           (((cumulativeExploratoryEpisodeCount episodes rounds : Nat) : Real) *
229             visitFloor / 2) =
230         Real.sqrt ((episodes : Real) * visitFloor / 2) *
231           Real.sqrt (rounds : Real) := by
232     rw [htotalMass, Real.sqrt_mul hbatchMass.le]
233   have hinner :
234       8 * (Fintype.card State : Real) * (mdp.horizon : Real) *
235               Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
236               Real.sqrt visitFloor /
237             (Real.sqrt ((episodes : Real) * visitFloor / 2) *
238               Real.sqrt (rounds : Real)) <=
239         8 * (Fintype.card State : Real) * (mdp.horizon : Real) /
240             Real.sqrt visitFloor /
241           Real.sqrt (rounds : Real) := by
242     calc
243       8 * (Fintype.card State : Real) * (mdp.horizon : Real) *
244               Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
245               Real.sqrt visitFloor /
246             (Real.sqrt ((episodes : Real) * visitFloor / 2) *
247               Real.sqrt (rounds : Real)) =
248           (8 * (Fintype.card State : Real) * (mdp.horizon : Real) /
249               Real.sqrt visitFloor /
250             Real.sqrt (rounds : Real)) *
251           (Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
252             Real.sqrt ((episodes : Real) * visitFloor / 2)) := by
253               field_simp [ne_of_gt hsqrtVisit, ne_of_gt hsqrtBatch,
254                 ne_of_gt hsqrtRounds]
255       _ <= (8 * (Fintype.card State : Real) * (mdp.horizon : Real) /
256               Real.sqrt visitFloor /
257             Real.sqrt (rounds : Real)) * 1 := by
258               exact mul_le_mul_of_nonneg_left
259                 ((div_le_one hsqrtBatch).2 hsqrtLogLeBatch) (by positivity)
260       _ = 8 * (Fintype.card State : Real) * (mdp.horizon : Real) /
261               Real.sqrt visitFloor /
262             Real.sqrt (rounds : Real) := by ring
263   rw [show normalizedCumulativeInverseSqrtScheduledEpisodes
264       mdp rounds delta visitFloor = episodes from rfl]
265   rw [normalizedCumulativeInverseSqrtAverageRecommendedExpectedRegretBound_eq_totalEpisodes mdp
266     hepisodes hrounds delta hvisitFloor]
267   rw [hsqrtTotalMass]
268   calc
269     2 * (mdp.horizon : Real) *
270           min 1
271             (8 * (Fintype.card State : Real) * (mdp.horizon : Real) *
272                 Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
273                 Real.sqrt visitFloor /
274               (Real.sqrt ((episodes : Real) * visitFloor / 2) *
275                 Real.sqrt (rounds : Real))) <=
276         2 * (mdp.horizon : Real) *
277           (8 * (Fintype.card State : Real) * (mdp.horizon : Real) *
278               Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
279               Real.sqrt visitFloor /
280             (Real.sqrt ((episodes : Real) * visitFloor / 2) *
281               Real.sqrt (rounds : Real))) := by
282             exact mul_le_mul_of_nonneg_left (min_le_right _ _) (by positivity)
283     _ <= 2 * (mdp.horizon : Real) *
284           (8 * (Fintype.card State : Real) * (mdp.horizon : Real) /
285               Real.sqrt visitFloor /
286             Real.sqrt (rounds : Real)) :=
287       mul_le_mul_of_nonneg_left hinner (by positivity)
288     _ = normalizedCumulativeInverseSqrtScheduledAverageEnvelope
289           mdp rounds visitFloor := by
290       unfold normalizedCumulativeInverseSqrtScheduledAverageEnvelope
