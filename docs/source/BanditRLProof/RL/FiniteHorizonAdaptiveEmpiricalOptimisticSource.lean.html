4 # Adaptive empirical-transition optimistic batch source
5 
6 This module constructs the measurable source left abstract by
7 `FiniteHorizonAdaptiveEpisodeBatchLaw`.  Every observed batch is compressed to
8 its finite family of transition counts.  Those counts define a normalized
9 empirical transition kernel, which is combined with the known deterministic
10 MDP reward and a fixed transition bonus.  The resulting optimistic action
11 table is a measurable function of the batch because the count-summary space is
12 countable with measurable singletons.
13 
14 The table-indexed iid batch laws form a Markov kernel by
15 `Kernel.ofFunOfCountable`.  Comapping that kernel along the measurable
16 history-to-table selector gives an `AdaptiveEpisodeBatchSource` with the exact
17 selected-policy law by construction.  The final theorem therefore exposes the
18 adaptive simultaneous count-confidence conclusion without a caller-supplied
19 kernel law or selected-event measurability proof.
20 
21 This route uses only the most recently observed batch, known rewards, and a
22 fixed transition bonus.  It does not yet prove that the empirical plan is
23 confident, sum bonuses across rounds, or identify cumulative online regret.
24 -/

80 /-- Normalized empirical transition PMF, with an explicit zero-count fallback. -/
81 noncomputable def empiricalTransitionPMF
82     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
83     (defaultState : State) (stage : Fin mdp.horizon)
84     (state : State) (action : Action) : PMF State :=
85   if hzero : summary.visitCount stage state action = 0 then
86     PMF.pure defaultState
87   else
88     PMF.ofFintype
89       (fun nextState =>
90         (summary stage state action nextState : ENNReal) /
91           (summary.visitCount stage state action : ENNReal))
92       (by
93         have sum_div_visitCount (states : Finset State) :
94             (∑ nextState ∈ states,
95                 (summary stage state action nextState : ENNReal) /
96                   (summary.visitCount stage state action : ENNReal)) =
97               (∑ nextState ∈ states,
98                   (summary stage state action nextState : ENNReal)) /
99                 (summary.visitCount stage state action : ENNReal) := by
100           classical
101           induction states using Finset.induction_on with
102           | empty => simp
103           | @insert nextState states hnext ih =>
104               simp only [Finset.sum_insert hnext]
105               rw [ih, ENNReal.add_div]
106         calc
107           (∑ nextState,
108               (summary stage state action nextState : ENNReal) /
109                 (summary.visitCount stage state action : ENNReal)) =
110               (∑ nextState,
111                 (summary stage state action nextState : ENNReal)) /
112                   (summary.visitCount stage state action : ENNReal) := by
113             simpa using sum_div_visitCount (Finset.univ : Finset State)
114           _ = (summary.visitCount stage state action : ENNReal) /
115                 (summary.visitCount stage state action : ENNReal) := by
116             congr 1
117             simp [visitCount]
118           _ = 1 :=
119             ENNReal.div_self (by exact_mod_cast hzero)
120               (ENNReal.natCast_ne_top _))

142 Known-reward empirical-transition plan with zero reward radius and one fixed
143 transition bonus at every coordinate.

145 noncomputable def optimisticPlan
146     (mdp : MDP State Action) (summary : TransitionCountSummary mdp)
147     (defaultState : State) (transitionBonus : Real) :
148     mdp.EstimatedModelPlan where
149   estimatedReward _stage := mdp.reward
150   measurable_estimatedReward _stage := mdp.measurable_reward

153     summary.empiricalTransitionKernel_isMarkov defaultState stage

292 /--
293 Concrete adaptive source: after every batch, recompute the known-reward
294 empirical-transition optimistic table from that batch and sample the next iid
295 batch under the selected deterministic policy.
296 -/
297 noncomputable def source
298     (mdp : MDP State Action) (initialState : Measure State)
299     [IsProbabilityMeasure initialState] (episodes : Nat)
300     (initialTable : DeterministicMarkovPolicyTable mdp)
301     (defaultState : State) (transitionBonus : Real) :
302     AdaptiveEpisodeBatchSource mdp initialState episodes where
303   initialPolicy := initialTable.toMarkovPolicy
304   successorPolicy n history :=
305     (successorTable defaultState transitionBonus n history).toMarkovPolicy
306   batchKernel n :=
307     (DeterministicMarkovPolicyTable.iidEpisodeBatchKernel
308       initialState episodes).comap
309         (successorTable defaultState transitionBonus n)
310         (measurable_successorTable defaultState transitionBonus n)
311   batchKernel_isMarkov n := by
312     exact ProbabilityTheory.Kernel.IsMarkovKernel.comap _
313       (measurable_successorTable defaultState transitionBonus n)
314   batchKernel_eq_iidEpisodeBatchMeasure n history := by
315     rw [ProbabilityTheory.Kernel.comap_apply]
316     rfl

386 /-- The concrete source's next-batch conditional law is its latest empirical optimistic policy law. -/
387 theorem source_trajectoryMeasure_condDistrib_eq_empiricalOptimisticPolicyBatchLaw
388     {mdp : MDP State Action} {initialState : Measure State}
389     [IsProbabilityMeasure initialState] {episodes : Nat}
390     [StandardBorelSpace (EpisodeBatch mdp episodes)]
391     (initialTable : DeterministicMarkovPolicyTable mdp)
392     (defaultState : State) (transitionBonus : Real) (n : Nat) :
393     let concreteSource :=
394       source mdp initialState episodes initialTable defaultState transitionBonus
395     Filter.EventuallyEq
396       (ae (concreteSource.trajectoryMeasure.map (Preorder.frestrictLe n)))
397       (ProbabilityTheory.condDistrib
398         (fun trajectory : EpisodeBatchTrajectory mdp episodes =>
399           trajectory (n + 1))
400         (Preorder.frestrictLe n) concreteSource.trajectoryMeasure)
401       (fun history =>
402         MarkovPolicy.iidEpisodeBatchMeasure
403           ((latestBatch history).transitionCountSummary.optimisticPlan
404             mdp defaultState transitionBonus).optimisticPolicy
405           initialState episodes) := by
406   dsimp only
407   filter_upwards [
408     AdaptiveEpisodeBatchSource.trajectoryMeasure_condDistrib_eq_iidEpisodeBatchMeasure
409         (source mdp initialState episodes initialTable defaultState transitionBonus) n]

415 /--
416 Concrete adaptive count-confidence terminal for the empirical optimistic
417 source.  No selected-law or successor-event measurability premise remains.
418 -/
419 theorem source_trajectoryMeasure_adaptiveSimultaneousCountConfidence
420     {mdp : MDP State Action} {initialState : Measure State}
421     [IsProbabilityMeasure initialState] {episodes : Nat}
422     (initialTable : DeterministicMarkovPolicyTable mdp)
423     (defaultState : State) (transitionBonus : Real)
424     (rounds : Nat) (hrounds : 0 < rounds)
425     (hepisodes : 0 < episodes) (delta : Real) (hdelta : 0 < delta)
426     (hdelta_le_one : delta <= 1) :
427     let concreteSource :=
428       source mdp initialState episodes initialTable defaultState transitionBonus
429     MeasurableSet
430         (concreteSource.adaptiveSimultaneousCountBadEvent rounds delta) ∧
431       concreteSource.trajectoryMeasure
432           (concreteSource.adaptiveSimultaneousCountBadEvent rounds delta) <=
433         ENNReal.ofReal delta ∧
