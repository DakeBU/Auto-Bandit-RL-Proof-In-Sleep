27 /-- At a positive pooled count, the singleton mass of the planner's empirical
28 kernel is exactly the pooled numerator divided by the pooled denominator. -/
29 theorem aggregateEmpiricalTransitionKernel_real_singleton_of_pos
30     {mdp : MDP State Action} (summary : TransitionCountSummary mdp)
31     (defaultState : State) (state : State) (action : Action)
32     (nextState : State)
33     (hpos : 0 < summary.aggregateVisitCount state action) :
34     (summary.aggregateEmpiricalTransitionKernel defaultState
35         (state, action)).real {nextState} =
36       (summary.aggregateTransitionCount state action nextState : Real) /
37         (summary.aggregateVisitCount state action : Real) := by
38   change
39     ((summary.aggregateEmpiricalTransitionPMF defaultState state action).toMeasure).real
40         {nextState} = _
41   rw [Measure.real,
42     PMF.toMeasure_apply_singleton _ _ (MeasurableSet.singleton nextState)]
43   rw [summary.aggregateEmpiricalTransitionPMF_apply_of_aggregateVisitCount_pos
44     defaultState state action nextState hpos]
45   simp [ENNReal.toReal_div, ENNReal.toReal_natCast]

51 theorem bernsteinCoordinateThreshold_mul_probability_div_le
52     {probability logBudget visits : Real}
53     (hprobability : probability ∈ Set.Icc (0 : Real) 1)
54     (hlog : 0 <= logBudget) (hvisits : 0 < visits) :
55     bernsteinCoordinateThreshold logBudget
56           (probability * (1 - probability) * visits) / visits <=
57       2 * Real.sqrt (2 * logBudget / visits) * Real.sqrt probability +
58         2 * logBudget / visits := by
59   let leftRoot := Real.sqrt
60     (2 * (probability * (1 - probability) * visits) * logBudget) / visits
61   let rightRoot := Real.sqrt (2 * logBudget / visits) * Real.sqrt probability
62   have hvariance : 0 <= probability * (1 - probability) * visits :=
63     mul_nonneg (mul_nonneg hprobability.1 (sub_nonneg.mpr hprobability.2))
64       hvisits.le
65   have hleftRoot : 0 <= leftRoot :=
66     div_nonneg (Real.sqrt_nonneg _) hvisits.le
67   have hrightRoot : 0 <= rightRoot :=
68     mul_nonneg (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)
69   have hleftSq : leftRoot ^ 2 =
70       2 * probability * (1 - probability) * logBudget / visits := by
71     dsimp [leftRoot]
72     rw [div_pow, Real.sq_sqrt]
73     · field_simp
74     · positivity
75   have hrightSq : rightRoot ^ 2 =
76       2 * probability * logBudget / visits := by
77     dsimp [rightRoot]
78     rw [mul_pow, Real.sq_sqrt, Real.sq_sqrt]
79     · ring
80     · exact hprobability.1
81     · positivity
82   have hroot : leftRoot <= rightRoot := by
83     apply (sq_le_sq₀ hleftRoot hrightRoot).mp
84     rw [hleftSq, hrightSq]
85     have honeMinus : 0 <= 1 - probability := sub_nonneg.mpr hprobability.2
86     have hmul : probability * (1 - probability) <= probability := by
87       have hOneMinusLe : 1 - probability <= 1 := by linarith [hprobability.1]
88       simpa using mul_le_mul_of_nonneg_left hOneMinusLe hprobability.1
89     have hcoefficient : 0 <= 2 * logBudget / visits := by positivity
90     calc
91       2 * probability * (1 - probability) * logBudget / visits =
92           (2 * logBudget / visits) * (probability * (1 - probability)) := by ring
93       _ <= (2 * logBudget / visits) * probability :=
94         mul_le_mul_of_nonneg_left hmul hcoefficient
95       _ = 2 * probability * logBudget / visits := by ring
96   unfold bernsteinCoordinateThreshold
97   rw [add_div]
98   have hleftRewrite :
99       2 * Real.sqrt
100           (2 * (probability * (1 - probability) * visits) * logBudget) /
101             visits = 2 * leftRoot := by
102     dsimp [leftRoot]
103     ring
104   have hrightRewrite :
105       2 * Real.sqrt (2 * logBudget / visits) * Real.sqrt probability =
106         2 * rightRoot := by
107     dsimp [rightRoot]
108     ring
109   rw [hleftRewrite, hrightRewrite]
110   exact add_le_add
111     (mul_le_mul_of_nonneg_left hroot (by norm_num : (0 : Real) <= 2)) le_rfl

113 /-- A finite weighted Bernstein coordinate family controls a bounded
114 continuation value.  The small `z/(32H)` term is the self-bounding part used by
115 the Bellman recursion; the remaining term is harmonic in the actual count. -/
116 theorem abs_sum_weight_mul_massError_le_transitionValue_div_thirtyTwo_add
117     (probability empirical value : State -> Real)
118     (horizon logBudget visits : Real)
119     (hhorizon : 1 <= horizon) (hlog : 0 <= logBudget)
120     (hvisits : 0 < visits)
121     (hprobability : forall state, 0 <= probability state)
122     (hvalue : forall state, value state ∈ Set.Icc (0 : Real) horizon)
123     (hcoordinate : forall state,
124       |empirical state - probability state| <=
125         2 * Real.sqrt (2 * logBudget / visits) *
126             Real.sqrt (probability state) +
127           2 * logBudget / visits) :
128     |∑ state : State, value state *
129         (empirical state - probability state)| <=
130       (∑ state : State, value state * probability state) / (32 * horizon) +
131         66 * Fintype.card State * horizon ^ 2 * logBudget / visits := by
132   let z : Real := ∑ state : State, value state * probability state
133   let c : Real := Real.sqrt (2 * logBudget / visits)
134   let cardH : Real := Fintype.card State * horizon
135   have hhorizonPos : 0 < horizon := lt_of_lt_of_le (by norm_num) hhorizon
136   have hz : 0 <= z := by
137     exact Finset.sum_nonneg fun state _ =>
138       mul_nonneg (hvalue state).1 (hprobability state)
139   have hsumValue : (∑ state : State, value state) <= cardH := by
140     calc
141       (∑ state : State, value state) <=
142           ∑ _state : State, horizon :=
143         Finset.sum_le_sum fun state _ => (hvalue state).2
144       _ = cardH := by simp [cardH]
145   have hcardH : 0 <= cardH := by
146     exact mul_nonneg (Nat.cast_nonneg _) hhorizonPos.le
147   have hweightedSqrt :
148       (∑ state : State, value state * Real.sqrt (probability state)) <=
149         Real.sqrt z * Real.sqrt cardH := by
150     have hcs := Real.sum_sqrt_mul_sqrt_le (Finset.univ : Finset State)
151       (fun state => mul_nonneg (hprobability state) (hvalue state).1)
152       (fun state => (hvalue state).1)
153     have hleft :
154         (∑ state : State,
155             Real.sqrt (probability state * value state) *
156               Real.sqrt (value state)) =
157           ∑ state : State, value state * Real.sqrt (probability state) := by
158       apply Finset.sum_congr rfl
159       intro state _
160       rw [Real.sqrt_mul (hprobability state)]
161       have hsquare := Real.sq_sqrt (hvalue state).1
162       nlinarith [Real.sqrt_nonneg (probability state),
163         Real.sqrt_nonneg (value state)]
164     rw [hleft] at hcs
165     have hright :
166         (∑ state : State, probability state * value state) = z := by
167       simp only [z]
168       apply Finset.sum_congr rfl
169       intro state _
170       ring
171     rw [hright] at hcs
172     exact hcs.trans (mul_le_mul_of_nonneg_left
173       (Real.sqrt_le_sqrt hsumValue) (Real.sqrt_nonneg z))
174   have hc : 0 <= c := Real.sqrt_nonneg _
175   have hcSq : c ^ 2 = 2 * logBudget / visits := by
176     exact Real.sq_sqrt (div_nonneg (mul_nonneg (by norm_num) hlog) hvisits.le)
177   have hcardHSq : Real.sqrt cardH ^ 2 = cardH := Real.sq_sqrt hcardH
178   let b : Real := c * Real.sqrt cardH
179   have hb : 0 <= b := mul_nonneg hc (Real.sqrt_nonneg _)
180   have hbSq : b ^ 2 = 2 * Fintype.card State * horizon * logBudget / visits := by
181     dsimp [b]
182     rw [mul_pow, hcSq, hcardHSq]
183     field_simp
184     ring
185   have hsqrtZSq : Real.sqrt z ^ 2 = z := Real.sq_sqrt hz
186   have hyoung :
187       2 * c * (Real.sqrt z * Real.sqrt cardH) <=
188         z / (32 * horizon) +
189           64 * Fintype.card State * horizon ^ 2 * logBudget / visits := by
190     have hrhs :
191         z / (32 * horizon) +
192             64 * Fintype.card State * horizon ^ 2 * logBudget / visits =
193           (z + 2048 * Fintype.card State * horizon ^ 3 * logBudget / visits) /
194             (32 * horizon) := by
195       field_simp
196       ring
197     rw [hrhs, le_div_iff₀ (mul_pos (by norm_num) hhorizonPos)]
198     have hsquare := sq_nonneg (Real.sqrt z - 32 * horizon * b)
199     rw [show 2 * c * (Real.sqrt z * Real.sqrt cardH) =
200         2 * Real.sqrt z * b by simp [b]; ring]
201     ring_nf at hsquare
202     rw [hbSq, hsqrtZSq] at hsquare
203     ring_nf at hsquare ⊢
204     nlinarith
205   have hfirst :
206       2 * c *
207           (∑ state : State, value state * Real.sqrt (probability state)) <=
208         z / (32 * horizon) +
209           64 * Fintype.card State * horizon ^ 2 * logBudget / visits :=
210     (mul_le_mul_of_nonneg_left hweightedSqrt
211       (mul_nonneg (by norm_num) hc)).trans hyoung
212   have hsecond :
213       (2 * logBudget / visits) * (∑ state : State, value state) <=
214         2 * Fintype.card State * horizon ^ 2 * logBudget / visits := by
215     have hcoefficient : 0 <= 2 * logBudget / visits :=
216       div_nonneg (mul_nonneg (by norm_num) hlog) hvisits.le
217     calc
218       (2 * logBudget / visits) * (∑ state : State, value state) <=
219           (2 * logBudget / visits) * cardH :=
220         mul_le_mul_of_nonneg_left hsumValue hcoefficient
221       _ <= 2 * Fintype.card State * horizon ^ 2 * logBudget / visits := by
222         dsimp [cardH]
223         have hhorizonSq : horizon <= horizon ^ 2 := by
224           have hmul := mul_le_mul_of_nonneg_left hhorizon hhorizonPos.le
225           nlinarith
226         have hfactor :
227             0 <= 2 * Fintype.card State * logBudget / visits := by
228           positivity
229         calc
230           (2 * logBudget / visits) *
231               (Fintype.card State * horizon) =
232               (2 * Fintype.card State * logBudget / visits) * horizon := by ring
233           _ <= (2 * Fintype.card State * logBudget / visits) * horizon ^ 2 :=
234             mul_le_mul_of_nonneg_left hhorizonSq hfactor
235           _ = _ := by ring
236   calc
237     |∑ state : State, value state *
238         (empirical state - probability state)| <=
239         ∑ state : State,
240           |value state * (empirical state - probability state)| :=
241       Finset.abs_sum_le_sum_abs _ _
242     _ = ∑ state : State,
243         value state * |empirical state - probability state| := by
244       apply Finset.sum_congr rfl
245       intro state _
246       rw [abs_mul, abs_of_nonneg (hvalue state).1]
247     _ <= ∑ state : State,
248         value state *
249           (2 * c * Real.sqrt (probability state) +
250             2 * logBudget / visits) := by
251       apply Finset.sum_le_sum
252       intro state _
253       exact mul_le_mul_of_nonneg_left
254         (by simpa [c, mul_assoc] using hcoordinate state)
255         (hvalue state).1
256     _ = 2 * c *
257           (∑ state : State, value state * Real.sqrt (probability state)) +
258         (2 * logBudget / visits) * (∑ state : State, value state) := by
259       simp_rw [mul_add]
260       rw [Finset.sum_add_distrib]
261       congr 1
262       · rw [Finset.mul_sum]
263         apply Finset.sum_congr rfl
264         intro state _
265         ring
266       · rw [Finset.mul_sum]
267         apply Finset.sum_congr rfl
268         intro state _
269         ring
270     _ <= z / (32 * horizon) +
271           64 * Fintype.card State * horizon ^ 2 * logBudget / visits +
272         2 * Fintype.card State * horizon ^ 2 * logBudget / visits :=
273       add_le_add hfirst hsecond
274     _ = (∑ state : State, value state * probability state) /
275           (32 * horizon) +
276         66 * Fintype.card State * horizon ^ 2 * logBudget / visits := by
277       simp only [z]
278       ring

284 /-- The simultaneous singleton event controls the exact empirical mass error
285 at the actual positive count.  No expected count or auxiliary sample appears. -/
286 theorem abs_empiricalTransitionMass_sub_lt_bernstein
287     {mdp : MDP State Action} {initialState : Measure State}
288     [IsProbabilityMeasure initialState]
289     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
290     {episodes : Nat} {logBudget : Real}
291     {trajectory : EpisodeBatchTrajectory mdp 1}
292     (htrajectory : trajectory ∉
293       simultaneousTransitionFailureEvent source episodes logBudget)
294     (defaultState : State) (index : BernsteinCoordinateIndex mdp episodes)
295     (hactual : adaptiveCumulativeAggregateVisitCountAt trajectory index.round
296         index.state index.action = index.count + 1) :
297     |((adaptiveCumulativeEmpiricalModelStateAt trajectory index.round).1
298           |>.aggregateEmpiricalTransitionKernel defaultState
299             (index.state, index.action)).real {index.nextState} -
300         (mdp.transition (index.state, index.action)).real {index.nextState}| <
301       bernsteinCoordinateThreshold logBudget
302           (mdp.transitionCoordinateVariance
303             index.state index.action index.nextState * (index.count + 1 : Nat)) /
304         (index.count + 1 : Nat) := by
305   let summary := (adaptiveCumulativeEmpiricalModelStateAt trajectory index.round).1
306   let visits : Real := (index.count + 1 : Nat)
307   let truth : Real :=
308     (mdp.transition (index.state, index.action)).real {index.nextState}
309   have hpos : 0 < summary.aggregateVisitCount index.state index.action := by
310     change 0 < adaptiveCumulativeAggregateVisitCountAt trajectory index.round
311       index.state index.action
312     rw [hactual]
313     exact Nat.succ_pos _
314   have hvisitSum :
315       (∑ i ∈ Finset.range (index.round + 1),
316           source.aggregateVisitIncrement index.state index.action i trajectory) =
317         (index.count + 1 : Nat) := by
318     rw [source.sum_aggregateVisitIncrement_eq_prefixAggregateVisitCount]
319     exact_mod_cast hactual
320   have hresidual :=
321     abs_coordinateResidual_lt_of_not_mem_simultaneousTransitionFailureEvent
322       source htrajectory index hvisitSum
323   rw [source.sum_aggregateTransitionResidualIncrement_eq_prefixAggregateResidual]
324       at hresidual
325   have hmass := summary.aggregateEmpiricalTransitionKernel_real_singleton_of_pos
326     defaultState index.state index.action index.nextState hpos
327   have hcount :
328       summary.aggregateTransitionCount index.state index.action index.nextState =
329         adaptiveCumulativeAggregateTransitionCountAt trajectory index.round
330           index.state index.action index.nextState := rfl
331   have hvisit :
332       summary.aggregateVisitCount index.state index.action =
333         adaptiveCumulativeAggregateVisitCountAt trajectory index.round
334           index.state index.action := rfl
335   have hvisitPos : 0 < visits := by
336     change 0 < ((index.count + 1 : Nat) : Real)
337     positivity
338   rw [hmass, hcount, hvisit, hactual]
339   change |(adaptiveCumulativeAggregateTransitionCountAt trajectory index.round
340       index.state index.action index.nextState : Real) / visits - truth| < _
341   have hrewrite :
342       (adaptiveCumulativeAggregateTransitionCountAt trajectory index.round
343           index.state index.action index.nextState : Real) / visits - truth =
344         ((adaptiveCumulativeAggregateTransitionCountAt trajectory index.round
345             index.state index.action index.nextState : Real) - visits * truth) /
346           visits := by
347     field_simp
348   rw [hrewrite, abs_div, abs_of_pos hvisitPos]
349   apply (div_lt_div_iff_of_pos_right hvisitPos).2
350   simpa [visits, truth, hactual] using hresidual

352 /-- The exact generated singleton family yields the self-bounding transition
353 value estimate used in the UCBVI recursion. -/
354 theorem abs_empiricalTransition_integral_sub_le_transitionValue_div_thirtyTwo_add
355     {mdp : MDP State Action} {initialState : Measure State}
356     [IsProbabilityMeasure initialState]
357     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
358     {episodes : Nat} {logBudget : Real}
359     {trajectory : EpisodeBatchTrajectory mdp 1}
360     (htrajectory : trajectory ∉
361       simultaneousTransitionFailureEvent source episodes logBudget)
362     (defaultState : State) (round : Fin episodes)
363     (state : State) (action : Action)
364     (count : Fin (episodes * mdp.horizon))
365     (hactual : adaptiveCumulativeAggregateVisitCountAt trajectory round
366         state action = count + 1)
367     (value : State -> Real)
368     (hvalue : forall nextState,
369       value nextState ∈ Set.Icc (0 : Real) mdp.horizon)
370     (hhorizon : 0 < mdp.horizon) (hlog : 0 <= logBudget) :
371     |(∫ nextState, value nextState
372           ∂TransitionCountSummary.aggregateEmpiricalTransitionKernel
373             (adaptiveCumulativeEmpiricalModelStateAt trajectory round).1
374             defaultState (state, action)) -
375         mdp.transitionValue value state action| <=
376       mdp.transitionValue value state action / (32 * mdp.horizon) +
377         66 * Fintype.card State * (mdp.horizon : Real) ^ 2 * logBudget /
378           (count + 1 : Nat) := by
379   let summary := (adaptiveCumulativeEmpiricalModelStateAt trajectory round).1
380   let empiricalMeasure :=
381     summary.aggregateEmpiricalTransitionKernel defaultState (state, action)
382   let truthMeasure := mdp.transition (state, action)
383   let empiricalMass : State -> Real := fun nextState =>
384     empiricalMeasure.real {nextState}
385   let truthMass : State -> Real := fun nextState =>
386     truthMeasure.real {nextState}
387   let visits : Real := (count + 1 : Nat)
388   have hhorizonReal : (1 : Real) <= mdp.horizon := by
389     exact_mod_cast hhorizon
390   have hvisits : 0 < visits := by
391     change 0 < ((count + 1 : Nat) : Real)
392     positivity
393   have hcoordinate : forall nextState,
394       |empiricalMass nextState - truthMass nextState| <=
395         2 * Real.sqrt (2 * logBudget / visits) *
396             Real.sqrt (truthMass nextState) +
397           2 * logBudget / visits := by
398     intro nextState
399     let index : BernsteinCoordinateIndex mdp episodes :=
400       { round := round
401         state := state
402         action := action
403         nextState := nextState
404         count := count }
405     have hraw := abs_empiricalTransitionMass_sub_lt_bernstein
406       source htrajectory defaultState index hactual
407     have hmass : truthMass nextState ∈ Set.Icc (0 : Real) 1 :=
408       ⟨measureReal_nonneg, measureReal_le_one⟩
409     have htune := bernsteinCoordinateThreshold_mul_probability_div_le
410       hmass hlog hvisits
411     exact (le_of_lt (by simpa [index, visits, truthMass, empiricalMass,
412       empiricalMeasure, truthMeasure, MDP.transitionCoordinateVariance,
413       mul_assoc] using hraw)).trans (by
414         simpa [visits, truthMass, MDP.transitionCoordinateVariance,
415           mul_assoc] using htune)
416   have hanalytic :=
417     abs_sum_weight_mul_massError_le_transitionValue_div_thirtyTwo_add
418       truthMass empiricalMass value (mdp.horizon : Real) logBudget visits
419       hhorizonReal hlog hvisits (fun _ => measureReal_nonneg) hvalue hcoordinate
420   letI : IsProbabilityMeasure empiricalMeasure := by
421     dsimp [empiricalMeasure]
422     exact (summary.aggregateEmpiricalTransitionKernel_isMarkov defaultState)
423       |>.isProbabilityMeasure (state, action)
424   have hempiricalIntegrable : Integrable value empiricalMeasure :=
425     integrable_of_fintype _ _ (measurable_of_finite _)
426   have hempirical :
427       (∫ nextState, value nextState ∂empiricalMeasure) =
428         ∑ nextState : State,
429           empiricalMass nextState * value nextState := by
430     rw [integral_fintype hempiricalIntegrable]
431     rfl
432   have htruth :
433       mdp.transitionValue value state action =
434         ∑ nextState : State,
435           value nextState * truthMass nextState := by
436     simpa [truthMass, truthMeasure] using
437       mdp.transitionValue_eq_sum_measureReal value state action
438   rw [hempirical, htruth]
439   have hleft :
440       (∑ nextState : State,
441           empiricalMass nextState * value nextState) -
442           ∑ nextState : State, value nextState * truthMass nextState =
443         ∑ nextState : State,
444           value nextState * (empiricalMass nextState - truthMass nextState) := by
445     rw [← Finset.sum_sub_distrib]
446     apply Finset.sum_congr rfl
447     intro nextState _
448     ring
449   rw [hleft]
450   simpa [visits] using hanalytic
