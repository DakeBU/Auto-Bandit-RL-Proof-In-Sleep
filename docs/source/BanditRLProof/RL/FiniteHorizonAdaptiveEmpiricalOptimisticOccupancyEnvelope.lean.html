4 # Explicit occupancy envelope for the fixed-bonus adaptive empirical plan
5 
6 The current known-reward empirical optimistic plan uses zero reward radius and
7 one fixed transition bonus at every coordinate. This module evaluates its
8 recursive occupancy-radius sum exactly and attaches that explicit envelope to
9 the compiled path-support episode-threshold event.
10 
11 The result is linear in rounds and horizon because the bonus is fixed. It is
12 not a count-dependent statistical regret rate.
13 -/

32 /-- A probability occupancy sum evaluates a constant stage cost exactly. -/
33 theorem MarkovPolicy.occupancySumRemaining_const
34     {mdp : MDP State Action} (policy : MarkovPolicy mdp) (c : Real)
35     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
36     (mu : Measure State) [IsProbabilityMeasure mu] :
37     policy.occupancySumRemaining
38         (fun _remaining _hremaining _state => c)
39         remaining hremaining mu =
40       (remaining : Real) * c := by
41   induction remaining generalizing mu with
42   | zero =>
43       simp [MarkovPolicy.occupancySumRemaining]
44   | succ remaining ih =>
45       let stage : Fin mdp.horizon :=
46         ⟨mdp.horizon - (remaining + 1), by omega⟩
47       let nextMu : Measure State := policy.inducedStateKernel stage ∘ₘ mu
48       letI : IsProbabilityMeasure nextMu := by
49         dsimp [nextMu]
50         infer_instance
51       rw [policy.occupancySumRemaining_succ]
52       have hnext := ih (by omega) nextMu
53       rw [hnext]
54       simp [Nat.cast_add, add_mul, add_comm]

71 /-- Each round's selected-radius occupancy term is the horizon times the fixed cost. -/
72 theorem adaptiveEmpiricalOptimisticPlanAt_occupancySelectedRadiusRemaining_eq
73     {mdp : MDP State Action} {initialState : Measure State}
74     [IsProbabilityMeasure initialState] {episodes : Nat}
75     (trajectory : EpisodeBatchTrajectory mdp episodes)
76     (defaultState : State) (transitionBonus : Real) (round : Nat) :
77     let plan := adaptiveEmpiricalOptimisticPlanAt
78       (mdp := mdp) (episodes := episodes)
79       trajectory defaultState transitionBonus round
80     plan.optimisticPolicy.occupancySumRemaining
81         (fun remaining hremaining state =>
82           2 * plan.selectedRadiusRemaining remaining hremaining state)
83         mdp.horizon le_rfl initialState =
84       (mdp.horizon : Real) * (2 * transitionBonus) := by
85   dsimp only
86   rw [show
87     (fun remaining hremaining state =>
88       2 * (adaptiveEmpiricalOptimisticPlanAt
89         (mdp := mdp) (episodes := episodes)
90         trajectory defaultState transitionBonus round).selectedRadiusRemaining
91           remaining hremaining state) =
92       (fun _remaining _hremaining _state => 2 * transitionBonus) by
93         funext remaining hremaining state
94         rw [adaptiveEmpiricalOptimisticPlanAt_selectedRadiusRemaining]]
95   exact MarkovPolicy.occupancySumRemaining_const
96     (policy :=
97       (adaptiveEmpiricalOptimisticPlanAt
98         (mdp := mdp) (episodes := episodes)
99         trajectory defaultState transitionBonus round).optimisticPolicy)
100     (2 * transitionBonus) mdp.horizon le_rfl initialState
