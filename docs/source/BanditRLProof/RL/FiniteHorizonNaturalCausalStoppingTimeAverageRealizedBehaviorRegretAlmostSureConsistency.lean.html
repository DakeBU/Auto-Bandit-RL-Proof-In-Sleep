5 # Diverging stopping-time almost-sure natural causal consistency
6 
7 This module equips the heterogeneous causal batch trajectory with its dependent
8 `Filtration.piLE` natural filtration. The exact per-batch-normalized,
9 equal-round-weighted average realized behavior-regret process is strongly
10 adapted to that filtration: its value at prefix `r` uses only successor batches
11 with coordinates at most `r`.
12 
13 Mathlib's `measurable_stoppedValue` then makes evaluation at every stopping
14 time measurable. The compiled all-prefix almost-sure theorem is transported
15 through a sequence of stopping times whose `untopA` values diverge. Mathlib
16 maps `⊤.untopA` to an arbitrary natural default, so this divergence premise
17 implies that the sequence is eventually finite while still permitting finitely
18 many `⊤` values. This is pathwise stopped-subsequence consistency, not optional
19 stopping and not an anytime finite-sample rate.
20 -/

69 /-- A trajectory coordinate is measurable at every later natural-filtration level. -/
70 theorem measurable_naturalTrajectory_coordinate_of_le
71     {mdp : MDP State Action} {initialState : Measure State}
72     [IsProbabilityMeasure initialState] {episodes : Nat -> Nat}
73     (source : HeterogeneousAdaptiveStochasticEpisodeBatchSource
74       mdp initialState episodes) {coordinate horizon : Nat}
75     (hcoordinate : coordinate <= horizon) :
76     @Measurable
77       (HeterogeneousStochasticEpisodeBatchTrajectory mdp episodes)
78       (StochasticEpisodeBatch mdp (episodes coordinate))
79       (source.naturalTrajectoryFiltration horizon) inferInstance
80       (fun trajectory => trajectory coordinate) := by
81   have hprefix :
82       @Measurable
83         (HeterogeneousStochasticEpisodeBatchTrajectory mdp episodes)
84         ((i : Finset.Iic horizon) ->
85           StochasticEpisodeBatch mdp (episodes i))
86         (source.naturalTrajectoryFiltration horizon) inferInstance
87         (Preorder.frestrictLe horizon) := by
88     rw [naturalTrajectoryFiltration_apply,
89       Filtration.piLE_eq_comap_frestrictLe]
90     exact Measurable.of_comap_le le_rfl
91   let index : Finset.Iic horizon :=
92     ⟨coordinate, Finset.mem_Iic.mpr hcoordinate⟩
93   have heval :
94       Measurable
95         (fun history : (i : Finset.Iic horizon) ->
96             StochasticEpisodeBatch mdp (episodes i) =>
97           history index) :=
98     measurable_pi_apply index
99   simpa [index, Preorder.frestrictLe] using heval.comp hprefix
