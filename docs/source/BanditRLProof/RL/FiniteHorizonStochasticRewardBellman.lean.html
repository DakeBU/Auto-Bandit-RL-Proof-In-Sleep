6 /-!
7 # Stochastic reward kernels preserve finite-horizon mean planning
8 
9 This module extends the finite-horizon MDP planning surface with a separate
10 Real reward kernel.  The compatibility contract says that every selected
11 reward is integrable and has the deterministic `mdp.reward` field as its mean.
12 The product with the transition kernel models conditional independence of the
13 reward and next state given the current state-action pair.
14 
15 The resulting one-step, policy, and backward stochastic Bellman values equal
16 the existing mean-reward definitions.  This is a planning transport theorem;
17 it does not construct a stochastic-reward trajectory law or a concentration
18 bound.
19 -/

35 /--
36 A stochastic Real reward kernel whose selected rewards are integrable and
37 have the mean stored in the finite-horizon MDP reward field.
38 -/
39 structure MeanCompatibleRewardKernel (mdp : MDP State Action) where
40   rewardKernel : RewardKernel.MarkovRewardKernel (Prod State Action) Real
41   integrable_reward : forall state action,
42     Integrable (fun reward : Real => reward)
43       (rewardKernel.kernel (state, action))
44   integral_reward_eq : forall state action,
45     integral (rewardKernel.kernel (state, action))
46         (fun reward : Real => reward) =
47       mdp.reward state action

53 /-- The conditionally independent joint law of reward and next state. -/
54 noncomputable def rewardNextStateKernel
55     (source : MeanCompatibleRewardKernel mdp) :
56     ProbabilityTheory.Kernel (Prod State Action) (Prod Real State) :=
57   ProbabilityTheory.Kernel.prod source.rewardKernel.kernel mdp.transition
58 
59 instance instRewardNextStateKernelIsMarkovKernel
60     (source : MeanCompatibleRewardKernel mdp) :
61     ProbabilityTheory.IsMarkovKernel source.rewardNextStateKernel := by
62   unfold rewardNextStateKernel
63   infer_instance

65 /-- Expected sampled reward plus continuation value under the joint kernel. -/
66 noncomputable def stochasticBellmanQ
67     (source : MeanCompatibleRewardKernel mdp)
68     (value : State -> Real) (state : State) (action : Action) : Real :=
69   integral (source.rewardNextStateKernel (state, action))
70     (fun pair : Prod Real State => pair.1 + value pair.2)
71 
72 /-- The sampled one-step return is integrable for measurable continuation values. -/
73 theorem integrable_reward_add_value
74     (source : MeanCompatibleRewardKernel mdp)
75     {value : State -> Real} (hvalue : Measurable value)
76     (state : State) (action : Action) :
77     Integrable (fun pair : Prod Real State => pair.1 + value pair.2)
78       (source.rewardNextStateKernel (state, action)) := by
79   haveI : IsProbabilityMeasure
80       (source.rewardKernel.kernel (state, action)) :=
81     RewardKernel.isProbabilityMeasure_apply source.rewardKernel (state, action)
82   haveI : IsProbabilityMeasure (mdp.transition (state, action)) := by
83     infer_instance
84   rw [rewardNextStateKernel, ProbabilityTheory.Kernel.prod_apply]
85   exact
86     (source.integrable_reward state action).comp_fst
87         (mdp.transition (state, action)) |>.add
88       ((integrable_of_fintype (mdp.transition (state, action)) value hvalue).comp_snd
89         (source.rewardKernel.kernel (state, action)))

91 /-- Sampling a mean-compatible reward preserves the existing Bellman action value. -/
92 theorem stochasticBellmanQ_eq_bellmanQ
93     (source : MeanCompatibleRewardKernel mdp)
94     {value : State -> Real} (hvalue : Measurable value)
95     (state : State) (action : Action) :
96     source.stochasticBellmanQ value state action =
97       mdp.bellmanQ value state action := by
98   haveI : IsProbabilityMeasure
99       (source.rewardKernel.kernel (state, action)) :=
100     RewardKernel.isProbabilityMeasure_apply source.rewardKernel (state, action)
101   haveI : IsProbabilityMeasure (mdp.transition (state, action)) := by
102     infer_instance
103   have hreward := source.integrable_reward state action
104   have hnext :=
105     integrable_of_fintype (mdp.transition (state, action)) value hvalue
106   have hjoint :
107       Integrable (fun pair : Prod Real State => pair.1 + value pair.2)
108         ((source.rewardKernel.kernel (state, action)).prod
109           (mdp.transition (state, action))) :=
110     (hreward.comp_fst (mdp.transition (state, action))).add
111       (hnext.comp_snd (source.rewardKernel.kernel (state, action)))
112   rw [stochasticBellmanQ, rewardNextStateKernel,
113     ProbabilityTheory.Kernel.prod_apply]
114   rw [integral_prod _ hjoint]
115   have hinner (reward : Real) :
116       integral (mdp.transition (state, action))
117           (fun nextState => reward + value nextState) =
118         reward + mdp.transitionValue value state action := by
119     rw [integral_add (integrable_const _) hnext]
120     simp [MDP.transitionValue]
121   simp_rw [hinner]
122   rw [integral_add hreward (integrable_const _)]
123   simp [source.integral_reward_eq, MDP.bellmanQ, MDP.transitionValue]
124 
125 /-- Policy expectation formed from the sampled stochastic Bellman action value. -/
126 noncomputable def stochasticBellman
127     (source : MeanCompatibleRewardKernel mdp)
128     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
129     (value : State -> Real) (state : State) : Real :=
130   integral (policy.actionKernel stage state)
131     (fun action => source.stochasticBellmanQ value state action)
132 
133 /-- The stochastic policy Bellman operator equals the existing mean operator. -/
134 theorem stochasticBellman_eq_bellman
135     (source : MeanCompatibleRewardKernel mdp)
136     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
137     {value : State -> Real} (hvalue : Measurable value) :
138     source.stochasticBellman policy stage value =
139       policy.bellman stage value := by
140   funext state
141   unfold stochasticBellman MarkovPolicy.bellman
142   apply integral_congr_ae
143   exact Filter.Eventually.of_forall fun action =>
144     source.stochasticBellmanQ_eq_bellmanQ hvalue state action
145 
146 /-- Backward policy value computed with sampled stochastic reward laws. -/
147 noncomputable def stochasticValueRemaining
148     (source : MeanCompatibleRewardKernel mdp)
149     (policy : MarkovPolicy mdp) :
150     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
151   | 0, _ => fun _ => 0
152   | remaining + 1, hremaining =>
153       source.stochasticBellman policy
154         ⟨mdp.horizon - (remaining + 1), by omega⟩
155         (source.stochasticValueRemaining policy remaining (by omega))
156 
157 /-- Stochastic backward policy evaluation equals mean-reward evaluation. -/
158 theorem stochasticValueRemaining_eq_valueRemaining
159     (source : MeanCompatibleRewardKernel mdp)
160     (policy : MarkovPolicy mdp)
161     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
162     source.stochasticValueRemaining policy remaining hremaining =
163       policy.valueRemaining remaining hremaining := by
164   induction remaining with
165   | zero =>
166       rfl
167   | succ remaining ih =>
168       rw [stochasticValueRemaining, MarkovPolicy.valueRemaining]
169       rw [source.stochasticBellman_eq_bellman]
170       · congr 1
171         exact ih (by omega)
172       · rw [ih (by omega)]
173         exact policy.measurable_valueRemaining remaining (by omega)
174 
175 /-- Stochastic policy value at a chronological stage. -/
176 noncomputable def stochasticValueAt
177     (source : MeanCompatibleRewardKernel mdp)
178     (policy : MarkovPolicy mdp)
179     (stage : Nat) (_hstage : stage <= mdp.horizon) : State -> Real :=
180   source.stochasticValueRemaining policy
181     (mdp.horizon - stage) (Nat.sub_le _ _)
182 
183 /-- Every chronological stochastic value equals the existing policy value. -/
184 theorem stochasticValueAt_eq_valueAt
185     (source : MeanCompatibleRewardKernel mdp)
186     (policy : MarkovPolicy mdp)
187     (stage : Nat) (hstage : stage <= mdp.horizon) :
188     source.stochasticValueAt policy stage hstage =
189       policy.valueAt stage hstage := by
190   exact source.stochasticValueRemaining_eq_valueRemaining policy _ _

192 /-- The deterministic MDP reward, viewed as a kernel, is mean-compatible. -/
193 noncomputable def deterministic (mdp : MDP State Action) :
194     MeanCompatibleRewardKernel mdp where
195   rewardKernel := RewardKernel.deterministic
196     (Function.uncurry mdp.reward) mdp.measurable_reward
197   integrable_reward := by
198     intro state action
199     rw [RewardKernel.deterministic,
200       ProbabilityTheory.Kernel.deterministic_apply]
201     exact integrable_dirac' measurable_id.stronglyMeasurable (by simp)
202   integral_reward_eq := by
203     intro state action
204     rw [RewardKernel.deterministic,
205       ProbabilityTheory.Kernel.deterministic_apply]
206     exact integral_dirac' (fun reward : Real => reward)
207       (mdp.reward state action) measurable_id.stronglyMeasurable

209 /--
210 Terminal mean-planning transport: the product kernel is Markov and all sampled
211 Bellman and backward values agree with the existing deterministic-mean model.
212 -/
213 theorem meanPlanningTransport
214     (source : MeanCompatibleRewardKernel mdp)
215     (policy : MarkovPolicy mdp) :
216     ProbabilityTheory.IsMarkovKernel source.rewardNextStateKernel ∧
217     (forall (value : State -> Real), Measurable value ->
218       forall state action,
219         source.stochasticBellmanQ value state action =
220           mdp.bellmanQ value state action) ∧
221     (forall (stage : Fin mdp.horizon) (value : State -> Real),
222       Measurable value ->
223         source.stochasticBellman policy stage value =
224           policy.bellman stage value) ∧
225     (forall (remaining : Nat) (hremaining : remaining <= mdp.horizon),
226       source.stochasticValueRemaining policy remaining hremaining =
227         policy.valueRemaining remaining hremaining) ∧
228     (forall (stage : Nat) (hstage : stage <= mdp.horizon),
229       source.stochasticValueAt policy stage hstage =
230         policy.valueAt stage hstage) := by
231   refine ⟨inferInstance, ?_, ?_, ?_, ?_⟩
232   · intro value hvalue state action
233     exact source.stochasticBellmanQ_eq_bellmanQ hvalue state action
234   · intro stage value hvalue
235     exact source.stochasticBellman_eq_bellman policy stage hvalue
236   · intro remaining hremaining
237     exact source.stochasticValueRemaining_eq_valueRemaining policy remaining hremaining
238   · intro stage hstage
239     exact source.stochasticValueAt_eq_valueAt policy stage hstage
