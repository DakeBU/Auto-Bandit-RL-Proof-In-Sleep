4 /-!
5 # Explicit stochastic-reward iid empirical-model calibration
6 
7 This module replaces the coordinatewise margin and cover inputs of the
8 stochastic-reward iid empirical-model terminal by one common expected-count
9 floor and one scalar half-contraction condition. The reward and transition
10 budgets are explicit functions of that floor.
11 -/

27 /-- Uniform reward-mean radius obtained from one common expected-count floor. -/
28 noncomputable def uniformFloorStochasticRewardCoordinateRadius
29     (mdp : MDP State Action) (episodes : Nat) (varianceProxy : NNReal)
30     (countDelta rewardDelta visitFloor : Real) : Real :=
31   MDP.MeanCompatibleRewardKernel.simultaneousRewardSumConfidenceRadius
32       mdp episodes varianceProxy rewardDelta /
33     ((episodes : Real) * visitFloor -
34       simultaneousCountConfidenceRadius mdp episodes countDelta)

39 /-- The uniform reward radius is nonnegative under the strict count margin. -/
40 theorem uniformFloorStochasticRewardCoordinateRadius_nonneg
41     {mdp : MDP State Action} {episodes : Nat} {varianceProxy : NNReal}
42     {countDelta rewardDelta visitFloor : Real}
43     (hmargin : simultaneousCountConfidenceRadius mdp episodes countDelta <
44       (episodes : Real) * visitFloor) :
45     0 <= uniformFloorStochasticRewardCoordinateRadius mdp episodes varianceProxy
46   countDelta rewardDelta visitFloor := by
47   unfold uniformFloorStochasticRewardCoordinateRadius
48   exact div_nonneg
49     (Concentration.subGaussianSumConfidenceRadius_nonneg _ _)
50     (sub_nonneg.mpr hmargin.le)

52 /-- The explicit transition budget paired with the uniform reward budget. -/
53 def uniformFloorStochasticTransitionBudget
54     (rewardBound rewardBudget : Real) : Real :=
55   rewardBound + 2 * rewardBudget
