4 /-!
5 # Adaptive cumulative count martingale confidence
6 
7 This module supplies the statistical producer required by the cumulative
8 count-radius planner.  Each raw episode-batch count is centered by its
9 history-kernel integral.  The exact adaptive iid batch law then gives a
10 conditionally sub-Gaussian increment with the within-batch Bernoulli proxy,
11 rather than the weaker whole-batch bounded-range proxy.
12 
13 The final route unions cumulative prefix tails over the finite round/count
14 coordinate family and feeds the resulting empirical-transition coordinate
15 confidence into the compiled optimism and recommended-policy expected-regret
16 terminal.  It does not identify behavior or realized regret.
17 -/

85 /-- The kernel integral of a selected raw count is its policy mean. -/
86 theorem integral_rawCount_iidEpisodeBatchMeasure
87     {mdp : MDP State Action} (coordinate : CountCoordinate mdp)
88     (policy : MarkovPolicy mdp)
89     (initialState : Measure State) [IsProbabilityMeasure initialState]
90     (episodes : Nat) :
91     integral (policy.iidEpisodeBatchMeasure initialState episodes)
92         coordinate.rawCount =
93       coordinate.policyMean policy initialState episodes := by
94   cases coordinate with
95   | visit stage state action =>
96       let mu := policy.iidEpisodeBatchMeasure initialState episodes
97       have hint : forall episode : Fin episodes,
98           Integrable
99             (fun batch : EpisodeBatch mdp episodes =>
100               EpisodeStep.visitIndicator state action (batch episode stage)) mu := by
101         intro episode
102         have hmeas : Measurable
103             (fun batch : EpisodeBatch mdp episodes =>
104               EpisodeStep.visitIndicator state action (batch episode stage)) :=
105           (EpisodeStep.measurable_visitIndicator state action).comp
106             ((measurable_pi_apply stage).comp (measurable_pi_apply episode))
107         refine Integrable.of_bound hmeas.aestronglyMeasurable 1 ?_
108         exact Filter.Eventually.of_forall fun batch => by
109           rw [Real.norm_eq_abs]
110           have hmem :=
111             EpisodeStep.visitIndicator_mem_Icc state action (batch episode stage)
112           exact abs_le.2 ⟨by linarith [hmem.1], hmem.2⟩
113       calc
114         integral mu (rawCount (.visit stage state action)) =
115             integral mu (fun batch : EpisodeBatch mdp episodes =>
116               ∑ episode : Fin episodes,
117                 EpisodeStep.visitIndicator state action (batch episode stage)) := by
118               apply integral_congr_ae
119               exact Filter.Eventually.of_forall fun batch => by
120                 simpa [rawCount] using
121                   (MarkovPolicy.sum_visitIndicator_eq_cast_visitCount
122                     batch stage state action).symm
123         _ = ∑ episode : Fin episodes,
124               integral mu (fun batch : EpisodeBatch mdp episodes =>
125                 EpisodeStep.visitIndicator state action (batch episode stage)) := by
126               exact integral_finset_sum Finset.univ (fun episode _ => hint episode)
127         _ = policyMean (.visit stage state action) policy initialState episodes := by
128               simp [mu, policyMean,
129                 policy.integral_visitIndicator_iidEpisodeBatchMeasure_eval
130                   initialState]
131   | transition stage state action nextState =>
132       let mu := policy.iidEpisodeBatchMeasure initialState episodes
133       have hint : forall episode : Fin episodes,
134           Integrable
135             (fun batch : EpisodeBatch mdp episodes =>
136               EpisodeStep.transitionIndicator state action nextState
137                 (batch episode stage)) mu := by
138         intro episode
139         have hmeas : Measurable
140             (fun batch : EpisodeBatch mdp episodes =>
141               EpisodeStep.transitionIndicator state action nextState
142                 (batch episode stage)) :=
143           (EpisodeStep.measurable_transitionIndicator state action nextState).comp
144             ((measurable_pi_apply stage).comp (measurable_pi_apply episode))
145         refine Integrable.of_bound hmeas.aestronglyMeasurable 1 ?_
146         exact Filter.Eventually.of_forall fun batch => by
147           rw [Real.norm_eq_abs]
148           have hmem := EpisodeStep.transitionIndicator_mem_Icc
149             state action nextState (batch episode stage)
150           exact abs_le.2 ⟨by linarith [hmem.1], hmem.2⟩
151       calc
152         integral mu (rawCount (.transition stage state action nextState)) =
153             integral mu (fun batch : EpisodeBatch mdp episodes =>
154               ∑ episode : Fin episodes,
155                 EpisodeStep.transitionIndicator state action nextState
156                   (batch episode stage)) := by
157               apply integral_congr_ae
158               exact Filter.Eventually.of_forall fun batch => by
159                 simpa [rawCount] using
160                   (MarkovPolicy.sum_transitionIndicator_eq_cast_transitionCount
161                     batch stage state action nextState).symm
162         _ = ∑ episode : Fin episodes,
163               integral mu (fun batch : EpisodeBatch mdp episodes =>
164                 EpisodeStep.transitionIndicator state action nextState
165                   (batch episode stage)) := by
166               exact integral_finset_sum Finset.univ (fun episode _ => hint episode)
167         _ = policyMean (.transition stage state action nextState)
168               policy initialState episodes := by
169               simp [mu, policyMean,
170                 policy.integral_transitionIndicator_iidEpisodeBatchMeasure_eval
171                   initialState]
