4 # Finite-horizon optimistic Bellman certificates
5 
6 This module isolates the deterministic dynamic-programming interface needed by
7 optimistic finite-horizon RL.  A certificate supplies upper values with zero
8 terminal value and a one-step upper Bellman inequality.  Backward induction
9 turns that local contract into global optimism.  For any Markov policy, the
10 upper-value minus policy-value difference is then exactly a recursive sum of
11 Bellman residuals under the true policy-induced state laws.  A pointwise bonus
12 bound on those residuals yields a finite-episode expected-regret bound.
13 -/

30 /-- The finite-action optimal Bellman operator is monotone in its continuation value. -/
31 theorem optimalBellman_mono
32     (mdp : MDP State Action) {left right : State -> Real}
33     (hle : forall state, left state <= right state) (state : State) :
34     mdp.optimalBellman left state <= mdp.optimalBellman right state := by
35   calc
36     mdp.optimalBellman left state =
37         mdp.bellmanQ left state (mdp.optimalAction left state) := rfl
38     _ <= mdp.bellmanQ right state (mdp.optimalAction left state) :=
39       mdp.bellmanQ_mono hle state (mdp.optimalAction left state)
40     _ <= mdp.optimalBellman right state :=
41       mdp.bellmanQ_le_optimalBellman right state (mdp.optimalAction left state)

43 /--
44 An upper-value plan whose terminal value is zero and whose successor surface
45 dominates one true optimal Bellman backup of its tail surface.
46 -/
47 structure OptimisticBellmanCertificate (mdp : MDP State Action) where
48   upperValueRemaining :
49     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
50   upperValueRemaining_zero :
51     upperValueRemaining 0 (Nat.zero_le mdp.horizon) = fun _ => 0
52   optimalBellman_le_upperValueRemaining_succ :
53     forall (remaining : Nat)
54       (hremaining : remaining + 1 <= mdp.horizon) (state : State),
55       mdp.optimalBellman
56           (upperValueRemaining remaining (by omega)) state <=
57         upperValueRemaining (remaining + 1) hremaining state
58 
59 /-- The true optimal value itself is the canonical exact optimistic certificate. -/
60 noncomputable def optimalBellmanCertificate (mdp : MDP State Action) :
61     OptimisticBellmanCertificate mdp where
62   upperValueRemaining := mdp.optimalValueRemaining
63   upperValueRemaining_zero := rfl
64   optimalBellman_le_upperValueRemaining_succ := by
65     intro remaining hremaining state
66     rw [optimalValueRemaining]

70 /-- Every upper-value surface is measurable on the finite discrete state space. -/

72     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)

77 /-- Local Bellman optimism implies the upper values dominate the true optimal values. -/
78 theorem optimalValueRemaining_le_upperValueRemaining
79     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
80     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
81     (state : State) :
82     mdp.optimalValueRemaining remaining hremaining state <=
83       certificate.upperValueRemaining remaining hremaining state := by
84   induction remaining generalizing state with
85   | zero =>
86       rw [MDP.optimalValueRemaining, certificate.upperValueRemaining_zero]
87   | succ remaining ih =>
88       rw [MDP.optimalValueRemaining]
89       exact
90         (mdp.optimalBellman_mono
91           (fun nextState => ih (by omega) nextState) state).trans
92         (certificate.optimalBellman_le_upperValueRemaining_succ
93           remaining hremaining state)

100 /--
101 Recursive occupancy sum for a remaining-horizon-indexed stage cost.  At a
102 successor step, cost index `remaining` is chronological stage
103 `horizon - (remaining + 1)`.
104 -/
105 noncomputable def occupancySumRemaining
106     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
107     (cost : (remaining : Nat) ->
108       remaining + 1 <= mdp.horizon -> State -> Real) :
109     (remaining : Nat) -> remaining <= mdp.horizon -> Measure State -> Real
110   | 0, _, _ => 0
111   | remaining + 1, hremaining, mu =>
112       let stage : Fin mdp.horizon :=
113         ⟨mdp.horizon - (remaining + 1), by omega⟩
114       (∫ state, cost remaining hremaining state ∂mu) +
115         policy.occupancySumRemaining cost remaining (by omega)
116           (policy.inducedStateKernel stage ∘ₘ mu)
117

119 /-- Successor equation for a generic recursive occupancy sum. -/
120 theorem occupancySumRemaining_succ

122     (cost : (remaining : Nat) ->
123       remaining + 1 <= mdp.horizon -> State -> Real)
124     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
125     (mu : Measure State) :

128         policy.occupancySumRemaining cost remaining (by omega)
129           (policy.inducedStateKernel
130             ⟨mdp.horizon - (remaining + 1), by omega⟩ ∘ₘ mu) := by

133 omit [Nonempty Action] in
134 /-- Pointwise domination of stage costs lifts to domination of their occupancy sums. -/
135 theorem occupancySumRemaining_mono
136     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
137     {left right : (remaining : Nat) ->
138       remaining + 1 <= mdp.horizon -> State -> Real}
139     (hle : forall (remaining : Nat)
140       (hremaining : remaining + 1 <= mdp.horizon) (state : State),
141       left remaining hremaining state <= right remaining hremaining state)
142     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
143     (mu : Measure State) [IsProbabilityMeasure mu] :
144     policy.occupancySumRemaining left remaining hremaining mu <=
145       policy.occupancySumRemaining right remaining hremaining mu := by
146   induction remaining generalizing mu with
147   | zero =>
148       simp [occupancySumRemaining]
149   | succ remaining ih =>
150       let stage : Fin mdp.horizon :=
151         ⟨mdp.horizon - (remaining + 1), by omega⟩
152       let nextMu : Measure State := policy.inducedStateKernel stage ∘ₘ mu
153       letI : IsProbabilityMeasure nextMu := by
154         dsimp [nextMu]
155         infer_instance
156       rw [occupancySumRemaining]
157       apply add_le_add
158       · apply integral_mono
159         · exact integrable_of_fintype _ _ (measurable_of_finite _)
160         · exact integrable_of_fintype _ _ (measurable_of_finite _)
161         · exact fun state => hle remaining hremaining state
162       · exact ih (by omega) nextMu
163 
164 end MarkovPolicy

168 /-- Policy Bellman residual of an optimistic upper-value plan. -/
169 noncomputable def policyBellmanResidual
170     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
171     (policy : MarkovPolicy mdp) (remaining : Nat)
172     (hremaining : remaining + 1 <= mdp.horizon) (state : State) : Real :=
173   certificate.upperValueRemaining (remaining + 1) hremaining state -
174     policy.bellman ⟨mdp.horizon - (remaining + 1), by omega⟩
175       (certificate.upperValueRemaining remaining (by omega)) state
176 
177 /-- The optimistic policy Bellman residual is measurable. -/
178 theorem measurable_policyBellmanResidual
179     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
180     (policy : MarkovPolicy mdp) (remaining : Nat)
181     (hremaining : remaining + 1 <= mdp.horizon) :
182     Measurable (certificate.policyBellmanResidual policy remaining hremaining) := by
183   exact (certificate.measurable_upperValueRemaining (remaining + 1) hremaining).sub
184     (policy.measurable_bellman _
185       (certificate.measurable_upperValueRemaining remaining (by omega)))
186 
187 /-- Every policy Bellman residual of an optimistic certificate is nonnegative. -/
188 theorem policyBellmanResidual_nonneg
189     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
190     (policy : MarkovPolicy mdp) (remaining : Nat)
191     (hremaining : remaining + 1 <= mdp.horizon) (state : State) :
192     0 <= certificate.policyBellmanResidual policy remaining hremaining state := by
193   unfold policyBellmanResidual
194   apply sub_nonneg.mpr
195   exact
196     (policy.bellman_le_optimalBellman
197       ⟨mdp.horizon - (remaining + 1), by omega⟩
198       (certificate.upperValueRemaining remaining (by omega)) state).trans
199     (certificate.optimalBellman_le_upperValueRemaining_succ
200       remaining hremaining state)

202 /-- Recursive true-occupancy sum of the certificate's policy Bellman residuals. -/
203 noncomputable def residualOccupancyRemaining
204     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
205     (policy : MarkovPolicy mdp) (remaining : Nat)
206     (hremaining : remaining <= mdp.horizon) (mu : Measure State) : Real :=
207   policy.occupancySumRemaining
208     (certificate.policyBellmanResidual policy) remaining hremaining mu

210 /--
211 The residual occupancy sum is exactly the integrated difference between the
212 certificate upper value and the supplied policy value.
213 -/
214 theorem residualOccupancyRemaining_eq_integral_upperValueRemaining_sub_valueRemaining
215     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
216     (policy : MarkovPolicy mdp) (remaining : Nat)
217     (hremaining : remaining <= mdp.horizon)
218     (mu : Measure State) [IsProbabilityMeasure mu] :
219     certificate.residualOccupancyRemaining policy remaining hremaining mu =
220       ∫ state, certificate.upperValueRemaining remaining hremaining state -
221         policy.valueRemaining remaining hremaining state ∂mu := by
222   induction remaining generalizing mu with
223   | zero =>
224       simp [residualOccupancyRemaining, MarkovPolicy.occupancySumRemaining,
225         certificate.upperValueRemaining_zero, MarkovPolicy.valueRemaining]
226   | succ remaining ih =>
227       let stage : Fin mdp.horizon :=
228         ⟨mdp.horizon - (remaining + 1), by omega⟩
229       let nextMu : Measure State := policy.inducedStateKernel stage ∘ₘ mu
230       letI : IsProbabilityMeasure nextMu := by
231         dsimp [nextMu]
232         infer_instance
233       unfold residualOccupancyRemaining
234       rw [MarkovPolicy.occupancySumRemaining]
235       change
236         (∫ state,
237             certificate.policyBellmanResidual policy remaining hremaining state ∂mu) +
238           certificate.residualOccupancyRemaining policy remaining (by omega) nextMu = _
239       rw [ih (by omega) nextMu]
240       rw [policy.integral_sub_comp_inducedStateKernel_eq_integral_bellman_sub
241         stage
242         (certificate.upperValueRemaining remaining (by omega))
243         (policy.valueRemaining remaining (by omega)) mu]
244       rw [MarkovPolicy.valueRemaining]
245       unfold policyBellmanResidual
246       have hfirst : Integrable
247           (fun state =>
248             certificate.upperValueRemaining (remaining + 1) hremaining state -
249               policy.bellman stage
250                 (certificate.upperValueRemaining remaining (by omega)) state) mu :=
251         integrable_of_fintype _ _ (measurable_of_finite _)
252       have hsecond : Integrable
253           (fun state =>
254             policy.bellman stage
255                 (certificate.upperValueRemaining remaining (by omega)) state -
256               policy.bellman stage
257                 (policy.valueRemaining remaining (by omega)) state) mu :=
258         integrable_of_fintype _ _ (measurable_of_finite _)
259       rw [← integral_add hfirst hsecond]
260       apply integral_congr_ae
261       filter_upwards [] with state
262       ring

264 /-- For the canonical exact certificate, residual occupancy is the previously compiled optimality
265 gap occupancy sum. -/
266 theorem optimalBellmanCertificate_residualOccupancyRemaining_eq_occupancyGapRemaining
267     {mdp : MDP State Action} (policy : MarkovPolicy mdp)

269     (mu : Measure State) [IsProbabilityMeasure mu] :

277 /-- The residual occupancy sum is nonnegative. -/
278 theorem residualOccupancyRemaining_nonneg
279     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
280     (policy : MarkovPolicy mdp) (remaining : Nat)
281     (hremaining : remaining <= mdp.horizon)
282     (mu : Measure State) [IsProbabilityMeasure mu] :
283     0 <= certificate.residualOccupancyRemaining policy remaining hremaining mu := by
284   rw [certificate.residualOccupancyRemaining_eq_integral_upperValueRemaining_sub_valueRemaining]
285   apply integral_nonneg
286   intro state
287   apply sub_nonneg.mpr
288   exact
289     (policy.valueRemaining_le_optimalValueRemaining remaining hremaining state).trans
290     (certificate.optimalValueRemaining_le_upperValueRemaining
291       remaining hremaining state)

293 /-- Single-episode expected regret is bounded by the optimistic residual occupancy sum. -/
294 theorem expectedRegret_le_residualOccupancyRemaining
295     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
296     (policy : MarkovPolicy mdp) (initialState : Measure State)
297     [IsProbabilityMeasure initialState] :
298     policy.expectedRegret initialState <=
299       certificate.residualOccupancyRemaining
300         policy mdp.horizon le_rfl initialState := by
301   rw [policy.expectedRegret_eq_integral_optimalValueAt_sub_valueAt]
302   rw [certificate.residualOccupancyRemaining_eq_integral_upperValueRemaining_sub_valueRemaining]
303   apply integral_mono
304   · exact integrable_of_fintype _ _ (measurable_of_finite _)
305   · exact integrable_of_fintype _ _ (measurable_of_finite _)
306   · intro state
307     exact sub_le_sub_right
308       (certificate.optimalValueRemaining_le_upperValueRemaining
309         mdp.horizon le_rfl state)
310       (policy.valueAt 0 (Nat.zero_le mdp.horizon) state)

312 /-- A pointwise bonus bound on Bellman residuals bounds the residual occupancy sum. -/
313 theorem residualOccupancyRemaining_le_occupancySumRemaining
314     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
315     (policy : MarkovPolicy mdp)
316     (bonus : (remaining : Nat) ->
317       remaining + 1 <= mdp.horizon -> State -> Real)
318     (hbonus : forall (remaining : Nat)
319       (hremaining : remaining + 1 <= mdp.horizon) (state : State),
320       certificate.policyBellmanResidual policy remaining hremaining state <=
321         bonus remaining hremaining state)
322     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
323     (mu : Measure State) [IsProbabilityMeasure mu] :
324     certificate.residualOccupancyRemaining policy remaining hremaining mu <=
325       policy.occupancySumRemaining bonus remaining hremaining mu := by
326   unfold residualOccupancyRemaining
327   exact policy.occupancySumRemaining_mono hbonus remaining hremaining mu

329 /--
330 Route endpoint: local true-Bellman optimism induces global value optimism;
331 single-episode expected regret is bounded by the true-occupancy residual sum;
332 and any pointwise bonus dominating those residuals bounds regret.
333 -/
334 theorem expectedRegret_le_residual_le_occupancyBonusRemaining
335     {mdp : MDP State Action} (certificate : OptimisticBellmanCertificate mdp)
336     (policy : MarkovPolicy mdp) (initialState : Measure State)
337     [IsProbabilityMeasure initialState]
338     (bonus : (remaining : Nat) ->
339       remaining + 1 <= mdp.horizon -> State -> Real)
340     (hbonus : forall (remaining : Nat)
341       (hremaining : remaining + 1 <= mdp.horizon) (state : State),
342       certificate.policyBellmanResidual policy remaining hremaining state <=
343         bonus remaining hremaining state) :
344     0 <= certificate.residualOccupancyRemaining
345         policy mdp.horizon le_rfl initialState /\
346       policy.expectedRegret initialState <=
347         certificate.residualOccupancyRemaining
348           policy mdp.horizon le_rfl initialState /\
349       certificate.residualOccupancyRemaining
350           policy mdp.horizon le_rfl initialState <=
351         policy.occupancySumRemaining bonus mdp.horizon le_rfl initialState /\
352       policy.expectedRegret initialState <=
353         policy.occupancySumRemaining bonus mdp.horizon le_rfl initialState := by
354   have hregret := certificate.expectedRegret_le_residualOccupancyRemaining
355     policy initialState
356   have hresidual := certificate.residualOccupancyRemaining_le_occupancySumRemaining
357     policy bonus hbonus mdp.horizon le_rfl initialState
358   exact ⟨certificate.residualOccupancyRemaining_nonneg
359       policy mdp.horizon le_rfl initialState,
360     hregret, hresidual, hregret.trans hresidual⟩
