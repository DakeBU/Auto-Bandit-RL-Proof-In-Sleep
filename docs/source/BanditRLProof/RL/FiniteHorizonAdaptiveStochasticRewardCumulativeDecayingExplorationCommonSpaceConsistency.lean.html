5 /-!
6 # Stochastic common-space realized behavior consistency
7 
8 The compiled stochastic cumulative route gives a sharp certificate on each
9 scheduled finite-window trajectory space. This module places those complete
10 finite-window experiments on one dependent infinite product and proves that
11 the scheduled stochastic realized-regret process converges to zero in measure.
12 
13 The coupling is intentionally the independent-coordinate product coupling. It
14 has the exact scheduled laws as marginals, but it is not a nested causal stream
15 of one online run and yields no pathwise, almost-sure, or anytime conclusion.
16 -/

97 An expected-regret upper bound and a two-sided global return-deviation bound
98 control the absolute stochastic realized average regret.
99 -/
100 theorem abs_realizedSuccessorAverageRegret_le_of_expected_le_of_deviation_abs_le
101     {mdp : MDP State Action} {initialState : Measure State}
102     [IsProbabilityMeasure initialState] {episodes : Nat}
103     (source : AdaptiveStochasticEpisodeBatchSource mdp initialState episodes)
104     (trajectory : StochasticEpisodeBatchTrajectory mdp episodes)
105     (rounds : Nat) (hrounds : 0 < rounds) (hepisodes : 0 < episodes)
106     (expectedBound deviationBound : Real)
107     (hexpected :
108       source.successorExpectedAverageRegret trajectory rounds <= expectedBound)
109     (hdeviation :
110       |source.cumulativeSuccessorGlobalReturnDeviation rounds trajectory| <=
111         deviationBound) :
112     |source.realizedSuccessorAverageRegret trajectory rounds| <=
113       expectedBound + deviationBound / ((episodes : Real) * (rounds : Real)) := by
114   have hdenom : 0 < (episodes : Real) * (rounds : Real) := by positivity
115   rw [source.realizedSuccessorAverageRegret_eq_expected_sub_deviation
116     trajectory rounds hrounds hepisodes]
117   calc
118     |source.successorExpectedAverageRegret trajectory rounds -
119         source.cumulativeSuccessorGlobalReturnDeviation rounds trajectory /
120           ((episodes : Real) * (rounds : Real))| <=
121         |source.successorExpectedAverageRegret trajectory rounds| +
122           |source.cumulativeSuccessorGlobalReturnDeviation rounds trajectory /
123             ((episodes : Real) * (rounds : Real))| := abs_sub _ _
124     _ = source.successorExpectedAverageRegret trajectory rounds +
125           |source.cumulativeSuccessorGlobalReturnDeviation rounds trajectory| /
126             ((episodes : Real) * (rounds : Real)) := by
127       rw [abs_of_nonneg
128         (source.successorExpectedAverageRegret_nonneg trajectory rounds),
129         abs_div, abs_of_pos hdenom]
130     _ <= expectedBound + deviationBound / ((episodes : Real) * (rounds : Real)) :=
131       add_le_add hexpected (div_le_div_of_nonneg_right hdeviation hdenom.le)
