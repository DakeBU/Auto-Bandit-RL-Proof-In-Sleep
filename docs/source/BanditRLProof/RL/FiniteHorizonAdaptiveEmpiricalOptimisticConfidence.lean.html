4 /-!
5 # Adaptive empirical optimistic all-coordinate confidence
6 
7 This module calibrates a latest-batch empirical-transition source whose behavior
8 policy uniformly explores around the latest optimistic deterministic table.
9 For every policy that generates a batch, a local contract records genuine
10 expected-visit margins and a finite-state coordinate-radius cover for the fixed
11 transition bonus.  Outside the existing adaptive simultaneous-count event,
12 those contracts produce coordinate confidence for every known-reward empirical
13 plan.
14 
15 The compiled one-episode expected-regret inequalities are summed over the
16 optimistic policies recommended by the generated batches.  Behavior-policy
17 exploration and recommendation regret are deliberately distinct.  This remains
18 a latest-batch, known-reward result with assumed state-action reachability and
19 bonus calibration; it is not behavior-policy regret, realized online regret,
20 or complete UCB-VI.
21 -/

104 /--
105 The known-reward empirical plan has the same explicit linear value envelope as
106 the raw empirical model when the fixed transition bonus is nonnegative.

108 theorem optimisticPlan_upperValueRemaining_abs_le
109     (mdp : MDP State Action) (summary : TransitionCountSummary mdp)
110     (defaultState : State) (rewardBound transitionBonus : Real)
111     (hrewardBound : forall state action,
112       |mdp.reward state action| <= rewardBound)
113     (htransitionBonus_nonneg : 0 <= transitionBonus) :
114     forall (remaining : Nat) (hremaining : remaining <= mdp.horizon)
115       (state : State),
116       |(summary.optimisticPlan mdp defaultState transitionBonus).upperValueRemaining
117           remaining hremaining state| <=
118         empiricalFiniteBatchValueEnvelope rewardBound transitionBonus remaining := by
119   let plan := summary.optimisticPlan mdp defaultState transitionBonus
120   intro remaining
121   induction remaining with
122   | zero =>
123       intro hremaining state
124       simp [MDP.EstimatedModelPlan.upperValueRemaining,
125         empiricalFiniteBatchValueEnvelope]
126   | succ remaining ih =>
127       intro hremaining state
128       let stage := mdp.decisionStageRemaining remaining hremaining
129       let tail := plan.upperValueRemaining remaining (by omega)
130       let action := plan.optimisticAction stage tail state
131       have htail : forall nextState,
132           |tail nextState| <=
133             empiricalFiniteBatchValueEnvelope
134               rewardBound transitionBonus remaining := by
135         intro nextState
136         exact ih (by omega) nextState
137       have htransition :
138           |plan.transitionValue stage tail state action| <=
139             empiricalFiniteBatchValueEnvelope
140               rewardBound transitionBonus remaining := by
141         rw [← Real.norm_eq_abs]
142         have hnorm := norm_integral_le_of_norm_le_const
143           (μ := plan.estimatedTransition stage (state, action))
144           (C := empiricalFiniteBatchValueEnvelope
145             rewardBound transitionBonus remaining)
146           (Filter.Eventually.of_forall fun nextState => by
147             rw [Real.norm_eq_abs]
148             exact htail nextState)
149         simpa [MDP.EstimatedModelPlan.transitionValue] using hnorm
150       rw [MDP.EstimatedModelPlan.upperValueRemaining]
151       change
152         |mdp.reward state action +
153               plan.transitionValue stage tail state action +
154             0 + transitionBonus| <=
155           empiricalFiniteBatchValueEnvelope
156             rewardBound transitionBonus (remaining + 1)
157       calc
158         |mdp.reward state action +
159               plan.transitionValue stage tail state action +
160             0 + transitionBonus| =
161             |(mdp.reward state action +
162               plan.transitionValue stage tail state action) +
163                 transitionBonus| := by ring_nf
164         _ <= |mdp.reward state action +
165               plan.transitionValue stage tail state action| +
166                 |transitionBonus| := abs_add_le _ _
167         _ <= (|mdp.reward state action| +
168               |plan.transitionValue stage tail state action|) +
169                 |transitionBonus| := by
170             have hsum := abs_add_le (mdp.reward state action)
171               (plan.transitionValue stage tail state action)
172             linarith
173         _ <= rewardBound +
174               empiricalFiniteBatchValueEnvelope
175                 rewardBound transitionBonus remaining +
176               transitionBonus := by
177             rw [abs_of_nonneg htransitionBonus_nonneg]
178             linarith [hrewardBound state action, htransition]
179         _ = empiricalFiniteBatchValueEnvelope
180               rewardBound transitionBonus (remaining + 1) := by
181             unfold empiricalFiniteBatchValueEnvelope
182             rw [show ((remaining + 1 : Nat) : Real) =
183               (remaining : Real) + 1 by norm_num]
184             ring
