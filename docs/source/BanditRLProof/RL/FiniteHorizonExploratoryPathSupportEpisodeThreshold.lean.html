3 /-!
4 # Episode-threshold calibration from exploratory path support
5 
6 This module solves the scalar square-root/logarithm obligations left by the
7 explicit path-support calibration route. A closed-form lower bound on the
8 number of episodes implies both the strict count margin and the finite-state,
9 finite-horizon half contraction, then recovers the same adaptive confidence,
10 optimism, and recommended-policy expected-regret endpoint.
11 -/

27 /-- Dimension factor sufficient to turn a radius margin into the half contraction. -/
28 noncomputable def exploratoryPathCalibrationDimensionFactor
29     (mdp : MDP State Action) : Real :=
30   4 * (Fintype.card State : Real) * (mdp.horizon : Real) + 1

37     (mdp : MDP State Action) :
38     1 <= exploratoryPathCalibrationDimensionFactor mdp := by

40   have hproduct :
41       0 <= 4 * (Fintype.card State : Real) * (mdp.horizon : Real) := by
42     positivity
43   linarith

45 /--
46 Closed-form Real episode threshold for the local simultaneous-count budget.
47 Its denominator is positive when `visitFloor` is positive.
48 -/
49 noncomputable def exploratoryPathCalibrationEpisodeThreshold
50     (mdp : MDP State Action) (rounds : Nat)
51     (delta visitFloor : Real) : Real :=
52   (exploratoryPathCalibrationDimensionFactor mdp) ^ 2 *
53       Real.log
54         (2 / simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta)) /
55     (2 * visitFloor ^ 2)

59 /--
60 Above the explicit episode threshold, the simultaneous count radius is less
61 than the common expected-count floor divided by the calibration dimension
62 factor.
63 -/
64 theorem simultaneousCountConfidenceRadius_lt_episodes_mul_visitFloor_div_dimensionFactor_of_threshold
65     (mdp : MDP State Action) (witnessState : State)
66     {rounds episodes : Nat} {delta visitFloor : Real}
67     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
68     (hepisodes : 0 < episodes) (hdelta : 0 < delta)
69     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor)
70     (hthreshold : exploratoryPathCalibrationEpisodeThreshold mdp rounds delta visitFloor <
71       (episodes : Real)) :
72     simultaneousCountConfidenceRadius mdp episodes
73         (multiBatchLocalDelta rounds delta) <
74       (episodes : Real) * visitFloor /
75         exploratoryPathCalibrationDimensionFactor mdp := by
76   let witnessAction : Action := Classical.choice inferInstance
77   have hcoordinate : Nonempty (CountCoordinate mdp) :=
78     ⟨CountCoordinate.visit ⟨0, hhorizon⟩ witnessState witnessAction⟩
79   have hroundsReal : (0 : Real) < rounds := by exact_mod_cast hrounds
80   have hlocalDeltaPos : 0 < multiBatchLocalDelta rounds delta := by
81     exact div_pos hdelta hroundsReal
82   have hroundsOne : (1 : Real) <= rounds := by exact_mod_cast hrounds
83   have hlocalDeltaLeOne : multiBatchLocalDelta rounds delta <= 1 := by
84     exact (div_le_self (le_of_lt hdelta) hroundsOne).trans hdelta_le_one
85   have hcoordinateDeltaPos :
86       0 < simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta) :=
87     MarkovPolicy.simultaneousCountDelta_pos hcoordinate hlocalDeltaPos
88   have hcoordinateDeltaLeOne :
89       simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta) <= 1 :=
90     MarkovPolicy.simultaneousCountDelta_le_one hcoordinate
91       hlocalDeltaPos hlocalDeltaLeOne
92   let q := exploratoryPathCalibrationDimensionFactor mdp
93   let radius := simultaneousCountConfidenceRadius mdp episodes
94     (multiBatchLocalDelta rounds delta)
95   let logarithm := Real.log
96     (2 / simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta))
97   have hqOne : (1 : Real) <= q := by
98     exact one_le_exploratoryPathCalibrationDimensionFactor mdp
99   have hqPos : 0 < q := zero_lt_one.trans_le hqOne
100   have hepisodesReal : (0 : Real) < episodes := by exact_mod_cast hepisodes
101   have hradiusNonneg : 0 <= radius :=
102     Concentration.subGaussianSumConfidenceRadius_nonneg _ _
103   have hradiusSq : radius ^ 2 = (episodes : Real) / 2 * logarithm := by
104     change
105       (Concentration.subGaussianSumConfidenceRadius
106           (MarkovPolicy.iidBernoulliVarianceProxy episodes)
107           (simultaneousCountDelta mdp (multiBatchLocalDelta rounds delta))) ^ 2 =
108         (episodes : Real) / 2 * logarithm
109     rw [Concentration.subGaussianSumConfidenceRadius_sq _ _
110       hcoordinateDeltaPos hcoordinateDeltaLeOne]
111     rw [MarkovPolicy.iidBernoulliVarianceProxy_eq]
112     dsimp [logarithm]
113     push_cast
114     ring
115   have hthresholdProduct :
116       q ^ 2 * logarithm <
117         (episodes : Real) * (2 * visitFloor ^ 2) := by
118     have hdenominator : 0 < 2 * visitFloor ^ 2 := by positivity
119     exact (div_lt_iff₀ hdenominator).mp (by
120       simpa [exploratoryPathCalibrationEpisodeThreshold, q, logarithm]
121         using hthreshold)
122   have hsquare :
123       (q * radius) ^ 2 < ((episodes : Real) * visitFloor) ^ 2 := by
124     rw [mul_pow, hradiusSq]
125     nlinarith
126   have hscaled : q * radius < (episodes : Real) * visitFloor :=
127     (sq_lt_sq₀ (mul_nonneg (le_of_lt hqPos) hradiusNonneg)
128       (mul_nonneg (le_of_lt hepisodesReal) (le_of_lt hvisitFloor))).mp hsquare
129   exact (lt_div_iff₀ hqPos).2 (by simpa [mul_comm] using hscaled)

133 /--
134 The episode threshold simultaneously discharges the strict count margin and
135 the half-contraction premise of the explicit path-support calibration route.
136 -/
137 theorem episodeThreshold_countMargin_and_halfContraction
138     (mdp : MDP State Action) (witnessState : State)
139     {rounds episodes : Nat} {delta visitFloor : Real}
140     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
141     (hepisodes : 0 < episodes) (hdelta : 0 < delta)
142     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor)
143     (hthreshold : exploratoryPathCalibrationEpisodeThreshold mdp rounds delta visitFloor <
144       (episodes : Real)) :
145     simultaneousCountConfidenceRadius mdp episodes
146         (multiBatchLocalDelta rounds delta) <
147         (episodes : Real) * visitFloor /\
148       (Fintype.card State : Real) *
149           uniformFloorTransitionCoordinateRadius mdp episodes
150             (multiBatchLocalDelta rounds delta) visitFloor *
151           (mdp.horizon : Real) <=
152         1 / 2 := by
153   let q := exploratoryPathCalibrationDimensionFactor mdp
154   let radius := simultaneousCountConfidenceRadius mdp episodes
155     (multiBatchLocalDelta rounds delta)
156   have hqOne : (1 : Real) <= q :=
157     one_le_exploratoryPathCalibrationDimensionFactor mdp
158   have hqPos : 0 < q := zero_lt_one.trans_le hqOne
159   have hepisodesReal : (0 : Real) < episodes := by exact_mod_cast hepisodes
160   have hradiusNonneg : 0 <= radius :=
161     Concentration.subGaussianSumConfidenceRadius_nonneg _ _
162   have hratio : radius < (episodes : Real) * visitFloor / q :=
163     simultaneousCountConfidenceRadius_lt_episodes_mul_visitFloor_div_dimensionFactor_of_threshold
164       mdp witnessState hhorizon hrounds hepisodes hdelta hdelta_le_one
165       hvisitFloor hthreshold
166   have hmargin : radius < (episodes : Real) * visitFloor :=
167     hratio.trans_le
168       (div_le_self
169         (mul_nonneg (le_of_lt hepisodesReal) (le_of_lt hvisitFloor)) hqOne)
170   refine ⟨hmargin, ?_⟩
171   have hscaled : q * radius < (episodes : Real) * visitFloor := by
172     have hradiusScaled : radius * q < (episodes : Real) * visitFloor :=
173       (lt_div_iff₀ hqPos).mp hratio
174     simpa [mul_comm] using hradiusScaled
175   have hdenominator : 0 < (episodes : Real) * visitFloor - radius := by
176     linarith
177   have hnumerator :
178       2 * (Fintype.card State : Real) * (mdp.horizon : Real) * radius <=
179         (1 / 2 : Real) * ((episodes : Real) * visitFloor - radius) := by
180     dsimp [q, exploratoryPathCalibrationDimensionFactor] at hscaled
181     nlinarith
182   rw [uniformFloorTransitionCoordinateRadius]
183   calc
184     (Fintype.card State : Real) *
185           (2 * radius / ((episodes : Real) * visitFloor - radius)) *
186           (mdp.horizon : Real) =
187         (2 * (Fintype.card State : Real) * (mdp.horizon : Real) * radius) /
188           ((episodes : Real) * visitFloor - radius) := by ring
189     _ <= 1 / 2 := (div_le_iff₀ hdenominator).2 (by
190       simpa [mul_comm, mul_left_comm, mul_assoc] using hnumerator)
