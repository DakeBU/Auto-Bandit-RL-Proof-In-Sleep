5 # Finite-horizon stochastic reward cumulative concentration
6 
7 This module centers every sampled reward at the mean of its actual pre-step
8 state and sampled action.  A recursive kernel composition proof combines the
9 common one-step sub-Gaussian proxy additively, yielding a total proxy linear in
10 the remaining horizon and a fixed-horizon two-sided delta tail.
11 -/

21 theorem kernel_hasSubgaussianMGF_of_ae
22     {Alpha : Type u} {Omega : Type v}
23     [MeasurableSpace Alpha] [MeasurableSpace Omega]
24     (mu : Measure Alpha) [IsFiniteMeasure mu]
25     (kernel : ProbabilityTheory.Kernel Alpha Omega)
26     (X : Omega -> Real) (varianceProxy : NNReal)
27     (hX : Measurable X)
28     (hfiber : ∀ᵐ alpha ∂mu,
29       ProbabilityTheory.HasSubgaussianMGF X varianceProxy (kernel alpha)) :
30     ProbabilityTheory.Kernel.HasSubgaussianMGF X varianceProxy kernel mu := by
31   have hintegrable : forall t : Real,
32       Integrable (fun omega => Real.exp (t * X omega)) (kernel ∘ₘ mu) := by
33     intro t
34     let f : Omega -> Real := fun omega => Real.exp (t * X omega)
35     have hf : Measurable f :=
36       Real.measurable_exp.comp (measurable_const.mul hX)
37     refine (Measure.integrable_comp_iff hf.aestronglyMeasurable).2 ⟨?_, ?_⟩
38     · filter_upwards [hfiber] with alpha hsub
39       simpa [f] using hsub.integrable_exp_mul t
40     · have hinner : StronglyMeasurable
41           (fun alpha => ∫ omega, ‖f omega‖ ∂kernel alpha) :=
42         (hf.stronglyMeasurable.norm.integral_kernel)
43       refine Integrable.of_bound hinner.aestronglyMeasurable
44         (Real.exp (((varianceProxy : NNReal) : Real) * t ^ 2 / 2)) ?_
45       filter_upwards [hfiber] with alpha hsub
46       have heq :
47           (∫ omega, ‖f omega‖ ∂kernel alpha) =
48             ProbabilityTheory.mgf X (kernel alpha) t := by
49         simp [f, ProbabilityTheory.mgf, Real.norm_eq_abs]
50       rw [heq, Real.norm_of_nonneg ProbabilityTheory.mgf_nonneg]
51       exact hsub.mgf_le t
52   refine ProbabilityTheory.Kernel.HasSubgaussianMGF.of_rat hintegrable ?_
53   intro q
54   filter_upwards [hfiber] with alpha hsub
55   simpa using hsub.mgf_le (q : Real)

57 theorem hasSubgaussianMGF_apply_of_kernel_dirac
58     {Alpha : Type u} {Omega : Type v}
59     [MeasurableSpace Alpha] [MeasurableSpace Omega]
60     [MeasurableSingletonClass Alpha]
61     (kernel : ProbabilityTheory.Kernel Alpha Omega)
62     (alpha : Alpha) (X : Omega -> Real) (varianceProxy : NNReal)
63     (hkernel : ProbabilityTheory.Kernel.HasSubgaussianMGF
64       X varianceProxy kernel (Measure.dirac alpha)) :
65     ProbabilityTheory.HasSubgaussianMGF X varianceProxy (kernel alpha) := by
66   constructor
67   · intro t
68     have hfiber := Measure.ae_integrable_of_integrable_comp
69       (hkernel.integrable_exp_mul t)
70     simpa only [ae_dirac_eq, Filter.eventually_pure] using hfiber
71   · intro t
72     have hpointwise : forall s : Real,
73         ProbabilityTheory.mgf X (kernel alpha) s <=
74           Real.exp ((varianceProxy : Real) * s ^ 2 / 2) := by
75       simpa only [ae_dirac_eq, Filter.eventually_pure] using hkernel.mgf_le
76     exact hpointwise t

88 def sampledCumulativeRewardDeviationFrom (mdp : MDP State Action) :
89     (remaining : Nat) -> State -> RewardStepTrace Action State remaining -> Real
90   | 0, _, _ => 0
91   | remaining + 1, state, trace =>
92       (trace 0).2.1 - mdp.reward state (trace 0).1 +
93         sampledCumulativeRewardDeviationFrom mdp remaining
94           (trace 0).2.2 (Fin.tail trace)

96 theorem measurable_sampledCumulativeRewardDeviationFrom (mdp : MDP State Action)
97     (remaining : Nat) :
98     Measurable (fun p : State × RewardStepTrace Action State remaining =>
99       mdp.sampledCumulativeRewardDeviationFrom remaining p.1 p.2) := by
100   induction remaining with
101   | zero =>
102       simp [sampledCumulativeRewardDeviationFrom]
103   | succ remaining ih =>
104       let head : State × RewardStepTrace Action State (remaining + 1) ->
105           Action × (Real × State) := fun p => p.2 0
106       have hhead : Measurable head := (measurable_pi_apply 0).comp measurable_snd
107       have htail : Measurable
108           (fun p : State × RewardStepTrace Action State (remaining + 1) =>
109             ((head p).2.2, Fin.tail p.2)) :=
110         hhead.snd.snd.prodMk
111           ((RewardStepTrace.measurable_tail remaining).comp measurable_snd)
112       simpa only [sampledCumulativeRewardDeviationFrom] using
113         (hhead.snd.fst.sub
114           (mdp.measurable_reward.comp (measurable_fst.prodMk hhead.fst))).add
115             (ih.comp htail)

161 theorem stochasticTrajectoryKernelRemaining_sampledCumulativeRewardDeviationFrom_hasSubgaussianMGF
162     [StandardBorelSpace State] [StandardBorelSpace Action]
163     [Nonempty Action]
164     (source : MeanCompatibleRewardKernel mdp)
165     (policy : MarkovPolicy mdp)
166     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
167     (state : State) (varianceProxy : NNReal)
168     (law : source.UniformSubgaussianRewardLaw varianceProxy) :
169     ProbabilityTheory.HasSubgaussianMGF
170       (mdp.sampledCumulativeRewardDeviationFrom remaining state)
171       ((remaining : NNReal) * varianceProxy)
172       (source.stochasticTrajectoryKernelRemaining policy
173         remaining hremaining state) := by
174   induction remaining generalizing state with
175   | zero =>
176       simpa [MDP.sampledCumulativeRewardDeviationFrom] using
177         (ProbabilityTheory.HasSubgaussianMGF.fun_zero
178           (μ := source.stochasticTrajectoryKernelRemaining policy
179             0 hremaining state))
180   | succ remaining ih =>
181       let stage : Fin mdp.horizon :=
182         ⟨mdp.horizon - (remaining + 1), by omega⟩
183       let headKernel : ProbabilityTheory.Kernel State
184           (Action × (Real × State)) :=
185         source.actionRewardStateKernel policy stage
186       let tailKernel : ProbabilityTheory.Kernel
187           (State × (Action × (Real × State)))
188           (RewardStepTrace Action State remaining) :=
189         (source.stochasticTrajectoryKernelRemaining policy remaining (by omega)).comap
190           (fun p : State × (Action × (Real × State)) => p.2.2.2)
191           measurable_snd.snd.snd
192       let retainNext :
193           ((State × (Action × (Real × State))) ×
194             RewardStepTrace Action State remaining) ->
195             State × RewardStepTrace Action State remaining :=
196         fun p => (p.1.2.2.2, p.2)
197       have hretainNext : Measurable retainNext :=
198         measurable_fst.snd.snd.snd.prodMk measurable_snd
199       let tailStateKernel : ProbabilityTheory.Kernel
200           (State × (Action × (Real × State)))
201           (State × RewardStepTrace Action State remaining) :=
202         (ProbabilityTheory.Kernel.id ×ₖ tailKernel).map retainNext
203       let X : Action × (Real × State) -> Real :=
204         fun head => head.2.1 - mdp.reward state head.1
205       have hX : Measurable X :=
206         measurable_snd.fst.sub
207           (mdp.measurable_reward.comp (measurable_const.prodMk measurable_fst))
208       let Y : State × RewardStepTrace Action State remaining -> Real :=
209         fun p => mdp.sampledCumulativeRewardDeviationFrom remaining p.1 p.2
210       have hY : Measurable Y :=
211         mdp.measurable_sampledCumulativeRewardDeviationFrom remaining
212       have hXkernel : ProbabilityTheory.Kernel.HasSubgaussianMGF
213           X varianceProxy headKernel (Measure.dirac state) := by
214         apply Concentration.kernel_hasSubgaussianMGF_of_ae
215           (Measure.dirac state) headKernel X varianceProxy hX
216         rw [ae_dirac_eq]
217         simpa [X, headKernel, stage] using
218           source.actionRewardStateKernel_rewardDeviation_hasSubgaussianMGF
219             policy remaining hremaining state varianceProxy law
220       have hYkernel : ProbabilityTheory.Kernel.HasSubgaussianMGF
221           Y ((remaining : NNReal) * varianceProxy) tailStateKernel
222           (Measure.dirac state ⊗ₘ headKernel) := by
223         apply Concentration.kernel_hasSubgaussianMGF_of_ae
224           (Measure.dirac state ⊗ₘ headKernel) tailStateKernel Y
225           ((remaining : NNReal) * varianceProxy) hY
226         exact Filter.Eventually.of_forall fun input => by
227           let nextState : State := input.2.2.2
228           have htail := ih (by omega) nextState
229           have htailMeas : Measurable
230               (mdp.sampledCumulativeRewardDeviationFrom remaining nextState) :=
231             (mdp.measurable_sampledCumulativeRewardDeviationFrom remaining).comp
232               (measurable_const.prodMk measurable_id)
233           have htailStateApply : tailStateKernel input =
234               (source.stochasticTrajectoryKernelRemaining policy remaining
235                 (by omega) nextState).map (fun tail => (nextState, tail)) := by
236             dsimp [tailStateKernel]
237             rw [ProbabilityTheory.Kernel.map_apply _ hretainNext]
238             rw [ProbabilityTheory.Kernel.prod_apply,
239               ProbabilityTheory.Kernel.id_apply, Measure.dirac_prod]
240             rw [Measure.map_map hretainNext measurable_prodMk_left]
241             rfl
242           apply htail.congr_identDistrib
243           refine
244             { aemeasurable_fst := htailMeas.aemeasurable
245               aemeasurable_snd := hY.aemeasurable
246               map_eq := ?_ }
247           rw [htailStateApply]
248           let embed : RewardStepTrace Action State remaining ->
249               State × RewardStepTrace Action State remaining :=
250             fun tail => (nextState, tail)
251           have hembed : Measurable embed := measurable_const.prodMk measurable_id
252           calc
253             _ = Measure.map (Y ∘ embed)
254                 (source.stochasticTrajectoryKernelRemaining policy remaining
255                   (by omega) nextState) := by rfl
256             _ = _ := (Measure.map_map hY hembed).symm
257       have hcombinedKernel := hXkernel.add_compProd hYkernel
258       have hcombined : ProbabilityTheory.HasSubgaussianMGF
259           (fun p => X p.1 + Y p.2)
260           (varianceProxy + (remaining : NNReal) * varianceProxy)
261           ((headKernel ⊗ₖ tailStateKernel) state) :=
262         Concentration.hasSubgaussianMGF_apply_of_kernel_dirac
263           (headKernel ⊗ₖ tailStateKernel) state
264           (fun p => X p.1 + Y p.2)
265           (varianceProxy + (remaining : NNReal) * varianceProxy)
266           hcombinedKernel
267       let assemble :
268           (Action × (Real × State)) ×
269               (State × RewardStepTrace Action State remaining) ->
270             RewardStepTrace Action State (remaining + 1) :=
271         fun p => @Fin.cons remaining (fun _ => Action × (Real × State))
272           (p.1.1, (p.1.2.1, p.2.1)) p.2.2
273       have hassemble : Measurable assemble := by
274         have hpair : Measurable
275             (fun p : (Action × (Real × State)) ×
276                 (State × RewardStepTrace Action State remaining) =>
277               ((p.1.1, (p.1.2.1, p.2.1)), p.2.2)) :=
278           (measurable_fst.fst.prodMk
279               (measurable_fst.snd.fst.prodMk measurable_snd.fst)).prodMk
280             measurable_snd.snd
281         simpa [assemble, Function.comp_def] using
282           (RewardStepTrace.measurable_cons remaining).comp hpair
283       have hgenerated :
284           ((headKernel ⊗ₖ tailStateKernel) state).map assemble =
285             source.stochasticTrajectoryKernelRemaining policy
286               (remaining + 1) hremaining state := by
287         rw [stochasticTrajectoryKernelRemaining]
288         rw [ProbabilityTheory.Kernel.map_apply _
289           (RewardStepTrace.measurable_cons remaining)]
290         ext rewardSet hrewardSet
291         rw [Measure.map_apply hassemble hrewardSet]
292         rw [ProbabilityTheory.Kernel.compProd_apply
293           (hrewardSet.preimage hassemble)]
294         rw [Measure.map_apply (RewardStepTrace.measurable_cons remaining)
295           hrewardSet]
296         rw [ProbabilityTheory.Kernel.compProd_apply
297           (hrewardSet.preimage (RewardStepTrace.measurable_cons remaining))]
298         congr 1
299         funext head
300         have hbaseSet : MeasurableSet
301             (Prod.mk head ⁻¹' (assemble ⁻¹' rewardSet)) :=
302           (hrewardSet.preimage hassemble).preimage measurable_prodMk_left
303         have hinnerSet : MeasurableSet
304             (retainNext ⁻¹' (Prod.mk head ⁻¹' (assemble ⁻¹' rewardSet))) :=
305           hbaseSet.preimage hretainNext
306         dsimp [tailStateKernel]
307         rw [ProbabilityTheory.Kernel.map_apply _ hretainNext]
308         rw [Measure.map_apply hretainNext hbaseSet]
309         rw [ProbabilityTheory.Kernel.id_prod_apply' _ _ hinnerSet]
310         rfl
311       have hident : ProbabilityTheory.IdentDistrib
312           (fun p => X p.1 + Y p.2)
313           (mdp.sampledCumulativeRewardDeviationFrom (remaining + 1) state)
314           ((headKernel ⊗ₖ tailStateKernel) state)
315           (source.stochasticTrajectoryKernelRemaining policy
316             (remaining + 1) hremaining state) := by
317         refine
318           { aemeasurable_fst := (hX.comp measurable_fst).add
319               (hY.comp measurable_snd) |>.aemeasurable
320             aemeasurable_snd :=
321               (mdp.measurable_sampledCumulativeRewardDeviationFrom
322                 (remaining + 1)).comp
323                 (measurable_const.prodMk measurable_id) |>.aemeasurable
324             map_eq := ?_ }
325         have hfinal : Measurable
326             (mdp.sampledCumulativeRewardDeviationFrom
327               (remaining + 1) state) :=
328           (mdp.measurable_sampledCumulativeRewardDeviationFrom
329             (remaining + 1)).comp (measurable_const.prodMk measurable_id)
330         change Measure.map
331           ((mdp.sampledCumulativeRewardDeviationFrom
332             (remaining + 1) state) ∘ assemble)
333           ((headKernel ⊗ₖ tailStateKernel) state) = _
334         calc
335           _ = Measure.map
336               (mdp.sampledCumulativeRewardDeviationFrom
337                 (remaining + 1) state)
338               (((headKernel ⊗ₖ tailStateKernel) state).map assemble) := by
339                 rw [Measure.map_map hfinal hassemble]
340           _ = _ := by rw [hgenerated]
341       have htarget := hcombined.congr_identDistrib hident
342       simpa [Nat.cast_succ, add_mul, add_comm, add_left_comm, add_assoc] using htarget

344 theorem stochasticTrajectoryKernelRemaining_sampledCumulativeRewardDeviationFrom_abs_tail_le
345     [StandardBorelSpace State] [StandardBorelSpace Action]
346     [Nonempty Action]
347     (source : MeanCompatibleRewardKernel mdp)
348     (policy : MarkovPolicy mdp)
349     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
350     (state : State) (varianceProxy : NNReal)
351     (law : source.UniformSubgaussianRewardLaw varianceProxy)
352     (htotal : 0 < ((((remaining : NNReal) * varianceProxy : NNReal) : Real)))
353     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
354     (source.stochasticTrajectoryKernelRemaining policy remaining hremaining state)
355       {trace |
356         Concentration.subGaussianSumConfidenceRadius
357             ((remaining : NNReal) * varianceProxy) delta <=
358           |mdp.sampledCumulativeRewardDeviationFrom remaining state trace|} <=
359       ENNReal.ofReal delta := by
360   let mu := source.stochasticTrajectoryKernelRemaining policy
361     remaining hremaining state
362   let F : Filtration Nat
363       (inferInstance : MeasurableSpace (RewardStepTrace Action State remaining)) :=
364     Filtration.const Nat inferInstance le_rfl
365   let Y : Nat -> RewardStepTrace Action State remaining -> Real :=
366     fun _ => mdp.sampledCumulativeRewardDeviationFrom remaining state
367   let cY : Nat -> NNReal := fun _ => (remaining : NNReal) * varianceProxy
368   have hstateMeas : Measurable
369       (mdp.sampledCumulativeRewardDeviationFrom remaining state) :=
370     (mdp.measurable_sampledCumulativeRewardDeviationFrom remaining).comp
371       (measurable_const.prodMk measurable_id)
372   have hadapted : StronglyAdapted F Y := by
373     intro i
374     simpa [F, Y] using hstateMeas.stronglyMeasurable
375   have hzero : ProbabilityTheory.HasSubgaussianMGF (Y 0) (cY 0) mu := by
376     simpa [Y, cY, mu] using
377       source.stochasticTrajectoryKernelRemaining_sampledCumulativeRewardDeviationFrom_hasSubgaussianMGF
378         policy remaining hremaining state varianceProxy law
379   have hsucc : forall i, i < 1 - 1 ->
380       ProbabilityTheory.HasCondSubgaussianMGF
381         (F i) (F.le i) (Y (i + 1)) (cY (i + 1)) mu := by
382     intro i hi
383     omega
384   have hvarianceSum :
385       0 < ((((Finset.range 1).sum cY : NNReal) : Real)) := by
386     simpa [cY] using htotal
387   simpa [mu, Y, cY] using
388     (Concentration.condSubGaussian_sum_abs_tail_ennreal_delta_of_stronglyAdapted
389       hadapted hzero 1 hsucc hvarianceSum delta hdelta hdelta_le_one)
