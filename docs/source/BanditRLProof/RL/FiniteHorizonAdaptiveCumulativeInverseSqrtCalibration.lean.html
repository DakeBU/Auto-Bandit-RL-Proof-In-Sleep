4 /-!
5 # Adaptive cumulative inverse-square-root calibration
6 
7 This module calibrates the cumulative count-martingale confidence producer to
8 a concrete count-dependent optimistic planner.  Path-support exploration gives
9 every adaptive batch a common predictable visit floor.  Outside the compiled
10 global count event, the accumulated realized visits exceed that predictable
11 floor minus the cumulative confidence radius.
12 
13 The usable planner radius is `budget` at zero visits and
14 `min budget (scale / sqrt(count))` after the first visit.  Separating the
15 zero-count value-envelope cap from the inverse-square-root scale avoids the
16 uninhabitable one-scale cover.  A deterministic roundwise two-scale calibration
17 then supplies both cap and inverse-square-root covers.  The terminal remains
18 about recommended-policy expected regret, not behavior or realized regret.
19 -/

37 /--
38 A concrete count radius: the supplied budget at zero visits and inverse-square
39 root decay after the first visit.
40 -/
41 noncomputable def inverseSqrt (budget : Real) (hbudget : 0 <= budget) :
42     TransitionCountRadius where
43   radius count := if count = 0 then budget else budget / Real.sqrt count
44   nonneg count := by
45     split_ifs
46     · exact hbudget
47     · exact div_nonneg hbudget (Real.sqrt_nonneg _)
48   antitone := by
49     intro left right hle
50     by_cases hleft : left = 0
51     · subst left
52       by_cases hright : right = 0
53       · simp [hright]
54       · have hrightOne : (1 : Real) <= right := by
55           exact_mod_cast (Nat.one_le_iff_ne_zero.mpr hright)
56         have hsqrtOne : (1 : Real) <= Real.sqrt right :=
57           Real.one_le_sqrt.mpr hrightOne
58         simp only [if_neg hright]
59         exact div_le_self hbudget hsqrtOne
60     · have hright : right ≠ 0 := by omega
61       have hleftPos : (0 : Real) < left := by
62         exact_mod_cast (Nat.pos_of_ne_zero hleft)
63       have hsqrtLeftPos : 0 < Real.sqrt (left : Real) :=
64         Real.sqrt_pos.2 hleftPos
65       have hsqrtLe : Real.sqrt (left : Real) <= Real.sqrt (right : Real) :=
66         Real.sqrt_le_sqrt (by exact_mod_cast hle)
67       simp only [if_neg hleft, if_neg hright]
68       exact div_le_div_of_nonneg_left hbudget hsqrtLeftPos hsqrtLe

82 /--
83 A usable capped inverse-square-root radius.  `budget` controls the zero-count
84 value envelope, while `scale` controls the statistical decay after enough
85 visits; the cap preserves antitonicity across the first visit.
86 -/
87 noncomputable def cappedInverseSqrt (budget scale : Real)
88     (hbudget : 0 <= budget) (hscale : 0 <= scale) : TransitionCountRadius where
89   radius count :=
90     if count = 0 then budget else min budget (scale / Real.sqrt count)
91   nonneg count := by
92     split_ifs
93     · exact hbudget
94     · exact le_min hbudget (div_nonneg hscale (Real.sqrt_nonneg _))
95   antitone := by
96     intro left right hle
97     by_cases hleft : left = 0
98     · subst left
99       by_cases hright : right = 0
100       · simp [hright]
101       · simp only [if_neg hright]
102         exact min_le_left _ _
103     · have hright : right ≠ 0 := by omega
104       have hleftPos : (0 : Real) < left := by
105         exact_mod_cast (Nat.pos_of_ne_zero hleft)
106       have hsqrtLeftPos : 0 < Real.sqrt (left : Real) :=
107         Real.sqrt_pos.2 hleftPos
108       have hsqrtLe : Real.sqrt (left : Real) <= Real.sqrt (right : Real) :=
109         Real.sqrt_le_sqrt (by exact_mod_cast hle)
110       simp only [if_neg hleft, if_neg hright]
111       exact min_le_min_left budget
112         (div_le_div_of_nonneg_left hscale hsqrtLeftPos hsqrtLe)

130 /-- Predictable cumulative visit floor supplied by one path-support batch floor. -/
131 def cumulativePathVisitExpectedFloor
132     (episodes prefixRounds : Nat) (visitFloor : Real) : Real :=

135 /-- Predictable visit floor after subtracting the cumulative confidence radius. -/
136 noncomputable def cumulativePathVisitLowerMargin
137     (mdp : MDP State Action) (episodes rounds : Nat)
138     (delta visitFloor : Real) (round : Fin rounds) : Real :=
139   cumulativePathVisitExpectedFloor episodes (round + 1) visitFloor -
140     cumulativeCoordinateConfidenceRadius episodes (round + 1)
141       (cumulativeCountLocalDelta mdp rounds delta)

143 /-- Deterministic selected-radius envelope obtained from the lower visit margin. -/
144 noncomputable def cumulativeInverseSqrtRadiusEnvelope
145     (mdp : MDP State Action) (episodes rounds : Nat)
146     (delta visitFloor budget scale : Real) (round : Fin rounds) : Real :=
147   min budget (scale / Real.sqrt
148     (cumulativePathVisitLowerMargin mdp episodes rounds delta visitFloor round))

150 /--
151 Scalar regularity needed to fit all transition-coordinate errors under the
152 inverse-square-root planner radius.  It is deterministic and roundwise; path
153 support and the martingale event supply the corresponding realized counts.
154 -/
155 structure CumulativeInverseSqrtPathCalibration
156     (mdp : MDP State Action) (episodes rounds : Nat)
157     (delta visitFloor rewardBound budget scale : Real) : Prop where
158   visitFloor_pos : 0 < visitFloor
159   rewardBound_nonneg : 0 <= rewardBound
160   budget_nonneg : 0 <= budget
161   scale_nonneg : 0 <= scale
162   lowerMargin_pos : forall round : Fin rounds,
163     0 < cumulativePathVisitLowerMargin
164       mdp episodes rounds delta visitFloor round
165   coverBudget : forall round : Fin rounds,
166     (Fintype.card State : Real) *
167           (2 * cumulativeCoordinateConfidenceRadius episodes (round + 1)
168             (cumulativeCountLocalDelta mdp rounds delta)) *
169           ((mdp.horizon : Real) * (rewardBound + budget)) <=
170       budget * cumulativePathVisitLowerMargin
171         mdp episodes rounds delta visitFloor round
172   coverScale : forall round : Fin rounds,
173     (Fintype.card State : Real) *
174           (2 * cumulativeCoordinateConfidenceRadius episodes (round + 1)
175             (cumulativeCountLocalDelta mdp rounds delta)) *
176           ((mdp.horizon : Real) * (rewardBound + budget)) <=
177       scale * Real.sqrt
178         (cumulativePathVisitLowerMargin
179           mdp episodes rounds delta visitFloor round)

181 end AdaptiveEpisodeBatchSource
