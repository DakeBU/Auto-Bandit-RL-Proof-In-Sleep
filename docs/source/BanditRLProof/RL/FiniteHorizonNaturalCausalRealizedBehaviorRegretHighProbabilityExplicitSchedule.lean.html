3 /-!
4 # Explicit polynomial-prefix high-probability natural causal regret schedule
5 
6 This module chooses the concrete schedule
7 `scale n = n + 1`, `burnin n = scale n`, `rounds n = scale n ^ 4`, and
8 `returnDelta n = exp (-scale n)` for the compiled natural-causal burn-in
9 terminal.  The fourth-power prefix simultaneously absorbs the linear burn-in
10 charge and the fixed-prefix normalized-return confidence radius.
11 
12 The result concerns this deterministic cofinal subsequence of prefixes.  It is
13 not an all-prefix or anytime statement, and the model-tail and return events
14 are still combined only by a union bound.
15 -/

37 Each positive-count normalized successor coordinate contributes at most one
38 copy of the one-episode global return proxy.

40 theorem naturalCumulativeSuccessorAverageReturnVarianceProxy_le_rounds_mul
41     (mdp : MDP State Action) (episodes : Nat -> Nat) (rounds : Nat)
42     (rewardBound rewardVarianceProxy : NNReal)
43     (hepisodes : forall n, 0 < episodes n) :
44     naturalCumulativeSuccessorAverageReturnVarianceProxy mdp episodes
45         rounds rewardBound rewardVarianceProxy <=
46       (rounds : NNReal) *
47         mdp.globalReturnDeviationPerEpisodeVarianceProxy
48           rewardBound rewardVarianceProxy := by
49   rw [naturalCumulativeSuccessorAverageReturnVarianceProxy]
50   rw [Finset.sum_range_succ']
51   simp only [naturalSuccessorAverageReturnVarianceProxyAt, add_zero]
52   calc
53     (Finset.range rounds).sum (fun n =>
54         ((episodes (n + 1) : NNReal)⁻¹) ^ 2 *
55           mdp.iidGlobalSampledCumulativeReturnDeviationVarianceProxy
56             (episodes (n + 1)) rewardBound rewardVarianceProxy) <=
57         (Finset.range rounds).sum (fun _ =>
58           mdp.globalReturnDeviationPerEpisodeVarianceProxy
59             rewardBound rewardVarianceProxy) := by
60       apply Finset.sum_le_sum
61       intro n _hn
62       rw [MDP.iidGlobalSampledCumulativeReturnDeviationVarianceProxy_eq]
63       have he : (episodes (n + 1) : NNReal) ≠ 0 := by
64         exact_mod_cast (Nat.ne_of_gt (hepisodes (n + 1)))
65       have he_one : (1 : NNReal) <= (episodes (n + 1) : NNReal) := by
66         exact_mod_cast (hepisodes (n + 1))
67       have hinv : (episodes (n + 1) : NNReal)⁻¹ <= 1 :=
68         inv_le_one_of_one_le₀ he_one
69       calc
70         ((episodes (n + 1) : NNReal)⁻¹) ^ 2 *
71               ((episodes (n + 1) : NNReal) *
72                 mdp.globalReturnDeviationPerEpisodeVarianceProxy
73                   rewardBound rewardVarianceProxy) =
74             (episodes (n + 1) : NNReal)⁻¹ *
75               mdp.globalReturnDeviationPerEpisodeVarianceProxy
76                 rewardBound rewardVarianceProxy := by
77                   field_simp [he]
78         _ <= 1 * mdp.globalReturnDeviationPerEpisodeVarianceProxy
79               rewardBound rewardVarianceProxy :=
80           mul_le_mul_of_nonneg_right hinv (zero_le _)
81         _ = mdp.globalReturnDeviationPerEpisodeVarianceProxy
82               rewardBound rewardVarianceProxy := one_mul _
83     _ = (rounds : NNReal) *
84         mdp.globalReturnDeviationPerEpisodeVarianceProxy
85           rewardBound rewardVarianceProxy := by simp

91 /-- Positive scale used by the explicit prefix schedule. -/

94 /-- The model-tail burn-in is linear in the schedule scale. -/
95 def explicitHighProbabilityBurnin (n : Nat) : Nat :=
96   explicitHighProbabilityScale n

98 /-- Natural successor prefixes are sampled along a fourth-power subsequence. -/

100   explicitHighProbabilityScale n ^ 4

102 /-- Exponentially vanishing fixed-prefix return failure share. -/
103 noncomputable def explicitHighProbabilityReturnDelta (n : Nat) : Real :=
104   Real.exp (-(explicitHighProbabilityScale n : Real))
