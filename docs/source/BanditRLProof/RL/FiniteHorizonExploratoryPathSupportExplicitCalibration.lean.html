3 /-!
4 # Explicit count and bonus calibration from exploratory path support
5 
6 This module removes the two remaining abstract calibration inputs from the
7 path-support endpoint. A common state-action visit floor controls every
8 expected-count denominator. If the resulting finite-state, finite-horizon
9 transition coefficient is at most one half, the deterministic reward bound
10 itself is a sufficient transition bonus.
11 -/

27 /--
28 Uniform transition-coordinate radius obtained from one common expected-count
29 floor. The denominator is useful when the count radius is strictly smaller
30 than `episodes * visitFloor`.
31 -/
32 noncomputable def uniformFloorTransitionCoordinateRadius
33     (mdp : MDP State Action) (episodes : Nat) (delta visitFloor : Real) : Real :=
34   2 * simultaneousCountConfidenceRadius mdp episodes delta /
35     ((episodes : Real) * visitFloor -
36       simultaneousCountConfidenceRadius mdp episodes delta)
37

41 /-- A positive common denominator makes the uniform coordinate radius nonnegative. -/
42 theorem uniformFloorTransitionCoordinateRadius_nonneg
43     {mdp : MDP State Action} {episodes : Nat} {delta visitFloor : Real}
44     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
45       (episodes : Real) * visitFloor) :
46     0 <= uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor := by
47   apply div_nonneg
48   · exact mul_nonneg (by norm_num)
49       (Concentration.subGaussianSumConfidenceRadius_nonneg _ _)
50   · linarith

56 /--
57 A common expected-count floor bounds every deterministic transition-coordinate
58 radius by the corresponding uniform-denominator radius.
59 -/
60 theorem expectedCountTransitionCoordinateRadius_le_uniformFloor
61     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
62     (initialState : Measure State) [IsProbabilityMeasure initialState]
63     (episodes : Nat) (delta visitFloor : Real)
64     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
65       (episodes : Real) * visitFloor)
66     (hcountFloor : forall coordinate : VisitCoordinate mdp,
67       (episodes : Real) * visitFloor <=
68         coordinate.expectedCount policy initialState episodes)
69     (stage : Fin mdp.horizon) (state : State) (action : Action)
70     (nextState : State) :
71     policy.expectedCountTransitionCoordinateRadius initialState episodes delta
72         stage state action nextState <=
73       uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor := by
74   let coordinate : VisitCoordinate mdp :=
75     { stage := stage, state := state, action := action }
76   have hradius : 0 <= simultaneousCountConfidenceRadius mdp episodes delta :=
77     Concentration.subGaussianSumConfidenceRadius_nonneg _ _
78   have hdenominator :
79       0 < (episodes : Real) * visitFloor -
80         simultaneousCountConfidenceRadius mdp episodes delta := by
81     linarith
82   have hdenominator_le :
83       (episodes : Real) * visitFloor -
84           simultaneousCountConfidenceRadius mdp episodes delta <=
85         coordinate.expectedCount policy initialState episodes -
86           simultaneousCountConfidenceRadius mdp episodes delta := by
87     linarith [hcountFloor coordinate]
88   exact div_le_div_of_nonneg_left (mul_nonneg (by norm_num) hradius)
89     hdenominator hdenominator_le
90 
91 end MarkovPolicy

93 /-- One state-action floor shared by every stage and target state on the path certificate. -/
94 def ExploratoryPathUniformVisitFloor
95     {mdp : MDP State Action} {initialState : Measure State}
96     (support : ExploratoryPathSupport mdp initialState)
97     (explorationRate : NNReal) (visitFloor : Real) : Prop :=
98   forall stage state,
99     visitFloor <=
100       exploratoryPathStateLower support explorationRate stage state *
101         exploratoryActionProbabilityFloor Action explorationRate

107 theorem ExploratoryPathUniformVisitFloor.exploratoryStateCountMargin
108     {mdp : MDP State Action} {initialState : Measure State}

128     (episodes : Nat) (support : ExploratoryPathSupport mdp initialState)

159 /--
160 If the uniform transition coefficient is at most one half, `rewardBound`
161 covers every transition-radius/value-envelope sum when it is used as the
162 transition bonus.
163 -/
164 theorem transitionBonusCover_rewardBound_of_uniformExpectedCountFloor
165     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
166     (initialState : Measure State) [IsProbabilityMeasure initialState]
167     (episodes : Nat) (delta visitFloor rewardBound : Real)
168     (hmargin : simultaneousCountConfidenceRadius mdp episodes delta <
169       (episodes : Real) * visitFloor)
170     (hcountFloor : forall coordinate : VisitCoordinate mdp,
171       (episodes : Real) * visitFloor <=
172         coordinate.expectedCount policy initialState episodes)
173     (hrewardBound_nonneg : 0 <= rewardBound)
174     (hcontraction :
175       (Fintype.card State : Real) *
176           uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor *
177           (mdp.horizon : Real) <=
178         1 / 2) :
179     policy.TransitionBonusCover initialState episodes delta rewardBound rewardBound := by
180   intro remaining hremaining state action
181   have huniform_nonneg :
182       0 <= uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor :=
183     uniformFloorTransitionCoordinateRadius_nonneg hmargin
184   have henvelope_nonneg :
185       0 <= empiricalFiniteBatchValueEnvelope rewardBound rewardBound remaining := by
186     unfold empiricalFiniteBatchValueEnvelope
187     positivity
188   have hremaining_le : (remaining : Real) <= (mdp.horizon : Real) := by
189     exact_mod_cast (show remaining <= mdp.horizon by omega)
190   calc
191     (∑ nextState,
192         policy.expectedCountTransitionCoordinateRadius initialState episodes delta
193             (mdp.decisionStageRemaining remaining hremaining)
194             state action nextState *
195           empiricalFiniteBatchValueEnvelope rewardBound rewardBound remaining) <=
196         ∑ _nextState : State,
197           uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor *
198             empiricalFiniteBatchValueEnvelope rewardBound rewardBound remaining := by
199       apply Finset.sum_le_sum
200       intro nextState _hnextState
201       exact mul_le_mul_of_nonneg_right
202         (policy.expectedCountTransitionCoordinateRadius_le_uniformFloor
203           initialState episodes delta visitFloor hmargin hcountFloor
204           (mdp.decisionStageRemaining remaining hremaining)
205           state action nextState)
206         henvelope_nonneg
207     _ = ((Fintype.card State : Real) *
208           uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor *
209           (remaining : Real)) * (2 * rewardBound) := by
210       simp [empiricalFiniteBatchValueEnvelope]
211       ring
212     _ <= ((Fintype.card State : Real) *
213           uniformFloorTransitionCoordinateRadius mdp episodes delta visitFloor *
214           (mdp.horizon : Real)) * (2 * rewardBound) := by
215       gcongr
216     _ <= (1 / 2 : Real) * (2 * rewardBound) := by
217       exact mul_le_mul_of_nonneg_right hcontraction (mul_nonneg (by norm_num) hrewardBound_nonneg)
218     _ = rewardBound := by ring
219 
220 end MarkovPolicy
