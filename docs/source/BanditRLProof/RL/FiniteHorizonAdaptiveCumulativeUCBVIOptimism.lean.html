84 /-- Statistical premise for one deterministic clipped update.  Downstream it
85 is discharged from the same-source finite event; it is kept abstract here so
86 the Bellman induction remains reusable and auditable. -/
87 def HasOptimalTailConfidence
88     (mdp : MDP State Action) (summary : TransitionCountSummary mdp)
89     (defaultState : State) (bonusScale : Real) : Prop :=
90   ∀ (stage : Fin mdp.horizon) state action,
91     0 < summary.aggregateVisitCount state action ->
92     |(∫ nextState,

95             ∂summary.aggregateEmpiricalTransitionKernel defaultState
96               (state, action)) -
97         mdp.transitionValue
98           (mdp.optimalValueAt (stage + 1)
99             (Nat.succ_le_of_lt stage.isLt)) state action| <=
100       bonusScale / Real.sqrt (summary.aggregateVisitCount state action)

252 /-- A pooled state-action count through prefix `round` is at most the number
253 of observed episodes times `H`. -/
254 theorem adaptiveCumulativeAggregateVisitCountAt_le
255     {mdp : MDP State Action}
256     (trajectory : EpisodeBatchTrajectory mdp 1) (round : Nat)
257     (state : State) (action : Action) :
258     adaptiveCumulativeAggregateVisitCountAt trajectory round state action <=
259       (round + 1) * mdp.horizon := by
260   rw [adaptiveCumulativeAggregateVisitCountAt_eq_sum]
261   calc
262     (∑ i : Fin (round + 1), ∑ stage : Fin mdp.horizon,
263         (trajectory i).visitCount stage state action) <=
264       ∑ _i : Fin (round + 1), mdp.horizon := by
265         apply Finset.sum_le_sum
266         intro i _hi
267         simpa [TransitionCountSummary.aggregateVisitCount,
268           EpisodeBatch.transitionCountSummary_visitCount] using
269           (trajectory i).aggregateVisitCount_le_horizon state action
270     _ = (round + 1) * mdp.horizon := by simp [mul_comm]
