4 /-!
5 # IID fixed-coordinate count concentration for finite-horizon RL
6 
7 This module applies the Mathlib-backed bounded-variable Hoeffding route to the
8 fixed-policy iid episode-batch law. It proves two-sided confidence tails for
9 one visit coordinate and one transition coordinate. Simultaneous finite
10 coordinate events, random-denominator ratios, and adaptive episode policies
11 remain downstream.
12 -/

31 /-- Real indicator that one record visits a fixed state-action coordinate. -/
32 def visitIndicator (state : State) (action : Action)
33     (step : EpisodeStep State Action) : Real :=
34   if step.state = state /\ step.action = action then 1 else 0
35 
36 /-- Real indicator that one record realizes a fixed transition coordinate. -/
37 def transitionIndicator (state : State) (action : Action) (nextState : State)
38     (step : EpisodeStep State Action) : Real :=
39   if step.state = state /\ step.action = action /\
40       step.nextState = nextState then 1 else 0

45     Measurable (visitIndicator state action) := by

116     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
117     (initialState : Measure State) [IsProbabilityMeasure initialState]
118     (stage : Fin mdp.horizon) (state : State) (action : Action) :
119     policy.stageVisitProbability initialState stage state action =
120       (policy.trajectoryMeasure initialState).real
121         {trajectory |

123             (mdp.episodeStepOfTrajectory trajectory stage).action = action} := by

128           (mdp.episodeStepOfTrajectory trajectory stage).action = action} :=

130       (hstep (EpisodeStep.measurable_action (measurableSet_singleton action)))

132   refine integral_congr_ae (Filter.Eventually.of_forall fun trajectory => ?_)

134       (mdp.episodeStepOfTrajectory trajectory stage).state = state /\

208 /-- Every mapped episode coordinate has the common genuine visit-indicator mean. -/
209 theorem integral_visitIndicator_iidEpisodeBatchMeasure_eval
210     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
211     (initialState : Measure State) [IsProbabilityMeasure initialState]
212     {episodes : Nat} (episode : Fin episodes) (stage : Fin mdp.horizon)
213     (state : State) (action : Action) :
214     integral (policy.iidEpisodeBatchMeasure initialState episodes)
215         (fun batch => EpisodeStep.visitIndicator state action
216           (batch episode stage)) =
217       policy.stageVisitProbability initialState stage state action := by
218   have heval : Measurable
219       (fun batch : EpisodeBatch mdp episodes => batch episode stage) :=
220     (measurable_pi_apply stage).comp (measurable_pi_apply episode)
221   have hstep : Measurable

227         (fun batch => EpisodeStep.visitIndicator state action

231           (fun batch => batch episode stage))
232         (EpisodeStep.visitIndicator state action) := by
233           rw [integral_map heval.aemeasurable hindicator.aestronglyMeasurable]
234     _ = integral
235         ((policy.trajectoryMeasure initialState).map
236           (fun trajectory => mdp.episodeStepOfTrajectory trajectory stage))
237         (EpisodeStep.visitIndicator state action) := by
238           rw [policy.iidEpisodeBatchMeasure_map_eval initialState episode stage]

432 /-- Total Hoeffding variance proxy for a finite iid family of `[0,1]` indicators. -/
433 noncomputable def iidBernoulliVarianceProxy (episodes : Nat) : NNReal :=
434   ∑ _episode : Fin episodes, Concentration.intervalVarianceProxy 0 1
435 
436 /-- The iid Bernoulli Hoeffding proxy is exactly one quarter per episode. -/
437 theorem iidBernoulliVarianceProxy_eq (episodes : Nat) :
438     iidBernoulliVarianceProxy episodes = (episodes : NNReal) / 4 := by
439   simp [iidBernoulliVarianceProxy, Concentration.intervalVarianceProxy]
440   norm_num [div_eq_mul_inv]
441 
442 /-- A positive episode count gives a positive total Bernoulli variance proxy. -/
443 theorem iidBernoulliVarianceProxy_pos {episodes : Nat} (hepisodes : 0 < episodes) :
444     0 < ((iidBernoulliVarianceProxy episodes : NNReal) : Real) := by
445   have hsingle : 0 < Concentration.intervalVarianceProxy 0 1 := by
446     rw [Concentration.intervalVarianceProxy]
447     norm_num
448   have hsum : 0 < iidBernoulliVarianceProxy episodes := by
449     unfold iidBernoulliVarianceProxy
450     refine Finset.sum_pos' (fun _episode _ => zero_le _) ?_
451     let episode : Fin episodes := ⟨0, hepisodes⟩
452     exact ⟨episode, Finset.mem_univ episode, hsingle⟩
453   exact_mod_cast hsum

458     {mdp : MDP State Action} (policy : MarkovPolicy mdp)

460     {episodes : Nat} (stage : Fin mdp.horizon) (state : State) (action : Action)
461     (episode : Fin episodes) :
462     ProbabilityTheory.HasSubgaussianMGF
463       (policy.centeredVisitIndicator initialState stage state action episode)
464       (Concentration.intervalVarianceProxy 0 1)
465       (policy.iidEpisodeBatchMeasure initialState episodes) := by
466   have hmeas : AEMeasurable
467       (fun batch : EpisodeBatch mdp episodes =>
468         EpisodeStep.visitIndicator state action (batch episode stage))
469       (policy.iidEpisodeBatchMeasure initialState episodes) :=
470     ((EpisodeStep.measurable_visitIndicator state action).comp
471       ((measurable_pi_apply stage).comp
472         (measurable_pi_apply episode))).aemeasurable
