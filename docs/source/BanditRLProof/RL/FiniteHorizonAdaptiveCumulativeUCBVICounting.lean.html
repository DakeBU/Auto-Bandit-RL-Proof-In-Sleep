5 # Batched visit-count summation for recurrent UCBVI-CH
6 
7 The policy is frozen within each length-`H` episode, so every visit in that
8 episode uses the same strict-prefix count.  These lemmas retain that batching
9 exactly: low-count overshoot costs at most one batch, while positive-count
10 terms telescope through square-root and logarithmic potentials.
11 -/

17 /-- Strict-prefix cumulative mass of a batched nonnegative increment stream. -/
18 def batchedPrefixCount (increment : Nat -> Nat) (round : Nat) : Nat :=
19   ∑ i ∈ Finset.range round, increment i
20 
21 @[simp] theorem batchedPrefixCount_zero (increment : Nat -> Nat) :
22     batchedPrefixCount increment 0 = 0 := by
23   simp [batchedPrefixCount]
24 
25 theorem batchedPrefixCount_succ (increment : Nat -> Nat) (round : Nat) :
26     batchedPrefixCount increment (round + 1) =
27       batchedPrefixCount increment round + increment round := by
28   simp [batchedPrefixCount, Finset.sum_range_succ]

30 theorem batchedPrefixCount_mono (increment : Nat -> Nat) :
31     Monotone (batchedPrefixCount increment) := by
32   intro a b hab
33   exact Finset.sum_le_sum_of_subset_of_nonneg (Finset.range_mono hab)
34     (fun _ _ _ => Nat.zero_le _)

36 private theorem sum_range_forwardDifference (f : Nat -> Real) (rounds : Nat) :
37     (∑ round ∈ Finset.range rounds, (f (round + 1) - f round)) =
38       f rounds - f 0 := by
39   induction rounds with
40   | zero => simp
41   | succ rounds ih =>
42       rw [Finset.sum_range_succ, ih]

45 /-- All increments whose strict-prefix mass is below a threshold occupy at
46 most the threshold plus one batch. -/
47 theorem sum_low_batchedIncrement_le
48     (increment : Nat -> Nat) (batch threshold rounds : Nat)
49     (hincrement : forall round, increment round <= batch) :
50     (∑ round ∈ Finset.range rounds,
51       if batchedPrefixCount increment round < threshold then increment round
52       else 0) <= threshold + batch := by
53   induction rounds with
54   | zero => simp
55   | succ rounds ih =>
56       rw [Finset.sum_range_succ]
57       by_cases hlow : batchedPrefixCount increment rounds < threshold
58       · rw [if_pos hlow]
59         have hprevious :
60             (∑ round ∈ Finset.range rounds,
61               if batchedPrefixCount increment round < threshold then
62                 increment round else 0) <= batchedPrefixCount increment rounds := by
63           unfold batchedPrefixCount
64           apply Finset.sum_le_sum
65           intro i hi
66           split_ifs
67           · exact le_rfl
68           · exact Nat.zero_le _
69         exact (Nat.add_le_add hprevious (hincrement rounds)).trans
70           (Nat.add_le_add_right (Nat.le_of_lt hlow) batch)
71       · rw [if_neg hlow]
72         exact ih

74 private theorem batched_ratio_le_two_log_increment
75     {mass increment : Nat} (hmass : 0 < mass) (hincrement : increment <= mass) :
76     (increment : Real) / mass <=
77       2 * (Real.log (mass + increment : Nat) - Real.log mass) := by
78   have hmassReal : 0 < (mass : Real) := by exact_mod_cast hmass
79   have hnewReal : 0 < ((mass + increment : Nat) : Real) := by positivity
80   have hratio : (0 : Real) < ((mass + increment : Nat) : Real) / mass :=
81     div_pos hnewReal hmassReal
82   have hlog := Real.one_sub_inv_le_log_of_pos hratio
83   rw [Real.log_div hnewReal.ne' hmassReal.ne'] at hlog
84   have htwice : (increment : Real) / mass <=
85       2 * ((increment : Real) / (mass + increment : Nat)) := by
86     rw [show 2 * ((increment : Real) / (mass + increment : Nat)) =
87         (2 * increment : Real) / (mass + increment : Nat) by ring]
88     apply (div_le_div_iff₀ hmassReal hnewReal).2
89     norm_num only [Nat.cast_add, Nat.cast_ofNat]
90     have hcast : (increment : Real) <= mass := by exact_mod_cast hincrement
91     nlinarith
92   have hone : 1 - (((mass + increment : Nat) : Real) / mass)⁻¹ =
93       (increment : Real) / (mass + increment : Nat) := by
94     field_simp
95     norm_num
96   rw [hone] at hlog
97   linarith

99 /-- High-prefix reciprocal masses telescope into one logarithm. -/
100 theorem sum_high_batchedRatio_le_log
101     (increment : Nat -> Nat) (batch threshold rounds : Nat)
102     (hbatch : 0 < batch) (hthreshold : batch <= threshold)
103     (hincrement : forall round, increment round <= batch) :
104     (∑ round ∈ Finset.range rounds,
105       if threshold <= batchedPrefixCount increment round then
106         (increment round : Real) / batchedPrefixCount increment round else 0) <=
107       2 * Real.log (max 1 (batchedPrefixCount increment rounds) : Nat) := by
108   induction rounds with
109   | zero => simp
110   | succ rounds ih =>
111       rw [Finset.sum_range_succ, batchedPrefixCount_succ]
112       by_cases hhigh : threshold <= batchedPrefixCount increment rounds
113       · rw [if_pos hhigh]
114         have hmass : 0 < batchedPrefixCount increment rounds := by omega
115         have hincLeMass : increment rounds <= batchedPrefixCount increment rounds :=
116           (hincrement rounds).trans (hthreshold.trans hhigh)
117         have hstep := batched_ratio_le_two_log_increment hmass hincLeMass
118         have hmaxOld : max 1 (batchedPrefixCount increment rounds) =
119             batchedPrefixCount increment rounds := max_eq_right (by omega)
120         have hmaxNew : max 1
121             (batchedPrefixCount increment rounds + increment rounds) =
122             batchedPrefixCount increment rounds + increment rounds :=
123           max_eq_right (by omega)
124         rw [hmaxOld] at ih
125         rw [hmaxNew]
126         linarith
127       · rw [if_neg hhigh]
128         have hmaxMono : (max 1 (batchedPrefixCount increment rounds) : Nat) <=
129             max 1 (batchedPrefixCount increment rounds + increment rounds) := by
130           exact max_le_max_left 1 (Nat.le_add_right _ _)
131         have hmaxPos : (0 : Real) <
132             (max 1 (batchedPrefixCount increment rounds) : Nat) := by
133           positivity
134         have hmaxMonoReal :
135             ((max 1 (batchedPrefixCount increment rounds) : Nat) : Real) <=
136               (max 1 (batchedPrefixCount increment rounds + increment rounds) : Nat) := by
137           exact_mod_cast hmaxMono
138         have hlogMono := Real.log_le_log
139           hmaxPos hmaxMonoReal
140         linarith

142 private theorem batched_inverseSqrt_step
143     {mass increment batch : Nat} (hmass : 0 < mass)
144     (hincrement : increment <= batch) :
145     (increment : Real) / Real.sqrt mass <=
146       2 * (Real.sqrt (mass + increment : Nat) - Real.sqrt mass) +
147         (batch : Real) * ((increment : Real) / mass) := by
148   have hx : 0 < Real.sqrt mass := Real.sqrt_pos.2 (by exact_mod_cast hmass)
149   have hxOne : 1 <= Real.sqrt mass := by
150     rw [← Real.sqrt_one]
151     exact Real.sqrt_le_sqrt (by exact_mod_cast hmass)
152   have hy : 0 <= Real.sqrt (mass + increment : Nat) := Real.sqrt_nonneg _
153   have hxy : Real.sqrt mass <= Real.sqrt (mass + increment : Nat) :=
154     Real.sqrt_le_sqrt (by exact_mod_cast Nat.le_add_right mass increment)
155   have hxsq : Real.sqrt mass ^ 2 = (mass : Real) :=
156     Real.sq_sqrt (Nat.cast_nonneg mass)
157   have hysq : Real.sqrt (mass + increment : Nat) ^ 2 =
158       ((mass + increment : Nat) : Real) :=
159     Real.sq_sqrt (Nat.cast_nonneg (mass + increment))
160   have hinc : (increment : Real) <= batch := by exact_mod_cast hincrement
161   have hprodGlobal :
162       (Real.sqrt (mass + increment : Nat) - Real.sqrt mass) *
163         (Real.sqrt (mass + increment : Nat) + Real.sqrt mass) =
164           (increment : Real) := by
165     calc
166       _ = Real.sqrt (mass + increment : Nat) ^ 2 -
167             Real.sqrt mass ^ 2 := by ring
168       _ = (increment : Real) := by
169         rw [hysq, hxsq]
170         norm_num
171   have hfactor : Real.sqrt (mass + increment : Nat) - Real.sqrt mass <=
172       (increment : Real) / Real.sqrt mass := by
173     apply (le_div_iff₀ hx).2
174     have hdiff : 0 <= Real.sqrt (mass + increment : Nat) - Real.sqrt mass :=
175       sub_nonneg.mpr hxy
176     have hsum : Real.sqrt mass <=
177         Real.sqrt (mass + increment : Nat) + Real.sqrt mass := by
178       linarith
179     have hmul := mul_le_mul_of_nonneg_left hsum hdiff
180     rw [hprodGlobal] at hmul
181     simpa [mul_comm] using hmul
182   have hfactorBatch : Real.sqrt (mass + increment : Nat) - Real.sqrt mass <=
183       (batch : Real) := by
184     calc
185       _ <= (increment : Real) / Real.sqrt mass := hfactor
186       _ <= (increment : Real) := by
187         exact div_le_self (Nat.cast_nonneg increment) hxOne
188       _ <= (batch : Real) := hinc
189   let d := Real.sqrt (mass + increment : Nat) - Real.sqrt mass
190   have hd : 0 <= d := sub_nonneg.mpr hxy
191   have hfactorDiv : d / Real.sqrt mass <=
192       (increment : Real) / mass := by
193     calc
194       d / Real.sqrt mass <=
195           ((increment : Real) / Real.sqrt mass) / Real.sqrt mass :=
196         (div_le_div_iff_of_pos_right hx).2 (by simpa [d] using hfactor)
197       _ = (increment : Real) / mass := by
198         rw [div_div, ← pow_two, hxsq]
199   have hremainder : d * (d / Real.sqrt mass) <=
200       (batch : Real) * ((increment : Real) / mass) :=
201     mul_le_mul hfactorBatch hfactorDiv (by positivity) (Nat.cast_nonneg batch)
202   have hid : (increment : Real) / Real.sqrt mass =
203       2 * d + d * (d / Real.sqrt mass) := by
204     apply (div_eq_iff hx.ne').2
205     calc
206       (increment : Real) = d *
207           (Real.sqrt (mass + increment : Nat) + Real.sqrt mass) := by
208         simpa [d] using hprodGlobal.symm
209       _ = (2 * d + d * (d / Real.sqrt mass)) * Real.sqrt mass := by
210         rw [add_mul, mul_assoc d, div_mul_cancel₀ _ hx.ne']
211         dsimp [d]
212         ring
213   rw [hid]
214   simpa [d] using add_le_add_left hremainder (2 * d)

405           min cap (coefficient / batchedPrefixCount increment round)) <=
