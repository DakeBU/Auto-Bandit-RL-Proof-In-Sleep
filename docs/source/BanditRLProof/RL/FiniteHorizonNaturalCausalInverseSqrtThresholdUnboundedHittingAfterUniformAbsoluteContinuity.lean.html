30 /-- Probability-theory uniform integrability at exponent one gives uniform
31 absolute continuity of the expected norm over measurable events.
32 
33 This is a thin real-valued wrapper around Mathlib's `UnifIntegrable`
34 epsilon-delta interface. -/
35 theorem integral_abs_restrict_le_of_uniformIntegrable_one
36     {Omega : Type w} [MeasurableSpace Omega] {mu : Measure Omega}
37     {f : Nat -> Omega -> Real}
38     (hui : UniformIntegrable f 1 mu) :
39     forall epsilon : Real, 0 < epsilon ->
40       exists delta : Real, 0 < delta /\
41         forall i event, MeasurableSet event ->
42           mu event <= ENNReal.ofReal delta ->
43             integral (mu.restrict event) (fun omega => |f i omega|) <=
44               epsilon := by
45   intro epsilon hepsilon
46   obtain ⟨delta, hdelta, hsmall⟩ := hui.unifIntegrable hepsilon
47   refine ⟨delta, hdelta, ?_⟩
48   intro i event hevent hmeasure
49   have hmem : MemLp (event.indicator (f i)) 1 mu :=
50     (hui.memLp i).indicator hevent
51   have hnorm := hsmall i event hevent hmeasure
52   have heq :
53       eLpNorm (event.indicator (f i)) 1 mu =
54         ENNReal.ofReal
55           (integral (mu.restrict event) (fun omega => |f i omega|)) := by
56     rw [hmem.eLpNorm_eq_integral_rpow_norm one_ne_zero ENNReal.one_ne_top]
57     simp only [ENNReal.toReal_one, Real.rpow_one, inv_one]
58     simp_rw [norm_indicator_eq_indicator_norm]
59     rw [integral_indicator hevent]
60     simp only [Real.norm_eq_abs]
61   rw [heq] at hnorm
62   exact (ENNReal.ofReal_le_ofReal_iff hepsilon.le).1 hnorm
