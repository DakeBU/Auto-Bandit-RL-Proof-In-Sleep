43 /-- A trajectory generated by a deterministic table records exactly that
44 table's action at every chronological stage, almost surely. -/
45 theorem trajectoryMeasure_action_eq_table_ae
46     {mdp : MDP State Action} (table : DeterministicMarkovPolicyTable mdp)
47     (initialState : Measure State) [IsProbabilityMeasure initialState] :
48     ∀ᵐ trajectory ∂table.toMarkovPolicy.trajectoryMeasure initialState,
49       ∀ stage : Fin mdp.horizon,
50         (mdp.episodeStepOfTrajectory trajectory stage).action =
51           table stage (mdp.trajectoryStateAt trajectory stage) := by
52   rw [ae_all_iff]
53   intro stage
54   rw [ae_iff]
55   have hbad :
56       {trajectory : State × StepTrace Action State mdp.horizon |
57           ¬(mdp.episodeStepOfTrajectory trajectory stage).action =
58             table stage (mdp.trajectoryStateAt trajectory stage)} =
59         ⋃ state : State, ⋃ action : Action,
60           if action ≠ table stage state then
61             {trajectory |
62               (mdp.episodeStepOfTrajectory trajectory stage).state = state ∧
63                 (mdp.episodeStepOfTrajectory trajectory stage).action = action}
64           else ∅ := by
65     ext trajectory
66     simp only [Set.mem_setOf_eq, Set.mem_iUnion, Set.mem_ite_empty_right,
67       Set.mem_empty_iff_false, and_false]
68     constructor
69     · intro hne
70       refine ⟨(mdp.episodeStepOfTrajectory trajectory stage).state,
71         (mdp.episodeStepOfTrajectory trajectory stage).action, ?_⟩
72       have hstate :
73           mdp.trajectoryStateAt trajectory stage =
74             (mdp.episodeStepOfTrajectory trajectory stage).state := by
75         simp [MDP.episodeStepOfTrajectory]
76       simpa [hstate] using hne
77     · rintro ⟨state, action, hne, hstate, haction⟩
78       have htrajectoryState : mdp.trajectoryStateAt trajectory stage = state := by
79         simpa [MDP.episodeStepOfTrajectory] using hstate
80       simpa [haction, htrajectoryState] using hne
81   rw [hbad]
82   apply measure_iUnion_null
83   intro state
84   apply measure_iUnion_null
85   intro action
86   by_cases hne : action ≠ table stage state
87   · rw [if_pos hne]
88     rw [table.toMarkovPolicy.trajectoryMeasure_visitEvent_eq_stateEvent_mul_action
89       initialState stage state action]
90     simp [DeterministicMarkovPolicyTable.toMarkovPolicy,
91       ProbabilityTheory.Kernel.deterministic_apply, hne]
92   · rw [if_neg hne]
93     exact measure_empty
94 
95 end DeterministicMarkovPolicyTable

99 /-- Under the one-episode mapped batch law, every reconstructed state/action
100 record coincides with the genuine generated trajectory and deterministic
101 table. -/
102 theorem iidEpisodeBatchMeasure_successorBatchAligned_ae
103     (mdp : MDP State Action) (initialState : Measure State)
104     [IsProbabilityMeasure initialState]
105     (defaultState : State) (table : DeterministicMarkovPolicyTable mdp)
106     (hhorizon : 0 < mdp.horizon) :
107     ∀ᵐ batch ∂table.toMarkovPolicy.iidEpisodeBatchMeasure initialState 1,
108       let reconstructed : State × StepTrace Action State mdp.horizon :=
109         (batch.reconstructedInitialState defaultState,
110           batch.reconstructedStepTrace)
111       ∀ stage : Fin mdp.horizon,
112         (batch 0 stage).state = mdp.trajectoryStateAt reconstructed stage ∧
113           (batch 0 stage).action =
114             table stage (mdp.trajectoryStateAt reconstructed stage) := by
115   let trajectoryLaw := table.toMarkovPolicy.iidTrajectoryFamilyMeasure initialState 1
116   let batchMap := mdp.episodeBatchOfTrajectories 1
117   have haction :
118       ∀ᵐ trajectories ∂trajectoryLaw,
119         ∀ stage : Fin mdp.horizon,
120           (mdp.episodeStepOfTrajectory (trajectories 0) stage).action =
121             table stage
122               (mdp.trajectoryStateAt (trajectories 0) stage) := by
123     have hsingle := table.trajectoryMeasure_action_eq_table_ae initialState
124     have heval : trajectoryLaw.map (Function.eval (0 : Fin 1)) =
125         table.toMarkovPolicy.trajectoryMeasure initialState :=
126       table.toMarkovPolicy.iidTrajectoryFamilyMeasure_map_eval initialState 0
127     rw [← heval] at hsingle
128     exact ae_of_ae_map (measurable_pi_apply (0 : Fin 1)).aemeasurable hsingle
129   have hsource :
130       ∀ᵐ trajectories ∂trajectoryLaw,
131         let batch := batchMap trajectories
132         let reconstructed : State × StepTrace Action State mdp.horizon :=
133           (batch.reconstructedInitialState defaultState,
134             batch.reconstructedStepTrace)
135         ∀ stage : Fin mdp.horizon,
136           (batch 0 stage).state = mdp.trajectoryStateAt reconstructed stage ∧
137             (batch 0 stage).action =
138               table stage (mdp.trajectoryStateAt reconstructed stage) := by
139     filter_upwards [haction] with trajectories htrajectory
140     have hinitial :
141         EpisodeBatch.reconstructedInitialState defaultState
142             (batchMap trajectories) = (trajectories 0).1 := by
143       simp [batchMap, EpisodeBatch.reconstructedInitialState, hhorizon,
144         MDP.episodeBatchOfTrajectories, MDP.episodeStepOfTrajectory,
145         MDP.trajectoryStateAt]
146     have htrace :
147         EpisodeBatch.reconstructedStepTrace (batchMap trajectories) =
148           (trajectories 0).2 := by
149       funext stage
150       simp [batchMap, EpisodeBatch.reconstructedStepTrace,
151         MDP.episodeBatchOfTrajectories, MDP.episodeStepOfTrajectory]
152     dsimp only
153     intro stage
154     constructor
155     · simp [batchMap, hinitial, htrace, MDP.episodeBatchOfTrajectories,
156         MDP.episodeStepOfTrajectory]
157     · simpa [batchMap, hinitial, htrace, MDP.episodeBatchOfTrajectories,
158         MDP.episodeStepOfTrajectory] using htrajectory stage
159   unfold MarkovPolicy.iidEpisodeBatchMeasure
160   have hreconstructed : Measurable
161       (fun batch : EpisodeBatch mdp 1 =>
162         (batch.reconstructedInitialState defaultState,
163           batch.reconstructedStepTrace)) :=
164     (EpisodeBatch.measurable_reconstructedInitialState defaultState).prodMk
165       EpisodeBatch.measurable_reconstructedStepTrace
166   have hstage (stage : Fin mdp.horizon) : MeasurableSet
167       {batch : EpisodeBatch mdp 1 |
168         (batch 0 stage).state = mdp.trajectoryStateAt
169               (batch.reconstructedInitialState defaultState,
170                 batch.reconstructedStepTrace) stage ∧
171           (batch 0 stage).action = table stage
172             (mdp.trajectoryStateAt
173               (batch.reconstructedInitialState defaultState,
174                 batch.reconstructedStepTrace) stage)} :=
175     let hstep := (measurable_pi_apply stage).comp (measurable_pi_apply (0 : Fin 1))
176     let hstate := EpisodeStep.measurable_state.comp hstep
177     let haction := EpisodeStep.measurable_action.comp hstep
178     let htrajectoryState : Measurable
179         (fun batch : EpisodeBatch mdp 1 => mdp.trajectoryStateAt
180           (batch.reconstructedInitialState defaultState,
181             batch.reconstructedStepTrace) stage) := by
182       exact (mdp.measurable_trajectoryStateAt stage).comp hreconstructed
183     (measurableSet_eq_fun hstate htrajectoryState).inter
184       (measurableSet_eq_fun haction
185         ((measurable_of_finite (table stage)).comp htrajectoryState))
186   apply (ae_map_iff
187     (μ := trajectoryLaw)
188     (mdp.measurable_episodeBatchOfTrajectories 1).aemeasurable ?_).2
189   · exact hsource
190   · simpa only [Set.iInter_setOf] using MeasurableSet.iInter hstage

192 /-- Every successor coordinate of the actual recurrent
193 `Kernel.trajMeasure` is aligned with the deterministic table computed from
194 that same trajectory's strict prefix. -/
195 theorem recurrentSource_trajectoryMeasure_successorBatchAligned_ae
196     (mdp : MDP State Action) (initialState : Measure State)
197     [IsProbabilityMeasure initialState]
198     (defaultState : State) (episodes : Nat) (delta : Real)
199     (hhorizon : 0 < mdp.horizon) (n : Nat) :
200     ∀ᵐ trajectory ∂
201         (recurrentSource mdp initialState defaultState episodes delta).trajectoryMeasure,
202       SuccessorBatchAligned mdp defaultState episodes delta trajectory n := by
203   let source := recurrentSource mdp initialState defaultState episodes delta
204   let prefixMeasure := source.trajectoryMeasure.map (Preorder.frestrictLe n)
205   let pairMap : EpisodeBatchTrajectory mdp 1 →
206       EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 :=
207     fun trajectory =>
208       (Preorder.frestrictLe n trajectory, trajectory (n + 1))
209   have hpairMap : Measurable pairMap := by fun_prop
210   let tableOfPrefix : EpisodeBatchPrefix mdp 1 n →
211       DeterministicMarkovPolicyTable mdp :=
212     recurrentSuccessorTable defaultState
213       (scale (State := State) (Action := Action) mdp episodes delta) n
214   have htable : Measurable tableOfPrefix :=
215     measurable_recurrentSuccessorTable defaultState
216       (scale (State := State) (Action := Action) mdp episodes delta) n
217   have hmeas : MeasurableSet
218       {pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 |
219         let reconstructed : State × StepTrace Action State mdp.horizon :=
220           (pair.2.reconstructedInitialState defaultState,
221             pair.2.reconstructedStepTrace)
222         ∀ stage : Fin mdp.horizon,
223           (pair.2 0 stage).state = mdp.trajectoryStateAt reconstructed stage ∧
224             (pair.2 0 stage).action =
225               tableOfPrefix pair.1 stage
226                 (mdp.trajectoryStateAt reconstructed stage)} := by
227     have hstage (stage : Fin mdp.horizon) : MeasurableSet
228         {pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 |
229           (pair.2 0 stage).state = mdp.trajectoryStateAt
230                 (pair.2.reconstructedInitialState defaultState,
231                   pair.2.reconstructedStepTrace) stage ∧
232             (pair.2 0 stage).action = tableOfPrefix pair.1 stage
233               (mdp.trajectoryStateAt
234                 (pair.2.reconstructedInitialState defaultState,
235                   pair.2.reconstructedStepTrace) stage)} := by
236       let hbatch0 : Measurable
237           (fun pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 =>
238             pair.2 (0 : Fin 1)) :=
239         (measurable_pi_apply (0 : Fin 1)).comp measurable_snd
240       let hstep : Measurable
241           (fun pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 =>
242             pair.2 (0 : Fin 1) stage) :=
243         (measurable_pi_apply stage).comp hbatch0
244       let hstate := EpisodeStep.measurable_state.comp hstep
245       let haction := EpisodeStep.measurable_action.comp hstep
246       let htrajectoryState : Measurable
247         (fun pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 =>
248             mdp.trajectoryStateAt
249               (pair.2.reconstructedInitialState defaultState,
250                 pair.2.reconstructedStepTrace) stage) := by
251         exact (mdp.measurable_trajectoryStateAt stage).comp
252           (((EpisodeBatch.measurable_reconstructedInitialState defaultState).comp
253             measurable_snd).prodMk
254             (EpisodeBatch.measurable_reconstructedStepTrace.comp measurable_snd))
255       have hstateEq := measurableSet_eq_fun hstate htrajectoryState
256       have hactionPiece (state : State) : MeasurableSet
257           {pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 |
258             mdp.trajectoryStateAt
259                 (pair.2.reconstructedInitialState defaultState,
260                   pair.2.reconstructedStepTrace) stage = state ∧
261               (pair.2 0 stage).action = tableOfPrefix pair.1 stage state} := by
262         have htableState : Measurable
263             (fun pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 =>
264               tableOfPrefix pair.1 stage state) :=
265           (measurable_pi_apply state).comp
266             ((measurable_pi_apply stage).comp (htable.comp measurable_fst))
267         exact (measurableSet_eq_fun htrajectoryState measurable_const).inter
268           (measurableSet_eq_fun haction htableState)
269       have hactionEq : MeasurableSet
270           {pair : EpisodeBatchPrefix mdp 1 n × EpisodeBatch mdp 1 |
271             (pair.2 0 stage).action = tableOfPrefix pair.1 stage
272               (mdp.trajectoryStateAt
273                 (pair.2.reconstructedInitialState defaultState,
274                   pair.2.reconstructedStepTrace) stage)} := by
275         have hunion := MeasurableSet.iUnion hactionPiece
276         convert hunion using 1
277         ext pair
278         simp only [Set.mem_setOf_eq, Set.mem_iUnion]
279         constructor
280         · intro h
281           exact ⟨mdp.trajectoryStateAt
282             (pair.2.reconstructedInitialState defaultState,
283               pair.2.reconstructedStepTrace) stage, rfl, h⟩
284         · rintro ⟨state, hstateValue, hactionValue⟩
285           simpa [hstateValue] using hactionValue
286       exact hstateEq.inter hactionEq
287     simpa only [Set.iInter_setOf] using MeasurableSet.iInter hstage
288   have hpair : ∀ᵐ pair ∂prefixMeasure ⊗ₘ source.batchKernel n,
289       let reconstructed : State × StepTrace Action State mdp.horizon :=
290         (pair.2.reconstructedInitialState defaultState,
291           pair.2.reconstructedStepTrace)
292       ∀ stage : Fin mdp.horizon,
293         (pair.2 0 stage).state = mdp.trajectoryStateAt reconstructed stage ∧
294           (pair.2 0 stage).action = tableOfPrefix pair.1 stage
295             (mdp.trajectoryStateAt reconstructed stage) := by
296     apply Measure.ae_compProd_of_ae_ae hmeas
297     exact Filter.Eventually.of_forall fun history => by
298       rw [source.batchKernel_eq_iidEpisodeBatchMeasure]
299       exact iidEpisodeBatchMeasure_successorBatchAligned_ae
300         mdp initialState defaultState (tableOfPrefix history) hhorizon
301   have hpair' := hpair
302   rw [source.trajectoryMeasure_prefix_compProd n] at hpair'
303   have hpull := ae_of_ae_map hpairMap.aemeasurable hpair'
304   simpa [pairMap, tableOfPrefix, source, SuccessorBatchAligned,
305     successorPolicyTable, successorPreviousQ, successorSummary,
306     recurrentSource, recurrentSuccessorTable, prefixTransitionSummaries,
307     adaptiveCumulativeEmpiricalModelStateAt] using hpull
