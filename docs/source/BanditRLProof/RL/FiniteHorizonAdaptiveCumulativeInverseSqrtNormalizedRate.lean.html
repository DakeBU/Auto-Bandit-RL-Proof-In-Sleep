3 /-!
4 # Normalized adaptive cumulative inverse-square-root rate
5 
6 This module specializes the explicit two-scale calibration to rewards bounded
7 in absolute value by one.  It fixes the zero-count budget to one, chooses the
8 inverse-square-root scale from the logarithmic factor and visit floor, and
9 replaces both scalar calibration premises by one episode threshold.
10 -/

28 /-- Canonical statistical scale for rewards and zero-count budget bounded by one. -/
29 noncomputable def normalizedCumulativeInverseSqrtScale
30     (mdp : MDP State Action) (rounds : Nat)
31     (delta visitFloor : Real) : Real :=
32   4 * (Fintype.card State : Real) * (mdp.horizon : Real) *
33       Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
34     Real.sqrt visitFloor
35

37 noncomputable def normalizedCumulativeInverseSqrtEpisodeThreshold
38     (mdp : MDP State Action) (rounds : Nat)
39     (delta visitFloor : Real) : Real :=
40   32 * (Fintype.card State : Real) ^ 2 * (mdp.horizon : Real) ^ 2 *
41       cumulativeInverseSqrtLogFactor mdp rounds delta /
42     visitFloor ^ 2

73 /-- The normalized scale exactly covers the explicit two-scale coefficient. -/
74 theorem normalizedCumulativeInverseSqrtScale_cover
75     (mdp : MDP State Action) {rounds : Nat}
76     (hhorizon : 0 < mdp.horizon) (hrounds : 0 < rounds)
77     {delta visitFloor : Real} (hdelta : 0 < delta)
78     (hdelta_le_one : delta <= 1) (hvisitFloor : 0 < visitFloor) :
79     (cumulativeInverseSqrtCoverCoefficient mdp 1 1) ^ 2 *
80         cumulativeInverseSqrtLogFactor mdp rounds delta <=
81       (normalizedCumulativeInverseSqrtScale mdp rounds delta visitFloor) ^ 2 *
82         visitFloor := by
83   have hlog : 0 <= cumulativeInverseSqrtLogFactor mdp rounds delta :=
84     cumulativeInverseSqrtLogFactor_nonneg mdp hhorizon hrounds
85       hdelta hdelta_le_one
86   have hsqrtVisit : 0 < Real.sqrt visitFloor := Real.sqrt_pos.2 hvisitFloor
87   rw [normalizedCumulativeInverseSqrtScale]
88   apply le_of_eq
89   calc
90     (cumulativeInverseSqrtCoverCoefficient mdp 1 1) ^ 2 *
91           cumulativeInverseSqrtLogFactor mdp rounds delta =
92         (4 * (Fintype.card State : Real) * (mdp.horizon : Real)) ^ 2 *
93           cumulativeInverseSqrtLogFactor mdp rounds delta := by
94             unfold cumulativeInverseSqrtCoverCoefficient
95             ring
96     _ = (4 * (Fintype.card State : Real) * (mdp.horizon : Real)) ^ 2 *
97           (Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta)) ^ 2 := by
98             rw [Real.sq_sqrt hlog]
99     _ = (4 * (Fintype.card State : Real) * (mdp.horizon : Real) *
100             Real.sqrt (cumulativeInverseSqrtLogFactor mdp rounds delta) /
101               Real.sqrt visitFloor) ^ 2 * visitFloor := by
102             rw [div_pow, Real.sq_sqrt hvisitFloor.le]
103             field_simp

105 /-- The old max threshold is bounded by the single normalized threshold. -/
