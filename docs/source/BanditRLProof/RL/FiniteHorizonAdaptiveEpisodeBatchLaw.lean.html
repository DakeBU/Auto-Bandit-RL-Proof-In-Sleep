4 /-!
5 # Adaptive finite-horizon episode-batch laws
6 
7 This module replaces the independent finite product of episode batches by an
8 Ionescu--Tulcea trajectory whose successor batch kernel may depend on the full
9 finite batch history.  A source records the selected Markov policy and the
10 exact equality between its generated iid batch law and the configured history
11 kernel.  The resulting trajectory exposes both the regular conditional law of
12 the next batch and a finite-horizon union bound for arbitrary measurable
13 adapted bad events.
14 
15 This is a law and confidence-budget transport layer.  It does not yet prove
16 that an empirical optimistic-policy update is measurable, instantiate the bad
17 events with policy-dependent count radii, bound cumulative bonuses, or identify
18 the finite sum of expected regrets with realized online regret.
19 -/

58 /--
59 An adaptive batch source.
60 
61 Coordinate zero is generated by `initialPolicy`.  After observing the prefix
62 through coordinate `n`, `successorPolicy n history` selects the policy for
63 coordinate `n + 1`.  `batchKernel_eq_iidEpisodeBatchMeasure` is the exact law
64 transport contract: it also certifies that the history-indexed family is a
65 measurable Markov kernel rather than merely a pointwise family of measures.
66 -/
67 structure AdaptiveEpisodeBatchSource
68     (mdp : MDP State Action) (initialState : Measure State)
69     [IsProbabilityMeasure initialState] (episodes : Nat) where
70   initialPolicy : MarkovPolicy mdp
71   successorPolicy : (n : Nat) -> EpisodeBatchPrefix mdp episodes n ->
72     MarkovPolicy mdp
73   batchKernel : (n : Nat) ->
74     ProbabilityTheory.Kernel (EpisodeBatchPrefix mdp episodes n)
75       (EpisodeBatch mdp episodes)
76   batchKernel_isMarkov : forall n,
77     ProbabilityTheory.IsMarkovKernel (batchKernel n)
78   batchKernel_eq_iidEpisodeBatchMeasure : forall n history,
79     batchKernel n history =
80       (successorPolicy n history).iidEpisodeBatchMeasure initialState episodes

295 /--
296 An adapted successor event inherits a uniform bound on every history fiber.
297 -/
298 theorem trajectoryMeasure_successorBadEvent_le
299     {mdp : MDP State Action} {initialState : Measure State}
300     [IsProbabilityMeasure initialState] {episodes : Nat}
301     (source : AdaptiveEpisodeBatchSource mdp initialState episodes)
302     (n : Nat)
303     {bad : Set (EpisodeBatchPrefix mdp episodes n ×
304       EpisodeBatch mdp episodes)}
305     (hbad : MeasurableSet bad) (budget : ENNReal)
306     (hfiber : forall history,
307       source.batchKernel n history (Prod.mk history ⁻¹' bad) <= budget) :
308     source.trajectoryMeasure (successorBadEvent n bad) <= budget := by
309   let prefixMeasure :=
310     source.trajectoryMeasure.map (Preorder.frestrictLe n)
311   let pairMap : EpisodeBatchTrajectory mdp episodes ->
312       EpisodeBatchPrefix mdp episodes n × EpisodeBatch mdp episodes :=
313     fun trajectory =>
314       (Preorder.frestrictLe n trajectory, trajectory (n + 1))
315   have hpairMap : Measurable pairMap := by fun_prop
316   haveI : IsProbabilityMeasure prefixMeasure := by
317     dsimp only [prefixMeasure]
318     exact Measure.isProbabilityMeasure_map (by fun_prop)
319   calc
320     source.trajectoryMeasure (successorBadEvent n bad) =
321         source.trajectoryMeasure.map pairMap bad := by
322       rw [Measure.map_apply hpairMap hbad]
323       rfl
324     _ = (prefixMeasure ⊗ₘ source.batchKernel n) bad := by
325       rw [source.trajectoryMeasure_prefix_compProd n]
326     _ = ∫⁻ history, source.batchKernel n history
327           (Prod.mk history ⁻¹' bad) ∂prefixMeasure :=
328       Measure.compProd_apply hbad
329     _ <= ∫⁻ _history, budget ∂prefixMeasure :=
330       lintegral_mono hfiber
331     _ = budget := by simp [prefixMeasure]
