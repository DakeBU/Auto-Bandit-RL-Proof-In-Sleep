5 /-!
6 # Bounded stopping-time high-probability natural causal realized regret
7 
8 The exact natural average realized behavior-regret process and its scheduled
9 fixed-prefix logarithmic rate are evaluated at one positive bounded Mathlib
10 stopping time. The stopped violation is measurable at the deterministic bound
11 and is covered pathwise by the finite union of fixed-prefix violations.
12 
13 The probability proof is finite subadditivity. It does not use optional
14 stopping, an expectation identity, or independence between prefix events.
15 -/

33 /-- A positive `WithTop Nat` time bounded by a finite horizon has a positive
34 finite `untopA` value in the same deterministic range. -/
35 theorem one_le_untopA_and_untopA_le_of_withTop_bounds
36     {Omega : Type*} (tau : Omega -> WithTop Nat) (maxRounds : Nat)
37     (htau_pos : forall omega, (1 : WithTop Nat) <= tau omega)
38     (htau_le : forall omega, tau omega <= (maxRounds : WithTop Nat))
39     (omega : Omega) :
40     1 <= (tau omega).untopA /\ (tau omega).untopA <= maxRounds := by
41   have hne : tau omega ≠ (⊤ : WithTop Nat) :=
42     ne_top_of_le_ne_top WithTop.coe_ne_top (htau_le omega)
43   have hcoe : (((tau omega).untopA : Nat) : WithTop Nat) = tau omega := by
44     rw [WithTop.untopA_eq_untop hne]
45     exact WithTop.coe_untop _ hne
46   have hlower :
47       ((1 : Nat) : WithTop Nat) <=
48         (((tau omega).untopA : Nat) : WithTop Nat) := by
49     rw [hcoe]
50     exact htau_pos omega
51   have hupper :
52       (((tau omega).untopA : Nat) : WithTop Nat) <=
53         ((maxRounds : Nat) : WithTop Nat) := by
54     rw [hcoe]
55     exact htau_le omega
56   exact And.intro (WithTop.coe_le_coe.mp hlower)
57     (WithTop.coe_le_coe.mp hupper)
