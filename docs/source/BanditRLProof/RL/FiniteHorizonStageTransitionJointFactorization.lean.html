5 # Generated stage-transition joint-law factorization
6 
7 This module proves that a fixed generated trajectory-stage joint transition
8 mass factors into its state-action visit mass and the true MDP transition
9 kernel singleton mass. The proof follows the recursive finite trajectory kernel;
10 it does not divide by visit probabilities or introduce empirical confidence.
11 -/

261     {mdp : MDP State Action} (policy : MarkovPolicy mdp)

263     (stage : Fin mdp.horizon) (state : State) (action : Action)
264     (nextState : State) :

267           ((state, action), nextState)} =

270             (state, action)} *
271         mdp.transition (state, action) {nextState} := by

273       {trajectory : State × StepTrace Action State mdp.horizon |
274         StepTrace.stateActionNextAt trajectory.1 trajectory.2 stage =
275           ((state, action), nextState)} := by

291 theorem stageTransitionJointProbability_eq_stageVisitProbability_mul_transition
292     [DecidableEq State] [DecidableEq Action]
293     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
294     (initialState : Measure State) [IsProbabilityMeasure initialState]
295     (stage : Fin mdp.horizon) (state : State) (action : Action)

297     policy.stageTransitionJointProbability initialState stage state action nextState =
298       policy.stageVisitProbability initialState stage state action *

304       {trajectory : State × StepTrace Action State mdp.horizon |
305         (mdp.episodeStepOfTrajectory trajectory stage).state = state /\
306           (mdp.episodeStepOfTrajectory trajectory stage).action = action /\
307             (mdp.episodeStepOfTrajectory trajectory stage).nextState = nextState} =
308         {trajectory | StepTrace.stateActionNextAt trajectory.1 trajectory.2 stage =
309           ((state, action), nextState)} := by

315       {trajectory : State × StepTrace Action State mdp.horizon |
316         (mdp.episodeStepOfTrajectory trajectory stage).state = state /\
317           (mdp.episodeStepOfTrajectory trajectory stage).action = action} =
318         {trajectory | StepTrace.stateActionAt trajectory.1 trajectory.2 stage =
319           (state, action)} := by
