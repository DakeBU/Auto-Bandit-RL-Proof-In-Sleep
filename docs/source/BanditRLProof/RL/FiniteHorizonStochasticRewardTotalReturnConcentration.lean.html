3 /-!
4 # Finite-horizon sampled-return concentration
5 
6 This module combines selected-reward noise with policy-action and transition
7 noise. The target random variable is the actual sampled cumulative return minus
8 the recursive policy value. The additive proxy relies on the product structure
9 of reward and next-state sampling given the current state and action.
10 -/

23 /--
24 Sampling from `kappa` and then from a kernel that ignores the sampled value is
25 the same as sampling from the product kernel at the original input.
26 -/
27 theorem compProd_prodMkRight_eq_prod
28     (kappa : Kernel Alpha Beta) (eta : Kernel Alpha Gamma)
29     [IsFiniteKernel kappa] [IsFiniteKernel eta] :
30     kappa ⊗ₖ prodMkRight Beta eta = kappa ×ₖ eta := by
31   apply Kernel.ext
32   intro a
33   apply Measure.ext_prod
34   intro s t hs ht
35   rw [compProd_apply_prod hs ht, prod_apply_prod]
36   simp [mul_comm]

38 /-- Push a first-coordinate-dependent map through the second stage of `compProd`. -/
39 theorem map_compProd_prodMk
40     {Delta : Type*} [MeasurableSpace Delta]
41     (kappa : Kernel Alpha Beta)
42     (eta : Kernel (Alpha × Beta) Gamma)
43     (eta' : Kernel (Alpha × Beta) Delta)
44     [IsSFiniteKernel kappa] [IsSFiniteKernel eta] [IsSFiniteKernel eta']
45     (g : Beta -> Gamma -> Delta) (hg : Measurable g.uncurry)
46     (hmap : forall a b, (eta (a, b)).map (g b) = eta' (a, b)) :
47     (kappa ⊗ₖ eta).map (fun p => (p.1, g p.1 p.2)) = kappa ⊗ₖ eta' := by
48   have hwhole : Measurable (fun p : Beta × Gamma => (p.1, g p.1 p.2)) :=
49     measurable_fst.prodMk (hg.comp (measurable_fst.prodMk measurable_snd))
50   apply Kernel.ext
51   intro a
52   ext s hs
53   rw [Kernel.map_apply _ hwhole]
54   rw [Measure.map_apply hwhole hs]
55   rw [compProd_apply (hs.preimage hwhole), compProd_apply hs]
56   congr with b
57   have hgb : Measurable (g b) :=
58     hg.comp (measurable_const.prodMk measurable_id)
59   rw [← hmap a b, Measure.map_apply hgb (measurable_prodMk_left hs)]
60   rfl
61 
62 end Kernel
63 end ProbabilityTheory

97 /-- Actual sampled cumulative return centered at the recursive policy value. -/
98 noncomputable def sampledCumulativeReturnDeviationFrom
99     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
100     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
101     (state : State) (trace : RewardStepTrace Action State remaining) : Real :=
102   sampledCumulativeRewardFrom remaining trace -
103     policy.valueRemaining remaining hremaining state
104 
105 /-- The sampled-return deviation is jointly measurable in start state and trace. -/
106 theorem measurable_sampledCumulativeReturnDeviationFrom
107     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
108     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
109     Measurable (fun p : State × RewardStepTrace Action State remaining =>
110       mdp.sampledCumulativeReturnDeviationFrom policy remaining
111         hremaining p.1 p.2) := by
112   exact
113     ((measurable_sampledCumulativeRewardFrom remaining).comp measurable_snd).sub
114       ((policy.measurable_valueRemaining remaining hremaining).comp measurable_fst)

116 /--
117 The centered sampled return splits pathwise into selected-reward noise and the
118 mean Bellman innovation. No probabilistic independence is used in this identity.
119 -/
120 theorem sampledCumulativeReturnDeviationFrom_eq_rewardDeviation_add_meanBellmanInnovation
121     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
122     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
123     (state : State) (trace : RewardStepTrace Action State remaining) :
124     mdp.sampledCumulativeReturnDeviationFrom policy remaining
125         hremaining state trace =
126       mdp.sampledCumulativeRewardDeviationFrom remaining state trace +
127         mdp.sampledCumulativeMeanBellmanInnovationFrom policy remaining
128           hremaining state trace := by
129   induction remaining generalizing state with
130   | zero =>
131       simp [sampledCumulativeReturnDeviationFrom, sampledCumulativeRewardFrom,
132         sampledCumulativeRewardDeviationFrom,
133         sampledCumulativeMeanBellmanInnovationFrom, MarkovPolicy.valueRemaining]
134   | succ remaining ih =>
135       have htail := ih (by omega) (trace 0).2.2 (Fin.tail trace)
136       simp only [sampledCumulativeReturnDeviationFrom] at htail
137       simp only [sampledCumulativeReturnDeviationFrom,
138         sampledCumulativeRewardFrom, sampledCumulativeRewardDeviationFrom,
139         sampledCumulativeMeanBellmanInnovationFrom]
140       linarith

450 /-- Retaining the current state preserves the one-step Bellman innovation MGF. -/
451 theorem retainedActionStateKernel_meanBellmanInnovation_hasSubgaussianMGF
452     (policy : MarkovPolicy mdp)
453     (rewardBound : NNReal)
454     (hrewardBound : forall state action,
455       |mdp.reward state action| <= (rewardBound : Real))
456     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
457     (state : State) :
458     ProbabilityTheory.HasSubgaussianMGF
459       (fun retained : State × (Action × State) =>
460         mdp.reward retained.1 retained.2.1 +
461           policy.valueRemaining remaining (by omega) retained.2.2 -
462             policy.valueRemaining (remaining + 1) hremaining state)
463       (meanBellmanInnovationStepVarianceProxy rewardBound (remaining + 1))
464       (policy.retainedActionStateKernel
465         ⟨mdp.horizon - (remaining + 1), by omega⟩ state) := by
466   let stage : Fin mdp.horizon :=
467     ⟨mdp.horizon - (remaining + 1), by omega⟩
468   let X : Action × State -> Real := fun head =>
469     mdp.reward state head.1 +
470       policy.valueRemaining remaining (by omega) head.2 -
471         policy.valueRemaining (remaining + 1) hremaining state
472   let retainedX : State × (Action × State) -> Real := fun retained =>
473     mdp.reward retained.1 retained.2.1 +
474       policy.valueRemaining remaining (by omega) retained.2.2 -
475         policy.valueRemaining (remaining + 1) hremaining state
476   have hX : Measurable X :=
477     ((mdp.measurable_reward.comp
478       (measurable_const.prodMk measurable_fst)).add
479       ((policy.measurable_valueRemaining remaining (by omega)).comp
480         measurable_snd)).sub measurable_const
481   have hretainedX : Measurable retainedX :=
482     ((mdp.measurable_reward.comp
483       (measurable_fst.prodMk measurable_snd.fst)).add
484       ((policy.measurable_valueRemaining remaining (by omega)).comp
485         measurable_snd.snd)).sub measurable_const
486   have hretainedLaw :
487       policy.retainedActionStateKernel stage state =
488         (policy.actionStateKernel stage state).map (Prod.mk state) := by
489     rw [MarkovPolicy.retainedActionStateKernel,
490       ProbabilityTheory.Kernel.prod_apply,
491       ProbabilityTheory.Kernel.id_apply, Measure.dirac_prod]
492   have hbase := actionStateKernel_meanBellmanInnovation_hasSubgaussianMGF
493     policy rewardBound hrewardBound remaining hremaining state
494   apply hbase.congr_identDistrib
495   refine
496     { aemeasurable_fst := hX.aemeasurable
497       aemeasurable_snd := hretainedX.aemeasurable
498       map_eq := ?_ }
499   rw [hretainedLaw]
500   change Measure.map X (policy.actionStateKernel stage state) =
501     Measure.map retainedX
502       (Measure.map (Prod.mk state) (policy.actionStateKernel stage state))
503   have hembed : Measurable
504       (Prod.mk state : Action × State -> State × (Action × State)) :=
505     measurable_const.prodMk measurable_id
506   rw [Measure.map_map hretainedX hembed]
507   rfl
