4 # Finite-horizon stochastic reward trajectory head marginals
5 
6 This module identifies the first generated stochastic-reward coordinate at
7 every positive recursive horizon.  It exposes the exact action/reward joint
8 law and reward-only policy mixture needed before adding conditional-law or
9 concentration assumptions.
10 -/

68 /-- The action/reward marginal of one generated stochastic MDP step. -/
69 noncomputable def actionRewardKernel
70     (source : MeanCompatibleRewardKernel mdp)
71     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon) :
72     ProbabilityTheory.Kernel State (Prod Action Real) :=
73   (source.actionRewardStateKernel policy stage).map
74     (fun head => (head.1, head.2.1))

76 instance instActionRewardKernelIsMarkovKernel
77     (source : MeanCompatibleRewardKernel mdp)
78     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon) :
79     ProbabilityTheory.IsMarkovKernel
80       (source.actionRewardKernel policy stage) := by
81   unfold actionRewardKernel

101 /-- The first generated coordinate has exactly the compiled one-step law. -/
102 theorem stochasticTrajectoryKernelRemaining_map_head
103     (source : MeanCompatibleRewardKernel mdp)
104     (policy : MarkovPolicy mdp)
105     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
106     (state : State) :
107     (source.stochasticTrajectoryKernelRemaining policy
108         (remaining + 1) hremaining state).map
109         (RewardStepTrace.head (Action := Action) (State := State) remaining) =
110       source.actionRewardStateKernel policy
111         ⟨mdp.horizon - (remaining + 1), by omega⟩ state := by
112   rw [stochasticTrajectoryKernelRemaining]
113   rw [← ProbabilityTheory.Kernel.map_apply _
114     (RewardStepTrace.measurable_head remaining)]
115   rw [← ProbabilityTheory.Kernel.map_comp_right _
116     (RewardStepTrace.measurable_cons remaining)
117     (RewardStepTrace.measurable_head remaining)]
118   change (((source.actionRewardStateKernel policy _).compProd _).map Prod.fst)
119       state = _
120   rw [← ProbabilityTheory.Kernel.fst_eq]
121   rw [ProbabilityTheory.Kernel.fst_compProd]

186 /-- Exact action/reward rectangle law for a policy-mixed stochastic step. -/
187 theorem actionRewardKernel_apply_prod
188     (source : MeanCompatibleRewardKernel mdp)
189     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
190     (state : State) {actionSet : Set Action} {rewardSet : Set Real}
191     (hactionSet : MeasurableSet actionSet)
192     (hrewardSet : MeasurableSet rewardSet) :
193     source.actionRewardKernel policy stage state (actionSet ×ˢ rewardSet) =
194       ∫⁻ action in actionSet,
195         source.rewardKernel.kernel (state, action) rewardSet
196           ∂(policy.actionKernel stage state) := by
197   let dropNext : Prod Action (Prod Real State) -> Prod Action Real :=
198     fun head => (head.1, head.2.1)
199   have hdropNext : Measurable dropNext :=
200     measurable_fst.prodMk measurable_snd.fst
201   rw [actionRewardKernel,
202     ProbabilityTheory.Kernel.map_apply' _ hdropNext state
203       (hactionSet.prod hrewardSet)]
204   have hpreimage :
205       dropNext ⁻¹' (actionSet ×ˢ rewardSet) =
206         actionSet ×ˢ (rewardSet ×ˢ (Set.univ : Set State)) := by
207     ext head
208     simp [dropNext]
209   rw [hpreimage, actionRewardStateKernel,
210     ProbabilityTheory.Kernel.compProd_apply_prod hactionSet
211       (hrewardSet.prod MeasurableSet.univ)]
212   simp_rw [rewardNextStateKernel,
213     ProbabilityTheory.Kernel.prod_apply_prod]
214   simp

216 /-- Reward-event probability is the selected reward law mixed over policy actions. -/
217 theorem rewardMarginalKernel_apply
218     (source : MeanCompatibleRewardKernel mdp)
219     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
220     (state : State) {rewardSet : Set Real}
221     (hrewardSet : MeasurableSet rewardSet) :
222     source.rewardMarginalKernel policy stage state rewardSet =
223       ∫⁻ action,
224         source.rewardKernel.kernel (state, action) rewardSet
225           ∂(policy.actionKernel stage state) := by
226   rw [rewardMarginalKernel,
227     ProbabilityTheory.Kernel.map_apply' _ measurable_snd state hrewardSet]
228   have hpreimage :
229       Prod.snd ⁻¹' rewardSet =
230         (Set.univ : Set Action) ×ˢ rewardSet := by
231     ext pair
232     simp
233   rw [hpreimage,
234     source.actionRewardKernel_apply_prod policy stage state
235       MeasurableSet.univ hrewardSet]
236   simp

238 /--
239 Route endpoint: the generated first reward event has the exact randomized
240 policy mixture law, while the joint action/reward rectangle retains the
241 selected-law factorization.
242 -/
243 theorem stochasticTrajectoryKernelRemaining_headMarginalFactorization
244     (source : MeanCompatibleRewardKernel mdp)
245     (policy : MarkovPolicy mdp)
246     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
247     (state : State) {actionSet : Set Action} {rewardSet : Set Real}
248     (hactionSet : MeasurableSet actionSet)
249     (hrewardSet : MeasurableSet rewardSet) :
250     (source.stochasticTrajectoryKernelRemaining policy
251         (remaining + 1) hremaining state)
252         ((RewardStepTrace.headActionReward
253           (Action := Action) (State := State) remaining) ⁻¹'
254             (actionSet ×ˢ rewardSet)) =
255       ∫⁻ action in actionSet,
256         source.rewardKernel.kernel (state, action) rewardSet
257           ∂(policy.actionKernel
258             ⟨mdp.horizon - (remaining + 1), by omega⟩ state) ∧
259     (source.stochasticTrajectoryKernelRemaining policy
260         (remaining + 1) hremaining state)
261         ((RewardStepTrace.headReward
262           (Action := Action) (State := State) remaining) ⁻¹' rewardSet) =
263       ∫⁻ action,
264         source.rewardKernel.kernel (state, action) rewardSet
265           ∂(policy.actionKernel
266             ⟨mdp.horizon - (remaining + 1), by omega⟩ state) := by
267   constructor
268   · rw [← Measure.map_apply
269       (RewardStepTrace.measurable_headActionReward remaining)
270       (hactionSet.prod hrewardSet)]
271     rw [source.stochasticTrajectoryKernelRemaining_map_headActionReward
272       policy remaining hremaining state]
273     exact source.actionRewardKernel_apply_prod policy _ state
274       hactionSet hrewardSet
275   · rw [← Measure.map_apply
276       (RewardStepTrace.measurable_headReward remaining) hrewardSet]
277     rw [source.stochasticTrajectoryKernelRemaining_map_headReward
278       policy remaining hremaining state]
