39 /-- Every finite Q-table fold through the first `n` actually generated
40 episodes is optimistic on the proved same-source event. -/
41 theorem recurrentQTableOfTrajectory_dominatesOptimal
42     {mdp : MDP State Action} {initialState : Measure State}
43     [IsProbabilityMeasure initialState]
44     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
45     {episodes : Nat} {delta : Real}
46     {trajectory : EpisodeBatchTrajectory mdp 1}
47     (hhorizon : 0 < mdp.horizon) (hepisodes : 0 < episodes)
48     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
49     (hrewardNonneg : ∀ state action, 0 <= mdp.reward state action)
50     (hrewardOne : ∀ state action, mdp.reward state action <= 1)
51     (htrajectory : trajectory ∉ simultaneousTransitionFailureEvent source episodes
52       (logFactor (State := State) (Action := Action) mdp episodes delta))
53     (defaultState : State) :
54     ∀ n, n <= episodes ->
55       QDominatesOptimal mdp
56         (recurrentQTableOfSummaries mdp defaultState
57           (scale (State := State) (Action := Action) mdp episodes delta) n
58           (fun i => (trajectory i).transitionCountSummary)) := by
59   intro n
60   induction n with
61   | zero =>
62       intro _hn
63       exact initialQTable_dominatesOptimal mdp (by
64         intro state action
65         rw [abs_of_nonneg (hrewardNonneg state action)]
66         exact hrewardOne state action)
67   | succ n ih =>
68       intro hn
69       have hnlt : n < episodes := by omega
70       let round : Fin episodes := ⟨n, hnlt⟩
71       let previous := recurrentQTableOfSummaries mdp defaultState
72         (scale (State := State) (Action := Action) mdp episodes delta) n
73         (fun i => (trajectory i).transitionCountSummary)
74       let summary := cumulativeSummaryOfSequence
75         (fun i : Fin (n + 1) => (trajectory i).transitionCountSummary)
76       have hprevious : QDominatesOptimal mdp previous := by
77         simpa [previous] using ih (by omega)
78       have hsummary : summary =
79           (adaptiveCumulativeEmpiricalModelStateAt trajectory round).1 := by
80         simpa [summary, round] using
81           cumulativeSummaryOfSequence_prefixTransitionSummaries_eq
82             trajectory n
83       have hconfidence : HasOptimalTailConfidence mdp summary defaultState
84           (scale (State := State) (Action := Action) mdp episodes delta) := by
85         rw [hsummary]
86         exact hasOptimalTailConfidence_of_not_mem_simultaneousTransitionFailureEvent
87           source hhorizon hepisodes hdelta hdelta_le_one htrajectory
88             defaultState round
89       change QDominatesOptimal mdp
90         (clippedQTable mdp previous summary defaultState
91           (scale (State := State) (Action := Action) mdp episodes delta))
92       exact clippedQTable_dominatesOptimal mdp previous summary defaultState
93         (scale (State := State) (Action := Action) mdp episodes delta)
94         hrewardNonneg hrewardOne hprevious hconfidence
