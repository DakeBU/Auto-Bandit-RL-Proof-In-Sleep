3 /-!
4 # Previous-Q clipped recurrent UCBVI-CH planner
5 
6 This module defines the actual recurrent planner used by the Chapter 9 route.
7 Coordinate zero uses the all-`H` initial Q table and its fixed finite argmax.
8 After observing coordinates `0,...,n`, the successor planner folds those exact
9 transition summaries, normalizes their cross-stage aggregate row, and applies
10 the backward recurrence
11 
12 `Q = H` at zero count and
13 `Q = min Q_previous (min H (reward + P_hat V_next + 7 H L / sqrt N))`
14 at positive count.
15 
16 The adaptive source below therefore contains no arbitrary uncharged initial
17 policy.  Its successor policy is a measurable function of the strict generated
18 prefix, and its batch kernel is exactly the one-episode trajectory law of that
19 selected policy.
20 -/

38 /-- A chronological action-value table for all actual decision stages. -/
39 abbrev QTable (mdp : MDP State Action) :=
40   Fin mdp.horizon -> State -> Action -> Real

42 /-- The UCBVI-CH initialization `Q_{0,h}(x,a)=H`. -/
43 def initialQTable (mdp : MDP State Action) : QTable mdp :=
44   fun _stage _state _action => (mdp.horizon : Real)

46 /-- Sum a finite sequence of observed transition summaries coordinatewise. -/
47 def cumulativeSummaryOfSequence
48     {mdp : MDP State Action} {n : Nat}
49     (summaries : Fin n -> TransitionCountSummary mdp) :
50     TransitionCountSummary mdp :=
51   fun stage state action nextState =>
52     ∑ i : Fin n, summaries i stage state action nextState

54 /--
55 The recursively selected clipped value with a given previous-episode Q table
56 and one pooled empirical transition model.  The index is decisions remaining;
57 its successor case uses the chronological stage `H-(remaining+1)`.
58 -/
59 noncomputable def clippedValueRemaining
60     {mdp : MDP State Action} (previousQ : QTable mdp)
61     (summary : TransitionCountSummary mdp) (defaultState : State)
62     (bonusScale : Real) :
63     (remaining : Nat) -> remaining <= mdp.horizon -> State -> Real
64   | 0, _ => fun _state => 0
65   | remaining + 1, hremaining =>
66       let stage := mdp.decisionStageRemaining remaining hremaining
67       let tail := clippedValueRemaining previousQ summary defaultState bonusScale
68         remaining (by omega)
69       let score : State -> Action -> Real := fun state action =>
70         let count := summary.aggregateVisitCount state action
71         if count = 0 then (mdp.horizon : Real)
72         else
73           min (previousQ stage state action)
74             (min (mdp.horizon : Real)
75               (mdp.reward state action +
76                 (∫ nextState, tail nextState ∂
77                   summary.aggregateEmpiricalTransitionKernel defaultState
78                     (state, action)) +
79                 bonusScale / Real.sqrt count))
80       fun state =>
81         score state
82           (FiniteRealArgmax.choose (fun action => score state action))
83 
84 /-- The action-value used at one remaining-horizon coordinate. -/
85 noncomputable def clippedQRemaining
86     {mdp : MDP State Action} (previousQ : QTable mdp)
87     (summary : TransitionCountSummary mdp) (defaultState : State)
88     (bonusScale : Real) (remaining : Nat)
89     (hremaining : remaining + 1 <= mdp.horizon)
90     (state : State) (action : Action) : Real :=
91   let stage := mdp.decisionStageRemaining remaining hremaining
92   let count := summary.aggregateVisitCount state action
93   if count = 0 then (mdp.horizon : Real)
94   else
95     min (previousQ stage state action)
96       (min (mdp.horizon : Real)
97         (mdp.reward state action +
98           (∫ nextState,
99             clippedValueRemaining previousQ summary defaultState bonusScale
100               remaining (by omega) nextState ∂
101               summary.aggregateEmpiricalTransitionKernel defaultState
102                 (state, action)) +
103           bonusScale / Real.sqrt count))
104 
105 /-- The complete chronological Q table produced by one recurrent update. -/
106 noncomputable def clippedQTable
107     (mdp : MDP State Action) (previousQ : QTable mdp)
108     (summary : TransitionCountSummary mdp) (defaultState : State)
109     (bonusScale : Real) : QTable mdp :=
110   fun stage =>
111     clippedQRemaining previousQ summary defaultState bonusScale
112       (mdp.horizon - (stage.val + 1)) (by omega)

182 /-- Fold observed summaries into the genuine sequence of previous-Q updates. -/
183 noncomputable def recurrentQTableOfSummaries
184     (mdp : MDP State Action) (defaultState : State) (bonusScale : Real) :
185     (n : Nat) -> (Fin n -> TransitionCountSummary mdp) -> QTable mdp
186   | 0, _summaries => initialQTable mdp
187   | n + 1, summaries =>
188       let previous := recurrentQTableOfSummaries mdp defaultState bonusScale n
189         (fun i => summaries i.castSucc)
190       let cumulative := cumulativeSummaryOfSequence summaries
191       clippedQTable mdp previous cumulative defaultState bonusScale

193 /-- Policy table obtained after the same finite previous-Q fold. -/
194 noncomputable def recurrentPolicyTableOfSummaries
195     (mdp : MDP State Action) (defaultState : State) (bonusScale : Real)
196     (n : Nat) (summaries : Fin n -> TransitionCountSummary mdp) :
197     DeterministicMarkovPolicyTable mdp :=
198   fun stage state =>
199     FiniteRealArgmax.choose
200       (fun action =>
201         recurrentQTableOfSummaries mdp defaultState bonusScale n summaries

204 /-- Extract exactly the transition summaries stored in one finite prefix. -/
205 def prefixTransitionSummaries
206     {mdp : MDP State Action} {episodes n : Nat}
207     (history : EpisodeBatchPrefix mdp episodes n) :
208     Fin (n + 1) -> TransitionCountSummary mdp :=
209   fun i =>
210     (history ⟨i, Finset.mem_Iic.mpr (Nat.le_of_lt_succ i.isLt)⟩)
211       |>.transitionCountSummary

252     refine measurable_pi_lambda _ fun state => ?_

263 /--
264 Canonical one-episode recurrent UCBVI source.  It has no arbitrary initial
265 policy parameter: coordinate zero executes `recurrentInitialTable`; coordinate
266 `n+1` executes the fold of exactly the observed coordinates `0,...,n`.
267 -/
268 noncomputable def recurrentSource
269     (mdp : MDP State Action) (initialState : Measure State)
270     [IsProbabilityMeasure initialState]
271     (defaultState : State) (episodes : Nat) (delta : Real) :
272     AdaptiveEpisodeBatchSource mdp initialState 1 where
273   initialPolicy := (recurrentInitialTable mdp).toMarkovPolicy
274   successorPolicy n history :=
275     (recurrentSuccessorTable defaultState
276       (scale (State := State) (Action := Action) mdp episodes delta)
277       n history).toMarkovPolicy
278   batchKernel n :=
279     (DeterministicMarkovPolicyTable.iidEpisodeBatchKernel initialState 1).comap
280       (recurrentSuccessorTable defaultState
281         (scale (State := State) (Action := Action) mdp episodes delta) n)
282       (measurable_recurrentSuccessorTable defaultState
283         (scale (State := State) (Action := Action) mdp episodes delta) n)
284   batchKernel_isMarkov n := by
285     exact ProbabilityTheory.Kernel.IsMarkovKernel.comap _
286       (measurable_recurrentSuccessorTable defaultState
287         (scale (State := State) (Action := Action) mdp episodes delta) n)
288   batchKernel_eq_iidEpisodeBatchMeasure n history := by
289     rw [ProbabilityTheory.Kernel.comap_apply]
290     rfl
