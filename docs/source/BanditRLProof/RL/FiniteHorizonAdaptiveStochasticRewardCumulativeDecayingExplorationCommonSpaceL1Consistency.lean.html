4 /-!
5 # Stochastic common-space L1 realized-behavior consistency
6 
7 This module strengthens the scheduled stochastic-reward common-space theorem
8 from convergence in probability to convergence of the expected absolute
9 realized-behavior regret and then to convergence in `L1`.
10 
11 Sampled rewards remain unbounded.  The deterministic `2H` envelope is used
12 only for selected-policy expected regret, whose values are computed from the
13 bounded mean-reward MDP.  The globally centered sampled-return deviation is
14 integrated through its compiled sub-Gaussian MGF at the scaled tilt
15 `1 / sqrt(proxy)`.
16 -/

34 /-- Mean-reward expected regret has the deterministic `2H` envelope. -/
35 theorem expectedRegret_le_two_mul_horizon_of_rewardBound
36     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
37     (initialState : Measure State) [IsProbabilityMeasure initialState]
38     (hrewardBound : forall state action, |mdp.reward state action| <= 1) :
39     policy.expectedRegret initialState <= 2 * (mdp.horizon : Real) := by
40   have hoptimal :=
41     AdaptiveEpisodeBatchSource.abs_optimalInitialExpectedReturn_le_horizon
42       mdp initialState hrewardBound
43   have hpolicy :
44       |integral (policy.trajectoryMeasure initialState) mdp.cumulativeReward| <=
45         (mdp.horizon : Real) := by
46     rw [← Real.norm_eq_abs]
47     have hnorm := norm_integral_le_of_norm_le_const
48       (μ := policy.trajectoryMeasure initialState)
49       (C := (mdp.horizon : Real))
50       (Filter.Eventually.of_forall fun trajectory => by
51         rw [Real.norm_eq_abs]
52         exact mdp.abs_cumulativeReward_le_horizon hrewardBound trajectory)
53     simpa using hnorm
54   unfold expectedRegret
55   calc
56     integral initialState
57           (mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon)) -
58         integral (policy.trajectoryMeasure initialState) mdp.cumulativeReward <=
59       |integral initialState
60           (mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon))| +
61         |integral (policy.trajectoryMeasure initialState)
62           mdp.cumulativeReward| := by
63         have hleft := le_abs_self
64           (integral initialState
65             (mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon)))
66         have hright := neg_le_abs
67           (integral (policy.trajectoryMeasure initialState)
68             mdp.cumulativeReward)
69         linarith
70     _ <= (mdp.horizon : Real) + (mdp.horizon : Real) :=
71       add_le_add hoptimal hpolicy
72     _ = 2 * (mdp.horizon : Real) := by ring
