3 /-!
4 # Expected optimality of stopped average sampled return
5 
6 This module exposes the observed successor-batch sample means whose complement
7 from the optimal initial value is the natural average realized-regret process.
8 The empty prefix is assigned the optimal value, so the complement identity is
9 total and remains valid at the `WithTop.untopA` fallback. The identity is then
10 transported through the capped and genuine uncapped stopping prefixes and
11 through Bochner integration. No expectation/stopping-index interchange or
12 optional-stopping theorem is used.
13 -/

31 /-- Observed sample mean in successor batch `t + 1`.

33 Lean's division on `Real` is total, so a zero-size successor batch gives zero.
34 The self-consistent scheduled source used below has positive batch sizes. -/

39       mdp initialState episodes)
40     (trajectory : HeterogeneousStochasticEpisodeBatchTrajectory mdp episodes)
41     (t : Nat) : Real :=
42   mdp.sampledCumulativeRewardSum (episodes (t + 1)) (trajectory (t + 1)) /
43     (episodes (t + 1) : Real)

45 /-- Average of observed successor-batch sample means over a natural prefix.
46 At the empty prefix it uses the optimal initial expected return. -/
47 noncomputable def naturalAverageSampledReturn
48     {mdp : MDP State Action} {initialState : Measure State}
49     [IsProbabilityMeasure initialState] {episodes : Nat -> Nat}
50     (source : HeterogeneousAdaptiveStochasticEpisodeBatchSource
51       mdp initialState episodes)
52     (trajectory : HeterogeneousStochasticEpisodeBatchTrajectory mdp episodes)
53     (rounds : Nat) : Real :=
54   if rounds = 0 then
55     AdaptiveStochasticEpisodeBatchSource.optimalInitialExpectedReturn
56       mdp initialState
57   else
58     (∑ t ∈ Finset.range rounds,
59       source.naturalSuccessorBatchAverageSampledReturn trajectory t) /
60         (rounds : Real)

65 /-- At every prefix, observed average return is optimal value minus realized
66 regret. The empty-prefix convention makes the identity valid at zero. -/
67 theorem naturalAverageSampledReturn_eq_optimal_sub_naturalAverageRealizedBehaviorRegret
68     {mdp : MDP State Action} {initialState : Measure State}
69     [IsProbabilityMeasure initialState] {episodes : Nat -> Nat}
70     (source : HeterogeneousAdaptiveStochasticEpisodeBatchSource
71       mdp initialState episodes)
72     (trajectory : HeterogeneousStochasticEpisodeBatchTrajectory mdp episodes)
73     (rounds : Nat) :
74     source.naturalAverageSampledReturn trajectory rounds =
75       AdaptiveStochasticEpisodeBatchSource.optimalInitialExpectedReturn
76           mdp initialState -
77         source.naturalAverageRealizedBehaviorRegret trajectory rounds := by
78   by_cases hrounds : rounds = 0
79   · subst rounds
80     simp [naturalAverageSampledReturn, naturalAverageRealizedBehaviorRegret,
81       naturalCumulativeRealizedBehaviorRegret]
82   · have hroundsReal : (rounds : Real) ≠ 0 := by
83       exact_mod_cast hrounds
84     simp only [naturalAverageSampledReturn, hrounds, if_false,
85       naturalAverageRealizedBehaviorRegret,
86       naturalCumulativeRealizedBehaviorRegret,
87       naturalSuccessorBatchAverageRealizedRegret,
88       naturalSuccessorBatchAverageSampledReturn]
89     rw [Finset.sum_sub_distrib, Finset.sum_const, Finset.card_range,
90       nsmul_eq_mul, sub_div, mul_div_cancel_left₀ _ hroundsReal]
91     ring
