5 /-!
6 # IID generated-trajectory batches for finite-horizon RL
7 
8 This module maps a finite product of genuine policy trajectory laws into the
9 finite-batch empirical-model surface.  It exposes both the episode/stage
10 marginal law and independence across episode coordinates.  The policy is fixed
11 across episodes; adaptive cross-episode policy updates and concentration of
12 visit-conditioned empirical ratios remain downstream.
13 -/

32 /-- Current state immediately before a recorded trajectory action. -/
33 def trajectoryStateAt (mdp : MDP State Action)
34     (trajectory : State × StepTrace Action State mdp.horizon)
35     (stage : Fin mdp.horizon) : State :=
36   if hzero : stage.val = 0 then
37     trajectory.1
38   else
39     (trajectory.2 ⟨stage.val - 1, by omega⟩).2

41 /-- A full generated trajectory viewed as one empirical record at a stage. -/
42 def episodeStepOfTrajectory (mdp : MDP State Action)
43     (trajectory : State × StepTrace Action State mdp.horizon)
44     (stage : Fin mdp.horizon) : EpisodeStep State Action where
45   state := mdp.trajectoryStateAt trajectory stage
46   action := (trajectory.2 stage).1
47   reward := mdp.reward (mdp.trajectoryStateAt trajectory stage)
48     (trajectory.2 stage).1
49   nextState := (trajectory.2 stage).2

51 /-- A finite family of full trajectories mapped to empirical episode records. -/
52 def episodeBatchOfTrajectories (mdp : MDP State Action) (episodes : Nat)
53     (trajectories : Fin episodes ->
54       State × StepTrace Action State mdp.horizon) :
55     EpisodeBatch mdp episodes :=
56   fun episode stage => mdp.episodeStepOfTrajectory (trajectories episode) stage

58 /-- One trajectory's contribution to a stage/state/action visit count. -/
59 def trajectoryVisitContribution (mdp : MDP State Action)
60     (stage : Fin mdp.horizon) (state : State) (action : Action)
61     (trajectory : State × StepTrace Action State mdp.horizon) : Nat :=
62   let step := mdp.episodeStepOfTrajectory trajectory stage
63   if step.state = state /\ step.action = action then 1 else 0

65 /-- One trajectory's contribution to a stage/state/action reward sum. -/
66 def trajectoryRewardContribution (mdp : MDP State Action)
67     (stage : Fin mdp.horizon) (state : State) (action : Action)
68     (trajectory : State × StepTrace Action State mdp.horizon) : Real :=
69   let step := mdp.episodeStepOfTrajectory trajectory stage
70   if step.state = state /\ step.action = action then step.reward else 0

72 /-- One trajectory's contribution to a stage/state/action/next-state count. -/
73 def trajectoryTransitionContribution (mdp : MDP State Action)
74     (stage : Fin mdp.horizon) (state : State) (action : Action)

235 /--
236 Each episode/stage coordinate of the mapped batch has the corresponding
237 pushforward of the genuine generated single-trajectory law.
238 -/
239 theorem iidEpisodeBatchMeasure_map_eval
240     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
241     (initialState : Measure State) [IsProbabilityMeasure initialState]
242     {episodes : Nat} (episode : Fin episodes) (stage : Fin mdp.horizon) :
243     (policy.iidEpisodeBatchMeasure initialState episodes).map
244         (fun batch => batch episode stage) =
245       (policy.trajectoryMeasure initialState).map
246         (fun trajectory => mdp.episodeStepOfTrajectory trajectory stage) := by
247   have hbatch : Measurable (mdp.episodeBatchOfTrajectories episodes) :=
248     mdp.measurable_episodeBatchOfTrajectories episodes
249   have hbatchEval : Measurable
250       (fun batch : EpisodeBatch mdp episodes => batch episode stage) :=
251     (measurable_pi_apply stage).comp (measurable_pi_apply episode)
252   have htrajectoryEval : Measurable
253       (Function.eval episode :
254         (Fin episodes -> State × StepTrace Action State mdp.horizon) ->
255           State × StepTrace Action State mdp.horizon) :=
256     measurable_pi_apply episode
257   have hstep : Measurable
258       (fun trajectory => mdp.episodeStepOfTrajectory trajectory stage) :=
259     mdp.measurable_episodeStepOfTrajectory stage
260   unfold iidEpisodeBatchMeasure
261   rw [Measure.map_map hbatchEval hbatch]
262   rw [show
263       (fun batch => batch episode stage) ∘
264           mdp.episodeBatchOfTrajectories episodes =
265         (fun trajectory => mdp.episodeStepOfTrajectory trajectory stage) ∘
266           Function.eval episode by rfl]
267   rw [← Measure.map_map hstep htrajectoryEval]
268   rw [policy.iidTrajectoryFamilyMeasure_map_eval initialState episode]

287 /-- Fixed-stage record coordinates are independent under the mapped batch law. -/
288 theorem iIndepFun_iidEpisodeBatch_eval
289     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
290     (initialState : Measure State) [IsProbabilityMeasure initialState]
291     (episodes : Nat) (stage : Fin mdp.horizon) :
292     ProbabilityTheory.iIndepFun
293       (fun episode (batch : EpisodeBatch mdp episodes) => batch episode stage)
294       (policy.iidEpisodeBatchMeasure initialState episodes) := by
295   have hbatch : Measurable (mdp.episodeBatchOfTrajectories episodes) :=
296     mdp.measurable_episodeBatchOfTrajectories episodes
297   have hbatchCoordinates : Measurable
298       (fun batch : EpisodeBatch mdp episodes =>
299         fun episode => batch episode stage) :=
300     measurable_pi_lambda _ fun episode =>
301       (measurable_pi_apply stage).comp (measurable_pi_apply episode)
302   have hbatchCoordinate : forall episode : Fin episodes,
303       Measurable
304         (fun batch : EpisodeBatch mdp episodes => batch episode stage) :=
305     fun episode => (measurable_pi_apply stage).comp (measurable_pi_apply episode)
306   have hsource :=
307     policy.iIndepFun_episodeStepOfTrajectory initialState episodes stage
308   have hsourceEq :=
309     (ProbabilityTheory.iIndepFun_iff_map_fun_eq_pi_map
310       (fun episode =>
311         ((mdp.measurable_episodeStepOfTrajectory stage).comp
312           (measurable_pi_apply episode)).aemeasurable)).1 hsource
313   have hsourceEq' :
314       (policy.iidTrajectoryFamilyMeasure initialState episodes).map
315           (fun trajectories episode =>
316             mdp.episodeStepOfTrajectory (trajectories episode) stage) =
317         Measure.pi fun episode =>
318           (policy.iidTrajectoryFamilyMeasure initialState episodes).map
319             (fun trajectories =>
320               mdp.episodeStepOfTrajectory (trajectories episode) stage) := by
321     simpa only [Function.comp_apply] using hsourceEq
322   apply
323     (ProbabilityTheory.iIndepFun_iff_map_fun_eq_pi_map
324       (fun episode => (hbatchCoordinate episode).aemeasurable)).2
325   unfold iidEpisodeBatchMeasure
326   rw [Measure.map_map hbatchCoordinates hbatch]
327   rw [show
328       (fun batch : EpisodeBatch mdp episodes =>
329         fun episode => batch episode stage) ∘
330           mdp.episodeBatchOfTrajectories episodes =
331         (fun trajectories episode =>
332           mdp.episodeStepOfTrajectory (trajectories episode) stage) by rfl]
333   rw [hsourceEq']
334   congr 1
335   funext episode
336   rw [Measure.map_map (hbatchCoordinate episode) hbatch]
337   rfl
