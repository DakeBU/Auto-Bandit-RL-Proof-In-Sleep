3 /-!
4 # Explicit three-quarter bounded-stopping natural causal realized regret
5 
6 The existing self-consistent schedule assigns two shifted high-power model
7 confidence shares to every coordinate. This module proves that their complete
8 finite-prefix budget is at most `1/8`, fixes the global return budget to
9 `1/8`, and turns the single-model bounded-stopping theorem into a joint bad
10 event of probability at most `1/4`. Its measurable complement therefore has
11 real probability at least `3/4` and carries the stopped logarithmic rate.
12 
13 No optional-stopping identity or independence argument is used.
14 -/

21 /-- A shifted inverse-power prefix with exponent at least six is at most one
22 sixteenth. This refines the existing inverse-square comparison. -/
23 theorem sum_range_one_div_natCast_add_two_pow_le_one_div_sixteen
24     (rounds exponent : Nat) (hexponent : 6 <= exponent) :
25     (Finset.range rounds).sum (fun t =>
26       1 / (((t + 2 : Nat) : Real) ^ exponent)) <= 1 / 16 := by
27   calc
28     (Finset.range rounds).sum (fun t =>
29         1 / (((t + 2 : Nat) : Real) ^ exponent)) <=
30         (Finset.range rounds).sum (fun t =>
31           (1 / 16 : Real) *
32             (1 / (((t + 2 : Nat) : Real) ^ 2))) := by
33       apply Finset.sum_le_sum
34       intro t _ht
35       let base : Real := ((t + 2 : Nat) : Real)
36       have hbase : (2 : Real) <= base := by
37         dsimp [base]
38         exact_mod_cast (show 2 <= t + 2 by omega)
39       have hbase_pos : 0 < base := lt_of_lt_of_le (by norm_num) hbase
40       have hbase_sq : (4 : Real) <= base ^ 2 := by nlinarith
41       have hbase_fourth : (16 : Real) <= base ^ 4 := by
42         nlinarith [sq_nonneg (base ^ 2 - 4)]
43       calc
44         1 / (((t + 2 : Nat) : Real) ^ exponent) <=
45             1 / (((t + 2 : Nat) : Real) ^ 6) :=
46           one_div_pow_le_one_div_pow_of_le
47             (by exact_mod_cast (show 1 <= t + 2 by omega)) hexponent
48         _ = (1 / base ^ 4) * (1 / base ^ 2) := by
49           dsimp [base]
50           field_simp [ne_of_gt hbase_pos]
51         _ <= (1 / 16 : Real) * (1 / base ^ 2) := by
52           exact mul_le_mul_of_nonneg_right
53             (one_div_le_one_div_of_le (by norm_num) hbase_fourth)
54             (by positivity)
55         _ = (1 / 16 : Real) *
56             (1 / (((t + 2 : Nat) : Real) ^ 2)) := by
57           rfl
58     _ = (1 / 16 : Real) *
59         (Finset.range rounds).sum (fun t =>
60           1 / (((t + 2 : Nat) : Real) ^ 2)) := by
61       rw [Finset.mul_sum]
62     _ <= (1 / 16 : Real) * 1 :=
63       mul_le_mul_of_nonneg_left
64         (sum_range_one_div_natCast_add_two_sq_le_one rounds) (by norm_num)
65     _ = 1 / 16 := by ring
