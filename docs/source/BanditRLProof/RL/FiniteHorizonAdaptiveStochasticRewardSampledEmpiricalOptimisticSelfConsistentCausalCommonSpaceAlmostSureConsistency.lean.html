4 /-!
5 # Natural-causal almost-sure consistency
6 
7 This module upgrades the heterogeneous natural-causal realized-regret process
8 from `L1` convergence to almost-sure convergence.  The proof keeps the same
9 dependent trajectory measure.  It applies the first Borel-Cantelli lemma to
10 the summable coordinate model events and to the mass-adapted successor-return
11 events, then consumes the existing fixed-burn-in absolute-regret envelope.
12 -/

19 /-- The stretched-exponential sequence `exp (-sqrt n)` is summable. -/
20 theorem summable_exp_neg_sqrt_natCast :
21     Summable (fun n : Nat => Real.exp (-Real.sqrt (n : Real))) := by
22   let f : Nat -> Real := fun n => Real.exp (-Real.sqrt (n : Real))
23   let u : Nat -> Nat := fun n => (n + 1) ^ 2
24   have hf_nonneg : forall n, 0 <= f n := fun n => Real.exp_nonneg _
25   have hf_antitone : forall {m n : Nat}, 0 < m -> m <= n -> f n <= f m := by
26     intro m n _hm hmn
27     apply Real.exp_monotone
28     exact neg_le_neg (Real.sqrt_le_sqrt (by exact_mod_cast hmn))
29   have hu_pos : forall n, 0 < u n := fun n => by
30     dsimp [u]
31     positivity
32   have hu_strict : StrictMono u := by
33     intro m n hmn
34     dsimp [u]
35     exact pow_lt_pow_left₀ (by omega) (by omega) (by omega)
36   have hu_diff : SuccDiffBounded 3 u := by
37     intro n
38     have hleft : (n + 2 + 1) ^ 2 = (n + 1 + 1) ^ 2 + (2 * n + 5) := by ring
39     have hright : (n + 1 + 1) ^ 2 = (n + 1) ^ 2 + (2 * n + 3) := by ring
40     dsimp [u]
41     omega
42   apply (summable_schlomilch_iff_of_nonneg hf_nonneg (@hf_antitone) hu_pos
43     hu_strict (by norm_num) hu_diff).1
44   have hexp : Summable (fun n : Nat => Real.exp (-n)) :=
45     Real.summable_exp_neg_nat
46   have hnexp : Summable (fun n : Nat => (n : Real) * Real.exp (-n)) := by
47     simpa using Real.summable_pow_mul_exp_neg_nat_mul 1 one_pos
48   have hbase : Summable
49       (fun n : Nat => (2 * (n : Real) + 3) * Real.exp (-n)) := by
50     exact ((hnexp.mul_left 2).add (hexp.mul_left 3)).congr fun n => by ring
51   have hmajor : Summable
52       (fun n : Nat => (2 * (n : Real) + 3) * Real.exp (-((n : Real) + 1))) := by
53     exact (hbase.mul_left (Real.exp (-1))).congr fun n => by
54       rw [show Real.exp (-((n : Real) + 1)) = Real.exp (-1) * Real.exp (-n) by
55         rw [← Real.exp_add]
56         congr 1
57         ring]
58       ring
59   apply hmajor.congr
60   intro n
61   dsimp [u, f]
62   rw [show (((n + 1 + 1) ^ 2 : Nat) : Real) - (((n + 1) ^ 2 : Nat) : Real) =
63       2 * (n : Real) + 3 by push_cast; ring]
64   rw [show Real.sqrt ((((n + 1) ^ 2 : Nat) : Real)) = (n : Real) + 1 by
65     push_cast
66     rw [Real.sqrt_sq_eq_abs, abs_of_nonneg]
67     positivity]

83 theorem natCast_le_successorEpisodeMass
84     (episodes : Nat -> Nat) (hepisodes : forall t, 0 < episodes t)
85     (rounds : Nat) :
86     (rounds : Real) <= successorEpisodeMass episodes rounds := by
87   rw [successorEpisodeMass_eq_sum_range]

91       apply Finset.sum_le_sum
