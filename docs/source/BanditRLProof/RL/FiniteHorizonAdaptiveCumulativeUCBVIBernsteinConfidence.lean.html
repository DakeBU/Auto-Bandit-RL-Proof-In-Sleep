4 /-!
5 # Variance-sensitive same-source transition coordinates for UCBVI-CH
6 
7 The Hoeffding producer records the exact generated visit compensator.  The
8 UCBVI-CH analysis additionally needs the Bernoulli variance of each next-state
9 coordinate.  This module proves that stronger fixed-tilt leaf from the same
10 transition kernel.  It is not an independent sample model.
11 -/

29 /-- True Bernoulli variance of one transition singleton. -/
30 noncomputable def transitionCoordinateVariance
31     (mdp : MDP State Action) (state : State) (action : Action)
32     (nextState : State) : Real :=
33   let p := (mdp.transition (state, action)).real {nextState}
34   p * (1 - p)

45 /-- A centered transition indicator has exact second moment `p(1-p)`. -/
46 theorem integral_sq_indicator_sub_transitionMass
47     (mdp : MDP State Action) (state : State) (action : Action)
48     (nextState : State) :
49     (∫ y, ((if y = nextState then 1 else 0) -
50           (mdp.transition (state, action)).real {nextState}) ^ 2
51         ∂mdp.transition (state, action)) =
52       mdp.transitionCoordinateVariance state action nextState := by
53   let p : Real := (mdp.transition (state, action)).real {nextState}
54   let indicator : State -> Real := fun y => if y = nextState then 1 else 0
55   have hindicator : Measurable indicator := measurable_of_finite _
56   have hindicator_eq : indicator =
57       Set.indicator ({nextState} : Set State) (fun _ : State => (1 : Real)) := by
58     funext y
59     by_cases hy : y = nextState <;> simp [indicator, hy]
60   have hmean : (∫ y, indicator y ∂mdp.transition (state, action)) = p := by
61     rw [hindicator_eq,
62       integral_indicator_const (1 : Real) (MeasurableSet.singleton nextState)]
63     simp [p]
64   have hsq : (∫ y, (indicator y) ^ 2 ∂mdp.transition (state, action)) = p := by
65     have hpoint : (fun y => (indicator y) ^ 2) = indicator := by
66       funext y
67       simp [indicator]
68     rw [hpoint, hmean]
69   have hi : Integrable indicator (mdp.transition (state, action)) :=
70     integrable_of_fintype _ _ hindicator
71   have hi2 : Integrable (fun y => (indicator y) ^ 2)
72       (mdp.transition (state, action)) :=
73     integrable_of_fintype _ _ (hindicator.pow_const 2)
74   have himem : MemLp indicator 2 (mdp.transition (state, action)) :=
75     memLp_of_bounded
76       (Filter.Eventually.of_forall (fun y => by
77         show indicator y ∈ Set.Icc (0 : Real) 1
78         simp only [indicator]
79         split <;> norm_num))
80       hindicator.aestronglyMeasurable 2
81   change (∫ y, (indicator y - p) ^ 2
82       ∂mdp.transition (state, action)) = _
83   calc
84     (∫ y, (indicator y - p) ^ 2 ∂mdp.transition (state, action)) =
85         ProbabilityTheory.variance indicator (mdp.transition (state, action)) := by
86       rw [ProbabilityTheory.variance_eq_integral hindicator.aemeasurable, hmean]
87     _ = p - p ^ 2 := by
88       rw [ProbabilityTheory.variance_eq_sub himem]
89       change (∫ y, indicator y ^ 2 ∂mdp.transition (state, action)) -
90           (∫ y, indicator y ∂mdp.transition (state, action)) ^ 2 = _
91       rw [hsq, hmean]
92     _ = mdp.transitionCoordinateVariance state action nextState := by
93       simp [transitionCoordinateVariance, p]
94       ring

97 /-- The visited transition singleton has a variance-sensitive compensated MGF.
98 The hard `|tilt| <= 1` contract is the standard Bernstein small-tilt range. -/
99 theorem transitionResidualHead_variance_compensated_hasMGFUpperBoundAt
100     (mdp : MDP State Action) (targetState : State) (targetAction : Action)
101     (targetNextState : State) (currentState : State) (chosenAction : Action)
102     (tilt : Real) (htilt : |tilt| <= 1) :
103     Concentration.HasMGFUpperBoundAt
104       (fun nextState =>
105         tilt * mdp.transitionResidualHead targetState targetAction targetNextState
106             currentState (chosenAction, nextState) -
107           tilt ^ 2 *
108             mdp.transitionCoordinateVariance
109               targetState targetAction targetNextState *
110             mdp.transitionVisitHead targetState targetAction
111               currentState (chosenAction, nextState))
112       1 0 (mdp.transition (currentState, chosenAction)) := by
113   by_cases hvisit : currentState = targetState ∧ chosenAction = targetAction
114   · rcases hvisit with ⟨hstate, haction⟩
115     subst currentState
116     subst chosenAction
117     let p : Real :=
118       (mdp.transition (targetState, targetAction)).real {targetNextState}
119     let X : State -> Real := fun y =>
120       (if y = targetNextState then 1 else 0) - p
121     have hX : Measurable X := measurable_of_finite _
122     have hXabs : ∀ y, |tilt * X y| <= 1 := by
123       intro y
124       have hp : p ∈ Set.Icc (0 : Real) 1 :=
125         ⟨measureReal_nonneg, measureReal_le_one⟩
126       rcases hp with ⟨hp0, hp1⟩
127       have hxy : |X y| <= 1 := by
128         simp only [X]
129         split <;> rw [abs_le] <;> constructor <;> linarith
130       calc
131         |tilt * X y| = |tilt| * |X y| := abs_mul _ _
132         _ <= 1 * 1 := mul_le_mul htilt hxy (abs_nonneg _) (by norm_num)
133         _ = 1 := by norm_num
134     have hmean : (∫ y, X y ∂mdp.transition (targetState, targetAction)) = 0 := by
135       let indicator : State -> Real := fun y => if y = targetNextState then 1 else 0
136       have hindicator_eq : indicator =
137           Set.indicator ({targetNextState} : Set State) (fun _ : State => (1 : Real)) := by
138         funext y
139         by_cases hy : y = targetNextState <;> simp [indicator, hy]
140       have hi : Integrable indicator (mdp.transition (targetState, targetAction)) :=
141         integrable_of_fintype _ _ (measurable_of_finite _)
142       change (∫ y, indicator y - p ∂mdp.transition (targetState, targetAction)) = 0
143       rw [integral_sub hi (integrable_const _)]
144       rw [hindicator_eq,
145         integral_indicator_const (1 : Real) (MeasurableSet.singleton targetNextState)]
146       simp [p]
147     have hsecond : (∫ y, (X y) ^ 2
148         ∂mdp.transition (targetState, targetAction)) =
149         mdp.transitionCoordinateVariance
150           targetState targetAction targetNextState := by
151       simpa [X, p] using
152         mdp.integral_sq_indicator_sub_transitionMass
153           targetState targetAction targetNextState
154     have hbase : Concentration.HasMGFUpperBoundAt X tilt
155         (tilt ^ 2 * mdp.transitionCoordinateVariance
156           targetState targetAction targetNextState)
157         (mdp.transition (targetState, targetAction)) := by
158       constructor
159       · intro s
160         exact integrable_of_fintype _ _ (by fun_prop)
161       · rw [ProbabilityTheory.mgf]
162         have hexp : ∀ y, Real.exp (tilt * X y) <=
163             1 + tilt * X y + (tilt * X y) ^ 2 := fun y =>
164           Concentration.exp_le_one_add_self_add_sq_of_abs_le_one (hXabs y)
165         have hleft : Integrable (fun y => Real.exp (tilt * X y))
166             (mdp.transition (targetState, targetAction)) :=
167           integrable_of_fintype _ _ (by fun_prop)
168         have hright : Integrable
169             (fun y => 1 + tilt * X y + (tilt * X y) ^ 2)
170             (mdp.transition (targetState, targetAction)) :=
171           integrable_of_fintype _ _ (by fun_prop)
172         calc
173           (∫ y, Real.exp (tilt * X y)
174               ∂mdp.transition (targetState, targetAction)) <=
175               ∫ y, (1 + tilt * X y + (tilt * X y) ^ 2)
176                 ∂mdp.transition (targetState, targetAction) :=
177             integral_mono_ae hleft hright (Filter.Eventually.of_forall hexp)
178           _ = 1 + tilt ^ 2 * mdp.transitionCoordinateVariance
179                 targetState targetAction targetNextState := by
180             rw [integral_add, integral_add]
181             · simp_rw [mul_pow]
182               rw [integral_const, integral_const_mul, hmean,
183                 integral_const_mul, hsecond]
184               simp
185             all_goals exact integrable_of_fintype _ _ (by fun_prop)
186           _ <= Real.exp (tilt ^ 2 * mdp.transitionCoordinateVariance
187                 targetState targetAction targetNextState) := by
188             simpa [add_comm] using Real.add_one_le_exp
189               (tilt ^ 2 * mdp.transitionCoordinateVariance
190                 targetState targetAction targetNextState)
191     have hcomp := hbase.compensated
192     convert hcomp using 1
193     funext y
194     simp [transitionResidualHead, transitionVisitHead, X, p]
195   · have hzero : Concentration.HasMGFUpperBoundAt
196         (fun _ : State => (0 : Real)) 1 0
197         (mdp.transition (currentState, chosenAction)) := by
198       constructor
199       · intro s
200         simpa using (integrable_const
201           (mu := mdp.transition (currentState, chosenAction)) (Real.exp (s * 0)))
202       · simp [ProbabilityTheory.mgf]
203     convert hzero using 1
204     funext y
205     simp [transitionResidualHead, transitionVisitHead, hvisit]

208 /-- The generated action/transition head preserves the exact coordinate
209 variance compensator. -/
210 theorem actionStateKernel_transitionResidualHead_variance_compensated_hasMGFUpperBoundAt
211     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
212     (targetState : State) (targetAction : Action) (targetNextState : State)
213     (currentState : State) (stage : Fin mdp.horizon) (tilt : Real)
214     (htilt : |tilt| <= 1) :
215     Concentration.HasMGFUpperBoundAt
216       (fun head : Action × State =>
217         tilt * mdp.transitionResidualHead targetState targetAction targetNextState
218             currentState head -
219           tilt ^ 2 *
220             mdp.transitionCoordinateVariance targetState targetAction targetNextState *
221             mdp.transitionVisitHead targetState targetAction currentState head)
222       1 0 (policy.actionStateKernel stage currentState) := by
223   let Z : Action × State -> Real := fun head =>
224     tilt * mdp.transitionResidualHead targetState targetAction targetNextState
225         currentState head -
226       tilt ^ 2 *
227         mdp.transitionCoordinateVariance targetState targetAction targetNextState *
228         mdp.transitionVisitHead targetState targetAction currentState head
229   have hint : Integrable (fun head => Real.exp (1 * Z head))
230       (policy.actionStateKernel stage currentState) :=
231     integrable_of_fintype _ _ (by fun_prop)
232   constructor
233   · intro s
234     exact integrable_of_fintype _ _ (by fun_prop)
235   · rw [ProbabilityTheory.mgf]
236     change (∫ head, Real.exp (1 * Z head)
237       ∂policy.actionStateKernel stage currentState) <= Real.exp 0
238     unfold MarkovPolicy.actionStateKernel
239     rw [ProbabilityTheory.integral_compProd hint]
240     simp only [Real.exp_zero]
241     have hinner : ∀ chosenAction : Action,
242         (∫ nextState, Real.exp (Z (chosenAction, nextState))
243             ∂mdp.transition (currentState, chosenAction)) <= 1 := by
244       intro chosenAction
245       have h := mdp.transitionResidualHead_variance_compensated_hasMGFUpperBoundAt
246         targetState targetAction targetNextState currentState chosenAction tilt htilt
247       simpa [ProbabilityTheory.mgf, Z] using h.mgf_le
248     have hleft : Integrable
249         (fun chosenAction =>
250           ∫ nextState, Real.exp (1 * Z (chosenAction, nextState))
251             ∂mdp.transition (currentState, chosenAction))
252         (policy.actionKernel stage currentState) :=
253       integrable_of_fintype _ _ (measurable_of_finite _)
254     calc
255       (∫ chosenAction,
256           ∫ nextState, Real.exp (1 * Z (chosenAction, nextState))
257             ∂mdp.transition (currentState, chosenAction)
258           ∂policy.actionKernel stage currentState) <=
259           ∫ _chosenAction, (1 : Real)
260             ∂policy.actionKernel stage currentState := by
261               exact integral_mono_ae hleft (integrable_const 1)
262                 (Filter.Eventually.of_forall (by
263                   intro chosenAction
264                   simpa only [one_mul] using hinner chosenAction))
265       _ = 1 := by simp

268 /-- The entire generated episode trace has the exact coordinate-variance
269 compensator, with the compensating count equal to literal visits. -/
270 theorem trajectoryKernelRemaining_transitionResidual_variance_compensated_hasMGFUpperBoundAt
271     (mdp : MDP State Action) (policy : MarkovPolicy mdp)
272     (targetState : State) (targetAction : Action) (targetNextState : State)
273     (tilt : Real) (htilt : |tilt| <= 1)
274     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
275     (currentState : State) :
276     Concentration.HasMGFUpperBoundAt
277       (fun trace =>
278         tilt * mdp.transitionResidualFrom targetState targetAction targetNextState
279             remaining currentState trace -
280           tilt ^ 2 *
281             mdp.transitionCoordinateVariance targetState targetAction targetNextState *
282             mdp.transitionVisitFrom targetState targetAction
283               remaining currentState trace)
284       1 0 (policy.trajectoryKernelRemaining remaining hremaining currentState) := by
285   induction remaining generalizing currentState with
286   | zero =>
287       have hzero : Concentration.HasMGFUpperBoundAt
288           (fun _trace : StepTrace Action State 0 => (0 : Real)) 1 0
289           (policy.trajectoryKernelRemaining 0 hremaining currentState) := by
290         constructor
291         · intro s
292           exact integrable_of_fintype _ _ (by fun_prop)
293         · simp [ProbabilityTheory.mgf]
294       simpa [transitionResidualFrom, transitionVisitFrom] using hzero
295   | succ remaining ih =>
296       let stage : Fin mdp.horizon :=
297         ⟨mdp.horizon - (remaining + 1), by omega⟩
298       let tailKernel : ProbabilityTheory.Kernel
299           (State × (Action × State)) (StepTrace Action State remaining) :=
300         (policy.trajectoryKernelRemaining remaining (by omega)).comap
301           (fun p : State × (Action × State) => p.2.2) measurable_snd.snd
302       let consStep :=
303         fun p : (Action × State) × StepTrace Action State remaining =>
304           @Fin.cons remaining (fun _ => Action × State) p.1 p.2
305       let variance : Real :=
306         mdp.transitionCoordinateVariance targetState targetAction targetNextState
307       let Z : StepTrace Action State (remaining + 1) -> Real := fun trace =>
308         tilt * mdp.transitionResidualFrom targetState targetAction targetNextState
309             (remaining + 1) currentState trace -
310           tilt ^ 2 * variance *
311             mdp.transitionVisitFrom targetState targetAction
312               (remaining + 1) currentState trace
313       let Zhead : Action × State -> Real := fun head =>
314         tilt * mdp.transitionResidualHead targetState targetAction targetNextState
315             currentState head -
316           tilt ^ 2 * variance *
317             mdp.transitionVisitHead targetState targetAction currentState head
318       let Ztail : (Action × State) × StepTrace Action State remaining -> Real :=
319         fun p =>
320           tilt * mdp.transitionResidualFrom targetState targetAction targetNextState
321               remaining p.1.2 p.2 -
322             tilt ^ 2 * variance *
323               mdp.transitionVisitFrom targetState targetAction remaining p.1.2 p.2
324       have hcons : Measurable consStep := StepTrace.measurable_cons remaining
325       have hZ : Measurable Z :=
326         ((mdp.measurable_transitionResidualFrom targetState targetAction
327           targetNextState (remaining + 1)).comp
328             (measurable_const.prodMk measurable_id)).const_mul tilt |>.sub
329           (((mdp.measurable_transitionVisitFrom targetState targetAction
330             (remaining + 1)).comp
331               (measurable_const.prodMk measurable_id)).const_mul (tilt ^ 2 * variance))
332       have hpair : Integrable (fun p => Real.exp (1 * Z (consStep p)))
333           (((policy.actionStateKernel stage).compProd tailKernel) currentState) :=
334         integrable_of_fintype _ _ (Real.measurable_exp.comp
335           ((hZ.comp hcons).const_mul 1))
336       have hhead :=
337         mdp.actionStateKernel_transitionResidualHead_variance_compensated_hasMGFUpperBoundAt
338           policy targetState targetAction targetNextState currentState stage tilt htilt
339       have htail : ∀ head : Action × State,
340           Concentration.HasMGFUpperBoundAt
341             (fun tail => Ztail (head, tail)) 1 0
342             (policy.trajectoryKernelRemaining remaining (by omega) head.2) := by
343         intro head
344         simpa [Ztail, variance] using ih (by omega) head.2
345       constructor
346       · intro s
347         exact integrable_of_fintype _ _ (by fun_prop)
348       · rw [ProbabilityTheory.mgf]
349         change (∫ trace, Real.exp (1 * Z trace)
350           ∂policy.trajectoryKernelRemaining (remaining + 1) hremaining currentState) <=
351             Real.exp 0
352         rw [MarkovPolicy.trajectoryKernelRemaining]
353         rw [ProbabilityTheory.Kernel.map_apply _ hcons]
354         change (∫ trace, (fun trace => Real.exp (1 * Z trace)) trace
355           ∂Measure.map consStep
356             (((policy.actionStateKernel stage).compProd tailKernel)
357               currentState)) <= Real.exp 0
358         have hfm : AEStronglyMeasurable
359             (fun trace => Real.exp (1 * Z trace))
360             (Measure.map consStep
361               (((policy.actionStateKernel stage).compProd tailKernel)
362                 currentState)) :=
363           (Real.measurable_exp.comp (hZ.const_mul 1)).aestronglyMeasurable
364         rw [integral_map hcons.aemeasurable hfm]
365         rw [ProbabilityTheory.integral_compProd hpair]
366         simp only [Real.exp_zero]
367         have hinner : ∀ head : Action × State,
368             (∫ tail, Real.exp (1 * Z (consStep (head, tail)))
369               ∂tailKernel (currentState, head)) <= Real.exp (Zhead head) := by
370           intro head
371           have htailApply : tailKernel (currentState, head) =
372               policy.trajectoryKernelRemaining remaining (by omega) head.2 := by
373             rfl
374           rw [htailApply]
375           have hz : ∀ tail,
376               Z (consStep (head, tail)) = Zhead head + Ztail (head, tail) := by
377             intro tail
378             simp [Z, Zhead, Ztail, consStep, variance, transitionResidualFrom,
379               transitionVisitFrom]
380             ring
381           simp_rw [hz, mul_add, Real.exp_add, one_mul, integral_const_mul]
382           have htailMgf :
383               (∫ tail, Real.exp (Ztail (head, tail))
384                 ∂policy.trajectoryKernelRemaining remaining (by omega) head.2) <= 1 := by
385             simpa [ProbabilityTheory.mgf] using (htail head).mgf_le
386           simpa using
387             (mul_le_mul_of_nonneg_left htailMgf (Real.exp_pos (Zhead head)).le)
388         have hleft : Integrable
389             (fun head =>
390               ∫ tail, Real.exp (1 * Z (consStep (head, tail)))
391                 ∂tailKernel (currentState, head))
392             (policy.actionStateKernel stage currentState) :=
393           integrable_of_fintype _ _ (measurable_of_finite _)
394         have hright : Integrable (fun head => Real.exp (Zhead head))
395             (policy.actionStateKernel stage currentState) :=
396           integrable_of_fintype _ _ (measurable_of_finite _)
397         calc
398           (∫ head,
399               ∫ tail, Real.exp (1 * Z (consStep (head, tail)))
400                 ∂tailKernel (currentState, head)
401               ∂policy.actionStateKernel stage currentState) <=
402               ∫ head, Real.exp (Zhead head)
403                 ∂policy.actionStateKernel stage currentState := by
404                   exact integral_mono_ae hleft hright
405                     (Filter.Eventually.of_forall hinner)
406           _ <= 1 := by
407             simpa [ProbabilityTheory.mgf, Zhead, variance] using hhead.mgf_le

546           simp only [Function.comp_apply]

689 theorem measure_aggregateTransitionResidualSum_ge_inter_visitSum_le_variance

691     [IsProbabilityMeasure initialState]
692     [StandardBorelSpace (EpisodeBatch mdp 1)]
693     [StandardBorelSpace (EpisodeBatchTrajectory mdp 1)]

697     (htilt : 0 < tilt) (htilt_le : tilt <= 1) :

701               ∑ i ∈ Finset.range rounds,
702                 source.aggregateTransitionResidualIncrement
703                   state action nextState i trajectory ∧
704             (∑ i ∈ Finset.range rounds,
705                 source.aggregateVisitIncrement state action i trajectory) <=
706               visitBudget} <=
707       ENNReal.ofReal

716     source.aggregateVisitIncrement state action i trajectory
717   let Z : Nat -> EpisodeBatchTrajectory mdp 1 -> Real := fun i trajectory =>
718     tilt * Y i trajectory - tilt ^ 2 * variance * V i trajectory

721     simpa [F, Y, V, Z, variance] using
722       source.aggregateTransitionResidual_compensated_stronglyAdapted_piLE
723         state action nextState tilt (tilt ^ 2 * variance)
724   have hzero : Concentration.HasMGFUpperBoundAt (Z 0) 1 0
725       source.trajectoryMeasure := by
726     simpa [Y, V, Z, variance] using
727       source.aggregateTransitionResidual_zero_variance_compensated_hasMGFUpperBoundAt
728         state action nextState tilt htiltAbs
729   have hsucc : ∀ i, i < rounds - 1 ->
730       Concentration.HasCondMGFUpperBoundAt
731         (F i) (F.le i) (Z (i + 1)) 1 0 source.trajectoryMeasure := by

739       (tilt ^ 2 * variance) threshold visitBudget hadapted hzero hsucc htilt.le

767   let positive : Set (EpisodeBatchTrajectory mdp 1) :=
