72 /-- The selected value in the backward recursion is the selected entry of the
73 chronological clipped Q table. -/
74 theorem clippedValueRemaining_succ_eq_selectedQ
75     {mdp : MDP State Action} (previousQ : QTable mdp)
76     (summary : TransitionCountSummary mdp) (defaultState : State)
77     (bonusScale : Real) (remaining : Nat)
78     (hremaining : remaining + 1 <= mdp.horizon) (state : State) :
79     clippedValueRemaining previousQ summary defaultState bonusScale
80         (remaining + 1) hremaining state =
81       clippedQRemaining previousQ summary defaultState bonusScale remaining
82         hremaining state
83         (clippedPolicyTable mdp previousQ summary defaultState bonusScale
84           (mdp.decisionStageRemaining remaining hremaining) state) := by
85   let stage := mdp.decisionStageRemaining remaining hremaining
86   let q : Action -> Real := fun action =>
87     clippedQRemaining previousQ summary defaultState bonusScale remaining
88       hremaining state action
89   have hq (action : Action) :
90       clippedQTable mdp previousQ summary defaultState bonusScale
91           stage state action = q action := by
92     unfold clippedQTable
93     change clippedQRemaining previousQ summary defaultState bonusScale
94       (mdp.horizon - (stage.val + 1)) _ state action = q action
95     have hinverse : mdp.horizon - (stage.val + 1) = remaining := by
96       simp [stage, MDP.decisionStageRemaining]
97       omega
98     exact congrFun (congrFun
99       (clippedQRemaining_eq_of_eq previousQ summary defaultState bonusScale
100         _ hremaining hinverse) state) action
101   have hchoice :
102       FiniteRealArgmax.choose
103           (fun action => clippedQTable mdp previousQ summary defaultState
104             bonusScale stage state action) =
105         FiniteRealArgmax.choose q := by
106     congr 1
107     funext action
108     exact hq action
109   simp only [clippedValueRemaining]
110   change q (FiniteRealArgmax.choose q) =
111     q (clippedPolicyTable mdp previousQ summary defaultState bonusScale
112       stage state)
113   unfold clippedPolicyTable
114   rw [hchoice]

150 /-- Pointwise optimal-Q dominance implies that the selected clipped value
151 dominates the optimal value at the matching chronological stage. -/
152 theorem optimalValueAt_le_clippedValueRemaining
153     {mdp : MDP State Action} (previousQ : QTable mdp)
154     (summary : TransitionCountSummary mdp) (defaultState : State)
155     (bonusScale : Real)
156     (hdominates : QDominatesOptimal mdp
157       (clippedQTable mdp previousQ summary defaultState bonusScale))
158     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
159     (state : State) :
160     mdp.optimalValueAt (mdp.horizon - remaining) (Nat.sub_le _ _ ) state <=
161       clippedValueRemaining previousQ summary defaultState bonusScale
162         remaining hremaining state := by
163   cases remaining with
164   | zero =>
165       simp [clippedValueRemaining, mdp.optimalValueAt_horizon]
166   | succ remaining =>
167       let stage := mdp.decisionStageRemaining remaining hremaining
168       let tail := mdp.optimalValueAt (stage + 1)
169         (Nat.succ_le_of_lt stage.isLt)
170       let optimalAction := mdp.optimalAction tail state
171       let selected := clippedPolicyTable mdp previousQ summary defaultState
172         bonusScale stage state
173       have hstage : mdp.horizon - (remaining + 1) = stage := by
174         rfl
175       have hopt : mdp.optimalValueAt stage (Nat.le_of_lt stage.isLt) state =
176           optimalQAt mdp stage state optimalAction := by
177         rw [mdp.optimalValueAt_bellman stage stage.isLt]
178         rfl
179       have hentry := hdominates stage state optimalAction
180       have hselected := clippedQTable_le_selected mdp previousQ summary
181         defaultState bonusScale stage state optimalAction
182       have htarget :
183           mdp.optimalValueAt (mdp.horizon - (remaining + 1))
184               (Nat.sub_le _ _) state =
185             optimalQAt mdp stage state optimalAction := by
186         calc
187           mdp.optimalValueAt (mdp.horizon - (remaining + 1))
188               (Nat.sub_le _ _) state =
189               mdp.optimalValueAt stage (Nat.le_of_lt stage.isLt) state := by
190             congr 2
191           _ = optimalQAt mdp stage state optimalAction := hopt
192       rw [htarget]
193       rw [clippedValueRemaining_succ_eq_selectedQ]
194       have hq :
195           clippedQTable mdp previousQ summary defaultState bonusScale
196               stage state selected =
197             clippedQRemaining previousQ summary defaultState bonusScale
198               remaining hremaining state selected := by
199         unfold clippedQTable
200         change clippedQRemaining previousQ summary defaultState bonusScale
201           (mdp.horizon - (stage.val + 1)) _ state selected = _
202         have hinverse : mdp.horizon - (stage.val + 1) = remaining := by
203           simp [stage, MDP.decisionStageRemaining]
204           omega
205         exact congrFun (congrFun
206           (clippedQRemaining_eq_of_eq previousQ summary defaultState bonusScale
207             _ hremaining hinverse) state) selected
208       change optimalQAt mdp stage state optimalAction <=
209         clippedQRemaining previousQ summary defaultState bonusScale remaining
210           hremaining state selected
211       exact (hentry.trans hselected).trans_eq hq

213 /-- At a positive actual count, the selected clipped value minus the selected
214 policy value propagates through the true transition kernel, plus exactly the
215 empirical-model error and the configured bonus. -/

222       (clippedPolicyTable mdp previousQ summary defaultState bonusScale
223         (mdp.decisionStageRemaining remaining hremaining) state))
224     (hmodel :
225       (∫ nextState,
226           clippedValueRemaining previousQ summary defaultState bonusScale
227               remaining (by omega) nextState
228         ∂summary.aggregateEmpiricalTransitionKernel defaultState
229           (state,
230             clippedPolicyTable mdp previousQ summary defaultState bonusScale
231               (mdp.decisionStageRemaining remaining hremaining) state)) -
232         mdp.transitionValue
233           (clippedValueRemaining previousQ summary defaultState bonusScale
234             remaining (by omega)) state
235           (clippedPolicyTable mdp previousQ summary defaultState bonusScale
236             (mdp.decisionStageRemaining remaining hremaining) state) <=
237         modelError) :

239           (remaining + 1) hremaining state -
240         (clippedPolicyTable mdp previousQ summary defaultState bonusScale
241           |>.toMarkovPolicy).valueRemaining (remaining + 1) hremaining state <=
242       mdp.transitionValue
243           (fun nextState =>
244             clippedValueRemaining previousQ summary defaultState bonusScale
245                 remaining (by omega) nextState -
246               (clippedPolicyTable mdp previousQ summary defaultState bonusScale
247                 |>.toMarkovPolicy).valueRemaining remaining (by omega) nextState)
248           state
249           (clippedPolicyTable mdp previousQ summary defaultState bonusScale
250             (mdp.decisionStageRemaining remaining hremaining) state) +
251         modelError +
252         bonusScale /
253           Real.sqrt (summary.aggregateVisitCount state
254             (clippedPolicyTable mdp previousQ summary defaultState bonusScale
255               (mdp.decisionStageRemaining remaining hremaining) state)) := by
256   let stage := mdp.decisionStageRemaining remaining hremaining
257   let table := clippedPolicyTable mdp previousQ summary defaultState bonusScale
258   let selected := table stage state
259   let upperTail := clippedValueRemaining previousQ summary defaultState bonusScale
260       remaining (by omega)
261   let policy := table.toMarkovPolicy
262   let policyTail := policy.valueRemaining remaining (by omega)
263   let count := summary.aggregateVisitCount state selected
264   have hq := clippedQRemaining_of_aggregateVisitCount_pos previousQ summary
265     defaultState bonusScale remaining hremaining state selected hpos
266   have hupper :
267       clippedValueRemaining previousQ summary defaultState bonusScale
268           (remaining + 1) hremaining state <=
269         mdp.reward state selected +
270           (∫ nextState, upperTail nextState
271             ∂summary.aggregateEmpiricalTransitionKernel defaultState
272               (state, selected)) +
273           bonusScale / Real.sqrt count := by
274     rw [clippedValueRemaining_succ_eq_selectedQ, hq]
275     exact (min_le_right _ _).trans (min_le_right _ _)
276   have hpolicy :
277       policy.valueRemaining (remaining + 1) hremaining state =
278         mdp.reward state selected +
279           mdp.transitionValue policyTail state selected := by
280     rw [MarkovPolicy.valueRemaining]
281     change policy.bellman stage policyTail state = _
282     unfold MarkovPolicy.bellman policy
283     unfold DeterministicMarkovPolicyTable.toMarkovPolicy
284     rw [ProbabilityTheory.Kernel.deterministic_apply]
285     rw [integral_dirac']
286     · rfl
287     · exact
288         ((mdp.measurable_bellmanQ (measurable_of_finite policyTail)).comp
289           (measurable_const.prodMk measurable_id)).stronglyMeasurable
290   have htransitionSub :
291       mdp.transitionValue upperTail state selected -
292           mdp.transitionValue policyTail state selected =
293         mdp.transitionValue (fun nextState =>
294           upperTail nextState - policyTail nextState) state selected := by
295     unfold MDP.transitionValue
296     have hupperInt : Integrable upperTail (mdp.transition (state, selected)) :=
297       integrable_of_fintype _ _ (measurable_of_finite _)
298     have hpolicyInt : Integrable policyTail (mdp.transition (state, selected)) :=
299       integrable_of_fintype _ _ (measurable_of_finite _)
300     rw [integral_sub hupperInt hpolicyInt]
301   change _ <= _
302   rw [hpolicy]
303   change _ <= mdp.transitionValue
304       (fun nextState => upperTail nextState - policyTail nextState)
305       state selected + modelError + bonusScale / Real.sqrt count
306   rw [← htransitionSub]
307   have hmodel' :
308       (∫ nextState, upperTail nextState
309         ∂summary.aggregateEmpiricalTransitionKernel defaultState
310           (state, selected)) -
311         mdp.transitionValue upperTail state selected <= modelError := by
312     simpa [upperTail, selected, table, stage] using hmodel
