6 # Finite-horizon stochastic reward head concentration
7 
8 This module transports a selected reward-kernel sub-Gaussian law through the
9 compiled generated-head conditional distribution.  It exposes the centered
10 head reward as a conditional and unconditional sub-Gaussian random variable,
11 then specializes the existing finite-sum concentration route to a one-step
12 two-sided tail.
13 -/

30 def headRewardMean (mdp : MDP State Action) (state : State)
31     (remaining : Nat) :
32     RewardStepTrace Action State (remaining + 1) -> Real :=
33   fun trace => mdp.reward state
34     (RewardStepTrace.headAction

58 /-- Actual first sampled reward centered by the mean of its selected action law. -/
59 def headRewardDeviation (mdp : MDP State Action) (state : State)
60     (remaining : Nat) :
61     RewardStepTrace Action State (remaining + 1) -> Real :=
62   fun trace =>
63     RewardStepTrace.headReward
64         (Action := Action) (State := State) remaining trace -

163     [StandardBorelSpace State] [StandardBorelSpace Action]
164     [Nonempty Action]

200 theorem stochasticTrajectoryKernelRemaining_headRewardDeviation_abs_tail_le
201     [StandardBorelSpace State] [StandardBorelSpace Action]
202     [Nonempty Action]
203     (source : MeanCompatibleRewardKernel mdp)
204     (policy : MarkovPolicy mdp)
205     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
206     (state : State) (varianceProxy : NNReal)
207     (law : source.UniformSubgaussianRewardLaw varianceProxy)
208     (hvariance : 0 < (varianceProxy : Real))
209     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1) :
210     (source.stochasticTrajectoryKernelRemaining policy
211       (remaining + 1) hremaining state)
212       {trace |
213         Concentration.subGaussianSumConfidenceRadius varianceProxy delta <=
214           |mdp.headRewardDeviation state remaining trace|} <=
215       ENNReal.ofReal delta := by
216   letI : Nonempty State := ⟨state⟩
217   let mu := source.stochasticTrajectoryKernelRemaining policy
218     (remaining + 1) hremaining state
219   let F : Filtration Nat
220       (inferInstance : MeasurableSpace
221         (RewardStepTrace Action State (remaining + 1))) :=
222     Filtration.const Nat inferInstance le_rfl
223   let Y : Nat -> RewardStepTrace Action State (remaining + 1) -> Real :=
224     fun _ => mdp.headRewardDeviation state remaining
225   let cY : Nat -> NNReal := fun _ => varianceProxy
226   have hadapted : StronglyAdapted F Y := by
227     intro i
