4 # Self-consistent stochastic transition calibration
5 
6 The half-contraction calibration uses the coarse fixed point
7 `transitionBudget = rewardBound + 2 * rewardBudget`.  Here the actual
8 contraction factor `q < 1` is retained and the fixed point is solved exactly,
9 so the transition budget shrinks with `q`.
10 -/

19 /-- Exact nonnegative fixed-point budget for `q * (base + budget) <= budget`. -/
20 noncomputable def selfConsistentTransitionBudget
21     (q base : Real) : Real :=
22   q * base / (1 - q)
23 
24 /-- The fixed-point budget is nonnegative when `0 <= q < 1` and `base >= 0`. -/
25 theorem selfConsistentTransitionBudget_nonneg
26     {q base : Real} (hq_nonneg : 0 <= q) (hq : q < 1)
27     (hbase : 0 <= base) :
28     0 <= selfConsistentTransitionBudget q base := by
29   unfold selfConsistentTransitionBudget
30   exact div_nonneg (mul_nonneg hq_nonneg hbase) (sub_nonneg.mpr hq.le)
31 
32 /-- The chosen budget solves the transition-envelope fixed point exactly. -/
33 theorem selfConsistentTransitionBudget_fixedPoint
34     {q base : Real} (hq : q < 1) :
35     q * (base + selfConsistentTransitionBudget q base) =
36       selfConsistentTransitionBudget q base := by
37   have hdenom : 1 - q ≠ 0 := ne_of_gt (sub_pos.mpr hq)
38   unfold selfConsistentTransitionBudget
39   field_simp [hdenom]
40   ring

49 /-- The exact common-floor transition contraction factor. -/
50 noncomputable def uniformFloorStochasticTransitionContraction
51     (mdp : MDP State Action) (episodes : Nat)
52     (countDelta visitFloor : Real) : Real :=
53   (Fintype.card State : Real) *
54     uniformFloorTransitionCoordinateRadius mdp episodes countDelta visitFloor *
55     (mdp.horizon : Real)

60 /-- The common-floor contraction factor is nonnegative under the count margin. -/
61 theorem uniformFloorStochasticTransitionContraction_nonneg
62     {mdp : MDP State Action} {episodes : Nat}
63     {countDelta visitFloor : Real}
64     (hmargin : simultaneousCountConfidenceRadius mdp episodes countDelta <
65       (episodes : Real) * visitFloor) :
66     0 <= uniformFloorStochasticTransitionContraction
67       mdp episodes countDelta visitFloor := by

71       (uniformFloorTransitionCoordinateRadius_nonneg hmargin))

74 /-- Shrinking transition budget obtained from the exact contraction factor. -/
75 noncomputable def uniformFloorStochasticSelfConsistentTransitionBudget
76     (mdp : MDP State Action) (episodes : Nat)
77     (countDelta visitFloor rewardBound rewardBudget : Real) : Real :=
78   selfConsistentTransitionBudget
79     (uniformFloorStochasticTransitionContraction
80       mdp episodes countDelta visitFloor)
81     (rewardBound + 2 * rewardBudget)

107 /-- The common-floor shrinking budget satisfies the exact envelope identity. -/
108 theorem uniformFloorStochasticSelfConsistentTransitionBudget_fixedPoint
109     {mdp : MDP State Action} {episodes : Nat}
110     {countDelta visitFloor rewardBound rewardBudget : Real}
111     (hq : uniformFloorStochasticTransitionContraction
112       mdp episodes countDelta visitFloor < 1) :

128 The exact `q < 1` fixed point covers every transition-radius/value-envelope
129 sum with a budget that shrinks as `q` tends to zero.

131 theorem stochasticTransitionCover_of_uniformExpectedCountFloor_selfConsistent
132     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
133     (initialState : Measure State) [IsProbabilityMeasure initialState]
134     (episodes : Nat) (varianceProxy : NNReal)

136     (hmargin : simultaneousCountConfidenceRadius mdp episodes countDelta <

139       (episodes : Real) * visitFloor <=

149         countDelta visitFloor rewardBound rewardBudget

152       (∑ nextState,
153           policy.expectedCountTransitionCoordinateRadius initialState episodes
154               countDelta (mdp.decisionStageRemaining remaining hremaining)
155               state action nextState *
156             stochasticEmpiricalFiniteBatchValueEnvelope rewardBound
157               rewardBudget transitionBudget remaining) <=
158         transitionBudget := by

163       countDelta rewardDelta visitFloor
164   let transitionBudget :=

166       countDelta visitFloor rewardBound rewardBudget
167   let q := uniformFloorStochasticTransitionContraction
168     mdp episodes countDelta visitFloor
169   have hrewardBudget_nonneg : 0 <= rewardBudget :=
170     uniformFloorStochasticRewardCoordinateRadius_nonneg hmargin
171   have htransitionBudget_nonneg : 0 <= transitionBudget := by
172     exact uniformFloorStochasticSelfConsistentTransitionBudget_nonneg
173       hmargin hrewardBound_nonneg hrewardBudget_nonneg hq
174   have huniform_nonneg :
175       0 <= uniformFloorTransitionCoordinateRadius mdp episodes countDelta
176         visitFloor :=
177     uniformFloorTransitionCoordinateRadius_nonneg hmargin
178   have henvelope_nonneg :
179       0 <= stochasticEmpiricalFiniteBatchValueEnvelope rewardBound rewardBudget
180         transitionBudget remaining := by
181     unfold stochasticEmpiricalFiniteBatchValueEnvelope
182     positivity
183   have hremaining_le : (remaining : Real) <= (mdp.horizon : Real) := by
184     exact_mod_cast (show remaining <= mdp.horizon by omega)
185   have hcoefficient :
186       (Fintype.card State : Real) *
187           uniformFloorTransitionCoordinateRadius mdp episodes countDelta
188             visitFloor *
189           (remaining : Real) <=
190         q := by
191     dsimp [q, uniformFloorStochasticTransitionContraction]
192     exact mul_le_mul_of_nonneg_left hremaining_le
193       (mul_nonneg (Nat.cast_nonneg _) huniform_nonneg)
194   have hbase_nonneg :
195       0 <= rewardBound + 2 * rewardBudget + transitionBudget := by
196     positivity
197   change
198     (∑ nextState,
199         policy.expectedCountTransitionCoordinateRadius initialState episodes
200             countDelta (mdp.decisionStageRemaining remaining hremaining)
201             state action nextState *
202           stochasticEmpiricalFiniteBatchValueEnvelope rewardBound rewardBudget
203             transitionBudget remaining) <=
204       transitionBudget
205   calc
206     (∑ nextState,
207         policy.expectedCountTransitionCoordinateRadius initialState episodes
208             countDelta (mdp.decisionStageRemaining remaining hremaining)
209             state action nextState *
210           stochasticEmpiricalFiniteBatchValueEnvelope rewardBound rewardBudget
211             transitionBudget remaining) <=
212         ∑ _nextState : State,
213           uniformFloorTransitionCoordinateRadius mdp episodes countDelta
214               visitFloor *
215             stochasticEmpiricalFiniteBatchValueEnvelope rewardBound rewardBudget
216               transitionBudget remaining := by
217       apply Finset.sum_le_sum
218       intro nextState _hnextState
219       exact mul_le_mul_of_nonneg_right
220         (policy.expectedCountTransitionCoordinateRadius_le_uniformFloor
221           initialState episodes countDelta visitFloor hmargin hcountFloor
222           (mdp.decisionStageRemaining remaining hremaining)
223           state action nextState)
224         henvelope_nonneg
225     _ = ((Fintype.card State : Real) *
226           uniformFloorTransitionCoordinateRadius mdp episodes countDelta
227             visitFloor *
228           (remaining : Real)) *
229         (rewardBound + 2 * rewardBudget + transitionBudget) := by
230       simp [stochasticEmpiricalFiniteBatchValueEnvelope]
231       ring
232     _ <= q * (rewardBound + 2 * rewardBudget + transitionBudget) :=
233       mul_le_mul_of_nonneg_right hcoefficient hbase_nonneg
234     _ = transitionBudget := by
