5 # Finite-horizon stochastic reward trajectory head conditional law
6 
7 This module identifies the selected reward kernel as the conditional law of
8 the first sampled reward given the first sampled action.  The result is first
9 proved for the one-step action/reward marginal, then transported to every
10 positive generated stochastic trajectory and to the corresponding trimmed
11 `condExpKernel.map` surface.
12 -/

29 /-- The first sampled action of a positive reward-bearing trace. -/
30 def headAction (remaining : Nat) :
31     RewardStepTrace Action State (remaining + 1) -> Action :=
32   fun trace => (trace 0).1

45 /-- The reward kernel selected after freezing the current state. -/
46 noncomputable def selectedRewardKernelAt
47     (source : MeanCompatibleRewardKernel mdp) (state : State) :
48     ProbabilityTheory.Kernel Action Real :=
49   ProbabilityTheory.Kernel.sectR source.rewardKernel.kernel state

51 instance instSelectedRewardKernelAtIsMarkovKernel
52     (source : MeanCompatibleRewardKernel mdp) (state : State) :
53     ProbabilityTheory.IsMarkovKernel
54       (source.selectedRewardKernelAt state) := by
55   unfold selectedRewardKernelAt
56   infer_instance

58 /-- The one-step action/reward marginal is the policy law composed with the selected reward law. -/
59 theorem actionRewardKernel_eq_compProd_selectedRewardKernelAt
60     (source : MeanCompatibleRewardKernel mdp)
61     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
62     (state : State) :
63     source.actionRewardKernel policy stage state =
64       policy.actionKernel stage state ⊗ₘ source.selectedRewardKernelAt state := by
65   apply Measure.ext_prod
66   intro actionSet rewardSet hactionSet hrewardSet
67   rw [source.actionRewardKernel_apply_prod policy stage state
68     hactionSet hrewardSet]
69   rw [Measure.compProd_apply_prod hactionSet hrewardSet]
70   rfl
71 
72 /-- The action marginal of the one-step action/reward law is the policy action law. -/
73 theorem actionRewardKernel_map_fst
74     (source : MeanCompatibleRewardKernel mdp)
75     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
76     (state : State) :
77     (source.actionRewardKernel policy stage state).map Prod.fst =
78       policy.actionKernel stage state := by
79   rw [source.actionRewardKernel_eq_compProd_selectedRewardKernelAt]
80   change
81     (policy.actionKernel stage state ⊗ₘ
82       source.selectedRewardKernelAt state).fst = _
83   rw [Measure.fst_compProd]

85 /-- Mapping a generated trace to its first action recovers the policy action law. -/
86 theorem stochasticTrajectoryKernelRemaining_map_headAction
87     (source : MeanCompatibleRewardKernel mdp)
88     (policy : MarkovPolicy mdp)
89     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
90     (state : State) :
91     (source.stochasticTrajectoryKernelRemaining policy
92         (remaining + 1) hremaining state).map
93         (RewardStepTrace.headAction
94           (Action := Action) (State := State) remaining) =
95       policy.actionKernel
96         ⟨mdp.horizon - (remaining + 1), by omega⟩ state := by
97   calc
98     _ = (Measure.map
99           (RewardStepTrace.headActionReward
100             (Action := Action) (State := State) remaining)
101           (source.stochasticTrajectoryKernelRemaining policy
102             (remaining + 1) hremaining state)).map Prod.fst := by
103           rw [Measure.map_map measurable_fst
104             (RewardStepTrace.measurable_headActionReward remaining)]
105           rfl
106     _ = (source.actionRewardKernel policy
107           ⟨mdp.horizon - (remaining + 1), by omega⟩ state).map Prod.fst := by
108           rw [source.stochasticTrajectoryKernelRemaining_map_headActionReward
109             policy remaining hremaining state]
110     _ = policy.actionKernel
111           ⟨mdp.horizon - (remaining + 1), by omega⟩ state := by
112           rw [source.actionRewardKernel_map_fst]

114 /-- Under the one-step joint law, reward conditioned on action is the selected reward kernel. -/
115 theorem actionRewardKernel_condDistrib_reward_given_action
116     (source : MeanCompatibleRewardKernel mdp)
117     (policy : MarkovPolicy mdp) (stage : Fin mdp.horizon)
118     (state : State) :
119     ProbabilityTheory.condDistrib Prod.snd Prod.fst
120         (source.actionRewardKernel policy stage state) =ᵐ[
121       (source.actionRewardKernel policy stage state).map Prod.fst]
122       source.selectedRewardKernelAt state := by
123   apply (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
124     (μ := source.actionRewardKernel policy stage state)
125     Prod.fst measurable_snd.aemeasurable
126     (source.selectedRewardKernelAt state)).2
127   rw [source.actionRewardKernel_map_fst]
128   change
129     (source.actionRewardKernel policy stage state).map id =
130       policy.actionKernel stage state ⊗ₘ source.selectedRewardKernelAt state
131   rw [Measure.map_id]
132   exact source.actionRewardKernel_eq_compProd_selectedRewardKernelAt
133     policy stage state

175 /--
176 The generated head conditional law on `Real` is also the mapped conditional
177 expectation kernel on the sigma-algebra generated by the sampled head action.
178 -/
179 theorem stochasticTrajectoryKernelRemaining_condExpKernel_map_headReward_given_headAction
180     [StandardBorelSpace State] [StandardBorelSpace Action]
181     [Nonempty Action]
182     (source : MeanCompatibleRewardKernel mdp)
183     (policy : MarkovPolicy mdp)
184     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
185     (state : State) :
186     ∀ᵐ trace ∂
187       (source.stochasticTrajectoryKernelRemaining policy
188         (remaining + 1) hremaining state).trim
189         (RewardStepTrace.measurable_headAction remaining).comap_le,
190       Measure.map
191           (RewardStepTrace.headReward
192             (Action := Action) (State := State) remaining)
193           (ProbabilityTheory.condExpKernel
194             (source.stochasticTrajectoryKernelRemaining policy
195               (remaining + 1) hremaining state)
196             (MeasurableSpace.comap
197               (RewardStepTrace.headAction
198                 (Action := Action) (State := State) remaining)
199               inferInstance)
200             trace) =
201         source.selectedRewardKernelAt state
202           (RewardStepTrace.headAction
203             (Action := Action) (State := State) remaining trace) := by
204   letI : Nonempty State := ⟨state⟩
205   exact
206     ConditionalExpectationReward.condExpKernel_map_eq_of_condDistrib_ae_eq_real_trim
207       (source.stochasticTrajectoryKernelRemaining policy
208         (remaining + 1) hremaining state)

210         (Action := Action) (State := State) remaining)

212         (Action := Action) (State := State) remaining)
