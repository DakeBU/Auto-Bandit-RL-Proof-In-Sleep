12 /-- A function with a countable discrete left coordinate is measurable when
13 every right section is measurable. -/
14 theorem measurable_prod_of_countable_left
15     {Alpha Beta Gamma : Type*}
16     [MeasurableSpace Alpha] [MeasurableSpace Beta] [MeasurableSpace Gamma]
17     [Countable Alpha] [MeasurableSingletonClass Alpha]
18     (f : Alpha × Beta -> Gamma)
19     (hf : forall a, Measurable (fun b => f (a, b))) :
20     Measurable f := by
21   intro s hs
22   have hpreimage : f ⁻¹' s =
23       ⋃ a : Alpha, ({a} : Set Alpha) ×ˢ
24         ((fun b => f (a, b)) ⁻¹' s) := by
25     ext p
26     simp only [Set.mem_preimage, Set.mem_iUnion, Set.mem_prod,
27       Set.mem_singleton_iff]
28     constructor
29     · intro hp
30       exact ⟨p.1, rfl, hp⟩
31     · rintro ⟨a, ha, hp⟩
32       subst a
33       exact hp
34   rw [hpreimage]
35   exact MeasurableSet.iUnion fun a =>
36     MeasurableSet.singleton a |>.prod ((hf a) hs)
37 
38 end BanditRLProof

42 /-- Mapping a composition-product by a function which may also read the
43 conditioning coordinate is the composition-product with the correspondingly
44 mapped copy-and-kernel law. -/
45 lemma compProd_map_dependent
46     {Alpha : Type u} {Beta : Type v} {Gamma : Type w}
47     [MeasurableSpace Alpha] [MeasurableSpace Beta] [MeasurableSpace Gamma]
48     (mu : Measure Alpha) [SFinite mu]
49     (kappa : ProbabilityTheory.Kernel Alpha Beta)
50     [ProbabilityTheory.IsSFiniteKernel kappa]
51     (f : Alpha × Beta -> Gamma) (hf : Measurable f) :
52     (mu ⊗ₘ kappa).map (fun p => (p.1, f p)) =
53       mu ⊗ₘ ((ProbabilityTheory.Kernel.id ×ₖ kappa).map f) := by
54   ext s hs
55   rw [Measure.map_apply (measurable_fst.prodMk hf) hs,
56     Measure.compProd_apply
57       (hs.preimage (measurable_fst.prodMk hf)),
58     Measure.compProd_apply hs]
59   congr with a
60   rw [ProbabilityTheory.Kernel.map_apply _ hf]
61   rw [ProbabilityTheory.Kernel.prod_apply]
62   rw [ProbabilityTheory.Kernel.id_apply, Measure.dirac_prod]
63   rw [Measure.map_map hf measurable_prodMk_left]
64   rw [Measure.map_apply (hf.comp measurable_prodMk_left)
65     (hs.preimage measurable_prodMk_left)]
66   rfl
67 
68 end MeasureTheory.Measure

221 /-- Inflation factor generated by the `z/(32H)` self-bounding term. -/
222 noncomputable def bellmanInflation (mdp : MDP State Action) : Real :=
223   1 + 1 / (32 * (mdp.horizon : Real))

225 /-- Uniform cap for every finite-horizon power of `bellmanInflation`. -/
226 noncomputable def bellmanWeightCap : Real := 32 / 31

228 theorem bellmanInflation_one_le
229     (mdp : MDP State Action) :
230     1 <= bellmanInflation mdp := by
231   unfold bellmanInflation
232   have : (0 : Real) <= 1 / (32 * (mdp.horizon : Real)) := by positivity
233   linarith

235 theorem bellmanInflation_pow_le_cap
236     (mdp : MDP State Action) (hhorizon : 0 < mdp.horizon)
237     (remaining : Nat) (hremaining : remaining <= mdp.horizon) :
238     bellmanInflation mdp ^ remaining <= bellmanWeightCap := by
239   let x : Real := 1 / (32 * (mdp.horizon : Real))
240   have hhorizonReal : 0 < (mdp.horizon : Real) := by exact_mod_cast hhorizon
241   have hx : 0 <= x := by dsimp [x]; positivity
242   have hbase : bellmanInflation mdp <= Real.exp x := by
243     simpa [bellmanInflation, x, add_comm] using Real.add_one_le_exp x
244   have hpow : bellmanInflation mdp ^ remaining <= Real.exp x ^ remaining := by
245     have hnonneg : 0 <= bellmanInflation mdp := by
246       linarith [bellmanInflation_one_le mdp]
247     gcongr
248   have hremainingReal : (remaining : Real) <= mdp.horizon := by exact_mod_cast hremaining
249   have hmul : (remaining : Real) * x <= (1 : Real) / 32 := by
250     dsimp [x]
251     have hdiv := div_le_div_of_nonneg_right hremainingReal
252       (by positivity : (0 : Real) <= 32 * mdp.horizon)
253     calc
254       (remaining : Real) * (1 / (32 * (mdp.horizon : Real))) =
255           (remaining : Real) / (32 * (mdp.horizon : Real)) := by ring
256       _ <= (mdp.horizon : Real) / (32 * (mdp.horizon : Real)) := hdiv
257       _ = (1 : Real) / 32 := by field_simp
258   have hexp : Real.exp x ^ remaining <= Real.exp ((1 : Real) / 32) := by
259     rw [← Real.exp_nat_mul]
260     exact Real.exp_le_exp.mpr hmul
261   have hcap := Real.exp_bound_div_one_sub_of_interval
262     (x := (1 : Real) / 32) (by norm_num) (by norm_num)
263   calc
264     bellmanInflation mdp ^ remaining <= Real.exp x ^ remaining := hpow
265     _ <= Real.exp ((1 : Real) / 32) := hexp
266     _ <= 1 / (1 - (1 : Real) / 32) := hcap
267     _ = bellmanWeightCap := by norm_num [bellmanWeightCap]

269 /-- Normalized chronological recursion weight.  Multiplication by the global
270 cap `32/31` later recovers the exact factor `alpha^(stage+1)`. -/
271 noncomputable def normalizedBellmanWeight
272     (mdp : MDP State Action) (stage : Fin mdp.horizon) : Real :=
273   (31 / 32 : Real) * bellmanInflation mdp ^ (stage.val + 1)

330 /-- The clipped, normalized and chronologically weighted continuation gap used
331 by the predictable martingale.  Clipping is inactive on the joint confidence
332 event; normalization keeps the global MGF range exactly `[0,H]`. -/
333 noncomputable def clippedSuccessorGapFeatureOfSummaries
334     (mdp : MDP State Action) (defaultState : State)
335     (episodes : Nat) (delta : Real) (n : Nat)
336     (summaries : Fin (n + 1) -> TransitionCountSummary mdp)
337     (stage : Fin mdp.horizon) (state : State) : Real :=
338   let previous := recurrentQTableOfSummaries mdp defaultState
339     (scale (State := State) (Action := Action) mdp episodes delta) n
340     (fun i => summaries i.castSucc)
341   let summary := cumulativeSummaryOfSequence summaries
342   let remaining := mdp.horizon - (stage.val + 1)
343   let raw := clippedPolicyGapRemaining previous summary defaultState
344     (scale (State := State) (Action := Action) mdp episodes delta)
345     remaining (by omega) state
346   normalizedBellmanWeight mdp stage *
347     max 0 (min raw (mdp.horizon : Real))

349 /-- The martingale feature is globally in `[0,H]`, independently of whether
350 the statistical event holds. -/
351 theorem clippedSuccessorGapFeatureOfSummaries_mem_Icc
352     (mdp : MDP State Action) (defaultState : State)
353     (episodes : Nat) (delta : Real) (n : Nat)
354     (summaries : Fin (n + 1) -> TransitionCountSummary mdp)
355     (stage : Fin mdp.horizon) (state : State) :
356     clippedSuccessorGapFeatureOfSummaries mdp defaultState episodes delta n
357       summaries stage state ∈ Set.Icc (0 : Real) mdp.horizon := by
358   have hstage := stage.isLt
359   have hhorizon : 0 < mdp.horizon := by omega
360   have hweight := normalizedBellmanWeight_mem_Icc mdp hhorizon stage
361   have hgap : max 0 (min
362       (clippedPolicyGapRemaining
363         (recurrentQTableOfSummaries mdp defaultState
364           (scale (State := State) (Action := Action) mdp episodes delta) n
365           (fun i => summaries i.castSucc))
366         (cumulativeSummaryOfSequence summaries) defaultState
367         (scale (State := State) (Action := Action) mdp episodes delta)
368         (mdp.horizon - (stage.val + 1)) (by omega) state)
369       (mdp.horizon : Real)) ∈ Set.Icc (0 : Real) mdp.horizon := by
