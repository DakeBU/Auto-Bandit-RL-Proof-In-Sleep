4 # Generated stage-visit factorization
5 
6 This module factors a generated trajectory's stage/state/action visit mass
7 into the corresponding stage-state mass and the policy action-kernel singleton
8 mass.  It is a population-law identity only: no reachability, action support,
9 episode count, concentration, or regret premise is introduced.
10 -/

27 /-- Chronological MDP stage represented by one coordinate of a remaining trace. -/
28 def stageOfRemainingCoordinate (mdp : MDP State Action)
29     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
30     (coordinate : Fin remaining) : Fin mdp.horizon :=
31   ⟨mdp.horizon - remaining + coordinate.val, by omega⟩

59 /--
60 Inside a remaining generated trace, a state/action visit factors into the
61 state event and the action singleton selected by the chronological policy
62 kernel.
63 -/
64 theorem trajectoryKernelRemaining_visitEvent_eq_stateEvent_mul_action
65     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
66     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
67     (initial state : State) (action : Action)
68     (coordinate : Fin remaining) :
69     (policy.trajectoryKernelRemaining remaining hremaining initial)
70         {trace | StepTrace.stateActionAt initial trace coordinate =
71           (state, action)} =
72       (policy.trajectoryKernelRemaining remaining hremaining initial)
73           {trace | StepTrace.stateAt initial trace coordinate = state} *
74         policy.actionKernel
75           (mdp.stageOfRemainingCoordinate remaining hremaining coordinate)
76           state {action} := by
77   classical
78   induction remaining generalizing initial with
79   | zero => exact Fin.elim0 coordinate
80   | succ remaining ih =>
81       refine Fin.cases ?_ (fun tailCoordinate => ?_) coordinate
82       · by_cases hinitial : initial = state
83         · subst initial
84           have hhead := policy.trajectoryKernelRemaining_map_head
85             remaining hremaining state
86           have hactionSet : MeasurableSet
87               ({action} ×ˢ (Set.univ : Set State)) :=
88             (measurableSet_singleton action).prod MeasurableSet.univ
89           have hactionPreimage :
90               (fun trace : StepTrace Action State (remaining + 1) => trace 0) ⁻¹'
91                   ({action} ×ˢ (Set.univ : Set State)) =
92                 {trace | StepTrace.stateActionAt state trace 0 =
93                   (state, action)} := by
94             ext trace
95             simp [StepTrace.stateActionAt]
96           calc
97             _ = (policy.trajectoryKernelRemaining
98                   (remaining + 1) hremaining state).map
99                 (fun trace => trace 0) ({action} ×ˢ Set.univ) := by
100                   rw [Measure.map_apply (measurable_pi_apply 0) hactionSet]
101                   rw [hactionPreimage]
102             _ = policy.actionStateKernel
103                 ⟨mdp.horizon - (remaining + 1), by omega⟩ state
104                 ({action} ×ˢ Set.univ) := by rw [hhead]
105             _ = policy.actionKernel
106                 ⟨mdp.horizon - (remaining + 1), by omega⟩ state {action} :=
107               policy.actionStateKernel_apply_actionSet _ _ _
108             _ = _ := by
109               simp [StepTrace.stateAt, MDP.stageOfRemainingCoordinate]
110         · simp [StepTrace.stateActionAt, StepTrace.stateAt, hinitial]
111       · rw [mdp.stageOfRemainingCoordinate_succ remaining hremaining tailCoordinate]
112         rw [trajectoryKernelRemaining]
113         have hvisitSet : MeasurableSet
114             {trace : StepTrace Action State (remaining + 1) |
115               StepTrace.stateActionAt initial trace tailCoordinate.succ =
116                 (state, action)} :=
117           (StepTrace.measurable_stateActionAt initial tailCoordinate.succ)
118             (measurableSet_singleton (state, action))
119         have hstateSet : MeasurableSet
120             {trace : StepTrace Action State (remaining + 1) |
121               StepTrace.stateAt initial trace tailCoordinate.succ = state} :=
122           (measurable_of_finite _) (measurableSet_singleton state)
123         rw [ProbabilityTheory.Kernel.map_apply' _
124           (StepTrace.measurable_cons remaining) _ hvisitSet]
125         rw [ProbabilityTheory.Kernel.map_apply' _
126           (StepTrace.measurable_cons remaining) _ hstateSet]
127         have hvisitPreimage := hvisitSet.preimage
128           (StepTrace.measurable_cons remaining)
129         have hstatePreimage := hstateSet.preimage
130           (StepTrace.measurable_cons remaining)
131         rw [ProbabilityTheory.Kernel.compProd_apply hvisitPreimage]
132         rw [ProbabilityTheory.Kernel.compProd_apply hstatePreimage]
133         simp only [Set.preimage_setOf_eq, StepTrace.stateActionAt_cons_succ,
134           StepTrace.stateAt_succ, ProbabilityTheory.Kernel.comap_apply]
135         simp_rw [ih (by omega) _ tailCoordinate]
136         rw [MeasureTheory.lintegral_mul_const _ (measurable_of_finite _)]

139 /-- The full trajectory visit event factors into its state event and action mass. -/
140 theorem trajectoryMeasure_visitEvent_eq_stateEvent_mul_action
141     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
142     (initialState : Measure State) [IsProbabilityMeasure initialState]
143     (stage : Fin mdp.horizon) (state : State) (action : Action) :
144     policy.trajectoryMeasure initialState
145         {trajectory |
146           (mdp.episodeStepOfTrajectory trajectory stage).state = state /\
147             (mdp.episodeStepOfTrajectory trajectory stage).action = action} =
148       policy.trajectoryMeasure initialState
149           {trajectory | mdp.trajectoryStateAt trajectory stage = state} *
150         policy.actionKernel stage state {action} := by
151   have hvisitEvent :
152       {trajectory : State × StepTrace Action State mdp.horizon |
153         (mdp.episodeStepOfTrajectory trajectory stage).state = state /\
154           (mdp.episodeStepOfTrajectory trajectory stage).action = action} =
155         {trajectory | StepTrace.stateActionAt trajectory.1 trajectory.2 stage =
156           (state, action)} := by
157     ext trajectory
158     simp [MDP.episodeStepOfTrajectory, StepTrace.stateActionAt,
159       mdp.stepTrace_stateAt_eq_trajectoryStateAt, Prod.ext_iff]
160   have hstateEvent :
161       {trajectory : State × StepTrace Action State mdp.horizon |
162         mdp.trajectoryStateAt trajectory stage = state} =
163         {trajectory | StepTrace.stateAt trajectory.1 trajectory.2 stage = state} := by
164     ext trajectory
165     simp [mdp.stepTrace_stateAt_eq_trajectoryStateAt]
166   rw [hvisitEvent, hstateEvent]
167   have hvisitSet : MeasurableSet
168       {trajectory : State × StepTrace Action State mdp.horizon |
169         StepTrace.stateActionAt trajectory.1 trajectory.2 stage =
170           (state, action)} :=
171     (measurable_of_finite _) (measurableSet_singleton (state, action))
172   have hstateSet : MeasurableSet
173       {trajectory : State × StepTrace Action State mdp.horizon |
174         StepTrace.stateAt trajectory.1 trajectory.2 stage = state} :=
175     (measurable_of_finite _) (measurableSet_singleton state)
176   unfold trajectoryMeasure
177   rw [Measure.compProd_apply hvisitSet]
178   rw [Measure.compProd_apply hstateSet]
179   simp only [Set.preimage_setOf_eq]
180   simp_rw [policy.trajectoryKernelRemaining_visitEvent_eq_stateEvent_mul_action
181     mdp.horizon le_rfl _ state action stage]
182   simp only [mdp.stageOfRemainingCoordinate_full]
183   rw [MeasureTheory.lintegral_mul_const _ (measurable_of_finite _)]

185 /-- Genuine state probability at one chronological stage of a generated trajectory. -/
186 noncomputable def stageStateProbability
187     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
188     (initialState : Measure State) [IsProbabilityMeasure initialState]
189     (stage : Fin mdp.horizon) (state : State) : Real :=
190   (policy.trajectoryMeasure initialState).real
191     {trajectory | mdp.trajectoryStateAt trajectory stage = state}

193 /-- A generated state/action visit probability is state mass times action mass. -/
194 theorem stageVisitProbability_eq_stageStateProbability_mul_action
195     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
196     (initialState : Measure State) [IsProbabilityMeasure initialState]
197     (stage : Fin mdp.horizon) (state : State) (action : Action) :
198     policy.stageVisitProbability initialState stage state action =
199       policy.stageStateProbability initialState stage state *
200         (policy.actionKernel stage state {action}).toReal := by
201   rw [policy.stageVisitProbability_eq_measureReal initialState stage state action]
202   rw [stageStateProbability]
203   unfold Measure.real
204   rw [policy.trajectoryMeasure_visitEvent_eq_stateEvent_mul_action
205     initialState stage state action]
206   exact ENNReal.toReal_mul

208 end MarkovPolicy
