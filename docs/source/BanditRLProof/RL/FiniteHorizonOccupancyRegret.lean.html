4 /-!
5 # Finite-horizon occupancy regret identity
6 
7 This module connects finite-horizon Bellman optimality to expected trajectory
8 regret.  It constructs chronological state occupancies, records the expected
9 one-step Bellman optimality gap under a policy, and recursively sums those gaps
10 along the policy-induced state laws.  The resulting occupancy functional is
11 exactly the difference between optimal and policy value, hence exactly the
12 expected trajectory regret.  The measurable greedy policy has zero regret.
13 -/

30 /-- State law at a chronological stage under a Markov policy. -/
31 noncomputable def stateOccupancy
32     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
33     (initialState : Measure State) :
34     (stage : Nat) -> stage <= mdp.horizon -> Measure State
35   | 0, _ => initialState
36   | stage + 1, hstage =>
37       policy.inducedStateKernel ⟨stage, by omega⟩ ∘ₘ
38         policy.stateOccupancy initialState stage (by omega)

40 /-- Every chronological state occupancy is a probability measure. -/
41 instance instStateOccupancyIsProbabilityMeasure
42     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
43     (initialState : Measure State) [IsProbabilityMeasure initialState]
44     (stage : Nat) (hstage : stage <= mdp.horizon) :
45     IsProbabilityMeasure (policy.stateOccupancy initialState stage hstage) := by
46   induction stage with
47   | zero =>
48       rw [stateOccupancy]
49       infer_instance
50   | succ stage ih =>
51       rw [stateOccupancy]
52       letI : IsProbabilityMeasure
53           (policy.stateOccupancy initialState stage (by omega)) :=
54         ih (by omega)
55       infer_instance

57 /--
58 Expected one-step optimality gap after averaging the optimal action-value gap
59 over the policy action kernel at a chronological stage.
60 -/
61 noncomputable def policyBellmanGap
62     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
63     (stage : Fin mdp.horizon) (state : State) : Real :=
64   mdp.optimalValueAt stage (Nat.le_of_lt stage.isLt) state -
65     policy.bellman stage
66       (mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt)) state
67 
68 /-- The policy-averaged one-step Bellman optimality gap is measurable. -/
69 theorem measurable_policyBellmanGap
70     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
71     (stage : Fin mdp.horizon) :
72     Measurable (policy.policyBellmanGap stage) := by
73   exact (mdp.measurable_optimalValueAt stage (Nat.le_of_lt stage.isLt)).sub
74     (policy.measurable_bellman stage
75       (mdp.measurable_optimalValueAt (stage + 1)
76         (Nat.succ_le_of_lt stage.isLt)))
77 
78 /-- Every policy-averaged one-step Bellman optimality gap is nonnegative. -/
79 theorem policyBellmanGap_nonneg
80     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
81     (stage : Fin mdp.horizon) (state : State) :
82     0 <= policy.policyBellmanGap stage state := by
83   unfold policyBellmanGap
84   rw [mdp.optimalValueAt_bellman stage stage.isLt]
85   exact sub_nonneg.mpr
86     (policy.bellman_le_optimalBellman stage
87       (mdp.optimalValueAt (stage + 1)
88         (Nat.succ_le_of_lt stage.isLt)) state)

91 /-- Reindex a chronological policy Bellman gap by the number of decisions remaining. -/
92 theorem policyBellmanGap_stageOfRemaining
93     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
94     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
95     (state : State) :
96     policy.policyBellmanGap
97         ⟨mdp.horizon - (remaining + 1), by omega⟩ state =
98       mdp.optimalValueRemaining (remaining + 1) hremaining state -
99         policy.bellman ⟨mdp.horizon - (remaining + 1), by omega⟩
100           (mdp.optimalValueRemaining remaining (by omega)) state := by
101   unfold policyBellmanGap MDP.optimalValueAt
102   have hcurrent := mdp.optimalValueRemaining_eq_of_eq
103     (Nat.sub_le mdp.horizon (mdp.horizon - (remaining + 1))) hremaining
104     (by omega : mdp.horizon - (mdp.horizon - (remaining + 1)) = remaining + 1)
105   have htail := mdp.optimalValueRemaining_eq_of_eq
106     (Nat.sub_le mdp.horizon (mdp.horizon - (remaining + 1) + 1)) (by omega)
107     (by omega : mdp.horizon - (mdp.horizon - (remaining + 1) + 1) = remaining)
108   rw [hcurrent, htail]

111 /-- Integrating a continuation value against the induced state kernel averages its transition
112 expectation over the policy action kernel. -/
113 theorem integral_inducedStateKernel_eq_integral_transitionValue
114     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
115     (stage : Fin mdp.horizon) (value : State -> Real)
116     (state : State) :
117     (∫ nextState, value nextState ∂policy.inducedStateKernel stage state) =
118       ∫ action, mdp.transitionValue value state action
119         ∂policy.actionKernel stage state := by
120   have hvalue : Measurable value := measurable_of_finite _
121   have hpair : Integrable (fun pair : Action × State => value pair.2)
122       ((policy.actionKernel stage).compProd mdp.transition state) := by
123     apply integrable_of_fintype
124     exact hvalue.comp measurable_snd
125   unfold inducedStateKernel
126   rw [ProbabilityTheory.Kernel.map_apply _ measurable_snd]
127   rw [integral_map measurable_snd.aemeasurable hvalue.aestronglyMeasurable]
128   rw [ProbabilityTheory.integral_compProd hpair]
129   rfl

132 /-- Changing only the continuation value in a policy Bellman step is integration against the
133 induced next-state kernel. -/
134 theorem bellman_sub_bellman_eq_integral_inducedStateKernel_sub
135     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
136     (stage : Fin mdp.horizon) (left right : State -> Real)
137     (state : State) :
138     policy.bellman stage left state - policy.bellman stage right state =
139       ∫ nextState, left nextState - right nextState
140         ∂policy.inducedStateKernel stage state := by
141   have hleftState : Integrable left (policy.inducedStateKernel stage state) :=
142     integrable_of_fintype _ _ (measurable_of_finite _)
143   have hrightState : Integrable right (policy.inducedStateKernel stage state) :=
144     integrable_of_fintype _ _ (measurable_of_finite _)
145   rw [integral_sub hleftState hrightState]
146   rw [policy.integral_inducedStateKernel_eq_integral_transitionValue stage left state]
147   rw [policy.integral_inducedStateKernel_eq_integral_transitionValue stage right state]
148   unfold bellman
149   have hleftBellmanQ : Measurable
150       (fun action => mdp.bellmanQ left state action) :=
151     (mdp.measurable_bellmanQ (measurable_of_finite left)).comp
152       (measurable_const.prodMk measurable_id)
153   have hrightBellmanQ : Measurable
154       (fun action => mdp.bellmanQ right state action) :=
155     (mdp.measurable_bellmanQ (measurable_of_finite right)).comp
156       (measurable_const.prodMk measurable_id)
157   have hleftAction : Integrable
158       (fun action => mdp.bellmanQ left state action)
159       (policy.actionKernel stage state) :=
160     integrable_of_fintype _ _ hleftBellmanQ
161   have hrightAction : Integrable
162       (fun action => mdp.bellmanQ right state action)
163       (policy.actionKernel stage state) :=
164     integrable_of_fintype _ _ hrightBellmanQ
165   have hleftTransition : Integrable
166       (fun action => mdp.transitionValue left state action)
167       (policy.actionKernel stage state) :=
168     integrable_of_fintype _ _
169       ((mdp.measurable_transitionValue (measurable_of_finite left)).comp
170         (measurable_const.prodMk measurable_id))
171   have hrightTransition : Integrable
172       (fun action => mdp.transitionValue right state action)
173       (policy.actionKernel stage state) :=
174     integrable_of_fintype _ _
175       ((mdp.measurable_transitionValue (measurable_of_finite right)).comp
176         (measurable_const.prodMk measurable_id))
177   rw [← integral_sub hleftAction hrightAction]
178   rw [← integral_sub hleftTransition hrightTransition]
179   apply integral_congr_ae
180   filter_upwards [] with action
181   simp only [MDP.bellmanQ]
182   ring

185 /-- Push an integrated continuation-value difference through one policy-induced state step. -/
186 theorem integral_sub_comp_inducedStateKernel_eq_integral_bellman_sub
187     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
188     (stage : Fin mdp.horizon) (left right : State -> Real)
189     (mu : Measure State) [IsProbabilityMeasure mu] :
190     (∫ nextState, left nextState - right nextState
191         ∂policy.inducedStateKernel stage ∘ₘ mu) =
192       ∫ state, policy.bellman stage left state -
193         policy.bellman stage right state ∂mu := by
194   have hsub : Measurable (fun state => left state - right state) :=
195     (measurable_of_finite left).sub (measurable_of_finite right)
196   rw [← Measure.snd_compProd mu (policy.inducedStateKernel stage)]
197   rw [Measure.snd]
198   rw [integral_map measurable_snd.aemeasurable hsub.aestronglyMeasurable]
199   have hpair : Integrable
200       (fun pair : State × State => left pair.2 - right pair.2)
201       (mu.compProd (policy.inducedStateKernel stage)) := by
202     apply integrable_of_fintype
203     exact hsub.comp measurable_snd
204   rw [Measure.integral_compProd hpair]
205   apply integral_congr_ae
206   filter_upwards [] with state
207   exact (policy.bellman_sub_bellman_eq_integral_inducedStateKernel_sub
208     stage left right state).symm

210 /--
211 Finite occupancy sum of policy Bellman gaps, indexed by decisions remaining.
212 Each recursive call advances the state law by the policy-induced kernel.
213 -/
214 noncomputable def occupancyGapRemaining
215     {mdp : MDP State Action} (policy : MarkovPolicy mdp) :
216     (remaining : Nat) -> remaining <= mdp.horizon -> Measure State -> Real
217   | 0, _, _ => 0
218   | remaining + 1, hremaining, mu =>
219       let stage : Fin mdp.horizon :=
220         ⟨mdp.horizon - (remaining + 1), by omega⟩
221       (∫ state,
222           mdp.optimalValueRemaining (remaining + 1) hremaining state -
223             policy.bellman stage
224               (mdp.optimalValueRemaining remaining (by omega)) state ∂mu) +
225         policy.occupancyGapRemaining remaining (by omega)
226           (policy.inducedStateKernel stage ∘ₘ mu)

229 /-- The successor recursion is an integral of the chronological policy Bellman gap plus the
230 remaining occupancy gaps under the next-state law. -/
231 theorem occupancyGapRemaining_succ
232     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
233     (remaining : Nat) (hremaining : remaining + 1 <= mdp.horizon)
234     (mu : Measure State) :
235     policy.occupancyGapRemaining (remaining + 1) hremaining mu =
236       (∫ state, policy.policyBellmanGap
237           ⟨mdp.horizon - (remaining + 1), by omega⟩ state ∂mu) +
238         policy.occupancyGapRemaining remaining (by omega)
239           (policy.inducedStateKernel
240             ⟨mdp.horizon - (remaining + 1), by omega⟩ ∘ₘ mu) := by
241   rw [occupancyGapRemaining]
242   congr 1
243   apply integral_congr_ae
244   filter_upwards [] with state
245   exact (policy.policyBellmanGap_stageOfRemaining remaining hremaining state).symm

247 /-- The recursive occupancy-gap sum is exactly the integrated optimal-policy value gap. -/
248 theorem occupancyGapRemaining_eq_integral_optimalValueRemaining_sub_valueRemaining
249     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
250     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
251     (mu : Measure State) [IsProbabilityMeasure mu] :
252     policy.occupancyGapRemaining remaining hremaining mu =
253       ∫ state, mdp.optimalValueRemaining remaining hremaining state -
254         policy.valueRemaining remaining hremaining state ∂mu := by
255   induction remaining generalizing mu with
256   | zero =>
257       simp [occupancyGapRemaining, MDP.optimalValueRemaining, valueRemaining]
258   | succ remaining ih =>
259       let stage : Fin mdp.horizon :=
260         ⟨mdp.horizon - (remaining + 1), by omega⟩
261       let nextMu : Measure State := policy.inducedStateKernel stage ∘ₘ mu
262       letI : IsProbabilityMeasure nextMu := by
263         dsimp [nextMu]
264         infer_instance
265       rw [occupancyGapRemaining]
266       rw [ih (by omega) nextMu]
267       rw [policy.integral_sub_comp_inducedStateKernel_eq_integral_bellman_sub
268         stage
269         (mdp.optimalValueRemaining remaining (by omega))
270         (policy.valueRemaining remaining (by omega)) mu]
271       rw [valueRemaining]
272       have hfirst : Integrable
273           (fun state =>
274             mdp.optimalValueRemaining (remaining + 1) hremaining state -
275               policy.bellman stage
276                 (mdp.optimalValueRemaining remaining (by omega)) state) mu :=
277         integrable_of_fintype _ _ (measurable_of_finite _)
278       have hsecond : Integrable
279           (fun state =>
280             policy.bellman stage
281                 (mdp.optimalValueRemaining remaining (by omega)) state -
282               policy.bellman stage
283                 (policy.valueRemaining remaining (by omega)) state) mu :=
284         integrable_of_fintype _ _ (measurable_of_finite _)
285       rw [← integral_add hfirst hsecond]
286       apply integral_congr_ae
287       filter_upwards [] with state
288       ring

290 /-- Backward policy value is bounded by backward optimal value. -/
291 theorem valueRemaining_le_optimalValueRemaining
292     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
293     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
294     (state : State) :
295     policy.valueRemaining remaining hremaining state <=
296       mdp.optimalValueRemaining remaining hremaining state := by
297   induction remaining generalizing state with
298   | zero =>
299       rfl
300   | succ remaining ih =>
301       rw [valueRemaining, MDP.optimalValueRemaining]
302       exact
303         (policy.bellman_mono
304           (stage := ⟨mdp.horizon - (remaining + 1), by omega⟩)
305           (fun nextState => ih (by omega) nextState) state).trans
306         (policy.bellman_le_optimalBellman
307           ⟨mdp.horizon - (remaining + 1), by omega⟩
308           (mdp.optimalValueRemaining remaining (by omega)) state)

310 /-- The finite occupancy Bellman-gap sum is nonnegative. -/
311 theorem occupancyGapRemaining_nonneg
312     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
313     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
314     (mu : Measure State) [IsProbabilityMeasure mu] :
315     0 <= policy.occupancyGapRemaining remaining hremaining mu := by
316   rw [policy.occupancyGapRemaining_eq_integral_optimalValueRemaining_sub_valueRemaining]
317   apply integral_nonneg
318   intro state
319   exact sub_nonneg.mpr

322 /-- Expected finite-horizon regret: optimal initial value minus generated trajectory reward. -/
323 noncomputable def expectedRegret
324     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
325     (initialState : Measure State) : Real :=
326   (∫ state, mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state ∂initialState) -
327     ∫ trajectory, mdp.cumulativeReward trajectory
328       ∂policy.trajectoryMeasure initialState

330 /-- Expected trajectory regret is the initial-law integral of the optimal-policy value gap. -/
331 theorem expectedRegret_eq_integral_optimalValueAt_sub_valueAt
332     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
333     (initialState : Measure State) [IsProbabilityMeasure initialState] :
334     policy.expectedRegret initialState =
335       ∫ state,
336         mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon) state -
337           policy.valueAt 0 (Nat.zero_le mdp.horizon) state ∂initialState := by
338   unfold expectedRegret
339   rw [policy.integral_cumulativeReward_trajectoryMeasure_eq_integral_valueAt_zero]
340   have hoptimal : Integrable
341       (mdp.optimalValueAt 0 (Nat.zero_le mdp.horizon)) initialState :=
342     integrable_of_fintype _ _
343       (mdp.measurable_optimalValueAt 0 (Nat.zero_le mdp.horizon))
344   have hpolicy : Integrable
345       (policy.valueAt 0 (Nat.zero_le mdp.horizon)) initialState :=
346     integrable_of_fintype _ _
347       (policy.measurable_valueAt 0 (Nat.zero_le mdp.horizon))
348   exact (integral_sub hoptimal hpolicy).symm

350 /-- Expected trajectory regret equals the full recursively accumulated occupancy Bellman gap. -/
351 theorem expectedRegret_eq_occupancyGapRemaining
352     {mdp : MDP State Action} (policy : MarkovPolicy mdp)
353     (initialState : Measure State) [IsProbabilityMeasure initialState] :
354     policy.expectedRegret initialState =
355       policy.occupancyGapRemaining mdp.horizon le_rfl initialState := by
356   rw [policy.expectedRegret_eq_integral_optimalValueAt_sub_valueAt]
357   rw [policy.occupancyGapRemaining_eq_integral_optimalValueRemaining_sub_valueRemaining]
