21 /-- Integrating the normalized optimal-tail probe against two probability
22 measures and subtracting transports exactly back to the `V*` difference. -/
23 theorem integral_optimalValue_sub_eq_two_mul_horizon_mul_integral_probe_sub
24     (mdp : MDP State Action) (summary : TransitionCountSummary mdp)
25     (defaultState : State) (stage : Fin mdp.horizon)
26     (state : State) (action : Action) :
27     (∫ nextState,
28         mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt) nextState
29           ∂summary.aggregateEmpiricalTransitionKernel defaultState (state, action)) -
30       mdp.transitionValue
31         (mdp.optimalValueAt (stage + 1) (Nat.succ_le_of_lt stage.isLt))
32         state action =
33       2 * (mdp.horizon : Real) *
34         ((∫ nextState, mdp.optimalTailProbe stage nextState
35             ∂summary.aggregateEmpiricalTransitionKernel defaultState (state, action)) -
36           mdp.transitionValue (mdp.optimalTailProbe stage) state action) := by
37   let empirical := summary.aggregateEmpiricalTransitionKernel defaultState
38       (state, action)
39   let truth := mdp.transition (state, action)
40   let value := mdp.optimalValueAt (stage + 1)
41       (Nat.succ_le_of_lt stage.isLt)
42   let probe := mdp.optimalTailProbe stage
43   let scale : Real := 2 * (mdp.horizon : Real)
44   have hhorizon : (mdp.horizon : Real) ≠ 0 := by
45     exact_mod_cast Nat.ne_of_gt (Nat.pos_of_ne_zero (by
46       intro hzero
47       have : stage.val < 0 := by simpa [hzero] using stage.isLt
48       omega))
49   have hpoint : ∀ nextState, value nextState =
50       scale * probe nextState - (mdp.horizon : Real) := by
51     intro nextState
52     simp only [value, scale, probe, MDP.optimalTailProbe]
53     field_simp
54     ring
55   letI : IsProbabilityMeasure empirical :=
56     (summary.aggregateEmpiricalTransitionKernel_isMarkov defaultState).isProbabilityMeasure
57       (state, action)
58   letI : IsProbabilityMeasure truth := mdp.transition_isMarkov.isProbabilityMeasure
59       (state, action)
60   have hprobeMeasurable : Measurable probe := measurable_of_finite _
61   have hprobeEmpirical : Integrable probe empirical :=
62     integrable_of_fintype _ _ hprobeMeasurable
63   have hprobeTruth : Integrable probe truth :=
64     integrable_of_fintype _ _ hprobeMeasurable
65   have hempirical :
66       (∫ nextState, value nextState ∂empirical) =
67         scale * (∫ nextState, probe nextState ∂empirical) -
68           (mdp.horizon : Real) := by
69     rw [integral_congr_ae (Filter.Eventually.of_forall hpoint)]
70     rw [integral_sub]
71     · rw [integral_const, measureReal_univ_eq_one, one_smul]
72       rw [integral_const_mul scale probe]
73     · exact hprobeEmpirical.const_mul _
74     · exact integrable_const _
75   have htruth :
76       (∫ nextState, value nextState ∂truth) =
77         scale * (∫ nextState, probe nextState ∂truth) -
78           (mdp.horizon : Real) := by
79     rw [integral_congr_ae (Filter.Eventually.of_forall hpoint)]
80     rw [integral_sub]
81     · rw [integral_const, measureReal_univ_eq_one, one_smul]
82       rw [integral_const_mul scale probe]
83     · exact hprobeTruth.const_mul _
84     · exact integrable_const _
85   change (∫ nextState, value nextState ∂empirical) -
86       (∫ nextState, value nextState ∂truth) =
87     scale * ((∫ nextState, probe nextState ∂empirical) -
88       (∫ nextState, probe nextState ∂truth))
89   rw [hempirical, htruth]
90   ring

94 /-- Outside the proved joint event, the empirical transition kernel consumed
95 by the planner has the sharp `V*` projection error at every peeled positive
96 actual count.  No confidence statement is supplied by the caller. -/
97 theorem abs_empiricalTransition_optimalValue_sub_lt
98     {mdp : MDP State Action} {initialState : Measure State}
99     [IsProbabilityMeasure initialState]
100     (source : AdaptiveEpisodeBatchSource mdp initialState 1)
101     {episodes : Nat} {logBudget : Real}
102     {trajectory : EpisodeBatchTrajectory mdp 1}
103     (htrajectory : trajectory ∉
104       simultaneousTransitionFailureEvent source episodes logBudget)
105     (defaultState : State) (index : OptimalTailIndex mdp episodes)
106     (hactual : adaptiveCumulativeAggregateVisitCountAt trajectory index.round
107         index.state index.action = index.count + 1) :
108     |(∫ nextState,
109           mdp.optimalValueAt (index.stage + 1)
110               (Nat.succ_le_of_lt index.stage.isLt) nextState
111             ∂TransitionCountSummary.aggregateEmpiricalTransitionKernel
112               (adaptiveCumulativeEmpiricalModelStateAt trajectory index.round).1
113               defaultState (index.state, index.action)) -
114         mdp.transitionValue
115           (mdp.optimalValueAt (index.stage + 1)
116             (Nat.succ_le_of_lt index.stage.isLt))
117           index.state index.action| <
118       2 * (mdp.horizon : Real) * logBudget /
119         Real.sqrt (index.count + 1 : Nat) := by
120   let summary := (adaptiveCumulativeEmpiricalModelStateAt trajectory index.round).1
121   let count : Real := (index.count + 1 : Nat)
122   let probeDifference : Real :=
123     (∫ nextState, mdp.optimalTailProbe index.stage nextState
124         ∂summary.aggregateEmpiricalTransitionKernel defaultState
125           (index.state, index.action)) -
126       mdp.transitionValue (mdp.optimalTailProbe index.stage)
127         index.state index.action
128   have hcountPosNat : 0 < adaptiveCumulativeAggregateVisitCountAt trajectory
129       index.round index.state index.action := by
130     rw [hactual]
131     exact Nat.succ_pos _
132   have hvisitSum :
133       (∑ i ∈ Finset.range (index.round + 1),
134           source.aggregateVisitIncrement index.state index.action i trajectory) =
135         (index.count + 1 : Nat) := by
136     rw [source.sum_aggregateVisitIncrement_eq_prefixAggregateVisitCount]
137     exact_mod_cast hactual
138   have hresidual :=
139     abs_optimalTailResidual_lt_of_not_mem_simultaneousTransitionFailureEvent
140       source htrajectory index hvisitSum
141   have halignment :=
142     source.aggregateTransitionFunctionalResidual_eq_count_mul_transitionValue_sub
143       trajectory index.round defaultState (mdp.optimalTailProbe index.stage)
144       index.state index.action hcountPosNat
145   have hresidual' : |count * probeDifference| <
146       logBudget * Real.sqrt count := by
147     rw [halignment, hactual] at hresidual
148     simpa [count, probeDifference, summary] using hresidual
149   have hcountPos : 0 < count := by
150     change 0 < ((index.count + 1 : Nat) : Real)
151     positivity
152   have hsqrtPos : 0 < Real.sqrt count := Real.sqrt_pos.2 hcountPos
153   have hsqrtSq : Real.sqrt count ^ 2 = count := Real.sq_sqrt hcountPos.le
154   have hsqrtMul : Real.sqrt count * Real.sqrt count = count := by
155     simpa [pow_two] using hsqrtSq
156   have hprobe : |probeDifference| < logBudget / Real.sqrt count := by
157     rw [abs_mul, abs_of_pos hcountPos] at hresidual'
158     apply (lt_div_iff₀ hsqrtPos).2
159     refine lt_of_mul_lt_mul_right (a := Real.sqrt count) ?_ hsqrtPos.le
160     calc
161       |probeDifference| * Real.sqrt count * Real.sqrt count =
162           |probeDifference| * count := by rw [mul_assoc, hsqrtMul]
163       _ = count * |probeDifference| := by ring
164       _ < logBudget * Real.sqrt count := hresidual'
165   have haffine :=
166     integral_optimalValue_sub_eq_two_mul_horizon_mul_integral_probe_sub
167       mdp summary defaultState index.stage index.state index.action
168   have hhorizon : 0 < (mdp.horizon : Real) := by
169     exact_mod_cast Nat.pos_of_ne_zero (by
170       intro hzero
171       have : index.stage.val < 0 := by simpa [hzero] using index.stage.isLt
172       omega)
173   rw [haffine, abs_mul, abs_of_pos (mul_pos (by norm_num) hhorizon)]
174   simpa [count, probeDifference, summary, mul_div_assoc] using
175     (mul_lt_mul_of_pos_left hprobe (mul_pos (by norm_num) hhorizon))
