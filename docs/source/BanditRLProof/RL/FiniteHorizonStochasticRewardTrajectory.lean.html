4 /-!
5 # Finite-horizon stochastic reward trajectory value identity
6 
7 This module generates finite policy trajectories whose coordinates retain the
8 sampled action, sampled Real reward, and next state.  Selected rewards need
9 only the `L1` mean-compatibility contract from the stochastic Bellman layer.
10 The cumulative sampled reward is proved integrable recursively and its
11 expectation is identified with both stochastic and mean policy evaluation.
12 -/

54 /-- An a.e. strongly measurable Real function on a finite type is integrable. -/
55 theorem integrable_of_fintype_aestronglyMeasurable
56     {Omega : Type*} [Fintype Omega] [MeasurableSpace Omega]
57     (mu : Measure Omega) [IsFiniteMeasure mu]
58     (f : Omega -> Real) (hf : AEStronglyMeasurable f mu) : Integrable f mu := by
59   apply Integrable.of_bound hf
60     (Finset.sum Finset.univ (fun x => |f x|))
61   exact Filter.Eventually.of_forall fun x => by
62     rw [Real.norm_eq_abs]
63     exact Finset.single_le_sum
64       (fun y _ => abs_nonneg (f y)) (Finset.mem_univ x)

88 /--
89 Kernel of the next `remaining` sampled action-reward-state coordinates.
90 The first chronological stage is `mdp.horizon - remaining`.
91 -/
92 noncomputable def stochasticTrajectoryKernelRemaining
93     (source : MeanCompatibleRewardKernel mdp)
94     (policy : MarkovPolicy mdp) :
95     (remaining : Nat) -> remaining <= mdp.horizon ->
96       ProbabilityTheory.Kernel State
97         (RewardStepTrace Action State remaining)
98   | 0, _ =>
99       ProbabilityTheory.Kernel.deterministic
100         (fun _ => fun i => Fin.elim0 i) measurable_const
101   | remaining + 1, hremaining =>
102       let stage : Fin mdp.horizon :=
103         ⟨mdp.horizon - (remaining + 1), by omega⟩
104       let tailKernel : ProbabilityTheory.Kernel
105           (Prod State (Prod Action (Prod Real State)))
106           (RewardStepTrace Action State remaining) :=
107         (source.stochasticTrajectoryKernelRemaining policy remaining (by omega)).comap
108           (fun p : Prod State (Prod Action (Prod Real State)) => p.2.2.2)
109           (show Measurable
110               (fun p : Prod State (Prod Action (Prod Real State)) => p.2.2.2) from
111             measurable_snd.snd.snd)
112       ((source.actionRewardStateKernel policy stage).compProd tailKernel).map
113         (fun p =>
114           @Fin.cons remaining (fun _ => Prod Action (Prod Real State)) p.1 p.2)

133 /-- Sum of the actual sampled rewards in a reward-bearing finite trace. -/
134 def sampledCumulativeRewardFrom :
135     (remaining : Nat) -> RewardStepTrace Action State remaining -> Real
136   | 0, _ => 0
137   | remaining + 1, trace =>
138       (trace 0).2.1 + sampledCumulativeRewardFrom remaining (Fin.tail trace)

161 /--
162 The sampled cumulative reward is `L1` under every statewise stochastic
163 trajectory law.  No boundedness or second-moment hypothesis is used.
164 -/
165 theorem integrable_sampledCumulativeRewardFrom_stochasticTrajectoryKernelRemaining
166     (source : MeanCompatibleRewardKernel mdp)
167     (policy : MarkovPolicy mdp)
168     (remaining : Nat) (hremaining : remaining <= mdp.horizon)
169     (state : State) :
170     Integrable
171       (MDP.sampledCumulativeRewardFrom (Action := Action) (State := State) remaining)
172       (source.stochasticTrajectoryKernelRemaining policy remaining hremaining state) := by
173   induction remaining generalizing state with
174   | zero =>
175       rw [stochasticTrajectoryKernelRemaining,
176         ProbabilityTheory.Kernel.deterministic_apply]
177       exact integrable_dirac'
178         (MDP.measurable_sampledCumulativeRewardFrom
179           (Action := Action) (State := State) 0).stronglyMeasurable
180         (by simp)
181   | succ remaining ih =>
182       let stage : Fin mdp.horizon :=
183         ⟨mdp.horizon - (remaining + 1), by omega⟩
184       let tailKernel : ProbabilityTheory.Kernel
185           (Prod State (Prod Action (Prod Real State)))
186           (RewardStepTrace Action State remaining) :=
187         (source.stochasticTrajectoryKernelRemaining policy remaining (by omega)).comap
188           (fun p : Prod State (Prod Action (Prod Real State)) => p.2.2.2)
189           (show Measurable
190               (fun p : Prod State (Prod Action (Prod Real State)) => p.2.2.2) from
191             measurable_snd.snd.snd)
192       let consStep :=
193         fun p : Prod (Prod Action (Prod Real State))
194             (RewardStepTrace Action State remaining) =>
195           @Fin.cons remaining (fun _ => Prod Action (Prod Real State)) p.1 p.2
196       let sampledTail :=
197         MDP.sampledCumulativeRewardFrom
198           (Action := Action) (State := State) remaining
199       let tailNorm : State -> Real := fun nextState =>
200         integral
201           (source.stochasticTrajectoryKernelRemaining policy remaining
202             (by omega) nextState)
203           (fun tail => ‖sampledTail tail‖)
204       have hcons : Measurable consStep := RewardStepTrace.measurable_cons remaining
205       have hreturn : Measurable
206           (MDP.sampledCumulativeRewardFrom
207             (Action := Action) (State := State) (remaining + 1)) :=
208         MDP.measurable_sampledCumulativeRewardFrom (remaining + 1)
209       have htailNorm : Measurable tailNorm := by
210         exact
211           (MDP.measurable_sampledCumulativeRewardFrom
212             (Action := Action) (State := State) remaining).norm.stronglyMeasurable
213             |>.integral_kernel.measurable
214       have hheadBound : Integrable
215           (fun head : Prod Action (Prod Real State) =>
216             |head.2.1| + tailNorm head.2.2)
217           (source.actionRewardStateKernel policy stage state) := by
218         let bound := fun head : Prod Action (Prod Real State) =>
219           |head.2.1| + tailNorm head.2.2
220         have hbound : StronglyMeasurable bound :=
221           (measurable_snd.fst.abs.add (htailNorm.comp measurable_snd.snd)).stronglyMeasurable
222         unfold actionRewardStateKernel
223         apply (ProbabilityTheory.integrable_compProd_iff
224           hbound.aestronglyMeasurable).2
225         constructor
226         · exact Filter.Eventually.of_forall fun action => by
227             haveI : IsProbabilityMeasure
228                 (source.rewardKernel.kernel (state, action)) :=
229               RewardKernel.isProbabilityMeasure_apply source.rewardKernel (state, action)
230             haveI : IsProbabilityMeasure (mdp.transition (state, action)) := by
231               infer_instance
232             rw [rewardNextStateKernel, ProbabilityTheory.Kernel.prod_apply]
233             have hrewardAbs : Integrable (fun reward : Real => |reward|)
234                 (source.rewardKernel.kernel (state, action)) := by
235               simpa [Real.norm_eq_abs] using
236                 (source.integrable_reward state action).norm
237             exact
238               (hrewardAbs.comp_fst (mdp.transition (state, action))).add
239                 ((integrable_of_fintype (mdp.transition (state, action))
240                   tailNorm htailNorm).comp_snd
241                     (source.rewardKernel.kernel (state, action)))
242         · have hboundAE : AEStronglyMeasurable bound
243               (((policy.actionKernel stage).compProd
244                 source.rewardNextStateKernel) state) :=
245             hbound.aestronglyMeasurable
246           exact integrable_of_fintype_aestronglyMeasurable
247             (policy.actionKernel stage state)
248             (fun action =>
249               integral (source.rewardNextStateKernel (state, action))
250                 (fun rewardState => ‖bound (action, rewardState)‖))
251             hboundAE.norm.integral_kernel_compProd
252       rw [stochasticTrajectoryKernelRemaining]
253       rw [ProbabilityTheory.Kernel.map_apply _ hcons]
254       apply (integrable_map_measure hreturn.aestronglyMeasurable
255         hcons.aemeasurable).2
256       have hpreMeas : AEStronglyMeasurable
257           (fun p =>
258             MDP.sampledCumulativeRewardFrom
259               (Action := Action) (State := State) (remaining + 1) (consStep p))
260           (((source.actionRewardStateKernel policy stage).compProd tailKernel) state) :=
261         (hreturn.comp hcons).aestronglyMeasurable
262       apply (ProbabilityTheory.integrable_compProd_iff hpreMeas).2
263       constructor
264       · exact Filter.Eventually.of_forall fun head => by
265           rw [ProbabilityTheory.Kernel.comap_apply]
266           change Integrable
267             (fun tail => head.2.1 + sampledTail tail)
268             (source.stochasticTrajectoryKernelRemaining policy remaining
269               (by omega) head.2.2)
270           exact (integrable_const _).add (ih (by omega) head.2.2)
271       · have hinnerNormAE : AEStronglyMeasurable
272             (fun head =>
273               integral (tailKernel (state, head))
274                 (fun tail =>
275                   ‖MDP.sampledCumulativeRewardFrom
276                     (Action := Action) (State := State) (remaining + 1)
277                       (consStep (head, tail))‖))
278             (source.actionRewardStateKernel policy stage state) :=
279           hpreMeas.norm.integral_kernel_compProd
280         refine hheadBound.mono' hinnerNormAE ?_
281         exact Filter.Eventually.of_forall fun head => by
282           rw [Real.norm_eq_abs,
283             abs_of_nonneg (integral_nonneg fun _ => norm_nonneg _)]
284           rw [ProbabilityTheory.Kernel.comap_apply]
285           change
286             integral
287                 (source.stochasticTrajectoryKernelRemaining policy remaining
288                   (by omega) head.2.2)
289                 (fun tail => ‖head.2.1 + sampledTail tail‖) <=
290               |head.2.1| + tailNorm head.2.2
291           calc
292             _ <= integral
293                 (source.stochasticTrajectoryKernelRemaining policy remaining
294                   (by omega) head.2.2)
295                 (fun tail => |head.2.1| + ‖sampledTail tail‖) := by
296               apply integral_mono_ae
297               · exact ((integrable_const _).add
298                   (ih (by omega) head.2.2)).norm
299               · exact (integrable_const _).add (ih (by omega) head.2.2).norm
300               · exact Filter.Eventually.of_forall fun tail => by
301                   simpa [Real.norm_eq_abs] using
302                     norm_add_le head.2.1 (sampledTail tail)
303             _ = |head.2.1| + tailNorm head.2.2 := by
304               rw [integral_add (integrable_const _)
305                 (ih (by omega) head.2.2).norm]
306               simp [tailNorm, sampledTail]
