4 /-!
5 # Finite-window uniform OFUL confidence
6 
7 This module assembles the compiled deterministic-horizon scalar-ridge
8 confidence theorem over every horizon in a finite inclusive window.  It
9 supports arbitrary confidence schedules with values in `(0, 1]` and an equal
10 allocation whose budgets sum exactly to `delta`, giving failure probability
11 at most `delta`.
12 -/

22 /-- Failure of the scalar-ridge confidence ellipsoid at one horizon. -/
23 def scalarRidgeConfidenceFailureAt
24     {Omega : Type u} {Feature : Type v}
25     [Fintype Feature] [DecidableEq Feature]
26     (lambda : Real)
27     (thetaStar : Feature -> Real)
28     (S : Real)
29     (feature : Nat -> Omega -> Feature -> Real)
30     (response : Nat -> Omega -> Real)
31     (R delta : Real)
32     (n : Nat) : Set Omega :=
33   {omega |
34     matrixNorm
35         (Matrix.scalar Feature lambda +
36           finiteHorizonFeatureGram feature n omega)
37         (finiteHorizonRidgeEstimate
38             (Matrix.scalar Feature lambda) feature response n omega -
39           thetaStar) >
40       finiteHorizonScalarConfidenceRadius
41         feature R delta lambda S n omega}
42 
43 /--
44 Union of scalar-ridge confidence failures over all `n <= horizon`, with the
45 confidence level selected by `deltaAt n`.
46 -/
47 def finiteHorizonScheduledScalarRidgeConfidenceFailureSet
48     {Omega : Type u} {Feature : Type v}
49     [Fintype Feature] [DecidableEq Feature]
50     (lambda : Real)
51     (thetaStar : Feature -> Real)
52     (S : Real)
53     (feature : Nat -> Omega -> Feature -> Real)
54     (response : Nat -> Omega -> Real)
55     (R : Real)
56     (deltaAt : Nat -> Real)
57     (horizon : Nat) : Set Omega :=
58   ⋃ n ∈ Finset.range (horizon + 1),
59     scalarRidgeConfidenceFailureAt
60       lambda thetaStar S feature response R (deltaAt n) n

62 /-- Membership in the scheduled failure set is failure at some `n <= horizon`. -/
63 theorem mem_finiteHorizonScheduledScalarRidgeConfidenceFailureSet_iff
64     {Omega : Type u} {Feature : Type v}
65     [Fintype Feature] [DecidableEq Feature]
66     (lambda : Real)
67     (thetaStar : Feature -> Real)
68     (S : Real)
69     (feature : Nat -> Omega -> Feature -> Real)
70     (response : Nat -> Omega -> Real)
71     (R : Real)
72     (deltaAt : Nat -> Real)
73     (horizon : Nat) (omega : Omega) :
74     omega ∈ finiteHorizonScheduledScalarRidgeConfidenceFailureSet
75         lambda thetaStar S feature response R deltaAt horizon ↔
76       ∃ n, n <= horizon ∧
77         omega ∈ scalarRidgeConfidenceFailureAt
78           lambda thetaStar S feature response R (deltaAt n) n := by
79   simp only [finiteHorizonScheduledScalarRidgeConfidenceFailureSet,
80     Set.mem_iUnion, exists_prop, Finset.mem_range, Nat.lt_succ_iff]

112 /--
113 Scheduled finite-window confidence: the failure-union probability is bounded
114 by the sum of the fixed-time confidence budgets.
115 -/
116 theorem measure_finiteHorizonScheduledScalarRidgeConfidenceFailureSet_le_sum
117     {Omega : Type u} {Feature : Type v}
118     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
119     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
120     (mu : Measure Omega) [IsProbabilityMeasure mu]
121     (lambda : Real) (hlambda : 0 < lambda)
122     (thetaStar : Feature -> Real) (S : Real)
123     (htheta : euclideanLength thetaStar <= S)
124     (F : Filtration Nat mOmega)
125     (feature : Nat -> Omega -> Feature -> Real)
126     (response noise : Nat -> Omega -> Real)
127     (R : Real) (hR : 0 < R)
128     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
129     (hfeature : forall i j,
130       StronglyMeasurable[F i] (fun omega => feature i omega j))
131     (hnoise : StronglyAdapted F (fun t omega =>
132       match t with
133       | 0 => 0
134       | i + 1 => noise i omega))
135     (hprojectionBound_nonneg : forall theta i,
136       0 <= projectionBound theta i)
137     (hprojectionBound : forall theta i omega,
138       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
139         projectionBound theta i)
140     (horizon : Nat)
141     (hsubGaussian : forall i, i < horizon ->
142       HasCondSubgaussianMGF
143         (F i) (F.le i) (noise i)
144         (constantSquaredVarianceProxy R i) mu)
145     (hresponse : forall omega i, i < horizon ->
146       response i omega =
147         dotProduct thetaStar (feature i omega) + noise i omega)
148     (deltaAt : Nat -> Real)
149     (hdeltaAt : forall n, n <= horizon -> 0 < deltaAt n)
150     (hdeltaAt_one : forall n, n <= horizon -> deltaAt n <= 1) :
151     mu (finiteHorizonScheduledScalarRidgeConfidenceFailureSet
152         lambda thetaStar S feature response R deltaAt horizon) <=
153       (Finset.range (horizon + 1)).sum
154         (fun n => ENNReal.ofReal (deltaAt n)) := by
155   unfold finiteHorizonScheduledScalarRidgeConfidenceFailureSet
156   calc
157     mu (⋃ n ∈ Finset.range (horizon + 1),
158         scalarRidgeConfidenceFailureAt
159           lambda thetaStar S feature response R (deltaAt n) n) <=
160         (Finset.range (horizon + 1)).sum (fun n =>
161           mu (scalarRidgeConfidenceFailureAt
162             lambda thetaStar S feature response R (deltaAt n) n)) := by
163       exact ProbabilityUnionBound.measure_biUnion_finset_le
164         mu (Finset.range (horizon + 1)) fun n =>
165           scalarRidgeConfidenceFailureAt
166             lambda thetaStar S feature response R (deltaAt n) n
167     _ <= (Finset.range (horizon + 1)).sum
168           (fun n => ENNReal.ofReal (deltaAt n)) := by
169       apply Finset.sum_le_sum
170       intro n hn
171       have hnle : n <= horizon :=
172         Nat.lt_succ_iff.mp (Finset.mem_range.mp hn)
173       simpa only [scalarRidgeConfidenceFailureAt] using
174         measure_finiteHorizonScalarRidgeEstimate_error_matrixNorm_gt_confidenceRadius_le
175           mu lambda hlambda thetaStar S htheta F feature response noise R hR
176           projectionBound hfeature hnoise hprojectionBound_nonneg
177           hprojectionBound n
178           (fun i hi => hsubGaussian i (lt_of_lt_of_le hi hnle))
179           (deltaAt n) (hdeltaAt n hnle) (hdeltaAt_one n hnle)
180           (fun omega i hi =>
181             hresponse omega i (lt_of_lt_of_le hi hnle))

183 /--
184 Equal-share finite-window failure union.  The inclusive window has
185 `horizon + 1` horizons, so each fixed-time theorem receives
186 `delta / (horizon + 1)`.
187 -/
188 def finiteHorizonUniformScalarRidgeConfidenceFailureSet
189     {Omega : Type u} {Feature : Type v}
190     [Fintype Feature] [DecidableEq Feature]
191     (lambda : Real)
192     (thetaStar : Feature -> Real)
193     (S : Real)
194     (feature : Nat -> Omega -> Feature -> Real)
195     (response : Nat -> Omega -> Real)
196     (R delta : Real)
197     (horizon : Nat) : Set Omega :=
198   finiteHorizonScheduledScalarRidgeConfidenceFailureSet
199     lambda thetaStar S feature response R
200     (fun _ => delta / ((horizon + 1 : Nat) : Real)) horizon

202 /--
203 Uniform scalar-ridge confidence over every deterministic horizon
204 `n <= horizon`, obtained by equal allocation of the total failure budget.
205 -/
206 theorem measure_finiteHorizonUniformScalarRidgeConfidenceFailureSet_le
207     {Omega : Type u} {Feature : Type v}
208     [mOmega : MeasurableSpace Omega] [StandardBorelSpace Omega]
209     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
210     (mu : Measure Omega) [IsProbabilityMeasure mu]
211     (lambda : Real) (hlambda : 0 < lambda)
212     (thetaStar : Feature -> Real) (S : Real)
213     (htheta : euclideanLength thetaStar <= S)
214     (F : Filtration Nat mOmega)
215     (feature : Nat -> Omega -> Feature -> Real)
216     (response noise : Nat -> Omega -> Real)
217     (R : Real) (hR : 0 < R)
218     (projectionBound : EuclideanSpace Real Feature -> Nat -> Real)
219     (hfeature : forall i j,
220       StronglyMeasurable[F i] (fun omega => feature i omega j))
221     (hnoise : StronglyAdapted F (fun t omega =>
222       match t with
223       | 0 => 0
224       | i + 1 => noise i omega))
225     (hprojectionBound_nonneg : forall theta i,
226       0 <= projectionBound theta i)
227     (hprojectionBound : forall theta i omega,
228       |dotProduct (WithLp.ofLp theta) (feature i omega)| <=
229         projectionBound theta i)
230     (horizon : Nat)
231     (hsubGaussian : forall i, i < horizon ->
232       HasCondSubgaussianMGF
233         (F i) (F.le i) (noise i)
234         (constantSquaredVarianceProxy R i) mu)
235     (hresponse : forall omega i, i < horizon ->
236       response i omega =
237         dotProduct thetaStar (feature i omega) + noise i omega)
238     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1) :
239     mu (finiteHorizonUniformScalarRidgeConfidenceFailureSet
240         lambda thetaStar S feature response R delta horizon) <=
241       ENNReal.ofReal delta := by
242   unfold finiteHorizonUniformScalarRidgeConfidenceFailureSet
243     finiteHorizonScheduledScalarRidgeConfidenceFailureSet
244   apply ProbabilityUnionBound.measure_biUnion_finset_le_of_uniform
245     mu (Finset.range (horizon + 1)) (by simp) delta
246   intro n hn
247   have hnle : n <= horizon :=
248     Nat.lt_succ_iff.mp (Finset.mem_range.mp hn)
249   have hden_pos : 0 < (((horizon + 1 : Nat) : Real)) := by positivity
250   have hden_one : (1 : Real) <= ((horizon + 1 : Nat) : Real) := by
251     exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
252   have hlocal_pos :
253       0 < delta / ((horizon + 1 : Nat) : Real) :=
254     div_pos hdelta hden_pos
255   have hlocal_one :
256       delta / ((horizon + 1 : Nat) : Real) <= 1 :=
257     (div_le_one hden_pos).2 (hdelta_one.trans hden_one)
258   simpa only [Finset.card_range, scalarRidgeConfidenceFailureAt] using
259     measure_finiteHorizonScalarRidgeEstimate_error_matrixNorm_gt_confidenceRadius_le
260       mu lambda hlambda thetaStar S htheta F feature response noise R hR
261       projectionBound hfeature hnoise hprojectionBound_nonneg hprojectionBound
262       n (fun i hi => hsubGaussian i (lt_of_lt_of_le hi hnle))
263       (delta / ((horizon + 1 : Nat) : Real)) hlocal_pos hlocal_one
264       (fun omega i hi =>
265         hresponse omega i (lt_of_lt_of_le hi hnle))
