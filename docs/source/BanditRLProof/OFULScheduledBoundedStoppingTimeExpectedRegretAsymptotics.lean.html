19 /--
20 The additional scheduled confidence logarithm at outer budget
21 `1 / (T + 1)` is `O(log (T + 1))`.
22 -/
23 theorem telescopingStandardExpectedConfidenceLog_isBigO_log_succ :
24     (fun horizon : Nat =>
25       2 * Real.log
26         ((((horizon + 1 : Nat) : Real) ^ 2) *
27           ((horizon + 2 : Nat) : Real))) =O[atTop]
28       (fun horizon : Nat =>
29         Real.log (((horizon + 1 : Nat) : Real))) := by
30   rw [Asymptotics.isBigO_iff]
31   refine ⟨8, ?_⟩
32   filter_upwards [eventually_ge_atTop 1] with horizon hhorizon
33   let n : Real := (((horizon + 1 : Nat) : Real))
34   have hn_two : 2 <= n := by
35     dsimp [n]
36     exact_mod_cast Nat.succ_le_succ hhorizon
37   have hn_one : 1 <= n := le_trans (by norm_num) hn_two
38   have hn_pos : 0 < n := lt_of_lt_of_le (by norm_num) hn_two
39   have hnext :
40       (((horizon + 2 : Nat) : Real)) = n + 1 := by
41     dsimp [n]
42     push_cast
43     ring
44   rw [hnext]
45   change
46     ‖2 * Real.log (n ^ 2 * (n + 1))‖ <=
47       8 * ‖Real.log n‖
48   have hnext_le_sq : n + 1 <= n ^ 2 := by
49     nlinarith
50   have hlog_next :
51       Real.log (n + 1) <= 2 * Real.log n := by
52     calc
53       Real.log (n + 1) <= Real.log (n ^ 2) :=
54         Real.log_le_log (by positivity) hnext_le_sq
55       _ = 2 * Real.log n := by
56         exact Real.log_pow n 2
57   have hlog_n_nonneg : 0 <= Real.log n := Real.log_nonneg hn_one
58   have hsquare_one : 1 <= n ^ 2 := by nlinarith
59   have hnext_one : 1 <= n + 1 := by linarith
60   have hproduct_one : 1 <= n ^ 2 * (n + 1) := by
61     simpa only [one_mul] using
62       mul_le_mul hsquare_one hnext_one zero_le_one
63         (le_trans zero_le_one hsquare_one)
64   have hlog_product_nonneg :
65       0 <= Real.log (n ^ 2 * (n + 1)) :=
66     Real.log_nonneg hproduct_one
67   have hlog_product :
68       Real.log (n ^ 2 * (n + 1)) =
69         2 * Real.log n + Real.log (n + 1) := by
70     rw [Real.log_mul (pow_ne_zero _ hn_pos.ne') (by positivity),
71       Real.log_pow]
72     norm_num
73   rw [Real.norm_eq_abs, abs_of_nonneg (mul_nonneg (by norm_num) hlog_product_nonneg),
74     Real.norm_eq_abs, abs_of_nonneg hlog_n_nonneg, hlog_product]
75   nlinarith

97 /--
98 The explicit tuned bounded-stopping-time expected pseudo-regret budget is
99 `O(sqrt (T + 1) * log (T + 1))`.
100 -/
101 theorem
102     telescopingStandardExpectedPseudoRegretBound_isBigO_sqrt_mul_log
103     {Feature : Type u}
104     [Fintype Feature] [Nonempty Feature]
105     (R : Real) (lambda : Real) (hlambda : 0 < lambda)
106     (S : Real) (L2 : Real) (hL2 : 0 <= L2) :
107     (fun horizon : Nat =>
108       telescopingStandardExpectedPseudoRegretBound
109         (Feature := Feature) R lambda S horizon L2) =O[atTop]
110       (fun horizon : Nat =>
111         Real.sqrt (((horizon + 1 : Nat) : Real)) *
112           Real.log (((horizon + 1 : Nat) : Real))) := by
113   let n : Nat -> Real := fun horizon => (((horizon + 1 : Nat) : Real))
114   let logN : Nat -> Real := fun horizon => Real.log (n horizon)
115   let sqrtN : Nat -> Real := fun horizon => Real.sqrt (n horizon)
116   let B : Nat -> Real := fun horizon =>
117     standardScalarLogDetBudget
118       (Feature := Feature) lambda (horizon + 1) L2
119   let C : Nat -> Real := fun horizon =>
120     telescopingStandardExpectedRegretLogBudget
121       (Feature := Feature) lambda horizon L2
122   have hlog_nonneg :
123       ∀ᶠ horizon in atTop, 0 <= logN horizon := by
124     exact Filter.Eventually.of_forall fun horizon => by
125       dsimp [logN, n]
126       exact Real.log_nonneg (by
127         exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon))
128   have hB : B =O[atTop] logN := by
129     simpa [B, logN, n] using
130       standardScalarLogDetBudget_isBigO_log
131         (Feature := Feature) lambda hlambda L2 hL2
132   have hC : C =O[atTop] logN := by
133     simpa [C, logN, n] using
134       telescopingStandardExpectedRegretLogBudget_isBigO_log
135         (Feature := Feature) lambda hlambda L2 hL2
136   have hsqrtC :
137       (fun horizon => Real.sqrt (C horizon)) =O[atTop]
138         (fun horizon => Real.sqrt (logN horizon)) :=
139     hC.sqrt hlog_nonneg
140   have hsqrt2B :
141       (fun horizon => Real.sqrt (2 * B horizon)) =O[atTop]
142         (fun horizon => Real.sqrt (logN horizon)) :=
143     (hB.const_mul_left 2).sqrt hlog_nonneg
144   have hsqrtLogLog :
145       (fun horizon => Real.sqrt (logN horizon)) =O[atTop] logN := by
146     simpa [logN, n] using sqrt_log_succ_isBigO_log_succ
147   have hsqrtN : sqrtN =O[atTop] sqrtN :=
148     Asymptotics.isBigO_refl _ _
149   have hsqrtN_sqrtLog_scale :
150       (fun horizon => sqrtN horizon * Real.sqrt (logN horizon)) =O[atTop]
151         (fun horizon => sqrtN horizon * logN horizon) :=
152     hsqrtN.mul hsqrtLogLog
153   have hvariance :
154       (fun horizon => sqrtN horizon * Real.sqrt (2 * B horizon)) =O[atTop]
155         (fun horizon => sqrtN horizon * logN horizon) :=
156     (hsqrtN.mul hsqrt2B).trans hsqrtN_sqrtLog_scale
157   have hconfidenceVariance :
158       (fun horizon =>
159         Real.sqrt (C horizon) *
160           (sqrtN horizon * Real.sqrt (2 * B horizon))) =O[atTop]
161         (fun horizon => sqrtN horizon * logN horizon) := by
162     have hproduct := hsqrtC.mul hsqrt2B
163     have hproductLog :
164         (fun horizon =>
165           Real.sqrt (C horizon) * Real.sqrt (2 * B horizon)) =O[atTop]
166           logN := by
167       refine hproduct.congr' Filter.EventuallyEq.rfl ?_
168       filter_upwards [hlog_nonneg] with horizon hlog
169       exact Real.mul_self_sqrt hlog
170     have hmul := hsqrtN.mul hproductLog
171     simpa [mul_assoc, mul_left_comm, mul_comm] using hmul
172   have hn_tendsto : Tendsto n atTop atTop := by
173     dsimp [n]
174     exact (tendsto_add_atTop_iff_nat 1).2 tendsto_natCast_atTop_atTop
175   have hconstantLog :
176       (fun _horizon : Nat => 4 * S * Real.sqrt L2) =o[atTop] logN := by
177     change
178       (fun _horizon : Nat => 4 * S * Real.sqrt L2) =o[atTop]
179         (fun horizon => Real.log (n horizon))
180     simpa only [Function.comp_apply] using
181       (Real.isLittleO_const_log_atTop
182         (c := 4 * S * Real.sqrt L2)).comp_tendsto hn_tendsto
183   have hlogScale :
184       logN =O[atTop] (fun horizon => sqrtN horizon * logN horizon) := by
185     refine (Asymptotics.isBigO_refl logN atTop).trans_le ?_
186     intro horizon
187     have hn_one : 1 <= n horizon := by
188       dsimp [n]
189       exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
190     have hsqrt_one : 1 <= sqrtN horizon := by
191       dsimp [sqrtN]
192       rw [← Real.sqrt_one]
193       exact Real.sqrt_le_sqrt hn_one
194     have hlog : 0 <= logN horizon := by
195       dsimp [logN]
196       exact Real.log_nonneg hn_one
197     rw [Real.norm_eq_abs, abs_of_nonneg hlog,
198       Real.norm_eq_abs, abs_of_nonneg
199         (mul_nonneg (Real.sqrt_nonneg _) hlog)]
200     nlinarith
201   have hconstant :
202       (fun _horizon : Nat => 4 * S * Real.sqrt L2) =O[atTop]
203         (fun horizon => sqrtN horizon * logN horizon) :=
204     hconstantLog.isBigO.trans hlogScale
205   have hconfidenceTerm :
206       (fun horizon =>
207         2 * R *
208           (Real.sqrt (C horizon) *
209             (sqrtN horizon * Real.sqrt (2 * B horizon)))) =O[atTop]
210         (fun horizon => sqrtN horizon * logN horizon) :=
211     hconfidenceVariance.const_mul_left (2 * R)
212   have hbiasTerm :
213       (fun horizon =>
214         (2 * (Real.sqrt lambda * S)) *
215           (sqrtN horizon * Real.sqrt (2 * B horizon))) =O[atTop]
216         (fun horizon => sqrtN horizon * logN horizon) :=
217     hvariance.const_mul_left (2 * (Real.sqrt lambda * S))
218   have hsum := hconstant.add (hconfidenceTerm.add hbiasTerm)
219   refine hsum.congr (fun horizon => ?_) (fun _ => rfl)
220   simp only [telescopingStandardExpectedPseudoRegretBound]
221   dsimp [B, C, sqrtN, logN, n]
222   ring

224 /--
225 For fixed model parameters and any horizon-indexed canonical stopping-time
226 family bounded pointwise by its horizon, the named expected stopped
227 pseudo-regret is `O(sqrt (T + 1) * log (T + 1))`.
228 -/
229 theorem
230     canonicalTelescopingStandardExpectedStoppedPseudoRegret_isBigO_sqrt_mul_log
231     {K : Nat} {Feature : Type u}
232     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
233     (hK : 0 < K)
234     (lambda : Real) (hlambda : 0 < lambda)
235     (thetaStar : Feature -> Real)
236     (actionFeature : Fin K -> Feature -> Real)
237     (R : Real) (hR : 0 < R)
238     (S : Real) (hS : 0 <= S)
239     (environment : Thompson.HistoryEnvironment (Fin K) Real)
240     (L2 : Real) (hL2 : 0 <= L2)
241     (hactionFeatureBound : forall action,
242       dotProduct (actionFeature action) (actionFeature action) <= L2)
243     (hL2lambda : L2 <= lambda)
244     (best : Fin K)
245     (hbest : IsOptimalLinearArm thetaStar actionFeature best)
246     (source : CanonicalLinearSubgaussianEnvironmentLaw
247       hK thetaStar actionFeature R S environment)
248     (tau : Nat -> (Nat -> Fin K × Real) -> WithTop Nat)
249     (htau : forall maxHorizon,
250       IsStoppingTime
251         (canonicalHistoryTrajectoryAllRoundFiltration (K := K))
252         (tau maxHorizon))
253     (htau_le : forall maxHorizon trajectory,
254       tau maxHorizon trajectory <= (maxHorizon : WithTop Nat)) :
255     (canonicalTelescopingStandardExpectedStoppedPseudoRegret
256       hK lambda thetaStar actionFeature R S environment best tau) =O[atTop]
257       (fun horizon : Nat =>
258         Real.sqrt (((horizon + 1 : Nat) : Real)) *
259           Real.log (((horizon + 1 : Nat) : Real))) := by
260   have hbound :=
261     telescopingStandardExpectedPseudoRegretBound_isBigO_sqrt_mul_log
262       (Feature := Feature) R lambda hlambda S L2 hL2
263   rw [Asymptotics.isBigO_iff] at hbound ⊢
264   obtain ⟨c, hc⟩ := hbound
265   refine ⟨c, ?_⟩
266   filter_upwards [hc] with horizon hcbound
267   have hpoint :=
268     canonicalTelescopingStandardExpectedStoppedPseudoRegret_nonneg_and_le
269       hK lambda hlambda thetaStar actionFeature R hR S hS environment
270       L2 hL2 hactionFeatureBound hL2lambda best hbest source
271       tau htau htau_le horizon
272   rw [Real.norm_eq_abs, abs_of_nonneg hpoint.1]
273   exact hpoint.2.trans ((le_abs_self _).trans hcbound)
