4 # Policy measurability surfaces
5 
6 This module records a narrow `MEAS-POLICY` leaf: a measurable policy applied to
7 a measurable history/context state yields a measurable action.  It is only a
8 measurability and predictability surface; it does not construct policy kernels,
9 trajectory laws, reward laws, or adaptive regret theorems.
10 -/
11 
12 universe u v w x
13

17 /--
18 A policy whose action map is measurable from a history/context state into the
19 action space.
20 -/
21 structure MeasurablePolicy
22     (State : Type u) (Action : Type v)
23     [MeasurableSpace State] [MeasurableSpace Action] where
24   action : State -> Action
25   measurable_action : Measurable action
26 
27 /-- A measurable history/context state composed with a measurable policy is a
28 measurable action random variable. -/
29 theorem measurable_action_of_measurable_state
30     {Omega : Type w} {State : Type u} {Action : Type v}
31     [MeasurableSpace Omega] [MeasurableSpace State] [MeasurableSpace Action]
32     (policy : MeasurablePolicy State Action)
33     (state : Omega -> State)
34     (hstate : Measurable state) :
35     Measurable (fun omega : Omega => policy.action (state omega)) := by
36   exact policy.measurable_action.comp hstate
37 
38 /--
39 If a history/context state is measurable with respect to a filtration at time
40 `t`, then the action selected by a measurable policy from that state is also
41 measurable with respect to the same filtration.
42 -/
43 theorem measurable_action_mem_filtration_of_measurable_state
44     {Omega : Type w} {State : Type u} {Action : Type v}
45     [mOmega : MeasurableSpace Omega]
46     [MeasurableSpace State] [MeasurableSpace Action]
47     (F : MeasureTheory.Filtration Nat mOmega)
48     (policy : MeasurablePolicy State Action)
49     (state : Omega -> State)
50     (t : Nat)
51     (hstate :
52       @Measurable Omega State
53         (F t) inferInstance state) :
54     @Measurable Omega Action
55       (F t) inferInstance
56       (fun omega : Omega => policy.action (state omega)) := by
57   exact policy.measurable_action.comp hstate
58 
59 /--
60 Specialization of policy measurability to the generated history filtration:
61 once a policy state is measurable from the past action/reward history at time
62 `t`, the policy-selected action is measurable from that same history.
63 -/
64 theorem measurable_action_mem_historyFiltration_of_measurable_state
65     {Omega : Type w} {TraceAction : Type v} {Reward : Type x}
66     {State : Type u} {PolicyAction : Type v}
67     [mOmega : MeasurableSpace Omega]
68     [MeasurableSpace TraceAction] [MeasurableSingletonClass TraceAction]
69     [MeasurableSpace Reward] [MeasurableSingletonClass Reward]
70     [MeasurableSpace State] [MeasurableSpace PolicyAction]
71     (traceAction : Omega -> ActionTrace TraceAction)
72     (reward : Omega -> RewardTrace Reward)
73     (haction : forall t : Nat,
74       Measurable (fun omega : Omega => traceAction omega t))
75     (hreward : forall t : Nat,
76       Measurable (fun omega : Omega => reward omega t))
77     (policy : MeasurablePolicy State PolicyAction)
78     (state : Omega -> State)
79     (t : Nat)
80     (hstate :
81       @Measurable Omega State
82         (History.historyFiltration traceAction reward haction hreward t)
83         inferInstance
84         state) :
85     @Measurable Omega PolicyAction
86       (History.historyFiltration traceAction reward haction hreward t)
87       inferInstance
88       (fun omega : Omega => policy.action (state omega)) := by
89   exact
90     measurable_action_mem_filtration_of_measurable_state
91       (History.historyFiltration traceAction reward haction hreward)
92       policy state t hstate

94 /--
95 The action trace generated by applying a fixed measurable policy to a
96 time-indexed history/context state process.
97 -/
98 def generatedActionTrace
99     {Omega : Type w} {State : Type u} {Action : Type v}
100     [MeasurableSpace State] [MeasurableSpace Action]
101     (policy : MeasurablePolicy State Action)
102     (state : Nat -> Omega -> State) :
103     Omega -> ActionTrace Action :=
104   fun omega t => policy.action (state t omega)
105 
106 /--
107 If every history/context state coordinate is ambient-measurable, every
108 coordinate of the generated policy action trace is ambient-measurable.
109 -/
110 theorem measurable_generatedActionTrace_eval_of_measurable_state
111     {Omega : Type w} {State : Type u} {Action : Type v}
112     [MeasurableSpace Omega] [MeasurableSpace State] [MeasurableSpace Action]
113     (policy : MeasurablePolicy State Action)
114     (state : Nat -> Omega -> State)
115     (hstate : forall t : Nat, Measurable (state t))
116     (t : Nat) :
117     Measurable
118       (fun omega : Omega =>
119         (generatedActionTrace policy state omega) t) := by
120   simpa [generatedActionTrace] using
121     measurable_action_of_measurable_state policy (state t) (hstate t)
122 
123 /--
124 The shifted action trace generated by a time-indexed policy.
125 
126 The value at time `t + 1` is selected by `policy t` from the state available at
127 time `t`.  This matches the one-step kernel convention used later by
128 `RewardKernel.actionRewardHistoryStepKernelFamily`.
129 -/
130 def generatedActionTraceSucc
131     {Omega : Type w} {State : Type u} {Action : Type v}
132     [MeasurableSpace State] [MeasurableSpace Action]
133     (policy : Nat -> MeasurablePolicy State Action)
134     (state : Nat -> Omega -> State)
135     (defaultAction : Action) :
136     Omega -> ActionTrace Action :=
137   fun omega t =>
138     match t with
139     | 0 => defaultAction
140     | Nat.succ n => (policy n).action (state n omega)
141 
142 /-- The initial value of the shifted generated trace is the chosen default. -/
143 @[simp]
144 theorem generatedActionTraceSucc_zero
145     {Omega : Type w} {State : Type u} {Action : Type v}
146     [MeasurableSpace State] [MeasurableSpace Action]
147     (policy : Nat -> MeasurablePolicy State Action)
148     (state : Nat -> Omega -> State)
149     (defaultAction : Action)
150     (omega : Omega) :
151     (generatedActionTraceSucc policy state defaultAction omega) 0 =
152       defaultAction := rfl
153 
154 /--
155 At time `t + 1`, the shifted generated trace is exactly the action selected by
156 `policy t`.
157 -/
158 @[simp]
159 theorem generatedActionTraceSucc_succ
160     {Omega : Type w} {State : Type u} {Action : Type v}
161     [MeasurableSpace State] [MeasurableSpace Action]
162     (policy : Nat -> MeasurablePolicy State Action)
163     (state : Nat -> Omega -> State)
164     (defaultAction : Action)
165     (omega : Omega) (t : Nat) :
166     (generatedActionTraceSucc policy state defaultAction omega) (t + 1) =
167       (policy t).action (state t omega) := by
168   rfl
169 
170 /-- Function-level equality for the predictable `t + 1` coordinate. -/
171 theorem generatedActionTraceSucc_succ_eq
172     {Omega : Type w} {State : Type u} {Action : Type v}
173     [MeasurableSpace State] [MeasurableSpace Action]
174     (policy : Nat -> MeasurablePolicy State Action)
175     (state : Nat -> Omega -> State)
176     (defaultAction : Action)
177     (t : Nat) :
178     (fun omega : Omega =>
179       (generatedActionTraceSucc policy state defaultAction omega) (t + 1)) =
180     (fun omega : Omega => (policy t).action (state t omega)) := by
181   funext omega
182   simp [generatedActionTraceSucc]
183 
184 /--
185 If every state coordinate is ambient-measurable, every coordinate of the
186 shifted generated policy action trace is ambient-measurable.
187 -/
188 theorem measurable_generatedActionTraceSucc_eval_of_measurable_state
189     {Omega : Type w} {State : Type u} {Action : Type v}
190     [MeasurableSpace Omega] [MeasurableSpace State] [MeasurableSpace Action]
191     (policy : Nat -> MeasurablePolicy State Action)
192     (state : Nat -> Omega -> State)
193     (defaultAction : Action)
194     (hstate : forall t : Nat, Measurable (state t))
195     (t : Nat) :
196     Measurable
197       (fun omega : Omega =>
198         (generatedActionTraceSucc policy state defaultAction omega) t) := by
199   cases t with
200   | zero =>
201       simp [generatedActionTraceSucc]
202   | succ n =>
203       simpa [generatedActionTraceSucc] using
204         measurable_action_of_measurable_state (policy n) (state n) (hstate n)
205 
206 /--
207 Predictable-coordinate measurability for the shifted generated action trace:
208 if the state at time `t` is measurable in `F t`, then action coordinate
209 `t + 1` is also measurable in `F t`.
210 -/
211 theorem measurable_generatedActionTraceSucc_succ_mem_filtration_of_measurable_state
212     {Omega : Type w} {State : Type u} {Action : Type v}
213     [mOmega : MeasurableSpace Omega]
214     [MeasurableSpace State] [MeasurableSpace Action]
215     (F : MeasureTheory.Filtration Nat mOmega)
216     (policy : Nat -> MeasurablePolicy State Action)
217     (state : Nat -> Omega -> State)
218     (defaultAction : Action)
219     (hstate :
220       forall t : Nat,
221         @Measurable Omega State
222           (F t) inferInstance (state t))
223     (t : Nat) :
224     @Measurable Omega Action
225       (F t) inferInstance
226       (fun omega : Omega =>
227         (generatedActionTraceSucc policy state defaultAction omega) (t + 1)) := by
228   simpa [generatedActionTraceSucc] using
229     measurable_action_mem_filtration_of_measurable_state
230       F (policy t) (state t) t (hstate t)
