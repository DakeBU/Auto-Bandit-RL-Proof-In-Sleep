9 # Curvature--Noise--Gap geometry: finite algebraic leaves
10 
11 This file contains only route-independent finite-dimensional algebra.  It does
12 not formalize the full Curvature--Noise--Gap calculus, Tsallis-INF, or a new
13 bandit theorem.  The intended later use is to test whether these abstractions
14 replace repeated route-specific proof subgraphs and transfer to held-out proof
15 families.
16 
17 The assumptions are deliberately explicit.  Tangent invariance needs only a
18 zero-sum direction.  The weighted shift minimum needs nonnegative weights and
19 strictly positive total weight; it does not hide an interiority or
20 positive-definiteness premise.
21 -/

28 /-- A finite direction tangent to the affine simplex hyperplane. -/
29 def IsSimplexTangent {Index : Type*} [DecidableEq Index]
30     (indices : Finset Index) (direction : Index -> Real) : Prop :=
31   indices.sum direction = 0
32 
33 /-- The finite pairing used to test a signal against a tangent direction. -/
34 def tangentPairing {Index : Type*} [DecidableEq Index]
35     (indices : Finset Index) (signal direction : Index -> Real) : Real :=
36   indices.sum (fun i => signal i * direction i)
37 
38 /-- Adding a constant to a signal is invisible on the simplex tangent space. -/
39 theorem tangentPairing_add_const_of_isSimplexTangent
40     {Index : Type*} [DecidableEq Index]
41     (indices : Finset Index) (signal direction : Index -> Real) (shift : Real)
42     (hdirection : IsSimplexTangent indices direction) :
43     tangentPairing indices (fun i => signal i + shift) direction =
44       tangentPairing indices signal direction := by
45   unfold tangentPairing IsSimplexTangent at *
46   simp_rw [add_mul]
47   rw [Finset.sum_add_distrib, <- Finset.mul_sum, hdirection, mul_zero, add_zero]

49 /-- Weighted squared energy after removing one common scalar shift. -/
50 def weightedShiftEnergy {Index : Type*} [DecidableEq Index]
51     (indices : Finset Index) (weight signal : Index -> Real) (shift : Real) : Real :=
52   indices.sum (fun i => weight i * (signal i - shift) ^ 2)
53 
54 /-- The scalar shift selected by a nondegenerate weighted quadratic energy. -/
55 noncomputable def weightedCenter {Index : Type*} [DecidableEq Index]
56     (indices : Finset Index) (weight signal : Index -> Real) : Real :=
57   indices.sum (fun i => weight i * signal i) / indices.sum weight
58 
59 /-- The weighted residual about `weightedCenter` has zero weighted sum. -/
60 theorem sum_weight_mul_sub_weightedCenter_eq_zero
61     {Index : Type*} [DecidableEq Index]
62     (indices : Finset Index) (weight signal : Index -> Real)
63     (hweightSum : indices.sum weight ≠ 0) :
64     indices.sum (fun i =>
65       weight i * (signal i - weightedCenter indices weight signal)) = 0 := by
66   simp_rw [mul_sub]
67   rw [Finset.sum_sub_distrib, <- Finset.sum_mul]
68   unfold weightedCenter
69   field_simp
70   ring
71 
72 /-- Completing the square around any weighted-centered scalar. -/
73 theorem weightedShiftEnergy_decomposition_of_centered
74     {Index : Type*} [DecidableEq Index]
75     (indices : Finset Index) (weight signal : Index -> Real)
76     (center shift : Real)
77     (hcenter : indices.sum (fun i => weight i * (signal i - center)) = 0) :
78     weightedShiftEnergy indices weight signal shift =
79       weightedShiftEnergy indices weight signal center +
80         indices.sum weight * (shift - center) ^ 2 := by
81   unfold weightedShiftEnergy
82   calc
83     indices.sum (fun i => weight i * (signal i - shift) ^ 2) =
84         indices.sum (fun i =>
85           (weight i * (signal i - center) ^ 2 +
86             2 * (center - shift) * (weight i * (signal i - center))) +
87               weight i * (center - shift) ^ 2) := by
88       apply Finset.sum_congr rfl
89       intro i _
90       ring
91     _ = indices.sum (fun i => weight i * (signal i - center) ^ 2) +
92           2 * (center - shift) *
93               indices.sum (fun i => weight i * (signal i - center)) +
94             indices.sum weight * (center - shift) ^ 2 := by
95       rw [Finset.sum_add_distrib, Finset.sum_add_distrib,
96         <- Finset.mul_sum, <- Finset.sum_mul]
97     _ = indices.sum (fun i => weight i * (signal i - center) ^ 2) +
98           indices.sum weight * (shift - center) ^ 2 := by
99       rw [hcenter]
100       ring

102 /-- Exact min-shift decomposition at the weighted center. -/
103 theorem weightedShiftEnergy_decomposition
104     {Index : Type*} [DecidableEq Index]
105     (indices : Finset Index) (weight signal : Index -> Real) (shift : Real)
106     (hweightSum : indices.sum weight ≠ 0) :
107     weightedShiftEnergy indices weight signal shift =
108       weightedShiftEnergy indices weight signal (weightedCenter indices weight signal) +
109         indices.sum weight *
110           (shift - weightedCenter indices weight signal) ^ 2 := by
111   exact weightedShiftEnergy_decomposition_of_centered
112     indices weight signal (weightedCenter indices weight signal) shift
113     (sum_weight_mul_sub_weightedCenter_eq_zero
114       indices weight signal hweightSum)

116 /-- The weighted center minimizes the quadratic shift energy. -/
117 theorem weightedShiftEnergy_center_le
118     {Index : Type*} [DecidableEq Index]
119     (indices : Finset Index) (weight signal : Index -> Real) (shift : Real)
120     (hweight : forall i, i ∈ indices -> 0 <= weight i)
121     (hweightSum : 0 < indices.sum weight) :
122     weightedShiftEnergy indices weight signal (weightedCenter indices weight signal) <=
123       weightedShiftEnergy indices weight signal shift := by
124   rw [weightedShiftEnergy_decomposition indices weight signal shift hweightSum.ne']
125   exact le_add_of_nonneg_right
126     (mul_nonneg (Finset.sum_nonneg hweight) (sq_nonneg _))
127 
128 /-- Strict positivity of the total weight makes the minimizing shift unique. -/
129 theorem weightedShiftEnergy_eq_center_iff
130     {Index : Type*} [DecidableEq Index]
131     (indices : Finset Index) (weight signal : Index -> Real) (shift : Real)
132     (hweightSum : 0 < indices.sum weight) :
133     weightedShiftEnergy indices weight signal shift =
134         weightedShiftEnergy indices weight signal (weightedCenter indices weight signal) <->
135       shift = weightedCenter indices weight signal := by
136   rw [weightedShiftEnergy_decomposition indices weight signal shift hweightSum.ne']
137   constructor
138   · intro henergy
139     have hmul : indices.sum weight *
140         (shift - weightedCenter indices weight signal) ^ 2 = 0 := by
141       linarith
142     have hsquare :
143         (shift - weightedCenter indices weight signal) ^ 2 = 0 :=
144       (mul_eq_zero.mp hmul).resolve_left hweightSum.ne'
145     have hsub : shift - weightedCenter indices weight signal = 0 := by
146       nlinarith
147     exact sub_eq_zero.mp hsub
148   · intro hshift
149     rw [hshift]
150     ring

152 /-- Exact signal--noise decomposition, including the interaction term. -/
153 theorem weightedShiftEnergy_add_decomposition
154     {Index : Type*} [DecidableEq Index]
155     (indices : Finset Index) (weight signal noise : Index -> Real)
156     (signalShift noiseShift : Real) :
157     weightedShiftEnergy indices weight (fun i => signal i + noise i)
158         (signalShift + noiseShift) =
159       weightedShiftEnergy indices weight signal signalShift +
160         weightedShiftEnergy indices weight noise noiseShift +
161           2 * indices.sum (fun i =>
162             weight i * (signal i - signalShift) * (noise i - noiseShift)) := by
163   unfold weightedShiftEnergy
164   calc
165     indices.sum (fun i =>
166         weight i * (signal i + noise i - (signalShift + noiseShift)) ^ 2) =
167       indices.sum (fun i =>
168         (weight i * (signal i - signalShift) ^ 2 +
169           weight i * (noise i - noiseShift) ^ 2) +
170             2 * (weight i * (signal i - signalShift) *
171               (noise i - noiseShift))) := by
172         apply Finset.sum_congr rfl
173         intro i _
174         ring
175     _ = indices.sum (fun i => weight i * (signal i - signalShift) ^ 2) +
176           indices.sum (fun i => weight i * (noise i - noiseShift) ^ 2) +
177             2 * indices.sum (fun i =>
178               weight i * (signal i - signalShift) * (noise i - noiseShift)) := by
179         rw [Finset.sum_add_distrib, Finset.sum_add_distrib, <- Finset.mul_sum]

181 /-- A conservative two-term signal--noise energy bound. -/
182 theorem weightedShiftEnergy_add_le_two
183     {Index : Type*} [DecidableEq Index]
184     (indices : Finset Index) (weight signal noise : Index -> Real)
185     (signalShift noiseShift : Real)
186     (hweight : forall i, i ∈ indices -> 0 <= weight i) :
187     weightedShiftEnergy indices weight (fun i => signal i + noise i)
188         (signalShift + noiseShift) <=
189       2 * weightedShiftEnergy indices weight signal signalShift +
190         2 * weightedShiftEnergy indices weight noise noiseShift := by
191   unfold weightedShiftEnergy
192   calc
193     indices.sum (fun i =>
194         weight i * (signal i + noise i - (signalShift + noiseShift)) ^ 2) <=
195       indices.sum (fun i =>
196         2 * (weight i * (signal i - signalShift) ^ 2) +
197           2 * (weight i * (noise i - noiseShift) ^ 2)) := by
198         apply Finset.sum_le_sum
199         intro i hi
200         have hw := hweight i hi
201         nlinarith [sq_nonneg ((signal i - signalShift) -
202           (noise i - noiseShift))]
203     _ = 2 * indices.sum (fun i => weight i * (signal i - signalShift) ^ 2) +
204           2 * indices.sum (fun i => weight i * (noise i - noiseShift) ^ 2) := by
205         rw [Finset.sum_add_distrib, <- Finset.mul_sum, <- Finset.mul_sum]
