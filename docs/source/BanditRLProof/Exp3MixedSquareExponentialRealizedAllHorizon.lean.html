5 # All-horizon exponential mixed-square realized EXP3 route
6 
7 The explicit exponential-square schedule has its refined `14 * gamma * T`
8 threshold when four horizon inequalities make clipping inactive. Outside that
9 regime this module reuses the compiled almost-sure horizon bound and the strict
10 `T + 1` zero-probability threshold. The fallback closes the active-clipping
11 presentation without claiming a sharp short-horizon, Freedman, or EXP3.P rate.
12 -/

20 /-- The regime in which all four components of the exponential-square
21 exploration schedule are at most one half. -/
22 def exponentialSquareBernsteinLargeHorizonCondition
23     (K T delta : Real) : Prop :=
24   4 * (K * Real.log K) <= T ∧
25     64 * (K ^ 2 * Real.log K ^ 2 * Real.log (4 / delta) / 2) <=
26       T ^ 3 ∧
27     8 * (K * Real.log (4 / delta)) <= T ∧
28     8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
29       Real.log (4 / delta) <= T

31 /-- All-horizon threshold for the exponential-square route: use the explicit
32 large-horizon rate in its valid regime and `T + 1` otherwise. -/
33 noncomputable def exponentialSquareBernsteinAllHorizonRegretThreshold
34     {Action : Type v} (arms : Finset Action) (horizon : Nat)
35     (delta : Real) : Real := by
36   classical
37   let gamma := exponentialSquareBernsteinClippedExplorationRate
38     (arms.card : Real) (horizon : Real) delta
39   exact if exponentialSquareBernsteinLargeHorizonCondition
40       (arms.card : Real) (horizon : Real) delta then
41     exponentialSquareBernsteinRealizedExplicitThreshold
42       arms gamma horizon delta
43   else
44     (horizon : Real) + 1
