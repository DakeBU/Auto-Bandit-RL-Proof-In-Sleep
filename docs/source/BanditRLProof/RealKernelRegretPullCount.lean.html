4 /-!
5 # Real kernel regret and pull-count decomposition
6 
7 This module specializes the Real mean-regret bookkeeping surface to an
8 arm-indexed Mathlib kernel.  The mean of arm `a` is the Bochner integral of the
9 identity under the measure `nu a`, exactly as in the LML finite-bandit regret
10 definitions.  Algorithm laws and concentration assumptions remain downstream.
11 -/
12 
13 namespace BanditRLProof
14 
15 open MeasureTheory
16 
17 universe u
18 
19 /-- Identity-integral mean of a Real-valued arm kernel. -/
20 noncomputable def realKernelMean {K : Nat}
21     (nu : ProbabilityTheory.Kernel (Fin K) Real) (a : Fin K) : Real :=
22   integral (nu a) id
23 
24 /-- Gap of a Real-valued arm kernel from its supremum arm mean. -/
25 noncomputable def realKernelGap {K : Nat}
26     (nu : ProbabilityTheory.Kernel (Fin K) Real) (a : Fin K) : Real :=
27   realMeanGap (realKernelMean nu) a
28 
29 /-- Finite-horizon regret of an action trace against a Real-valued arm kernel. -/
30 noncomputable def realKernelRegret {K : Nat}
31     (nu : ProbabilityTheory.Kernel (Fin K) Real)
32     (action : ActionTrace (Fin K)) (n : Nat) : Real :=
33   realMeanRegret (realKernelMean nu) action n
34 
35 /-- Every kernel arm gap is nonnegative when the finite arm type is nonempty. -/
36 theorem realKernelGap_nonneg
37     {K : Nat} [Nonempty (Fin K)]
38     (nu : ProbabilityTheory.Kernel (Fin K) Real) (a : Fin K) :
39     0 <= realKernelGap nu a := by
40   rw [realKernelGap, realMeanGap, sub_nonneg]
41   exact le_ciSup (f := realKernelMean nu) (by simp) a
42 
43 /-- Kernel regret is the finite time-indexed sum of selected kernel gaps. -/
44 theorem realKernelRegret_eq_finset_sum_gap
45     {K : Nat}
46     (nu : ProbabilityTheory.Kernel (Fin K) Real)
47     (action : ActionTrace (Fin K)) (n : Nat) :
48     realKernelRegret nu action n =
49       (Finset.range n).sum (fun t => realKernelGap nu (action t)) := by
50   simpa [realKernelRegret, realKernelGap] using
51     realMeanRegret_eq_finset_sum_gap (realKernelMean nu) action n
52 
53 /-- Kernel regret is the gap-weighted finite sum of arm pull counts. -/
54 theorem realKernelRegret_eq_sum_gap_mul_pullCount
55     {K : Nat}
56     (nu : ProbabilityTheory.Kernel (Fin K) Real)
57     (action : ActionTrace (Fin K)) (n : Nat) :
58     realKernelRegret nu action n =
59       (Finset.univ : Finset (Fin K)).sum
60         (fun a => realKernelGap nu a * (pullCount action a n : Real)) := by
61   simpa [realKernelRegret, realKernelGap] using
62     realMeanRegret_eq_sum_gap_mul_pullCount (realKernelMean nu) action n
63 
64 /-- Pull-count integrability implies integrability of Real kernel regret. -/
65 theorem integrable_realKernelRegret_of_integrable_pullCount
66     {Omega : Type u} {K : Nat}
67     [MeasurableSpace Omega]
68     (mu : Measure Omega)
69     (nu : ProbabilityTheory.Kernel (Fin K) Real)
70     (action : Omega -> ActionTrace (Fin K))
71     (n : Nat)
72     (hcount : forall a : Fin K,
73       Integrable (fun omega => (pullCount (action omega) a n : Real)) mu) :
74     Integrable (fun omega => realKernelRegret nu (action omega) n) mu := by
75   simpa [realKernelRegret] using
76     integrable_realMeanRegret_of_integrable_pullCount
77       mu (realKernelMean nu) action n hcount
78 
79 /--
80 The Bochner expectation of Real kernel regret is the gap-weighted sum of
81 expected arm pull counts.
82 
83 This is the kernel-facing bookkeeping endpoint for the exact ETC route.  It
84 does not assume a Markov kernel, probability measure, identity integrability,
85 algorithm/environment law, sub-Gaussian proxy, or argmax semantics.  Those
86 contracts are required only by downstream statistical theorems.
87 -/
88 theorem integral_realKernelRegret_eq_sum_gap_mul_integral_pullCount
89     {Omega : Type u} {K : Nat}
90     [MeasurableSpace Omega]
91     (mu : Measure Omega)
92     (nu : ProbabilityTheory.Kernel (Fin K) Real)
93     (action : Omega -> ActionTrace (Fin K))
94     (n : Nat)
95     (hcount : forall a : Fin K,
96       Integrable (fun omega => (pullCount (action omega) a n : Real)) mu) :
97     integral mu (fun omega => realKernelRegret nu (action omega) n) =
98       (Finset.univ : Finset (Fin K)).sum
99         (fun a =>
100           realKernelGap nu a *
101             integral mu (fun omega => (pullCount (action omega) a n : Real))) := by
102   simpa [realKernelRegret, realKernelGap] using
103     integral_realMeanRegret_eq_sum_gap_mul_integral_pullCount
104       mu (realKernelMean nu) action n hcount
105 
106 end BanditRLProof
