6 /-!
7 # Half-Tsallis conjugate-potential stability
8 
9 This module formalizes the deterministic potential quantity used in the
10 paper-faithful refined Tsallis-INF stability route.  The local learning-rate
11 normalization is half the paper normalization, so the translated Lemma 19
12 coefficients are `eta` for the quadratic term and `2 * eta^2` for the positive
13 cubic remainder.
14 
15 The constrained potential is represented by the negative regularized
16 objective value at a certified simplex minimizer, plus `1 / eta`.  That final
17 constant matches the paper normalization under `eta_paper = 2 * eta_local`; it
18 cancels in every fixed-eta step here but must be present for future cross-eta
19 telescoping.  The explicit coordinate increment is the unconstrained
20 half-Tsallis conjugate upper bound.  Their connection is proved below from
21 interior stationarity and simplex normalization; no measure, kernel, or
22 trajectory premise is involved.
23 -/

30 /--
31 The paper-normalized negative regularized-objective value.  When `probability`
32 is a certified minimizer and `eta > 0`, this is the constrained half-Tsallis
33 potential under the local-to-paper learning-rate translation.
34 -/
35 noncomputable def halfTsallisPotentialValue
36     {Action : Type u}
37     (arms : Finset Action) (eta : Real)
38     (score probability : Action -> Real) : Real :=
39   -FTRL.regularizedObjective arms eta
40       (negEntropyRegularizer arms (1 / 2 : Real)) score probability / eta +
41     1 / eta
42 
43 /--
44 The one-step candidate-potential expression before taking a sampling average.
45 The low-level feasible-next bridge treats `next` only as a candidate point;
46 the algorithm-facing theorems require both current and next minimizer
47 certificates.
48 -/
49 noncomputable def halfTsallisPotentialStability
50     {Action : Type u}
51     (arms : Finset Action) (eta : Real)
52     (score probability estimate next : Action -> Real) : Real :=
53   FTRL.linearLoss arms probability estimate +
54     halfTsallisPotentialValue arms eta
55       (fun action => score action + estimate action) next -
56     halfTsallisPotentialValue arms eta score probability

58 /--
59 The coordinate increment obtained from the explicit unconstrained conjugate.
60 The shift is `estimate action - baseline`.
61 -/
62 noncomputable def halfTsallisConjugateCoordinateIncrement
63     (eta probability shift : Real) : Real :=
64   probability * shift +
65     Real.sqrt probability /
66       (eta * (1 + eta * shift * Real.sqrt probability)) -
67     Real.sqrt probability / eta
68 
69 /-- Finite sum of explicit conjugate coordinate increments. -/
70 noncomputable def halfTsallisConjugatePotentialUpper
71     {Action : Type u}
72     (arms : Finset Action) (eta : Real)
73     (probability estimate : Action -> Real) (baseline : Real) : Real :=
74   arms.sum (fun action =>
75     halfTsallisConjugateCoordinateIncrement eta (probability action)
76       (estimate action - baseline))

86 /-- Exact rational form of the explicit conjugate coordinate increment. -/
87 theorem halfTsallisConjugateCoordinateIncrement_eq
88     {eta probability shift : Real}
89     (heta : 0 < eta) (hprobability : 0 < probability)
90     (hdomain : -1 <= 2 * eta * shift * Real.sqrt probability) :
91     halfTsallisConjugateCoordinateIncrement eta probability shift =
92       eta * Real.sqrt probability * probability * shift ^ 2 /
93         (1 + eta * shift * Real.sqrt probability) := by
94   let s := Real.sqrt probability
95   let denominator := 1 + eta * shift * s
96   have hsqrt : 0 < s := by simpa [s] using Real.sqrt_pos.2 hprobability
97   have hsqrt_sq : s ^ 2 = probability := by
98     simpa [s] using Real.sq_sqrt hprobability.le
99   have hden : 0 < denominator := by
100     simpa [denominator, s] using
101       one_add_eta_mul_shift_mul_sqrt_pos heta hprobability hdomain
102   have hetaNe : eta ≠ 0 := ne_of_gt heta
103   have hdenNe : denominator ≠ 0 := ne_of_gt hden
104   unfold halfTsallisConjugateCoordinateIncrement
105   change probability * shift + s / (eta * denominator) - s / eta =
106     eta * s * probability * shift ^ 2 / denominator
107   apply (eq_div_iff hdenNe).2
108   have hcancel : s / (eta * denominator) * denominator = s / eta := by
109     field_simp [hetaNe, hdenNe]
110   calc
111     (probability * shift + s / (eta * denominator) - s / eta) *
112         denominator =
113       probability * shift * denominator +
114         (s / (eta * denominator) * denominator) -
115         (s / eta) * denominator := by ring
116     _ = probability * shift * denominator + s / eta -
117         (s / eta) * denominator := by rw [hcancel]
118     _ = eta * s * probability * shift ^ 2 := by
119       field_simp [hetaNe]
120       linear_combination -eta * shift * hsqrt_sq

122 /--
123 Scalar translated Lemma 19 bound.  The positive cubic term is active only
124 when the shifted estimate is negative.
125 -/
126 theorem halfTsallisConjugateCoordinateIncrement_le
127     {eta probability shift : Real}
128     (heta : 0 < eta) (hprobability : 0 < probability)
129     (hdomain : -1 <= 2 * eta * shift * Real.sqrt probability) :
130     halfTsallisConjugateCoordinateIncrement eta probability shift <=
131       eta * Real.sqrt probability * probability * shift ^ 2 +
132         2 * eta ^ 2 * probability ^ 2 * (max (-shift) 0) ^ 3 := by
133   rw [halfTsallisConjugateCoordinateIncrement_eq
134     heta hprobability hdomain]
135   have hsqrt : 0 < Real.sqrt probability := Real.sqrt_pos.2 hprobability
136   have hsqrt_sq : (Real.sqrt probability) ^ 2 = probability :=
137     Real.sq_sqrt hprobability.le
138   have hden := one_add_eta_mul_shift_mul_sqrt_pos
139     heta hprobability hdomain
140   by_cases hshift : 0 <= shift
141   · rw [max_eq_right]
142     · simp only [zero_pow (by norm_num : (3 : Nat) ≠ 0), mul_zero, add_zero]
143       apply (div_le_iff₀ hden).2
144       have hmain :
145           0 <= eta * Real.sqrt probability * probability * shift ^ 2 := by
146         positivity
147       have hdenOne : 1 <= 1 + eta * shift * Real.sqrt probability := by
148         have : 0 <= eta * shift * Real.sqrt probability := by positivity
149         linarith
150       simpa only [mul_one] using
151         (mul_le_mul_of_nonneg_left hdenOne hmain)
152     · linarith
153   · have hshiftNeg : shift < 0 := lt_of_not_ge hshift
154     rw [max_eq_left (by linarith : 0 <= -shift)]
155     apply (div_le_iff₀ hden).2
156     have hbase :
157         0 <= eta * Real.sqrt probability * probability * shift ^ 2 := by
158       positivity
159     have hremainder :
160         0 <= 2 * eta ^ 2 * probability ^ 2 * (-shift) ^ 3 := by
161       exact mul_nonneg
162         (mul_nonneg (by positivity) (by positivity))
163         (pow_nonneg (by linarith) 3)
164     have hfactor :
165         (eta * Real.sqrt probability * probability * shift ^ 2 +
166               2 * eta ^ 2 * probability ^ 2 * (-shift) ^ 3) *
167             (1 + eta * shift * Real.sqrt probability) -
168           eta * Real.sqrt probability * probability * shift ^ 2 =
169         eta ^ 2 * probability ^ 2 * (-shift) ^ 3 *
170           (1 + 2 * eta * shift * Real.sqrt probability) := by
171       linear_combination
172         eta ^ 2 * probability * shift ^ 3 * hsqrt_sq
173     have hfactorNonneg :
174         0 <= eta ^ 2 * probability ^ 2 * (-shift) ^ 3 *
175           (1 + 2 * eta * shift * Real.sqrt probability) := by
176       exact mul_nonneg
177         (mul_nonneg
178           (mul_nonneg (sq_nonneg eta) (sq_nonneg probability))
179           (pow_nonneg (by linarith) 3))
180         (by linarith)
181     linarith

183 /-- Coordinatewise Fenchel upper bound for a nonnegative competitor weight. -/
184 theorem halfTsallis_fenchelCoordinate_le
185     {eta probability competitor shift : Real}
186     (heta : 0 < eta) (hprobability : 0 < probability)
187     (hcompetitor : 0 <= competitor)
188     (hdomain : -1 <= 2 * eta * shift * Real.sqrt probability) :
189     -competitor / (eta * Real.sqrt probability) - competitor * shift +
190         2 * Real.sqrt competitor / eta <=
191       Real.sqrt probability /
192         (eta * (1 + eta * shift * Real.sqrt probability)) := by
193   let s := Real.sqrt probability
194   let r := Real.sqrt competitor
195   let denominator := 1 + eta * shift * s
196   have hs : 0 < s := by simpa [s] using Real.sqrt_pos.2 hprobability
197   have hrSq : r ^ 2 = competitor := by
198     simpa [r] using Real.sq_sqrt hcompetitor
199   have hden : 0 < denominator := by
200     simpa [denominator, s] using
201       one_add_eta_mul_shift_mul_sqrt_pos heta hprobability hdomain
202   have hetaS : 0 < eta * s := mul_pos heta hs
203   have hetaDen : 0 < eta * denominator := mul_pos heta hden
204   have hleft :
205       -competitor / (eta * s) - competitor * shift + 2 * r / eta =
206         (-competitor * denominator + 2 * s * r) / (eta * s) := by
207     field_simp [ne_of_gt heta, ne_of_gt hs]
208     ring
209   have hcore :
210       (-competitor * denominator + 2 * s * r) * denominator <= s ^ 2 := by
211     rw [← hrSq]
212     nlinarith [sq_nonneg (r * denominator - s)]
213   rw [hleft]
214   apply (div_le_div_iff₀ hetaS hetaDen).2
215   calc
216     (-competitor * denominator + 2 * s * r) *
217         (eta * denominator) =
218       eta * ((-competitor * denominator + 2 * s * r) * denominator) := by
219         ring
220     _ <= eta * s ^ 2 := mul_le_mul_of_nonneg_left hcore heta.le
221     _ = s * (eta * s) := by ring

223 /-- The explicit conjugate finite sum is bounded by shifted quadratic/cubic moments. -/
224 theorem halfTsallisConjugatePotentialUpper_le_shiftedMoments
225     {Action : Type u} [DecidableEq Action]
226     (arms : Finset Action) (eta : Real)
227     (probability estimate : Action -> Real) (baseline : Real)
228     (heta : 0 < eta)
229     (hprobability : forall action, action ∈ arms -> 0 < probability action)
230     (hdomain : forall action, action ∈ arms ->
231       -1 <= 2 * eta * (estimate action - baseline) *
232         Real.sqrt (probability action)) :
233     halfTsallisConjugatePotentialUpper
234         arms eta probability estimate baseline <=
235       eta * arms.sum (fun action =>
236         Real.sqrt (probability action) * probability action *
237           (estimate action - baseline) ^ 2) +
238       2 * eta ^ 2 * arms.sum (fun action =>
239         probability action ^ 2 *
240           (max (baseline - estimate action) 0) ^ 3) := by
241   unfold halfTsallisConjugatePotentialUpper
242   calc
243     arms.sum (fun action =>
244         halfTsallisConjugateCoordinateIncrement eta (probability action)
245           (estimate action - baseline)) <=
246       arms.sum (fun action =>
247         eta * Real.sqrt (probability action) * probability action *
248             (estimate action - baseline) ^ 2 +
249           2 * eta ^ 2 * probability action ^ 2 *
250             (max (-(estimate action - baseline)) 0) ^ 3) := by
251       apply Finset.sum_le_sum
252       intro action haction
253       exact halfTsallisConjugateCoordinateIncrement_le heta
254         (hprobability action haction) (hdomain action haction)
255     _ = eta * arms.sum (fun action =>
256           Real.sqrt (probability action) * probability action *
257             (estimate action - baseline) ^ 2) +
258         2 * eta ^ 2 * arms.sum (fun action =>
259           probability action ^ 2 *
260             (max (baseline - estimate action) 0) ^ 3) := by
261       rw [Finset.sum_add_distrib, Finset.mul_sum, Finset.mul_sum]
262       congr 1
263       · apply Finset.sum_congr rfl
264         intro action _haction
265         ring
266       · apply Finset.sum_congr rfl
267         intro action _haction
268         have hshift :
269             -(estimate action - baseline) = baseline - estimate action := by
270           ring
271         rw [hshift]
272         ring

275 theorem importanceWeightedLoss_sub_selectedLoss_conjugate_domain
276     {Action : Type u} [DecidableEq Action]
277     (arms : Finset Action) (eta : Real)
278     (probability loss : Action -> Real)
279     (heta : 0 < eta) (heta_le : eta <= 1 / 2)
280     (hprobability : FTRL.finiteSimplex arms probability)
281     (hprobabilityPos : forall action, action ∈ arms -> 0 < probability action)
282     (hloss : forall action, action ∈ arms ->
283       0 <= loss action ∧ loss action <= 1)
284     {chosen action : Action} (hchosen : chosen ∈ arms) (haction : action ∈ arms) :
285     -1 <= 2 * eta *
286       (Exp3.importanceWeightedLoss probability loss chosen action - loss chosen) *
287         Real.sqrt (probability action) := by
288   by_cases hsame : chosen = action
289   · subst action
290     have hp := hprobabilityPos chosen hchosen
291     have hpLe := finiteSimplex_apply_le_one hprobability hchosen
292     have hl := hloss chosen hchosen
293     have hlDiv : loss chosen <= loss chosen / probability chosen := by
294       apply (le_div_iff₀ hp).2
295       calc
296         loss chosen * probability chosen <= loss chosen * 1 :=
297           mul_le_mul_of_nonneg_left hpLe hl.1
298         _ = loss chosen := by ring
299     have hshift :
300         0 <= loss chosen / probability chosen - loss chosen := sub_nonneg.2 hlDiv
301     simp only [Exp3.importanceWeightedLoss, if_true]
302     have : 0 <= 2 * eta *
303         (loss chosen / probability chosen - loss chosen) *
304           Real.sqrt (probability chosen) := by positivity
305     linarith
306   · have hpNonneg := hprobability.1 action haction
307     have hpLe := finiteSimplex_apply_le_one hprobability haction
308     have hsqrtLe : Real.sqrt (probability action) <= 1 :=
309       Real.sqrt_le_one.2 hpLe
310     have hl := hloss chosen hchosen
311     have htwoEtaNonneg : 0 <= 2 * eta := by positivity
312     have htwoEtaLe : 2 * eta <= 1 := by nlinarith
313     have hscaledLoss : 2 * eta * loss chosen <= 1 := by
314       calc
315         2 * eta * loss chosen <= 2 * eta * 1 :=
316           mul_le_mul_of_nonneg_left hl.2 htwoEtaNonneg
317         _ <= 1 := by simpa using htwoEtaLe
318     have hproduct :
319         2 * eta * loss chosen * Real.sqrt (probability action) <= 1 := by
320       calc
321         2 * eta * loss chosen * Real.sqrt (probability action) <=
322             2 * eta * loss chosen * 1 :=
323           mul_le_mul_of_nonneg_left hsqrtLe
324             (mul_nonneg htwoEtaNonneg hl.1)
325         _ <= 1 := by simpa using hscaledLoss
326     simp [Exp3.importanceWeightedLoss, hsame]
327     nlinarith

359 /-- Stationarity removes the score and its common simplex multiplier. -/
360 theorem linearLoss_sub_linearLoss_score_eq_sum_div_sqrt_of_stationary
361     {Action : Type u} [DecidableEq Action]
362     (arms : Finset Action) (eta : Real)
363     (score p q : Action -> Real) (multiplier : Real)
364     (heta : 0 < eta)
365     (hp : FTRL.finiteSimplex arms p)
366     (hq : FTRL.finiteSimplex arms q)
367     (hpPos : forall action, action ∈ arms -> 0 < p action)
368     (hstationary : HalfTsallisInteriorStationary
369       arms eta score p multiplier) :
370     FTRL.linearLoss arms p score - FTRL.linearLoss arms q score =
371       arms.sum (fun action =>
372         (p action - q action) / (eta * Real.sqrt (p action))) := by
373   rw [linearLoss_sub_linearLoss_eq_sum_sub_mul]

375       score action = multiplier / eta +
376         1 / (eta * Real.sqrt (p action)) := by
377     intro action haction

379     have hrpow :

387       score action = (multiplier + 1 / Real.sqrt (p action)) / eta := by

394     rw [hscore action haction])]

396     rw [Finset.sum_sub_distrib, hp.2, hq.2]
397     ring

400         (p action - q action) *
401           (multiplier / eta + 1 / (eta * Real.sqrt (p action)))) =
402       multiplier / eta * arms.sum (fun action => p action - q action) +
403         arms.sum (fun action =>
404           (p action - q action) / (eta * Real.sqrt (p action))) := by
405       simp_rw [mul_add]
406       rw [Finset.sum_add_distrib]
407       congr 1
408       · rw [Finset.mul_sum]
409         apply Finset.sum_congr rfl
410         intro action _haction
411         ring
412       · apply Finset.sum_congr rfl
413         intro action _haction
414         ring
415     _ = arms.sum (fun action =>
416           (p action - q action) / (eta * Real.sqrt (p action))) := by
417       rw [hsumDiff]

420 /--
421 The feasible-next candidate-potential expression is bounded by the explicit
422 unconstrained conjugate sum.  Only the current point needs stationarity.  This
423 is an algebraic bridge, not an actual constrained-potential theorem unless the
424 caller separately certifies that `next` minimizes the updated objective.
425 -/
426 theorem halfTsallisPotentialStability_le_conjugatePotentialUpper_of_feasible
427     {Action : Type u} [DecidableEq Action]
428     (arms : Finset Action) (eta : Real)
429     (score probability estimate next : Action -> Real)
430     (baseline multiplier : Real)
431     (heta : 0 < eta)
432     (hprobability : FTRL.finiteSimplex arms probability)
433     (hnext : FTRL.finiteSimplex arms next)
434     (hprobabilityPos : forall action, action ∈ arms -> 0 < probability action)
435     (hstationary : HalfTsallisInteriorStationary
436       arms eta score probability multiplier)
437     (hdomain : forall action, action ∈ arms ->
438       -1 <= 2 * eta * (estimate action - baseline) *
439         Real.sqrt (probability action)) :
440     halfTsallisPotentialStability
441         arms eta score probability estimate next <=
442       halfTsallisConjugatePotentialUpper
443         arms eta probability estimate baseline := by
444   have hetaNe : eta ≠ 0 := ne_of_gt heta
445   have hlinearAdd :
446       FTRL.linearLoss arms next (fun action => score action + estimate action) =
447         FTRL.linearLoss arms next score +
448           FTRL.linearLoss arms next estimate := by
449     unfold FTRL.linearLoss
450     simp_rw [mul_add]
451     rw [Finset.sum_add_distrib]
452   have hstability :
453       halfTsallisPotentialStability arms eta score probability estimate next =
454         (FTRL.linearLoss arms probability estimate -
455             FTRL.linearLoss arms next estimate) +
456           (FTRL.linearLoss arms probability score -
457             FTRL.linearLoss arms next score) +
458           2 / eta *
459             (arms.sum (fun action => Real.sqrt (next action)) -
460               arms.sum (fun action => Real.sqrt (probability action))) := by
461     unfold halfTsallisPotentialStability halfTsallisPotentialValue
462     rw [regularizedObjective_half_eq, regularizedObjective_half_eq, hlinearAdd]
463     field_simp [hetaNe]
464     ring
465   have hscore :=
466     linearLoss_sub_linearLoss_score_eq_sum_div_sqrt_of_stationary
467       arms eta score probability next multiplier heta hprobability hnext
468       hprobabilityPos hstationary
469   have hestimate :
470       FTRL.linearLoss arms probability estimate -
471           FTRL.linearLoss arms next estimate =
472         arms.sum (fun action =>
473           (probability action - next action) *
474             (estimate action - baseline)) := by
475     rw [linearLoss_sub_linearLoss_eq_sum_sub_mul]
476     symm
477     exact sum_sub_mul_sub_baseline_eq_sum_sub_mul
478       arms probability next estimate baseline hprobability.2 hnext.2
479   have hsqrt :
480       2 / eta *
481           (arms.sum (fun action => Real.sqrt (next action)) -
482             arms.sum (fun action => Real.sqrt (probability action))) =
483         arms.sum (fun action =>
484           2 / eta *
485             (Real.sqrt (next action) - Real.sqrt (probability action))) := by
486     rw [mul_sub, Finset.mul_sum, Finset.mul_sum,
487       ← Finset.sum_sub_distrib]
488     apply Finset.sum_congr rfl
489     intro action _haction
490     ring
491   rw [hstability, hscore, hestimate, hsqrt]
492   rw [← Finset.sum_add_distrib, ← Finset.sum_add_distrib]
493   unfold halfTsallisConjugatePotentialUpper
494   apply Finset.sum_le_sum
495   intro action haction
496   have hfenchel := halfTsallis_fenchelCoordinate_le
497     heta (hprobabilityPos action haction) (hnext.1 action haction)
498     (hdomain action haction)
499   have hpAction := hprobabilityPos action haction
500   have hsqrtSq : (Real.sqrt (probability action)) ^ 2 =
501       probability action := Real.sq_sqrt hpAction.le
502   have hprobDiv :
503       probability action /
504           (eta * Real.sqrt (probability action)) =
505         Real.sqrt (probability action) / eta := by
506     field_simp [hetaNe, ne_of_gt (Real.sqrt_pos.2 hpAction)]
507     linear_combination -hsqrtSq
508   calc
509     (probability action - next action) * (estimate action - baseline) +
510           (probability action - next action) /
511             (eta * Real.sqrt (probability action)) +
512         2 / eta *
513           (Real.sqrt (next action) - Real.sqrt (probability action)) =
514       probability action * (estimate action - baseline) -
515           Real.sqrt (probability action) / eta +
