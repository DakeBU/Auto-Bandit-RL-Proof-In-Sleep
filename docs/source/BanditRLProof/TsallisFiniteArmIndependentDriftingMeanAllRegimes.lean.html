4 # All-regimes regret for independent reward laws with drifting means
5 
6 This module combines the compact-window refined endpoint with the logarithmic
7 fallback for the same independent, nonidentical generated reward law. The
8 comparator remains the fixed baseline arm `model.bestArm`.
9 -/

17 /-- Total explicit fixed-comparator bound for independent nonidentical reward
18 laws with drifting means. The refined expression is used exactly inside its
19 coefficient-aware window; the logarithmic explicit-budget expression is used
20 on the complement. -/
21 noncomputable def finiteArmIndependentDriftingMeanAllRegimeBound
22     {K : Nat} (model : FiniteBanditModel K) (horizon : Nat)
23     (meanDeviation : Nat -> Fin K -> Real) : Real := by
24   classical
25   let actions := (Finset.univ : Finset (Fin K)).erase model.bestArm
26   let armCount : Real := (actions.card : Real)
27   let horizonMass : Real := ((horizon + 1 : Nat) : Real)
28   let reciprocalGap := actions.sum
29     (fun arm => 1 / ((model.gap arm : Rat) : Real))
30   let scale := 2 * armCount * horizonMass
31   let corruption :=
32     finiteArmIndependentMeanDeviationBudget model horizon meanDeviation
33   exact if actions.Nonempty ∧
34       finiteArmIndependentDriftingMeanRefinedCorruptionWindow
35         model horizon meanDeviation then
36     1 + Real.log horizonMass +
37       10 * Real.sqrt (corruption * reciprocalGap) *
38         (2 + Real.sqrt
39           (Real.log (scale / (corruption * reciprocalGap)) + 1))
40   else
41     (1 + Real.log horizonMass) * (1 + 25 * reciprocalGap) + corruption

43 /-- With one arm there is no suboptimal coordinate, so the all-regimes
44 envelope reduces to the logarithmic base term. -/
45 @[simp]
46 theorem finiteArmIndependentDriftingMeanAllRegimeBound_fin_one
47     (model : FiniteBanditModel 1) (horizon : Nat)
48     (meanDeviation : Nat -> Fin 1 -> Real) :
49     finiteArmIndependentDriftingMeanAllRegimeBound
50         model horizon meanDeviation =
51       1 + Real.log (((horizon + 1 : Nat) : Real)) := by
52   have hbest : model.bestArm = 0 := Subsingleton.elim _ _
53   simp [finiteArmIndependentDriftingMeanAllRegimeBound,
54     finiteArmIndependentMeanDeviationBudget_eq, hbest]
