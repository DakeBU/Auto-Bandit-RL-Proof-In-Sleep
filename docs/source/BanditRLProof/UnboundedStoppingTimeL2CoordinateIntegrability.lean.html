7 /-!
8 # L2 stopping fibers and unbounded stopped-value integrability
9 
10 This module provides the countable-fiber transport needed when an unbounded
11 `WithTop Nat` stopping time has an L2 round count and the deterministic-time
12 coordinates have a uniform L2 bound. The argument is a measurable equality-
13 fiber decomposition plus Holder; it is not optional stopping.
14 -/

23 /-- The squared successor-round count is the countable sum of its weighted
24 equality fibers. This is an equality in `ENNReal`, so no integrability
25 assumption is needed. -/
26 theorem tsum_natSuccSquare_mul_stoppingFiberMeasure_eq_lintegral_rounds_sq
27     {Omega : Type u} [MeasurableSpace Omega]
28     (mu : Measure Omega)
29     (tau : Omega -> WithTop Nat) (htau : Measurable tau)
30     (hfinite : ∀ᵐ omega ∂mu, tau omega ≠ ⊤) :
31     (∑' n : Nat,
32       ((((n + 1) ^ 2 : Nat) : ENNReal) *
33         mu {omega | tau omega = (n : WithTop Nat)})) =
34       ∫⁻ omega,
35         ENNReal.ofReal
36           (((((tau omega).untopA + 1 : Nat) : Real)) ^ 2) ∂mu := by
37   let rounds : Omega -> Real := fun omega =>
38     (((tau omega).untopA + 1 : Nat) : Real)
39   let fiber : Nat -> Set Omega := fun n =>
40     {omega | tau omega = (n : WithTop Nat)}
41   let term : Nat -> Omega -> ENNReal := fun n =>
42     (fiber n).indicator (fun _ => ((((n + 1) ^ 2 : Nat) : ENNReal)))
43   have hfiberMeasurable (n : Nat) : MeasurableSet (fiber n) := by
44     exact measurableSet_eq_fun htau measurable_const
45   have htermMeasurable (n : Nat) : AEMeasurable (term n) mu := by
46     exact (Measurable.indicator measurable_const
47       (hfiberMeasurable n)).aemeasurable
48   have htermIntegral (n : Nat) :
49       ∫⁻ omega, term n omega ∂mu =
50         ((((n + 1) ^ 2 : Nat) : ENNReal) * mu (fiber n)) := by
51     unfold term
52     rw [lintegral_indicator_const (hfiberMeasurable n)]
53   have hpoint : ∀ᵐ omega ∂mu,
54       (∑' n : Nat, term n omega) = ENNReal.ofReal (rounds omega ^ 2) := by
55     filter_upwards [hfinite] with omega htop
56     lift tau omega to Nat using htop with hit hhit
57     dsimp [term, fiber, rounds]
58     rw [tsum_eq_single hit]
59     · rw [Set.indicator_of_mem (by simpa using hhit.symm)]
60       have huntopA : (tau omega).untopA = hit := by
61         rw [← hhit]
62         rfl
63       rw [huntopA, ENNReal.ofReal_pow (by positivity) 2,
64         ENNReal.ofReal_natCast]
65       norm_cast
66     · intro n hn
67       rw [Set.indicator_of_notMem]
68       · intro hmem
69         have heq : hit = n := by
70           simpa using hhit.trans hmem
71         exact hn heq.symm
72   calc
73     (∑' n : Nat,
74         ((((n + 1) ^ 2 : Nat) : ENNReal) * mu (fiber n))) =
75         ∑' n : Nat, ∫⁻ omega, term n omega ∂mu := by
76           apply tsum_congr
77           intro n
78           exact (htermIntegral n).symm
79     _ = ∫⁻ omega, ∑' n : Nat, term n omega ∂mu := by
80       rw [lintegral_tsum htermMeasurable]
81     _ = ∫⁻ omega, ENNReal.ofReal (rounds omega ^ 2) ∂mu :=
82       lintegral_congr_ae hpoint
83     _ = ∫⁻ omega,
84         ENNReal.ofReal
85           (((((tau omega).untopA + 1 : Nat) : Real)) ^ 2) ∂mu := rfl

166         ((((n + 1) ^ 2 : Nat) : ENNReal) * mu (fiber n))) ∞ := by

171       (((n + 1 : Nat) : Real) ^ 2) * mu.real (fiber n)) := by

173     simpa only [ENNReal.toReal_mul, ENNReal.toReal_natCast,

176       1 / (((n + 1 : Nat) : Real) ^ 2)) := by
177     let base : Nat -> Real := fun n => 1 / ((n : Real) ^ 2)
178     have hbase : Summable base := by
179       exact Real.summable_one_div_nat_pow.mpr (by norm_num)

181       ((summable_nat_add_iff (f := base) 1).2 hbase)
182   have hmajorant : Summable (fun n : Nat =>

184         ((((n + 1 : Nat) : Real) ^ 2) * mu.real (fiber n) +
185           1 / (((n + 1 : Nat) : Real) ^ 2))) :=
186     (hweightedReal.add hinverseSquare).mul_left (1 / 2 : Real)
187   apply hmajorant.of_nonneg_of_le
188   · intro n
189     exact Real.sqrt_nonneg _
190   · intro n
191     let x : Real := ((n + 1 : Nat) : Real)
192     let p : Real := mu.real (fiber n)
193     change Real.sqrt p <= (1 / 2 : Real) * (x ^ 2 * p + 1 / x ^ 2)

195       dsimp [x]
196       positivity
197     have hp : 0 <= p := by
198       dsimp [p]
199       exact measureReal_nonneg
200     have hsqrtSq : (Real.sqrt p) ^ 2 = p := Real.sq_sqrt hp
201     have hxInv : x * (1 / x) = 1 := by
202       field_simp
203     have hyoung := sq_nonneg (x * Real.sqrt p - 1 / x)
204     have hyoungIdentity :
205         (x * Real.sqrt p - 1 / x) ^ 2 =
206           x ^ 2 * p - 2 * Real.sqrt p + (1 / x) ^ 2 := by
207       calc
208         (x * Real.sqrt p - 1 / x) ^ 2 =
209             x ^ 2 * (Real.sqrt p) ^ 2 -
210               2 * (x * (1 / x)) * Real.sqrt p + (1 / x) ^ 2 := by
211                 ring
212         _ = x ^ 2 * p - 2 * Real.sqrt p + (1 / x) ^ 2 := by
213           rw [hsqrtSq, hxInv]
214           ring
215     rw [hyoungIdentity] at hyoung
216     have hraw :
217         2 * Real.sqrt p <= x ^ 2 * p + (1 / x) ^ 2 := by
218       nlinarith
219     have hinvSq : (1 / x) ^ 2 = 1 / x ^ 2 := by ring
220     rw [← hinvSq]
221     nlinarith

434 /-- Uniform deterministic-coordinate second moments and an L2 finite stopping
435 time make the corresponding unbounded stopped value integrable. The proof is
436 a countable equality-fiber decomposition; it does not use optional stopping. -/
437 theorem integrable_stoppedValue_of_uniform_secondMoment_of_memLp_two_rounds
438     {Omega : Type u} [MeasurableSpace Omega]
439     (mu : Measure Omega) [IsFiniteMeasure mu]
440     (tau : Omega -> WithTop Nat) (htau : Measurable tau)
441     (hfinite : ∀ᵐ omega ∂mu, tau omega ≠ ⊤)
442     (hrounds :
443       MemLp (fun omega => (((tau omega).untopA + 1 : Nat) : Real)) 2 mu)
444     (process : Nat -> Omega -> Real)
445     (hstoppedMeasurable : Measurable (stoppedValue process tau))
446     (secondMomentEnvelope : Real)
447     (hprocessMemLp : ∀ n, MemLp (process n) 2 mu)
448     (hprocessSecondMoment : ∀ n,
449       integral mu (fun omega => process n omega ^ 2) <=
450         secondMomentEnvelope) :
451     Integrable (stoppedValue process tau) mu := by
452   let fiber : Nat -> Set Omega := fun n =>
453     {omega | tau omega = (n : WithTop Nat)}
454   let term : Nat -> Omega -> ENNReal := fun n =>
455     (fiber n).indicator (fun omega => ENNReal.ofReal |process n omega|)
456   let majorant : Nat -> Real := fun n =>
457     Real.sqrt secondMomentEnvelope * Real.sqrt (mu.real (fiber n))
458   have hfiberMeasurable (n : Nat) : MeasurableSet (fiber n) := by
459     exact measurableSet_eq_fun htau measurable_const
