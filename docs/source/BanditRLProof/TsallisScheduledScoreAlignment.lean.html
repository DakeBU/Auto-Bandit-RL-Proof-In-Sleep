4 /-!
5 # Scheduled half-Tsallis score alignment
6 
7 This module identifies the recursively accumulated scheduled half-Tsallis
8 history score with `FTRL.cumulativeLoss` of the actual observed
9 importance-weighted loss vectors.  It then rewrites the deterministic
10 time-varying penalty theorem onto one generated trajectory sample.
11 
12 The result is pathwise.  It does not yet integrate the penalty, prove the
13 roundwise refined stability bound under a schedule, handle early large rates,
14 or conclude Tsallis-INF regret.
15 -/

22 /-- The pure scheduled half-Tsallis sampling probability at an actual time. -/
23 noncomputable def sampledScheduledHalfTsallisProbabilityAtTime
24     {Env : Type u} {Action : Type v} [DecidableEq Action]
25     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real) :
26     Nat -> Env × ((k : Nat) -> Action × Real) -> Action -> Real
27   | 0, _sample => initialHalfTsallisDistribution arms harms (eta 0)
28   | n + 1, sample =>
29       sampledScheduledHalfTsallisHistoryDistribution arms harms eta n
30         (Preorder.frestrictLe n sample.2)
31 
32 /-- The actual observed-scalar IW loss vector under the scheduled policy. -/
33 noncomputable def sampledScheduledHalfTsallisObservedEstimatedLossAt
34     {Env : Type u} {Action : Type v} [DecidableEq Action]
35     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
36     (t : Nat) (sample : Env × ((k : Nat) -> Action × Real)) :
37     Action -> Real :=
38   Exp3.importanceWeightedLoss
39     (sampledScheduledHalfTsallisProbabilityAtTime arms harms eta t sample)
40     (fun _ => (sample.2 t).2) (sample.2 t).1

42 /-- The cumulative observed IW loss through time `n` is exactly the inclusive
43 scheduled finite-history score at level `n`. -/
44 theorem cumulativeLoss_sampledScheduledHalfTsallisObservedEstimatedLossAt_succ
45     {Env : Type u} {Action : Type v} [DecidableEq Action]
46     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
47     (sample : Env × ((k : Nat) -> Action × Real)) (n : Nat) :
48     FTRL.cumulativeLoss
49         (fun t => sampledScheduledHalfTsallisObservedEstimatedLossAt
50           arms harms eta t sample) (n + 1) =
51       sampledScheduledHalfTsallisHistoryScore arms harms eta n
52         (Preorder.frestrictLe n sample.2) := by
53   induction n with
54   | zero =>
55       funext action
56       simp [FTRL.cumulativeLoss_succ,
57         sampledScheduledHalfTsallisObservedEstimatedLossAt,
58         sampledScheduledHalfTsallisProbabilityAtTime]
59   | succ n ih =>
60       rw [FTRL.cumulativeLoss_succ]
61       funext candidate
62       change
63         FTRL.cumulativeLoss
64               (fun t => sampledScheduledHalfTsallisObservedEstimatedLossAt
65                 arms harms eta t sample) (n + 1) candidate +
66             sampledScheduledHalfTsallisObservedEstimatedLossAt
67               arms harms eta (n + 1) sample candidate = _
68       rw [ih, sampledScheduledHalfTsallisHistoryScore_succ,
69         Exp3.previousPairHistory_frestrictLe]
70       rfl

72 /-- The canonical scheduled cumulative selector is the actual generated
73 sampling probability at the same time. -/
74 @[simp]
75 theorem halfTsallisScheduledMinimizer_observedEstimatedLoss_eq_probabilityAtTime
76     {Env : Type u} {Action : Type v} [DecidableEq Action]
77     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
78     (sample : Env × ((k : Nat) -> Action × Real)) (t : Nat) :
79     halfTsallisScheduledMinimizer arms harms eta
80         (fun s => sampledScheduledHalfTsallisObservedEstimatedLossAt
81           arms harms eta s sample) t =
82       sampledScheduledHalfTsallisProbabilityAtTime
83         arms harms eta t sample := by
84   cases t with
85   | zero =>
86       unfold halfTsallisScheduledMinimizer
87         sampledScheduledHalfTsallisProbabilityAtTime
88         initialHalfTsallisDistribution
89       apply congrArg (halfTsallisMinimizer arms harms (eta 0))
90       funext action
91       simp [FTRL.cumulativeLoss]
92   | succ n =>
93       unfold halfTsallisScheduledMinimizer
94       rw [cumulativeLoss_sampledScheduledHalfTsallisObservedEstimatedLossAt_succ]
95       rfl

108 /-- The same-rate conjugate-potential stability term at an actual trajectory
109 time. -/
110 noncomputable def sampledScheduledHalfTsallisPotentialStabilityAtTime
111     {Env : Type u} {Action : Type v} [DecidableEq Action]
112     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
113     (sample : Env × ((k : Nat) -> Action × Real)) (t : Nat) : Real :=
114   let observedLoss := fun s =>
115     sampledScheduledHalfTsallisObservedEstimatedLossAt
116       arms harms eta s sample

119     (sampledScheduledHalfTsallisProbabilityAtTime

125 /-- The scheduled potential-change penalty at an actual trajectory time. -/
126 noncomputable def sampledScheduledHalfTsallisPotentialPenaltyAtTime
127     {Env : Type u} {Action : Type v} [DecidableEq Action]
128     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
129     (q : Action -> Real)
130     (sample : Env × ((k : Nat) -> Action × Real)) (t : Nat) : Real :=
131   let observedLoss := fun s =>
132     sampledScheduledHalfTsallisObservedEstimatedLossAt
133       arms harms eta s sample
134   halfTsallisPotentialValue arms (eta t)
135       (FTRL.cumulativeLoss observedLoss t)
136       (sampledScheduledHalfTsallisProbabilityAtTime
137         arms harms eta t sample) -
138     halfTsallisPotentialValue arms (eta t)
139       (FTRL.cumulativeLoss observedLoss (t + 1))
140       (sampledScheduledHalfTsallisSameRateNextAt
141         arms harms eta sample t) -
142     FTRL.linearLoss arms q (observedLoss t)

144 /-- Pathwise estimated regret against a finite-simplex comparator through the
145 inclusive terminal time `n`. -/
146 noncomputable def sampledScheduledHalfTsallisEstimatedRegret
147     {Env : Type u} {Action : Type v} [DecidableEq Action]
148     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
149     (q : Action -> Real) (n : Nat)
150     (sample : Env × ((k : Nat) -> Action × Real)) : Real :=
151   (Finset.range (n + 1)).sum (fun t =>
152     FTRL.linearLoss arms
153         (sampledScheduledHalfTsallisProbabilityAtTime
154           arms harms eta t sample)
155         (sampledScheduledHalfTsallisObservedEstimatedLossAt
156           arms harms eta t sample) -

159           arms harms eta t sample))

161 /-- Exact pathwise decomposition of generated scheduled estimated regret into
162 same-rate potential stability plus the learning-rate-change penalty. -/
163 theorem sampledScheduledHalfTsallisEstimatedRegret_eq_stability_add_penalty
164     {Env : Type u} {Action : Type v} [DecidableEq Action]
165     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
166     (q : Action -> Real) (n : Nat)
167     (sample : Env × ((k : Nat) -> Action × Real)) :
168     sampledScheduledHalfTsallisEstimatedRegret
169         arms harms eta q n sample =
170       (Finset.range (n + 1)).sum (fun t =>
171         sampledScheduledHalfTsallisPotentialStabilityAtTime
172           arms harms eta sample t) +
173       (Finset.range (n + 1)).sum (fun t =>
174         sampledScheduledHalfTsallisPotentialPenaltyAtTime
175           arms harms eta q sample t) := by
176   rw [← Finset.sum_add_distrib]
177   apply Finset.sum_congr rfl
178   intro t _ht
179   unfold sampledScheduledHalfTsallisPotentialStabilityAtTime
180     sampledScheduledHalfTsallisPotentialPenaltyAtTime
181     halfTsallisPotentialStability
182   dsimp only
183   rw [FTRL.cumulativeLoss_succ]
184   ring

186 /-- Pathwise generated-trajectory specialization of the scheduled point-mass
187 penalty theorem, retaining the explicit terminal `-1 / eta n` contribution. -/
188 theorem sum_sampledScheduledHalfTsallisPotentialPenalty_pointMass_le
189     {Env : Type u} {Action : Type v} [DecidableEq Action]
190     (arms : Finset Action) (harms : arms.Nonempty)
191     (eta : Nat -> Real)
192     (sample : Env × ((k : Nat) -> Action × Real))
193     {best : Action} (hbest : best ∈ arms) (n : Nat)
194     (heta : forall t, t <= n -> 0 < eta t)
195     (hetaMono : forall t, t < n -> eta (t + 1) <= eta t) :
196     (Finset.range (n + 1)).sum (fun t =>
197         sampledScheduledHalfTsallisPotentialPenaltyAtTime
198           arms harms eta (pointMass best) sample t) <=
199       halfTsallisPotentialMass arms
200           (sampledScheduledHalfTsallisProbabilityAtTime
201             arms harms eta 0 sample) / eta n -
202         1 / eta n := by
203   simpa [sampledScheduledHalfTsallisPotentialPenaltyAtTime,
204     sampledScheduledHalfTsallisSameRateNextAt] using

206       arms harms eta

211 /-- Generated pathwise best-arm estimated regret is bounded by the scheduled
212 same-rate stability sum plus the explicit initial-minus-terminal penalty. -/
213 theorem sampledScheduledHalfTsallisEstimatedRegret_pointMass_le_stability_add_penalty
214     {Env : Type u} {Action : Type v} [DecidableEq Action]
215     (arms : Finset Action) (harms : arms.Nonempty)
216     (eta : Nat -> Real)
217     (sample : Env × ((k : Nat) -> Action × Real))
218     {best : Action} (hbest : best ∈ arms) (n : Nat)
219     (heta : forall t, t <= n -> 0 < eta t)
220     (hetaMono : forall t, t < n -> eta (t + 1) <= eta t) :
221     sampledScheduledHalfTsallisEstimatedRegret
222         arms harms eta (pointMass best) n sample <=
223       (Finset.range (n + 1)).sum (fun t =>
224         sampledScheduledHalfTsallisPotentialStabilityAtTime
225           arms harms eta sample t) +
226         halfTsallisPotentialMass arms
227           (sampledScheduledHalfTsallisProbabilityAtTime
