5 /-!
6 # Tsallis finite-simplex regularizer wrappers
7 
8 This module records the first deterministic `Real.rpow` surface for
9 Tsallis-INF/FTRL routes.  It defines the finite-action Tsallis power sum,
10 Tsallis entropy, and the negative-entropy regularizer used as an FTRL
11 regularizer, then packages the small well-definedness side conditions over the
12 finite simplex.
13 
14 It does not prove convexity, stability/penalty bounds, self-bounding
15 conversion, learning-rate optimization, or regret.
16 -/
17 
18 namespace BanditRLProof
19 namespace Tsallis
20 
21 universe u
22 
23 /-- Finite Tsallis power sum `sum_a p_a^alpha`. -/
24 noncomputable def powerSum {Action : Type u}
25     (arms : Finset Action) (alpha : Real) (p : Action -> Real) : Real :=
26   arms.sum (fun a => (p a) ^ alpha)
27 
28 /-- Tsallis entropy on a finite action set, with denominator left explicit. -/
29 noncomputable def entropy {Action : Type u}
30     (arms : Finset Action) (alpha : Real) (p : Action -> Real) : Real :=
31   (powerSum arms alpha p - 1) / (1 - alpha)
32 
33 /-- Negative Tsallis entropy as the FTRL regularizer convention. -/
34 noncomputable def negEntropyRegularizer {Action : Type u}
35     (arms : Finset Action) (alpha : Real) (p : Action -> Real) : Real :=
36   - entropy arms alpha p
37 
38 /-- The Tsallis denominator is nonzero when `alpha != 1`. -/
39 theorem one_sub_exponent_ne_zero {alpha : Real}
40     (halpha : alpha ≠ 1) :
41     1 - alpha ≠ 0 := by
42   intro h
43   apply halpha
44   linarith
45 
46 /-- The Tsallis power sum is nonnegative on the finite simplex. -/
47 theorem powerSum_nonneg_of_finiteSimplex {Action : Type u}
48     (arms : Finset Action) (alpha : Real) (p : Action -> Real)
49     (hp : FTRL.finiteSimplex arms p) :
50     0 <= powerSum arms alpha p := by
51   exact Finset.sum_nonneg (fun a ha =>
52     Real.rpow_nonneg (hp.1 a ha) alpha)
53 
54 /--
55 Well-definedness package for the finite-simplex Tsallis regularizer.
56 
57 The two facts exposed here are the local obligations needed before later
58 Tsallis/FTRL leaves can use `Real.rpow` algebra and division by `1 - alpha`.
59 -/
60 theorem negEntropyRegularizer_wellDefined_on_finiteSimplex
61     {Action : Type u}
62     (arms : Finset Action) (alpha : Real) (p : Action -> Real)
63     (hp : FTRL.finiteSimplex arms p) (halpha : alpha ≠ 1) :
64     0 <= powerSum arms alpha p ∧
65       1 - alpha ≠ 0 ∧
66       negEntropyRegularizer arms alpha p =
67         - ((powerSum arms alpha p - 1) / (1 - alpha)) := by
68   exact ⟨
69     powerSum_nonneg_of_finiteSimplex arms alpha p hp,
70     one_sub_exponent_ne_zero halpha,
71     rfl⟩
72 
73 end Tsallis
74 end BanditRLProof
