4 /-!
5 # Finite union probability bounds
6 
7 Thin Mathlib-backed finite-union wrappers.  These are outer-measure bounds:
8 no event measurability or probability-measure assumption is required.
9 -/
10 
11 namespace BanditRLProof
12 namespace ProbabilityUnionBound
13 
14 open MeasureTheory
15 
16 universe u v
17 
18 /--
19 The measure of a finite union of events is bounded by the finite sum of their
20 measures.
21 
22 This is an outer-measure wrapper around
23 `MeasureTheory.measure_biUnion_finset_le`, so it does not require the events to
24 be measurable and does not require `mu` to be a probability measure.
25 -/
26 theorem measure_biUnion_finset_le
27     {Omega : Type u} [MeasurableSpace Omega]
28     {Idx : Type v}
29     (mu : Measure Omega)
30     (s : Finset Idx)
31     (E : Idx -> Set Omega) :
32     mu (⋃ i ∈ s, E i) <=
33       s.sum (fun i => mu (E i)) := by
34   simpa using
35     (MeasureTheory.measure_biUnion_finset_le
36       (μ := mu)
37       (I := s)
38       (s := E))
39 
40 /--
41 Equal-share finite-union bound.
42 
43 Every event receives confidence budget `delta / s.card`; nonemptiness of the
44 index set lets the finite sum normalize back to `delta`.  As with
45 `measure_biUnion_finset_le`, no event measurability or probability-measure
46 assumption is required.
47 -/
48 theorem measure_biUnion_finset_le_of_uniform
49     {Omega : Type u} [MeasurableSpace Omega]
50     {Idx : Type v} [DecidableEq Idx]
51     (mu : Measure Omega)
52     (s : Finset Idx) (hs : s.Nonempty)
53     (delta : Real)
54     (E : Idx -> Set Omega)
55     (hE : forall i, i ∈ s ->
56       mu (E i) <= ENNReal.ofReal (delta / (s.card : Real))) :
57     mu (⋃ i ∈ s, E i) <= ENNReal.ofReal delta := by
58   have hcardNat : 0 < s.card := Finset.card_pos.mpr hs
59   have hcardReal : 0 < (s.card : Real) := Nat.cast_pos.mpr hcardNat
60   have hcardENN_ne_zero : (s.card : ENNReal) ≠ 0 := by
61     simp [Finset.card_ne_zero.mpr hs]
62   have hcardENN_ne_top : (s.card : ENNReal) ≠ ⊤ := by simp
63   calc
64     mu (⋃ i ∈ s, E i) <= s.sum (fun i => mu (E i)) :=
65       measure_biUnion_finset_le mu s E
66     _ <= s.sum (fun _i =>
67         ENNReal.ofReal (delta / (s.card : Real))) := by
68       exact Finset.sum_le_sum fun i hi => hE i hi
69     _ = (s.card : ENNReal) *
70         ENNReal.ofReal (delta / (s.card : Real)) := by
71       simp [nsmul_eq_mul]
72     _ = ENNReal.ofReal delta := by
73       rw [ENNReal.ofReal_div_of_pos hcardReal]
74       simp only [ENNReal.ofReal_natCast]
75       exact ENNReal.mul_div_cancel hcardENN_ne_zero hcardENN_ne_top
76 
77 /--
78 Fintype specialization of the finite-union probability bound.
79 
80 The index set is `(Finset.univ : Finset Idx)`, matching the finite-arm style
81 used by the bandit proofs.
82 -/
83 theorem measure_iUnion_fintype_le_sum
84     {Omega : Type u} [MeasurableSpace Omega]
85     {Idx : Type v} [Fintype Idx]
86     (mu : Measure Omega)
87     (E : Idx -> Set Omega) :
88     mu (⋃ i, E i) <=
89       (Finset.univ : Finset Idx).sum (fun i => mu (E i)) := by
90   simpa using
91     (MeasureTheory.measure_iUnion_fintype_le
92       (μ := mu)
93       (s := E))
