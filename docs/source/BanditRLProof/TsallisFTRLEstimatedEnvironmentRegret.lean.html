134 /-- The deterministic FTRL stability sum splits into the initial term and the
135 generated successor terms. -/
136 theorem sum_observedEstimated_stability_eq_initial_add_successor
137     {Env : Type u} {Action : Type v} [DecidableEq Action]
138     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
139     (sample : Env × ((k : Nat) -> Action × Real)) (horizon : Nat) :
140     let estimatedLoss := fun t =>
141       sampledHalfTsallisObservedEstimatedLossAt arms harms eta t sample
142     let p := halfTsallisCumulativeMinimizer arms harms eta estimatedLoss
143     (Finset.range (horizon + 1)).sum (fun t =>
144         FTRL.linearLoss arms (p t) (estimatedLoss t) -
145           FTRL.linearLoss arms (p (t + 1)) (estimatedLoss t)) =
146       sampledHalfTsallisInitialStability arms harms eta sample +
147         (Finset.range horizon).sum (fun n =>
148           sampledHalfTsallisObservedSuccessorStabilityAt
149             arms harms eta n sample) := by
150   dsimp only
151   rw [Finset.sum_range_succ']
152   conv_lhs => rw [add_comm]
153   congr 1
154   · simp [sampledHalfTsallisInitialStability,
155       sampledHalfTsallisProbabilityAtTime,
156       halfTsallisCumulativeMinimizer_observedEstimatedLoss_eq_probabilityAtTime]
157   · apply Finset.sum_congr rfl
158     intro n hn
159     rw [halfTsallisCumulativeMinimizer_observedEstimatedLoss_eq_probabilityAtTime,
160       halfTsallisCumulativeMinimizer_observedEstimatedLoss_eq_probabilityAtTime]
161     simp [sampledHalfTsallisObservedSuccessorStabilityAt,
162       sampledHalfTsallisProbabilityAtTime]

164 /-- Pathwise estimated-loss regret decomposition for time zero followed by
165 `horizon` generated successor rounds. -/
166 theorem sampledHalfTsallis_observedEstimatedRegret_le
167     {Env : Type u} {Action : Type v} [DecidableEq Action]
168     (arms : Finset Action) (harms : arms.Nonempty) (eta : Real)
169     (heta : 0 < eta) (q : Action -> Real)
170     (hq : FTRL.finiteSimplex arms q)
171     (sample : Env × ((k : Nat) -> Action × Real)) (horizon : Nat) :
172     (Finset.range (horizon + 1)).sum (fun t =>
173         FTRL.linearLoss arms
174             (sampledHalfTsallisProbabilityAtTime arms harms eta t sample)
175             (sampledHalfTsallisObservedEstimatedLossAt
176               arms harms eta t sample) -
177           FTRL.linearLoss arms q
178             (sampledHalfTsallisObservedEstimatedLossAt
179               arms harms eta t sample)) <=
180       sampledHalfTsallisInitialStability arms harms eta sample +
181         (Finset.range horizon).sum (fun n =>
182           sampledHalfTsallisObservedSuccessorStabilityAt
183             arms harms eta n sample) +
184         ((powerSum arms (1 / 2 : Real)
185               (initialHalfTsallisDistribution arms harms eta) -
186             powerSum arms (1 / 2 : Real) q) /
187           (1 - (1 / 2 : Real))) / eta := by
188   let estimatedLoss := fun t =>
189     sampledHalfTsallisObservedEstimatedLossAt arms harms eta t sample
190   have hdecomp :=
191     cumulativeLinearLoss_sub_comparator_le_stability_add_powerSumPenalty_half_canonical
192       arms harms eta estimatedLoss q (horizon + 1) heta hq
193   dsimp [estimatedLoss] at hdecomp
194   rw [sum_observedEstimated_stability_eq_initial_add_successor
195     arms harms eta sample horizon] at hdecomp
196   simpa only
197       [halfTsallisCumulativeMinimizer_observedEstimatedLoss_eq_probabilityAtTime]
198     using hdecomp
