23 /-- The refined all-arm half-Tsallis budget is bounded by the square-root
24 mass of the arms other than any distinguished supported arm. -/
25 theorem refinedPotentialStabilityBound_le_two_mul_eta_mul_sqrt_erase_card
26     {History : Type u} {Action : Type v} [DecidableEq Action]
27     (arms : Finset Action) {best : Action} (hbest : best ∈ arms)
28     (eta : Real) (heta : 0 <= eta)
29     (probability : History -> Action -> Real) (history : History)
30     (hprobability : FTRL.finiteSimplex arms (probability history)) :
31     refinedPotentialStabilityBound arms eta probability history <=
32       2 * eta * Real.sqrt ((arms.erase best).card : Real) +
33         2 * eta ^ 2 := by
34   have hall :=
35     sum_sqrt_mul_one_sub_le_two_mul_sum_erase_sqrt
36       arms hbest (probability history) hprobability
37   have herase :=
38     sum_erase_sqrt_le_sqrt_card
39       arms hbest (probability history) hprobability
40   unfold refinedPotentialStabilityBound
41   calc
42     eta * arms.sum (fun action =>
43         Real.sqrt (probability history action) *
44           (1 - probability history action)) +
45         2 * eta ^ 2 <=
46       eta * (2 * (arms.erase best).sum (fun action =>
47         Real.sqrt (probability history action))) +
48         2 * eta ^ 2 := by
49           exact add_le_add (mul_le_mul_of_nonneg_left hall heta) le_rfl
50     _ <= eta * (2 * Real.sqrt ((arms.erase best).card : Real)) +
51         2 * eta ^ 2 := by
52           have htwoEta : 0 <= 2 * eta := mul_nonneg (by norm_num) heta
53           have hscaled :
54               eta * (2 * (arms.erase best).sum (fun action =>
55                   Real.sqrt (probability history action))) <=
56                 eta * (2 * Real.sqrt ((arms.erase best).card : Real)) := by
57             calc
58               eta * (2 * (arms.erase best).sum (fun action =>
59                   Real.sqrt (probability history action))) =
60                   (2 * eta) * (arms.erase best).sum (fun action =>
61                     Real.sqrt (probability history action)) := by ring
62               _ <= (2 * eta) *
63                   Real.sqrt ((arms.erase best).card : Real) :=
64                 mul_le_mul_of_nonneg_left herase htwoEta
65               _ = eta * (2 *
66                   Real.sqrt ((arms.erase best).card : Real)) := by ring
67           exact add_le_add hscaled le_rfl
68     _ = 2 * eta * Real.sqrt ((arms.erase best).card : Real) +
69         2 * eta ^ 2 := by ring
