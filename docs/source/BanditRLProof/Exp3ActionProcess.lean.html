3 /-!
4 # A generated one-round EXP3 action process
5 
6 This module constructs a history-adaptive finite-action Markov kernel from
7 measurable probability coordinates.  Its canonical sample measure is the
8 history law composed with that kernel, so the sampled action has the requested
9 conditional distribution by Mathlib's `condDistrib`/`compProd` uniqueness
10 theorem.  The final wrappers discharge the law premises of the one-round EXP3
11 importance-weighted moment transport.
12 -/

21 /-- Pointwise probability-vector and coordinate-measurability contracts. -/
22 structure MeasurableFiniteActionDistribution
23     {History : Type u} {Action : Type v}
24     [MeasurableSpace History] (arms : Finset Action)
25     (prob : History -> Action -> Real) : Prop where
26   distribution : forall history, FiniteActionDistribution arms (prob history)
27   measurable_prob : forall action, action ∈ arms ->
28     Measurable (fun history => prob history action)

30 /-- The history-adaptive finite action kernel generated by `prob`. -/
31 noncomputable def finiteActionKernel
32     {History : Type u} {Action : Type v}
33     [MeasurableSpace History] [MeasurableSpace Action]
34     [MeasurableSingletonClass Action]
35     (arms : Finset Action) (prob : History -> Action -> Real)
36     (source : MeasurableFiniteActionDistribution arms prob) :
37     Kernel History Action where
38   toFun history := finiteActionMeasure arms (prob history)
39   measurable' := by
40     refine Measure.measurable_of_measurable_coe _ fun event hevent => ?_
41     simp only [finiteActionMeasure, Measure.finset_sum_apply,
42       Measure.smul_apply, smul_eq_mul, Measure.dirac_apply' _ hevent]
43     refine Finset.measurable_sum arms fun action haction => ?_
44     exact (ENNReal.measurable_ofReal.comp
45       (source.measurable_prob action haction)).mul measurable_const

48 theorem finiteActionKernel_apply
49     {History : Type u} {Action : Type v}
50     [MeasurableSpace History] [MeasurableSpace Action]
51     [MeasurableSingletonClass Action]
52     (arms : Finset Action) (prob : History -> Action -> Real)
53     (source : MeasurableFiniteActionDistribution arms prob)
54     (history : History) :

104 /-- History coordinate of the generated one-round process. -/

124 /-- The generated process preserves the supplied history marginal. -/
125 theorem actionProcess_history_map_eq
126     {History : Type u} {Action : Type v}
127     [MeasurableSpace History] [MeasurableSpace Action]
128     [MeasurableSingletonClass Action]
129     (historyMu : Measure History) [IsFiniteMeasure historyMu]
130     (arms : Finset Action) (prob : History -> Action -> Real)
131     (source : MeasurableFiniteActionDistribution arms prob) :
132     (actionProcessMeasure historyMu arms prob source).map
133         actionProcessHistory = historyMu := by
134   change (historyMu ⊗ₘ finiteActionKernel arms prob source).fst = historyMu
135   rw [Measure.fst_compProd]

137 /-- The sampled action has the generated policy as its a.e. conditional law. -/
138 theorem actionProcess_condDistrib_action_ae_eq_finiteActionKernel
139     {History : Type u} {Action : Type v}
140     [MeasurableSpace History] [MeasurableSpace Action]
141     [MeasurableSingletonClass Action]
142     [StandardBorelSpace Action] [Nonempty Action]
143     (historyMu : Measure History) [IsFiniteMeasure historyMu]
144     (arms : Finset Action) (prob : History -> Action -> Real)
145     (source : MeasurableFiniteActionDistribution arms prob) :
146     condDistrib actionProcessAction actionProcessHistory
147         (actionProcessMeasure historyMu arms prob source) =ᵐ[
148       (actionProcessMeasure historyMu arms prob source).map
149         actionProcessHistory]
150       finiteActionKernel arms prob source := by
151   apply (condDistrib_ae_eq_iff_measure_eq_compProd
152     (μ := actionProcessMeasure historyMu arms prob source)
153     actionProcessHistory actionProcessAction_measurable.aemeasurable
154     (finiteActionKernel arms prob source)).2
155   rw [actionProcess_history_map_eq]
156   change (historyMu ⊗ₘ finiteActionKernel arms prob source).map id = _
157   rw [Measure.map_id]

160 theorem finiteActionKernel_ae_eq_finiteActionMeasure
161     {History : Type u} {Action : Type v}
