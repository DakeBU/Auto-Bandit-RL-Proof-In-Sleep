5 # All-horizon Bernstein mixed-square realized EXP3 route
6 
7 The explicit variance-sensitive schedule has its refined `14 * gamma * T`
8 threshold when four horizon inequalities make clipping inactive. Outside that
9 regime this module reuses the compiled almost-sure horizon bound and the strict
10 `T + 1` zero-probability threshold. The fallback covers every positive horizon
11 without claiming a sharp active-clipping, Freedman, or EXP3.P rate.
12 -/

20 /-- The regime in which all four components of the reused variance-sensitive
21 exploration schedule are at most one half. -/
22 def bernsteinSquareLargeHorizonCondition
23     (K T delta : Real) : Prop :=
24   4 * (K * Real.log K) <= T ∧
25     64 * (K ^ 2 * Real.log K ^ 2 * Real.log (4 / delta) / 2) <=
26       T ^ 3 ∧
27     8 * (K * Real.log (4 / delta)) <= T ∧
28     8 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
29       Real.log (4 / delta) <= T
30 
31 /-- All-horizon threshold for the variance-sensitive mixed-square route: use
32 the explicit large-horizon rate in its valid regime and `T + 1` otherwise. -/
33 noncomputable def bernsteinSquareAllHorizonRegretThreshold
34     {Action : Type v} (arms : Finset Action) (horizon : Nat)
35     (delta : Real) : Real := by
36   classical
37   let gamma := bernsteinSquareClippedExplorationRate
38     (arms.card : Real) (horizon : Real) delta
39   exact if bernsteinSquareLargeHorizonCondition
40       (arms.card : Real) (horizon : Real) delta then
41     bernsteinSquareRealizedExplicitThreshold arms gamma horizon delta
42   else
43     (horizon : Real) + 1
