4 # Finite-horizon UCB tail summability wrappers
5 
6 This file records the finite-union/summation layer used by UCB-style bad
7 events: once each arm-time event has a tail bound, the bad-event union over
8 finite arms and a finite horizon is bounded by the corresponding double sum.
9 -/
10 
11 namespace BanditRLProof
12 namespace UCBSummability
13 
14 open MeasureTheory
15 
16 universe u v
17 
18 /-- The union of arm-time bad events over all finite arms and times `< T`. -/
19 def finiteHorizonBadEvent
20     {Omega : Type u} {Arm : Type v}
21     (bad : Arm -> Nat -> Set Omega) (T : Nat) : Set Omega :=
22   ⋃ a, ⋃ t ∈ Finset.range T, bad a t
23 
24 /--
25 Finite-horizon union bound for a UCB-style arm-time bad-event family.
26 
27 No event measurability is required: this is an outer-measure bound inherited
28 from Mathlib's finite-union measure inequality.
29 -/
30 theorem measure_finiteHorizonBadEvent_le_sum
31     {Omega : Type u} [MeasurableSpace Omega]
32     {Arm : Type v} [Fintype Arm]
33     (mu : Measure Omega)
34     (bad : Arm -> Nat -> Set Omega)
35     (T : Nat) :
36     mu (finiteHorizonBadEvent bad T) <=
37       (Finset.univ : Finset Arm).sum
38         (fun a => (Finset.range T).sum (fun t => mu (bad a t))) := by
39   unfold finiteHorizonBadEvent
40   calc
41     mu (⋃ a, ⋃ t ∈ Finset.range T, bad a t)
42         <= (Finset.univ : Finset Arm).sum
43             (fun a => mu (⋃ t ∈ Finset.range T, bad a t)) := by
44           exact ProbabilityUnionBound.measure_iUnion_fintype_le_sum
45             mu (fun a => ⋃ t ∈ Finset.range T, bad a t)
46     _ <= (Finset.univ : Finset Arm).sum
47           (fun a => (Finset.range T).sum (fun t => mu (bad a t))) := by
48           exact Finset.sum_le_sum (fun a _ha =>
49             ProbabilityUnionBound.measure_biUnion_finset_le
50               mu (Finset.range T) (bad a))
51 
52 /--
53 Tail-bound consumer for finite-horizon UCB bad events.
54 
55 The hypothesis `htail` is the per-arm/per-time concentration result; this
56 wrapper only assembles those local bounds into the finite bad-event sum.
57 -/
58 theorem measure_finiteHorizonBadEvent_le_tail_sum
59     {Omega : Type u} [MeasurableSpace Omega]
60     {Arm : Type v} [Fintype Arm]
61     (mu : Measure Omega)
62     (bad : Arm -> Nat -> Set Omega)
63     (tail : Arm -> Nat -> ENNReal)
64     (T : Nat)
65     (htail : forall a t, t < T -> mu (bad a t) <= tail a t) :
66     mu (finiteHorizonBadEvent bad T) <=
67       (Finset.univ : Finset Arm).sum
68         (fun a => (Finset.range T).sum (fun t => tail a t)) := by
69   exact (measure_finiteHorizonBadEvent_le_sum mu bad T).trans
70     (Finset.sum_le_sum (fun a _ha =>
71       Finset.sum_le_sum (fun t ht =>
72         htail a t (Finset.mem_range.mp ht))))
73 
74 end UCBSummability
75 end BanditRLProof
