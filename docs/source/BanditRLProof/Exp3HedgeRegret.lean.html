10 # Deterministic exponential-weights regret
11 
12 This module closes the full-information exponential-weights/Hedge potential
13 argument that sits immediately below EXP3.  It defines weights from cumulative
14 losses, normalizes them on an explicit nonempty finite action set, proves the
15 one-step logarithmic potential inequality, and telescopes it into second-order
16 and `[0,1]` finite-horizon regret bounds.
17 
18 No probability space or importance-weighted estimator is used here.  The EXP3
19 route must separately construct an estimator, prove its conditional
20 unbiasedness and second-moment bound, and then take expectations of this
21 pathwise theorem.
22 -/

29 /-- Cumulative loss of one action before round `t`. -/
30 noncomputable def cumulativeLoss {Action : Type u}
31     (loss : Nat -> Action -> Real) (t : Nat) (a : Action) : Real :=
32   (Finset.range t).sum (fun s => loss s a)
33 
34 /-- Exponential weight generated by the cumulative loss before round `t`. -/
35 noncomputable def weight {Action : Type u}
36     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) (a : Action) : Real :=
37   Real.exp (-eta * cumulativeLoss loss t a)
38 
39 /-- Total exponential weight on an explicit finite action set. -/
40 noncomputable def totalWeight {Action : Type u}
41     (arms : Finset Action) (eta : Real)
42     (loss : Nat -> Action -> Real) (t : Nat) : Real :=
43   arms.sum (weight eta loss t)
44 
45 /-- Normalized exponential weight at round `t`. -/
46 noncomputable def distribution {Action : Type u}
47     (arms : Finset Action) (eta : Real)
48     (loss : Nat -> Action -> Real) (t : Nat) (a : Action) : Real :=
49   weight eta loss t a / totalWeight arms eta loss t
50 
51 /-- Mixed loss incurred by the normalized exponential weights at round `t`. -/
52 noncomputable def mixedLoss {Action : Type u}
53     (arms : Finset Action) (eta : Real)
54     (loss : Nat -> Action -> Real) (t : Nat) : Real :=
55   arms.sum (fun a => distribution arms eta loss t a * loss t a)
56 
57 /-- Mixed second moment of the losses at round `t`. -/
58 noncomputable def mixedSquaredLoss {Action : Type u}
59     (arms : Finset Action) (eta : Real)
60     (loss : Nat -> Action -> Real) (t : Nat) : Real :=
61   arms.sum (fun a => distribution arms eta loss t a * (loss t a) ^ 2)

63 theorem cumulativeLoss_succ {Action : Type u}
64     (loss : Nat -> Action -> Real) (t : Nat) (a : Action) :
65     cumulativeLoss loss (t + 1) a = cumulativeLoss loss t a + loss t a := by
66   simp [cumulativeLoss, Finset.sum_range_succ]

68 theorem weight_pos {Action : Type u}
69     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) (a : Action) :
70     0 < weight eta loss t a := by
71   exact Real.exp_pos _

73 theorem weight_succ {Action : Type u}
74     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) (a : Action) :
75     weight eta loss (t + 1) a =
76       weight eta loss t a * Real.exp (-eta * loss t a) := by
77   rw [weight, cumulativeLoss_succ]
78   rw [show -eta * (cumulativeLoss loss t a + loss t a) =
79       -eta * cumulativeLoss loss t a + -eta * loss t a by ring]
80   rw [Real.exp_add]
81   rfl

83 theorem totalWeight_pos {Action : Type u}
84     (arms : Finset Action) (harms : arms.Nonempty)
85     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) :
86     0 < totalWeight arms eta loss t := by
87   unfold totalWeight
88   refine Finset.sum_pos' (fun a _ha => (weight_pos eta loss t a).le) ?_
89   rcases harms with ⟨a, ha⟩
90   exact ⟨a, ha, weight_pos eta loss t a⟩

92 theorem distribution_nonneg {Action : Type u}
93     (arms : Finset Action) (harms : arms.Nonempty)
94     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) (a : Action) :
95     0 <= distribution arms eta loss t a := by
96   exact div_nonneg (weight_pos eta loss t a).le
97     (totalWeight_pos arms harms eta loss t).le

99 theorem distribution_pos {Action : Type u}
100     (arms : Finset Action) (harms : arms.Nonempty)
101     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) (a : Action) :
102     0 < distribution arms eta loss t a := by
103   exact div_pos (weight_pos eta loss t a)
104     (totalWeight_pos arms harms eta loss t)

106 theorem sum_distribution {Action : Type u}
107     (arms : Finset Action) (harms : arms.Nonempty)
108     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) :
109     arms.sum (distribution arms eta loss t) = 1 := by
110   rw [show arms.sum (distribution arms eta loss t) =
111       arms.sum (weight eta loss t) / totalWeight arms eta loss t by
112     simp only [distribution]
113     rw [Finset.sum_div]]
114   rw [show arms.sum (weight eta loss t) = totalWeight arms eta loss t by rfl]
115   exact div_self (ne_of_gt (totalWeight_pos arms harms eta loss t))

117 theorem totalWeight_zero {Action : Type u}
118     (arms : Finset Action) (eta : Real) (loss : Nat -> Action -> Real) :
119     totalWeight arms eta loss 0 = arms.card := by
120   simp [totalWeight, weight, cumulativeLoss]

122 /-- Global quadratic upper bound for a nonnegative exponential argument. -/
123 theorem exp_neg_le_one_sub_add_sq {x : Real} (hx : 0 <= x) :
124     Real.exp (-x) <= 1 - x + x ^ 2 := by
125   by_cases hx_le : x <= 1
126   · have hneg : -x <= 0 := neg_nonpos.mpr hx
127     have habs : |-x| <= 1 := by simpa [abs_of_nonpos hneg]
128     have hrem := Real.abs_exp_sub_one_sub_id_le habs
129     have hrem_le : Real.exp (-x) - 1 - (-x) <= (-x) ^ 2 :=
130       (le_abs_self _).trans hrem
131     nlinarith [hrem_le]
132   · have hexp : Real.exp (-x) <= 1 := by
133       simpa only [Real.exp_zero] using
134         Real.exp_le_exp.mpr (neg_nonpos.mpr hx)
135     have hone : 1 <= 1 - x + x ^ 2 := by
136       nlinarith [lt_of_not_ge hx_le]
137     exact hexp.trans hone

139 /-- Quadratic exponential bound for arbitrary nonnegative learning rates and losses. -/
140 theorem exp_neg_mul_le_one_sub_add_sq_of_nonneg
141     {eta ell : Real} (heta : 0 <= eta) (hell : 0 <= ell) :
142     Real.exp (-eta * ell) <= 1 - eta * ell + eta ^ 2 * ell ^ 2 := by
143   have h := exp_neg_le_one_sub_add_sq (mul_nonneg heta hell)
144   simpa [mul_pow] using h

146 /-- Backwards-compatible bounded-loss specialization of the global bound. -/
147 theorem exp_neg_mul_le_one_sub_add_sq
148     {eta ell : Real} (heta : 0 <= eta) (_heta_le : eta <= 1)
149     (hell : 0 <= ell) (_hell_le : ell <= 1) :
150     Real.exp (-eta * ell) <= 1 - eta * ell + eta ^ 2 * ell ^ 2 :=
151   exp_neg_mul_le_one_sub_add_sq_of_nonneg heta hell

153 theorem totalWeight_succ_div_eq_sum_distribution_mul_exp
154     {Action : Type u}
155     (arms : Finset Action)
156     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat) :
157     totalWeight arms eta loss (t + 1) /
158         totalWeight arms eta loss t =
159       arms.sum (fun a =>
160         distribution arms eta loss t a * Real.exp (-eta * loss t a)) := by
161   rw [totalWeight]
162   simp_rw [weight_succ]
163   rw [Finset.sum_div]
164   apply Finset.sum_congr rfl
165   intro a _ha
166   simp only [distribution, totalWeight]
167   ring

169 theorem totalWeight_succ_div_le_of_nonneg
170     {Action : Type u}
171     (arms : Finset Action) (harms : arms.Nonempty)
172     (eta : Real) (heta : 0 <= eta)
173     (loss : Nat -> Action -> Real) (t : Nat)
174     (hloss : forall a, a ∈ arms -> 0 <= loss t a) :
175     totalWeight arms eta loss (t + 1) /
176         totalWeight arms eta loss t <=
177       1 - eta * mixedLoss arms eta loss t +
178         eta ^ 2 * mixedSquaredLoss arms eta loss t := by
179   rw [totalWeight_succ_div_eq_sum_distribution_mul_exp arms]
180   calc
181     arms.sum (fun a =>
182         distribution arms eta loss t a * Real.exp (-eta * loss t a)) <=
183       arms.sum (fun a =>
184         distribution arms eta loss t a *
185           (1 - eta * loss t a + eta ^ 2 * (loss t a) ^ 2)) := by
186         apply Finset.sum_le_sum
187         intro a ha
188         exact mul_le_mul_of_nonneg_left
189           (exp_neg_mul_le_one_sub_add_sq_of_nonneg heta (hloss a ha))
190           (distribution_nonneg arms harms eta loss t a)
191     _ = arms.sum (distribution arms eta loss t) -
192           eta * mixedLoss arms eta loss t +
193           eta ^ 2 * mixedSquaredLoss arms eta loss t := by
194         simp only [mixedLoss, mixedSquaredLoss]
195         calc
196           arms.sum (fun a =>
197               distribution arms eta loss t a *
198                 (1 - eta * loss t a + eta ^ 2 * (loss t a) ^ 2)) =
199             arms.sum (fun a =>
200               (distribution arms eta loss t a -
201                 eta * (distribution arms eta loss t a * loss t a)) +
202                 eta ^ 2 *
203                   (distribution arms eta loss t a * (loss t a) ^ 2)) := by
204               apply Finset.sum_congr rfl
205               intro a _ha
206               ring
207           _ = arms.sum (distribution arms eta loss t) -
208                 eta * arms.sum (fun a =>
209                   distribution arms eta loss t a * loss t a) +
210                 eta ^ 2 * arms.sum (fun a =>
211                   distribution arms eta loss t a * (loss t a) ^ 2) := by
212               rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
213                 ← Finset.mul_sum, ← Finset.mul_sum]
214     _ = 1 - eta * mixedLoss arms eta loss t +
215           eta ^ 2 * mixedSquaredLoss arms eta loss t := by
216         rw [sum_distribution arms harms]

231 theorem log_totalWeight_succ_sub_le_of_nonneg
232     {Action : Type u}
233     (arms : Finset Action) (harms : arms.Nonempty)
234     (eta : Real) (heta : 0 <= eta)

262 theorem mixedSquaredLoss_le_one
263     {Action : Type u}
264     (arms : Finset Action) (harms : arms.Nonempty)
265     (eta : Real) (loss : Nat -> Action -> Real) (t : Nat)
266     (hloss : forall a, a ∈ arms -> 0 <= loss t a ∧ loss t a <= 1) :
267     mixedSquaredLoss arms eta loss t <= 1 := by
268   calc
269     mixedSquaredLoss arms eta loss t <=
270         arms.sum (distribution arms eta loss t) := by
271       unfold mixedSquaredLoss
272       apply Finset.sum_le_sum
273       intro a ha
274       have hsquare : (loss t a) ^ 2 <= 1 := by
275         nlinarith [(hloss a ha).1, (hloss a ha).2]
276       simpa using mul_le_mul_of_nonneg_left hsquare
277         (distribution_nonneg arms harms eta loss t a)
278     _ = 1 := sum_distribution arms harms eta loss t

280 /--
281 Second-order finite-horizon Hedge regret bound against one comparator action.
282 -/
283 theorem hedge_regret_le_log_card_div_add_eta_mul_mixedSquaredLoss_of_nonneg
284     {Action : Type u}
285     (arms : Finset Action) (harms : arms.Nonempty)
286     (eta : Real) (heta : 0 < eta)
287     (loss : Nat -> Action -> Real) (T : Nat)
288     (hloss : forall t, t < T -> forall a, a ∈ arms ->
289       0 <= loss t a)
290     (comparator : Action) (hcomparator : comparator ∈ arms) :
291     (Finset.range T).sum (fun t => mixedLoss arms eta loss t) -
292         cumulativeLoss loss T comparator <=
293       Real.log arms.card / eta +
294         eta * (Finset.range T).sum
295           (fun t => mixedSquaredLoss arms eta loss t) := by
296   have hlog_upper :
297       Real.log (totalWeight arms eta loss T) -
298           Real.log (totalWeight arms eta loss 0) <=
299         -eta * (Finset.range T).sum
300             (fun t => mixedLoss arms eta loss t) +
301           eta ^ 2 * (Finset.range T).sum
302             (fun t => mixedSquaredLoss arms eta loss t) := by
303     calc
304       Real.log (totalWeight arms eta loss T) -
305           Real.log (totalWeight arms eta loss 0) =
306         (Finset.range T).sum (fun t =>
307           Real.log (totalWeight arms eta loss (t + 1)) -
308             Real.log (totalWeight arms eta loss t)) := by
309               symm
310               exact Exp3Potential.sum_range_forward_difference
311                 (fun t => Real.log (totalWeight arms eta loss t)) T
312       _ <= (Finset.range T).sum (fun t =>
313           -eta * mixedLoss arms eta loss t +
314             eta ^ 2 * mixedSquaredLoss arms eta loss t) := by
315               apply Finset.sum_le_sum
316               intro t ht
317               exact log_totalWeight_succ_sub_le_of_nonneg arms harms eta heta.le
318                 loss t (hloss t (Finset.mem_range.mp ht))
319       _ = -eta * (Finset.range T).sum
320             (fun t => mixedLoss arms eta loss t) +
321           eta ^ 2 * (Finset.range T).sum
322             (fun t => mixedSquaredLoss arms eta loss t) := by
323               rw [Finset.sum_add_distrib, ← Finset.mul_sum,
324                 ← Finset.mul_sum]
325   have hweight_le :
326       weight eta loss T comparator <= totalWeight arms eta loss T := by
327     exact Finset.single_le_sum
328       (fun a _ha => (weight_pos eta loss T a).le) hcomparator
329   have hcomparator_log :
330       -eta * cumulativeLoss loss T comparator <=
331         Real.log (totalWeight arms eta loss T) := by
332     have hlog := Real.log_le_log (weight_pos eta loss T comparator) hweight_le
333     simpa [weight] using hlog
334   have hinitial :
335       Real.log (totalWeight arms eta loss 0) = Real.log arms.card := by
336     rw [totalWeight_zero]
337   rw [hinitial] at hlog_upper
338   have hscaled :
339       eta * ((Finset.range T).sum (fun t => mixedLoss arms eta loss t) -
340           cumulativeLoss loss T comparator) <=
341         Real.log arms.card +
342           eta ^ 2 * (Finset.range T).sum
343             (fun t => mixedSquaredLoss arms eta loss t) := by
344     linarith
345   calc
346     (Finset.range T).sum (fun t => mixedLoss arms eta loss t) -
347         cumulativeLoss loss T comparator <=
348       (Real.log arms.card +
349           eta ^ 2 * (Finset.range T).sum
350             (fun t => mixedSquaredLoss arms eta loss t)) / eta := by
351         apply (le_div_iff₀ heta).2
352         nlinarith
353     _ = Real.log arms.card / eta +
354         eta * (Finset.range T).sum
355           (fun t => mixedSquaredLoss arms eta loss t) := by
356       field_simp

376 /--
377 Finite-horizon Hedge regret for losses in `[0,1]`.
378 
379 This is the deterministic full-information theorem consumed by the future
380 importance-weighted EXP3 route.
381 -/
382 theorem hedge_regret_le_log_card_div_add_eta_mul_horizon
383     {Action : Type u}
384     (arms : Finset Action) (harms : arms.Nonempty)
385     (eta : Real) (heta : 0 < eta) (heta_le : eta <= 1)
386     (loss : Nat -> Action -> Real) (T : Nat)
387     (hloss : forall t, t < T -> forall a, a ∈ arms ->
388       0 <= loss t a ∧ loss t a <= 1)
389     (comparator : Action) (hcomparator : comparator ∈ arms) :
390     (Finset.range T).sum (fun t => mixedLoss arms eta loss t) -
391         cumulativeLoss loss T comparator <=
392       Real.log arms.card / eta + eta * T := by
393   have hsecond :=
394     hedge_regret_le_log_card_div_add_eta_mul_mixedSquaredLoss
395       arms harms eta heta heta_le loss T hloss comparator hcomparator
396   have hsquare_sum :
397       (Finset.range T).sum
398           (fun t => mixedSquaredLoss arms eta loss t) <= (T : Real) := by
399     calc
400       (Finset.range T).sum
401           (fun t => mixedSquaredLoss arms eta loss t) <=
402         (Finset.range T).sum (fun _t => (1 : Real)) := by
403           apply Finset.sum_le_sum
404           intro t ht
405           exact mixedSquaredLoss_le_one arms harms eta loss t
406             (hloss t (Finset.mem_range.mp ht))
407       _ = (T : Real) := by simp
408   exact hsecond.trans
409     (add_le_add (le_refl _) (mul_le_mul_of_nonneg_left hsquare_sum heta.le))
