3 /-!
4 # Parameter tuning for the realized EXP3 Bernstein-confidence route
5 
6 The compiled high-probability theorem retains a pathwise estimator-square
7 term. Consequently, the expected-regret choice `eta = gamma / K` leaves a
8 linear term. This module instead balances the Hedge terms with
9 `eta = sqrt (log K * gamma / (T * K))` and records the cubic exploration
10 conditions required by the current Bernstein confidence radii.
11 -/

19 /-- Learning rate balancing the entropy and pathwise estimator-square terms
20 for the current high-probability budget. -/
21 noncomputable def bernsteinHighProbabilityLearningRate
22     (K T gamma : Real) : Real :=
23   Real.sqrt (Real.log K * gamma / (T * K))

25 /-- A cubic exploration budget makes one current Bernstein confidence radius
26 at most `3 * gamma * T`. -/
27 theorem bernsteinConfidenceRadius_le_three_mul_gamma_mul_horizon
28     (K T budget gamma : Real)
29     (hK : 0 < K) (hT : 0 < T) (_hbudget : 0 <= budget)
30     (hgamma_pos : 0 < gamma) (hgamma_le_one : gamma <= 1)
31     (hcubic : K * budget <= gamma ^ 3 * T) :
32     2 * Real.sqrt (T * budget / (gamma / K)) +
33         budget / (gamma / K) <=
34       3 * gamma * T := by
35   have hK_ne : K ≠ 0 := ne_of_gt hK
36   have hgamma_ne : gamma ≠ 0 := ne_of_gt hgamma_pos
37   have hepsilon_pos : 0 < gamma / K := div_pos hgamma_pos hK
38   have hnum : T * (K * budget) <= T * (gamma ^ 3 * T) :=
39     mul_le_mul_of_nonneg_left hcubic hT.le
40   have hradicand : T * budget / (gamma / K) <= (gamma * T) ^ 2 := by
41     calc
42       T * budget / (gamma / K) = T * (K * budget) / gamma := by
43         field_simp [hK_ne, hgamma_ne]
44       _ <= T * (gamma ^ 3 * T) / gamma :=
45         (div_le_div_iff_of_pos_right hgamma_pos).2 hnum
46       _ = (gamma * T) ^ 2 := by
47         field_simp [hgamma_ne]
48   have hsqrt :
49       Real.sqrt (T * budget / (gamma / K)) <= gamma * T := by
50     rw [Real.sqrt_le_iff]
51     exact ⟨mul_nonneg hgamma_pos.le hT.le, hradicand⟩
52   have hlinear : budget / (gamma / K) <= gamma * T := by
53     have hlinear_sq : budget / (gamma / K) <= gamma ^ 2 * T := by
54       calc
55         budget / (gamma / K) = K * budget / gamma := by
56           field_simp [hK_ne, hgamma_ne]
57         _ <= gamma ^ 2 * T := by
58           rw [div_le_iff₀ hgamma_pos]
59           nlinarith [hcubic]
60     have hgamma_sq_le : gamma ^ 2 * T <= gamma * T := by
61       have hgamma_sq_le_gamma : gamma ^ 2 <= gamma := by
62         nlinarith [mul_nonneg hgamma_pos.le (sub_nonneg.mpr hgamma_le_one)]
63       exact mul_le_mul_of_nonneg_right hgamma_sq_le_gamma hT.le
64     exact hlinear_sq.trans hgamma_sq_le
65   linarith

67 /-- The bounded realized-loss deviation radius is at most `gamma * T` under
68 its matching quadratic exploration budget. -/
69 theorem realizedDeviationRadius_le_mul_gamma_mul_horizon
70     (T budget variance gamma : Real)
71     (hT : 0 < T) (_hbudget : 0 <= budget) (_hvariance : 0 <= variance)
72     (hgamma_pos : 0 < gamma)
73     (hquadratic : 2 * variance * budget <= gamma ^ 2 * T) :
74     Real.sqrt (2 * (T * variance) * budget) <= gamma * T := by
75   rw [Real.sqrt_le_iff]
76   constructor
77   · exact mul_nonneg hgamma_pos.le hT.le
78   have hmul := mul_le_mul_of_nonneg_left hquadratic hT.le
79   nlinarith

81 /-- The tuned learning rate is positive in the nondegenerate finite-arm,
82 positive-horizon regime. -/
83 theorem bernsteinHighProbabilityLearningRate_pos
84     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
85     (hgamma_pos : 0 < gamma) :
86     0 < bernsteinHighProbabilityLearningRate K T gamma := by
87   apply Real.sqrt_pos.2
88   exact div_pos (mul_pos (Real.log_pos hK_one) hgamma_pos)
89     (mul_pos hT (lt_trans zero_lt_one hK_one))

91 /-- Squared balance identity for the tuned learning rate. -/
92 theorem bernsteinHighProbabilityLearningRate_sq_mul
93     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
94     (hgamma_pos : 0 < gamma) :
95     bernsteinHighProbabilityLearningRate K T gamma ^ 2 * (T * K) =
96       Real.log K * gamma := by
97   rw [bernsteinHighProbabilityLearningRate, Real.sq_sqrt]
98   · field_simp [ne_of_gt hT, ne_of_gt (lt_trans zero_lt_one hK_one)]
99   · exact (div_nonneg
100       (mul_nonneg (Real.log_pos hK_one).le hgamma_pos.le)
101       (mul_nonneg hT.le (le_of_lt (lt_trans zero_lt_one hK_one))))

103 /-- The cubic exploration contract places the tuned learning rate below the
104 scale `gamma ^ 2 / K`. -/
105 theorem bernsteinHighProbabilityLearningRate_le_sq_div
106     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
107     (hgamma_pos : 0 < gamma)
108     (hcubic_log : K * Real.log K <= gamma ^ 3 * T) :
109     bernsteinHighProbabilityLearningRate K T gamma <= gamma ^ 2 / K := by
110   have hK : 0 < K := lt_trans zero_lt_one hK_one
111   have hmul := mul_le_mul_of_nonneg_left hcubic_log
112     (div_nonneg hgamma_pos.le hK.le)
113   rw [bernsteinHighProbabilityLearningRate, Real.sqrt_le_iff]
114   constructor
115   · exact div_nonneg (sq_nonneg gamma) hK.le
116   · rw [div_le_iff₀ (mul_pos hT hK)]
117     calc
118       Real.log K * gamma = (gamma / K) * (K * Real.log K) := by
119         field_simp [ne_of_gt hK]
120       _ <= (gamma / K) * (gamma ^ 3 * T) := hmul
121       _ = (gamma ^ 2 / K) ^ 2 * (T * K) := by
122         field_simp [ne_of_gt hK]
123 
124 /-- The entropy contribution is at most `gamma * T` under the cubic arm-log
125 budget. -/
126 theorem bernsteinEntropyBudget_le_mul_gamma_mul_horizon
127     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
128     (hgamma_pos : 0 < gamma)
129     (hcubic_log : K * Real.log K <= gamma ^ 3 * T) :
130     Real.log K / bernsteinHighProbabilityLearningRate K T gamma <=
131       gamma * T := by
132   let eta := bernsteinHighProbabilityLearningRate K T gamma
133   have hK : 0 < K := lt_trans zero_lt_one hK_one
134   have heta : 0 < eta :=
135     bernsteinHighProbabilityLearningRate_pos K T gamma hK_one hT hgamma_pos
136   have heta_le : eta <= gamma ^ 2 / K :=
137     bernsteinHighProbabilityLearningRate_le_sq_div
138       K T gamma hK_one hT hgamma_pos hcubic_log
139   have hbalance : eta ^ 2 * (T * K) = Real.log K * gamma :=
140     bernsteinHighProbabilityLearningRate_sq_mul
141       K T gamma hK_one hT hgamma_pos
142   have heq : Real.log K / eta = eta * T * K / gamma := by
143     field_simp [ne_of_gt heta, ne_of_gt hgamma_pos]
144     nlinarith [hbalance]
145   rw [heq]
146   calc
147     eta * T * K / gamma <= (gamma ^ 2 / K) * T * K / gamma := by
148       exact div_le_div_of_nonneg_right
149         (mul_le_mul_of_nonneg_right
150           (mul_le_mul_of_nonneg_right heta_le hT.le) hK.le)
151         hgamma_pos.le
152     _ = gamma * T := by
153       field_simp [ne_of_gt hK, ne_of_gt hgamma_pos]
154 
155 /-- Before the stability factor `1 / (1 - gamma)`, the pathwise square term
156 has the same tuned scale as the entropy term. -/
157 theorem bernsteinUnscaledSquareBudget_le_mul_gamma_mul_horizon
158     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
159     (hgamma_pos : 0 < gamma)
160     (hcubic_log : K * Real.log K <= gamma ^ 3 * T) :
161     bernsteinHighProbabilityLearningRate K T gamma *
162         (T * (1 / (gamma / K))) <=
163       gamma * T := by
164   have hK : 0 < K := lt_trans zero_lt_one hK_one
165   have heta_le := bernsteinHighProbabilityLearningRate_le_sq_div
166     K T gamma hK_one hT hgamma_pos hcubic_log
167   calc
168     bernsteinHighProbabilityLearningRate K T gamma *
169         (T * (1 / (gamma / K))) =
170         bernsteinHighProbabilityLearningRate K T gamma * T * K / gamma := by
171       field_simp [ne_of_gt hK, ne_of_gt hgamma_pos]
172     _ <= (gamma ^ 2 / K) * T * K / gamma := by
173       exact div_le_div_of_nonneg_right
174         (mul_le_mul_of_nonneg_right
175           (mul_le_mul_of_nonneg_right heta_le hT.le) hK.le)
176         hgamma_pos.le
177     _ = gamma * T := by
178       field_simp [ne_of_gt hK, ne_of_gt hgamma_pos]

180 /-- The entropy and pathwise square terms together cost at most
181 `3 * gamma * T` when `gamma <= 1 / 2`. -/
182 theorem bernsteinHedgeBudget_le_three_mul_gamma_mul_horizon
183     (K T gamma : Real) (hK_one : 1 < K) (hT : 0 < T)
184     (hgamma_pos : 0 < gamma) (hgamma_le_half : gamma <= 1 / 2)
185     (hcubic_log : K * Real.log K <= gamma ^ 3 * T) :
186     Real.log K / bernsteinHighProbabilityLearningRate K T gamma +
187         (bernsteinHighProbabilityLearningRate K T gamma *
188           (1 / (1 - gamma))) * (T * (1 / (gamma / K))) <=
189       3 * gamma * T := by
190   have hden : 0 < 1 - gamma := by linarith
191   have hentropy := bernsteinEntropyBudget_le_mul_gamma_mul_horizon
192     K T gamma hK_one hT hgamma_pos hcubic_log
193   have hraw := bernsteinUnscaledSquareBudget_le_mul_gamma_mul_horizon
194     K T gamma hK_one hT hgamma_pos hcubic_log
195   have hgammaT_nonneg : 0 <= gamma * T :=
196     mul_nonneg hgamma_pos.le hT.le
197   have hfactor : 1 <= 2 * (1 - gamma) := by linarith
198   have hstable :
199       (bernsteinHighProbabilityLearningRate K T gamma *
200           (1 / (1 - gamma))) * (T * (1 / (gamma / K))) <=
201         2 * gamma * T := by
202     rw [show
203       (bernsteinHighProbabilityLearningRate K T gamma *
204           (1 / (1 - gamma))) * (T * (1 / (gamma / K))) =
205         (bernsteinHighProbabilityLearningRate K T gamma *
206           (T * (1 / (gamma / K)))) / (1 - gamma) by ring]
207     rw [div_le_iff₀ hden]
208     calc
209       bernsteinHighProbabilityLearningRate K T gamma *
210           (T * (1 / (gamma / K))) <= gamma * T := hraw
211       _ <= (gamma * T) * (2 * (1 - gamma)) :=
212         by simpa using mul_le_mul_of_nonneg_left hfactor hgammaT_nonneg
213       _ = 2 * gamma * T * (1 - gamma) := by ring
214   linarith

223 /-- Under the explicit cubic and quadratic dominance contracts, the complete
224 three-event realized Bernstein budget is at most `11 * gamma * T`. -/
225 theorem sampledPredictableBernsteinRealizedHighProbabilityRegretBudget_le_eleven_mul
226     {Action : Type v} [DecidableEq Action]
227     (arms : Finset Action) (hcard_two : 2 <= arms.card)
228     (horizon : Nat) (hhorizon : 0 < horizon)
229     (gamma delta : Real) (hgamma_pos : 0 < gamma)
230     (hgamma_le_half : gamma <= 1 / 2)
231     (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
232     (hcubic_log :
233       (arms.card : Real) * Real.log (arms.card : Real) <=
234         gamma ^ 3 * (horizon : Real))
235     (hcubic_confidence :
236       (arms.card : Real) * Real.log (3 / delta) <=
237         gamma ^ 3 * (horizon : Real))
238     (hquadratic_realized :
239       2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
240           Real.log (3 / delta) <=
241         gamma ^ 2 * (horizon : Real)) :
242     sampledPredictableBernsteinRealizedHighProbabilityRegretBudget
243         arms
244         (bernsteinHighProbabilityLearningRate
245           (arms.card : Real) (horizon : Real) gamma)
246         gamma horizon (delta / 3) <=
247       11 * gamma * (horizon : Real) := by
248   let K : Real := arms.card
249   let T : Real := horizon
250   let budget : Real := Real.log (3 / delta)
251   let variance : Real :=
252     ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real)
253   have hK_one : 1 < K := by
254     dsimp [K]
255     exact_mod_cast hcard_two
256   have hK : 0 < K := lt_trans zero_lt_one hK_one
257   have hT : 0 < T := by
258     dsimp [T]
259     exact_mod_cast hhorizon
260   have hone_lt_three_div : 1 < 3 / delta := by
261     rw [lt_div_iff₀ hdelta]
262     nlinarith
263   have hbudget_pos : 0 < budget := Real.log_pos hone_lt_three_div
264   have hlog_third : Real.log (1 / (delta / 3)) = budget := by
265     dsimp [budget]
266     exact log_one_div_third_eq_log_three_div delta hdelta
267   have hmax_third : max (Real.log (1 / (delta / 3))) 0 = budget := by
268     rw [hlog_third]
269     exact max_eq_left hbudget_pos.le
270   have hhedge :
271       Real.log K / bernsteinHighProbabilityLearningRate K T gamma +
272           (bernsteinHighProbabilityLearningRate K T gamma *
273             (1 / (1 - gamma))) * (T * (1 / (gamma / K))) <=
274         3 * gamma * T :=
275     bernsteinHedgeBudget_le_three_mul_gamma_mul_horizon
276       K T gamma hK_one hT hgamma_pos hgamma_le_half (by
277         simpa [K, T] using hcubic_log)
278   have hconfidence :
279       2 * Real.sqrt (T * budget / (gamma / K)) +
280           budget / (gamma / K) <=
281         3 * gamma * T :=
282     bernsteinConfidenceRadius_le_three_mul_gamma_mul_horizon
283       K T budget gamma hK hT hbudget_pos.le hgamma_pos (by linarith)
284         (by simpa [K, T, budget] using hcubic_confidence)
285   have hvariance_pos : 0 < variance := by
286     simpa [variance] using intervalVarianceProxy_zero_one_pos
287   have hrealized :
288       Real.sqrt (2 * (T * variance) * budget) <= gamma * T :=
289     realizedDeviationRadius_le_mul_gamma_mul_horizon
290       T budget variance gamma hT hbudget_pos.le hvariance_pos.le hgamma_pos
291         (by simpa [T, budget, variance] using hquadratic_realized)
292   dsimp [sampledPredictableBernsteinRealizedHighProbabilityRegretBudget,
293     sampledPredictableBernsteinHighProbabilityRegretBudget,
294     sampledPurePredictableMinusObservedBernsteinConfidenceRadius,
295     sampledComparatorEstimatorBernsteinConfidenceRadius,
296     sampledPredictableRealizedDeviationConfidenceRadius]
297   rw [hmax_third, hlog_third]
298   change
299     Real.log K / bernsteinHighProbabilityLearningRate K T gamma +
300           (bernsteinHighProbabilityLearningRate K T gamma *
301             (1 / (1 - gamma))) * (T * (1 / (gamma / K))) +
302         gamma * T +
303         (2 * Real.sqrt (T * budget / (gamma / K)) +
304           budget / (gamma / K)) +
305         (2 * Real.sqrt (T * budget / (gamma / K)) +
306           budget / (gamma / K)) +
307         Real.sqrt (2 * (T * variance) * budget) <=
308       11 * gamma * T
309   linarith

311 /-- Tuned generated realized-regret tail with an explicit `11 * gamma * T`
312 threshold. The current confidence route yields a `T^(2/3)`-type contract:
313 the arm entropy and both importance-weighted confidence budgets must be
314 dominated by `gamma ^ 3 * T`, while the bounded realized deviation uses the
315 displayed quadratic contract. -/
316 theorem sampledPredictable_tunedBernsteinRealizedHighProbabilityRegret_tail
317     {Env : Type u} {Action : Type v}
318     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
319     [MeasurableSpace Action] [MeasurableSingletonClass Action]
320     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
321     (prior : Measure Env) [IsProbabilityMeasure prior]
322     (arms : Finset Action) (harms : arms.Nonempty)
323     (hcard_two : 2 <= arms.card)
324     (gamma : Real) (hgamma_pos : 0 < gamma)
325     (hgamma_le_half : gamma <= 1 / 2)
326     (loss : PredictableLossVector Env Action)
327     (comparator : Action) (hcomparator : comparator ∈ arms)
328     (horizon : Nat) (hhorizon : 0 < horizon)
329     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
330     (hcubic_log :
331       (arms.card : Real) * Real.log (arms.card : Real) <=
332         gamma ^ 3 * (horizon : Real))
333     (hcubic_confidence :
334       (arms.card : Real) * Real.log (3 / delta) <=
335         gamma ^ 3 * (horizon : Real))
336     (hquadratic_realized :
337       2 * ((Concentration.intervalVarianceProxy 0 1 : NNReal) : Real) *
338           Real.log (3 / delta) <=
339         gamma ^ 2 * (horizon : Real)) :
340     let eta := bernsteinHighProbabilityLearningRate
341       (arms.card : Real) (horizon : Real) gamma
342     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
343       eta gamma hgamma_pos.le (by linarith : gamma <= 1) loss.environment
344     mu {sample |
345         11 * gamma * (horizon : Real) <=
346           (Finset.range horizon).sum (fun t =>
347               sampledTrajectoryRealizedLossAt t sample) -
348             (Finset.range horizon).sum (fun t =>
349               predictableLossAt loss t sample comparator)} <=
350       ENNReal.ofReal delta := by
351   dsimp only
352   have hK_one : 1 < (arms.card : Real) := by
353     exact_mod_cast hcard_two
354   have hT : 0 < (horizon : Real) := by
355     exact_mod_cast hhorizon
356   have heta :
357       0 < bernsteinHighProbabilityLearningRate
358         (arms.card : Real) (horizon : Real) gamma :=
359     bernsteinHighProbabilityLearningRate_pos
360       (arms.card : Real) (horizon : Real) gamma hK_one hT hgamma_pos
361   have hgamma_lt_one : gamma < 1 := by linarith
362   have hbudget :=
363     sampledPredictableBernsteinRealizedHighProbabilityRegretBudget_le_eleven_mul
364       arms hcard_two horizon hhorizon gamma delta hgamma_pos hgamma_le_half
365         hdelta hdelta_le_one hcubic_log hcubic_confidence hquadratic_realized
366   have htail :=
367     sampledPredictable_bernsteinRealizedHighProbabilityRegret_tail_total_delta
368       prior arms harms
369         (bernsteinHighProbabilityLearningRate
370           (arms.card : Real) (horizon : Real) gamma)
371         gamma heta hgamma_pos hgamma_lt_one loss comparator hcomparator
372         horizon hhorizon delta hdelta
373   dsimp only at htail ⊢
374   calc
375     (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
376         (bernsteinHighProbabilityLearningRate
377           (arms.card : Real) (horizon : Real) gamma)
378         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
379         {sample |
380           11 * gamma * (horizon : Real) <=
381             (Finset.range horizon).sum (fun t =>
382                 sampledTrajectoryRealizedLossAt t sample) -
383               (Finset.range horizon).sum (fun t =>
384                 predictableLossAt loss t sample comparator)} <=
385       (prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
386         (bernsteinHighProbabilityLearningRate
387           (arms.card : Real) (horizon : Real) gamma)
388         gamma hgamma_pos.le hgamma_lt_one.le loss.environment)
389         {sample |
390           sampledPredictableBernsteinRealizedHighProbabilityRegretBudget
391               arms
392                 (bernsteinHighProbabilityLearningRate
393                   (arms.card : Real) (horizon : Real) gamma)
394               gamma horizon (delta / 3) <=
395             (Finset.range horizon).sum (fun t =>
396                 sampledTrajectoryRealizedLossAt t sample) -
397               (Finset.range horizon).sum (fun t =>
398                 predictableLossAt loss t sample comparator)} := by
399         apply measure_mono
400         intro sample hsample
