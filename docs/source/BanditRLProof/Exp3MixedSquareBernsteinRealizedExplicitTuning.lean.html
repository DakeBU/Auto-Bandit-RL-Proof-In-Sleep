4 /-!
5 # Explicit exploration tuning for Bernstein-square realized EXP3
6 
7 The existing four-scale clipped exploration schedule is strong enough for the
8 variance-sensitive mixed-square radius. Its sixth-power contract controls the
9 new square-root term when `gamma <= 1/2`, while its arm and confidence
10 contracts jointly control the linear `log_+ / epsilon` term. Thus the complete
11 tuned threshold remains bounded by `14 * gamma * T`.
12 
13 This is still deterministic fixed-tilt control, not random predictable
14 quadratic variation, a general Freedman theorem, or ideal EXP3.P.
15 -/

34 /-- Under the existing arm, sixth-power mixed, and confidence contracts, the
35 log-weighted Bernstein mixed-square radius is at most `3 * gamma^2 * T^2`.
36 The first two copies control the square-root term; the third controls the
37 linear `log_+ / epsilon` correction. -/
38 theorem log_mul_sampledMixedSquaredBernsteinConfidenceRadius_le_three_mul_sq_mul_horizon_sq
39     {Action : Type v} (arms : Finset Action) (hcard_two : 2 <= arms.card)
40     (gamma : Real) (hgamma_pos : 0 < gamma)
41     (hgamma_le_half : gamma <= 1 / 2)
42     (horizon : Nat) (hhorizon : 0 < horizon)
43     (delta : Real) (hdelta : 0 < delta) (hdelta_le_one : delta <= 1)
44     (hbase :
45       (arms.card : Real) * Real.log (arms.card : Real) <=
46         gamma ^ 2 * (horizon : Real))
47     (hmixed :
48       (arms.card : Real) ^ 2 * Real.log (arms.card : Real) ^ 2 *
49             Real.log (4 / delta) / 2 <=
50         gamma ^ 6 * (horizon : Real) ^ 3)
51     (hconfidence :
52       (arms.card : Real) * Real.log (4 / delta) <=
53         gamma ^ 3 * (horizon : Real)) :
54     Real.log (arms.card : Real) *
55         sampledMixedSquaredBernsteinConfidenceRadius
56           arms gamma horizon (delta / 4) <=
57       3 * gamma ^ 2 * (horizon : Real) ^ 2 := by
58   let K : Real := arms.card
59   let T : Real := horizon
60   let L : Real := Real.log (4 / delta)
61   let variance := sampledMixedSquaredBernsteinVarianceCoefficient arms gamma
62   have hK_one : 1 < K := by
63     dsimp [K]
64     exact_mod_cast hcard_two
65   have hK : 0 < K := lt_trans zero_lt_one hK_one
66   have hT : 0 < T := by
67     dsimp [T]
68     exact_mod_cast hhorizon
69   have hlogK : 0 <= Real.log K := (Real.log_pos hK_one).le
70   have hone_lt_four_div : 1 < 4 / delta := by
71     rw [lt_div_iff₀ hdelta]
72     nlinarith
73   have hL : 0 < L := by
74     dsimp [L]
75     exact Real.log_pos hone_lt_four_div
76   have hlog_fourth : Real.log (1 / (delta / 4)) = L := by
77     dsimp [L]
78     exact log_one_div_fourth_eq_log_four_div delta hdelta
79   have hmax_fourth : max (Real.log (1 / (delta / 4))) 0 = L := by
80     rw [hlog_fourth]
81     exact max_eq_left hL.le
82   have hvariance : variance = K ^ 2 / gamma := by
83     dsimp [variance, K]
84     exact sampledMixedSquaredBernsteinVarianceCoefficient_eq_card_sq_div_gamma
85       arms hcard_two gamma hgamma_pos
86   have hradicand : 0 <= T * variance * L := by
87     rw [hvariance]
88     positivity
89   have hroot_nonneg :
90       0 <= Real.log K * Real.sqrt (T * variance * L) :=
91     mul_nonneg hlogK (Real.sqrt_nonneg _)
92   have hroot_right_nonneg : 0 <= gamma ^ 2 * T ^ 2 := by positivity
93   have hmixed' : K ^ 2 * Real.log K ^ 2 * L <=
94       2 * gamma ^ 6 * T ^ 3 := by
95     dsimp [K, T, L] at hmixed ⊢
96     nlinarith [hmixed]
97   have hgamma_fifth_nonneg : 0 <= gamma ^ 5 := by positivity
98   have hpow : 2 * gamma ^ 6 <= gamma ^ 5 := by
99     have htwo_gamma : 2 * gamma <= 1 := by linarith
100     calc
101       2 * gamma ^ 6 = (2 * gamma) * gamma ^ 5 := by ring
102       _ <= 1 * gamma ^ 5 :=
103         mul_le_mul_of_nonneg_right htwo_gamma hgamma_fifth_nonneg
104       _ = gamma ^ 5 := one_mul _
105   have hpowT : 2 * gamma ^ 6 * T ^ 3 <= gamma ^ 5 * T ^ 3 :=
106     mul_le_mul_of_nonneg_right hpow (by positivity)
107   have hmixed_fifth :
108       K ^ 2 * Real.log K ^ 2 * L <= gamma ^ 5 * T ^ 3 :=
109     hmixed'.trans hpowT
110   have hroot :
111       Real.log K * Real.sqrt (T * variance * L) <=
112         gamma ^ 2 * T ^ 2 := by
113     apply le_of_sq_le_sq
114     · rw [mul_pow, Real.sq_sqrt hradicand, hvariance]
115       dsimp [K, T, L] at hmixed ⊢
116       field_simp [ne_of_gt hgamma_pos]
117       nlinarith [hmixed_fifth]
118     · exact hroot_right_nonneg
119   have hgamma_sq_le_one : gamma ^ 2 <= 1 := by
120     nlinarith [hgamma_pos, hgamma_le_half]
121   have hlog_le_Klog : Real.log K <= K * Real.log K := by
122     have hmul := mul_nonneg (sub_nonneg.mpr hK_one.le) hlogK
123     nlinarith
124   have hgamma_sq_T_le_T : gamma ^ 2 * T <= T := by
125     have hmul := mul_nonneg (sub_nonneg.mpr hgamma_sq_le_one) hT.le
126     nlinarith
127   have hlog_le_T : Real.log K <= T := by
128     have hbase' : K * Real.log K <= gamma ^ 2 * T := by
129       simpa [K, T] using hbase
130     linarith
131   have hconfidence' : K * L <= gamma ^ 3 * T := by
132     simpa [K, T, L] using hconfidence
133   have hconfidence_nonneg : 0 <= gamma ^ 3 * T := by positivity
134   have hprod_one := mul_le_mul_of_nonneg_right hconfidence' hlogK
135   have hprod_two :=
136     mul_le_mul_of_nonneg_left hlog_le_T hconfidence_nonneg
137   have hprod : (K * L) * Real.log K <= gamma ^ 3 * T ^ 2 := by
138     calc
139       (K * L) * Real.log K <= (gamma ^ 3 * T) * Real.log K := hprod_one
140       _ <= (gamma ^ 3 * T) * T := hprod_two
141       _ = gamma ^ 3 * T ^ 2 := by ring
142   have hlinear :
143       Real.log K * (L / (gamma / K)) <= gamma ^ 2 * T ^ 2 := by
144     calc
145       Real.log K * (L / (gamma / K)) =
146           ((K * L) * Real.log K) / gamma := by
147         field_simp [ne_of_gt hK, ne_of_gt hgamma_pos]
148       _ <= (gamma ^ 3 * T ^ 2) / gamma :=
149         div_le_div_of_nonneg_right hprod hgamma_pos.le
150       _ = gamma ^ 2 * T ^ 2 := by
151         field_simp [ne_of_gt hgamma_pos]
152   dsimp [sampledMixedSquaredBernsteinConfidenceRadius]
153   rw [hmax_fourth]
154   change
155     Real.log K * (2 * Real.sqrt (T * variance * L) + L / (gamma / K)) <=
156       3 * gamma ^ 2 * T ^ 2
157   rw [mul_add]
158   nlinarith [hroot, hlinear]

195     have hepsilon : 0 < gamma / K := div_pos hgamma_pos hK

201   have hmixed_bound :
202       Real.log K * radius <= 3 * gamma ^ 2 * T ^ 2 := by

208   have hbase_bound : Real.log K * (K * T) <= gamma ^ 2 * T ^ 2 := by
209     have hbase' : K * Real.log K <= gamma ^ 2 * T := by

211     nlinarith
212   rw [Real.sqrt_le_iff]

215   · dsimp [bernsteinSquareHighProbabilityScale, K, T, radius] at *

296         (by simpa [T, budget, variance] using hrealized)
297   dsimp [bernsteinSquareRealizedTunedThreshold,
298     bernsteinSquareRealizedExplicitThreshold,
299     sampledPurePredictableMinusObservedBernsteinConfidenceRadius,
300     sampledComparatorEstimatorBernsteinConfidenceRadius,
301     sampledPredictableRealizedDeviationConfidenceRadius]

307         gamma * T +
308         (2 * Real.sqrt (T * budget / (gamma / K)) +
309           budget / (gamma / K)) +
310         (2 * Real.sqrt (T * budget / (gamma / K)) +
311           budget / (gamma / K)) +
312         Real.sqrt (2 * (T * variance) * budget) <=
