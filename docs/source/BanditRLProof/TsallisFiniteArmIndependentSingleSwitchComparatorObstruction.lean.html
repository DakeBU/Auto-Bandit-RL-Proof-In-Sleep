3 /-!
4 # Single-switch obstruction for the current dynamic-comparator route
5 
6 This module gives an exact two-arm, one-switch calculation. The global
7 population-mean switch count is one, while the moving-comparator mean
8 advantage charged by the current fixed-plus-advantage decomposition grows
9 linearly with the horizon. This is a proof-route obstruction, not a dynamic
10 regret lower bound.
11 -/

288 /-- For every natural coefficient, a one-switch horizon exists where the
289 exact separately charged comparator advantage exceeds that coefficient
290 times `sqrt(horizon)`. This rules out obtaining a uniform square-root
291 switch-rate by bounding this term independently. -/

293     (coefficient : Nat) :
294     let horizon := (4 * coefficient + 1) ^ 2
295     finiteArmIndependentCumulativeGlobalMeanSwitchCount
296         finiteArmIndependentSingleSwitchObstructionLaw horizon = 1 ∧
297       (coefficient : Real) * Real.sqrt (horizon : Real) <
298         finiteArmIndependentSingleSwitchComparatorAdvantage horizon := by
299   dsimp only
300   have hpos : 0 < 4 * coefficient + 1 := by omega
301   have hcount :=
302     finiteArmIndependentSingleSwitchObstructionGlobalCount_succ
303       ((4 * coefficient + 1) ^ 2 - 1)
304   have hsucc :
305       (4 * coefficient + 1) ^ 2 - 1 + 1 =
306         (4 * coefficient + 1) ^ 2 := by
307     have hsquarePos : 0 < (4 * coefficient + 1) ^ 2 := by positivity
308     omega
309   rw [hsucc] at hcount
310   refine ⟨hcount, ?_⟩
311   rw [finiteArmIndependentSingleSwitchComparatorAdvantage_eq]
312   have hcast :
313       (((4 * coefficient + 1) ^ 2 : Nat) : Real) =
314         (((4 * coefficient + 1 : Nat) : Real)) ^ 2 := by
315     norm_cast
316   rw [hcast, Real.sqrt_sq_eq_abs]
