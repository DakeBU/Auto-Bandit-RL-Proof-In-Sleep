3 /-!
4 # Integrated predictable EXP3 bridge
5 
6 This module connects the pathwise sampled-Hedge inequality to the generated
7 predictable trajectory moments.  The key law transport uses the exploration
8 distribution `p_t` to sample an action while a distinct predictable pure-Hedge
9 distribution `q_t` weights the importance-weighted estimator.
10 
11 The route is: construct measurable `q_t` sources, prove cross-weighted score
12 regularity, transport the conditional action law, aggregate over a finite
13 horizon, and only then integrate the almost-sure Hedge inequality.
14 -/

32 /-- Measurable finite-distribution source for the pure Hedge probabilities at
33 every actual trajectory time. -/
34 noncomputable def sampledTrajectoryPureProbabilitySourceAt
35     {Env : Type u} {Action : Type v}
36     [MeasurableSpace Env]
37     [MeasurableSpace Action] [MeasurableSingletonClass Action]
38     [DecidableEq Action]
39     (arms : Finset Action) (harms : arms.Nonempty)
40     (eta gamma : Real) (t : Nat) :
41     MeasurableFiniteActionDistribution arms
42       (sampledTrajectoryPureProbabilityAt (Env := Env) arms eta gamma t) := by
43   refine
44     { distribution := fun sample =>
45         { nonneg := fun action _haction =>
46             distribution_nonneg arms harms eta
47               (sampledTrajectoryObservedLoss arms eta gamma sample) t action
48           sum_eq_one := sum_distribution arms harms eta
49             (sampledTrajectoryObservedLoss arms eta gamma sample) t }
50       measurable_prob := ?_ }
51   intro action haction
52   cases t with
53   | zero =>
54       simp [sampledTrajectoryPureProbabilityAt, distribution, totalWeight,
55         weight, cumulativeLoss]
56   | succ n =>
57       have hmeas := measurable_normalizedHistoryDistribution arms eta
58         (sampledHistoryScore arms eta gamma n)
59         (fun selected hselected =>
60           MeasurableFiniteHistoryScore.measurable_score
61             (measurableFiniteHistoryScore_sampledHistoryScore arms eta gamma)
62             n selected hselected)
63         action haction
64       simpa [sampledTrajectoryPureProbabilityAt,
65         distribution_sampledTrajectoryObservedLoss_succ] using
66         hmeas.comp ((Preorder.measurable_frestrictLe n).comp measurable_snd)

242 /-- The observed scalar reward can be replaced almost surely by its selected
243 predictable coordinate inside the pure-Hedge mixed estimator. -/
244 theorem sampledTrajectoryPureObservedLossAt_ae_eq_weightedPredictable
245     {Env : Type u} {Action : Type v}
246     [MeasurableSpace Env] [StandardBorelSpace Env]
247     [MeasurableSpace Action] [MeasurableSingletonClass Action]
248     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
249     (prior : Measure Env) [IsFiniteMeasure prior]
250     (arms : Finset Action) (harms : arms.Nonempty)
251     (eta gamma : Real) (hgamma_nonneg : 0 <= gamma)
252     (hgamma_le_one : gamma <= 1)
253     (loss : PredictableLossVector Env Action) (t : Nat) :
254     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
255       eta gamma hgamma_nonneg hgamma_le_one loss.environment
256     (fun sample => sampledTrajectoryPureObservedLossAt
257         arms eta gamma t sample) =ᵐ[mu]
258       (fun sample => weightedImportanceWeightedLoss arms
259         (sampledTrajectoryProbabilityAt arms eta gamma t sample)
260         (sampledTrajectoryPureProbabilityAt arms eta gamma t sample)
261         (predictableLossAt loss t sample) (sample.2 t).1) := by
262   dsimp only
263   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
264     eta gamma hgamma_nonneg hgamma_le_one loss.environment
265   have hreward :
266       (fun sample : Env × ((k : Nat) -> Action × Real) => (sample.2 t).2) =ᵐ[mu]
267         (fun sample => predictableLossAt loss t sample (sample.2 t).1) := by
268     cases t with
269     | zero =>
270         simpa [mu, predictableLossAt,
271           sampledImportanceWeightedTrajectoryKernel] using
272           (canonicalPredictableTrajectoryMeasure_reward_zero_eq_initialLoss_ae
273             prior
274               (sampledImportanceWeightedHistoryAlgorithm arms harms eta gamma
275                 hgamma_nonneg hgamma_le_one)
276               loss)
277     | succ n =>
278         simpa [mu, predictableLossAt] using
279           (sampledPredictableTrajectoryMeasure_reward_eq_successorLoss_ae
280             prior arms harms eta gamma hgamma_nonneg hgamma_le_one loss n)
281   filter_upwards [hreward] with sample hsample
282   unfold sampledTrajectoryPureObservedLossAt mixedLoss
283     weightedImportanceWeightedLoss
284   apply Finset.sum_congr rfl
285   intro action _haction
286   change
287     distribution arms eta
288           (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
289         sampledTrajectoryObservedLoss arms eta gamma sample t action =
290       distribution arms eta
291           (sampledTrajectoryObservedLoss arms eta gamma sample) t action *
292         importanceWeightedLoss
293           (sampledTrajectoryProbabilityAt arms eta gamma t sample)
294           (predictableLossAt loss t sample) (sample.2 t).1 action
295   congr 1
296   unfold sampledTrajectoryObservedLoss observedImportanceWeightedLossAt
297   by_cases hchosen : (sample.2 t).1 = action
298   · simp [importanceWeightedLoss, hchosen, hsample]
299   · simp [importanceWeightedLoss, hchosen]
