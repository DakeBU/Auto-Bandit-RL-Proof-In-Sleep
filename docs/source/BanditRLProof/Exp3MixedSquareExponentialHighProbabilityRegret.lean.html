3 /-!
4 # Predictable EXP3 regret with exponential mixed-square confidence
5 
6 This module replaces the Markov estimator-square event in the generated
7 predictable EXP3 regret assembly by the compiled exponential mixed-square
8 confidence theorem.  The pure-cross and fixed-comparator confidence terms
9 remain the existing variance-sensitive Bernstein radii.
10 
11 The square radius still uses the interval proxy of order
12 `(|arms| / gamma)^2`; this route is logarithmic in the square-event failure
13 probability but is not a Freedman or ideal EXP3.P theorem.
14 -/
15

22 /-- Predictable regret budget using the exponential mixed-square threshold
23 instead of the Markov `|arms| * T / deltaSquare` threshold. -/
24 noncomputable def sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
25     {Action : Type v} (arms : Finset Action) (eta gamma : Real)
26     (horizon : Nat) (deltaSquare deltaConfidence : Real) : Real :=
27   Real.log arms.card / eta +
28     (eta * (1 / (1 - gamma))) *
29       ((arms.card : Real) * (horizon : Real) +
30         sampledMixedSquaredConfidenceRadius arms gamma horizon deltaSquare) +
31     gamma * (horizon : Real) +
32     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
33       arms gamma horizon deltaConfidence +
34     sampledComparatorEstimatorBernsteinConfidenceRadius
35       arms gamma horizon deltaConfidence

37 /-- Generated predictable EXP3 regret with an exponential estimator-square
38 event and the two compiled Bernstein confidence events. -/
39 theorem sampledPredictable_exponentialSquareBernsteinHighProbabilityRegret_tail
40     {Env : Type u} {Action : Type v}
41     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
42     [MeasurableSpace Action] [MeasurableSingletonClass Action]
43     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
44     (prior : Measure Env) [IsProbabilityMeasure prior]
45     (arms : Finset Action) (harms : arms.Nonempty)
46     (eta gamma : Real) (heta : 0 < eta)
47     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
48     (loss : PredictableLossVector Env Action)
49     (comparator : Action) (hcomparator : comparator ∈ arms)
50     (horizon : Nat) (hhorizon : 0 < horizon)
51     (deltaSquare deltaConfidence : Real)
52     (hdeltaSquare : 0 < deltaSquare)
53     (hdeltaConfidence : 0 < deltaConfidence) :
54     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
55       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
56     mu {sample |
57         sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
58             arms eta gamma horizon deltaSquare deltaConfidence <=
59           (Finset.range horizon).sum (fun t =>
60               sampledTrajectoryExploredPredictableLossAt
61                 arms eta gamma loss t sample) -
62             (Finset.range horizon).sum (fun t =>
63               predictableLossAt loss t sample comparator)} <=
64       (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
65         ENNReal.ofReal deltaConfidence := by
66   dsimp only
67   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
68     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
69   let purePredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
70     (Finset.range horizon).sum (fun t =>
71       sampledTrajectoryPurePredictableLossAt arms eta gamma loss t sample)
72   let pureObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
73     (Finset.range horizon).sum (fun t =>
74       sampledTrajectoryPureObservedLossAt arms eta gamma t sample)
75   let comparatorObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
76     (Finset.range horizon).sum (fun t =>
77       observedImportanceWeightedLossAt arms eta gamma t sample comparator)
78   let comparatorTrue := fun sample : Env × ((k : Nat) -> Action × Real) =>
79     (Finset.range horizon).sum (fun t =>
80       predictableLossAt loss t sample comparator)
81   let exploredPredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
82     (Finset.range horizon).sum (fun t =>
83       sampledTrajectoryExploredPredictableLossAt arms eta gamma loss t sample)
84   let secondMoment := sampledObservedMixedSquaredSum
85     (Env := Env) arms eta gamma horizon
86   let secondMomentBudget :=
87     (arms.card : Real) * (horizon : Real) +
88       sampledMixedSquaredConfidenceRadius arms gamma horizon deltaSquare
89   let pureRadius :=
90     sampledPurePredictableMinusObservedBernsteinConfidenceRadius
91       arms gamma horizon deltaConfidence
92   let comparatorRadius := sampledComparatorEstimatorBernsteinConfidenceRadius
93     arms gamma horizon deltaConfidence
94   let secondMomentBad : Set (Env × ((k : Nat) -> Action × Real)) :=
95     {sample | secondMomentBudget <= secondMoment sample}
96   let pureBad : Set (Env × ((k : Nat) -> Action × Real)) :=
97     {sample | pureRadius <= purePredictable sample - pureObserved sample}
98   let comparatorBad : Set (Env × ((k : Nat) -> Action × Real)) :=
99     {sample | comparatorRadius <= comparatorObserved sample - comparatorTrue sample}
100   have hsecondTail : mu secondMomentBad <= ENNReal.ofReal deltaSquare := by
101     have h := sampledPredictableObservedMixedSquared_sum_tail_delta
102       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss horizon
103         hhorizon deltaSquare hdeltaSquare
104     dsimp only at h
105     simpa [mu, secondMomentBad, secondMomentBudget, secondMoment] using h
106   have hpureTail : mu pureBad <= ENNReal.ofReal deltaConfidence := by
107     have h := sampledPurePredictableMinusObserved_sum_tail_bernstein_delta
108       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss
109         horizon deltaConfidence hdeltaConfidence
110     dsimp only at h
111     simpa [mu, pureBad, pureRadius, purePredictable, pureObserved] using h
112   have hcomparatorTail : mu comparatorBad <= ENNReal.ofReal deltaConfidence := by
113     have h := sampledObservedComparatorEstimatorDeviation_sum_tail_bernstein_delta
114       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss comparator
115         hcomparator horizon deltaConfidence hdeltaConfidence
116     dsimp only at h
117     simpa [mu, comparatorBad, comparatorRadius, comparatorObserved,
118       comparatorTrue,
119       sampledTrajectoryObservedComparatorEstimatorDeviationAt] using h
120   have hhedge :=
121     sampledPredictableTrajectoryMeasure_hedge_exploredSecondMoment_le_ae
122       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss horizon
123         comparator hcomparator
124   dsimp only at hhedge
125   have hsubset : ∀ᵐ sample ∂mu,
126       sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
127           arms eta gamma horizon deltaSquare deltaConfidence <=
128         exploredPredictable sample - comparatorTrue sample ->
129       sample ∈ (secondMomentBad ∪ pureBad) ∪ comparatorBad := by
130     filter_upwards [hhedge] with sample hsampleHedge
131     intro hregret
132     by_cases hsecond : sample ∈ secondMomentBad
133     · exact Or.inl (Or.inl hsecond)
134     by_cases hpure : sample ∈ pureBad
135     · exact Or.inl (Or.inr hpure)
136     by_cases hcomparator : sample ∈ comparatorBad
137     · exact Or.inr hcomparator
138     exfalso
139     have hsecondGood : secondMoment sample < secondMomentBudget := by
140       change ¬ secondMomentBudget <= secondMoment sample at hsecond
141       exact lt_of_not_ge hsecond
142     have hpureGood :
143         purePredictable sample - pureObserved sample < pureRadius := by
144       change ¬ pureRadius <= purePredictable sample - pureObserved sample at hpure
145       exact lt_of_not_ge hpure
146     have hcomparatorGood :
147         comparatorObserved sample - comparatorTrue sample <
148           comparatorRadius := by
149       change ¬ comparatorRadius <=
150         comparatorObserved sample - comparatorTrue sample at hcomparator
151       exact lt_of_not_ge hcomparator
152     have hexploration :
153         exploredPredictable sample <=
154           purePredictable sample + gamma * (horizon : Real) := by
155       have h := (sampledTrajectory_finiteHorizon_explorationBias_secondMoment
156         arms harms eta gamma hgamma_pos.le hgamma_lt_one loss horizon sample).1
157       simpa [exploredPredictable, purePredictable,
158         sampledTrajectoryExploredPredictableLossAt,
159         sampledTrajectoryPurePredictableLossAt,
160         sampledTrajectoryPureProbabilityAt] using h
161     have hcoef_pos : 0 < eta * (1 / (1 - gamma)) :=
162       mul_pos heta (one_div_pos.mpr (sub_pos.mpr hgamma_lt_one))
163     have hweightedSecond :
164         (eta * (1 / (1 - gamma))) * secondMoment sample <
165           (eta * (1 / (1 - gamma))) * secondMomentBudget :=
166       mul_lt_mul_of_pos_left hsecondGood hcoef_pos
167     have hhedge' :
168         pureObserved sample - comparatorObserved sample <=
169           Real.log arms.card / eta +
170             (eta * (1 / (1 - gamma))) * secondMoment sample := by
171       simpa [pureObserved, comparatorObserved, secondMoment,
172         sampledObservedMixedSquaredSum] using hsampleHedge
173     have hfinal : exploredPredictable sample - comparatorTrue sample <
174         sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
175           arms eta gamma horizon deltaSquare deltaConfidence := by
176       dsimp [sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget,
177         secondMomentBudget, pureRadius, comparatorRadius]
178       linarith
179     exact (not_lt_of_ge hregret) hfinal
180   calc
181     mu {sample |
182         sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
183             arms eta gamma horizon deltaSquare deltaConfidence <=
184           (Finset.range horizon).sum (fun t =>
185               sampledTrajectoryExploredPredictableLossAt
186                 arms eta gamma loss t sample) -
187             (Finset.range horizon).sum (fun t =>
188               predictableLossAt loss t sample comparator)} <=
189         mu ((secondMomentBad ∪ pureBad) ∪ comparatorBad) := by
190       apply measure_mono_ae
191       simpa [exploredPredictable, comparatorTrue, mu] using hsubset
192     _ <= (mu secondMomentBad + mu pureBad) + mu comparatorBad := by
193       exact (measure_union_le _ _).trans
194         (add_le_add (measure_union_le _ _) le_rfl)
195     _ <= (ENNReal.ofReal deltaSquare + ENNReal.ofReal deltaConfidence) +
196         ENNReal.ofReal deltaConfidence :=
197       add_le_add (add_le_add hsecondTail hpureTail) hcomparatorTail

199 /-- Total-failure form: the exponential square event and both Bernstein
200 confidence events each receive `delta / 3`. -/
201 theorem sampledPredictable_exponentialSquareBernsteinHighProbabilityRegret_tail_total_delta
202     {Env : Type u} {Action : Type v}
203     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
204     [MeasurableSpace Action] [MeasurableSingletonClass Action]
205     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
206     (prior : Measure Env) [IsProbabilityMeasure prior]
207     (arms : Finset Action) (harms : arms.Nonempty)
208     (eta gamma : Real) (heta : 0 < eta)
209     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
210     (loss : PredictableLossVector Env Action)
211     (comparator : Action) (hcomparator : comparator ∈ arms)
212     (horizon : Nat) (hhorizon : 0 < horizon)
213     (delta : Real) (hdelta : 0 < delta) :
214     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
215       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
216     mu {sample |
217         sampledPredictableExponentialSquareBernsteinHighProbabilityRegretBudget
218             arms eta gamma horizon (delta / 3) (delta / 3) <=
219           (Finset.range horizon).sum (fun t =>
220               sampledTrajectoryExploredPredictableLossAt
221                 arms eta gamma loss t sample) -
222             (Finset.range horizon).sum (fun t =>
223               predictableLossAt loss t sample comparator)} <=
224       ENNReal.ofReal delta := by
225   have hthird_pos : 0 < delta / 3 := div_pos hdelta (by norm_num)
226   have h :=
227     sampledPredictable_exponentialSquareBernsteinHighProbabilityRegret_tail
228       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
229         hcomparator horizon hhorizon (delta / 3) (delta / 3)
230           hthird_pos hthird_pos
231   dsimp only at h ⊢
232   have hthird_nonneg : 0 <= delta / 3 := hthird_pos.le
233   have hprob :
234       (ENNReal.ofReal (delta / 3) + ENNReal.ofReal (delta / 3)) +
235           ENNReal.ofReal (delta / 3) = ENNReal.ofReal delta := by
236     rw [← ENNReal.ofReal_add hthird_nonneg hthird_nonneg]
237     rw [← ENNReal.ofReal_add (add_nonneg hthird_nonneg hthird_nonneg)
238       hthird_nonneg]
239     congr 1
240     ring
241   exact h.trans_eq hprob
