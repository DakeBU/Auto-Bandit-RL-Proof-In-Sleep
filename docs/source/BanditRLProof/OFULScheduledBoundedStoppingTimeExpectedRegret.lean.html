4 /-!
5 # Bounded stopping-time expected OFUL pseudo-regret rate
6 
7 This module integrates the compiled bounded-stopping-time high-probability
8 pseudo-regret theorem. Horizon monotonicity moves both the explicit scheduled
9 budget and the deterministic gap envelope to the deterministic stopping-time
10 bound. The expectation proof is an indicator decomposition, not optional
11 stopping.

22 /-- The explicit telescoping confidence logarithm is nonnegative. -/
23 theorem telescopingHighProbabilityRegretLogBudget_nonneg
24     {Feature : Type u} [Fintype Feature] [Nonempty Feature]
25     (lambda : Real) (hlambda : 0 < lambda)
26     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1)
27     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2) :
28     0 <= telescopingHighProbabilityRegretLogBudget
29       (Feature := Feature) lambda delta horizon L2 := by
30   have hdet :
31       0 <= standardScalarLogDetBudget
32         (Feature := Feature) lambda (horizon + 1) L2 :=
33     standardScalarLogDetBudget_nonneg
34       (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
35   have hprod_one :
36       (1 : Real) <=
37         (((horizon + 1 : Nat) : Real) *
38           ((horizon + 2 : Nat) : Real)) := by
39     have hfirst :
40         (1 : Real) <= (((horizon + 1 : Nat) : Real)) := by
41       exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
42     have hsecond :
43         (1 : Real) <= (((horizon + 2 : Nat) : Real)) := by
44       exact_mod_cast Nat.succ_le_succ (Nat.zero_le (horizon + 1))
45     simpa using
46       (mul_le_mul hfirst hsecond (by norm_num : (0 : Real) <= 1)
47         (by positivity : (0 : Real) <= (((horizon + 1 : Nat) : Real))))
48   have hratio_one :
49       (1 : Real) <=
50         ((((horizon + 1 : Nat) : Real) *
51             ((horizon + 2 : Nat) : Real)) / delta) := by
52     rw [le_div_iff₀ hdelta]
53     simpa using hdelta_one.trans hprod_one
54   unfold telescopingHighProbabilityRegretLogBudget
55   exact add_nonneg hdet
56     (mul_nonneg (by norm_num) (Real.log_nonneg hratio_one))

60     {Feature : Type u} [Fintype Feature] [Nonempty Feature]

67       telescopingHighProbabilityRegretLogBudget
68         (Feature := Feature) lambda delta horizon L2 := by

70       standardScalarLogDetBudget
71           (Feature := Feature) lambda (n + 1) L2 <=
72         standardScalarLogDetBudget
73           (Feature := Feature) lambda (horizon + 1) L2 :=
74     standardScalarLogDetBudget_mono
75       (Feature := Feature) lambda hlambda L2 hL2
76       (Nat.add_le_add_right hn 1)

78       (((n + 1 : Nat) : Real)) <=
79         (((horizon + 1 : Nat) : Real)) := by

81   have hsecond :
82       (((n + 2 : Nat) : Real)) <=
83         (((horizon + 2 : Nat) : Real)) := by
84     exact_mod_cast Nat.add_le_add_right hn 2
85   have hprod :
86       (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) <=
87         (((horizon + 1 : Nat) : Real) *
88           ((horizon + 2 : Nat) : Real)) :=
89     mul_le_mul hfirst hsecond (by positivity) (by positivity)
90   have hratio :
91       ((((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) / delta) <=
92         ((((horizon + 1 : Nat) : Real) *
93             ((horizon + 2 : Nat) : Real)) / delta) :=
94     (div_le_div_iff_of_pos_right hdelta).2 hprod
95   have hratio_pos :
96       0 <
97         ((((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) / delta) :=
98     div_pos (mul_pos (by positivity) (by positivity)) hdelta
99   have hlog :
100       Real.log
101           ((((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) / delta) <=
102         Real.log
103           ((((horizon + 1 : Nat) : Real) *
104               ((horizon + 2 : Nat) : Real)) / delta) :=
105     Real.log_le_log hratio_pos hratio
106   unfold telescopingHighProbabilityRegretLogBudget
107   exact add_le_add hdet (mul_le_mul_of_nonneg_left hlog (by norm_num))

140     telescopingHighProbabilityRegretLogBudget

143     telescopingHighProbabilityRegretLogBudget

159       (Feature := Feature) lambda hlambda L2 hL2

165   have hradius :

169       (mul_le_mul_of_nonneg_left (Real.sqrt_le_sqrt hlog) hR) le_rfl
170   have hradiusN :
171       0 <= R * Real.sqrt logN + Real.sqrt lambda * S := by
172     positivity
173   have hradiusT :
174       0 <= R * Real.sqrt logT + Real.sqrt lambda * S := by
175     positivity
176   have htime :
177       Real.sqrt (((n + 1 : Nat) : Real)) <=
178         Real.sqrt (((horizon + 1 : Nat) : Real)) := by
179     apply Real.sqrt_le_sqrt
180     exact_mod_cast Nat.add_le_add_right hn 1
181   have hwidthLog :
182       Real.sqrt (2 * detN) <= Real.sqrt (2 * detT) := by
183     exact Real.sqrt_le_sqrt
184       (mul_le_mul_of_nonneg_left hdet (by norm_num))
185   have hwidth :
186       Real.sqrt (((n + 1 : Nat) : Real)) * Real.sqrt (2 * detN) <=
187         Real.sqrt (((horizon + 1 : Nat) : Real)) *
188           Real.sqrt (2 * detT) := by
189     exact mul_le_mul htime hwidthLog
190       (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)
191   have hproduct :
192       (R * Real.sqrt logN + Real.sqrt lambda * S) *
193           (Real.sqrt (((n + 1 : Nat) : Real)) * Real.sqrt (2 * detN)) <=
194         (R * Real.sqrt logT + Real.sqrt lambda * S) *
195           (Real.sqrt (((horizon + 1 : Nat) : Real)) *
196             Real.sqrt (2 * detT)) :=
197     mul_le_mul hradius hwidth
198       (mul_nonneg (Real.sqrt_nonneg _) (Real.sqrt_nonneg _)) hradiusT
199   unfold telescopingHighProbabilityPseudoRegretBound
200   simpa [logN, logT, detN, detT, mul_assoc] using
201     (add_le_add_left
202       (mul_le_mul_of_nonneg_left hproduct (by norm_num : (0 : Real) <= 2))
203       (2 * S * Real.sqrt L2))
