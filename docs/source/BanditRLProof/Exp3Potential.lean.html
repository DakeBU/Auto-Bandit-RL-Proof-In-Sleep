    1 import Mathlib.Algebra.BigOperators.Group.Finset.Basic
2 import Mathlib.Analysis.SpecialFunctions.Exp
3 import Mathlib.Tactic.Ring

5 /-!
6 # EXP3 potential finite-sum wrappers
7 
8 This module records the deterministic finite-action potential surface used by
9 exponential-weights/EXP3 routes.  It deliberately stops before importance
10 weighted estimators, logarithmic inequalities, learning-rate optimization, or a
11 regret theorem.
12 -/
13 
14 namespace BanditRLProof
15 namespace Exp3Potential
16 
17 universe u
18 
19 /-- Finite-action exponential-weights potential. -/
20 noncomputable def potential {Action : Type u}
21     (arms : Finset Action) (w : Action -> Real) : Real :=
22   arms.sum w
23 
24 /-- Multiplicative exponential-weights update for one action. -/
25 noncomputable def updatedWeight {Action : Type u}
26     (eta : Real) (w loss : Action -> Real) (a : Action) : Real :=
27   w a * Real.exp (-eta * loss a)
28 
29 /-- Potential after applying the exponential-weights update on each action. -/
30 noncomputable def updatedPotential {Action : Type u}
31     (arms : Finset Action) (eta : Real)
32     (w loss : Action -> Real) : Real :=
33   potential arms (updatedWeight eta w loss)
34 
35 /-- Unfold the updated potential as an explicit finite sum. -/
36 theorem updatedPotential_eq_sum {Action : Type u}
37     (arms : Finset Action) (eta : Real)
38     (w loss : Action -> Real) :
39     updatedPotential arms eta w loss =
40       arms.sum (fun a => w a * Real.exp (-eta * loss a)) := by
41   rfl
42 
43 /-- Exponential updating preserves nonnegative weights. -/
44 theorem updatedWeight_nonneg_of_nonneg {Action : Type u}
45     (eta : Real) (w loss : Action -> Real) (a : Action)
46     (hw : 0 <= w a) :
47     0 <= updatedWeight eta w loss a := by
48   exact mul_nonneg hw (Real.exp_pos _).le
49 
50 /-- The updated potential is nonnegative when all current finite weights are. -/
51 theorem updatedPotential_nonneg_of_nonneg {Action : Type u}
52     (arms : Finset Action) (eta : Real)
53     (w loss : Action -> Real)
54     (hw : forall a, a ∈ arms -> 0 <= w a) :
55     0 <= updatedPotential arms eta w loss := by
56   exact Finset.sum_nonneg (fun a ha =>
57     updatedWeight_nonneg_of_nonneg eta w loss a (hw a ha))
58 
59 /--
60 One-step potential increment identity.
61 
62 This is the algebraic finite-sum surface used before applying any EXP/log
63 inequality such as `exp x <= 1 + x + x^2`.
64 -/
65 theorem updatedPotential_sub_potential_eq_sum_weight_mul_exp_sub_one
66     {Action : Type u}
67     (arms : Finset Action) (eta : Real)
68     (w loss : Action -> Real) :
69     updatedPotential arms eta w loss - potential arms w =
70       arms.sum (fun a => w a * (Real.exp (-eta * loss a) - 1)) := by
71   simp only [updatedPotential, potential, updatedWeight]
72   rw [← Finset.sum_sub_distrib]
73   refine Finset.sum_congr rfl ?_
74   intro a _ha
75   ring
76 
77 /-- Finite-horizon telescoping for a real-valued potential process. -/
78 theorem sum_range_forward_difference
79     (Phi : Nat -> Real) (T : Nat) :
80     (Finset.range T).sum (fun t => Phi (t + 1) - Phi t) =
81       Phi T - Phi 0 := by
82   induction T with
83   | zero =>
84       simp
85   | succ T ih =>
86       rw [Finset.sum_range_succ, ih]
87       ring
88 
89 /-- Potential process induced by a time-indexed finite-action weight family. -/
90 noncomputable def potentialProcess {Action : Type u}
91     (arms : Finset Action) (w : Nat -> Action -> Real) (t : Nat) : Real :=
92   potential arms (w t)
93 
94 /--
95 Finite-horizon telescope specialized to exponential-weights potentials.
96 
97 This is the compiled local replacement for treating "the potential telescopes"
98 as only a proof weapon.
99 -/
100 theorem potentialProcess_telescope_sum_range {Action : Type u}
101     (arms : Finset Action) (w : Nat -> Action -> Real) (T : Nat) :
102     (Finset.range T).sum
103         (fun t => potentialProcess arms w (t + 1) -
104           potentialProcess arms w t) =
105       potentialProcess arms w T - potentialProcess arms w 0 := by
106   exact sum_range_forward_difference (fun t => potentialProcess arms w t) T
107 
108 end Exp3Potential
109 end BanditRLProof
