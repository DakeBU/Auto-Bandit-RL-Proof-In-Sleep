4 /-!
5 # Expected scheduled half-Tsallis successor stability
6 
7 This module transports the one-round ordinary-importance-weighted
8 conjugate-potential bound through the generated scheduled action law.  Round
9 `n + 1` uses `eta (n + 1)`, so the finite sum can vary the learning rate
10 without weakening the deterministic stability theorem.
11 
12 Only successor rounds are treated here.  The initial action has a different
13 conditioning surface and remains a separate leaf.  Rates above `1 / 2` also
14 remain outside this theorem.
15 -/

169     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
170     (loss : Exp3.PredictableLossVector Env Action) (n : Nat)

173         (Env × History.FinitePairHistory Action Real n) × Action =>
174       sampledScheduledHalfTsallisUpdatedAt arms harms eta loss n
175         sample.1 sample.2 candidate) := by

177     canonicalHalfTsallisScheduleFiniteHistorySelectorMeasurability
178       arms harms eta
179   let updatedScore :
180       ((Env × History.FinitePairHistory Action Real n) × Action) ->
181         Action -> Real :=
182     fun sample selected =>
183       sampledScheduledHalfTsallisScoreAt arms harms eta n
184           sample.1 selected +
185         Exp3.importanceWeightedLoss
186           (sampledScheduledHalfTsallisProbabilityAt
187             arms harms eta n sample.1)
188           (sampledScheduledHalfTsallisPredictableLossAt loss n sample.1)
189           sample.2 selected
190   have hupdatedScore : forall selected, selected ∈ arms ->
191       Measurable (fun sample => updatedScore sample selected) := by
192     intro selected hselected
193     have hscore : Measurable (fun sample :

197       (measurable_sampledScheduledHalfTsallisHistoryScore
198         arms harms eta finiteSelector n selected hselected).comp
199           (measurable_snd.comp measurable_fst)
200     have hprob : Measurable (fun sample :
201         ((Env × History.FinitePairHistory Action Real n) × Action) =>
202         sampledScheduledHalfTsallisProbabilityAt arms harms eta n
203           sample.1 selected) :=
204       ((sampledScheduledHalfTsallisEnvironmentHistoryDistributionSource
205         (Env := Env) arms harms eta finiteSelector n).measurable_prob

207     have hloss : Measurable (fun sample :
208         ((Env × History.FinitePairHistory Action Real n) × Action) =>
209         sampledScheduledHalfTsallisPredictableLossAt loss n
210           sample.1 selected) :=
211       (measurable_sampledHalfTsallisPredictableLossAt loss n selected).comp
212         measurable_fst
213     have hincrement : Measurable (fun sample :
214         ((Env × History.FinitePairHistory Action Real n) × Action) =>
215         Exp3.importanceWeightedLoss
216           (sampledScheduledHalfTsallisProbabilityAt

218           (sampledScheduledHalfTsallisPredictableLossAt loss n sample.1)

220       unfold Exp3.importanceWeightedLoss
221       refine Measurable.ite ?_ (hloss.div hprob) measurable_const
222       simpa only [Set.mem_setOf_eq] using
223         measurable_snd (measurableSet_singleton selected)
224     exact hscore.add hincrement
225   have hcanonical : Measurable (fun sample =>
226       halfTsallisMinimizer arms harms (eta (n + 1))
227         (updatedScore sample) candidate) :=
228     measurable_halfTsallisMinimizer_comp arms harms (eta (n + 1))
229       updatedScore hupdatedScore candidate hcandidate
230   simpa only [updatedScore, sampledScheduledHalfTsallisUpdatedAt,
231     halfTsallisHistoryUpdatedMinimizer, halfTsallisUpdatedMinimizer] using

477 Time zero and rates above `1 / 2` are deliberately not claimed.

485     (horizon : Nat) (arms : Finset Action) (harms : arms.Nonempty)
486     (eta : Nat -> Real)

488     (heta_le : forall n, n < horizon -> eta (n + 1) <= 1 / 2)
489     (loss : Exp3.PredictableLossVector Env Action) :
490     let selector :=
491       canonicalHalfTsallisScheduleGeneratedSelectorMeasurability
492         arms harms eta loss
