20 /-- Sum of square roots of expected probabilities over suboptimal arms. -/
21 noncomputable def sampledScheduledHalfTsallisExpectedSuboptimalSqrtMassAt
22     {Env : Type u} {Action : Type v}
23     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
24     (mu : Measure (Env × ((k : Nat) -> Action × Real)))
25     (arms : Finset Action) (harms : arms.Nonempty) (eta : Nat -> Real)
26     (best : Action) (t : Nat) : Real :=
27   (arms.erase best).sum (fun action =>
28     Real.sqrt (sampledScheduledHalfTsallisExpectedProbabilityAt

56 /-- Combined coefficient multiplying `sqrt (E[p_t(a)])` after adding refined
57 stability and reciprocal-rate penalty terms. -/
58 noncomputable def sampledScheduledHalfTsallisRefinedCoefficient
59     (eta : Nat -> Real) : Nat -> Real
60   | 0 => 2 * eta 0 + 2 / eta 0
61   | t + 1 =>
62       2 * eta (t + 1) + 2 * (1 / eta (t + 1) - 1 / eta t)
