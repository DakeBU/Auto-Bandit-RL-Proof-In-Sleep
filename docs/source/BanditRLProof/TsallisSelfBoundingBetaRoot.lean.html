4 /-!
5 # Tsallis self-bounding beta root
6 
7 This module isolates the scalar intermediate-value step used by the refined
8 self-bounding tuning route.  It proves existence of a root in the paper's
9 admissible beta interval without assuming a Lambert-W API.  Quantitative bounds
10 on that root remain downstream obligations.
11 -/
12 
13 namespace BanditRLProof
14 namespace Tsallis
15 
16 open Set
17 
18 /-- The scalar equation used to tune the refined self-bounding parameter. -/
19 noncomputable def selfBoundingBetaEquation
20     (scale reciprocalGap corruption beta : Real) : Real :=
21   corruption * reciprocalGap / scale * beta - Real.log beta - 1
22 
23 /-- The beta equation is continuous on every interval bounded below by one. -/
24 theorem continuousOn_selfBoundingBetaEquation
25     (scale reciprocalGap corruption upper : Real) :
26     ContinuousOn (selfBoundingBetaEquation scale reciprocalGap corruption)
27       (Icc 1 upper) := by
28   unfold selfBoundingBetaEquation
29   apply ContinuousOn.sub
30   · apply ContinuousOn.sub
31     · fun_prop
32     · exact Real.continuousOn_log.mono (by
33         intro beta hbeta
34         exact ne_of_gt (zero_lt_one.trans_le hbeta.1))
35   · exact continuousOn_const
36 
37 /-- Under the corruption window used by the refined Tsallis-INF analysis, the
38 scalar beta equation has a zero between one and `scale / reciprocalGap ^ 2`.
39 This is the intermediate-value certificate that precedes any quantitative
40 Lambert-W estimate. -/
41 theorem exists_selfBoundingBetaEquation_eq_zero
42     (scale reciprocalGap corruption : Real)
43     (hscale : 0 < scale) (hreciprocalGap : 0 < reciprocalGap)
44     (hupper : 1 <= scale / reciprocalGap ^ 2)
45     (hcorruptionUpper : corruption * reciprocalGap <= scale)
46     (hcorruptionLower :
47       reciprocalGap *
48           (Real.log (scale / reciprocalGap ^ 2) + 1) <= corruption) :
49     exists beta,
50       beta ∈ Icc 1 (scale / reciprocalGap ^ 2) ∧
51         selfBoundingBetaEquation scale reciprocalGap corruption beta = 0 := by
52   let upper := scale / reciprocalGap ^ 2
53   let g := selfBoundingBetaEquation scale reciprocalGap corruption
54   have hgOne : g 1 <= 0 := by
55     dsimp [g, selfBoundingBetaEquation]
56     rw [Real.log_one]
57     have hdiv : corruption * reciprocalGap / scale <= 1 :=
58       (div_le_one hscale).2 hcorruptionUpper
59     linarith
60   have hgUpper : 0 <= g upper := by
61     have hreciprocalGapNe : reciprocalGap ≠ 0 := ne_of_gt hreciprocalGap
62     have hscaleNe : scale ≠ 0 := ne_of_gt hscale
63     have hnormalize :
64         corruption * reciprocalGap / scale * upper =
65           corruption / reciprocalGap := by
66       dsimp [upper]
67       field_simp
68     have hdiv :
69         Real.log upper + 1 <= corruption / reciprocalGap := by
70       apply (le_div_iff₀ hreciprocalGap).2
71       simpa [mul_comm] using hcorruptionLower
72     dsimp [g, selfBoundingBetaEquation]
73     rw [hnormalize]
74     linarith
75   have hcontinuous : ContinuousOn g (Icc 1 upper) := by
76     exact continuousOn_selfBoundingBetaEquation
77       scale reciprocalGap corruption upper
78   have hzero : (0 : Real) ∈ Icc (g 1) (g upper) := ⟨hgOne, hgUpper⟩
79   rcases intermediate_value_Icc hupper hcontinuous hzero with
80     ⟨beta, hbeta, hbetaZero⟩
81   exact ⟨beta, hbeta, hbetaZero⟩
82 
83 end Tsallis
