3 /-!
4 # Predictable-variance tail for realized EXP3 loss
5 
6 This module iterates the exact selected-loss variance-compensated conditional
7 MGF along the generated trajectory. It yields a Bernstein-shaped upper tail
8 for realized-minus-predictable selected loss jointly with a pathwise budget on
9 the cumulative selected-loss predictable variance.
10 -/

18 /-- Shifted exact-variance compensated realized-loss process. -/
19 noncomputable def sampledPredictableRealizedCompensatedProcess
20     {Env : Type u} {Action : Type v}
21     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
22     (arms : Finset Action) (eta gamma tilt : Real)
23     (loss : PredictableLossVector Env Action) :
24     Nat → Env × ((k : Nat) → Action × Real) → Real :=
25   fun i sample =>
26     tilt * sampledPredictableRealizedDeviationProcess
27         arms eta gamma loss i sample -
28       tilt ^ 2 * sampledPredictableRealizedVarianceProcess
29         arms eta gamma loss i sample
30 
31 theorem sampledPredictableRealizedCompensatedProcess_stronglyAdapted
32     {Env : Type u} {Action : Type v}
33     [MeasurableSpace Env]
34     [MeasurableSpace Action] [MeasurableSingletonClass Action]
35     [DecidableEq Action]
36     (arms : Finset Action) (harms : arms.Nonempty)
37     (eta gamma tilt : Real) (hgamma_pos : 0 < gamma)
38     (hgamma_le_one : gamma ≤ 1)
39     (loss : PredictableLossVector Env Action) :
40     StronglyAdapted (sampledPredictableDeviationFiltration Env Action)
41       (sampledPredictableRealizedCompensatedProcess
42         arms eta gamma tilt loss) := by
43   have hdeviation :=
44     sampledPredictableRealizedDeviationProcess_stronglyAdapted
45       arms harms eta gamma hgamma_pos.le hgamma_le_one loss
46   have hvariance :=
47     (sampledPredictableRealizedVarianceProcess_isPredictable
48       arms harms eta gamma hgamma_pos hgamma_le_one loss).adapted
49   intro i
50   exact ((hdeviation i).const_mul tilt).sub
51     ((hvariance i).const_mul (tilt ^ 2))

53 theorem sampledPredictableRealizedCompensatedProcess_sum_range_succ
54     {Env : Type u} {Action : Type v}
55     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
56     (arms : Finset Action) (eta gamma tilt : Real)
57     (loss : PredictableLossVector Env Action)
58     (horizon : Nat) (sample : Env × ((k : Nat) → Action × Real)) :
59     (Finset.range (horizon + 1)).sum (fun i =>
60         sampledPredictableRealizedCompensatedProcess
61           arms eta gamma tilt loss i sample) =
62       tilt * (Finset.range horizon).sum (fun i =>
63           sampledTrajectoryRealizedDeviationAt
64             arms eta gamma loss i sample) -
65         tilt ^ 2 * (Finset.range horizon).sum (fun i =>
66           sampledTrajectoryPredictableRealizedVarianceAt
67             arms eta gamma loss i sample) := by
68   simp_rw [sampledPredictableRealizedCompensatedProcess]
69   rw [Finset.sum_sub_distrib, ← Finset.mul_sum, ← Finset.mul_sum,
70     sampledPredictableRealizedDeviationProcess_sum_range_succ,
71     sampledPredictableRealizedVarianceProcess_sum_range_succ]

73 /-- Fixed-tilt upper tail retaining the random cumulative selected-loss
74 predictable variance. -/
75 theorem sampledPredictableRealizedDeviation_sum_tail_predictableVariance_fixedTilt
76     {Env : Type u} {Action : Type v}
77     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
78     [MeasurableSpace Action] [MeasurableSingletonClass Action]
79     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
80     (prior : Measure Env) [IsProbabilityMeasure prior]
81     (arms : Finset Action) (harms : arms.Nonempty)
82     (eta gamma : Real) (hgamma_pos : 0 < gamma)
83     (hgamma_le_one : gamma ≤ 1)
84     (loss : PredictableLossVector Env Action)
85     (horizon : Nat) (tilt : Real) (htilt_nonneg : 0 ≤ tilt)
86     (htilt_le_one : tilt ≤ 1)
87     (threshold varianceBudget : Real) :
88     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
89       eta gamma hgamma_pos.le hgamma_le_one loss.environment
90     mu {sample |
91         threshold ≤ (Finset.range horizon).sum (fun i =>
92           sampledTrajectoryRealizedDeviationAt
93             arms eta gamma loss i sample) ∧
94         (Finset.range horizon).sum (fun i =>
95           sampledTrajectoryPredictableRealizedVarianceAt
96             arms eta gamma loss i sample) ≤ varianceBudget} ≤
97       ENNReal.ofReal (Real.exp
98         (-tilt * threshold + tilt ^ 2 * varianceBudget)) := by
99   dsimp only
100   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
101     eta gamma hgamma_pos.le hgamma_le_one loss.environment
102   let F := sampledPredictableDeviationFiltration Env Action
103   let Y := sampledPredictableRealizedDeviationProcess arms eta gamma loss
104   let V := sampledPredictableRealizedVarianceProcess arms eta gamma loss
105   let Z := sampledPredictableRealizedCompensatedProcess
106     arms eta gamma tilt loss
107   let psiZ : Nat → Real := fun _ => 0
108   letI : IsProbabilityMeasure mu := by
109     dsimp [mu]
110     infer_instance
111   have hadapted : StronglyAdapted F Z := by
112     simpa [F, Z] using
113       sampledPredictableRealizedCompensatedProcess_stronglyAdapted
114         arms harms eta gamma tilt hgamma_pos hgamma_le_one loss
115   have hzero : Concentration.HasMGFUpperBoundAt (Z 0) 1 (psiZ 0) mu := by
116     have hzero' : Concentration.HasMGFUpperBoundAt
117         (fun _ : Env × ((k : Nat) → Action × Real) => 0) 1 0 mu := by
118       constructor
119       · intro s
120         simp
121       · simp [ProbabilityTheory.mgf]
122     have hZ0 : Z 0 =
123         (fun _ : Env × ((k : Nat) → Action × Real) => 0) := by
124       funext sample
125       simp [Z, sampledPredictableRealizedCompensatedProcess,
126         sampledPredictableRealizedDeviationProcess,
127         sampledPredictableRealizedVarianceProcess]
128     rw [hZ0]
129     simpa [psiZ] using hzero'
130   have hcond : ∀ i, i < (horizon + 1) - 1 →
131       Concentration.HasCondMGFUpperBoundAt
132         (F i) (F.le i) (Z (i + 1)) 1 (psiZ (i + 1)) mu := by
133     intro i _hi
134     cases i with
135     | zero =>
136         simpa [mu, F, Z, psiZ,
137           sampledPredictableRealizedCompensatedProcess,
138           sampledPredictableRealizedDeviationProcess,
139           sampledPredictableRealizedVarianceProcess] using
140           (sampledPredictableRealizedCompensated_zero_hasCondMGFUpperBoundAt
141             prior arms harms eta gamma hgamma_pos hgamma_le_one loss
142               tilt htilt_nonneg htilt_le_one)
143     | succ n =>
144         simpa [mu, F, Z, psiZ,
145           sampledPredictableRealizedCompensatedProcess,
146           sampledPredictableRealizedDeviationProcess,
147           sampledPredictableRealizedVarianceProcess,
148           Nat.succ_eq_add_one, Nat.add_assoc] using
149           (sampledPredictableRealizedCompensated_succ_hasCondMGFUpperBoundAt
150             prior arms harms eta gamma hgamma_pos hgamma_le_one loss n
151               tilt htilt_nonneg htilt_le_one)
152   have htail :=
153     Concentration.measure_sum_ge_inter_sum_le_of_compensated_hasCondMGFUpperBoundAt
154       (μ := mu) (ℱ := F) Y V (horizon + 1) tilt (tilt ^ 2)
155         threshold varianceBudget
156         (by simpa [Y, V, Z, sampledPredictableRealizedCompensatedProcess]
157           using hadapted)
158         (by simpa [Y, V, Z, psiZ,
159           sampledPredictableRealizedCompensatedProcess] using hzero)
160         (by simpa [Y, V, Z, psiZ,
161           sampledPredictableRealizedCompensatedProcess] using hcond)
162         htilt_nonneg (sq_nonneg tilt)
163   simpa [Y, V, sampledPredictableRealizedDeviationProcess_sum_range_succ,
164     sampledPredictableRealizedVarianceProcess_sum_range_succ] using htail

166 /-- Bernstein radius for realized selected-loss deviation under a pathwise
167 predictable-variance budget. -/
168 noncomputable def sampledRealizedPredictableVarianceRadius
169     (varianceBudget delta : Real) : Real :=
170   let budget := max (Real.log (1 / delta)) 0
171   2 * Real.sqrt (varianceBudget * budget) + budget

173 theorem sampledPredictableRealizedDeviation_sum_tail_predictableVariance_delta
174     {Env : Type u} {Action : Type v}
175     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
176     [MeasurableSpace Action] [MeasurableSingletonClass Action]
177     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
178     (prior : Measure Env) [IsProbabilityMeasure prior]
179     (arms : Finset Action) (harms : arms.Nonempty)
180     (eta gamma : Real) (hgamma_pos : 0 < gamma)
181     (hgamma_le_one : gamma ≤ 1)
182     (loss : PredictableLossVector Env Action)
183     (horizon : Nat) (varianceBudget delta : Real)
184     (hvarianceBudget : 0 < varianceBudget) (hdelta : 0 < delta) :
185     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
186       eta gamma hgamma_pos.le hgamma_le_one loss.environment
187     mu {sample |
188         sampledRealizedPredictableVarianceRadius varianceBudget delta ≤
189           (Finset.range horizon).sum (fun i =>
190             sampledTrajectoryRealizedDeviationAt
191               arms eta gamma loss i sample) ∧
192         (Finset.range horizon).sum (fun i =>
193           sampledTrajectoryPredictableRealizedVarianceAt
194             arms eta gamma loss i sample) ≤ varianceBudget} ≤
195       ENNReal.ofReal delta := by
196   dsimp only
197   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
198     eta gamma hgamma_pos.le hgamma_le_one loss.environment
199   let deviation := fun sample : Env × ((k : Nat) -> Action × Real) =>
200     (Finset.range horizon).sum (fun i =>
201       sampledTrajectoryRealizedDeviationAt arms eta gamma loss i sample)
202   let predictableVariance := fun sample : Env × ((k : Nat) -> Action × Real) =>
203     (Finset.range horizon).sum (fun i =>
204       sampledTrajectoryPredictableRealizedVarianceAt
205         arms eta gamma loss i sample)
206   have htail :=
207     Concentration.measure_deviation_ge_inter_variance_le_delta_of_fixedTilt_quadratic_tail
208       mu deviation predictableVariance 1 varianceBudget 1 delta
209         (by norm_num) hvarianceBudget (by norm_num) hdelta (by
210           intro tilt htilt_nonneg htilt_le_one
211           simpa [mu, deviation, predictableVariance] using
212             (sampledPredictableRealizedDeviation_sum_tail_predictableVariance_fixedTilt
213               prior arms harms eta gamma hgamma_pos hgamma_le_one loss horizon
214                 tilt htilt_nonneg htilt_le_one
215                 (Concentration.quadraticFixedMGFRadius
216                   1 varianceBudget 1 delta) varianceBudget))
217   simpa [sampledRealizedPredictableVarianceRadius,
218     Concentration.quadraticFixedMGFRadius, mu, deviation,
219     predictableVariance] using htail
