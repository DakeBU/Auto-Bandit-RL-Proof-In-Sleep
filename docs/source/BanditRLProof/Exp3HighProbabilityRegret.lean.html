3 /-!
4 # Generated predictable EXP3 high-probability regret
5 
6 This module completes the finite-horizon high-probability pseudo-regret route
7 for the generated predictable EXP3 trajectory.  A pathwise reciprocal-floor
8 bound controls the random estimator-square sum almost surely.  The final theorem
9 then combines the sampled Hedge inequality, exploration bias, the pure-Hedge
10 predictable-minus-observed confidence event, and the comparator-estimator
11 confidence event by a two-event union bound.
12 
13 The resulting theorem is valid but uses range-based Hoeffding radii.  Their
14 `(|arms| / gamma)^2` per-round proxy is intentionally recorded rather than
15 silently presented as the ideal EXP3.P/Freedman rate.
16 -/

24 theorem observedMixedSquaredImportanceWeightedLossAt_le_inv_explorationFloor
25     {Env : Type u} {Action : Type v}
26     [MeasurableSpace Env]
27     [MeasurableSpace Action] [MeasurableSingletonClass Action]
28     [DecidableEq Action]
29     (arms : Finset Action) (harms : arms.Nonempty)
30     (eta gamma : Real) (hgamma_pos : 0 < gamma)
31     (hgamma_le_one : gamma <= 1)
32     (loss : PredictableLossVector Env Action) (t : Nat)
33     (sample : Env × ((k : Nat) -> Action × Real))
34     (hreward : (sample.2 t).2 ∈ Set.Icc (0 : Real) 1) :
35     observedMixedSquaredImportanceWeightedLossAt
36         arms eta gamma t sample <=
37       1 / (gamma / (arms.card : Real)) := by
38   let chosen := (sample.2 t).1
39   let prob := sampledTrajectoryProbabilityAt arms eta gamma t sample
40   let regularity := sampledPredictableTrajectoryLossRegularityAt
41     arms harms eta gamma hgamma_pos hgamma_le_one loss t
42   by_cases hchosen : chosen ∈ arms
43   · have hfloor := regularity.prob_floor sample chosen hchosen
44     have hepsilon := regularity.epsilon_pos
45     have hprob_pos : 0 < prob chosen := lt_of_lt_of_le hepsilon hfloor
46     have hsq : ((sample.2 t).2) ^ 2 <= 1 := by
47       nlinarith [hreward.1, hreward.2]
48     have hone_le : 1 <= (1 / (gamma / (arms.card : Real))) * prob chosen := by
49       rw [one_div_mul_eq_div, le_div_iff₀ hepsilon]
50       simpa [regularity] using hfloor
51     unfold observedMixedSquaredImportanceWeightedLossAt
52     rw [mixedSquaredImportanceWeightedLoss_eq_selectedLoss_sq_div
53       arms prob (fun _ => (sample.2 t).2) chosen hchosen hprob_pos.ne']
54     apply (div_le_iff₀ hprob_pos).2
55     exact hsq.trans hone_le
56   · unfold observedMixedSquaredImportanceWeightedLossAt
57     have hzero : mixedSquaredImportanceWeightedLoss arms prob
58         (fun _ => (sample.2 t).2) chosen = 0 := by
59       unfold mixedSquaredImportanceWeightedLoss
60       apply Finset.sum_eq_zero
61       intro candidate hcandidate
62       have hne : chosen ≠ candidate := by
63         intro heq
64         exact hchosen (heq ▸ hcandidate)
65       simp [importanceWeightedLoss, hne]
66     rw [hzero]
67     exact one_div_nonneg.mpr
68       (explorationFloor_pos arms harms gamma hgamma_pos).le

109       rw [hsample]

158 noncomputable def sampledPredictableHighProbabilityRegretBudget
159     {Action : Type v} (arms : Finset Action) (eta gamma : Real)
160     (horizon : Nat) (delta : Real) : Real :=
161   Real.log arms.card / eta +
162     (eta * (1 / (1 - gamma))) *
163       ((horizon : Real) * (1 / (gamma / (arms.card : Real)))) +
164     gamma * (horizon : Real) +
165     sampledPureObservedDeviationConfidenceRadius arms gamma horizon delta +
166     sampledComparatorEstimatorConfidenceRadius arms gamma horizon delta

168 /-- Generated predictable EXP3 pseudo-regret exceeds the explicit Hedge,
169 exploration, estimator-square, and two confidence-radius budget with probability
170 at most the sum of the pure-q and comparator failure probabilities. -/
171 theorem sampledPredictable_highProbabilityRegret_tail_delta
172     {Env : Type u} {Action : Type v}
173     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
174     [MeasurableSpace Action] [MeasurableSingletonClass Action]
175     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
176     (prior : Measure Env) [IsProbabilityMeasure prior]
177     (arms : Finset Action) (harms : arms.Nonempty)
178     (eta gamma : Real) (heta : 0 < eta)
179     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
180     (loss : PredictableLossVector Env Action)
181     (comparator : Action) (hcomparator : comparator ∈ arms)
182     (horizon : Nat) (hhorizon : 0 < horizon)
183     (delta : Real) (hdelta : 0 < delta) :
184     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
185       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
186     mu {sample |
187         sampledPredictableHighProbabilityRegretBudget
188             arms eta gamma horizon delta <=
189           (Finset.range horizon).sum (fun t =>
190               sampledTrajectoryExploredPredictableLossAt
191                 arms eta gamma loss t sample) -
192             (Finset.range horizon).sum (fun t =>
193               predictableLossAt loss t sample comparator)} <=
194       ENNReal.ofReal delta + ENNReal.ofReal delta := by
195   dsimp only
196   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
197     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
198   let purePredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
199     (Finset.range horizon).sum (fun t =>
200       sampledTrajectoryPurePredictableLossAt arms eta gamma loss t sample)
201   let pureObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
202     (Finset.range horizon).sum (fun t =>
203       sampledTrajectoryPureObservedLossAt arms eta gamma t sample)
204   let comparatorObserved := fun sample : Env × ((k : Nat) -> Action × Real) =>
205     (Finset.range horizon).sum (fun t =>
206       observedImportanceWeightedLossAt arms eta gamma t sample comparator)
207   let comparatorTrue := fun sample : Env × ((k : Nat) -> Action × Real) =>
208     (Finset.range horizon).sum (fun t =>
209       predictableLossAt loss t sample comparator)
210   let exploredPredictable := fun sample : Env × ((k : Nat) -> Action × Real) =>
211     (Finset.range horizon).sum (fun t =>
212       sampledTrajectoryExploredPredictableLossAt
213         arms eta gamma loss t sample)
214   let secondMoment := fun sample : Env × ((k : Nat) -> Action × Real) =>
215     (Finset.range horizon).sum (fun t =>
216       observedMixedSquaredImportanceWeightedLossAt arms eta gamma t sample)
217   let pureRadius := sampledPureObservedDeviationConfidenceRadius
218     arms gamma horizon delta
219   let comparatorRadius := sampledComparatorEstimatorConfidenceRadius
220     arms gamma horizon delta
221   let pureBad : Set (Env × ((k : Nat) -> Action × Real)) := {sample | pureRadius <=
222     purePredictable sample - pureObserved sample}
223   let comparatorBad : Set (Env × ((k : Nat) -> Action × Real)) := {sample | comparatorRadius <=
224     comparatorObserved sample - comparatorTrue sample}
225   have hpureTail : mu pureBad <= ENNReal.ofReal delta := by
226     have h := sampledPurePredictableMinusObserved_sum_tail_delta
227       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss
228         horizon hhorizon delta hdelta
229     dsimp only at h
230     simpa [mu, pureBad, pureRadius, purePredictable, pureObserved] using h
231   have hcomparatorTail : mu comparatorBad <= ENNReal.ofReal delta := by
232     have h := sampledObservedComparatorEstimatorDeviation_sum_tail_delta
233       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss comparator
234         hcomparator horizon hhorizon delta hdelta
235     dsimp only at h
236     simpa [mu, comparatorBad, comparatorRadius, comparatorObserved,
237       comparatorTrue] using h
238   have hhedge :=
239     sampledPredictableTrajectoryMeasure_hedge_exploredSecondMoment_le_ae
240       prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss horizon
241         comparator hcomparator
242   dsimp only at hhedge
243   have hsecond :=
244     sampledPredictableTrajectoryMeasure_observedMixedSquared_sum_le_ae
245       prior arms harms eta gamma hgamma_pos hgamma_lt_one.le loss horizon
246   dsimp only at hsecond
247   have hsubset : ∀ᵐ sample ∂mu,
248       sampledPredictableHighProbabilityRegretBudget
249           arms eta gamma horizon delta <=
250         exploredPredictable sample - comparatorTrue sample ->
251       sample ∈ pureBad ∪ comparatorBad := by
252     filter_upwards [hhedge, hsecond] with sample hsampleHedge hsampleSecond
253     intro hregret
254     by_cases hpure : sample ∈ pureBad
255     · exact Or.inl hpure
256     by_cases hcomparator : sample ∈ comparatorBad
257     · exact Or.inr hcomparator
258     exfalso
259     have hpureGood :
260         purePredictable sample - pureObserved sample < pureRadius := by
261       change ¬ pureRadius <=
262         purePredictable sample - pureObserved sample at hpure
263       exact lt_of_not_ge hpure
264     have hcomparatorGood :
265         comparatorObserved sample - comparatorTrue sample <
266           comparatorRadius := by
267       change ¬ comparatorRadius <=
268         comparatorObserved sample - comparatorTrue sample at hcomparator
269       exact lt_of_not_ge hcomparator
270     have hexploration :
271         exploredPredictable sample <=
272           purePredictable sample + gamma * (horizon : Real) := by
273       have h := (sampledTrajectory_finiteHorizon_explorationBias_secondMoment
274         arms harms eta gamma hgamma_pos.le hgamma_lt_one loss horizon sample).1
275       simpa [exploredPredictable, purePredictable,
276         sampledTrajectoryExploredPredictableLossAt,
277         sampledTrajectoryPurePredictableLossAt,
278         sampledTrajectoryPureProbabilityAt] using h
279     have hcoef_nonneg : 0 <= eta * (1 / (1 - gamma)) :=
280       mul_nonneg heta.le (one_div_pos.mpr (sub_pos.mpr hgamma_lt_one)).le
281     have hweightedSecond :
282         (eta * (1 / (1 - gamma))) * secondMoment sample <=
283           (eta * (1 / (1 - gamma))) *
284             ((horizon : Real) *
285               (1 / (gamma / (arms.card : Real)))) :=
286       mul_le_mul_of_nonneg_left
287         (by simpa [secondMoment] using hsampleSecond) hcoef_nonneg
288     have hhedge' :
289         pureObserved sample - comparatorObserved sample <=
290           Real.log arms.card / eta +
291             (eta * (1 / (1 - gamma))) * secondMoment sample := by
292       simpa [pureObserved, comparatorObserved, secondMoment] using hsampleHedge
293     have hfinal : exploredPredictable sample - comparatorTrue sample <
294         sampledPredictableHighProbabilityRegretBudget
295           arms eta gamma horizon delta := by
296       dsimp [sampledPredictableHighProbabilityRegretBudget,
297         pureRadius, comparatorRadius]
298       linarith
299     exact (not_lt_of_ge hregret) hfinal
300   calc
301     mu {sample |
302         sampledPredictableHighProbabilityRegretBudget
303             arms eta gamma horizon delta <=
304           (Finset.range horizon).sum (fun t =>
305               sampledTrajectoryExploredPredictableLossAt
306                 arms eta gamma loss t sample) -
307             (Finset.range horizon).sum (fun t =>
308               predictableLossAt loss t sample comparator)} <=
309         mu (pureBad ∪ comparatorBad) := by
310       apply measure_mono_ae
311       simpa [exploredPredictable, comparatorTrue, mu] using hsubset
312     _ <= mu pureBad + mu comparatorBad := measure_union_le _ _
313     _ <= ENNReal.ofReal delta + ENNReal.ofReal delta :=
314       add_le_add hpureTail hcomparatorTail

316 /-- Standard total-failure-probability form of the generated predictable EXP3
317 pseudo-regret bound. Each of the two confidence events receives `delta / 2`,
318 so their union has probability at most `delta`. -/
319 theorem sampledPredictable_highProbabilityRegret_tail_total_delta
320     {Env : Type u} {Action : Type v}
321     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
322     [MeasurableSpace Action] [MeasurableSingletonClass Action]
323     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
324     (prior : Measure Env) [IsProbabilityMeasure prior]
325     (arms : Finset Action) (harms : arms.Nonempty)
326     (eta gamma : Real) (heta : 0 < eta)
327     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
328     (loss : PredictableLossVector Env Action)
329     (comparator : Action) (hcomparator : comparator ∈ arms)
330     (horizon : Nat) (hhorizon : 0 < horizon)
331     (delta : Real) (hdelta : 0 < delta) :
332     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
333       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
334     mu {sample |
335         sampledPredictableHighProbabilityRegretBudget
336             arms eta gamma horizon (delta / 2) <=
337           (Finset.range horizon).sum (fun t =>
338               sampledTrajectoryExploredPredictableLossAt
339                 arms eta gamma loss t sample) -
340             (Finset.range horizon).sum (fun t =>
341               predictableLossAt loss t sample comparator)} <=
342       ENNReal.ofReal delta := by
343   have h := sampledPredictable_highProbabilityRegret_tail_delta
344     prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss comparator
345       hcomparator horizon hhorizon (delta / 2) (half_pos hdelta)
346   dsimp only at h ⊢
347   rw [← ENNReal.ofReal_add (half_pos hdelta).le (half_pos hdelta).le,
348     add_halves] at h
349   exact h
