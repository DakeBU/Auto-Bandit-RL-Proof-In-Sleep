4 # Explicit finite-window OFUL expected pseudo-regret rate
5 
6 This module normalizes the horizon-tuned confidence parameter used by the
7 canonical OFUL expected-regret theorem and exposes its logarithmic
8 square-root bound without hiding the rate inside the radius-width definitions.
9 -/

17 /--
18 The log budget in the horizon-tuned expected-regret radius: the standard
19 log-determinant term plus the `4 * log (T+1)` confidence contribution.
20 -/
21 noncomputable def standardExpectedRegretLogBudget
22     {Feature : Type u} [Fintype Feature]
23     (lambda : Real) (horizon : Nat) (L2 : Real) : Real :=
24   standardScalarLogDetBudget
25       (Feature := Feature) lambda (horizon + 1) L2 +
26     4 * Real.log (((horizon + 1 : Nat) : Real))
27 
28 /--
29 The outer failure budget `1/(T+1)`, divided once more by the uniform-time
30 schedule size, is exactly the algorithm parameter `1/(T+1)^2`.
31 -/
32 theorem standardExpectedRegretAlgorithmDelta_eq_inv_sq
33     (horizon : Nat) :
34     standardExpectedRegretDelta horizon /
35         (((horizon + 1 : Nat) : Real)) =
36       1 / (((horizon + 1 : Nat) : Real) ^ 2) := by
37   unfold standardExpectedRegretDelta
38   ring
39 
40 /--
41 At the canonical horizon-tuned algorithm parameter, the standard confidence
42 radius has the explicit form
43 `R * sqrt (B_T + 4 * log (T+1)) + sqrt lambda * S`.
44 -/
45 theorem standardScalarConfidenceRadiusUpper_standardExpectedRegret
46     {Feature : Type u}
47     [Fintype Feature] [Nonempty Feature]
48     (R : Real) (hR : 0 < R)
49     (lambda : Real) (hlambda : 0 < lambda)
50     (S : Real)
51     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2) :
52     standardScalarConfidenceRadiusUpper
53         (Feature := Feature) R
54         (standardExpectedRegretDelta horizon /
55           (((horizon + 1 : Nat) : Real)))
56         lambda S (horizon + 1) L2 =
57       R * Real.sqrt
58           (standardExpectedRegretLogBudget
59             (Feature := Feature) lambda horizon L2) +
60         Real.sqrt lambda * S := by
61   let n : Real := (((horizon + 1 : Nat) : Real))
62   let B : Real :=
63     standardScalarLogDetBudget
64       (Feature := Feature) lambda (horizon + 1) L2
65   have hn_pos : 0 < n := by
66     dsimp [n]
67     positivity
68   have hn_one : 1 <= n := by
69     dsimp [n]
70     exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
71   have hB : 0 <= B := by
72     exact standardScalarLogDetBudget_nonneg
73       (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
74   have hlogn : 0 <= Real.log n := Real.log_nonneg hn_one
75   have hX : 0 <= B + 4 * Real.log n := by positivity
76   have hdelta :
77       standardExpectedRegretDelta horizon / n = 1 / n ^ 2 := by
78     dsimp [n]
79     exact standardExpectedRegretAlgorithmDelta_eq_inv_sq horizon
80   have hlog :
81       Real.log (Real.sqrt (Real.exp B) /
82           (standardExpectedRegretDelta horizon / n)) =
83         B / 2 + 2 * Real.log n := by
84     rw [hdelta]
85     have hquot :
86         Real.sqrt (Real.exp B) / (1 / n ^ 2) =
87           Real.sqrt (Real.exp B) * n ^ 2 := by
88       field_simp
89     rw [hquot]
90     rw [Real.log_mul (Real.sqrt_pos.2 (Real.exp_pos B)).ne'
91       (pow_ne_zero 2 hn_pos.ne')]
92     rw [Real.log_sqrt (Real.exp_pos B).le, Real.log_exp, Real.log_pow]
93     ring
94   unfold standardScalarConfidenceRadiusUpper
95   change Real.sqrt
96         (2 * R ^ 2 *
97           Real.log
98             (Real.sqrt (Real.exp B) /
99               (standardExpectedRegretDelta horizon / n))) +
100       Real.sqrt lambda * S =
101     R * Real.sqrt (B + 4 * Real.log n) + Real.sqrt lambda * S
102   rw [hlog]
103   congr 1
104   have hrad :
105       2 * R ^ 2 * (B / 2 + 2 * Real.log n) =
106         R ^ 2 * (B + 4 * Real.log n) := by ring
107   rw [hrad, Real.sqrt_mul (sq_nonneg R), Real.sqrt_sq hR.le]

109 /-- Explicit logarithmic square-root bound for expected OFUL pseudo-regret. -/
110 noncomputable def standardExpectedPseudoRegretBound
111     {Feature : Type u} [Fintype Feature]
112     (R lambda S : Real) (horizon : Nat) (L2 : Real) : Real :=
113   4 * S * Real.sqrt L2 +
114     2 *
115       (R * Real.sqrt
116           (standardExpectedRegretLogBudget
117             (Feature := Feature) lambda horizon L2) +
118         Real.sqrt lambda * S) *
119       (Real.sqrt (((horizon + 1 : Nat) : Real)) *
120         Real.sqrt
121           (2 *
122             standardScalarLogDetBudget
123               (Feature := Feature) lambda (horizon + 1) L2))

146 /--
147 Explicit finite-window expected pseudo-regret theorem for the canonical
148 scalar-ridge OFUL trajectory. The algorithm parameter is displayed directly
149 as `1/(T+1)^2`, and the upper bound is the logarithmic square-root expression
150 `standardExpectedPseudoRegretBound`.
151 -/
152 theorem
153     integral_canonicalHistoryTrajectoryPseudoRegret_nonneg_and_le_explicitStandardExpectedBound_of_linearSubgaussianEnvironment_of_featureBound_le_regularization
154     {K : Nat} {Feature : Type u}
155     [Fintype Feature] [DecidableEq Feature] [Nonempty Feature]
156     (hK : 0 < K)
157     (lambda : Real) (hlambda : 0 < lambda)
158     (thetaStar : Feature -> Real)
159     (actionFeature : Fin K -> Feature -> Real)
160     (R : Real) (hR : 0 < R)
161     (S : Real) (hS : 0 <= S)
162     (environment : Thompson.HistoryEnvironment (Fin K) Real)
163     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2)
164     (hactionFeatureBound : forall action,
165       dotProduct (actionFeature action) (actionFeature action) <= L2)
166     (hL2lambda : L2 <= lambda)
167     (best : Fin K)
168     (hbest : IsOptimalLinearArm thetaStar actionFeature best)
169     (source : CanonicalLinearSubgaussianEnvironmentLaw
170       hK thetaStar actionFeature R S environment) :
171     let algorithmDelta :=
172       1 / (((horizon + 1 : Nat) : Real) ^ 2)
173     let expectedPseudoRegret :=
174       integral
175         (Thompson.canonicalHistoryTrajectoryMeasure
176           (finiteHistoryScalarRidgeOptimisticAlgorithm
177             hK lambda actionFeature R algorithmDelta S)
178           environment)
179         (canonicalHistoryTrajectorySumRangeAllGap
180           thetaStar actionFeature horizon (fun _t => best))
181     0 <= expectedPseudoRegret ∧
182       expectedPseudoRegret <=
183         standardExpectedPseudoRegretBound
184           (Feature := Feature) R lambda S horizon L2 := by
185   dsimp only
186   have h :=
187     integral_canonicalHistoryTrajectoryPseudoRegret_nonneg_and_le_standardExpectedBound_of_linearSubgaussianEnvironment_of_featureBound_le_regularization

189       S hS environment horizon L2 hL2 hactionFeatureBound hL2lambda
190       best hbest source
191   dsimp only at h
192   rw [standardExpectedRegretAlgorithmDelta_eq_inv_sq,
193     standardScalarAllRoundGapBound_add_initial_eq_standardExpectedPseudoRegretBound
