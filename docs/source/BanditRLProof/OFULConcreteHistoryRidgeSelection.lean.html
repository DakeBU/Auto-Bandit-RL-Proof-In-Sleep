20 /-- A determinant is measurable when every random matrix entry is measurable. -/
21 theorem measurable_matrix_det_of_apply
22     {Omega Index : Type*} [MeasurableSpace Omega]
23     [Fintype Index] [DecidableEq Index]
24     (A : Omega -> Matrix Index Index Real)
25     (hA : forall i j, Measurable (fun omega => A omega i j)) :
26     Measurable (fun omega => Matrix.det (A omega)) := by
27   classical
28   simp_rw [Matrix.det_apply']
29   refine Finset.measurable_sum _ ?_
30   intro sigma _hsigma
31   exact measurable_const.mul <|
32     Finset.measurable_prod _ fun i _hi => hA (sigma i) i

34 /-- Every adjugate entry is measurable under coordinatewise matrix measurability. -/
35 theorem measurable_matrix_adjugate_apply_of_apply
36     {Omega Index : Type*} [MeasurableSpace Omega]
37     [Fintype Index] [DecidableEq Index]
38     (A : Omega -> Matrix Index Index Real)
39     (hA : forall i j, Measurable (fun omega => A omega i j))
40     (i j : Index) :
41     Measurable (fun omega => Matrix.adjugate (A omega) i j) := by
42   rw [show (fun omega => Matrix.adjugate (A omega) i j) =
43       fun omega => Matrix.det ((A omega).updateRow j (Pi.single i 1)) by
44     funext omega
45     exact Matrix.adjugate_apply (A omega) i j]
46   apply measurable_matrix_det_of_apply
47   intro row col
48   by_cases hrow : row = j
49   · subst row
50     simpa using measurable_const
51   · simpa [Matrix.updateRow_apply, hrow] using hA row col

53 /--
54 Every entry of Mathlib's nonsingular inverse is measurable.  The proof uses
55 the determinant/adjugate formula, including its zero-at-singular convention.
56 -/
57 theorem measurable_matrix_nonsingInv_apply_of_apply
58     {Omega Index : Type*} [MeasurableSpace Omega]
59     [Fintype Index] [DecidableEq Index]
60     (A : Omega -> Matrix Index Index Real)
61     (hA : forall i j, Measurable (fun omega => A omega i j))
62     (i j : Index) :
63     Measurable (fun omega => (A omega)⁻¹ i j) := by
64   have hdet : Measurable (fun omega => Matrix.det (A omega)) :=
65     measurable_matrix_det_of_apply A hA
66   have hadjugate :
67       Measurable (fun omega => Matrix.adjugate (A omega) i j) :=
68     measurable_matrix_adjugate_apply_of_apply A hA i j
69   simpa only [Matrix.inv_def, Pi.smul_apply, smul_eq_mul,
70     Ring.inverse_eq_inv] using hdet.inv.mul hadjugate
