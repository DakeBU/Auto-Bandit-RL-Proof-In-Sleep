4 /-!
5 # All-time predictable EXP3 regret
6 
7 This module gives every positive prefix of one fixed generated EXP3 process a
8 geometric confidence share and reuses the compiled fixed-horizon pathwise
9 potential, exploration, and comparator assembly. The result is countable
10 outer-measure subadditivity, not a Ville/Doob, mixture, optional-stopping,
11 self-normalized, general Freedman, or tuned horizon-free EXP3 theorem.
12 -/
13

21 /-- The fixed-horizon predictable-regret budget at prefix `n+1`. The outer
22 geometric share is divided by two because the parent total-delta theorem
23 allocates equal shares to its pure-cross and comparator-estimator events. -/
24 noncomputable def sampledPredictableRegretGeometricAllTimeBudget
25     {Action : Type v} (arms : Finset Action) (eta gamma delta : Real)
26     (n : Nat) : Real :=
27   sampledPredictableHighProbabilityRegretBudget arms eta gamma (n + 1)
28     (Concentration.geometricConfidenceShare delta n / 2)
29 
30 /-- Countable predictable-regret failure event over every positive prefix of
31 one generated EXP3 trajectory. -/
32 noncomputable def sampledPredictableRegretGeometricAllTimeFailureSet
33     {Env : Type u} {Action : Type v}
34     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
35     (arms : Finset Action) (eta gamma : Real)
36     (loss : PredictableLossVector Env Action) (comparator : Action)
37     (delta : Real) : Set (Env × ((k : Nat) -> Action × Real)) :=
38   ⋃ n, {sample |
39       sampledPredictableRegretGeometricAllTimeBudget
40             arms eta gamma delta n <=
41           (Finset.range (n + 1)).sum (fun t =>
42             sampledTrajectoryExploredPredictableLossAt
43               arms eta gamma loss t sample) -
44         (Finset.range (n + 1)).sum (fun t =>
45           predictableLossAt loss t sample comparator)}
46 
47 /-- Membership is predictable-regret failure at at least one positive
48 prefix. -/
49 theorem mem_sampledPredictableRegretGeometricAllTimeFailureSet_iff
50     {Env : Type u} {Action : Type v}
51     [MeasurableSpace Env] [MeasurableSpace Action] [DecidableEq Action]
52     (arms : Finset Action) (eta gamma : Real)
53     (loss : PredictableLossVector Env Action) (comparator : Action)
54     (delta : Real) (sample : Env × ((k : Nat) -> Action × Real)) :
55     sample ∈ sampledPredictableRegretGeometricAllTimeFailureSet
56         arms eta gamma loss comparator delta ↔
57       ∃ n,
58         sampledPredictableRegretGeometricAllTimeBudget
59               arms eta gamma delta n <=
60             (Finset.range (n + 1)).sum (fun t =>
61               sampledTrajectoryExploredPredictableLossAt
62                 arms eta gamma loss t sample) -
63           (Finset.range (n + 1)).sum (fun t =>
64             predictableLossAt loss t sample comparator) := by
65   simp [sampledPredictableRegretGeometricAllTimeFailureSet]

67 /-- On one fixed generated EXP3 process and against one fixed supported
68 comparator, predictable-regret failures over all positive prefixes have outer
69 measure at most the geometric confidence budget. -/
70 theorem measure_sampledPredictableRegretGeometricAllTimeFailureSet_le
71     {Env : Type u} {Action : Type v}
72     [MeasurableSpace Env] [StandardBorelSpace Env] [Nonempty Env]
73     [MeasurableSpace Action] [MeasurableSingletonClass Action]
74     [StandardBorelSpace Action] [Nonempty Action] [DecidableEq Action]
75     (prior : Measure Env) [IsProbabilityMeasure prior]
76     (arms : Finset Action) (harms : arms.Nonempty)
77     (eta gamma : Real) (heta : 0 < eta)
78     (hgamma_pos : 0 < gamma) (hgamma_lt_one : gamma < 1)
79     (loss : PredictableLossVector Env Action)
80     (comparator : Action) (hcomparator : comparator ∈ arms)
81     (delta : Real) (hdelta : 0 < delta) :
82     let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
83       eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
84     mu (sampledPredictableRegretGeometricAllTimeFailureSet
85         arms eta gamma loss comparator delta) <= ENNReal.ofReal delta := by
86   dsimp only
87   let mu := prior ⊗ₘ sampledImportanceWeightedTrajectoryKernel arms harms
88     eta gamma hgamma_pos.le hgamma_lt_one.le loss.environment
89   calc
90     mu (sampledPredictableRegretGeometricAllTimeFailureSet
91         arms eta gamma loss comparator delta) <=
92         ∑' n, mu {sample |
93           sampledPredictableRegretGeometricAllTimeBudget
94                 arms eta gamma delta n <=
95               (Finset.range (n + 1)).sum (fun t =>
96                 sampledTrajectoryExploredPredictableLossAt
97                   arms eta gamma loss t sample) -
98             (Finset.range (n + 1)).sum (fun t =>
99               predictableLossAt loss t sample comparator)} := by
100       unfold sampledPredictableRegretGeometricAllTimeFailureSet
101       exact MeasureTheory.measure_iUnion_le _
102     _ <= ∑' n, ENNReal.ofReal
103         (Concentration.geometricConfidenceShare delta n) := by
104       apply ENNReal.tsum_le_tsum
105       intro n
106       have h := sampledPredictable_highProbabilityRegret_tail_total_delta
107         prior arms harms eta gamma heta hgamma_pos hgamma_lt_one loss
108           comparator hcomparator (n + 1) (Nat.succ_pos n)
109           (Concentration.geometricConfidenceShare delta n)
110           (Concentration.geometricConfidenceShare_pos hdelta n)
111       dsimp only at h
112       simpa [mu, sampledPredictableRegretGeometricAllTimeBudget] using h
113     _ = ENNReal.ofReal delta :=
114       Concentration.tsum_ofReal_geometricConfidenceShare hdelta.le
