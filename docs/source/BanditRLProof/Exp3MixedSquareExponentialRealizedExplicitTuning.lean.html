4 /-!
5 # Explicit exploration tuning for exponential-square realized EXP3
6 
7 This module closes the remaining exploration-parameter leaf in the generated
8 exponential mixed-square route.  The interval sub-Gaussian proxy contributes a
9 sixth-root scale because its range is `|arms| / gamma`; the two Bernstein
10 radii contribute a cube-root scale, and the realized deviation contributes a
11 square-root scale.  The resulting rate is deliberately recorded as the output
12 of the current Hoeffding-proxy route, not as a Freedman or EXP3.P rate.
13 -/

32   rw [Real.norm_of_nonneg]
33   · field_simp [ne_of_gt hgamma_pos]
34     ring
35   · simpa using one_div_nonneg.mpr hfloor_nonneg

37 /-- The logarithmically weighted mixed-square confidence radius is controlled
38 by `gamma^2 T^2` under the sixth-power dominance contract. -/
39 theorem log_mul_sampledMixedSquaredConfidenceRadius_le_sq_mul_horizon_sq

45             Real.log (4 / delta) / 2 <=

47     Real.log (arms.card : Real) *
48         sampledMixedSquaredConfidenceRadius arms gamma horizon (delta / 4) <=
49       gamma ^ 2 * (horizon : Real) ^ 2 := by
50   let K : Real := arms.card
51   let T : Real := horizon

57     exact_mod_cast hhorizon
58   have hK_nonneg : 0 <= K := by positivity

60     by_cases hK : K = 0
61     · simp [hK]
62     · exact Real.log_nonneg (by
63         have hK_nat : 1 <= arms.card := Nat.one_le_iff_ne_zero.mpr (by
64           intro hcard
65           apply hK
66           simp [K, hcard])
67         dsimp [K]
68         exact_mod_cast hK_nat)

70     rw [lt_div_iff₀ hdelta]
71     nlinarith

74     dsimp [L]

79   have hradicand :
80       0 <= 2 * (T * ((sampledMixedSquaredVarianceProxy arms gamma : NNReal) : Real)) * L := by
81     positivity
82   have hradius_nonneg : 0 <= radius := by

84     exact Real.sqrt_nonneg _
85   have hradius_sq :

87         2 * (T * ((sampledMixedSquaredVarianceProxy arms gamma : NNReal) : Real)) * L := by

90     rw [hlog_fourth, Real.sq_sqrt hradicand]

93   have hright_nonneg : 0 <= gamma ^ 2 * T ^ 2 := by positivity

97     have hgamma_ne : gamma ≠ 0 := ne_of_gt hgamma_pos
98     have hT_nonneg : 0 <= (horizon : Real) := hT.le
99     field_simp [hgamma_ne]
100     nlinarith [hmixed]
101   · simpa [K, T, radius] using hright_nonneg

354 /-- A nonnegative sixth-root scale is at most one half when its numerator is
355 at most one sixty-fourth of its positive denominator. -/
356 theorem rpow_inv_six_le_half_of_sixtyfour_mul_le
357     (numerator denominator : Real) (hnumerator : 0 <= numerator)
358     (hdenominator : 0 < denominator)
359     (hlarge : 64 * numerator <= denominator) :
360     (numerator / denominator) ^ (6 : Real)⁻¹ <= 1 / 2 := by
361   have hsix :
362       ((numerator / denominator) ^ (6 : Real)⁻¹) ^ (6 : Nat) =
363         numerator / denominator := by
364     convert Real.rpow_inv_natCast_pow (n := 6)
365       (div_nonneg hnumerator hdenominator.le) (by norm_num) using 1
366   apply le_of_pow_le_pow_left₀ (n := 6) (by positivity) (by norm_num)
367   rw [hsix]
368   norm_num
369   rw [div_le_iff₀ hdenominator]
370   nlinarith
