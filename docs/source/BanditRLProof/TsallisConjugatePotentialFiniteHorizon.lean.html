6 /-!
7 # Finite-horizon half-Tsallis conjugate-potential stability
8 
9 This module transports the compiled ordinary-importance-weighted
10 conjugate-potential bound through identified finite conditional action laws and
11 sums it over a finite horizon.  It also records the exact deterministic
12 potential telescope.  The potential includes the paper-normalizing 1 / eta,
13 so the process is ready for later cross-learning-rate comparisons.
14 -/

31 Exact fixed-learning-rate finite-horizon telescope for the candidate potential
32 expression.  When every probability is the corresponding certified minimizer,
33 this is the constrained-potential telescope.

35 theorem sum_halfTsallisPotentialStability_eq_linearLoss_sum_add_terminal_sub_initial
36     {Action : Type u}
37     (arms : Finset Action) (eta : Real)

39     (T : Nat)
40     (hscoreSucc : forall t, t < T ->
41       score (t + 1) = fun action => score t action + estimate t action) :

43         halfTsallisPotentialStability arms eta
44           (score t) (probability t) (estimate t) (probability (t + 1))) =
45       (Finset.range T).sum (fun t =>
46         FTRL.linearLoss arms (probability t) (estimate t)) +
47         halfTsallisPotentialProcess arms eta score probability T -
48         halfTsallisPotentialProcess arms eta score probability 0 := by
49   have hpotential :
50       (Finset.range T).sum (fun t =>
51           halfTsallisPotentialValue arms eta
52               (fun action => score t action + estimate t action)
53               (probability (t + 1)) -
54             halfTsallisPotentialValue arms eta
55               (score t) (probability t)) =
56         (Finset.range T).sum (fun t =>
57           halfTsallisPotentialProcess arms eta score probability (t + 1) -
58             halfTsallisPotentialProcess arms eta score probability t) := by
59     apply Finset.sum_congr rfl
60     intro t ht
61     unfold halfTsallisPotentialProcess
62     rw [hscoreSucc t (Finset.mem_range.mp ht)]
63   calc
64     (Finset.range T).sum (fun t =>
65         halfTsallisPotentialStability arms eta
66           (score t) (probability t) (estimate t) (probability (t + 1))) =
67       (Finset.range T).sum (fun t =>
68           FTRL.linearLoss arms (probability t) (estimate t)) +
69         (Finset.range T).sum (fun t =>
70           halfTsallisPotentialValue arms eta
71               (fun action => score t action + estimate t action)
72               (probability (t + 1)) -
73             halfTsallisPotentialValue arms eta
74               (score t) (probability t)) := by
75       rw [← Finset.sum_add_distrib]
76       apply Finset.sum_congr rfl
77       intro t _ht
78       unfold halfTsallisPotentialStability
79       ring
80     _ = (Finset.range T).sum (fun t =>
81           FTRL.linearLoss arms (probability t) (estimate t)) +

94 noncomputable def importanceWeightedPotentialStabilityScore
95     {History : Type u} {Action : Type v}
96     (arms : Finset Action) (eta : Real)

106 /-- The paper-shaped one-round conditional expectation budget. -/
107 noncomputable def refinedPotentialStabilityBound
108     {History : Type u} {Action : Type v}
109     (arms : Finset Action) (eta : Real)
110     (prob : History -> Action -> Real) (history : History) : Real :=
111   eta * arms.sum (fun action =>
112       Real.sqrt (prob history action) * (1 - prob history action)) +
113     2 * eta ^ 2

129 /-- The paper-normalized potential is measurable from supported coordinate
130 measurability of its score and simplex candidate. -/
131 theorem measurable_halfTsallisPotentialValue
132     {History : Type u} {Action : Type v} [MeasurableSpace History]
133     (arms : Finset Action) (eta : Real)
134     (score probability : History -> Action -> Real)
135     (hscore : forall action, action ∈ arms ->
136       Measurable (fun history => score history action))
137     (hprobability : forall action, action ∈ arms ->
138       Measurable (fun history => probability history action)) :
139     Measurable (fun history =>
140       halfTsallisPotentialValue arms eta
141         (score history) (probability history)) := by
142   have hlinear := measurable_linearLoss_of_coordinatewise
143     arms probability score hprobability hscore
144   have hsqrt : Measurable (fun history =>
145       arms.sum (fun action => Real.sqrt (probability history action))) :=
146     Finset.measurable_sum arms fun action haction =>
147       (hprobability action haction).sqrt
148   have hobjective : Measurable (fun history =>
149       eta * FTRL.linearLoss arms (probability history) (score history) -
150         2 * arms.sum (fun action =>
151           Real.sqrt (probability history action)) + 2) :=
152     ((measurable_const.mul hlinear).sub (measurable_const.mul hsqrt)).add
153       measurable_const
154   have hformula : (fun history =>
155       halfTsallisPotentialValue arms eta
156         (score history) (probability history)) =
157       fun history =>
158         -(eta * FTRL.linearLoss arms (probability history) (score history) -
159             2 * arms.sum (fun action =>
160               Real.sqrt (probability history action)) + 2) / eta +
161           1 / eta := by
162     funext history
163     rw [halfTsallisPotentialValue,
164       regularizedObjective_half_eq]
165   rw [hformula]
166   exact (hobjective.neg.div measurable_const).add measurable_const

169 /-- The conjugate-potential score is measurable from supported current score,
170 probability, loss, and updated-selector coordinates. -/
171 theorem measurable_importanceWeightedPotentialStabilityScore
172     {History : Type u} {Action : Type v}
173     [MeasurableSpace History] [MeasurableSpace Action]
174     [MeasurableSingletonClass Action] [DecidableEq Action]
175     (arms : Finset Action) (eta : Real)
176     (score prob loss : History -> Action -> Real)
177     (next : History -> Action -> Action -> Real)
178     (hscore : forall candidate, candidate ∈ arms ->
179       Measurable (fun history => score history candidate))
180     (hprob : forall candidate, candidate ∈ arms ->
181       Measurable (fun history => prob history candidate))
182     (hloss : forall candidate, candidate ∈ arms ->
183       Measurable (fun history => loss history candidate))
184     (hnext : forall candidate, candidate ∈ arms ->
185       Measurable (fun sample : History × Action =>
186         next sample.1 sample.2 candidate)) :
187     Measurable (importanceWeightedPotentialStabilityScore
188       arms eta score prob loss next) := by
189   have hestimate : forall candidate, candidate ∈ arms ->
190       Measurable (fun sample : History × Action =>
191         Exp3.importanceWeightedLoss (prob sample.1) (loss sample.1)
192           sample.2 candidate) := by
193     intro candidate hcandidate
194     unfold Exp3.importanceWeightedLoss
195     refine Measurable.ite ?_
196       (((hloss candidate hcandidate).comp measurable_fst).div
197         ((hprob candidate hcandidate).comp measurable_fst))
198       measurable_const
199     simpa only [Set.mem_setOf_eq] using
200       measurable_snd (measurableSet_singleton candidate)
201   have hcurrentPotential : Measurable (fun sample : History × Action =>
202       halfTsallisPotentialValue arms eta
203         (score sample.1) (prob sample.1)) :=
204     measurable_halfTsallisPotentialValue arms eta
205       (fun sample : History × Action => score sample.1)
206       (fun sample : History × Action => prob sample.1)
207       (fun candidate hcandidate =>
208         (hscore candidate hcandidate).comp measurable_fst)
209       (fun candidate hcandidate =>
210         (hprob candidate hcandidate).comp measurable_fst)
211   have hnextPotential : Measurable (fun sample : History × Action =>
212       halfTsallisPotentialValue arms eta
213         (fun candidate => score sample.1 candidate +
214           Exp3.importanceWeightedLoss
215             (prob sample.1) (loss sample.1) sample.2 candidate)
216         (next sample.1 sample.2)) :=
217     measurable_halfTsallisPotentialValue arms eta
218       (fun (sample : History × Action) candidate => score sample.1 candidate +
219         Exp3.importanceWeightedLoss
220           (prob sample.1) (loss sample.1) sample.2 candidate)
221       (fun (sample : History × Action) candidate =>
222         next sample.1 sample.2 candidate)
223       (fun candidate hcandidate =>
224         ((hscore candidate hcandidate).comp measurable_fst).add
225           (hestimate candidate hcandidate))
226       hnext
227   have hlinear : Measurable (fun sample : History × Action =>
228       FTRL.linearLoss arms (prob sample.1)
229         (Exp3.importanceWeightedLoss
230           (prob sample.1) (loss sample.1) sample.2)) :=
231     measurable_linearLoss_of_coordinatewise arms
232       (fun sample : History × Action => prob sample.1)
233       (fun sample candidate =>
234         Exp3.importanceWeightedLoss
235           (prob sample.1) (loss sample.1) sample.2 candidate)
236       (fun candidate hcandidate =>
237         (hprob candidate hcandidate).comp measurable_fst)
238       hestimate
239   unfold importanceWeightedPotentialStabilityScore
240   unfold halfTsallisPotentialStability
241   exact hlinear.add hnextPotential |>.sub hcurrentPotential

258 /-- A true minimizer-to-minimizer potential step is nonnegative. -/
259 theorem halfTsallisPotentialStability_nonneg_of_minimizers
260     {Action : Type u}
261     (arms : Finset Action) (eta : Real)
262     (score probability estimate next : Action -> Real)
263     (heta : 0 < eta)
264     (hprobabilityMin : FTRL.IsRegularizedMinimizer
265       (FTRL.finiteSimplex arms) arms eta
266       (negEntropyRegularizer arms (1 / 2 : Real)) score probability)
267     (hnextMin : FTRL.IsRegularizedMinimizer
268       (FTRL.finiteSimplex arms) arms eta
269       (negEntropyRegularizer arms (1 / 2 : Real))
270       (fun action => score action + estimate action) next) :
271     0 <= halfTsallisPotentialStability
272       arms eta score probability estimate next := by
273   have hmin := hnextMin.2 probability hprobabilityMin.1
274   have hobjective :
275       FTRL.regularizedObjective arms eta
276           (negEntropyRegularizer arms (1 / 2 : Real))
277           (fun action => score action + estimate action) probability =
278         FTRL.regularizedObjective arms eta
279           (negEntropyRegularizer arms (1 / 2 : Real))
280           score probability +
281         eta * FTRL.linearLoss arms probability estimate := by
282     unfold FTRL.regularizedObjective
283     rw [FTRL.linearLoss_add_right]
284     ring
285   have hminDiv := div_le_div_of_nonneg_right hmin heta.le
286   unfold halfTsallisPotentialStability halfTsallisPotentialValue
287   rw [hobjective] at hminDiv
288   have heta_ne : eta ≠ 0 := ne_of_gt heta
289   rw [add_div, mul_div_cancel_left₀ _ heta_ne] at hminDiv
290   calc
291     0 <=
292         FTRL.regularizedObjective arms eta
293             (negEntropyRegularizer arms (1 / 2 : Real)) score probability / eta +
294           FTRL.linearLoss arms probability estimate -
295           FTRL.regularizedObjective arms eta
296             (negEntropyRegularizer arms (1 / 2 : Real))
297             (fun action => score action + estimate action) next / eta :=
298       sub_nonneg.mpr hminDiv
299     _ = FTRL.linearLoss arms probability estimate +
300           (-FTRL.regularizedObjective arms eta
301               (negEntropyRegularizer arms (1 / 2 : Real))
302               (fun action => score action + estimate action) next / eta + 1 / eta) -
303           (-FTRL.regularizedObjective arms eta
304               (negEntropyRegularizer arms (1 / 2 : Real)) score probability / eta +
305             1 / eta) := by ring

307 /-- The refined budget is uniformly integrable under a finite history measure. -/
308 theorem integrable_refinedPotentialStabilityBound_of_finiteSimplex
309     {History : Type u} {Action : Type v} [MeasurableSpace History]
310     (historyMu : Measure History) [IsFiniteMeasure historyMu]
311     (arms : Finset Action) (eta : Real)
312     (prob : History -> Action -> Real)
313     (source : Exp3.MeasurableFiniteActionDistribution arms prob)
314     (heta : 0 < eta) :
315     Integrable (refinedPotentialStabilityBound arms eta prob) historyMu := by
316   have hmeasurable := measurable_refinedPotentialStabilityBound
317     arms eta prob source.measurable_prob
318   refine Integrable.of_bound hmeasurable.aestronglyMeasurable
319     (eta * (arms.card : Real) + 2 * eta ^ 2) ?_
320   exact Filter.Eventually.of_forall fun history => by
321     have hsimplex : FTRL.finiteSimplex arms (prob history) := {
322       left := (source.distribution history).nonneg
323       right := (source.distribution history).sum_eq_one
324     }
325     have hsumNonneg : 0 <= arms.sum (fun action =>
326         Real.sqrt (prob history action) * (1 - prob history action)) := by
327       exact Finset.sum_nonneg fun action haction =>
328         mul_nonneg (Real.sqrt_nonneg _)
329           (sub_nonneg.mpr (finiteSimplex_apply_le_one
330             hsimplex haction))
331     have hsumLe : arms.sum (fun action =>
332         Real.sqrt (prob history action) * (1 - prob history action)) <=
333         (arms.card : Real) := by
334       calc
335         arms.sum (fun action =>
336             Real.sqrt (prob history action) * (1 - prob history action)) <=
337           arms.sum (fun _action => (1 : Real)) := by
338             apply Finset.sum_le_sum
339             intro action haction
340             have hpNonneg := hsimplex.1 action haction
341             have hpLe := finiteSimplex_apply_le_one
342               hsimplex haction
343             have hsqrtLe : Real.sqrt (prob history action) <= 1 :=
344               (Real.sqrt_le_one).2 hpLe
345             have honeSub : 0 <= 1 - prob history action := by linarith
346             calc
347               Real.sqrt (prob history action) *
348                     (1 - prob history action) <=
349                   1 * (1 - prob history action) :=
350                 mul_le_mul_of_nonneg_right hsqrtLe honeSub
351               _ <= 1 := by linarith
352         _ = (arms.card : Real) := by simp
353     rw [Real.norm_eq_abs, abs_of_nonneg]
354     · unfold refinedPotentialStabilityBound
355       nlinarith
356     · unfold refinedPotentialStabilityBound
357       nlinarith [sq_nonneg eta]

492 /-- Product-law integrability pulls back to a realized history/action pair
493 when the kernel is an identified conditional action law. -/
494 theorem integrable_score_comp_history_action_of_condDistrib_generic
495     {Omega : Type u} {History : Type v} {Action : Type w}
496     [MeasurableSpace Omega] [MeasurableSpace History]
497     [MeasurableSpace Action] [StandardBorelSpace Action] [Nonempty Action]
498     (mu : Measure Omega) [IsFiniteMeasure mu]
499     (history : Omega -> History) (hhistory : Measurable history)
500     (action : Omega -> Action) (haction : Measurable action)
501     (score : History × Action -> Real)
502     (policy : Kernel History Action) [IsMarkovKernel policy]
503     (hcond : condDistrib action history mu =ᵐ[mu.map history] policy)
504     (hIntegrable : Integrable score (mu.map history ⊗ₘ policy)) :
505     Integrable (fun omega => score (history omega, action omega)) mu := by
506   have hpair :
507       mu.map (fun omega => (history omega, action omega)) =
508         mu.map history ⊗ₘ policy :=
