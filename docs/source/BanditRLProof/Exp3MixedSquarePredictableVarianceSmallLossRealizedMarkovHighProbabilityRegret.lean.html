19 /-- Every generated predictable loss coordinate remains in the unit interval. -/
20 theorem predictableLossAt_mem_unitInterval
21     {Env : Type u} {Action : Type v}
22     [MeasurableSpace Env] [MeasurableSpace Action]
23     (loss : PredictableLossVector Env Action) (t : Nat)
24     (sample : Env × ((k : Nat) → Action × Real)) (action : Action) :
25     predictableLossAt loss t sample action ∈ Set.Icc (0 : Real) 1 := by
26   cases t with
27   | zero =>
28       simpa [predictableLossAt] using
29         loss.initial_mem_unitInterval sample.1 action
30   | succ n =>
31       simpa [predictableLossAt] using
32         loss.successor_mem_unitInterval n sample.1
33           (Preorder.frestrictLe n sample.2) action

44 /-- Unit-interval losses have armwise square mass at most armwise loss mass. -/
45 theorem sampledPredictableLossSquaredAt_le_lossMassAt
46     {Env : Type u} {Action : Type v}
47     [MeasurableSpace Env] [MeasurableSpace Action]
48     (arms : Finset Action) (loss : PredictableLossVector Env Action)
49     (t : Nat) (sample : Env × ((k : Nat) → Action × Real)) :
50     arms.sum (fun action => (predictableLossAt loss t sample action) ^ 2) ≤
51       arms.sum (fun action => predictableLossAt loss t sample action) := by
52   apply Finset.sum_le_sum
53   intro action _haction
54   have hloss := predictableLossAt_mem_unitInterval loss t sample action
55   have hproduct :
56       0 ≤ predictableLossAt loss t sample action *
57         (1 - predictableLossAt loss t sample action) :=
58     mul_nonneg hloss.1 (sub_nonneg.mpr hloss.2)
59   nlinarith
