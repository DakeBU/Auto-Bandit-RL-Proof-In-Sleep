3 /-!
4 # Learning-rate tuning with pathwise variance under probabilistic sparsity
5 
6 This module tunes `eta` for the four-event probabilistic-sparsity route. Its
7 variance budget is the deterministic good-path bound
8 
9 `(1 / (gamma / arms.card)) * (sparsity * horizon)`,
10 
11 and its confidence allocation is `delta / 4`. Thus the tuned theorem retains
12 the exact sparsity-failure residual without the global `arms.card * horizon`
13 Markov envelope or a fifth overflow event.
14 
15 Gamma remains caller-selected.
16 -/

24 /-- Complete Hedge scale using sparse loss mass and sparse pathwise variance. -/
25 noncomputable def pathwiseVarianceProbabilisticSparseLossHighProbabilityScale
26     {Action : Type v} [DecidableEq Action]
27     (arms : Finset Action) (gamma : Real)
28     (horizon sparsity : Nat) (delta : Real) : Real :=
29   (sparsity : Real) * (horizon : Real) +
30     sampledMixedSquaredPredictableVarianceRadius arms gamma
31       (sampledPredictableSparsePathwiseVarianceBudget
32         arms gamma horizon sparsity)
33       (delta / 4)

35 theorem pathwiseVarianceProbabilisticSparseLossHighProbabilityScale_pos
36     {Action : Type v} [DecidableEq Action]
37     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
38     (gamma : Real) (hgamma_pos : 0 < gamma)
39     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
40     (hsparsity : 0 < sparsity) (delta : Real) :
41     0 < pathwiseVarianceProbabilisticSparseLossHighProbabilityScale
42       arms gamma horizon sparsity delta := by
43   have hcard_pos : 0 < (arms.card : Real) := by positivity
44   have hfloor_pos : 0 < gamma / (arms.card : Real) :=
45     div_pos hgamma_pos hcard_pos
46   have hloss_pos :
47       0 < (sparsity : Real) * (horizon : Real) :=
48     mul_pos (Nat.cast_pos.2 hsparsity) (Nat.cast_pos.2 hhorizon)
49   have hvariance_nonneg :
50       0 ≤ sampledPredictableSparsePathwiseVarianceBudget
51         arms gamma horizon sparsity := by
52     dsimp [sampledPredictableSparsePathwiseVarianceBudget]
53     positivity
54   have hlog_nonneg : 0 ≤ max (Real.log (1 / (delta / 4))) 0 :=
55     le_max_right _ _
56   have hradius_nonneg :
57       0 ≤ sampledMixedSquaredPredictableVarianceRadius arms gamma
58         (sampledPredictableSparsePathwiseVarianceBudget
59           arms gamma horizon sparsity)
60         (delta / 4) := by
61     dsimp [sampledMixedSquaredPredictableVarianceRadius]
62     exact add_nonneg
63       (mul_nonneg (by norm_num)
64         (Real.sqrt_nonneg
65           (sampledPredictableSparsePathwiseVarianceBudget
66             arms gamma horizon sparsity *
67               max (Real.log (1 / (delta / 4))) 0)))
68       (div_nonneg hlog_nonneg hfloor_pos.le)
69   dsimp [pathwiseVarianceProbabilisticSparseLossHighProbabilityScale]
70   linarith

119 /-- Under `gamma ≤ 1/2`, entropy and stability cost at most three balanced
120 copies of the four-event sparse scale. -/
121 theorem pathwiseVarianceProbabilisticSparseLossHighProbabilityHedgeBudget_le_three_mul_sqrt
122     {Action : Type v} [DecidableEq Action]
123     (arms : Finset Action) (hcard_two : 2 ≤ arms.card)
124     (gamma : Real) (hgamma_pos : 0 < gamma)
125     (hgamma_le_half : gamma ≤ 1 / 2)
126     (horizon sparsity : Nat) (hhorizon : 0 < horizon)
127     (hsparsity : 0 < sparsity) (delta : Real) :
128     Real.log (arms.card : Real) /
129           pathwiseVarianceProbabilisticSparseLossHighProbabilityLearningRate
130             arms gamma horizon sparsity delta +
131         (pathwiseVarianceProbabilisticSparseLossHighProbabilityLearningRate
132             arms gamma horizon sparsity delta * (1 / (1 - gamma))) *
133           pathwiseVarianceProbabilisticSparseLossHighProbabilityScale
134             arms gamma horizon sparsity delta ≤
135       3 * Real.sqrt
136         (Real.log (arms.card : Real) *
137           pathwiseVarianceProbabilisticSparseLossHighProbabilityScale
138             arms gamma horizon sparsity delta) := by
139   let eta :=
140     pathwiseVarianceProbabilisticSparseLossHighProbabilityLearningRate
141       arms gamma horizon sparsity delta
142   let scale :=
143     pathwiseVarianceProbabilisticSparseLossHighProbabilityScale
144       arms gamma horizon sparsity delta
145   let entropy := Real.log (arms.card : Real) / eta
146   have hK_one : 1 < (arms.card : Real) := by
147     exact_mod_cast hcard_two
148   have heta : 0 < eta := by
149     dsimp [eta]
150     exact
151       pathwiseVarianceProbabilisticSparseLossHighProbabilityLearningRate_pos
152         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
153   have hscale : 0 < scale := by
154     dsimp [scale]
155     exact
156       pathwiseVarianceProbabilisticSparseLossHighProbabilityScale_pos
157         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
158   have hbalance : eta ^ 2 * scale = Real.log (arms.card : Real) := by
159     dsimp [eta, scale]
160     exact
161       pathwiseVarianceProbabilisticSparseLossHighProbabilityLearningRate_sq_mul_scale
162         arms hcard_two gamma hgamma_pos horizon sparsity hhorizon hsparsity delta
163   have hentropy_nonneg : 0 ≤ entropy :=
164     div_nonneg (Real.log_pos hK_one).le heta.le
165   have hentropy_eq_unscaled : entropy = eta * scale := by
166     dsimp [entropy]
167     field_simp [ne_of_gt heta]
168     nlinarith [hbalance]
169   have hden : 0 < 1 - gamma := by linarith
170   have hfactor : 1 ≤ 2 * (1 - gamma) := by linarith
171   have hstable :
172       (eta * (1 / (1 - gamma))) * scale ≤ 2 * entropy := by
173     rw [show (eta * (1 / (1 - gamma))) * scale =
174       (eta * scale) / (1 - gamma) by ring]
175     rw [← hentropy_eq_unscaled, div_le_iff₀ hden]
176     have hmul := mul_le_mul_of_nonneg_left hfactor hentropy_nonneg
177     nlinarith
178   have hbalanced_sq :
179       entropy ^ 2 = Real.log (arms.card : Real) * scale := by
180     rw [hentropy_eq_unscaled]
181     calc
182       (eta * scale) ^ 2 = (eta ^ 2 * scale) * scale := by ring
183       _ = Real.log (arms.card : Real) * scale := by rw [hbalance]
184   have hradicand_nonneg :
185       0 ≤ Real.log (arms.card : Real) * scale :=
186     mul_nonneg (Real.log_pos hK_one).le hscale.le
187   have hentropy_eq_sqrt :
188       entropy = Real.sqrt
189         (Real.log (arms.card : Real) * scale) := by
190     have hsqrt_nonneg :=
191       Real.sqrt_nonneg (Real.log (arms.card : Real) * scale)
192     have hsqrt_sq := Real.sq_sqrt hradicand_nonneg
193     nlinarith [hbalanced_sq]
194   change entropy + (eta * (1 / (1 - gamma))) * scale ≤
195     3 * Real.sqrt (Real.log (arms.card : Real) * scale)
196   rw [← hentropy_eq_sqrt]
197   linarith
