5 # All-time scheduled OFUL confidence
6 
7 This module upgrades the fixed-time scalar-ridge confidence theorem to one
8 countable union over every deterministic horizon on the same probability
9 space.  A telescoping confidence schedule allocates the total failure budget
10 exactly:
11 
12 `delta_n = delta / ((n + 1) * (n + 2))`.
13 
14 This is an all-time confidence theorem for one fixed process.  It does not
15 identify a generated OFUL policy; a one-policy anytime regret consumer must
16 first use a history algorithm whose confidence parameter follows this schedule.
17 -/
18

148 /-- Positive telescoping weight `1 / ((n+1)(n+2))`. -/
149 noncomputable def allTimeTelescopingWeight (n : Nat) : Real :=
150   1 / (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real))
151 
152 /-- The telescoping weight is a difference of consecutive reciprocals. -/
153 theorem allTimeTelescopingWeight_eq_sub (n : Nat) :
154     allTimeTelescopingWeight n =
155       1 / (((n + 1 : Nat) : Real)) -
156         1 / (((n + 2 : Nat) : Real)) := by
157   unfold allTimeTelescopingWeight
158   field_simp
159   norm_num [Nat.cast_add]
160 
161 /-- Exact finite partial sum of the telescoping weights. -/
162 theorem sum_range_allTimeTelescopingWeight (n : Nat) :
163     (Finset.range n).sum allTimeTelescopingWeight =
164       1 - 1 / (((n + 1 : Nat) : Real)) := by
165   rw [Finset.sum_congr rfl
166     (fun i _hi => allTimeTelescopingWeight_eq_sub i)]
167   rw [Finset.sum_range_sub']
168   norm_num
169 
170 /-- Every telescoping confidence weight is nonnegative. -/
171 theorem allTimeTelescopingWeight_nonneg (n : Nat) :
172     0 <= allTimeTelescopingWeight n := by
173   unfold allTimeTelescopingWeight
174   positivity
175 
176 /-- Every telescoping confidence weight is at most one. -/
177 theorem allTimeTelescopingWeight_le_one (n : Nat) :
178     allTimeTelescopingWeight n <= 1 := by
179   unfold allTimeTelescopingWeight
180   have hfirst :
181       (1 : Real) <= (((n + 1 : Nat) : Real)) := by
182     exact_mod_cast Nat.succ_le_succ (Nat.zero_le n)
183   have hsecond :
184       (1 : Real) <= (((n + 2 : Nat) : Real)) := by
185     exact_mod_cast Nat.succ_le_succ (Nat.zero_le (n + 1))
186   have hden :
187       (1 : Real) <=
188         (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) := by
189     nlinarith
190   have hden_pos :
191       0 <
192         (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) :=
193     lt_of_lt_of_le zero_lt_one hden
194   exact (div_le_one hden_pos).2 hden
195 
196 /-- The telescoping weights sum exactly to one. -/
197 theorem hasSum_allTimeTelescopingWeight :
198     HasSum allTimeTelescopingWeight 1 := by
199   rw [hasSum_iff_tendsto_nat_of_nonneg allTimeTelescopingWeight_nonneg]
200   simp_rw [sum_range_allTimeTelescopingWeight]
201   have hcast :
202       Filter.Tendsto
203         (fun n : Nat => (((n + 1 : Nat) : Real)))
204         Filter.atTop Filter.atTop := by
205     exact tendsto_natCast_atTop_atTop.comp
206       (Filter.tendsto_add_atTop_nat 1)
207   have hinv :
208       Filter.Tendsto
209         (fun n : Nat => (1 / (((n + 1 : Nat) : Real))))
210         Filter.atTop (nhds 0) := by
211     simpa only [one_div] using tendsto_inv_atTop_zero.comp hcast
212   simpa using tendsto_const_nhds.sub hinv

214 /-- Time-`n` share of an all-time failure budget. -/
215 noncomputable def allTimeTelescopingDelta
216     (delta : Real) (n : Nat) : Real :=
217   delta * allTimeTelescopingWeight n
218 
219 /-- Display the scheduled budget as `delta / ((n+1)(n+2))`. -/
220 theorem allTimeTelescopingDelta_eq_div (delta : Real) (n : Nat) :
221     allTimeTelescopingDelta delta n =
222       delta /
223         (((n + 1 : Nat) : Real) * ((n + 2 : Nat) : Real)) := by
224   simp [allTimeTelescopingDelta, allTimeTelescopingWeight, div_eq_mul_inv]
225 
226 /-- Positive outer budget gives positive timewise budgets. -/
227 theorem allTimeTelescopingDelta_pos
228     {delta : Real} (hdelta : 0 < delta) (n : Nat) :
229     0 < allTimeTelescopingDelta delta n :=
230   mul_pos hdelta (by
231     unfold allTimeTelescopingWeight
232     positivity)
233 
234 /-- If `delta <= 1`, every timewise budget is also at most one. -/
235 theorem allTimeTelescopingDelta_le_one
236     {delta : Real} (hdelta : 0 < delta) (hdelta_one : delta <= 1)
237     (n : Nat) :
238     allTimeTelescopingDelta delta n <= 1 := by
239   calc
240     allTimeTelescopingDelta delta n <= delta := by
241       unfold allTimeTelescopingDelta
242       simpa only [mul_one] using
243         mul_le_mul_of_nonneg_left
244           (allTimeTelescopingWeight_le_one n) hdelta.le
245     _ <= 1 := hdelta_one

247 /-- The `ENNReal` scheduled failure budgets sum exactly to `ofReal delta`. -/
248 theorem tsum_ofReal_allTimeTelescopingDelta
249     {delta : Real} (hdelta : 0 <= delta) :
250     ∑' n, ENNReal.ofReal (allTimeTelescopingDelta delta n) =
251       ENNReal.ofReal delta := by
252   have hreal :
253       HasSum (allTimeTelescopingDelta delta) delta := by
254     simpa [allTimeTelescopingDelta] using
255       hasSum_allTimeTelescopingWeight.mul_left delta
256   have hnonneg : forall n, 0 <= allTimeTelescopingDelta delta n := by
257     intro n
258     exact mul_nonneg hdelta (allTimeTelescopingWeight_nonneg n)
259   have hnnreal :
260       HasSum
261           (fun n => (allTimeTelescopingDelta delta n).toNNReal)
262           delta.toNNReal :=
263     hreal.toNNReal hnonneg
264   have hcoe :
265       HasSum
266           (fun n =>
267             ((allTimeTelescopingDelta delta n).toNNReal : ENNReal))
268           (delta.toNNReal : ENNReal) :=
269     ENNReal.hasSum_coe.mpr hnnreal
270   have hofReal :
271       HasSum
272           (fun n => ENNReal.ofReal (allTimeTelescopingDelta delta n))
273           (ENNReal.ofReal delta) := by
274     simpa only [ENNReal.ofReal_eq_coe_nnreal] using hcoe
275   exact hofReal.tsum_eq
