5 # Square-root schedule self-bounding tuning
6 
7 This module removes the caller-supplied cutoff from the refined generated
8 self-bounding bound.  It floors the continuous active-branch threshold and
9 records the large-horizon conditions needed to keep that cutoff positive and
10 inside the finite horizon.  Joint optimization of `lambda` and corruption is
11 left to a downstream scalar leaf.
12 -/

21 /-- A positive real threshold and its natural floor differ by at most a
22 factor of two once the threshold is at least one. -/
23 theorem natFloor_positive_and_half_le
24     (q : Real) (hq : 1 <= q) :
25     0 < ⌊q⌋₊ ∧
26       ((⌊q⌋₊ : Nat) : Real) <= q ∧
27       q <= 2 * ((⌊q⌋₊ : Nat) : Real) := by
28   have hqNonneg : 0 <= q := zero_le_one.trans hq
29   have hfloorPos : 0 < ⌊q⌋₊ := Nat.floor_pos.mpr hq
30   have hfloorLe : ((⌊q⌋₊ : Nat) : Real) <= q := Nat.floor_le hqNonneg
31   have hqLt : q < ((⌊q⌋₊ : Nat) : Real) + 1 := by
32     simpa using (Nat.lt_floor_add_one q)
33   have hfloorOne : (1 : Real) <= ((⌊q⌋₊ : Nat) : Real) := by
34     exact_mod_cast hfloorPos
35   exact ⟨hfloorPos, hfloorLe, by linarith⟩

37 /-- The continuous threshold whose floor is used for the active-prefix split. -/
38 noncomputable def sampledScheduledHalfTsallisSelfBoundingThreshold
39     {Action : Type u} [DecidableEq Action]
40     (arms : Finset Action) (best : Action) (gap : Action -> Real)
41     (lambda : Real) : Real :=
42   let reciprocalGap := (arms.erase best).sum (fun action =>
43     1 / (lambda * gap action))
44   let amplitude := 5 * (1 + lambda)
45   (amplitude * reciprocalGap /
46       (2 * Real.sqrt ((arms.erase best).card : Real))) ^ 2
