4 # Quadratic fixed-MGF optimization
5 
6 This module turns a family of fixed-tilt quadratic exponential tails into a
7 delta-shaped bound.  The probabilistic construction of each fixed-tilt tail
8 remains separate, so model-specific consumers only need to expose the common
9 quadratic exponent and admissible tilt cap.
10 -/

18 /-- Optimize a quadratic fixed-tilt MGF budget when the quadratic coefficient
19 and admissible tilt cap are separate parameters. -/
20 theorem exists_tilt_quadratic_fixedMGF_exponent_le_neg
21     (horizon variance cap budget : Real)
22     (hhorizon : 0 <= horizon) (hvariance : 0 < variance)
23     (hcap : 0 < cap) (hbudget : 0 <= budget) :
24     exists tilt : Real, 0 <= tilt ∧ tilt <= cap ∧
25       -tilt *
26           (2 * Real.sqrt (horizon * variance * budget) + budget / cap) +
27         horizon * (tilt ^ 2 * variance) <= -budget := by
28   by_cases hbudget_zero : budget = 0
29   · refine ⟨0, le_rfl, hcap.le, ?_⟩
30     simp [hbudget_zero]
31   have hbudget_pos : 0 < budget :=
32     lt_of_le_of_ne hbudget (Ne.symm hbudget_zero)
33   by_cases hsmall : budget <= cap ^ 2 * horizon * variance
34   · have hhorizon_pos : 0 < horizon := by
35       by_contra h
36       have hhorizon_zero : horizon = 0 := le_antisymm (le_of_not_gt h) hhorizon
37       simp [hhorizon_zero] at hsmall
38       exact (not_lt_of_ge hsmall) hbudget_pos
39     let tilt := Real.sqrt (budget / (horizon * variance))
40     let scale := Real.sqrt (horizon * variance * budget)
41     have htilt_nonneg : 0 <= tilt := Real.sqrt_nonneg _
42     have hscale_nonneg : 0 <= scale := Real.sqrt_nonneg _
43     have hden_pos : 0 < horizon * variance := mul_pos hhorizon_pos hvariance
44     have htilt_sq : tilt ^ 2 = budget / (horizon * variance) := by
45       dsimp only [tilt]
46       rw [Real.sq_sqrt]
47       positivity
48     have hscale_sq : scale ^ 2 = horizon * variance * budget := by
49       dsimp only [scale]
50       rw [Real.sq_sqrt]
51       positivity
52     have hprod_sq : (tilt * scale) ^ 2 = budget ^ 2 := by
53       rw [mul_pow, htilt_sq, hscale_sq]
54       field_simp
55     have hprod : tilt * scale = budget := by
56       nlinarith [mul_nonneg htilt_nonneg hscale_nonneg]
57     have htilt_sq_le : tilt ^ 2 <= cap ^ 2 := by
58       rw [htilt_sq, div_le_iff₀ hden_pos]
59       simpa [mul_assoc, mul_comm, mul_left_comm] using hsmall
60     have htilt_le : tilt <= cap :=
61       (sq_le_sq₀ htilt_nonneg hcap.le).mp htilt_sq_le
62     refine ⟨tilt, htilt_nonneg, htilt_le, ?_⟩
63     change -tilt * (2 * scale + budget / cap) +
64       horizon * (tilt ^ 2 * variance) <= -budget
65     have hbudget_term : horizon * (tilt ^ 2 * variance) = budget := by
66       rw [htilt_sq]
67       field_simp
68     have hextra : 0 <= tilt * (budget / cap) :=
69       mul_nonneg htilt_nonneg (div_nonneg hbudget hcap.le)
70     rw [hbudget_term]
71     nlinarith
72   · refine ⟨cap, hcap.le, le_rfl, ?_⟩
73     let scale := Real.sqrt (horizon * variance * budget)
74     have hscale_nonneg : 0 <= scale := Real.sqrt_nonneg _
75     have hscale_sq : scale ^ 2 = horizon * variance * budget := by
76       dsimp only [scale]
77       rw [Real.sq_sqrt]
78       positivity
79     have hlarge : cap ^ 2 * horizon * variance < budget := lt_of_not_ge hsmall
80     have hhv_nonneg : 0 <= horizon * variance :=
81       mul_nonneg hhorizon hvariance.le
82     have hbase_sq : (horizon * variance * cap) ^ 2 <= scale ^ 2 := by
83       rw [hscale_sq]
84       have hmul := mul_le_mul_of_nonneg_right hlarge.le hhv_nonneg
85       nlinarith
86     have hbase_nonneg : 0 <= horizon * variance * cap :=
87       mul_nonneg hhv_nonneg hcap.le
88     have hbase_le : horizon * variance * cap <= scale :=
89       (sq_le_sq₀ hbase_nonneg hscale_nonneg).mp hbase_sq
90     have hcover : horizon * variance * cap ^ 2 <= 2 * cap * scale := by
91       calc
92         horizon * variance * cap ^ 2 = cap * (horizon * variance * cap) := by ring
93         _ <= cap * scale := mul_le_mul_of_nonneg_left hbase_le hcap.le
94         _ <= 2 * cap * scale := by
95           nlinarith [mul_nonneg hcap.le hscale_nonneg]
96     calc
97       -cap * (2 * Real.sqrt (horizon * variance * budget) + budget / cap) +
98           horizon * (cap ^ 2 * variance) =
99         -2 * cap * scale - budget + horizon * variance * cap ^ 2 := by
100           field_simp
101           ring
102       _ <= -budget := by nlinarith

104 /-- Radius obtained by optimizing a quadratic fixed-tilt exponent over
105 `0 <= tilt <= tiltCap`. -/
106 noncomputable def quadraticFixedMGFRadius
107     (varianceScale varianceBudget tiltCap delta : Real) : Real :=
108   let budget := max (Real.log (1 / delta)) 0
109   2 * Real.sqrt (varianceScale * varianceBudget * budget) + budget / tiltCap
110 
111 /-- A family of fixed-tilt quadratic tails yields a delta-shaped joint
112 deviation and variance-budget tail. -/
113 theorem measure_deviation_ge_inter_variance_le_delta_of_fixedTilt_quadratic_tail
114     {Omega : Type*} [MeasurableSpace Omega]
115     (mu : Measure Omega) (deviation predictableVariance : Omega -> Real)
116     (varianceScale varianceBudget tiltCap delta : Real)
117     (hvarianceScale : 0 < varianceScale)
118     (hvarianceBudget : 0 < varianceBudget) (htiltCap : 0 < tiltCap)
119     (hdelta : 0 < delta)
120     (hfixed : forall tilt, 0 <= tilt -> tilt <= tiltCap ->
121       mu {omega |
122           quadraticFixedMGFRadius varianceScale varianceBudget tiltCap delta <=
123               deviation omega ∧
124             predictableVariance omega <= varianceBudget} <=
125         ENNReal.ofReal (Real.exp
126           (-tilt * quadraticFixedMGFRadius
127               varianceScale varianceBudget tiltCap delta +
128             varianceScale * (tilt ^ 2 * varianceBudget)))) :
129     mu {omega |
130         quadraticFixedMGFRadius varianceScale varianceBudget tiltCap delta <=
131             deviation omega ∧
132           predictableVariance omega <= varianceBudget} <=
133       ENNReal.ofReal delta := by
134   let budget := max (Real.log (1 / delta)) 0
135   let radius :=
136     2 * Real.sqrt (varianceScale * varianceBudget * budget) + budget / tiltCap
137   have hbudget : 0 <= budget := le_max_right _ _
138   obtain ⟨tilt, htilt_nonneg, htilt_le, hexponent⟩ :=
139     exists_tilt_quadratic_fixedMGF_exponent_le_neg
140       varianceScale varianceBudget tiltCap budget hvarianceScale.le
141         hvarianceBudget htiltCap hbudget
142   have htail := hfixed tilt htilt_nonneg htilt_le
143   have hexponent' :
144       -tilt * radius + varianceScale * (tilt ^ 2 * varianceBudget) <=
145         -budget := by
146     simpa [radius, mul_assoc] using hexponent
147   have hbudget_tail :
148       mu {omega | radius <= deviation omega ∧
149           predictableVariance omega <= varianceBudget} <=
150         ENNReal.ofReal (Real.exp (-budget)) := by
151     have htail' :
152         mu {omega | radius <= deviation omega ∧
153             predictableVariance omega <= varianceBudget} <=
154           ENNReal.ofReal (Real.exp
155             (-tilt * radius + varianceScale * (tilt ^ 2 * varianceBudget))) := by
156       simpa [quadraticFixedMGFRadius, radius, budget] using htail
157     exact htail'.trans
158       (ENNReal.ofReal_le_ofReal (Real.exp_le_exp.mpr hexponent'))
159   have hscale : 0 < 1 / delta := one_div_pos.mpr hdelta
160   have hexp_le_delta : Real.exp (-budget) <= delta := by
161     by_cases hdelta_le_one : delta <= 1
162     · have hone_le_inv : 1 <= 1 / delta := by
163         simpa using one_div_le_one_div_of_le hdelta hdelta_le_one
164       have hlog_nonneg : 0 <= Real.log (1 / delta) :=
165         Real.log_nonneg hone_le_inv
166       have hbudget_eq : budget = Real.log (1 / delta) :=
167         max_eq_left hlog_nonneg
168       rw [hbudget_eq, Real.exp_neg, Real.exp_log hscale]
169       field_simp
170       exact le_rfl
171     · have hone_lt_delta : 1 < delta := lt_of_not_ge hdelta_le_one
172       have hinv_le_one : 1 / delta <= 1 := by
173         simpa using one_div_le_one_div_of_le (by norm_num : (0 : Real) < 1)
174           hone_lt_delta.le
175       have hlog_nonpos : Real.log (1 / delta) <= 0 :=
176         Real.log_nonpos hscale.le hinv_le_one
177       have hbudget_eq : budget = 0 := max_eq_right hlog_nonpos
178       rw [hbudget_eq, neg_zero, Real.exp_zero]
179       exact hone_lt_delta.le
180   have hfinal := hbudget_tail.trans (ENNReal.ofReal_le_ofReal hexp_le_delta)
181   simpa [quadraticFixedMGFRadius, radius, budget] using hfinal
