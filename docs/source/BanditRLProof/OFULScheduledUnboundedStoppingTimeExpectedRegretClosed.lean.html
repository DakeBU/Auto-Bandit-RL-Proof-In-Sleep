4 # Closed square-integrable unbounded stopping-time OFUL expected-regret rate

19 /-- The scalar log-determinant budget is at most linear in the round count. -/
20 theorem standardScalarLogDetBudget_le_rounds_mul_div
21     {Feature : Type u} [Fintype Feature] [Nonempty Feature]
22     (lambda : Real) (hlambda : 0 < lambda)
23     (rounds : Nat) (L2 : Real) (hL2 : 0 <= L2) :
24     standardScalarLogDetBudget (Feature := Feature) lambda rounds L2 <=
25       (rounds : Real) * (L2 / lambda) := by
26   have hcard_pos : 0 < (Fintype.card Feature : Real) := by
27     exact_mod_cast (Fintype.card_pos_iff.mpr inferInstance :
28       0 < Fintype.card Feature)
29   have hdenom_pos :
30       0 < (Fintype.card Feature : Real) * lambda :=
31     mul_pos hcard_pos hlambda
32   let x :=
33     ((rounds : Real) * L2) /
34       ((Fintype.card Feature : Real) * lambda)
35   have hx_nonneg : 0 <= x := by
36     dsimp only [x]
37     positivity
38   have hlog : Real.log (1 + x) <= x := by
39     have := Real.log_le_sub_one_of_pos (show 0 < 1 + x by linarith)
40     linarith
41   unfold standardScalarLogDetBudget
42   change
43     (Fintype.card Feature : Real) * Real.log (1 + x) <=
44       (rounds : Real) * (L2 / lambda)
45   calc
46     (Fintype.card Feature : Real) * Real.log (1 + x) <=
47         (Fintype.card Feature : Real) * x := by
48       exact mul_le_mul_of_nonneg_left hlog hcard_pos.le
49     _ = (rounds : Real) * (L2 / lambda) := by
50       dsimp only [x]
51       field_simp

53 /--
54 Parameter-only coefficient in the quadratic envelope for the explicit
55 telescoping pseudo-regret budget.
56 -/
57 noncomputable def telescopingHighProbabilityPseudoRegretQuadraticCoefficient
58     {Feature : Type u} [Fintype Feature]
59     (R delta lambda S L2 : Real) : Real :=
60   standardScalarInitialGapBound S L2 +
61     2 *
62       (R * Real.sqrt (L2 / lambda + 4 / delta) +
63         Real.sqrt lambda * S) *
64       Real.sqrt (2 * (L2 / lambda))
65

72     (lambda : Real) (hlambda : 0 < lambda)

75     0 <=
76       telescopingHighProbabilityPseudoRegretQuadraticCoefficient
77         (Feature := Feature) R delta lambda S L2 := by
78   unfold telescopingHighProbabilityPseudoRegretQuadraticCoefficient
79   unfold standardScalarInitialGapBound
80   positivity

82 /--
83 The telescoping confidence logarithm is bounded by a parameter-only
84 coefficient times the square of the round count.
85 -/
86 theorem
87     telescopingHighProbabilityRegretLogBudget_le_rounds_sq_mul
88     {Feature : Type u} [Fintype Feature] [Nonempty Feature]
89     (lambda : Real) (hlambda : 0 < lambda)
90     (delta : Real) (hdelta : 0 < delta)
91     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2) :
92     telescopingHighProbabilityRegretLogBudget
93         (Feature := Feature) lambda delta horizon L2 <=
94       (((horizon + 1 : Nat) : Real) ^ 2) *
95         (L2 / lambda + 4 / delta) := by
96   let r : Real := ((horizon + 1 : Nat) : Real)
97   have hr_one : 1 <= r := by
98     dsimp only [r]
99     exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
100   have hr_nonneg : 0 <= r := le_trans (by norm_num) hr_one
101   have hD_nonneg : 0 <= L2 / lambda := div_nonneg hL2 hlambda.le
102   have hdeltaInv_nonneg : 0 <= 4 / delta := by positivity
103   have hdet :=
104     standardScalarLogDetBudget_le_rounds_mul_div
105       (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
106   have hdet_sq :
107       standardScalarLogDetBudget
108           (Feature := Feature) lambda (horizon + 1) L2 <=
109         r ^ 2 * (L2 / lambda) := by
110     calc
111       standardScalarLogDetBudget
112           (Feature := Feature) lambda (horizon + 1) L2 <=
113           r * (L2 / lambda) := by simpa [r] using hdet
114       _ <= r ^ 2 * (L2 / lambda) := by
115         apply mul_le_mul_of_nonneg_right _ hD_nonneg
116         nlinarith
117   let ratio : Real :=
118     ((((horizon + 1 : Nat) : Real) *
119         ((horizon + 2 : Nat) : Real)) / delta)
120   have hratio_pos : 0 < ratio := by
121     dsimp only [ratio]
122     positivity
123   have hlog_ratio : Real.log ratio <= ratio := by
124     exact (Real.log_le_sub_one_of_pos hratio_pos).trans (by linarith)
125   have hsucc_le : (((horizon + 2 : Nat) : Real)) <= 2 * r := by
126     have hsucc_nat : horizon + 2 <= 2 * (horizon + 1) := by omega
127     dsimp only [r]
128     exact_mod_cast hsucc_nat
129   have hratio_le :
130       ratio <= 2 * r ^ 2 / delta := by
131     dsimp only [ratio]
132     rw [div_le_div_iff_of_pos_right hdelta]
133     calc
134       (((horizon + 1 : Nat) : Real) *
135           ((horizon + 2 : Nat) : Real)) <=
136           r * (2 * r) := by
137         exact mul_le_mul_of_nonneg_left hsucc_le hr_nonneg
138       _ = 2 * r ^ 2 := by ring
139   have hlog_term :
140       2 * Real.log ratio <= r ^ 2 * (4 / delta) := by
141     calc
142       2 * Real.log ratio <= 2 * ratio := by linarith
143       _ <= 2 * (2 * r ^ 2 / delta) := by linarith
144       _ = r ^ 2 * (4 / delta) := by ring
145   unfold telescopingHighProbabilityRegretLogBudget
146   change
147     standardScalarLogDetBudget
148           (Feature := Feature) lambda (horizon + 1) L2 +
149         2 * Real.log ratio <=
150       r ^ 2 * (L2 / lambda + 4 / delta)
151   calc
152     standardScalarLogDetBudget
153           (Feature := Feature) lambda (horizon + 1) L2 +
154         2 * Real.log ratio <=
155         r ^ 2 * (L2 / lambda) + r ^ 2 * (4 / delta) := by
156       exact add_le_add hdet_sq hlog_term
157     _ = r ^ 2 * (L2 / lambda + 4 / delta) := by ring

159 /--
160 The explicit telescoping pseudo-regret budget grows at most quadratically in
161 the round count.
162 -/
163 theorem
164     telescopingHighProbabilityPseudoRegretBound_le_rounds_sq_mul_coefficient
165     {Feature : Type u} [Fintype Feature] [Nonempty Feature]
166     (R : Real) (hR : 0 <= R)
167     (delta : Real) (hdelta : 0 < delta) (hdelta_one : delta <= 1)
168     (lambda : Real) (hlambda : 0 < lambda)
169     (S : Real) (hS : 0 <= S)
170     (horizon : Nat) (L2 : Real) (hL2 : 0 <= L2) :
171     telescopingHighProbabilityPseudoRegretBound
172         (Feature := Feature) R delta lambda S horizon L2 <=
173       (((horizon + 1 : Nat) : Real) ^ 2) *
174         telescopingHighProbabilityPseudoRegretQuadraticCoefficient
175           (Feature := Feature) R delta lambda S L2 := by
176   let r : Real := ((horizon + 1 : Nat) : Real)
177   let D : Real := L2 / lambda
178   let C : Real := D + 4 / delta
179   let logBudget :=
180     telescopingHighProbabilityRegretLogBudget
181       (Feature := Feature) lambda delta horizon L2
182   let detBudget :=
183     standardScalarLogDetBudget
184       (Feature := Feature) lambda (horizon + 1) L2
185   let radiusBase :=
186     R * Real.sqrt C + Real.sqrt lambda * S
187   let widthBase := Real.sqrt (2 * D)
188   let initialGap := standardScalarInitialGapBound S L2
189   have hr_one : 1 <= r := by
190     dsimp only [r]
191     exact_mod_cast Nat.succ_le_succ (Nat.zero_le horizon)
192   have hr_nonneg : 0 <= r := le_trans (by norm_num) hr_one
193   have hD_nonneg : 0 <= D := by
194     dsimp only [D]
195     exact div_nonneg hL2 hlambda.le
196   have hC_nonneg : 0 <= C := by
197     dsimp only [C]
198     positivity
199   have hlogBudget_nonneg : 0 <= logBudget := by
200     dsimp only [logBudget]
201     exact
202       telescopingHighProbabilityRegretLogBudget_nonneg
203         (Feature := Feature) lambda hlambda delta hdelta hdelta_one
204         horizon L2 hL2
205   have hdetBudget_nonneg : 0 <= detBudget := by
206     dsimp only [detBudget]
207     exact
208       standardScalarLogDetBudget_nonneg
209         (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
210   have hlogBudget :
211       logBudget <= r ^ 2 * C := by
212     dsimp only [logBudget, r, C, D]
213     exact
214       telescopingHighProbabilityRegretLogBudget_le_rounds_sq_mul
215         (Feature := Feature) lambda hlambda delta hdelta horizon L2 hL2
216   have hdetBudget :
217       detBudget <= r * D := by
218     dsimp only [detBudget, r, D]
219     exact
220       standardScalarLogDetBudget_le_rounds_mul_div
221         (Feature := Feature) lambda hlambda (horizon + 1) L2 hL2
222   have hsqrtLogBudget :
223       Real.sqrt logBudget <= r * Real.sqrt C := by
224     calc
225       Real.sqrt logBudget <= Real.sqrt (r ^ 2 * C) :=
226         Real.sqrt_le_sqrt hlogBudget
227       _ = Real.sqrt (r ^ 2) * Real.sqrt C := by
228         rw [Real.sqrt_mul (sq_nonneg r)]
229       _ = r * Real.sqrt C := by rw [Real.sqrt_sq hr_nonneg]
230   have hradiusBase_nonneg : 0 <= radiusBase := by
231     dsimp only [radiusBase]
232     positivity
233   have hradius :
234       R * Real.sqrt logBudget + Real.sqrt lambda * S <=
235         r * radiusBase := by
236     calc
237       R * Real.sqrt logBudget + Real.sqrt lambda * S <=
238           R * (r * Real.sqrt C) + Real.sqrt lambda * S := by
239         exact add_le_add
240           (mul_le_mul_of_nonneg_left hsqrtLogBudget hR) le_rfl
241       _ <= r * radiusBase := by
242         dsimp only [radiusBase]
243         have hsqrtLambdaS_nonneg : 0 <= Real.sqrt lambda * S := by
244           positivity
245         nlinarith
246   have hwidth :
247       Real.sqrt r * Real.sqrt (2 * detBudget) <=
248         r * widthBase := by
249     calc
250       Real.sqrt r * Real.sqrt (2 * detBudget) <=
251           Real.sqrt r * Real.sqrt (2 * (r * D)) := by
252         exact mul_le_mul_of_nonneg_left
253           (Real.sqrt_le_sqrt
254             (mul_le_mul_of_nonneg_left hdetBudget (by norm_num)))
255           (Real.sqrt_nonneg r)
256       _ = Real.sqrt r * (Real.sqrt r * Real.sqrt (2 * D)) := by
257         rw [show 2 * (r * D) = r * (2 * D) by ring]
258         rw [Real.sqrt_mul hr_nonneg]
259       _ = r * widthBase := by
260         rw [← mul_assoc, Real.mul_self_sqrt hr_nonneg]
261   have hinitialGap_nonneg : 0 <= initialGap := by
262     dsimp only [initialGap, standardScalarInitialGapBound]
263     positivity
264   have hinitialGap :
265       initialGap <= r ^ 2 * initialGap := by
266     calc
267       initialGap = 1 * initialGap := by ring
268       _ <= r ^ 2 * initialGap := by
269         exact mul_le_mul_of_nonneg_right (by nlinarith [sq_nonneg r])
270           hinitialGap_nonneg
271   have hproduct :
272       2 *
273           (R * Real.sqrt logBudget + Real.sqrt lambda * S) *
274           (Real.sqrt r * Real.sqrt (2 * detBudget)) <=
275         r ^ 2 * (2 * radiusBase * widthBase) := by
276     calc
277       2 *
278           (R * Real.sqrt logBudget + Real.sqrt lambda * S) *
279           (Real.sqrt r * Real.sqrt (2 * detBudget)) <=
280           2 * (r * radiusBase) * (r * widthBase) := by
281         have hradius_nonneg :
282             0 <= R * Real.sqrt logBudget + Real.sqrt lambda * S := by
283           positivity
284         exact mul_le_mul
285           (mul_le_mul_of_nonneg_left hradius (by norm_num))
286           hwidth
287           (mul_nonneg (Real.sqrt_nonneg r)
288             (Real.sqrt_nonneg (2 * detBudget)))
289           (mul_nonneg (by norm_num)
290             (mul_nonneg hr_nonneg hradiusBase_nonneg))
291       _ = r ^ 2 * (2 * radiusBase * widthBase) := by ring
292   unfold telescopingHighProbabilityPseudoRegretBound
293   change
294     initialGap +
295         2 *
296           (R * Real.sqrt logBudget + Real.sqrt lambda * S) *
297           (Real.sqrt r * Real.sqrt (2 * detBudget)) <=
298       r ^ 2 *
299         telescopingHighProbabilityPseudoRegretQuadraticCoefficient
300           (Feature := Feature) R delta lambda S L2
301   calc
302     initialGap +
303         2 *
304           (R * Real.sqrt logBudget + Real.sqrt lambda * S) *
305           (Real.sqrt r * Real.sqrt (2 * detBudget)) <=
306         r ^ 2 * initialGap + r ^ 2 * (2 * radiusBase * widthBase) := by
307       exact add_le_add hinitialGap hproduct
308     _ =
309         r ^ 2 *
310           telescopingHighProbabilityPseudoRegretQuadraticCoefficient
311             (Feature := Feature) R delta lambda S L2 := by
312       dsimp only [initialGap, radiusBase, widthBase, C, D]
313       unfold telescopingHighProbabilityPseudoRegretQuadraticCoefficient
314       ring
