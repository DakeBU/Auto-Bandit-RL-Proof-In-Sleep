4 /-!
5 # Scheduled OFUL expected regret under a.e. aligned-window positive action costs
6 
7 This module replaces the earlier pointwise reach contract on every theoretical
8 trajectory by the measure-theoretically sufficient almost-everywhere contract
9 under the canonical generated trajectory law. It then specializes the route to
10 deterministic Nat-valued action costs.
11 
12 For a probability measure, the aligned-window contract is inconsistent when
13 `window = 0`, since every block is empty. At `budget = 0`, budget exhaustion is
14 immediate and the numerical round factor is one; the final uniform wrapper
15 still accepts the aligned-window contract, while the generic a.e.-reach
16 terminal can prove that zero-budget edge without it.
17 -/

28 /--
29 An almost-everywhere deterministic bound supplies the local square-integrable
30 finite-stopping contract. Values outside the support may remain unbounded.
31 -/
32 theorem squareIntegrableFiniteStoppingTime_of_bounded_ae
33     {Omega : Type v} [mOmega : MeasurableSpace Omega]
34     (mu : Measure Omega) [IsFiniteMeasure mu]
35     {F : Filtration Nat mOmega}
36     (tau : Omega -> WithTop Nat)
37     (htau : IsStoppingTime F tau)
38     (bound : Nat)
39     (htau_le : ∀ᵐ omega ∂mu, tau omega <= (bound : WithTop Nat)) :
40     SquareIntegrableFiniteStoppingTime mu tau := by
41   have hrounds_measurable :
42       Measurable
43         (fun omega =>
44           (((tau omega).untopA + 1 : Nat) : Real)) :=
45     (measurable_of_countable
46       (fun count : Nat => (((count + 1 : Nat) : Nat) : Real))).comp
47         htau.measurable'.untopA
48   refine ⟨?_, ?_⟩
49   · filter_upwards [htau_le] with omega hle
50     intro htop
51     simp [htop] at hle
52   · refine MemLp.of_bound hrounds_measurable.aestronglyMeasurable
53       (((bound + 1 : Nat) : Real)) ?_
54     filter_upwards [htau_le] with omega hle
55     rw [Real.norm_eq_abs, abs_of_nonneg (Nat.cast_nonneg _)]
56     exact_mod_cast Nat.succ_le_succ
57       (WithTop.untopA_le hle)

59 /--
60 The exact round-count second moment obeys the same numerical square bound when
61 the stopping-time bound is only almost everywhere.
62 -/
63 theorem stoppingTimeRoundSecondMoment_le_sq_of_bounded_ae
64     {Omega : Type v} [MeasurableSpace Omega]
65     (mu : Measure Omega) [IsProbabilityMeasure mu]
66     (tau : Omega -> WithTop Nat)
67     (hstop : SquareIntegrableFiniteStoppingTime mu tau)
68     (bound : Nat)
69     (htau_le : ∀ᵐ omega ∂mu, tau omega <= (bound : WithTop Nat)) :
70     stoppingTimeRoundSecondMoment mu tau hstop <=
71       (((bound + 1 : Nat) : Real)) ^ 2 := by
72   unfold stoppingTimeRoundSecondMoment
73   have hsquare :
74       Integrable
75         (fun omega =>
76           ((((tau omega).untopA + 1 : Nat) : Real)) ^ 2) mu :=
77     hstop.memLp_rounds.integrable_sq
78   have hconst :
79       Integrable
80         (fun _omega : Omega => (((bound + 1 : Nat) : Real)) ^ 2) mu :=
81     integrable_const _
82   calc
83     integral mu
84         (fun omega =>
85           ((((tau omega).untopA + 1 : Nat) : Real)) ^ 2) <=
86         integral mu
87           (fun _omega : Omega => (((bound + 1 : Nat) : Real)) ^ 2) := by
88       apply integral_mono_ae hsquare hconst
89       filter_upwards [htau_le] with omega hle
90       apply
91         (sq_le_sq₀
92           (Nat.cast_nonneg ((tau omega).untopA + 1))
93           (Nat.cast_nonneg (bound + 1))).2
94       have hnat :
95           (tau omega).untopA + 1 <= bound + 1 :=
96         Nat.succ_le_succ (WithTop.untopA_le hle)
97       exact Nat.cast_le.2 hnat
98     _ = (((bound + 1 : Nat) : Real)) ^ 2 := by
99       rw [integral_const]
100       simp [MeasureTheory.probReal_univ]

276 /--
277 Almost every trajectory selects an action with positive deterministic cost at
278 least once in every aligned half-open block.
279 -/
280 def AlignedWindowPositiveActionCostAE
281     {K : Nat}
282     (mu : Measure (Nat -> Fin K × Real))
283     (actionCost : Fin K -> Nat)
284     (window : Nat) : Prop :=
285   ∀ᵐ trajectory ∂mu, forall block,
286     exists s,
287       s ∈ Finset.Ico (block * window) ((block + 1) * window) /\
288         1 <= actionCost (trajectory s).1

290 /--
291 The all-block a.e. contract is equivalent to proving the selected positive-cost
292 witness almost surely for each natural block separately.
293 -/
294 theorem alignedWindowPositiveActionCostAE_iff_forall_ae
295     {K : Nat}
296     (mu : Measure (Nat -> Fin K × Real))
297     (actionCost : Fin K -> Nat)
298     (window : Nat) :
299     AlignedWindowPositiveActionCostAE mu actionCost window ↔
300       forall block,
301         ∀ᵐ trajectory ∂mu,
302           exists s,
303             s ∈ Finset.Ico (block * window) ((block + 1) * window) /\

307 /--
308 One positive selected-action cost in each aligned block makes the whole block
309 sum positive almost surely.
310 -/
311 theorem
312     canonicalActionCostProcess_alignedWindowPositive_ae
313     {K : Nat}
314     (mu : Measure (Nat -> Fin K × Real))
315     (actionCost : Fin K -> Nat)
316     (window : Nat)
317     (hwindow : AlignedWindowPositiveActionCostAE
318       mu actionCost window) :
319     ∀ᵐ trajectory ∂mu, forall block,
320       1 <=
321         (Finset.Ico (block * window) ((block + 1) * window)).sum
322           (fun s => canonicalActionCostProcess actionCost s trajectory) := by
323   filter_upwards [hwindow] with trajectory htrajectory
324   intro block
325   obtain ⟨s, hs, hpositive⟩ := htrajectory block
326   calc
327     1 <= canonicalActionCostProcess actionCost s trajectory := by
328       simpa only [canonicalActionCostProcess] using hpositive
329     _ <=
330         (Finset.Ico (block * window) ((block + 1) * window)).sum
331           (fun t => canonicalActionCostProcess actionCost t trajectory) := by
332       exact
333         Finset.single_le_sum
334           (fun t _ht => Nat.zero_le
335             (canonicalActionCostProcess actionCost t trajectory))
336           hs
