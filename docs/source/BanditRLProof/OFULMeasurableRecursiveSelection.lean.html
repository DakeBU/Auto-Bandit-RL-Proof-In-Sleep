6 # Measurable recursive OFUL selection
7 
8 This module replaces the nonconstructive finite OFUL argmax by the existing
9 strict finite fold on `Fin K`.  Under measurable score coordinates, the fold
10 defines a deterministic history policy.  The canonical history trajectory
11 then realizes that selector at every successor time.
12 -/

48     [MeasurableSpace Reward]
49     [Fintype Feature] [DecidableEq Feature]
50     (hK : 0 < K)
51     (thetaHat : (n : Nat) ->
52       History.FinitePairHistory (Fin K) Reward n -> Feature -> Real)
53     (V : (n : Nat) ->
54       History.FinitePairHistory (Fin K) Reward n ->
55         Matrix Feature Feature Real)
56     (beta : (n : Nat) ->
57       History.FinitePairHistory (Fin K) Reward n -> Real)
58     (candidateFeature : (n : Nat) ->
59       History.FinitePairHistory (Fin K) Reward n ->
60         Fin K -> Feature -> Real)

132       History.FinitePairHistory (Fin K) Reward n -> Feature -> Real)
133     (V : (n : Nat) ->
134       History.FinitePairHistory (Fin K) Reward n ->
135         Matrix Feature Feature Real)

137       History.FinitePairHistory (Fin K) Reward n -> Real)

183 /--
184 Along the canonical recursive trajectory, the successor action is almost
185 surely the measurable OFUL selector evaluated on the realized finite history.
186 -/
187 theorem canonicalHistoryTrajectory_action_succ_ae_eq_finiteHistoryOptimisticAction
188     {K : Nat} {Reward : Type u} {Feature : Type v}
189     [MeasurableSpace Reward] [StandardBorelSpace Reward] [Nonempty Reward]
190     [Fintype Feature] [DecidableEq Feature]
191     (hK : 0 < K)
192     (thetaHat : (n : Nat) ->
193       History.FinitePairHistory (Fin K) Reward n -> Feature -> Real)
194     (V : (n : Nat) ->
195       History.FinitePairHistory (Fin K) Reward n ->
196         Matrix Feature Feature Real)
197     (beta : (n : Nat) ->
198       History.FinitePairHistory (Fin K) Reward n -> Real)
199     (candidateFeature : (n : Nat) ->
200       History.FinitePairHistory (Fin K) Reward n ->
201         Fin K -> Feature -> Real)
202     (hscores : forall (n : Nat) (action : Fin K),
203       Measurable (fun history : History.FinitePairHistory (Fin K) Reward n =>
204         finiteHistoryOptimisticScore thetaHat V beta candidateFeature
205           n history action))
206     (environment : Thompson.HistoryEnvironment (Fin K) Reward)
207     (n : Nat) :
208     ∀ᵐ trajectory ∂
209         Thompson.canonicalHistoryTrajectoryMeasure
210           (finiteHistoryOptimisticAlgorithm
211             hK thetaHat V beta candidateFeature hscores)
212           environment,
213       Thompson.canonicalHistoryTrajectoryAction trajectory (n + 1) =
214         finiteHistoryOptimisticAction
215           hK thetaHat V beta candidateFeature n
216           (History.finitePairHistoryOfTrace
217             (Thompson.canonicalHistoryTrajectoryAction trajectory)
218             (Thompson.canonicalHistoryTrajectoryReward trajectory) n) := by
219   letI : Nonempty (Fin K) := ⟨Fin.mk 0 hK⟩
220   let algorithm :=
221     finiteHistoryOptimisticAlgorithm
222       hK thetaHat V beta candidateFeature hscores
223   let mu := Thompson.canonicalHistoryTrajectoryMeasure algorithm environment
224   let history := fun trajectory : (k : Nat) -> Fin K × Reward =>
225     History.finitePairHistoryOfTrace
226       (Thompson.canonicalHistoryTrajectoryAction trajectory)
227       (Thompson.canonicalHistoryTrajectoryReward trajectory) n
228   let nextAction := fun trajectory : (k : Nat) -> Fin K × Reward =>
229     Thompson.canonicalHistoryTrajectoryAction trajectory (n + 1)
230   let selector :=
231     finiteHistoryOptimisticAction
232       hK thetaHat V beta candidateFeature n
233   have hhistory : Measurable history :=
234     History.measurable_finitePairHistoryOfTrace
235       Thompson.canonicalHistoryTrajectoryAction
236       Thompson.canonicalHistoryTrajectoryReward
237       Thompson.measurable_canonicalHistoryTrajectoryAction_apply
238       Thompson.measurable_canonicalHistoryTrajectoryReward_apply n
239   have hnextAction : Measurable nextAction :=
240     Thompson.measurable_canonicalHistoryTrajectoryAction_apply (n + 1)
241   have hselector : Measurable selector :=
242     measurable_finiteHistoryOptimisticAction
243       hK thetaHat V beta candidateFeature n (hscores n)
244   have hpolicy :=
245     Thompson.policy_condDistrib_of_historyAlgorithmEnvironmentSequence
246       mu
247       Thompson.canonicalHistoryTrajectoryAction
248       Thompson.canonicalHistoryTrajectoryReward
249       algorithm environment
250       (Thompson.canonicalHistoryAlgorithmEnvironmentSequence
251         algorithm environment)
252       n
253   have hjoint :
254       mu.map (fun trajectory => (history trajectory, nextAction trajectory)) =
255         mu.map history ⊗ₘ algorithm.policy n :=
256     (ProbabilityTheory.condDistrib_ae_eq_iff_measure_eq_compProd
257       history hnextAction.aemeasurable (algorithm.policy n)).mp <| by
258         simpa [history, nextAction] using hpolicy
259   have hp : MeasurableSet
260       {pair : History.FinitePairHistory (Fin K) Reward n × Fin K |
261         pair.2 = selector pair.1} :=
262     measurableSet_eq_fun measurable_snd
263       (hselector.comp measurable_fst)
264   have hgraph :
265       ∀ᵐ pair ∂ mu.map history ⊗ₘ algorithm.policy n,
266         pair.2 = selector pair.1 := by
267     apply Measure.ae_compProd_of_ae_ae hp
268     filter_upwards [] with historyValue
269     simp [algorithm, selector]
270   rw [← hjoint] at hgraph
271   exact
272     (ae_map_iff
273       ((hhistory.prod hnextAction).aemeasurable) hp).mp hgraph
