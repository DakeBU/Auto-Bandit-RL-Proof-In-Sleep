4 /-!
5 # Variance-sensitive pure-Hedge cross-weight concentration
6 
7 This module replaces the range-squared Hoeffding proxy for the pure-Hedge
8 cross-weighted estimator by a fixed-tilt second-moment budget.  The sign is
9 `predictable pure loss - observed cross-weighted loss`, as consumed by the
10 sampled Hedge regret decomposition.
11 -/

20 /-- On the finite support, the cross-weighted estimator has only the sampled
21 coordinate as a nonzero summand. -/
22 theorem weightedImportanceWeightedLoss_eq_selected
23     {Action : Type u} [DecidableEq Action]
24     (arms : Finset Action) (prob weight loss : Action -> Real)
25     (chosen : Action) (hchosen : chosen ∈ arms) :
26     weightedImportanceWeightedLoss arms prob weight loss chosen =
27       weight chosen * loss chosen / prob chosen := by
28   unfold weightedImportanceWeightedLoss
29   rw [Finset.sum_eq_single chosen]
30   · simp [importanceWeightedLoss]
31     ring
32   · intro action _haction hne
33     simp [importanceWeightedLoss, Ne.symm hne]
34   · exact fun hnotmem => (hnotmem hchosen).elim

36 /-- The centered pure-Hedge cross-weighted estimator has second moment at most
37 the reciprocal exploration floor. -/
38 theorem sum_prob_mul_sq_weightedEstimatorMeanMinusRaw_le_inv_floor
39     {Action : Type u} [DecidableEq Action]
40     (arms : Finset Action) (prob weight loss : Action -> Real)
41     (hprob : FiniteActionDistribution arms prob)
42     (hweight : FiniteActionDistribution arms weight)
43     (epsilon : Real) (hepsilon : 0 < epsilon)
44     (hfloor : forall action, action ∈ arms -> epsilon <= prob action)
45     (hloss : forall action, action ∈ arms -> loss action ∈ Set.Icc (0 : Real) 1) :
46     let mean := arms.sum (fun action => weight action * loss action)
47     arms.sum (fun chosen =>
48         prob chosen *
49           (mean - weightedImportanceWeightedLoss arms prob weight loss chosen) ^ 2) <=
50       1 / epsilon := by
51   dsimp only
52   let mean := arms.sum (fun action => weight action * loss action)
53   let raw := fun chosen =>
54     weightedImportanceWeightedLoss arms prob weight loss chosen
55   have hprob_pos (action : Action) (haction : action ∈ arms) :
56       0 < prob action := hepsilon.trans_le (hfloor action haction)
57   have hfirst : arms.sum (fun chosen => prob chosen * raw chosen) = mean := by
58     simpa [raw, mean] using
59       sum_prob_mul_weightedImportanceWeightedLoss_eq_weightedLoss
60         arms prob weight loss (fun action haction =>
61           (hprob_pos action haction).ne')
62   have hsecond : arms.sum (fun chosen => prob chosen * (raw chosen) ^ 2) <=
63       1 / epsilon := by
64     calc
65       arms.sum (fun chosen => prob chosen * (raw chosen) ^ 2) <=
66           arms.sum (fun chosen => weight chosen / epsilon) := by
67         apply Finset.sum_le_sum
68         intro chosen hchosen
69         have hp := hprob_pos chosen hchosen
70         have hq_nonneg := hweight.nonneg chosen hchosen
71         have hq_le_one : weight chosen <= 1 := by
72           calc
73             weight chosen <= arms.sum weight :=
74               Finset.single_le_sum hweight.nonneg hchosen
75             _ = 1 := hweight.sum_eq_one
76         have hl := hloss chosen hchosen
77         have hq_sq_le : (weight chosen) ^ 2 <= weight chosen := by
78           nlinarith
79         have hl_sq_le : (loss chosen) ^ 2 <= 1 := by
80           simpa using (sq_le_sq₀ hl.1 zero_le_one).2 hl.2
81         have hnum :
82             (weight chosen) ^ 2 * (loss chosen) ^ 2 <= weight chosen := by
83           calc
84             (weight chosen) ^ 2 * (loss chosen) ^ 2 <=
85                 (weight chosen) ^ 2 * 1 :=
86               mul_le_mul_of_nonneg_left hl_sq_le (sq_nonneg _)
87             _ <= weight chosen := by simpa using hq_sq_le
88         change prob chosen *
89           (weightedImportanceWeightedLoss arms prob weight loss chosen) ^ 2 <= _
90         rw [weightedImportanceWeightedLoss_eq_selected
91           arms prob weight loss chosen hchosen]
92         calc
93           prob chosen * (weight chosen * loss chosen / prob chosen) ^ 2 =
94               (weight chosen) ^ 2 * (loss chosen) ^ 2 / prob chosen := by
95             field_simp
96           _ <= weight chosen / prob chosen :=
97             div_le_div_of_nonneg_right hnum hp.le
98           _ <= weight chosen / epsilon := by
99             simpa [div_eq_mul_inv] using
100               mul_le_mul_of_nonneg_left
101                 (one_div_le_one_div_of_le hepsilon (hfloor chosen hchosen))
102                 hq_nonneg
103       _ = 1 / epsilon := by
104         rw [← Finset.sum_div, hweight.sum_eq_one, one_div]
105   have hcenter (chosen : Action) :
106       prob chosen * (mean - raw chosen) ^ 2 =
107         (prob chosen * (raw chosen) ^ 2 -
108           2 * mean * (prob chosen * raw chosen)) +
109           mean ^ 2 * prob chosen := by
110     ring
111   change arms.sum (fun chosen => prob chosen * (mean - raw chosen) ^ 2) <= _
112   simp_rw [hcenter]
113   rw [Finset.sum_add_distrib, Finset.sum_sub_distrib,
114     ← Finset.mul_sum, ← Finset.mul_sum, hfirst, hprob.sum_eq_one]
115   nlinarith [sq_nonneg mean]

117 /-- Fixed-tilt MGF budget for the sign used by the pure-Hedge regret route. -/
118 theorem finiteActionWeightedEstimatorMeanMinusRaw_hasMGFUpperBoundAt
119     {Action : Type u} [MeasurableSpace Action] [MeasurableSingletonClass Action]
120     [DecidableEq Action]
121     (arms : Finset Action) (harms : arms.Nonempty)
122     (prob weight loss : Action -> Real)
123     (hprob : FiniteActionDistribution arms prob)
124     (hweight : FiniteActionDistribution arms weight)
125     (epsilon : Real) (hepsilon : 0 < epsilon)
126     (hfloor : forall action, action ∈ arms -> epsilon <= prob action)
127     (hloss : forall action, action ∈ arms -> loss action ∈ Set.Icc (0 : Real) 1)
128     (tilt : Real) (htilt_nonneg : 0 <= tilt) (htilt_le : tilt <= epsilon) :
129     let mean := arms.sum (fun action => weight action * loss action)
130     Concentration.HasMGFUpperBoundAt
131       (fun chosen =>
132         mean - weightedImportanceWeightedLoss arms prob weight loss chosen)
133       tilt (tilt ^ 2 / epsilon) (finiteActionMeasure arms prob) := by
134   dsimp only
135   let mean := arms.sum (fun action => weight action * loss action)
136   let raw := fun chosen =>
137     weightedImportanceWeightedLoss arms prob weight loss chosen
138   let X := fun chosen => mean - raw chosen
139   letI : IsProbabilityMeasure (finiteActionMeasure arms prob) :=
140     finiteActionMeasure_isProbabilityMeasure arms prob hprob
141   have hprob_pos (action : Action) (haction : action ∈ arms) :
142       0 < prob action := hepsilon.trans_le (hfloor action haction)
143   obtain ⟨anchor, hanchor⟩ := harms
144   have hprob_anchor_le_one : prob anchor <= 1 := by
145     calc
146       prob anchor <= arms.sum prob := Finset.single_le_sum hprob.nonneg hanchor
147       _ = 1 := hprob.sum_eq_one
148   have hepsilon_le_one : epsilon <= 1 :=
149     (hfloor anchor hanchor).trans hprob_anchor_le_one
150   have hinv_nonneg : 0 <= 1 / epsilon := (one_div_pos.mpr hepsilon).le
151   have hone_le_inv : 1 <= 1 / epsilon := by
152     rw [le_div_iff₀ hepsilon]
153     simpa using hepsilon_le_one
154   have hmean_nonneg : 0 <= mean := by
155     dsimp only [mean]
156     exact Finset.sum_nonneg fun action haction =>
157       mul_nonneg (hweight.nonneg action haction) (hloss action haction).1
158   have hmean_le_one : mean <= 1 := by
159     dsimp only [mean]
160     calc
161       arms.sum (fun action => weight action * loss action) <=
162           arms.sum weight := by
163         apply Finset.sum_le_sum
164         intro action haction
165         simpa using mul_le_of_le_one_right
166           (hweight.nonneg action haction) (hloss action haction).2
167       _ = 1 := hweight.sum_eq_one
168   have hX_bound (chosen : Action) (hchosen : chosen ∈ arms) :
169       |X chosen| <= 1 / epsilon := by
170     have hq_nonneg := hweight.nonneg chosen hchosen
171     have hq_le_one : weight chosen <= 1 := by
172       calc
173         weight chosen <= arms.sum weight :=
174           Finset.single_le_sum hweight.nonneg hchosen
175         _ = 1 := hweight.sum_eq_one
176     have hl := hloss chosen hchosen
177     have hraw_nonneg : 0 <= raw chosen := by
178       change 0 <= weightedImportanceWeightedLoss arms prob weight loss chosen
179       rw [weightedImportanceWeightedLoss_eq_selected
180         arms prob weight loss chosen hchosen]
181       exact div_nonneg (mul_nonneg hq_nonneg hl.1)
182         (hprob.nonneg chosen hchosen)
183     have hraw_le : raw chosen <= 1 / epsilon := by
184       change weightedImportanceWeightedLoss arms prob weight loss chosen <= 1 / epsilon
185       rw [weightedImportanceWeightedLoss_eq_selected
186         arms prob weight loss chosen hchosen]
187       calc
188         weight chosen * loss chosen / prob chosen <= 1 / prob chosen := by
189           apply div_le_div_of_nonneg_right _ (hprob_pos chosen hchosen).le
190           calc
191             weight chosen * loss chosen <= 1 * loss chosen :=
192               mul_le_mul_of_nonneg_right hq_le_one hl.1
193             _ <= 1 := by simpa using hl.2
194         _ <= 1 / epsilon :=
195           one_div_le_one_div_of_le hepsilon (hfloor chosen hchosen)
196     dsimp only [X]
197     rw [abs_le]
198     constructor
199     · linarith
200     · exact (sub_le_self mean hraw_nonneg).trans (hmean_le_one.trans hone_le_inv)
201   constructor
202   · intro s
203     rw [finiteActionMeasure]
204     refine integrable_finset_sum_measure.2 fun action haction => ?_
205     exact (integrable_dirac (by simp)).smul_measure ENNReal.ofReal_ne_top
206   · rw [ProbabilityTheory.mgf,
207       integral_finiteActionMeasure_eq_sum arms prob hprob]
208     have hexp (chosen : Action) (hchosen : chosen ∈ arms) :
209         Real.exp (tilt * X chosen) <=
210           1 + tilt * X chosen + (tilt * X chosen) ^ 2 := by
211       apply Concentration.exp_le_one_add_self_add_sq_of_abs_le_one
212       rw [abs_mul, abs_of_nonneg htilt_nonneg]
213       calc
214         tilt * |X chosen| <= tilt * (1 / epsilon) :=
215           mul_le_mul_of_nonneg_left (hX_bound chosen hchosen) htilt_nonneg
216         _ <= epsilon * (1 / epsilon) :=
217           mul_le_mul_of_nonneg_right htilt_le hinv_nonneg
218         _ = 1 := by field_simp
219     calc
220       arms.sum (fun chosen => prob chosen * Real.exp (tilt * X chosen)) <=
221           arms.sum (fun chosen =>
222             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2)) := by
223         apply Finset.sum_le_sum
224         intro chosen hchosen
225         exact mul_le_mul_of_nonneg_left (hexp chosen hchosen)
226           (hprob.nonneg chosen hchosen)
227       _ = 1 + tilt * arms.sum (fun chosen => prob chosen * X chosen) +
228           tilt ^ 2 * arms.sum (fun chosen => prob chosen * (X chosen) ^ 2) := by
229         have hterm (chosen : Action) :
230             prob chosen * (1 + tilt * X chosen + (tilt * X chosen) ^ 2) =
231               (prob chosen + tilt * (prob chosen * X chosen)) +
232                 tilt ^ 2 * (prob chosen * (X chosen) ^ 2) := by ring
233         simp_rw [hterm, Finset.sum_add_distrib, ← Finset.mul_sum,
234           hprob.sum_eq_one]
235       _ <= 1 + tilt ^ 2 * (1 / epsilon) := by
236         have hmean_zero : arms.sum (fun chosen => prob chosen * X chosen) = 0 := by
237           dsimp only [X, raw, mean]
238           simp_rw [mul_sub]
239           rw [Finset.sum_sub_distrib, ← Finset.sum_mul, hprob.sum_eq_one,
240             sum_prob_mul_weightedImportanceWeightedLoss_eq_weightedLoss
241               arms prob weight loss (fun action haction =>
242                 (hprob_pos action haction).ne')]
243           ring
244         rw [hmean_zero, mul_zero, add_zero]
245         have hsecond_base :
246             arms.sum (fun chosen => prob chosen * (X chosen) ^ 2) <=
247               1 / epsilon := by
248           simpa only [X, raw, mean] using
249             sum_prob_mul_sq_weightedEstimatorMeanMinusRaw_le_inv_floor
250               arms prob weight loss hprob hweight epsilon hepsilon hfloor hloss
251         have hsecond := mul_le_mul_of_nonneg_left hsecond_base
252           (sq_nonneg tilt)
253         simpa [add_comm] using add_le_add_left hsecond 1
254       _ <= Real.exp (tilt ^ 2 / epsilon) := by
255         rw [show tilt ^ 2 * (1 / epsilon) = tilt ^ 2 / epsilon by ring]
256         simpa [add_comm] using Real.add_one_le_exp (tilt ^ 2 / epsilon)
