namespace WeightedIdentityProbe

abbrev V : BanditRL.OnlineGradientDescent.Domain ℝ := BanditRL.OnlineHuber.fullSpace
def signedA (t : ℕ) : ℝ := if t = 0 then -2 else if t = 1 then 3 else if t = 2 then -1 else 4
def signedW (t : ℕ) : ℝ := if t = 0 then 0 else if t = 1 then -2 else 1
def eta (t : ℕ) : ℝ := if t = 1 then 2 else 1
def feedback (t : ℕ) : ℝ := if t = 0 then 1 else if t = 1 then -2 else 2
def realLoss (t : ℕ) (z : ℝ) : ℝ := inner ℝ (feedback t) z
def extendedLoss (t : ℕ) (z : ℝ) : EReal := ((inner ℝ (feedback t) z + 0 : ℝ) : EReal)
def policy : BanditRL.OnlineSubgradientPolicy.SupportPolicy (E := ℝ) := fun t _ _ _ => feedback t
abbrev ogd (t : ℕ) : ℝ := BanditRL.OnlineGradientDescent.iterateVariable V eta realLoss 0 t
abbrev osd (t : ℕ) : ℝ := BanditRL.OnlineSubgradientPolicy.output V eta extendedLoss 0 policy t

theorem signed_weights_complete :
    signedA 0 = -2 ∧ signedA 1 = 3 ∧ signedA 2 = -1 ∧ signedA 3 = 4 ∧
    signedW 0 = 0 ∧ signedW 1 = -2 ∧ signedW 2 = 1 ∧
    (∑ t ∈ range 3, (signedA t - signedA (t + 1)) * signedW t) =
      signedA 0 * signedW 0 - signedA 3 * signedW 2 +
        ∑ t ∈ range 2, (signedW (t + 1) - signedW t) * signedA (t + 1) ∧
    (∑ t ∈ range 3, (signedA t - signedA (t + 1)) * signedW t) = -13 ∧
    -(signedA 3 * signedW 2) = -4

theorem one_step_free_terminal_complete (a w : ℕ → ℝ) :
    (∑ t ∈ range 1, (a t - a (t + 1)) * w t) =
      a 0 * w 0 - a 1 * w 0 +
        ∑ t ∈ range 0, (w (t + 1) - w t) * a (t + 1) ∧
    (∑ t ∈ range 0, (w (t + 1) - w t) * a (t + 1)) = 0

theorem constant_zero_weights_complete (a : ℕ → ℝ) (c : ℝ) (T : ℕ) (hT : 0 < T) :
    (∑ t ∈ range T, (a t - a (t + 1)) * c) =
      a 0 * c - a T * c + ∑ t ∈ range (T - 1), (c - c) * a (t + 1) ∧
    (∑ t ∈ range T, (a t - a (t + 1)) * c) = (a 0 - a T) * c ∧
    (∑ t ∈ range T, (a t - a (t + 1)) * (0 : ℝ)) =
      a 0 * 0 - a T * 0 + ∑ t ∈ range (T - 1), ((0 : ℝ) - 0) * a (t + 1) ∧
    (∑ t ∈ range T, (a t - a (t + 1)) * (0 : ℝ)) = 0

theorem ogd_three_step_complete :
    eta 0 = 1 ∧ eta 1 = 2 ∧ eta 2 = 1 ∧
    (∀ t < 3, 0 < eta t) ∧ ¬ (∀ t, t + 1 < 3 → eta (t + 1) ≤ eta t) ∧
    feedback 0 = 1 ∧ feedback 1 = -2 ∧ feedback 2 = 2 ∧
    (∀ t < 3, BanditRL.OnlineGradientDescentSource.FeasibleRegularLoss V (realLoss t)) ∧
    (∀ t < 3, gradient (realLoss t) (ogd t) = feedback t) ∧
    ogd 0 = 0 ∧ ogd 1 = -1 ∧ ogd 2 = 3 ∧ ogd 3 = 1 ∧
    (∑ t ∈ range 3, (‖ogd t - 2‖ ^ 2 - ‖ogd (t + 1) - 2‖ ^ 2) / (2 * eta t)) =
      ‖ogd 0 - 2‖ ^ 2 / (2 * eta 0) - ‖ogd 3 - 2‖ ^ 2 / (2 * eta 2) +
        (1 / 2 : ℝ) * ∑ t ∈ range 2,
          (1 / eta (t + 1) - 1 / eta t) * ‖ogd (t + 1) - 2‖ ^ 2 ∧
    (∑ t ∈ range 3, (‖ogd t - 2‖ ^ 2 - ‖ogd (t + 1) - 2‖ ^ 2) / (2 * eta t)) = -(1 / 2 : ℝ) ∧
    ‖ogd 0 - 2‖ ^ 2 / (2 * eta 0) = 2 ∧
    -(‖ogd 3 - 2‖ ^ 2 / (2 * eta 2)) = -(1 / 2 : ℝ) ∧
    (1 / 2 : ℝ) * ∑ t ∈ range 2,
      (1 / eta (t + 1) - 1 / eta t) * ‖ogd (t + 1) - 2‖ ^ 2 = -2

theorem osd_three_step_complete :
    eta 0 = 1 ∧ eta 1 = 2 ∧ eta 2 = 1 ∧
    (∀ t < 3, 0 < eta t) ∧ ¬ (∀ t, t + 1 < 3 → eta (t + 1) ≤ eta t) ∧
    feedback 0 = 1 ∧ feedback 1 = -2 ∧ feedback 2 = 2 ∧
    (∀ t < 3, BanditRL.OnlineSubgradientDescent.SubdifferentiableOn V (extendedLoss t)) ∧
    BanditRL.OnlineSubgradientPolicy.LegalFeedback V eta extendedLoss 0 policy 3 ∧
    (∀ t < 3, BanditRL.OnlineSubgradientPolicy.selected V eta extendedLoss 0 policy t = feedback t) ∧
    osd 0 = 0 ∧ osd 1 = -1 ∧ osd 2 = 3 ∧ osd 3 = 1 ∧
    (∑ t ∈ range 3, (‖osd t - 2‖ ^ 2 - ‖osd (t + 1) - 2‖ ^ 2) / (2 * eta t)) =
      ‖osd 0 - 2‖ ^ 2 / (2 * eta 0) - ‖osd 3 - 2‖ ^ 2 / (2 * eta 2) +
        (1 / 2 : ℝ) * ∑ t ∈ range 2,
          (1 / eta (t + 1) - 1 / eta t) * ‖osd (t + 1) - 2‖ ^ 2 ∧
    (∑ t ∈ range 3, (‖osd t - 2‖ ^ 2 - ‖osd (t + 1) - 2‖ ^ 2) / (2 * eta t)) = -(1 / 2 : ℝ) ∧
    ‖osd 0 - 2‖ ^ 2 / (2 * eta 0) = 2 ∧
    -(‖osd 3 - 2‖ ^ 2 / (2 * eta 2)) = -(1 / 2 : ℝ) ∧
    (1 / 2 : ℝ) * ∑ t ∈ range 2,
      (1 / eta (t + 1) - 1 / eta t) * ‖osd (t + 1) - 2‖ ^ 2 = -2

end WeightedIdentityProbe
