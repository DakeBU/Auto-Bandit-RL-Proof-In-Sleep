variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [FiniteDimensional ℝ E]
abbrev Domain := BanditRL.OnlineGradientDescent.Domain E
abbrev SupportPolicy := BanditRL.OnlineSubgradientPolicy.SupportPolicy (E := E)
local instance : DecidableEq E := Classical.decEq E

def A (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) : (t : ℕ) → (Fin (t + 1) → E) × ℝ := by
  classical
  exact Nat.rec (motive := fun t => (Fin (t + 1) → E) × ℝ) ((fun _ => x₁), 0)
    (fun t s =>
      let g := p t (fun i => loss i.val) s.1 (loss t)
      let S := s.2 + ‖g‖ ^ 2
      (Fin.snoc s.1 (if g = 0 then s.1 (Fin.last t) else
        BanditRL.OnlineGradientDescent.project V
          (s.1 (Fin.last t) - (α * D / Real.sqrt S) • g)), S))

def H (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (t : ℕ) : Fin (t + 1) → E :=
  (A V α D loss x₁ p t).1

def Q (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (t : ℕ) : ℝ :=
  (A V α D loss x₁ p t).2

def X (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (t : ℕ) : E :=
  H V α D loss x₁ p t (Fin.last t)

def G (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (t : ℕ) : E :=
  p t (fun i => loss i.val) (H V α D loss x₁ p t) (loss t)

def R (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (t : ℕ) : ℝ :=
  α * D / Real.sqrt (Q V α D loss x₁ p t + ‖G V α D loss x₁ p t‖ ^ 2)

def L (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (T : ℕ) : Prop :=
  ∀ t < T, G V α D loss x₁ p t ∈
    SourceSubdifferential (loss t) (X V α D loss x₁ p t)

def F (V : Domain (E := E)) (α D : ℝ) (loss : ℕ → E → EReal)
    (x₁ : E) (p : SupportPolicy (E := E)) (u : E) (T : ℕ) : ℝ :=
  ∑ t ∈ range T, ((loss t (X V α D loss x₁ p t)).toReal - (loss t u).toReal)

noncomputable def B (A S η : ℝ) : ℝ := A / (2 * η) + η * S / 2

abbrev V := BanditRL.OnlineGradientDescent.unitInterval

def W : Domain (E := ℝ) where
  carrier := {0}
  nonempty := ⟨0, rfl⟩
  closed := isClosed_singleton
  convex := convex_singleton 0

def feedback (t : ℕ) : ℝ := if t = 1 then 3 else if t = 3 then -4 else 0
def loss (t : ℕ) (x : ℝ) : EReal := ((feedback t * x : ℝ) : EReal)
def policy : SupportPolicy (E := ℝ) := fun t _ _ _ => feedback t
def futureLoss (t : ℕ) (x : ℝ) : EReal := if t < 2 then loss t x else (7 : EReal)
def zeroLoss (_ : ℕ) (_ : ℝ) : EReal := 0
def zeroPolicy : SupportPolicy (E := ℝ) := fun _ _ _ _ => 0
abbrev x (α : ℝ) (t : ℕ) := X V α 1 loss (1 / 2) policy t
abbrev q (α : ℝ) (t : ℕ) := Q V α 1 loss (1 / 2) policy t
abbrev r (α : ℝ) (t : ℕ) := R V α 1 loss (1 / 2) policy t

theorem certificate_1 (U : Domain (E := ℝ)) (t : ℕ) :
    IsConvexExtended (loss t) ∧
    BanditRL.OnlineSubgradientDescent.SubdifferentiableOn U (loss t) ∧
    ∀ z : ℝ, feedback t ∈ SourceSubdifferential (loss t) z

theorem certificate_2 :
    (∀ α : ℝ, ∀ t : ℕ, G V α 1 loss (1 / 2) policy t = feedback t) ∧
    (∀ α : ℝ, ∀ T : ℕ, T ≤ 4 →
      q α T = if T ≤ 1 then 0 else if T ≤ 3 then 9 else 25)

theorem certificate_3 :
    x 1 0 = 1 / 2 ∧ x 1 1 = 1 / 2 ∧ x 1 2 = 0 ∧ x 1 3 = 0 ∧ x 1 4 = 4 / 5 ∧
    r 1 0 = 0 ∧ r 1 1 = 1 / 3 ∧ r 1 2 = 1 / 3 ∧ r 1 3 = 1 / 5 ∧
    F V 1 1 loss (1 / 2) policy 0 4 = 3 / 2 ∧ ‖x 1 4 - 0‖ ^ 2 = 16 / 25 ∧
    BanditRL.OnlineGradientDescent.project V (-1 / 2) = 0 ∧ (-1 / 2 : ℝ) ≠ 0

theorem certificate_4 :
    F V 1 1 loss (1 / 2) policy 0 4 ≤ 59 / 10 ∧
    F V 1 1 loss (1 / 2) policy 0 4 ≤ 15 / 2 ∧
    F V (Real.sqrt 2 / 2) 1 loss (1 / 2) policy 0 4 ≤ Real.sqrt 50 ∧
    Real.sqrt 50 = Real.sqrt 2 * sInf {b : ℝ | ∃ η : ℝ, 0 < η ∧
      b = B 1 25 η}

theorem certificate_5 :
    A V 1 1 loss (1 / 2) policy 2 = A V 1 1 futureLoss (1 / 2) policy 2 ∧
    loss 2 ≠ futureLoss 2 ∧ loss 1 = futureLoss 1

theorem certificate_6 :
    (∀ t : ℕ, X V 1 1 zeroLoss (1 / 2) zeroPolicy t = 1 / 2) ∧
    (∀ T : ℕ, Q V 1 1 zeroLoss (1 / 2) zeroPolicy T = 0) ∧
    F V 1 1 zeroLoss (1 / 2) zeroPolicy 0 4 = 0 ∧
    ‖X V 1 1 zeroLoss (1 / 2) zeroPolicy 4 - 0‖ ^ 2 = 1 / 4 ∧
    F V 1 1 zeroLoss (1 / 2) zeroPolicy 0 4 ≤ 0

theorem certificate_7 :
    X W 1 0 loss 0 policy 2 = 0 ∧ Q W 1 0 loss 0 policy 2 = 9 ∧
    G W 1 0 loss 0 policy 1 = 3 ∧ F W 1 0 loss 0 policy 0 2 = 0 ∧
    F W 1 0 loss 0 policy 0 2 ≤ 0

end NeutralDynamics

/- Exact imported notation needed to decode the real fixture:
unitInterval has carrier Icc (0:ℝ) 1; project is the unique nearest-point projection.
SourceSubdifferential f z = {g | ∀ y, f z + (inner ℝ g (y-z) : EReal) ≤ f y}.
SourceProper f = (∀ z, f z ≠ ⊥) ∧ ∃ z c, f z = (c : EReal).
SubdifferentiableOn U f = SourceProper f ∧ ∀ z ∈ U.carrier, (SourceSubdifferential f z).Nonempty.
SupportPolicy inputs: current natural time, full strict-past loss functions,
full feasible/history points through current point, current whole loss function.
No source identity or previous verdict is included. The concrete policy ignores
those inputs other than time; no universal off-path oracle law is asserted.
