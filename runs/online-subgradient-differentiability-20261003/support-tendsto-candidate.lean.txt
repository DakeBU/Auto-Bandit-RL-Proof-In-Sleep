variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [FiniteDimensional ℝ E]

def SourceDifferentiableAt (f : E → EReal) (x : E) : Prop :=
  ∃ h : E → ℝ, (fun y => (h y : EReal)) =ᶠ[𝓝 x] f ∧ DifferentiableAt ℝ h x

theorem subgradient_norm_le_lipschitz_ball (f : E → EReal) (x : E)
    (r : ℝ) (hr : 0 < r) (K : ℝ≥0)
    (hfinite : ∀ y ∈ Metric.ball x r, ∃ a : ℝ, f y = (a : EReal))
    (hlip : LipschitzOnWith K (fun y => (f y).toReal) (Metric.ball x r))
    (g : E) (hg : g ∈ SourceSubdifferential f x) :
    ‖g‖ ≤ (K : ℝ) := by
  by_cases hg0 : g = 0
  · simpa only [hg0, norm_zero] using K.coe_nonneg
  have hgn : 0 < ‖g‖ := norm_pos_iff.mpr hg0
  let t : ℝ := r / (2 * ‖g‖)
  have ht : 0 < t := div_pos hr (mul_pos (by norm_num) hgn)
  let z : E := x + t • g
  have hdist : dist z x = t * ‖g‖ := by
    simp only [z, dist_eq_norm, add_sub_cancel_left, norm_smul, Real.norm_of_nonneg ht.le]
  have hscale : t * ‖g‖ = r / 2 := by
    dsimp [t]
    field_simp
  have hz : z ∈ Metric.ball x r := by
    rw [Metric.mem_ball, hdist, hscale]
    linarith
  have hx : x ∈ Metric.ball x r := by simpa using hr
  have hxval : ((f x).toReal : EReal) = f x := by
    obtain ⟨a, ha⟩ := hfinite x hx
    rw [ha, EReal.toReal_coe]
  have hzval : ((f z).toReal : EReal) = f z := by
    obtain ⟨a, ha⟩ := hfinite z hz
    rw [ha, EReal.toReal_coe]
  have hs := hg z
  rw [← hxval, ← hzval, ← EReal.coe_add] at hs
  have hsreal : (f x).toReal + inner ℝ g (z - x) ≤ (f z).toReal := by exact_mod_cast hs
  have hinner : inner ℝ g (z - x) = t * ‖g‖^2 := by
    simp only [z, add_sub_cancel_left, inner_smul_right, real_inner_self_eq_norm_sq]
  rw [hinner] at hsreal
  have hl := hlip.dist_le_mul z hz x hx
  rw [Real.dist_eq, hdist] at hl
  have hdiff := (le_abs_self ((f z).toReal - (f x).toReal)).trans hl
  have hp : 0 < t * ‖g‖ := mul_pos ht hgn
  have hb : (t * ‖g‖) * ‖g‖ ≤ (t * ‖g‖) * (K : ℝ) := by
    nlinarith [hsreal, hdiff]
  exact le_of_mul_le_mul_left hb hp

end BanditRL.OnlineConvex
#print axioms BanditRL.OnlineConvex.subgradient_norm_le_lipschitz_ball

variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [FiniteDimensional ℝ E]
theorem subgradients_locally_bounded (f : E → EReal) (hbot : ∀ y, f y ≠ ⊥)
    (hf : IsConvexExtended f) (x : E) (hx : x ∈ interior (effectiveDomain f)) :
    ∃ r : ℝ, 0 < r ∧ ∃ K : ℝ≥0,
      ∀ y ∈ Metric.ball x r, ∀ g ∈ SourceSubdifferential f y, ‖g‖ ≤ (K : ℝ) := by
  have hc := (convexExtended_iff_toReal f hbot).mp hf
  obtain ⟨K, t, ht, hLip⟩ := hc.locallyLipschitzOn_interior hx
  rw [isOpen_interior.nhdsWithin_eq hx] at ht
  obtain ⟨R, hR, hsub⟩ := Metric.mem_nhds_iff.mp
    (inter_mem ht (isOpen_interior.mem_nhds hx))
  refine ⟨R / 2, half_pos hR, K, ?_⟩
  intro y hy g hg
  have hball : Metric.ball y (R / 2) ⊆ Metric.ball x R := Metric.ball_half_subset y hy
  apply subgradient_norm_le_lipschitz_ball f y (R / 2) (half_pos hR) K ?_ ?_ g hg
  · intro z hz
    have hzint := (hsub (hball hz)).2
    have hzdom : z ∈ effectiveDomain f := interior_subset hzint
    exact ⟨(f z).toReal, (EReal.coe_toReal (ne_of_lt hzdom) (hbot z)).symm⟩
  · exact hLip.mono (fun z hz => (hsub (hball hz)).1)

end BanditRL.OnlineConvex
#print axioms BanditRL.OnlineConvex.subgradients_locally_bounded

variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [FiniteDimensional ℝ E]
theorem subgradient_limit_of_continuousAt
    {ι : Type*} (l : Filter ι) [NeBot l]
    (f : E → EReal) (hbot : ∀ y, f y ≠ ⊥)
    (x g : E) (hx : x ∈ interior (effectiveDomain f))
    (hcont : ContinuousAt (fun y => (f y).toReal) x)
    (xs gs : ι → E) (hxs : Tendsto xs l (𝓝 x)) (hgs : Tendsto gs l (𝓝 g))
    (hs : ∀ᶠ i in l, gs i ∈ SourceSubdifferential f (xs i)) :
    g ∈ SourceSubdifferential f x := by
  have hfinite := finitePart_eventually f hbot x hx
  have hxfin := hfinite.self_of_nhds
  have hefin := hxs.eventually hfinite
  have hfconv := hcont.tendsto.comp hxs
  intro y
  by_cases hyt : f y = ⊤
  · rw [hyt]
    exact le_top
  · have hyfin := EReal.coe_toReal hyt (hbot y)
    have ht : Tendsto (fun i => (f (xs i)).toReal + inner ℝ (gs i) (y - xs i)) l
        (𝓝 ((f x).toReal + inner ℝ g (y - x))) :=
      hfconv.add (hgs.inner (tendsto_const_nhds.sub hxs))
    have hbound : (f x).toReal + inner ℝ g (y - x) ≤ (f y).toReal := by
      apply le_of_tendsto ht
      filter_upwards [hs, hefin] with i hi hifin
      have hb := hi y
      rw [← hifin, ← hyfin, ← EReal.coe_add] at hb
      exact_mod_cast hb
    rw [← hxfin, ← hyfin, ← EReal.coe_add]
    exact EReal.coe_le_coe hbound


theorem singleton_subgradient_tendsto
    {ι : Type*} (l : Filter ι) [NeBot l]
    (f : E → EReal) (hbot : ∀ y, f y ≠ ⊥) (hf : IsConvexExtended f)
    (x g : E) (hx : x ∈ interior (effectiveDomain f))
    (hg : SourceSubdifferential f x = {g})
    (xs gs : ι → E) (hxs : Tendsto xs l (𝓝 x))
    (hs : ∀ᶠ i in l, gs i ∈ SourceSubdifferential f (xs i)) :
    Tendsto gs l (𝓝 g) := by
  obtain ⟨r, hr, K, hK⟩ := subgradients_locally_bounded f hbot hf x hx
  have hm : ∀ᶠ i in l, gs i ∈ Metric.closedBall (0 : E) (K : ℝ) := by
    filter_upwards [hs, hxs.eventually (Metric.ball_mem_nhds x hr)] with i hi hxi
    simp only [Metric.mem_closedBall, dist_zero_right]
    exact hK (xs i) hxi (gs i) hi
  apply (isCompact_closedBall (0 : E) (K : ℝ)).tendsto_nhds_of_unique_mapClusterPt hm
  intro z hz hcl
  let l' : Filter ι := comap gs (𝓝 z) ⊓ l
  haveI : NeBot l' := neBot_inf_comap_iff_map'.mpr hcl
  have hxs' : Tendsto xs l' (𝓝 x) := hxs.mono_left inf_le_right
  have hgs' : Tendsto gs l' (𝓝 z) := tendsto_comap.mono_left inf_le_left
  have hs' : ∀ᶠ i in l', gs i ∈ SourceSubdifferential f (xs i) := hs.filter_mono inf_le_right
  have hc := (convexExtended_iff_toReal f hbot).mp hf
  have hcont := hc.continuousOn_interior.continuousAt (isOpen_interior.mem_nhds hx)
  have hzmem := subgradient_limit_of_continuousAt l' f hbot x z hx hcont xs gs hxs' hgs' hs'
  rw [hg, mem_singleton_iff] at hzmem
  exact hzmem

end BanditRL.OnlineConvex
