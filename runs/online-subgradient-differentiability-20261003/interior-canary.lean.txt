namespace BanditRL.OnlineConvex
variable {E : Type*} [NormedAddCommGroup E] [InnerProductSpace ℝ E] [FiniteDimensional ℝ E]

def SourceDifferentiableAt (f : E → EReal) (x : E) : Prop :=
  ∃ h : E → ℝ, (fun y => (h y : EReal)) =ᶠ[𝓝 x] f ∧ DifferentiableAt ℝ h x

theorem singleton_subdifferential_interior (f : E → EReal) (hf : IsConvexExtended f)
    (x : E) (hx : ∃ r : ℝ, f x = (r : EReal))
    (g : E) (hg : SourceSubdifferential f x = {g}) :
    x ∈ interior (effectiveDomain f) := by
  classical
  obtain ⟨r, hr⟩ := hx
  have hxdom : x ∈ effectiveDomain f := by
    change f x < ⊤
    rw [hr]
    exact EReal.coe_lt_top _
  have hgs : g ∈ SourceSubdifferential f x := by rw [hg]; exact mem_singleton g
  by_contra hxi
  obtain ⟨a, hane, ha⟩ := supporting_functional_at_closure (effectiveDomain f)
    (convex_effectiveDomain f hf) x (subset_closure hxdom) hxi
  let d : E := (InnerProductSpace.toDual ℝ E).symm a
  have hdrepr : InnerProductSpace.toDual ℝ E d = a :=
    (InnerProductSpace.toDual ℝ E).apply_symm_apply a
  have hdne : d ≠ 0 := by
    intro hd
    rw [hd, map_zero] at hdrepr
    exact hane hdrepr.symm
  have hbot : ∀ y, f y ≠ ⊥ := by
    intro y hy
    have h := hgs y
    rw [hr, hy, ← EReal.coe_add] at h
    exact (EReal.bot_lt_coe _).not_ge h
  have hgd : g + d ∈ SourceSubdifferential f x := by
    intro y
    by_cases hyt : f y = ⊤
    · rw [hyt]; exact le_top
    · have hyfin := EReal.coe_toReal hyt (hbot y)
      have hydom : y ∈ effectiveDomain f := by
        change f y < ⊤
        rw [← hyfin]
        exact EReal.coe_lt_top _
      have hn := ha y hydom
      have hdinner : inner ℝ d (y - x) = a y - a x := by
        rw [← InnerProductSpace.toDual_apply_apply, hdrepr, map_sub]
      have hs := hgs y
      rw [hr, ← hyfin, ← EReal.coe_add] at hs
      have hsreal : r + inner ℝ g (y - x) ≤ (f y).toReal := by exact_mod_cast hs
      rw [hr, ← hyfin, ← EReal.coe_add]
      apply EReal.coe_le_coe
      rw [inner_add_left, hdinner]
      linarith
  rw [hg, mem_singleton_iff] at hgd
  apply hdne
  apply add_left_cancel (a := g)
  simpa using hgd

theorem subgradient_eq_gradient_at_interior (f : E → EReal) (hbot : ∀ z, f z ≠ ⊥)
    (x : E) (hx : x ∈ interior (effectiveDomain f))
    (hd : DifferentiableAt ℝ (fun z => (f z).toReal) x)
    (g : E) (hg : g ∈ SourceSubdifferential f x) :
    g = gradient (fun z => (f z).toReal) x := by
  have he := finitePart_eventually f hbot x hx
  have hxval := he.self_of_nhds
  have hm : IsLocalMin (fun z => (f z).toReal - inner ℝ g z) x := by
    filter_upwards [he] with z hz
    have hb := hg z
    rw [← hxval, ← hz, ← EReal.coe_add] at hb
    have hr : (f x).toReal + inner ℝ g (z - x) ≤ (f z).toReal := by
      exact_mod_cast hb
    rw [inner_sub_right] at hr
    linarith
  have hd' := hd.hasGradientAt.hasFDerivAt
  have hl : HasFDerivAt (fun z => inner ℝ g z) (InnerProductSpace.toDual ℝ E g) x :=
    (InnerProductSpace.toDual ℝ E g).hasFDerivAt
  have hz := hm.hasFDerivAt_eq_zero (hd'.sub hl)
  have ht : InnerProductSpace.toDual ℝ E (gradient (fun z => (f z).toReal) x) =
      InnerProductSpace.toDual ℝ E g := sub_eq_zero.mp hz
  exact ((InnerProductSpace.toDual ℝ E).injective ht).symm


end BanditRL.OnlineConvex

namespace SingletonInteriorProbe
open BanditRL.OnlineConvex Set

theorem square_singleton_support :
    SourceSubdifferential (fun y : ℝ => ((y^2 : ℝ) : EReal)) 1 = {(2 : ℝ)} := by
  let f : ℝ → EReal := fun y => ((y^2 : ℝ) : EReal)
  have hbot : ∀ y, f y ≠ ⊥ := by intro y; exact EReal.coe_ne_bot _
  have hdom : effectiveDomain f = univ := by
    ext y
    simp only [effectiveDomain, mem_setOf_eq, mem_univ, iff_true]
    exact EReal.coe_lt_top _
  have hx : (1 : ℝ) ∈ interior (effectiveDomain f) := by rw [hdom, interior_univ]; trivial
  have hd : DifferentiableAt ℝ (fun y => (f y).toReal) 1 := by
    simpa only [f, EReal.toReal_coe] using (differentiableAt_id.pow 2 : DifferentiableAt ℝ (fun y : ℝ => y^2) 1)
  have h2 : (2 : ℝ) ∈ SourceSubdifferential f 1 := by
    simpa [f] using SubgradientProbe.square_support 1
  have hgrad := subgradient_eq_gradient_at_interior f hbot 1 hx hd 2 h2
  ext g
  constructor
  · intro hg
    have he := subgradient_eq_gradient_at_interior f hbot 1 hx hd g hg
    exact mem_singleton_iff.mpr (he.trans hgrad.symm)
  · intro hg
    have he : g = 2 := mem_singleton_iff.mp hg
    simpa only [he] using h2

theorem square_finite_interior :
    (1 : ℝ) ∈ interior (effectiveDomain (fun y : ℝ => ((y^2 : ℝ) : EReal))) := by
  apply singleton_subdifferential_interior _ _ _ ⟨1, by norm_num⟩ 2 square_singleton_support
  apply (convexExtended_iff_toReal _ (fun y => EReal.coe_ne_bot _)).mpr
  have hdom : effectiveDomain (fun y : ℝ => ((y^2 : ℝ) : EReal)) = univ := by
    ext y
    simp only [effectiveDomain, mem_setOf_eq, mem_univ, iff_true]
    exact EReal.coe_lt_top _
  rw [hdom]
  simpa only [EReal.toReal_coe] using SubgradientProbe.square_convex

end SingletonInteriorProbe
