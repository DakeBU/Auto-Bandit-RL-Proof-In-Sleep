def SourceSubgradientSum {ι : Type*} [Fintype ι]
    (f : ι → E → EReal) (x : E) : Set E :=
  {g | ∃ G : ι → E, (∀ i, G i ∈ SourceSubdifferential (f i) x) ∧ ∑ i, G i = g}


theorem theorem_2_23_inclusion {ι : Type*} [Fintype ι]
    (f : ι → E → EReal) (hp : ∀ i, SourceProper (f i)) (x : E) :
    SourceSubgradientSum f x ⊆ SourceSubdifferential (fun y => ∑ i, f i y) x

theorem theorem_2_23_equality (n : ℕ) (f : Fin (n + 1) → E → EReal)
    (hp : ∀ i, SourceProper (f i)) (hc : ∀ i, IsConvexExtended (f i))
    (hclosed : ∀ i, SourceClosed (f i))
    (hqual : ∃ z : E, z ∈ effectiveDomain (f (Fin.last n)) ∧
      ∀ i : Fin (n + 1), i ≠ Fin.last n → z ∈ interior (effectiveDomain (f i)))
    (x : E) :
    SourceSubdifferential (fun y => ∑ i, f i y) x = SourceSubgradientSum f x

Neutral imported definitions (copied from the actual imported project declarations):

def effectiveDomain (f : E → EReal) : Set E := {x | f x < ⊤}
def realEpigraph (f : E → EReal) : Set (E × ℝ) := {p | f p.1 ≤ (p.2 : EReal)}
def IsConvexExtended (f : E → EReal) : Prop := Convex ℝ (realEpigraph f)
def SourceClosed (f : E → EReal) : Prop :=
  ∀ r : ℝ, IsClosed {x | f x ≤ (r : EReal)}
def SourceProper (f : E → EReal) : Prop :=
  (∀ x, f x ≠ ⊥) ∧ ∃ x, ∃ r : ℝ, f x = (r : EReal)
def SourceSubdifferential (f : E → EReal) (x : E) : Set E :=
  {g | ∀ y, f x + (inner ℝ g (y - x) : EReal) ≤ f y}
