import BanditRLProof.OnlineGuessingSubgradientPolicy

/-! Actual paired historical support choices at a kink, legal only on played
source queries. All outputs, regrets, energies and terminal distances are proved
from the common recursion. No supplied stability/regret or external trajectory. -/
noncomputable section
namespace GuessingPolicyProbe
open Set Finset Filter BanditRL.OnlineConvex
open BanditRL.OnlineSubgradientPolicy
open BanditRL.OnlineGuessingSubgradient
abbrev V := BanditRL.OnlineGradientDescent.unitInterval

def policy : SupportPolicy (E := ℝ) := fun t past h f =>
  if (f 0).toReal < h (Fin.last t) then 1
  else if h (Fin.last t) < (f 0).toReal then -1
  else if ht : 0 < t then
    if (past ⟨0, ht⟩ 0).toReal = 0 then 1 else -1
  else 1

theorem policy_on_absolute (t : ℕ) (past : Fin t → ℝ → EReal)
    (h : Fin (t + 1) → ℝ) (y : ℝ) (hy : y ∈ Icc (0 : ℝ) 1) :
    policy t past h (loss y) ∈ SourceSubdifferential (loss y) (h (Fin.last t)) := by
  have hv : (loss y 0).toReal = y := by
    simp only [loss, EReal.toReal_coe, zero_sub, abs_neg]
    exact abs_of_nonneg hy.1
  simp only [policy, hv]
  by_cases hp : y < h (Fin.last t)
  · rw [if_pos hp, loss_subgradient_positive y _ hp]
    exact Set.mem_singleton 1
  · rw [if_neg hp]
    by_cases hn : h (Fin.last t) < y
    · rw [if_pos hn, loss_subgradient_negative y _ hn]
      exact Set.mem_singleton (-1)
    · rw [if_neg hn]
      have he : h (Fin.last t) = y := le_antisymm (le_of_not_gt hp) (le_of_not_gt hn)
      rw [he, loss_subgradient_zero]
      split_ifs <;> norm_num

theorem policy_legal (η : ℕ → ℝ) (y : ℕ → ℝ) (x₁ : ℝ) (T : ℕ)
    (hy : ∀ t < T, y t ∈ Icc (0 : ℝ) 1) :
    LegalFeedback V η (fun s => loss (y s)) x₁ policy T := by
  intro t ht
  exact policy_on_absolute t (fun i => loss (y i.val))
    (history V η (fun s => loss (y s)) x₁ policy t) (y t) (hy t ht)

theorem selected_rule (η : ℕ → ℝ) (y : ℕ → ℝ) (x₁ : ℝ) (t : ℕ) :
    selected V η (fun s => loss (y s)) x₁ policy t =
      if |y t| < output V η (fun s => loss (y s)) x₁ policy t then 1
      else if output V η (fun s => loss (y s)) x₁ policy t < |y t| then -1
      else if ht : 0 < t then if |y 0| = 0 then 1 else -1 else 1 := by
  simp only [selected, policy, loss, EReal.toReal_coe, zero_sub, abs_neg, output]

def labelsA (t : ℕ) : ℝ := if t = 0 then 0 else 1 / 2
def labelsB (t : ℕ) : ℝ := if t = 0 then 1 else 1 / 2
def eta : ℕ → ℝ := fun _ => 1 / 2
abbrev a (t : ℕ) := output V eta (fun s => loss (labelsA s)) (1 / 2) policy t
abbrev b (t : ℕ) := output V eta (fun s => loss (labelsB s)) (1 / 2) policy t
abbrev ga (t : ℕ) := selected V eta (fun s => loss (labelsA s)) (1 / 2) policy t
abbrev gb (t : ℕ) := selected V eta (fun s => loss (labelsB s)) (1 / 2) policy t

theorem labelsA_feasible (t : ℕ) : labelsA t ∈ Icc (0 : ℝ) 1 := by
  unfold labelsA; split_ifs <;> norm_num
theorem labelsB_feasible (t : ℕ) : labelsB t ∈ Icc (0 : ℝ) 1 := by
  unfold labelsB; split_ifs <;> norm_num

theorem a_zero : a 0 = 1 / 2 := rfl
theorem b_zero : b 0 = 1 / 2 := rfl
theorem ga_zero : ga 0 = 1 := by
  rw [selected_rule]
  norm_num [labelsA, a_zero]
theorem gb_zero : gb 0 = -1 := by
  rw [selected_rule]
  norm_num [labelsB, b_zero]

theorem a_one : a 1 = 0 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (a 0 - (1 / 2) * ga 0) 0) 1 = 0
  rw [a_zero, ga_zero]; norm_num
theorem b_one : b 1 = 1 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (b 0 - (1 / 2) * gb 0) 0) 1 = 1
  rw [b_zero, gb_zero]; norm_num
theorem ga_one : ga 1 = -1 := by
  rw [selected_rule]
  norm_num [labelsA, a_one]
theorem gb_one : gb 1 = 1 := by
  rw [selected_rule]
  norm_num [labelsB, b_one]

theorem a_two : a 2 = 1 / 2 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (a 1 - (1 / 2) * ga 1) 0) 1 = 1 / 2
  rw [a_one, ga_one]; norm_num
theorem b_two : b 2 = 1 / 2 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (b 1 - (1 / 2) * gb 1) 0) 1 = 1 / 2
  rw [b_one, gb_one]; norm_num
theorem ga_two : ga 2 = 1 := by
  rw [selected_rule]
  norm_num [labelsA, a_two]
theorem gb_two : gb 2 = -1 := by
  rw [selected_rule]
  norm_num [labelsB, b_two]

theorem a_three : a 3 = 0 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (a 2 - (1 / 2) * ga 2) 0) 1 = 0
  rw [a_two, ga_two]; norm_num
theorem b_three : b 3 = 1 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (b 2 - (1 / 2) * gb 2) 0) 1 = 1
  rw [b_two, gb_two]; norm_num
theorem ga_three : ga 3 = -1 := by
  rw [selected_rule]
  norm_num [labelsA, a_three]
theorem gb_three : gb 3 = 1 := by
  rw [selected_rule]
  norm_num [labelsB, b_three]
theorem a_four : a 4 = 1 / 2 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (a 3 - (1 / 2) * ga 3) 0) 1 = 1 / 2
  rw [a_three, ga_three]; norm_num
theorem b_four : b 4 = 1 / 2 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  change min (max (b 3 - (1 / 2) * gb 3) 0) 1 = 1 / 2
  rw [b_three, gb_three]; norm_num

theorem history_changes_kink_choice :
    a 2 = b 2 ∧ loss (labelsA 2) = loss (labelsB 2) ∧
      ga 2 = 1 ∧ gb 2 = -1 ∧ a 3 ≠ b 3 := by
  norm_num [a_two, b_two, labelsA, labelsB, ga_two, gb_two, a_three, b_three]

theorem a_real_regret :
    regret V eta (fun s => loss (labelsA s)) (1 / 2) policy (1 / 4) 4 = 1 / 2 := by
  simp only [regret, sum_range_succ, sum_range_zero, zero_add]
  change ((loss (labelsA 0) (a 0)).toReal - (loss (labelsA 0) (1 / 4)).toReal) +
    ((loss (labelsA 1) (a 1)).toReal - (loss (labelsA 1) (1 / 4)).toReal) +
    ((loss (labelsA 2) (a 2)).toReal - (loss (labelsA 2) (1 / 4)).toReal) +
    ((loss (labelsA 3) (a 3)).toReal - (loss (labelsA 3) (1 / 4)).toReal) = 1 / 2
  rw [a_zero, a_one, a_two, a_three]; norm_num [labelsA, loss]
theorem b_real_regret :
    regret V eta (fun s => loss (labelsB s)) (1 / 2) policy (1 / 4) 4 = 0 := by
  simp only [regret, sum_range_succ, sum_range_zero, zero_add]
  change ((loss (labelsB 0) (b 0)).toReal - (loss (labelsB 0) (1 / 4)).toReal) +
    ((loss (labelsB 1) (b 1)).toReal - (loss (labelsB 1) (1 / 4)).toReal) +
    ((loss (labelsB 2) (b 2)).toReal - (loss (labelsB 2) (1 / 4)).toReal) +
    ((loss (labelsB 3) (b 3)).toReal - (loss (labelsB 3) (1 / 4)).toReal) = 0
  rw [b_zero, b_one, b_two, b_three]; norm_num [labelsB, loss]
theorem a_energy : (∑ t ∈ range 4, ‖ga t‖ ^ 2) = (4 : ℝ) := by
  simp only [sum_range_succ, sum_range_zero, zero_add]
  rw [ga_zero, ga_one, ga_two, ga_three]; norm_num
theorem positive_terminal : ‖a 4 - (1 / 4)‖ ^ 2 = (1 / 16 : ℝ) := by
  rw [a_four]; norm_num

theorem a_actual_fixed :
    regret V eta (fun s => loss (labelsA s)) (1 / 2) policy (1 / 4) 4 ≤
      ‖(1 / 2 : ℝ) - 1 / 4‖ ^ 2 / (2 * (1 / 2)) +
        (1 / 2) / 2 * (∑ t ∈ range 4, ‖ga t‖ ^ 2) -
        ‖a 4 - (1 / 4)‖ ^ 2 / (2 * (1 / 2)) := by
  exact regret_fixed V (1 / 2) (by norm_num) (fun s => loss (labelsA s))
    (1 / 2) policy (by norm_num [V, BanditRL.OnlineGradientDescent.unitInterval]) 4
    (fun t ht => loss_on_unitInterval _) (policy_legal eta labelsA (1 / 2) 4
      (fun t ht => labelsA_feasible t)) (1 / 4)
    (by norm_num [V, BanditRL.OnlineGradientDescent.unitInterval])
theorem fixed_rhs_is_one :
    ‖(1 / 2 : ℝ) - 1 / 4‖ ^ 2 / (2 * (1 / 2)) +
      (1 / 2) / 2 * (∑ t ∈ range 4, ‖ga t‖ ^ 2) -
      ‖a 4 - (1 / 4)‖ ^ 2 / (2 * (1 / 2)) = 1 := by
  rw [a_energy, positive_terminal]; norm_num

theorem a_selected_bound (t : ℕ) (ht : t < 4) : ‖ga t‖ ≤ 1 := by
  exact BanditRL.OnlineGuessingSubgradientPolicy.selected_bound eta labelsA
    (1 / 2) policy 4 (policy_legal eta labelsA _ 4 (fun t ht => labelsA_feasible t)) t ht

theorem a_tuned_all_comparators : ∀ u ∈ Icc (0 : ℝ) 1,
    (∑ t ∈ range 4, (|a t - labelsA t| - |u - labelsA t|)) ≤ 2 := by
  have hs : Real.sqrt (4 : ℝ) = 2 := by norm_num
  have hb := BanditRL.OnlineGuessingSubgradientPolicy.example_2_32 labelsA
    (1 / 2) policy (by norm_num) 4 (by norm_num) (fun t ht => labelsA_feasible t)
    (policy_legal (fun _ => 1 / Real.sqrt 4) labelsA _ 4
      (fun t ht => labelsA_feasible t))
  simpa only [hs] using hb

theorem not_global_oracle : ¬ OracleLaw V policy := by
  intro horacle
  have hh := horacle 0 (fun _ => loss 0) (fun _ => (1 / 2 : ℝ)) (loss (-1))
    (loss_on_unitInterval (-1))
    (by norm_num [V, BanditRL.OnlineGradientDescent.unitInterval])
  have hsel : policy 0 (fun _ => loss 0) (fun _ => (1 / 2 : ℝ)) (loss (-1)) = -1 := by
    norm_num [policy, loss]
  rw [hsel, loss_subgradient_positive (-1) (1 / 2) (by norm_num)] at hh
  norm_num at hh

def futureLabels (t : ℕ) : ℝ := if t < 4 then labelsA t else -99
def futureEta (t : ℕ) : ℝ := if t < 4 then eta t else -99
theorem invalid_future_causal :
    output V futureEta (fun s => loss (futureLabels s)) (1 / 2) policy 4 = a 4 := by
  symm
  exact output_prefix V eta futureEta (fun s => loss (labelsA s))
    (fun s => loss (futureLabels s)) (1 / 2) policy 4
    (fun s hs => by simp [futureEta, hs]) (fun s hs => by simp [futureLabels, hs])

theorem zero_rate_infeasible_initial :
    output V (fun _ => 0) (fun _ => loss 0) 2 policy 1 = 1 := by
  rw [BanditRL.OnlineGuessingSubgradientPolicy.step_clamp]
  simp only [zero_mul, sub_zero, output_zero]
  norm_num

theorem a_horizon_family_average : ∀ᶠ T : ℕ in atTop,
    (∑ t ∈ range T,
      (|output V (fun _ => 1 / Real.sqrt T) (fun s => loss (labelsA s))
        (1 / 2) policy t - labelsA t| - |(1 / 4) - labelsA t|)) / T < 1 / 10 := by
  exact BanditRL.OnlineGuessingSubgradientPolicy.example_2_32_average_eventually
    labelsA (1 / 2) policy (by norm_num) labelsA_feasible
    (fun T hT => policy_legal _ labelsA _ T (fun t ht => labelsA_feasible t))
    (1 / 4) (by norm_num) (1 / 10) (by norm_num)

end GuessingPolicyProbe
