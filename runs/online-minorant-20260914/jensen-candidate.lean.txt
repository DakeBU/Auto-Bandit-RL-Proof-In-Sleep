theorem jensen_negativeIntegral_ne_top (μ : Measure Ω) [IsProbabilityMeasure μ]
    (f : E → EReal) (hbot : ∀ x, f x ≠ ⊥) (hf : IsConvexExtended f)
    (hfm : Measurable f) (X : Ω → E) (hXm : Measurable X) (hX : Integrable X μ)
    (hdom : ∀ᵐ ω ∂μ, X ω ∈ effectiveDomain f) :
    negativeIntegral μ (fun ω => f (X ω)) ≠ ∞ := by
  classical
  obtain ⟨ω, hω⟩ := hdom.exists
  obtain ⟨a, b, hab⟩ := convex_affine_minorant f hbot hf ⟨X ω, hω⟩
  have hi : Integrable (fun ω => a (X ω) + b) μ := (a.integrable_comp hX).add (integrable_const b)
  have hbound : negativeIntegral μ (fun ω => f (X ω)) ≤
      negativeIntegral μ (fun ω => ((a (X ω) + b : ℝ) : EReal)) := by
    apply lintegral_mono
    intro ω
    exact EReal.toENNReal_le_toENNReal (EReal.neg_le_neg_iff.mpr (hab (X ω)))
  exact ne_top_of_le_ne_top (negativeIntegral_coe_ne_top μ _ hi) hbound
theorem theorem_2_9 (μ : Measure Ω) [IsProbabilityMeasure μ]
    (f : E → EReal) (hbot : ∀ x, f x ≠ ⊥) (hf : IsConvexExtended f)
    (hfm : Measurable f) (X : Ω → E) (hXm : Measurable X) (hX : Integrable X μ)
    (hdom : ∀ᵐ ω ∂μ, X ω ∈ effectiveDomain f) :
    f (∫ ω, X ω ∂μ) ≤ signedExpectation μ (fun ω => f (X ω)) := by
  classical
  have hn := jensen_negativeIntegral_ne_top μ f hbot hf hfm X hXm hX hdom
  by_cases hp : positiveIntegral μ (fun ω => f (X ω)) = ∞
  · rw [signedExpectation_eq_top μ _ hp hn]
    exact le_top
  · let Y : Ω → ℝ := fun ω => (f (X ω)).toReal
    have hYm : Measurable Y := (hfm.comp hXm).ereal_toReal
    have hcoe : ∀ᵐ ω ∂μ, (Y ω : EReal) = f (X ω) := by
      filter_upwards [hdom] with ω hω
      exact EReal.coe_toReal (ne_of_lt hω) (hbot (X ω))
    have hpos : (∫⁻ ω, ENNReal.ofReal (Y ω) ∂μ) ≠ ∞ := by
      have he : (∫⁻ ω, ENNReal.ofReal (Y ω) ∂μ) = positiveIntegral μ (fun ω => f (X ω)) := by
        apply lintegral_congr_ae
        filter_upwards [hcoe] with ω hω
        rw [← hω]
        simp
      rwa [he]
    have hneg : (∫⁻ ω, ENNReal.ofReal (-Y ω) ∂μ) ≠ ∞ := by
      have he : (∫⁻ ω, ENNReal.ofReal (-Y ω) ∂μ) = negativeIntegral μ (fun ω => f (X ω)) := by
        apply lintegral_congr_ae
        filter_upwards [hcoe] with ω hω
        rw [← hω]
        simp [← EReal.coe_neg]
      rwa [he]
    have hip : Integrable (fun ω => max (Y ω) 0) μ := by
      apply (lintegral_ofReal_ne_top_iff_integrable (hYm.max measurable_const).aestronglyMeasurable
        (ae_of_all μ (fun ω => le_max_right (Y ω) 0))).mp
      simpa using hpos
    have hin : Integrable (fun ω => max (-Y ω) 0) μ := by
      apply (lintegral_ofReal_ne_top_iff_integrable (hYm.neg.max measurable_const).aestronglyMeasurable
        (ae_of_all μ (fun ω => le_max_right (-Y ω) 0))).mp
      simpa using hneg
    have hY : Integrable Y μ := by
      convert hip.sub hin using 1
      ext ω
      dsimp
      rcases le_total (Y ω) 0 with h | h <;> simp [max_eq_left, max_eq_right, h, neg_nonneg.mpr, neg_nonpos.mpr] <;> linarith
    have hepi : ∀ᵐ ω ∂μ, (X ω, Y ω) ∈ realEpigraph f := by
      filter_upwards [hcoe] with ω hω
      exact hω.ge
    have hb := integral_mem_convex_finiteDimensional μ (realEpigraph f) hf
      (fun ω => (X ω, Y ω)) (hX.prodMk hY) hepi
    rw [integral_pair hX hY] at hb
    have hexp : signedExpectation μ (fun ω => f (X ω)) = ((∫ ω, Y ω ∂μ : ℝ) : EReal) := by
      rw [← signedExpectation_coe_integrable μ Y hY]
      unfold signedExpectation positiveIntegral negativeIntegral
      congr 2
      · apply lintegral_congr_ae
        filter_upwards [hcoe] with ω hω
        rw [hω]
      · apply lintegral_congr_ae
        filter_upwards [hcoe] with ω hω
        rw [hω]
    rw [hexp]
    exact hb
end BanditRL.OnlineConvex
