noncomputable def empiricalMean (y : ℕ → ℝ) (n : ℕ) : ℝ :=
  (∑ t ∈ Finset.range n, y t) / n

noncomputable def meanPredict (y : ℕ → ℝ) (t : ℕ) : ℝ :=
  if t = 0 then 1/2 else empiricalMean y t

noncomputable def comparatorRegret {X : Type*} (loss : ℕ → X → ℝ)
    (prediction : ℕ → X) (u : X) (T : ℕ) : ℝ :=
  (∑ t ∈ Finset.range T, loss t (prediction t)) - ∑ t ∈ Finset.range T, loss t u

noncomputable def squaredBestRegret (y prediction : ℕ → ℝ) (T : ℕ) : ℝ :=
  (∑ t ∈ Finset.range T, (prediction t - y t)^2) -
    sInf ((fun u : ℝ => ∑ t ∈ Finset.range T, (u - y t)^2) ''
      Set.Icc (0 : ℝ) 1)

def LimitNoRegret {X : Type*} (V : Set X) (loss : ℕ → X → ℝ)
    (prediction : ℕ → X) : Prop :=
  ∀ u ∈ V, ∃ a : ℝ, a ≤ 0 ∧
    Tendsto (fun T : ℕ => comparatorRegret loss prediction u T / T) atTop (nhds a)

theorem meanPredict_bestLoss_nonneg (y : ℕ → ℝ) (T : ℕ) :
    0 ≤ comparatorRegret (fun t x => (x - y t)^2)
      (meanPredict y) (empiricalMean y T) T

theorem meanPredict_comparator_decomposition (y : ℕ → ℝ) (u : ℝ) (T : ℕ) :
    comparatorRegret (fun t x => (x - y t)^2) (meanPredict y) u T =
      comparatorRegret (fun t x => (x - y t)^2)
        (meanPredict y) (empiricalMean y T) T -
      (T : ℝ) * (u - empiricalMean y T)^2

theorem meanPredict_bestRegret_average_tendsto_zero (y : ℕ → ℝ)
    (hy : ∀ t, y t ∈ Set.Icc (0 : ℝ) 1) :
    Tendsto (fun T : ℕ => squaredBestRegret y (meanPredict y) T / (T : ℝ))
      atTop (nhds (0 : ℝ))

theorem meanPredict_fixedRegret_limit_iff (y : ℕ → ℝ)
    (hy : ∀ t, y t ∈ Set.Icc (0 : ℝ) 1) (u a : ℝ) :
    Tendsto (fun T : ℕ =>
      comparatorRegret (fun t x => (x - y t)^2) (meanPredict y) u T / (T : ℝ))
      atTop (nhds a) ↔
    Tendsto (fun T : ℕ => (u - empiricalMean y T)^2) atTop (nhds (-a))

theorem meanPredict_limitNoRegret_of_mean_converges (y : ℕ → ℝ)
    (hy : ∀ t, y t ∈ Set.Icc (0 : ℝ) 1) (m : ℝ)
    (hm : Tendsto (empiricalMean y) atTop (nhds m)) :
    (∀ u : ℝ, Tendsto (fun T : ℕ =>
      comparatorRegret (fun t x => (x - y t)^2) (meanPredict y) u T / (T : ℝ))
      atTop (nhds (-(u - m)^2))) ∧
    LimitNoRegret (Set.Icc (0 : ℝ) 1) (fun t x => (x - y t)^2) (meanPredict y)

end BanditRL.OnlineLearning
